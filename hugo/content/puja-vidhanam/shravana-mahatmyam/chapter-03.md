<div class="verse-block-wrapper" id="shravana-mahatmyam-b0110"><div class="verse-block verse-justify"><div class="line">नक्तव्रतेन योगीश श्रावणं यो नयेत्सुधीः।</div><div class="line" style="position:relative;">द्वादशस्वपि मासेषु स नक्तफलभाग्भवेत्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0112"><div class="verse-block verse-justify"><div class="line">ततः सन्ध्या त्रिघटिका अस्तादुपरि भास्वतः।</div><div class="line" style="position:relative;">चत्वारीमानि कर्माणि सन्ध्यायां परिवर्जयेत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0113"><div class="verse-block verse-justify"><div class="line">आहारं मैथुनं निद्रां स्वाध्यायं च चतुर्थकम्।</div><div class="line" style="position:relative;">गृहस्थयतिभेदेन तद्व्यवस्थां च मे शृणु॥६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0114"><div class="verse-block verse-justify"><div class="line">आत्मनो द्विगुणी छाया मन्दीभवति भास्करे।</div><div class="line" style="position:relative;">यतेर्नक्तं तु तत्प्रोक्तं न नक्तं निशि भोजनम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0115"><div class="verse-block verse-justify"><div class="line">नक्षत्रदर्शनान्नक्तं गृहस्थस्य बुधैः स्मृतम्।</div><div class="line" style="position:relative;">यतेर्दिनाष्टमे भागे रात्रौ तस्य निषिध्यते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0116"><div class="verse-block verse-justify"><div class="line">नक्तं निशायां कुर्वीत गृहस्थो विधिसंयुतः।</div><div class="line" style="position:relative;">यतिश्च विधवा चैव विधुरश्च ससूर्यकम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0117"><div class="verse-block verse-justify"><div class="line">विधुरश्चेत्पुत्रवान् स्यात्स तु रात्रौ समाचरेत्।</div><div class="line" style="position:relative;">अनाश्रमोऽप्याश्रमी स्यादपत्नीकोऽपि पुत्रवान्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0118"><div class="verse-block verse-justify"><div class="line">एवं यथाधिकारं तु कुर्यान्नक्तव्रतं सुधीः।</div><div class="line" style="position:relative;">अत्र नक्तव्रती मासे परां गतिमवाप्नुयात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0119"><div class="verse-block verse-justify"><div class="line">प्रातःस्नानं करिष्यामि ब्रह्मचर्यव्रते स्थितः।</div><div class="line" style="position:relative;">भोक्ष्यामि नक्तं भूशय्यां करिष्ये प्राणिनां दयाम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0120"><div class="verse-block verse-justify"><div class="line">प्रारब्धेऽस्मिन्वते देव यद्यपूर्णे म्रिये ह्यहम्।</div><div class="line" style="position:relative;">तदा सम्पूर्णतां यातु प्रसादात्ते जगत्पते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0121"><div class="verse-block verse-justify"><div class="line">इति सङ्कल्प्य मेधावी मासे नक्तव्रतं चरेत्।</div><div class="line" style="position:relative;">एवं नक्तव्रतं कुर्वन्मम प्रियतमो भवेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0122"><div class="verse-block verse-justify"><div class="line">विप्रद्वारातिरुद्रेण महारुद्रेण वा स्वयम्।</div><div class="line" style="position:relative;">अभिषेकं मासमात्रं रुद्रेण प्रत्यहं चरेत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0123"><div class="verse-block verse-justify"><div class="line">तस्य प्रीणाम्यहं वत्स जलधाराप्रियो यतः।</div><div class="line" style="position:relative;">कुर्याद्रुद्रेण वा होमं मम प्रीतिकरं परम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0124"><div class="verse-block verse-justify"><div class="line">स्वस्य यद्रोचतेऽत्यन्तं भोज्यं वा भोग्यमेव वा।</div><div class="line" style="position:relative;">सङ्कल्प्य द्विजवर्याय दत्वा मासे स्वयं त्यजेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0125"><div class="verse-block verse-justify"><div class="line">अतः परं शृणु मुने लक्षपूजाविधिं परम्।</div><div class="line" style="position:relative;">श्रीकामो बिल्वपत्रैश्च दूर्वाभिः शान्तिकामुकः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0126"><div class="verse-block verse-justify"><div class="line">आयुष्कामेन कर्तव्यं चम्पकैः पूजनं हरेः।</div><div class="line" style="position:relative;">विद्याकामेन कर्तव्यं मल्लिकाजातिभिस्तथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0127"><div class="verse-block verse-justify"><div class="line">शिवविष्ण्वोः प्रसन्नत्वं तुलसीभिः प्रसिध्यति।</div><div class="line" style="position:relative;">पुत्रकामेन कर्तव्यं बार्हतैः पूजनं शुभम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0128"><div class="verse-block verse-justify"><div class="line">दुःस्वप्नप्रशमार्थाय शस्तं धान्यप्रपूजनम्।</div><div class="line" style="position:relative;">रङ्गवल्ल्यादिभिर्देवं देवस्याग्रे विनिर्मितैः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0129"><div class="verse-block verse-justify"><div class="line">पद्मादिभिः स्वस्तिकाद्यैश्चक्राद्यैः पूजयेद्विभुम्।</div><div class="line" style="position:relative;">एवं हि सर्वपुष्यैश्च सर्वकामार्थसिद्धये॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0130"><div class="verse-block verse-justify"><div class="line">लक्षपूजां प्रकुर्याच्चेत्सुप्रसन्नो हरो भवेत्।</div><div class="line" style="position:relative;">उद्यापनं ततः कार्यं मण्डपं चैव साधयेत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0131"><div class="verse-block verse-justify"><div class="line">वेदिका च प्रकर्तव्या मण्डपस्य त्रिभागतः।</div><div class="line" style="position:relative;">पुण्याहवाचनं कृत्वा आचार्यं वरयेत्ततः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0132"><div class="verse-block verse-justify"><div class="line">गीतवादित्रनिर्घोषैर्ब्रह्मघोषेण भूयसा।</div><div class="line" style="position:relative;">प्रविश्य मण्डपे तस्मिन् रात्रौ जागरणं चरेत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0133"><div class="verse-block verse-justify"><div class="line">वेदिकायां प्रकर्तव्यं लिङ्गतोभद्रमुत्तमम्।</div><div class="line" style="position:relative;">तन्मध्ये तण्डुलैः कुर्यात्कैलासं च सुशोभनम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0134"><div class="verse-block verse-justify"><div class="line">कलशं स्थापयेत्तत्र ताम्र चैव महाप्रभम्।</div><div class="line" style="position:relative;">पञ्चपल्लवसंयुक्तं न्यसेद् वस्त्रं सुसूक्ष्मकम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0135"><div class="verse-block verse-justify"><div class="line">सौवर्णी प्रतिमां तत्र स्थापयेत्पार्वतीपतेः।</div><div class="line" style="position:relative;">पूजां तत्र प्रकुर्वीत पञ्चामृतपुरःसरैः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0136"><div class="verse-block verse-justify"><div class="line">सनैवेद्यैर्गीतवादित्रनृत्यकैः।</div><div class="line" style="position:relative;">वेदशास्त्रपुराणैश्च रात्रौ जागरणं चरेत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0137"><div class="verse-block verse-justify"><div class="line">प्रभातसमये सुस्नातः सुशुचिर्भवेत्।</div><div class="line" style="position:relative;">स्थण्डिलं कारयेत्तत्र स्वशाखोक्तविधानतः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0138"><div class="verse-block verse-justify"><div class="line">होमं च सतिलाज्येन पायसेन च कारयेत्।</div><div class="line" style="position:relative;">मूलमन्त्रेण गायत्र्या शिवनाम्नां सहस्रकैः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0139"><div class="verse-block verse-justify"><div class="line">येन मन्त्रेण पूजा तु कृता तेनैव होमयेत्।</div><div class="line" style="position:relative;">शर्कराघृतमिश्रेण चरुणा जुहुयात्ततः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0140"><div class="verse-block verse-justify"><div class="line">ततः स्विष्टकृतं हुत्वा पूर्णाहुतिमनन्तरम्।</div><div class="line" style="position:relative;">आचार्यं पूजयेत्सम्यग्वस्त्रालङ्कारभूषणैः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0141"><div class="verse-block verse-justify"><div class="line">ब्राह्मणान्पूजयेत्पश्चात्तेभ्यो दद्याच्च दक्षिणाम्।</div><div class="line" style="position:relative;">येन येन प्रकुर्याच्च लक्षपूजामुमापतेः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0142"><div class="verse-block verse-justify"><div class="line">तत्तद्दद्यात्सुवर्णेन कृत्वा शम्भुं प्रपूजयेत्।</div><div class="line" style="position:relative;">यदि दीपः कृतस्तेन तद्दानं चैव कारयेत्॥३५॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0144"><div class="verse-block verse-justify"><div class="line">क्षमापयेत्ततो देवं ब्राह्मणान्भोजयेच्छतम्।</div><div class="line" style="position:relative;">एवं यः कुरुते पूजां तस्य प्रीणाम्यहं मुने॥३७॥</div></div></div>
