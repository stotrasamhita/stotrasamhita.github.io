<div class="verse-block-wrapper" id="shravana-mahatmyam-b0328"><div class="verse-block verse-justify"><div class="line">उपचारैः षोडशभिर्मङ्गलागौरिसंज्ञिताम्।</div><div class="line" style="position:relative;">दूर्वादलैः षोडशभिरपामार्गदलैस्तथा॥६॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0332"><div class="verse-block verse-justify"><div class="line">प्रतिमां मङ्गलागौर्याः सुवर्णपलनिर्मिताम्।</div><div class="line" style="position:relative;">तदर्थेन तदर्धेन शक्त्या वाप्यथ कारयेत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0333"><div class="verse-block verse-justify"><div class="line">तण्डुलैः पूरिते भाण्डे शक्त्या स्वर्णादिनिर्मिते।</div><div class="line" style="position:relative;">संस्थाप्य परिधानीयं रमणीयां च कञ्चुकीम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0334"><div class="verse-block verse-justify"><div class="line">तयोरुपरि देव्यास्तु प्रतिमां स्थापयेत्ततः।</div><div class="line" style="position:relative;">समीपभागे संस्थाप्य दृषदं चोपलं तथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0335"><div class="verse-block verse-justify"><div class="line">रौप्येण निर्मितं मात्रे एवं दद्यात्तु वायनम्।</div><div class="line" style="position:relative;">षोडशापि सुवासिन्यो भोजनीयाः प्रयत्नतः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0336"><div class="verse-block verse-justify"><div class="line">एवं कृते व्रते व सौभाग्यं सप्तजन्मसु।</div><div class="line" style="position:relative;">पुत्रपौत्रादिभिश्चैव रमते सम्पदा युता॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0346"><div class="verse-block verse-justify"><div class="line">इति तस्य वचः श्रुत्वा देवी प्राह शुचिस्मिता।</div><div class="line" style="position:relative;">दुर्लभं याचितं राजन्दास्ये तुभ्यं कृपावशात्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0347"><div class="verse-block verse-justify"><div class="line">परं शृणुष्व राजेन्द्र पुत्रश्चेद् गुणवत्तरः।</div><div class="line" style="position:relative;">ईप्सितश्चेत्षोडशाब्दं जीविष्यति न चाधिकम्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0352"><div class="verse-block verse-justify"><div class="line">हृष्टो राजा तथा चक्रे पत्नी गर्भं च सा दधौ।</div><div class="line" style="position:relative;">दशमे मासि सुषुवे पुत्रं देवसुतोपमम्॥२७॥</div></div></div>
