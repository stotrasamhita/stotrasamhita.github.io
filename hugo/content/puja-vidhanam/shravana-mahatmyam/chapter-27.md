<div class="verse-block-wrapper" id="shravana-mahatmyam-b1322"><div class="verse-block verse-justify"><div class="line">सिंहकर्कटयोर्मध्ये सर्वा नद्यो रजस्वलाः।</div><div class="line" style="position:relative;">तासु स्नानं न कुर्वीत वर्जयित्वा समुद्रगाः॥२॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b1324"><div class="verse-block verse-justify"><div class="line">तावद्रजोवहा नद्य अल्पतोयाः प्रकीर्तिताः।</div><div class="line" style="position:relative;">याः शोषमुपगच्छन्ति ग्रीष्मे तु सरितो भुवि॥४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1325"><div class="verse-block verse-justify"><div class="line">तासु प्रावृषि न स्नायादपूर्णे दशवासरे।</div><div class="line" style="position:relative;">धनुःसहस्राण्यष्टौ च गतिर्यासां स्वतो नहि॥५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1326"><div class="verse-block verse-justify"><div class="line">न ता नदीशब्दवाच्या गर्तास्ते परिकीर्तिताः।</div><div class="line" style="position:relative;">प्रारम्भे कर्कसङ्क्रान्तेर्महानद्यो रजस्वलाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1327"><div class="verse-block verse-justify"><div class="line">त्रिदिनं तु चतुर्थेऽह्नि शुद्धाः स्युर्योषितो यथा।</div><div class="line" style="position:relative;">महानदीः प्रवक्ष्यामि शृणुष्वावहितो मुने॥७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1328"><div class="verse-block verse-justify"><div class="line">गोदावरी भीमरथी तुङ्गभद्रा च वेणिका।</div><div class="line" style="position:relative;">तापी पयोष्णी विन्ध्यस्य दक्षिणे षट् प्रकीर्तिताः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1329"><div class="verse-block verse-justify"><div class="line">भागीरथी नर्मदा च यमुना च सरस्वती।</div><div class="line" style="position:relative;">विशोका च वितस्ता च विन्ध्यस्योत्तरतोऽपि षट्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1330"><div class="verse-block verse-justify"><div class="line">द्वादशैता महानद्यो देवर्षिक्षेत्रसम्भवाः।</div><div class="line" style="position:relative;">महानद्यो देविका च कावेरी वञ्जरा तथा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1331"><div class="verse-block verse-justify"><div class="line">कृष्णा रजस्वला एताः कर्कटादावहर्मुने।</div><div class="line" style="position:relative;">कर्कटादौ रजोदुष्टा गौतमी वासरत्रयम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1332"><div class="verse-block verse-justify"><div class="line">चन्द्रभागा सती सिन्धुः शरयूर्नर्मदा तथा।</div><div class="line" style="position:relative;">गङ्गा च यमुना चैव प्लक्षजाला सरस्वती॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1333"><div class="verse-block verse-justify"><div class="line">रजसा नाभिभूयन्ते ये चान्ये नदसंज्ञिताः।</div><div class="line" style="position:relative;">शोणः सिन्धुर्हिरण्याख्यः कोकिलाहितघर्घराः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1334"><div class="verse-block verse-justify"><div class="line">शतद्रुश्च नदाः सप्त पावनाः परिकीर्तिताः।</div><div class="line" style="position:relative;">गङ्गा धर्मद्रवा पुण्या यमुना च सरस्वती॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1335"><div class="verse-block verse-justify"><div class="line">अन्तर्गता रजोदोषाः सर्वावस्थासु चामलाः।</div><div class="line" style="position:relative;">अपामयं रजोदोषो न भवेत्तीरवासिनाम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1336"><div class="verse-block verse-justify"><div class="line">जलं रजोदुष्टमपि गङ्गातोयेन पावनम्।</div><div class="line" style="position:relative;">अजा गावो महिष्यश्च योषितश्च प्रसूतिकाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1337"><div class="verse-block verse-justify"><div class="line">भूमेर्नवोदकं चैव दशरात्रेण शुध्यति।</div><div class="line" style="position:relative;">अभावे कूपवापीनामन्यासां च पयोऽमृतम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1338"><div class="verse-block verse-justify"><div class="line">रजोदुष्टेऽपि वयसि ग्रामभोगो न दुष्यति।</div><div class="line" style="position:relative;">अन्येन चोद्धृते नीरे रजोदोषो न विद्यते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1339"><div class="verse-block verse-justify"><div class="line">उपाकर्मणि चोत्सर्गे प्रातः स्नाने विपत्सु च।</div><div class="line" style="position:relative;">चन्द्रसूर्यग्रहे चैव रजोदोषो न विद्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1340"><div class="verse-block verse-justify"><div class="line">अतः परं प्रवक्ष्यामि सिंहे गोप्रसवो यदि।</div><div class="line" style="position:relative;">भानौ सिंहगते चैव यस्य गौः सम्प्रसूयते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1341"><div class="verse-block verse-justify"><div class="line">मरणं तस्य निर्दिष्टं षड्भिर्मासैर्न संशयः।</div><div class="line" style="position:relative;">तत्र शान्तिं प्रवक्ष्यामि येन सम्पाद्यते सुखम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1342"><div class="verse-block verse-justify"><div class="line">प्रसूतां तत्क्षणादेव तां गां विप्राय दापयेत्।</div><div class="line" style="position:relative;">ततो होमं प्रकुर्वीत घृताक्तै राजसर्षपैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1343"><div class="verse-block verse-justify"><div class="line">आहुतीनां घृताक्तानां तिलानां जुहुयात्ततः।</div><div class="line" style="position:relative;">सहस्रेण व्याहृतिभिरष्टसङ्ख्याधिकेन च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1344"><div class="verse-block verse-justify"><div class="line">सोपवासः प्रयत्नेन दद्याद् विप्राय दक्षिणाम्।</div><div class="line" style="position:relative;">सिंहराशौ गते सूर्ये गोष्ठसूतिर्यदा भवेत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1345"><div class="verse-block verse-justify"><div class="line">तदानिष्टं भवेत्किञ्चित्तच्छान्त्यै शान्तिकं चरेत्।</div><div class="line" style="position:relative;">अस्य वामेति सूक्तेन तद्विष्णोरिति मन्त्रतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1346"><div class="verse-block verse-justify"><div class="line">जुहुयाच्च तिलाज्येन शतमष्टोत्तराधिकम्।</div><div class="line" style="position:relative;">मृत्युञ्जयविधानेन जुहुयाच्च तथायुतम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1347"><div class="verse-block verse-justify"><div class="line">श्रीसूक्तेन ततः स्नायाच्छान्तिसूक्तेन वा पुनः।</div><div class="line" style="position:relative;">एवं कृतविधानेन न भयं जायते क्वचित्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1348"><div class="verse-block verse-justify"><div class="line">एवमेव नभोमासि सूयेत वडवा दिने।</div><div class="line" style="position:relative;">अत्रापि शान्तिकं कार्यं तदा दोषो विनश्यति॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1349"><div class="verse-block verse-justify"><div class="line">कर्के सिंहे नभोदानमथ वक्ष्ये शुभप्रदम्।</div><div class="line" style="position:relative;">घृतधेनुप्रदानं च कर्कटस्थे दिवाकरे॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1350"><div class="verse-block verse-justify"><div class="line">ससुवर्णं छत्रदानं शस्तं सिंहे निगद्यते।</div><div class="line" style="position:relative;">श्रावणे वस्त्रदानस्य कीर्तितं सुमहत्फलम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1351"><div class="verse-block verse-justify"><div class="line">घृतं च घृतकुम्भाश्च घृतधेनुः फलानि च।</div><div class="line" style="position:relative;">श्रावणे श्रीधरप्रीत्यै दातव्यानि विपश्चिते॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1352"><div class="verse-block verse-justify"><div class="line">अन्यान्यपि च दानानि मत्तोषाय कृतानि च।</div><div class="line" style="position:relative;">अक्षय्यफलदानि स्युरन्यमासेभ्य एव हि॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1353"><div class="verse-block verse-justify"><div class="line">द्वादशस्वपि मासेषु नास्ति चैतादृशः प्रियः।</div><div class="line" style="position:relative;">आगच्छति नभोमासि प्रतीक्षां च करोम्यहम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1354"><div class="verse-block verse-justify"><div class="line">करिष्यते व्रतं योऽत्र स मे प्रियतरो भवेत्।</div><div class="line" style="position:relative;">ब्राह्मणानां विधू राजा सूर्यः प्रत्यक्षदैवतम्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1355"><div class="verse-block verse-justify"><div class="line">ममाक्षिणी तयोरत्र सङ्क्रान्ती भवतो यतः।</div><div class="line" style="position:relative;">कर्कसंज्ञा सिंहसंज्ञा माहात्म्यं किमतः परम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1356"><div class="verse-block verse-justify"><div class="line">प्रातः स्नानं मासमात्रमत्र यः कुरुते नरः।</div><div class="line" style="position:relative;">द्वादशस्वपि मासेषु प्रातः स्नानफलं लभेत्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1357"><div class="verse-block verse-justify"><div class="line">न करोति नभोमासि प्रातः स्नानं यदा नरः।</div><div class="line" style="position:relative;">द्वादशस्वपि मासेषु कृतं निष्फलतामियात्॥३७॥</div></div></div>
