<div class="verse-block-wrapper" id="kartika-mahatmyam-b0191"><div class="verse-block verse-justify"><div class="line">स्मरेद्गङ्गादिका नद्यो विष्णुशर्वादिदेवताः।</div><div class="line" style="position:relative;">नाभिमात्रे जले स्थित्वा मन्त्रमेतमुदीरयेत्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0196"><div class="verse-block verse-justify"><div class="line">व्रतिनः कार्तिके मासि स्नातस्य विधिवन्मम।</div><div class="line" style="position:relative;">गृहाणार्घ्यं मया दत्तं दनुजेन्द्रनिषूदन॥८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0197"><div class="verse-block verse-justify"><div class="line">किरणा धूतपापा च पुण्यतोया सरस्वती।</div><div class="line" style="position:relative;">गङ्गा च यमुना चैव पञ्चनद्यः पुनन्तु माम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0198"><div class="verse-block verse-justify"><div class="line">अन्यासां च नदीनां च दद्यादर्घ्यं यथाविधि।</div><div class="line" style="position:relative;">जाह्नवीस्मरणं कुर्यात्सर्वतीर्थेषु मानवः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0199"><div class="verse-block verse-justify"><div class="line">नान्यत्तीर्थं तु जाह्नव्यां स्मरणीयं कदाचन।</div><div class="line" style="position:relative;">एतान्मन्त्रान्समुच्चार्य मलस्नानं समाचरेत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0200"><div class="verse-block verse-justify"><div class="line">मृत्स्नानं च पितृस्नानं गुरुस्नानं ततः परम्।</div><div class="line" style="position:relative;">ततस्तु पावमानीभिरभिषिञ्चेत्स्वमस्तकम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0201"><div class="verse-block verse-justify"><div class="line">अघमर्षणकं कृत्वा स्नानाङ्गं तर्पणं तथा।</div><div class="line" style="position:relative;">ततः पुरुषसूक्तेन जलं शिरसि सिञ्चयेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0202"><div class="verse-block verse-justify"><div class="line">ततस्तु बहिरागत्य तीर्थं शिरसि निक्षिपेत्।</div><div class="line" style="position:relative;">तीर्थं पीत्वा त्रिवारं तु तुलसीं गृह्य पाणिना॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0203"><div class="verse-block verse-justify"><div class="line">ततो जलाद्विनिष्क्रम्य चाञ्चलं पीडयेद्बहिः।</div><div class="line" style="position:relative;">यन्मया दूषितं तोयं शारीरमलसञ्चयैः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0204"><div class="verse-block verse-justify"><div class="line">तद्दोषपरिहारार्थं यक्ष्मणं तर्पयाम्यहम्।</div><div class="line" style="position:relative;">वस्त्रनिष्पीडनं कृत्वा कुर्याच्च तिलकादिकम्॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0206"><div class="verse-block verse-justify"><div class="line">शृणुध्वमृषयः सर्वे कार्तिकस्नानजं फलम्।</div><div class="line" style="position:relative;">अरुणं प्रति सूर्येण यदुक्तं च सविस्तरम्॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0211"><div class="verse-block verse-justify"><div class="line">ततो दशगुणं पुण्यं शीततोयनिमज्जनात्।</div><div class="line" style="position:relative;">ततः शतगुणं पुण्यं बहिः कूपोदके कृतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0212"><div class="verse-block verse-justify"><div class="line">कूपात्सहस्रगुणितं फलं वापीनिषेकतः।</div><div class="line" style="position:relative;">ततोऽयुतगुणं पुण्यं तडागस्नानतो भवेत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0213"><div class="verse-block verse-justify"><div class="line">ततो दशगुणं पुण्यं निर्झरेषु निमज्जनात्।</div><div class="line" style="position:relative;">ततोऽधिकतरं पुण्यं नदीस्नानस्य कार्तिके॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0214"><div class="verse-block verse-justify"><div class="line">नद्या दशगुणं प्रोक्तं तीर्थस्नानं खगोत्तम।</div><div class="line" style="position:relative;">ततो दशगुणं पुण्यं नद्योर्यत्र च सङ्गमः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0215"><div class="verse-block verse-justify"><div class="line">नदीत्रयस्य संयोगे पुण्यस्यान्तो न विद्यते।</div><div class="line" style="position:relative;">सिन्धुः कृष्णा च वेणी च यमुना च सरस्वती॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0216"><div class="verse-block verse-justify"><div class="line">गोदावरी विपाशा च नर्मदा तमसा मही।</div><div class="line" style="position:relative;">कावेरी शरयूः शिप्रा तथा चर्मण्वती नदी॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0217"><div class="verse-block verse-justify"><div class="line">वितस्ता वेदिका शोणो वेत्रवत्यपराजिता।</div><div class="line" style="position:relative;">गण्डकी गोमती पूर्णा ब्रह्मपुत्रा सरोवरम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0218"><div class="verse-block verse-justify"><div class="line">वाग्मती च शतद्रुश्च तथा बदरिकाश्रमः।</div><div class="line" style="position:relative;">दुर्लभाः कार्तिके त्वेते तीर्थान्यथ निबोध मे॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0219"><div class="verse-block verse-justify"><div class="line">सर्वेभ्यश्च स्थलेभ्यश्च आर्यावर्तं तु पुण्यदम्।</div><div class="line" style="position:relative;">कोल्हापुरी ततः श्रेष्ठा ततः काञ्चीद्वयं स्मृतम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0220"><div class="verse-block verse-justify"><div class="line">अनन्तसेनवसतिर्वराहक्षेत्रमेव च।</div><div class="line" style="position:relative;">चक्रक्षेत्रं ततः पुण्यं मुक्तिक्षेत्रं ततोऽधिकम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0221"><div class="verse-block verse-justify"><div class="line">अवन्तिका ततः श्रेष्ठा ततो बदरिकाश्रमः।</div><div class="line" style="position:relative;">अयोध्या च ततः श्रेष्ठा गङ्गाद्वारं ततोऽधिकम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0222"><div class="verse-block verse-justify"><div class="line">ततः कनखलं तीर्थं ततो मधुपुरी वरा।</div><div class="line" style="position:relative;">एकोऽपि कार्तिको मासो मथुरायमुनाजले॥३१॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0224"><div class="verse-block verse-justify"><div class="line" style="position:relative;">अतो मधुपुरी श्रेष्ठा यमुना च विशेषतः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0225"><div class="verse-block verse-justify"><div class="line">द्वारावती ततः श्रेष्ठा प्रत्यहं स्नाति केशवः।</div><div class="line" style="position:relative;">षोडशस्त्रीसहस्रेण सार्धं यादवसंयुतः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0226"><div class="verse-block verse-justify"><div class="line">द्वारकायां मृत्तिकायास्तिलको येन मस्तके।</div><div class="line">धार्यतेऽसौ नरो ज्ञेयो जीवन्मुक्तो न संशयः।</div><div class="line" style="position:relative;">द्वारकास्नानमाहात्म्यं न वक्तुं शक्यते मया॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0227"><div class="verse-block verse-justify"><div class="line">गोविन्दार्पितचित्तानां जायते पुण्यभास्करा।</div><div class="line" style="position:relative;">ततो भागीरथी श्रेष्ठा यत्र विन्ध्येन सङ्गता॥३६॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0243"><div class="verse-block verse-justify"><div class="line">कावेर्याश्चैव माहात्म्यं को वदेत्परमुत्तमम्।</div><div class="line" style="position:relative;">अत्र ते वर्णयिष्यामि इतिहासं पुरातनम्॥५१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0244"><div class="verse-block verse-justify"><div class="line">कावेर्या विषये ब्रह्मन्सावधानमनाः शृणु।</div><div class="line" style="position:relative;">गौतम्या उत्तरे तीरे विष्णुपादाब्जसम्भवा॥५२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0245"><div class="verse-block verse-justify"><div class="line">गङ्गा त्रैलोक्यपापघ्नी वर्तते लोकपूजिता।</div><div class="line" style="position:relative;">सा गङ्गा चिन्तयामास कदाचित्पापशङ्किता॥५३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0246"><div class="verse-block verse-justify"><div class="line">सर्वलोकाः समागत्य मयि पापं त्यजन्ति हि।</div><div class="line" style="position:relative;">तत्पापं तु कथं गच्छेदिति चिन्तापरा तदा॥५४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0247"><div class="verse-block verse-justify"><div class="line">प्रष्टुं जगाम कैलासं गिरिजावल्लभं भवम्।</div><div class="line" style="position:relative;">तत्र दृष्ट्वा महारुद्रं प्रोवाच हरिपादजा॥५५॥</div></div></div>
<p class="uvacha">गङ्गोवाच</p>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0249"><div class="verse-block verse-justify"><div class="line">महारुद्र नमस्तेऽस्तु त्वां प्रष्टुमहमागता।</div><div class="line" style="position:relative;">सर्वे लोकाः समागत्य मयि पापं त्यजन्ति हि॥५६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0250"><div class="verse-block verse-justify"><div class="line">तत्पापं तु मया सोढुं न शक्यं पार्वतीपते।</div><div class="line" style="position:relative;">येनोपायेन तत्पापं नाऽऽगच्छेन्मम तद्वद॥५७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0251"><div class="verse-block verse-justify"><div class="line" style="position:relative;">एवं गङ्गावचः श्रुत्वा प्रत्याह परमेश्वरः।</div></div></div>
<p class="uvacha">रुद्र उवाच</p>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0253"><div class="verse-block verse-justify"><div class="line" style="position:relative;">पापनिर्हरणायादौ पद्मनाभाङ्घ्रिपङ्कजात्॥५८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0254"><div class="verse-block verse-justify"><div class="line">प्रादुर्भूताऽसि त्वं देवि किमर्थं तप्यते त्वया।</div><div class="line" style="position:relative;">पापप्रहाराऽऽधिपत्यं कल्पितं तव विष्णुना॥५९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0255"><div class="verse-block verse-justify"><div class="line">तथाऽपि पापनिर्हार उपायं ते ब्रवीम्यहम्।</div><div class="line" style="position:relative;">कवेश्च तनया देवी कावेरी सरितां वरा॥६०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0256"><div class="verse-block verse-justify"><div class="line">सर्वोत्कृष्टा च सर्वेषां हरेर्बलवशात्तु सा।</div><div class="line" style="position:relative;">सर्वपापप्रहरणे सामर्थ्यं तत्र वर्तते॥६१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0257"><div class="verse-block verse-justify"><div class="line">कार्तिके मासि कावेर्यां यः स्नानं कुरुते नरः।</div><div class="line" style="position:relative;">स तु पापविनिर्मुक्तो याति विष्णोः परं पदम्॥६२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0258"><div class="verse-block verse-justify"><div class="line">तस्मात् तां गच्छ देवि त्वं ततः पापाद्विमोक्ष्यसे।</div><div class="line" style="position:relative;">इत्युक्ता सा तदागच्छत्कावेरीं पापहारिणीम्॥६३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0259"><div class="verse-block verse-justify"><div class="line">तज्जलस्पर्शमात्रेण कार्तिके विष्णुपादजा।</div><div class="line" style="position:relative;">निर्धूतपातका गङ्गा जगाम स्वनिकेतनम्॥६४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0260"><div class="verse-block verse-justify"><div class="line">कार्तिके प्रतिवर्षं तु गङ्गा त्रैलोक्यपावनीम्।</div><div class="line" style="position:relative;">स्नातुं भक्त्या समायाति कावेरीं पापहारिणीम्॥६५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0261"><div class="verse-block verse-justify"><div class="line">तज्जलस्पर्शमात्रेण कार्तिके विष्णुपादजा।</div><div class="line" style="position:relative;">निर्धूतपातका गङ्गा जगाम स्वनिकेतनम्॥६६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0262"><div class="verse-block verse-justify"><div class="line">तस्माच्छस्तं तुलास्नानं कावेर्यां शस्यते बुधैः।</div><div class="line" style="position:relative;">यः कावेर्यां तुलास्नानं भक्त्या तु कुरुते मुने॥६७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0263"><div class="verse-block verse-justify"><div class="line">विमुक्तदुरितः सद्यस्ततो याति परां गतिम्।</div><div class="line" style="position:relative;">तस्मात् स्नानं तु कावेर्यां कार्तिके मासि शस्यते॥६८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0264"><div class="verse-block verse-justify"><div class="line">इतिहासमिमं श्रुत्वा कार्तिकव्रततत्परः।</div><div class="line" style="position:relative;">स कावेरीस्नानफलं प्राप्नोति च परां गतिम्॥६९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0265"><div class="verse-block verse-justify"><div class="line">रात्रिशेषे भवेत् स्नानमुत्तमं विष्णुतुष्टिकृत्।</div><div class="line" style="position:relative;">सूर्योदये मध्यमं स्याद्यावन्नाऽऽस्ता तु कृत्तिका॥७०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0266"><div class="verse-block verse-justify"><div class="line">तावदेव भवेत् स्नानमन्यथा तन्न कार्तिकम्।</div><div class="line" style="position:relative;">स्नानं स्त्रीभिर्विधातव्यं गृहीत्वाऽऽज्ञां धवस्य च॥७१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0267"><div class="verse-block verse-justify"><div class="line">अपृष्ट्वा यत्कृतं धर्म्यं भर्तारं तत्क्षयं नयेत्।</div><div class="line" style="position:relative;">स्त्रीणां नास्त्यपरो धर्मो भर्तारं प्रोज्झ्य कश्चन॥७२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0268"><div class="verse-block verse-justify"><div class="line">कुर्यात्सहस्रपापानि भर्त्राज्ञां या समाचरेत्।</div><div class="line" style="position:relative;">सैषा धर्मवती लोके न जायेत व्रतादिना॥७३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0269"><div class="verse-block verse-justify"><div class="line">दरिद्रः पतितो मूर्खो दीनोऽपि यदि चेत्पतिः।</div><div class="line" style="position:relative;">तादृशः शरणं स्त्रीणां तत्त्यागान्निरयं व्रजेत्॥७४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0270"><div class="verse-block verse-justify"><div class="line">कलौ वत्स मनुष्याणां शैथिल्यं स्नानकर्मणि।</div><div class="line" style="position:relative;">तथाऽपि कथयिष्यामि स्नानं कार्तिकमाघयोः॥७५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0271"><div class="verse-block verse-justify"><div class="line">यस्य हस्तौ च पादौ च वाङ्मनश्च सुसंयतम्।</div><div class="line" style="position:relative;">विद्या तपश्च कीर्तिश्च स तीर्थफलभाङ्नरः॥७६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0272"><div class="verse-block verse-justify"><div class="line">अश्रद्दधानः पापात्मा नास्तिकश्छिन्नमानसः।</div><div class="line" style="position:relative;">हेतुवादी च पञ्चैते न तीर्थफलभागिनः॥७७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0273"><div class="verse-block verse-justify"><div class="line">प्रातरुत्थाय यो विप्रस्तीर्थस्नायी सदा भवेत्।</div><div class="line" style="position:relative;">सर्वपापविनिर्मुक्तः परं ब्रह्माधिगच्छति॥७८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0274"><div class="verse-block verse-justify"><div class="line">स्नानं चतुर्विधं प्रोक्तं स्नानविद्भिर्मनीषिभिः।</div><div class="line" style="position:relative;">वायव्यं वारुणं दिव्यं ब्राह्मं चेति तथा स्मृतम्॥७९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0275"><div class="verse-block verse-justify"><div class="line">वायव्यं गोरजःस्नानं वारुणं सागरादिषु।</div><div class="line" style="position:relative;">ब्राह्मं ब्राह्मणमन्त्रोक्तं दिव्यं मेघाम्बु भास्करम्॥८०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0276"><div class="verse-block verse-justify"><div class="line">स्नानानां चैव सर्वेषां विशिष्टं तत्र वारुणम्।</div><div class="line" style="position:relative;">ब्राह्मणः क्षत्रियो वैश्यो मन्त्रवत्स्नानमाचरेत्॥८१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0277"><div class="verse-block verse-justify"><div class="line">तूष्णीमेव हि शूद्रस्य स्त्रीणां चैव तथा स्मृतम्।</div><div class="line" style="position:relative;">बाला च तरुणी वृद्धा नरनारीनपुंसकाः॥८२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0278"><div class="verse-block verse-justify"><div class="line">पापैः सर्वैः प्रमुच्यन्ते स्नानात्कार्तिकमाघयोः।</div><div class="line" style="position:relative;">स्नाता वै कार्तिके लोकाः प्राप्नुवन्तीप्सितं फलम्॥८३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0279"><div class="verse-block verse-justify"><div class="line">पुष्करे तीर्थवर्ये तु नन्दायाः सङ्गमे पुरा।</div><div class="line" style="position:relative;">प्रभञ्जनश्च मुक्तोभूत्तदैव व्याघ्रजन्मतः॥८४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0280"><div class="verse-block verse-justify"><div class="line">नन्दाया वचनेनैव कार्तिके सा परं ययौ।</div><div class="line" style="position:relative;">एवं स्नानविधिः प्रोक्तः किं भूयः श्रोतुमिच्छसि॥८५॥</div></div></div>
