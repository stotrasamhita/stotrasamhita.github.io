<div class="verse-block-wrapper" id="kishkindha-kanda-b0361"><div class="verse-block verse-justify"><div class="line">विशालनयनं शान्तं स्मितचारुमुखाम्बुजम्।</div><div class="line" style="position:relative;">सीताविरहसन्तप्तं पश्यन्तं मृगपक्षिणः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0362"><div class="verse-block verse-justify"><div class="line">रथाद्दूरात्समुत्पत्य वेगात्सुग्रीवलक्ष्मणौ।</div><div class="line" style="position:relative;">रामस्य पादयोरग्रे पेततुर्भक्तिसंयुतौ॥३॥</div></div></div>

<div class="verse-block-wrapper" id="kishkindha-kanda-b0364"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीद् रघुश्रेष्ठं सुग्रीवो भक्तिनम्रधीः।</div><div class="line" style="position:relative;">देव पश्य समायान्तीं वानराणां महाचमूम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0365"><div class="verse-block verse-justify"><div class="line">कुलाचलाद्रिसम्भूता मेरुमन्दरसन्निभाः।</div><div class="line" style="position:relative;">नानाद्वीपसरिच्छैलवासिनः पर्वतोपमाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0366"><div class="verse-block verse-justify"><div class="line">असङ्ख्याताः समायान्ति हरयः कामरूपिणः।</div><div class="line" style="position:relative;">सर्वे देवांशसम्भूताः सर्वे युद्धविशारदाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0367"><div class="verse-block verse-justify"><div class="line">अत्र केचिद्गजबलाः केचिद्दशगजोपमाः।</div><div class="line" style="position:relative;">गजायुतबलाः केचिदन्येऽमितबलाः प्रभो॥८॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0368"><div class="verse-block verse-justify"><div class="line">केचिदञ्जनकूटाभाः केचित्कनकसन्निभाः।</div><div class="line" style="position:relative;">केचिद्रक्तान्तवदना दीर्घवालास्तथाऽपरे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0369"><div class="verse-block verse-justify"><div class="line">शुद्धस्फटिकसङ्काशाः केचिद्राक्षससन्निभाः।</div><div class="line" style="position:relative;">गर्जन्तः परितो यान्ति वानरा युद्धकाङ्क्षिणः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0370"><div class="verse-block verse-justify"><div class="line">त्वदाज्ञाकारिणः सर्वे फलमूलाशनाः प्रभो।</div><div class="line" style="position:relative;">ऋक्षाणामधिपो वीरो जाम्बवान्नाम बुद्धिमान्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0371"><div class="verse-block verse-justify"><div class="line">एष मे मन्त्रिणां श्रेष्ठः कोटिभल्लूकवृन्दपः।</div><div class="line" style="position:relative;">हनूमानेष विख्यातो महासत्त्वपराक्रमः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0372"><div class="verse-block verse-justify"><div class="line">वायुपुत्रोऽतितेजस्वी मन्त्री बुद्धिमतां वरः।</div><div class="line" style="position:relative;">नलो नीलश्च गवयो गवाक्षो गन्धमादनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0373"><div class="verse-block verse-justify"><div class="line">शरभो मैन्दवश्चैव गजः पनस एव च।</div><div class="line" style="position:relative;">वलीमुखो दधिमुखः सुषेणस्तार एव च॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0374"><div class="verse-block verse-justify"><div class="line">केसरी च महासत्त्वः पिता हनुमतो बली।</div><div class="line" style="position:relative;">एते ते यूथपा राम प्राधान्येन मयोदिताः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0375"><div class="verse-block verse-justify"><div class="line">महात्मानो महावीर्याः शक्रतुल्यपराक्रमाः।</div><div class="line" style="position:relative;">एते प्रत्येकतः कोटिकोटिवानरयूथपाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0376"><div class="verse-block verse-justify"><div class="line">तवाज्ञाकारिणः सर्वे सर्वे देवांशसम्भवाः।</div><div class="line" style="position:relative;">एष वालिसुतः श्रीमानङ्गदो नाम विश्रुतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0377"><div class="verse-block verse-justify"><div class="line">वालितुल्यबलो वीरो राक्षसानां बलान्तकः।</div><div class="line" style="position:relative;">एते चान्ये च बहवस्त्वदर्थे त्यक्तजीविताः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0378"><div class="verse-block verse-justify"><div class="line">योद्धारः पर्वताग्रैश्च निपुणाः शत्रुघातने।</div><div class="line" style="position:relative;">आज्ञापय रघुश्रेष्ठ सर्वे ते वशवर्तिनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0379"><div class="verse-block verse-justify"><div class="line">रामः सुग्रीवमालिङ्ग्य हर्षपूर्णाश्रुलोचनः।</div><div class="line" style="position:relative;">प्राह सुग्रीव जानासि सर्वं त्वं कार्यगौरवम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0380"><div class="verse-block verse-justify"><div class="line">मार्गणार्थं हि जानक्या नियुङ्क्ष्व यदि रोचते।</div><div class="line" style="position:relative;">श्रुत्वा रामस्य वचनं सुग्रीवः प्रीतमानसः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0381"><div class="verse-block verse-justify"><div class="line">प्रेषयामास बलिनो वानरान् वानरर्षभः।</div><div class="line" style="position:relative;">दिक्षु सर्वासु विविधान् वानरान् प्रेष्य सत्वरम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0382"><div class="verse-block verse-justify"><div class="line">दक्षिणां दिशमत्यर्थं प्रयत्नेन महाबलान्।</div><div class="line" style="position:relative;">युवराजं जाम्बवन्तं हनूमन्तं महाबलम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0383"><div class="verse-block verse-justify"><div class="line">नलं सुषेणं शरभं मैन्दं द्विविदमेव च।</div><div class="line" style="position:relative;">प्रेषयामास सुग्रीवो वचनं चेदमब्रवीत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0384"><div class="verse-block verse-justify"><div class="line">विचिन्वन्तु प्रयत्नेन भवन्तो जानकीं शुभाम्।</div><div class="line" style="position:relative;">मासादर्वाङ्निवर्तध्वं मच्छासनपुरःसराः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0385"><div class="verse-block verse-justify"><div class="line">सीतामदृष्ट्वा यदि वो मासादूर्ध्वं दिनं भवेत्।</div><div class="line" style="position:relative;">तदा प्राणान्तिकं दण्डं मत्तः प्राप्स्यथ वानराः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0386"><div class="verse-block verse-justify"><div class="line">इति प्रस्थाप्य सुग्रीवो वानरान् भीमविक्रमान्।</div><div class="line" style="position:relative;">रामस्य पार्श्वे श्रीरामं नत्वा चोपविवेश सः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0387"><div class="verse-block verse-justify"><div class="line">गच्छन्तं मारुतिं दृष्ट्वा रामो वचनमब्रवीत्।</div><div class="line" style="position:relative;">अभिज्ञानार्थमेतन्मे ह्यङ्गुलीयकमुत्तमम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0388"><div class="verse-block verse-justify"><div class="line">मन्नामाक्षरसंयुक्तं सीतायै दीयतां रहः।</div><div class="line">अस्मिन् कार्ये प्रमाणं हि त्वमेव कपिसत्तम।</div><div class="line" style="position:relative;">जानामि सत्त्वं ते सर्वं गच्छ पन्थाः शुभस्तव॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0389"><div class="verse-block verse-justify"><div class="line">एवं कपीनां राज्ञा ते विसृष्टाः परिमार्गणे।</div><div class="line" style="position:relative;">सीताया अङ्गदमुखा बभ्रमुस्तत्र तत्र ह॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0390"><div class="verse-block verse-justify"><div class="line">भ्रमन्तो विन्ध्यगहने ददृशुः पर्वतोपमम्।</div><div class="line" style="position:relative;">राक्षसं भीषणाकारं भक्षयन्तं मृगान् गजान्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0391"><div class="verse-block verse-justify"><div class="line">रावणोऽयमिति ज्ञात्वा केचिद् वानरपुङ्गवाः।</div><div class="line" style="position:relative;">जघ्नुः किलकिलाशब्दं मुञ्चन्तो मुष्टिभिः क्षणात्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0392"><div class="verse-block verse-justify"><div class="line">नायं रावण इत्युक्त्वा ययुरन्यन्महद्वनम्।</div><div class="line" style="position:relative;">तृषार्ता सलिलं तत्र नाविन्दन् हरिपुङ्गवाः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0393"><div class="verse-block verse-justify"><div class="line">विभ्रमन्तो महारण्ये शुष्ककण्ठोष्ठतालुकाः।</div><div class="line" style="position:relative;">ददृशुर्गह्वरं तत्र तृणगुल्मावृतं महत्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0394"><div class="verse-block verse-justify"><div class="line">आर्द्रपक्षान् क्रौञ्चहंसान्निःसृतान् ददृशुस्ततः।</div><div class="line" style="position:relative;">अत्रास्ते सलिलं नूनं प्रविशामो महागुहाम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0395"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा हनुमानग्रे प्रविवेश तमन्वयुः।</div><div class="line" style="position:relative;">सर्वे परस्परं धृत्वा बाहून् बाहुभिरुत्सुकाः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0396"><div class="verse-block verse-justify"><div class="line">अन्धकारे महद्दूरं गत्वाऽपश्यन् कपीश्वराः।</div><div class="line" style="position:relative;">जलाशयान् मणिनिभतोयान् कल्पद्रुमोपमान्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0397"><div class="verse-block verse-justify"><div class="line">वृक्षान् पक्वफलैर्नम्रान् मधुद्रोणसमन्वितान्।</div><div class="line" style="position:relative;">गृहान् सर्वगुणोपेतान् मणिवस्त्रादिपूरितान्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0398"><div class="verse-block verse-justify"><div class="line">दिव्यभक्ष्यान्नसहितान् मानुषैः परिवर्जितान्।</div><div class="line" style="position:relative;">विस्मितास्तत्र भवने दिव्ये कनकविष्टरे॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0399"><div class="verse-block verse-justify"><div class="line">प्रभया दीप्यमानां तु ददृशुः स्त्रियमेककाम्।</div><div class="line" style="position:relative;">ध्यायन्तीं चीरवसनां योगिनीं योगमास्थिताम्॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0400"><div class="verse-block verse-justify"><div class="line">प्रणेमुस्तां महाभागां भक्त्या भीत्या च वानराः।</div><div class="line" style="position:relative;">दृष्ट्वा तान् वानरान् देवी प्राह यूयं किमागताः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0401"><div class="verse-block verse-justify"><div class="line">कुतो वा कस्य दूता वा मत्स्थानं किं प्रधर्षथ।</div><div class="line" style="position:relative;">तच्छ्रुत्वा हनुमानाह शृणु वक्ष्यामि देवि ते॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0402"><div class="verse-block verse-justify"><div class="line">अयोध्याधिपतिः श्रीमान् राजा दशरथः प्रभुः।</div><div class="line" style="position:relative;">तस्य पुत्रो महाभागो ज्येष्ठो राम इति श्रुतः॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0403"><div class="verse-block verse-justify"><div class="line">पितुराज्ञां पुरस्कृत्य सभार्यः सानुजो वनम्।</div><div class="line" style="position:relative;">गतस्तत्र हृता भार्या तस्य साध्वी दुरात्मना॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0404"><div class="verse-block verse-justify"><div class="line">रावणेन ततो रामः सुग्रीवं सानुजो ययौ।</div><div class="line" style="position:relative;">सुग्रीवो मित्रभावेन रामस्य प्रियवल्लभाम्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0405"><div class="verse-block verse-justify"><div class="line">मृगयध्वमिति प्राह ततो वयमुपागताः।</div><div class="line" style="position:relative;">ततो वनं विचिन्वन्तो जानकीं जलकाङ्क्षिणः॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="kishkindha-kanda-b0406"><div class="verse-block verse-justify"><div class="line">प्रविष्टा गह्वरं घोरं दैवादत्र समागताः।</div><div class="line" style="position:relative;">त्वं वा किमर्थमत्रासि का वा त्वं वद नः शुभे॥४७॥</div></div></div>
