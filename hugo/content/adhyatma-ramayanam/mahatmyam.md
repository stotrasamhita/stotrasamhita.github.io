<div class="verse-block-wrapper" id="mahatmyam-b0010"><div class="verse-block verse-justify"><div class="line">चतुर्मुखं जगन्नाथं भक्ताभीष्टफलप्रदम्।</div><div class="line" style="position:relative;">प्रणम्य दण्डवद्भक्त्या तुष्टाव मुनिपुङ्गवः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0011"><div class="verse-block verse-justify"><div class="line">सन्तुष्टस्तं मुनिं प्राह स्वयम्भूर्वैष्णवोत्तमम्।</div><div class="line" style="position:relative;">किं प्रष्टुकामस्त्वमसि तद्वदिष्यामि ते मुने॥६॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0013"><div class="verse-block verse-justify"><div class="line">इदानीमेकमेवास्ति श्रोतव्यं सुरसत्तम।</div><div class="line" style="position:relative;">तद्रहस्यमपि ब्रूहि यदि तेऽनुग्रहो मयि॥८॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0015"><div class="verse-block verse-justify"><div class="line">परापवादनिरताः परद्रव्याभिलाषिणः।</div><div class="line" style="position:relative;">परस्त्रीसक्तमनसः परहिंसापरायणाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0016"><div class="verse-block verse-justify"><div class="line">देहात्मदृष्टयो मूढा नास्तिका पशुबुद्धयः।</div><div class="line" style="position:relative;">मातापितृकृतद्वेषाः स्त्रीदेवाः कामकिङ्कराः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0017"><div class="verse-block verse-justify"><div class="line">विप्रा लोभग्रहग्रस्ता वेदविक्रयजीविनः।</div><div class="line" style="position:relative;">धनार्जनार्थमभ्यस्तविद्या मदविमोहिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0018"><div class="verse-block verse-justify"><div class="line">त्यक्तस्वजातिकर्माणः प्रायशः परवञ्चकाः।</div><div class="line" style="position:relative;">क्षत्रियाश्च तथा वैश्याः स्वधर्मत्यागशीलिनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0019"><div class="verse-block verse-justify"><div class="line">तद्वच्छूद्राश्च ये केचिद्ब्राह्मणाचारतत्पराः।</div><div class="line" style="position:relative;">स्त्रियश्च प्रायशो भ्रष्टा भर्त्रवज्ञाननिर्भयाः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0020"><div class="verse-block verse-justify"><div class="line">श्वशुरद्रोहकारिण्यो भविष्यन्ति न संशयः।</div><div class="line" style="position:relative;">एतेषां नष्टबुद्धीनां परलोकः कथं भवेत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0021"><div class="verse-block verse-justify"><div class="line">इति चिन्ताकुलं चित्तं जायते मम सन्ततम्।</div><div class="line">लघूपायेन येनैषां परलोकगतिर्भवेत्।</div><div class="line" style="position:relative;">तमुपायमुपाख्याहि सर्वं वेत्ति यतो भवान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0022"><div class="verse-block verse-justify"><div class="line">इत्यृषेर्वाक्यमाकर्ण्य प्रत्युवाचाम्बुजासनः।</div><div class="line" style="position:relative;">साधु पृष्टं त्वया साधो वक्ष्ये तच्छृणु सादरम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0023"><div class="verse-block verse-justify"><div class="line">पुरा त्रिपुरहन्तारं पार्वती भक्तवत्सला।</div><div class="line" style="position:relative;">श्रीरामतत्त्वं जिज्ञासुः पप्रच्छ विनयान्विता॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0024"><div class="verse-block verse-justify"><div class="line">प्रियायै गिरिशस्तस्यै गूढं व्याख्यातवान् स्वयम्।</div><div class="line" style="position:relative;">पुराणोत्तममध्यात्मरामायणमिति स्मृतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0025"><div class="verse-block verse-justify"><div class="line">तत्पार्वती जगद्धात्री पूजयित्वा दिवानिशम्।</div><div class="line" style="position:relative;">आलोचयन्ती स्वानन्दमग्ना तिष्ठति साम्प्रतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0026"><div class="verse-block verse-justify"><div class="line">प्रचरिष्यति तल्लोके प्राण्यदृष्टवशाद्यदा।</div><div class="line" style="position:relative;">तस्याध्ययनमात्रेण जना यास्यन्ति सद्गतिम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0027"><div class="verse-block verse-justify"><div class="line">तावद्विजृम्भते पापं ब्रह्महत्यापुरःसरम्।</div><div class="line" style="position:relative;">यावज्जगति नाध्यात्मरामायणमुदेष्यति॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0029"><div class="verse-block verse-justify"><div class="line">तावद्यमभटाः शूराः सञ्चरिष्यन्ति निर्भयाः।</div><div class="line" style="position:relative;">यावज्जगति नाध्यात्मरामायणमुदेष्यति॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0031"><div class="verse-block verse-justify"><div class="line">तावत्स्वरूपं रामस्य दुर्बोधं महतामपि।</div><div class="line" style="position:relative;">यावज्जगति नाध्यात्मरामायणमुदेष्यति॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0036"><div class="verse-block verse-justify"><div class="line">यो भक्त्यार्चयतेऽध्यात्मरामायणमतन्द्रितः।</div><div class="line" style="position:relative;">दिने दिनेऽश्वमेधस्य फलं तस्य भवेन्मुने॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="mahatmyam-b0037"><div class="verse-block verse-justify"><div class="line">यदृच्छयाऽपि योऽध्यात्मरामायणमनादरात्।</div><div class="line" style="position:relative;">अन्यतः शृणुयान्मर्त्यः सोऽपि मुच्येत पातकात्॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="mahatmyam-b0044"><div class="verse-block verse-justify"><div class="line">कुरुक्षेत्रादिनिखिलपुण्यतीर्थेष्वनेकशः।</div><div class="line" style="position:relative;">आत्मतुल्यं धनं सूर्यग्रहणे सर्वतोमुखे॥३९॥</div></div></div>
