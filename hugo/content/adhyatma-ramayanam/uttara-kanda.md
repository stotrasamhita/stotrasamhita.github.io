<div class="verse-block-wrapper" id="uttara-kanda-b0013"><div class="verse-block verse-justify"><div class="line">ब्रूहि रामाय मुनयः समागत्य बहिःस्थिताः।</div><div class="line" style="position:relative;">अगस्त्यप्रमुखाः सर्वे आशीर्भिरभिनन्दितुम्॥९॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0015"><div class="verse-block verse-justify"><div class="line">कृताञ्जलिरुवाचेदमगस्त्यो मुनिभिः सह।</div><div class="line" style="position:relative;">देव त्वद्दर्शनार्थाय प्राप्तो बहिरुपस्थितः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0016"><div class="verse-block verse-justify"><div class="line">तमुवाच द्वारपालं प्रवेशय यथासुखम्।</div><div class="line" style="position:relative;">पूजिता विविशुर्वेश्म नानारत्नविभूषितम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0017"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा रामो मुनीन् शीघ्रं प्रत्युत्थाय कृताञ्जलिः।</div><div class="line" style="position:relative;">पाद्यार्घ्यादिभिरापूज्य गां निवेद्य यथाविधि॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0018"><div class="verse-block verse-justify"><div class="line">नत्वा तेभ्यो ददौ दिव्यान्यासनानि यथार्हतः।</div><div class="line" style="position:relative;">उपविष्टा प्रहृष्टाश्च मुनयो रामपूजिताः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0019"><div class="verse-block verse-justify"><div class="line">सम्पृष्टकुशलाः सर्वे रामं कुशलमब्रुवन्।</div><div class="line" style="position:relative;">कुशलं ते महाबाहो सर्वत्र रघुनन्दन॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0020"><div class="verse-block verse-justify"><div class="line">दिष्ट्येदानीं प्रपश्यामो हतशत्रुमरिन्दम।</div><div class="line" style="position:relative;">न हि भारः स ते राम रावणो राक्षसेश्वरः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0021"><div class="verse-block verse-justify"><div class="line">सधनुस्त्वं हि लोकांस्त्रीन् विजेतुं शक्त एव हि।</div><div class="line" style="position:relative;">दिष्ट्या त्वया हताः सर्वे राक्षसा रावणादयः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0022"><div class="verse-block verse-justify"><div class="line">सह्यमेतन्महाबाहो रावणस्य निबर्हणम्।</div><div class="line" style="position:relative;">असह्यमेतत्सम्प्राप्तं रावणेर्यन्निषूदनम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0023"><div class="verse-block verse-justify"><div class="line">अन्तकप्रतिमाः सर्वे कुम्भकर्णादयो मृधे।</div><div class="line" style="position:relative;">अन्तकप्रतिमैर्बाणैर्हतास्ते रघुसत्तम॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0024"><div class="verse-block verse-justify"><div class="line">दत्ता चेयं त्वयाऽस्माकं पुरा ह्यभयदक्षिणा।</div><div class="line" style="position:relative;">हत्वा रक्षोगणान् सङ्ख्ये कृतकृत्योऽद्य जीवसि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0025"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु भाषितं तेषां मुनीनां भावितात्मनाम्।</div><div class="line" style="position:relative;">विस्मयं परमं गत्वा रामः प्राञ्जलिरब्रवीत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0026"><div class="verse-block verse-justify"><div class="line">रावणादीनतिक्रम्य कुम्भकर्णादिराक्षसान्।</div><div class="line" style="position:relative;">त्रिलोकजयिनो हित्वा किं प्रशंसथ रावणिम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0027"><div class="verse-block verse-justify"><div class="line">ततस्तद्वचनं श्रुत्वा राघवस्य महात्मनः।</div><div class="line" style="position:relative;">कुम्भयोनिर्महातेजा रामं प्रीत्या वचोऽब्रवीत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0028"><div class="verse-block verse-justify"><div class="line">शृणु राम यथा वृत्तं रावणे रावणस्य च।</div><div class="line" style="position:relative;">जन्म कर्म वरादानं सङ्क्षेपाद्वदतो मम॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0029"><div class="verse-block verse-justify"><div class="line">पुरा कृतयुगे राम पुलस्त्यो ब्रह्मणः सुतः।</div><div class="line" style="position:relative;">तपस्तप्तुं गतो विद्वान् मेरोः पार्श्वं महामतिः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0030"><div class="verse-block verse-justify"><div class="line">तृणबिन्दोराश्रमेऽसौ न्यवसन्मुनिपुङ्गवः।</div><div class="line" style="position:relative;">तपस्तेपे महातेजाः स्वाध्यायनिरतः सदा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0031"><div class="verse-block verse-justify"><div class="line">तत्राऽऽश्रमे महारम्ये देवगन्धर्वकन्यकाः।</div><div class="line" style="position:relative;">गायन्त्यो ननृतुस्तत्र हसन्त्यो वादयन्ति च॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0032"><div class="verse-block verse-justify"><div class="line">पुलस्त्यस्य तपोविघ्नं चक्रुः सर्वा अनिन्दिताः।</div><div class="line" style="position:relative;">ततः क्रुद्धो महातेजा व्याजहार वचो महत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0033"><div class="verse-block verse-justify"><div class="line">या मे दृष्टिपथं गच्छेत् सा गर्भं धारयिष्यति।</div><div class="line" style="position:relative;">ताः सर्वाः शापसंविग्ना न तं देशं प्रचक्रमुः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0034"><div class="verse-block verse-justify"><div class="line">तृणबिन्दोस्तु राजर्षेः कन्या तन्नाशृणोद्वचः।</div><div class="line" style="position:relative;">विचचार मुनेरग्रे निर्भया तं प्रपश्यती॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0035"><div class="verse-block verse-justify"><div class="line">बभूव पाण्डुरतनुर्व्यञ्जितान्तःशरीरजा।</div><div class="line" style="position:relative;">दृष्ट्वा सा देहवैवर्ण्यं भीता पितरमन्वगात्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0036"><div class="verse-block verse-justify"><div class="line">तृणबिन्दुश्च तां दृष्ट्वा राजर्षिरमितद्युतिः।</div><div class="line" style="position:relative;">ध्यात्वा मुनिकृतं सर्वमवैद्विज्ञानचक्षुषा॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0037"><div class="verse-block verse-justify"><div class="line">तां कन्यां मुनिवर्याय पुलस्त्याय ददौ पिता।</div><div class="line" style="position:relative;">तां प्रगृह्याब्रवीत्कन्यां बाढमित्येव स द्विजः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0038"><div class="verse-block verse-justify"><div class="line">शुश्रूषणपरां दृष्ट्वा मुनिः प्रीतोऽब्रवीद् वचः।</div><div class="line" style="position:relative;">दास्यामि पुत्रमेकं ते उभयोर्वंशवर्धनम्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0039"><div class="verse-block verse-justify"><div class="line">ततः प्रासूत सा पुत्रं पुलस्त्याल्लोकविश्रुतम्।</div><div class="line" style="position:relative;">विश्रवा इति विख्यातः पौलस्त्यो ब्रह्मविन्मुनिः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0040"><div class="verse-block verse-justify"><div class="line">तस्य शीलादिकं दृष्ट्वा भरद्वाजो महामुनिः।</div><div class="line" style="position:relative;">भार्यार्थं स्वां दुहितरं ददौ विश्रवसे मुदा॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0041"><div class="verse-block verse-justify"><div class="line">तस्यां तु पुत्रः सञ्जज्ञे पौलस्त्याल्लोकसम्मतः।</div><div class="line" style="position:relative;">पितृतुल्यो वैश्रवणो ब्रह्मणा चानुमोदितः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0042"><div class="verse-block verse-justify"><div class="line">ददौ तत्तपसा तुष्टो ब्रह्मा तस्मै वरं शुभम्।</div><div class="line" style="position:relative;">मनोऽभिलषितं तस्य धनेशत्वमखण्डितम्॥३८॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0046"><div class="verse-block verse-justify"><div class="line">विश्रवा अपि तं प्राह लङ्कानाम पुरी शुभा।</div><div class="line" style="position:relative;">राक्षसानां निवासाय निर्मिता विश्वकर्मणा॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0047"><div class="verse-block verse-justify"><div class="line">त्यक्त्वा विष्णुभयाद्दैत्या विविशुस्ते रसातलम्।</div><div class="line" style="position:relative;">सा पुरी दुष्प्रधर्षान्यैर्मध्येसागरमास्थिता॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0048"><div class="verse-block verse-justify"><div class="line">तत्र वासाय गच्छ त्वं नान्यैः साधिष्ठिता पुरा।</div><div class="line" style="position:relative;">पित्रादिष्टस्त्वसौ गत्वा तां पुरीं धनदोऽविशत्॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0049"><div class="verse-block verse-justify"><div class="line">स तत्र सुचिरं कालमुवास पितृसम्मतः।</div><div class="line" style="position:relative;">कस्यचित्त्वथ कालस्य सुमाली नाम राक्षसः॥४५॥</div></div></div>

<h2 class="stotra-heading">चतुर्थः सर्गः</h2>
<p class="uvacha">श्री-महादेव उवाच</p>
<div class="verse-block-wrapper" id="uttara-kanda-b0220"><div class="verse-block verse-justify"><div class="line">एकदा ब्रह्मणो लोकादायान्तं नारदं मुनिम्।</div><div class="line" style="position:relative;">पर्यटन् रावणो लोकान् दृष्ट्वा नत्वाऽब्रवीद् वचः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0221"><div class="verse-block verse-justify"><div class="line">भगवन् ब्रूहि मे योद्धुं कुत्र सन्ति महाबलाः।</div><div class="line" style="position:relative;">योद्धुमिच्छामि बलिभिस्त्वं ज्ञाताऽसि जगत्त्रयम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0222"><div class="verse-block verse-justify"><div class="line">मुनिर्ध्यात्वाऽऽह सुचिरं श्वेतद्वीपनिवासिनः।</div><div class="line" style="position:relative;">महाबला महाकायास्तत्र याहि महामते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0223"><div class="verse-block verse-justify"><div class="line">विष्णुपूजारता ये वै विष्णुना निहताश्च ये।</div><div class="line" style="position:relative;">त एव तत्र सञ्जाता अजेयाश्च सुरासुरैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0224"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तद्रावणो वेगान्मन्त्रिभिः पुष्पकेण तान्।</div><div class="line" style="position:relative;">योद्धुकामः समागत्य श्वेतद्वीपसमीपतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0225"><div class="verse-block verse-justify"><div class="line">तत्प्रभाहततेजस्कं पुष्पकं नाचलत्ततः।</div><div class="line" style="position:relative;">त्यक्त्वा विमानं प्रययौ मन्त्रिणश्च दशाननः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0226"><div class="verse-block verse-justify"><div class="line">प्रविशन्नेव तद्द्वीपं धृतो हस्तेन योषिता।</div><div class="line" style="position:relative;">पृष्टश्च त्वं कुतः कोऽसि प्रेषितः केन वा वद॥७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0227"><div class="verse-block verse-justify"><div class="line">इत्युक्तो लीलया स्त्रीभिर्हसन्तीभिः पुनः पुनः।</div><div class="line" style="position:relative;">कृच्छ्राद्धस्ताद्विनिर्मुक्तस्तासां स्त्रीणां दशाननः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0228"><div class="verse-block verse-justify"><div class="line">आश्चर्यमतुलं लब्ध्वा चिन्तयामास दुर्मतिः।</div><div class="line" style="position:relative;">विष्णुना निहतो यामि वैकुण्ठमिति निश्चितः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0229"><div class="verse-block verse-justify"><div class="line">मयि विष्णुर्यथा कुप्येत्तथा कार्यं करोम्यहम्।</div><div class="line" style="position:relative;">इति निश्चित्य वैदेहीं जहार विपिनेऽसुरः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0230"><div class="verse-block verse-justify"><div class="line">जानन्नेव परात्मानं स जहारावनीसुताम्।</div><div class="line" style="position:relative;">मातृवत्पालयामास त्वत्तः काङ्क्षन् वधं स्वकम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0231"><div class="verse-block"><div class="line">राम त्वं परमेश्वरोऽसि सकलं जानासि विज्ञानदृग्</div><div class="line pada-even">भूतं भव्यमिदं त्रिकालकलनासाक्षी विकल्पोज्झितः।</div><div class="line">भक्तानामनुवर्तनाय सकलां कुर्वन् क्रियासंहतिं</div><div class="line pada-even" style="position:relative;">त्वं शृण्वन्मनुजाकृतिर्मुनिवचो भासीश लोकार्चितः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0232"><div class="verse-block verse-justify"><div class="line">स्तुत्वैवं राघवं तेन पूजितः कुम्भसम्भवः।</div><div class="line" style="position:relative;">स्वाश्रमं मुनिभिः सार्धं प्रययौ हृष्टमानसः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0233"><div class="verse-block verse-justify"><div class="line">रामस्तु सीतया सार्धं भ्रातृभिः सह मन्त्रिभिः।</div><div class="line" style="position:relative;">संसारीव रमानाथो रममाणोऽवसद्गृहे॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0239"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा रामचन्द्रोऽपि पौरकार्याणि सर्वशः।</div><div class="line" style="position:relative;">भ्रातृभिर्मन्त्रिभिः सार्धं यथान्यायं चकार सः॥२०॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0241"><div class="verse-block verse-justify"><div class="line">जना धर्मपराः सर्वे पतिभक्तिपराः स्त्रियः।</div><div class="line" style="position:relative;">नापश्यत्पुत्रमरणं कश्चिद्राजनि राघवे॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0245"><div class="verse-block verse-justify"><div class="line">जीवयामास शूद्रस्य ददौ स्वर्गमनुत्तमम्।</div><div class="line" style="position:relative;">लोकानामुपदेशार्थं परमात्मा रघूत्तमः॥२६॥</div></div></div>

<h2 class="stotra-heading">षष्ठः सर्गः</h2>
<p class="uvacha">श्री-महादेव उवाच</p>
<div class="verse-block-wrapper" id="uttara-kanda-b0353"><div class="verse-block verse-justify"><div class="line">एकदा मुनयः सर्वे यमुनातीरवासिनः।</div><div class="line" style="position:relative;">आजग्मू राघवं द्रष्टुं भयाल्लवणरक्षसः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0354"><div class="verse-block verse-justify"><div class="line">कृत्वाऽग्रे तु मुनिश्रेष्ठं भार्गवं च्यवनं द्विजाः।</div><div class="line" style="position:relative;">असङ्ख्याताः समायाता रामादभयकाङ्क्षिणः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0355"><div class="verse-block verse-justify"><div class="line">तान् पूजयित्वा परया भक्त्या रघुकुलोत्तमः।</div><div class="line" style="position:relative;">उवाच मधुरं वाक्यं हर्षयन् मुनिमण्डलम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0356"><div class="verse-block verse-justify"><div class="line">करवाणि मुनिश्रेष्ठाः किमागमनकारणम्।</div><div class="line" style="position:relative;">धन्योऽस्मि यदि यूयं मां प्रीत्या द्रष्टुमिहागताः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0357"><div class="verse-block verse-justify"><div class="line">दुष्करं चापि यत्कार्यं भवतां तत्करोम्यहम्।</div><div class="line" style="position:relative;">आज्ञापयन्तु मां भृत्यं ब्राह्मणा दैवतं हि मे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0358"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा सहसा हृष्टश्च्यवनो वाक्यमब्रवीत्।</div><div class="line" style="position:relative;">मधुनामा महादैत्यः पुरा कृतयुगे प्रभो॥६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0359"><div class="verse-block verse-justify"><div class="line">आसीदतीव धर्मात्मा देवब्राह्मणपूजकः।</div><div class="line" style="position:relative;">तस्य तुष्टो महादेवो ददौ शूलमनुत्तमम्॥७॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0361"><div class="verse-block verse-justify"><div class="line">तस्यां तु लवणो नाम राक्षसो भीमविक्रमः।</div><div class="line" style="position:relative;">आसीद्दुरात्मा दुर्धर्षो देवब्राह्मणहिंसकः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0362"><div class="verse-block verse-justify"><div class="line">पीडितास्तेन राजेन्द्र वयं त्वां शरणं गताः।</div><div class="line" style="position:relative;">तच्छ्रुत्वा राघवोऽप्याह मा भीर्वो मुनिपुङ्गवाः॥१०॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0364"><div class="verse-block verse-justify"><div class="line">लवणं राक्षसं दद्यात् ब्राह्मणेभ्योऽभयं महत्।</div><div class="line" style="position:relative;">तच्छ्रुत्वा प्राञ्जलिः प्राह भरतो राघवाय वै॥१२॥</div></div></div>

<h2 class="stotra-heading">अष्टमः सर्गः</h2>
<p class="uvacha">श्री-महादेव उवाच</p>
<div class="verse-block-wrapper" id="uttara-kanda-b0502"><div class="verse-block verse-justify"><div class="line">अथ काले गते कस्मिन् भरतो भीमविक्रमः।</div><div class="line" style="position:relative;">युधाजिता मातुलेन ह्याहूतोऽगात्ससैनिकः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0503"><div class="verse-block verse-justify"><div class="line">रामाज्ञया गतस्तत्र हत्वा गन्धर्वनायकान्।</div><div class="line" style="position:relative;">तिस्रः कोटीः पुरे द्वे तु निवेश्य रघुनन्दनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0504"><div class="verse-block verse-justify"><div class="line">पुष्करं पुष्करावत्यां तक्षं तक्षशिलाह्वये।</div><div class="line" style="position:relative;">अभिषिच्य सुतौ तत्र धनधान्यसुहृद्वृतौ॥३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0505"><div class="verse-block verse-justify"><div class="line">पुनरागत्य भरतो रामसेवापरोऽभवत्।</div><div class="line" style="position:relative;">ततः प्रीतो रघुश्रेष्ठो लक्ष्मणं प्राह सादरम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0506"><div class="verse-block verse-justify"><div class="line">उभौ कुमारौ सौमित्रे गृहीत्वा पश्चिमां दिशम्।</div><div class="line" style="position:relative;">तत्र भिल्लान् विनिर्जित्य दुष्टान् सर्वापकारिणः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0507"><div class="verse-block verse-justify"><div class="line">अङ्गदश्चित्रकेतुश्च महासत्त्वपराक्रमौ।</div><div class="line" style="position:relative;">द्वयोर्द्वे नगरे कृत्वा गजाश्वधनरत्नकैः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0508"><div class="verse-block verse-justify"><div class="line">अभिषिच्य सुतौ तत्र शीघ्रमागच्छ मां पुनः।</div><div class="line" style="position:relative;">रामस्याज्ञां पुरस्कृत्य गजाश्वबलवाहनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0509"><div class="verse-block verse-justify"><div class="line">गत्वा हत्वा रिपून् सर्वान् स्थापयित्वा कुमारकौ।</div><div class="line" style="position:relative;">सौमित्रिः पुनरागत्य रामसेवापरोऽभवत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0510"><div class="verse-block"><div class="line">ततस्तु काले महति प्रयाते</div><div class="line pada-even">रामं सदा धर्मपथे स्थितं हरिम्।</div><div class="line">द्रष्टुं समागादृषिवेषधारी</div><div class="line pada-even" style="position:relative;">कालस्ततो लक्ष्मणमित्युवाच॥९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0511"><div class="verse-block"><div class="line">निवेदयस्वातिबलस्य दूतं</div><div class="line pada-even">मां द्रष्टुकामं पुरुषोत्तमाय।</div><div class="line">रामाय विज्ञापनमस्ति तस्य</div><div class="line pada-even" style="position:relative;">महर्षिमुख्यस्य चिराय धीमन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0512"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा सौमित्रिस्त्वरयान्वितः।</div><div class="line" style="position:relative;">आचचक्षेऽथ रामाय स सम्प्राप्तं तपोधनम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0513"><div class="verse-block verse-justify"><div class="line">एवं ब्रुवन्तं प्रोवाच लक्ष्मणं राघवो वचः।</div><div class="line" style="position:relative;">शीघ्रं प्रवेश्यतां तात मुनिः सत्कारपूर्वकम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0514"><div class="verse-block verse-justify"><div class="line">लक्ष्मणस्तु तथेत्युक्त्वा प्रावेशयत तापसम्।</div><div class="line" style="position:relative;">स्वतेजसा ज्वलन्तं तं घृतसिक्तं यथाऽनलम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0515"><div class="verse-block verse-justify"><div class="line">सोऽभिगम्य रघुश्रेष्ठं दीप्यमानः स्वतेजसा।</div><div class="line" style="position:relative;">मुनिर्मधुरवाक्येन वर्धस्वेत्याह राघवम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0516"><div class="verse-block verse-justify"><div class="line">तस्मै स मुनये रामः पूजां कृत्वा यथाविधि।</div><div class="line" style="position:relative;">पृष्ट्वाऽनामयमव्यग्रो रामः पृष्टोऽथ तेन सः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0517"><div class="verse-block verse-justify"><div class="line">दिव्यासने समासीनो रामः प्रोवाच तापसम्।</div><div class="line" style="position:relative;">यदर्थमागतोऽसि त्वमिह तत्प्रापयस्व मे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0518"><div class="verse-block verse-justify"><div class="line">वाक्येन चोदितस्तेन रामेणाह मुनिर्वचः।</div><div class="line" style="position:relative;">द्वन्द्वमेव प्रयोक्तव्यमनालक्ष्यं तु तद्वचः॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0520"><div class="verse-block verse-justify"><div class="line">तथेति च प्रतिज्ञाय रामो लक्ष्मणमब्रवीत्।</div><div class="line" style="position:relative;">तिष्ठ त्वं द्वारि सौमित्रे नायात्वत्र जनो रहः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0521"><div class="verse-block verse-justify"><div class="line">यद्यागच्छति को वाऽपि स वध्यो मे न संशयः।</div><div class="line" style="position:relative;">ततः प्राह मुनिं रामो येन वा त्वं विसर्जितः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0522"><div class="verse-block verse-justify"><div class="line">यत्ते मनीषितं वाक्यं तद्वदस्व ममाग्रतः।</div><div class="line" style="position:relative;">ततः प्राह मुनिर्वाक्यं शृणु राम यथातथम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0523"><div class="verse-block verse-justify"><div class="line">ब्रह्मणा प्रेषितोऽस्मीश कार्यार्थे तेऽन्तिकं प्रभो।</div><div class="line" style="position:relative;">अहं हि पूर्वजो देव तव पुत्रः परन्तप॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0524"><div class="verse-block verse-justify"><div class="line">मायासङ्गमजो वीर कालः सर्वहरः स्मृतः।</div><div class="line" style="position:relative;">ब्रह्मा त्वामाह भगवान् सर्वदेवर्षिपूजितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0525"><div class="verse-block verse-justify"><div class="line">रक्षितुं स्वर्गलोकस्य समयस्ते महामते।</div><div class="line" style="position:relative;">पुरा त्वमेक एवासीर्लोकान् संहृत्य मायया॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0526"><div class="verse-block verse-justify"><div class="line">भार्यया सहितस्त्वं मामादौ पुत्रमजीजनः।</div><div class="line" style="position:relative;">तथा भोगवतं नागमनन्तमुदकेशयम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0527"><div class="verse-block verse-justify"><div class="line">मायया जनयित्वा त्वं द्वौ ससत्त्वौ महाबलौ।</div><div class="line" style="position:relative;">मधुकैटभकौ दैत्यौ हत्वा मेदोऽस्थिसञ्चयम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0528"><div class="verse-block verse-justify"><div class="line">इमां पर्वतसम्बद्धां मेदिनीं पुरुषर्षभ।</div><div class="line" style="position:relative;">पद्मे दिव्यार्कसङ्काशे नाभ्यामुत्पाद्य मामपि॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0529"><div class="verse-block verse-justify"><div class="line">मां विधाय प्रजाध्यक्षं मयि सर्वं न्यवेदयत्।</div><div class="line" style="position:relative;">सोऽहं संयुक्तसम्भारस्त्वामवोचं जगत्पते॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0530"><div class="verse-block verse-justify"><div class="line">रक्षां विधत्स्व भूतेभ्यो ये मे वीर्यापहारिणः।</div><div class="line" style="position:relative;">ततस्त्वं कश्यपाज्जातो विष्णुर्वामनरूपधृक्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0531"><div class="verse-block verse-justify"><div class="line">हृतवानसि भूभारं वधाद्रक्षोगणस्य च।</div><div class="line" style="position:relative;">सर्वासूत्सार्यमाणासु प्रजासु धरणीधर॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="uttara-kanda-b0532"><div class="verse-block verse-justify"><div class="line">रावणस्य वधाकाङ्क्षी मर्त्यलोकमुपागतः।</div><div class="line" style="position:relative;">दशवर्षसहस्राणि दशवर्षशतानि च॥३१॥</div></div></div>

<div class="verse-block-wrapper" id="uttara-kanda-b0640"><div class="verse-block"><div class="line">श्रुत्वा हरेर्वाक्यमथाब्रवीत्कः</div><div class="line pada-even">सान्तानिकान् यान्तु विचित्रभोगान्।</div><div class="line">लोकान्मदीयोपरि दीप्यमानान्</div><div class="line pada-even" style="position:relative;">त्वद्भावयुक्ताः कृतपुण्यपुञ्जाः॥६२॥</div></div></div>
