<div class="verse-block-wrapper" id="aranya-kanda-b0056"><div class="verse-block verse-justify"><div class="line">अभिगम्य सुसम्पूज्य विष्टरेषूपवेशयत्।</div><div class="line" style="position:relative;">आतिथ्यमकरोत्तेषां कन्दमूलफलादिभिः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0057"><div class="verse-block verse-justify"><div class="line">प्रीत्याऽऽह शरभङ्गोऽपि रामं भक्तिपरायणम्।</div><div class="line" style="position:relative;">बहुकालमिहैवाऽऽसं तपसे कृतनिश्चयः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0058"><div class="verse-block verse-justify"><div class="line">अद्य मत्तपसा सिद्धं यत्पुण्यं बहु विद्यते।</div><div class="line" style="position:relative;">तत्सर्वं तव दास्यामि ततो मुक्तिं व्रजाम्यहम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0059"><div class="verse-block"><div class="line">समर्प्य रामस्य महत्सुपुण्यफलं</div><div class="line pada-even">विरक्तः शरभङ्गयोगी।</div><div class="line">चितिं समारोहयदप्रमेयं</div><div class="line pada-even" style="position:relative;">रामं ससीतं सहसा प्रणम्य॥६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0060"><div class="verse-block"><div class="line">ध्यायन्श्चिरं राममशेषहृत्स्थं</div><div class="line pada-even">दूर्वादलश्यामलमम्बुजाक्षम्।</div><div class="line">चीराम्बरं स्निग्धजटाकलापं</div><div class="line pada-even" style="position:relative;">सीतासहायं सहलक्ष्मणं तम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0061"><div class="verse-block"><div class="line">को वा दयालुः स्मृतकामधेनुरन्यो</div><div class="line pada-even">जगत्यां रघुनायकादहो।</div><div class="line">स्मृतो मया नित्यमनन्यभाजा</div><div class="line pada-even" style="position:relative;">ज्ञात्वा स्मृतिं मे स्वयमेव यातः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0062"><div class="verse-block verse-justify"><div class="line">पश्यत्विदानीं देवेशो रामो दाशरथिः प्रभुः।</div><div class="line" style="position:relative;">दग्ध्वा स्वदेहं गच्छामि ब्रह्मलोकमकल्मषः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0063"><div class="verse-block verse-justify"><div class="line">अयोध्याधिपतिर्मेऽस्तु हृदये राघवः सदा।</div><div class="line" style="position:relative;">यद् वामाङ्के स्थिता सीता मेघस्येव तटिल्लता॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0064"><div class="verse-block verse-justify"><div class="line">इति रामं चिरं ध्यात्वा दृष्ट्वा च पुरतः स्थितम्।</div><div class="line" style="position:relative;">प्रज्वाल्य सहसा वह्निं दग्ध्वा पञ्चात्मकं वपुः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0065"><div class="verse-block verse-justify"><div class="line">दिव्यदेहधरः साक्षाद्ययौ लोकपतेः पदम्।</div><div class="line">ततो मुनिगणाः सर्वे दण्डकारण्यवासिनः।</div><div class="line" style="position:relative;">आजग्मू राघवं द्रष्टुं शरभङ्गनिवेशनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0066"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा मुनिसमूहं तं जानकीरामलक्ष्मणाः।</div><div class="line" style="position:relative;">प्रणेमुः सहसा भूमौ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0067"><div class="verse-block verse-justify"><div class="line">आशीर्भिरभिनन्द्याथ रामं सर्वहृदि स्थितम्।</div><div class="line" style="position:relative;">ऊचुः प्राञ्जलयः सर्वे धनुर्बाणधरं हरिम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0068"><div class="verse-block verse-justify"><div class="line">भूमेर्भारावताराय जातोऽसि ब्रह्मणार्थितः।</div><div class="line" style="position:relative;">जानीमस्त्वां हरिं लक्ष्मीं जानकीं लक्ष्मणं तथा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0069"><div class="verse-block verse-justify"><div class="line">शेषांशं शङ्खचक्रे द्वे भरतं सानुजं तथा।</div><div class="line" style="position:relative;">अतश्चादौ ऋषीणां त्वं दुःखं मोक्तुमिहार्हसि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0070"><div class="verse-block"><div class="line">आगच्छ यामो मुनिसेवितानि</div><div class="line pada-even">वनानि सर्वाणि रघूत्तम क्रमात्।</div><div class="line">द्रष्टुं सुमित्रासुतजानकीभ्यां</div><div class="line pada-even" style="position:relative;">तदा दयाऽस्मासु दृढा भविष्यति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0071"><div class="verse-block verse-justify"><div class="line">इति विज्ञापितो रामः कृताञ्जलिपुटैर्विभुः।</div><div class="line" style="position:relative;">जगाम मुनिभिः सार्धं द्रष्टुं मुनिवनानि सः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0072"><div class="verse-block verse-justify"><div class="line">ददर्श तत्र पतितान्यनेकानि शिरांसि सः।</div><div class="line" style="position:relative;">अस्थिभूतानि सर्वत्र रामो वचनमब्रवीत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0073"><div class="verse-block verse-justify"><div class="line">अस्थीनि केषामेतानि किमर्थं पतितानि वै।</div><div class="line" style="position:relative;">तमूचुर्मुनयो राम ऋषीणां मस्तकानि हि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0074"><div class="verse-block verse-justify"><div class="line">राक्षसैर्भक्षितानीश प्रमत्तानां समाधितः।</div><div class="line" style="position:relative;">अन्तरायं मुनीनां ते पश्यन्तोऽनुचरन्ति हि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0075"><div class="verse-block verse-justify"><div class="line">श्रुत्वा वाक्यं मुनीनां स भयदैन्यसमन्वितम्।</div><div class="line" style="position:relative;">प्रतिज्ञामकरोद्रामो वधायाशेषरक्षसाम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0076"><div class="verse-block verse-justify"><div class="line">पूज्यमानः सदा तत्र मुनिभिर्वनवासिभिः।</div><div class="line" style="position:relative;">जानक्या सहितो रामो लक्ष्मणेन समन्वितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0077"><div class="verse-block verse-justify"><div class="line">उवास कतिचित् तत्र वर्षाणि रघुनन्दनः।</div><div class="line" style="position:relative;">एवं क्रमेण सम्पश्यन्नृषीणामाश्रमान् विभुः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0078"><div class="verse-block verse-justify"><div class="line">सुतीक्ष्णस्याश्रमं प्रागात्प्रख्यातमृषिसङ्कुलम्।</div><div class="line" style="position:relative;">सर्वर्तुगुणसम्पन्नं सर्वकालसुखावहम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0079"><div class="verse-block verse-justify"><div class="line">राममागतमाकर्ण्य सुतीक्ष्णः स्वयमागतः।</div><div class="line">अगस्त्यशिष्यो रामस्य मन्त्रोपासनतत्परः।</div><div class="line" style="position:relative;">विधिवत्पूजयामास भक्त्युत्कण्ठितलोचनः॥२६॥</div></div></div>
<p class="uvacha">सुतीक्ष्ण उवाच</p>
<div class="verse-block-wrapper" id="aranya-kanda-b0081"><div class="verse-block"><div class="line">त्वन्मन्त्रजाप्यहमनन्तगुणाप्रमेय</div><div class="line pada-even">सीतापते शिवविरिञ्चिसमाश्रिताङ्घ्रे।</div><div class="line">संसारसिन्धुतरणामलपोतपाद</div><div class="line pada-even" style="position:relative;">रामाभिराम सततं तव दासदासः॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="aranya-kanda-b0085"><div class="verse-block"><div class="line">प्रत्यक्षतोऽद्य भवतश्चरणारविन्दं</div><div class="line pada-even">पश्यामि राम तमसः परतः स्थितस्य।</div><div class="line">दृग्रूपतस्त्वमसतामविगोचरोऽपि</div><div class="line pada-even" style="position:relative;">त्वन्मन्त्रपूतहृदयेषु सदा प्रसन्नः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0086"><div class="verse-block"><div class="line">पश्यामि राम तव रूपमरूपिणोऽपि</div><div class="line pada-even">मायाविडम्बनकृतं सुमनुष्यवेषम्।</div><div class="line">कन्दर्पकोटिसुभगं कमनीयचाप-</div><div class="line pada-even" style="position:relative;">बाणं दयार्द्रहृदयं स्मितचारुवक्त्रम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0087"><div class="verse-block"><div class="line">सीतासमेतमजिनाम्बरमप्रधृष्यं</div><div class="line pada-even">सौमित्रिणा नियतसेवितपादपद्मम्।</div><div class="line">नीलोत्पलद्युतिमनन्तगुणं प्रशान्तं</div><div class="line pada-even" style="position:relative;">मद्भागधेयमनिशं प्रणमामि रामम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0088"><div class="verse-block"><div class="line">जानन्तु राम तव रूपमशेषदेश-</div><div class="line pada-even">कालाद्युपाधिरहितं घनचित्प्रकाशम्।</div><div class="line">प्रत्यक्षतोऽद्य मम गोचरमेतदेव</div><div class="line pada-even" style="position:relative;">रूपं विभातु हृदये न परं विकाङ्क्षे॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0089"><div class="verse-block verse-justify"><div class="line">इत्येवं स्तुवतस्तस्य रामः सस्मितमब्रवीत्।</div><div class="line" style="position:relative;">मुने जानामि ते चित्तं निर्मलं मदुपासनात्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0090"><div class="verse-block verse-justify"><div class="line">अतोऽहमागतो द्रष्टुं मदृते नान्यसाधनम्।</div><div class="line" style="position:relative;">मन्मन्त्रोपासका लोके मामेव शरणं गताः॥३६॥</div></div></div>
