<div class="verse-block-wrapper" id="aranya-kanda-b0218"><div class="verse-block verse-justify"><div class="line">राक्षसी राघवं प्राह कस्य त्वं कः किमाश्रमे।</div><div class="line" style="position:relative;">युक्तो जटावल्कलाद्यैः साध्यं किं तेऽत्र मे वद॥५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0219"><div class="verse-block verse-justify"><div class="line">अहं शूर्पणखा नाम राक्षसी कामरूपिणी।</div><div class="line" style="position:relative;">भगिनी राक्षसेन्द्रस्य रावणस्य महात्मनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0220"><div class="verse-block verse-justify"><div class="line">खरेण सहिता भ्रात्रा वसाम्यत्रैव कानने।</div><div class="line" style="position:relative;">राज्ञा दत्तं च मे सर्वं मुनिभक्षा वसाम्यहम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0221"><div class="verse-block verse-justify"><div class="line">त्वां तु वेदितुमिच्छामि वद मे वदतां वर।</div><div class="line" style="position:relative;">तामाह रामनामाहमयोध्याधिपतेः सुतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0222"><div class="verse-block verse-justify"><div class="line">एषा मे सुन्दरी भार्या सीता जनकनन्दिनी।</div><div class="line" style="position:relative;">स तु भ्राता कनीयान् मे लक्ष्मणोऽतीवसुन्दरः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0223"><div class="verse-block verse-justify"><div class="line">किं कृत्यं ते मया ब्रूहि कार्यं भुवनसुन्दरि।</div><div class="line" style="position:relative;">इति रामवचः श्रुत्वा कामार्ता साब्रवीदिदम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0224"><div class="verse-block verse-justify"><div class="line">एहि राम मया सार्धं रमस्व गिरिकानने।</div><div class="line" style="position:relative;">कामार्ताहं न शक्नोमि त्यक्तुं त्वां कमलेक्षणम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0225"><div class="verse-block verse-justify"><div class="line">रामः सीतां कटाक्षेण पश्यन् सस्मितमब्रवीत्।</div><div class="line" style="position:relative;">भार्या ममैषा कल्याणी विद्यते ह्यनपायिनी॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0226"><div class="verse-block verse-justify"><div class="line">त्वं तु सापत्न्यदुःखेन कथं स्थास्यसि सुन्दरि।</div><div class="line" style="position:relative;">बहिरास्ते मम भ्राता लक्ष्मणोऽतीव सुन्दरः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0227"><div class="verse-block verse-justify"><div class="line">तवानुरूपो भविता पतिस्तेनैव सञ्चर।</div><div class="line" style="position:relative;">इत्युक्ता लक्ष्मणं प्राह पतिर्मे भव सुन्दर॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0228"><div class="verse-block verse-justify"><div class="line">भ्रातुराज्ञां पुरस्कृत्य सङ्गच्छावोऽद्य मा चिरम्।</div><div class="line" style="position:relative;">इत्याह राक्षसी घोरा लक्ष्मणं काममोहिता॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0229"><div class="verse-block verse-justify"><div class="line">तामाह लक्ष्मणः साध्वि दासोऽहं तस्य धीमतः।</div><div class="line" style="position:relative;">दासी भविष्यसि त्वं तु ततो दुःखतरं नु किम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0230"><div class="verse-block verse-justify"><div class="line">तमेव गच्छ भद्रं ते स तु राजाखिलेश्वरः।</div><div class="line" style="position:relative;">तच्छ्रुत्वा पुनरप्यागाद्राघवं दुष्टमानसा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0231"><div class="verse-block verse-justify"><div class="line">क्रोधाद्राम किमर्थं मां भ्रामयस्यनवस्थितः।</div><div class="line" style="position:relative;">इदानीमेव तां सीतां भक्षयामि तवाग्रतः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0232"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा विकटाकारा जानकीमनुधावति।</div><div class="line" style="position:relative;">ततो रामाज्ञया खड्गमादाय परिगृह्य ताम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0233"><div class="verse-block verse-justify"><div class="line">चिच्छेद नासां कर्णौ च लक्ष्मणो लघुविक्रमः।</div><div class="line" style="position:relative;">ततो घोरध्वनिं कृत्वा रुधिराक्तवपुर्द्रुतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0234"><div class="verse-block verse-justify"><div class="line">क्रन्दमाना पपाताग्रे खरस्य परुषाक्षरा।</div><div class="line" style="position:relative;">किमेतदिति तामाह खरः खरतराक्षरः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0235"><div class="verse-block verse-justify"><div class="line">केनैवं कारितासि त्वं मृत्योर्वक्त्रानुवर्तिना।</div><div class="line" style="position:relative;">वद मे तं वधिष्यामि कालकल्पमपि क्षणात्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0236"><div class="verse-block verse-justify"><div class="line">तमाह राक्षसी रामः सीतालक्ष्मणसंयुतः।</div><div class="line" style="position:relative;">दण्डकं निर्भयं कुर्वन्नास्ते गोदावरीतटे॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0237"><div class="verse-block verse-justify"><div class="line">मामेवं कृतवान्स्तस्य भ्राता तेनैव चोदितः।</div><div class="line" style="position:relative;">यदि त्वं कुलजातोऽसि वीरोऽसि जहि तौ रिपू॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0238"><div class="verse-block verse-justify"><div class="line">तयोस्तु रुधिरं पास्ये भक्षयैतौ सुदुर्मदौ।</div><div class="line" style="position:relative;">नो चेत्प्राणान् परित्यज्य यास्यामि यमसादनम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0239"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा त्वरितं प्रागात्खरः क्रोधेन मूर्च्छितः।</div><div class="line" style="position:relative;">चतुर्दश सहस्राणि रक्षसां भीमकर्मणाम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0240"><div class="verse-block verse-justify"><div class="line">चोदयामास रामस्य समीपं वधकाङ्क्षया।</div><div class="line" style="position:relative;">खरश्च त्रिशिराश्चैव दूषणश्चैव राक्षसः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0241"><div class="verse-block verse-justify"><div class="line">सर्वे रामं ययुः शीघ्रं नानाप्रहरणोद्यताः।</div><div class="line" style="position:relative;">श्रुत्वा कोलाहलं तेषां रामः सौमित्रिमब्रवीत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0242"><div class="verse-block verse-justify"><div class="line">श्रूयते विपुलः शब्दो नूनमायान्ति राक्षसाः।</div><div class="line" style="position:relative;">भविष्यति महद्युद्धं नूनमद्य मया सह॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0243"><div class="verse-block verse-justify"><div class="line">सीतां नीत्वा गुहां गत्वा तत्र तिष्ठ महाबल।</div><div class="line" style="position:relative;">हन्तुमिच्छाम्यहं सर्वान् राक्षसान् घोररूपिणः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0244"><div class="verse-block verse-justify"><div class="line">अत्र किञ्चिन्न वक्तव्यं शापितोऽसि ममोपरि।</div><div class="line" style="position:relative;">तथेति सीतामादाय लक्ष्मणो गह्वरं ययौ॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0245"><div class="verse-block verse-justify"><div class="line">रामः परिकरं बद्ध्वा धनुरादाय निष्ठुरम्।</div><div class="line" style="position:relative;">तूणीरावक्षयशरौ बद्ध्वायत्तोऽभवत्प्रभुः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0246"><div class="verse-block verse-justify"><div class="line">तत आगत्य रक्षांसि रामस्योपरि चिक्षिपुः।</div><div class="line" style="position:relative;">आयुधानि विचित्राणि पाषाणान् पादपानपि॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="aranya-kanda-b0249"><div class="verse-block verse-justify"><div class="line">लक्ष्मणोऽपि गुहामध्यात्सीतामादाय राघवे।</div><div class="line" style="position:relative;">समर्प्य राक्षसान् दृष्ट्वा हतान् विस्मयमाययौ॥३६॥</div></div></div>
