<div class="verse-block-wrapper" id="aranya-kanda-b0327"><div class="verse-block verse-justify"><div class="line">अथ रामोऽपि तत्सर्वं ज्ञात्वा रावणचेष्टितम्।</div><div class="line" style="position:relative;">उवाच सीतामेकान्ते शृणु जानकि मे वचः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0328"><div class="verse-block verse-justify"><div class="line">रावणो भिक्षुरूपेण आगमिष्यति तेऽन्तिकम्।</div><div class="line" style="position:relative;">त्वं तु छायां त्वदाकारां स्थापयित्वोटजे विश॥२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0329"><div class="verse-block verse-justify"><div class="line">अग्नावदृश्यरूपेण वर्षं तिष्ठ ममाऽऽज्ञया।</div><div class="line" style="position:relative;">रावणस्य वधान्ते मां पूर्ववत्प्राप्स्यसे शुभे॥३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0330"><div class="verse-block verse-justify"><div class="line">श्रुत्वा रामोदितं वाक्यं साऽपि तत्र तथाऽकरोत्।</div><div class="line" style="position:relative;">मायासीतां बहिः स्थाप्य स्वयमन्तर्दधेऽनले॥४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0331"><div class="verse-block verse-justify"><div class="line">मायासीता तदाऽपश्यन्मृगं मायाविनिर्मितम्।</div><div class="line" style="position:relative;">हसन्ती राममभ्येत्य प्रोवाच विनयान्विता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0332"><div class="verse-block verse-justify"><div class="line">पश्य राम मृगं चित्रं कानकं रत्नभूषितम्।</div><div class="line">विचित्रबिन्दुभिर्युक्तं चरन्तमकुतोभयम्।</div><div class="line" style="position:relative;">बद्ध्वा देहि मम क्रीडामृगो भवतु सुन्दरः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0333"><div class="verse-block verse-justify"><div class="line">तथेति धनुरादाय गच्छन् लक्ष्मणमब्रवीत्।</div><div class="line" style="position:relative;">रक्ष त्वमतियत्नेन सीतां मत्प्राणवल्लभाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0334"><div class="verse-block verse-justify"><div class="line">मायिनः सन्ति विपिने राक्षसा घोरदर्शनाः।</div><div class="line" style="position:relative;">अतोऽत्रावहितः साध्वीं रक्ष सीतामनिन्दिताम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0335"><div class="verse-block verse-justify"><div class="line">लक्ष्मणो राममाहेदं देवायं मृगरूपधृक्।</div><div class="line" style="position:relative;">मारीचोऽत्र न सन्देह एवम्भूतो मृगः कुतः॥९॥</div></div></div>
<p class="uvacha">श्रीराम उवाच</p>
<div class="verse-block-wrapper" id="aranya-kanda-b0337"><div class="verse-block verse-justify"><div class="line">यदि मारीच एवायं तदा हन्मि न संशयः।</div><div class="line" style="position:relative;">मृगश्चेदानयिष्यामि सीताविश्रमहेतवे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0338"><div class="verse-block verse-justify"><div class="line">गमिष्यामि मृगं बद्ध्वा ह्यानयिष्यामि सत्वरः।</div><div class="line" style="position:relative;">त्वं प्रयत्नेन सन्तिष्ठ सीतासंरक्षणोद्यतः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0339"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा प्रययौ रामो मायामृगमनुद्रुतः।</div><div class="line" style="position:relative;">माया यदाश्रया लोकमोहिनी जगदाकृतिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0340"><div class="verse-block verse-justify"><div class="line">निर्विकारश्चिदात्माऽपि पूर्णोऽपि मृगमन्वगात्।</div><div class="line" style="position:relative;">भक्तानुकम्पी भगवानिति सत्यं वचो हरिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0341"><div class="verse-block verse-justify"><div class="line">कर्तुं सीताप्रियार्थाय जानन्नपि मृगं ययौ।</div><div class="line" style="position:relative;">अन्यथा पूर्णकामस्य रामस्य विदितात्मनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0342"><div class="verse-block verse-justify"><div class="line" style="position:relative;">मृगेण वा स्त्रिया वाऽपि किं कार्यं परमात्मनः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0343"><div class="verse-block verse-justify"><div class="line">कदाचिद् दृश्यतेऽभ्याशे क्षणं धावति लीयते।</div><div class="line">दृश्यते च ततो दूरादेवं राममपाहरत्।</div><div class="line" style="position:relative;">ततो रामोऽपि विज्ञाय राक्षसोऽयमिति स्फुटम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0344"><div class="verse-block verse-justify"><div class="line">विव्याध शरमादाय राक्षसं मृगरूपिणम्।</div><div class="line" style="position:relative;">पपात रुधिराक्तास्यो मारीचः पूर्वरूपधृक्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0345"><div class="verse-block verse-justify"><div class="line">हा हतोऽस्मि महाबाहो त्राहि लक्ष्मण मां द्रुतम्।</div><div class="line" style="position:relative;">इत्युक्त्वा रामवद् वाचा पपात रुधिराशनः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0346"><div class="verse-block verse-justify"><div class="line">यन्नामाज्ञोऽपि मरणे स्मृत्वा तत्साम्यमाप्नुयात्।</div><div class="line" style="position:relative;">किमुताग्रे हरिं पश्यन्स्तेनैव निहतोऽसुरः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0347"><div class="verse-block verse-justify"><div class="line">तद्देहादुत्थितं तेजः सर्वलोकस्य पश्यतः।</div><div class="line" style="position:relative;">राममेवाविशद् देवा विस्मयं परमं ययुः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0348"><div class="verse-block verse-justify"><div class="line">किं कर्म कृत्वा किं प्राप्तः पातकी मुनिहिंसकः।</div><div class="line" style="position:relative;">अथवा राघवस्यायं महिमा नात्र संशयः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0349"><div class="verse-block verse-justify"><div class="line">रामबाणेन संविद्धः पूर्वं राममनुस्मरन्।</div><div class="line" style="position:relative;">भयात्सर्वं परित्यज्य गृहवित्तादिकं च यत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0350"><div class="verse-block verse-justify"><div class="line">हृदि रामं सदा ध्यात्वा निर्धूताशेषकल्मषः।</div><div class="line" style="position:relative;">अन्ते रामेण निहतः पश्यन् राममवाप सः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0351"><div class="verse-block verse-justify"><div class="line">द्विजो वा राक्षसो वाऽपि पापी वा धार्मिकोऽपि वा।</div><div class="line" style="position:relative;">त्यजन् कलेवरं रामं स्मृत्वा याति परं पदम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0352"><div class="verse-block verse-justify"><div class="line">इति तेऽन्योन्यमाभाष्य ततो देवा दिवं ययुः।</div><div class="line" style="position:relative;">रामस्तच्चिन्तयामास म्रियमाणोऽसुराधमः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0353"><div class="verse-block verse-justify"><div class="line">हा लक्ष्मणेति मद्वाक्यमनुकुर्वन् ममार किम्।</div><div class="line" style="position:relative;">श्रुत्वा मद्वाक्यसदृशं वाक्यं सीताऽपि किं भवेत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0354"><div class="verse-block verse-justify"><div class="line">इति चिन्तापरीतात्मा रामो दूरान्न्यवर्तत।</div><div class="line" style="position:relative;">सीता तद् भाषितं श्रुत्वा मारीचस्य दुरात्मनः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0355"><div class="verse-block verse-justify"><div class="line">भीतातिदुःखसंविग्ना लक्ष्मणं त्विदमब्रवीत्।</div><div class="line" style="position:relative;">गच्छ लक्ष्मण वेगेन भ्राता तेऽसुरपीडितः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0356"><div class="verse-block verse-justify"><div class="line">हा लक्ष्मणेति वचनं भ्रातुस्ते न शृणोषि किम्।</div><div class="line" style="position:relative;">तामाह लक्ष्मणो देवि रामवाक्यं न तद्भवेत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0357"><div class="verse-block verse-justify"><div class="line">यः कश्चिद्राक्षसो देवि म्रियमाणोऽब्रवीद् वचः।</div><div class="line" style="position:relative;">रामस्त्रैलोक्यमपि यः क्रुद्धो नाशयति क्षणात्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0358"><div class="verse-block verse-justify"><div class="line">स कथं दीनवचनं भाषतेऽमरपूजितः।</div><div class="line" style="position:relative;">क्रुद्धा लक्ष्मणमालोक्य सीता बाष्पविलोचना॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0359"><div class="verse-block verse-justify"><div class="line">प्राह लक्ष्मण दुर्बुद्धे भ्रातुर्व्यसनमिच्छसि।</div><div class="line" style="position:relative;">प्रेषितो भरतेनैव रामनाशाभिकाङ्क्षिणा॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="aranya-kanda-b0361"><div class="verse-block verse-justify"><div class="line">न जानातीदृशं रामस्त्वां भार्याहरणोद्यतम्।</div><div class="line" style="position:relative;">रामादन्यं न स्पृशामि त्वां वा भरतमेव वा॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="aranya-kanda-b0362"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा वध्यमाना सा स्वबाहुभ्यां रुरोद ह।</div><div class="line" style="position:relative;">तच्छ्रुत्वा लक्ष्मणः कर्णौ पिधायातीव दुःखितः॥३५॥</div></div></div>
