<div class="verse-block-wrapper" id="sundara-kanda-b0066"><div class="verse-block verse-justify"><div class="line">सीतान्वेषणकार्यार्थी प्रविवेश नृपालयम्।</div><div class="line" style="position:relative;">तत्र सर्वप्रदेशेषु विविच्य हनुमान् कपिः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0067"><div class="verse-block verse-justify"><div class="line">नापश्यज्जानकीं स्मृत्वा ततो लङ्काभिभाषितम्।</div><div class="line" style="position:relative;">जगाम हनुमान् शीघ्रमशोकवनिकां शुभाम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="sundara-kanda-b0069"><div class="verse-block verse-justify"><div class="line">फलैरानम्रशाखाग्रपादपैः परिवारिताम्।</div><div class="line" style="position:relative;">विचिन्वन् जानकीं तत्र प्रतिवृक्षं मरुत्सुतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0070"><div class="verse-block verse-justify"><div class="line">ददर्शाभ्रंलिहं तत्र चैत्यप्रासादमुत्तमम्।</div><div class="line" style="position:relative;">दृष्ट्वा विस्मयमापन्नो मणिस्तम्भशतान्वितम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0071"><div class="verse-block verse-justify"><div class="line">समतीत्य पुनर्गत्वा किञ्चिद् दूरं स मारुतिः।</div><div class="line" style="position:relative;">ददर्श शिंशपावृक्षमत्यन्तनिबिडच्छदम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0072"><div class="verse-block verse-justify"><div class="line">अदृष्टातपमाकीर्णं स्वर्णवर्णविहङ्गमम्।</div><div class="line" style="position:relative;">तन्मूले राक्षसीमध्ये स्थितां जनकनन्दिनीम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0073"><div class="verse-block verse-justify"><div class="line">ददर्श हनुमान् वीरो देवतामिव भूतले।</div><div class="line" style="position:relative;">एकवेणीं कृशां दीनां मलिनाम्बरधारिणीम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0074"><div class="verse-block verse-justify"><div class="line">भूमौ शयानां शोचन्तीं रामरामेति भाषिणीम्।</div><div class="line" style="position:relative;">त्रातारं नाधिगच्छन्तीमुपवासकृशां शुभाम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0075"><div class="verse-block verse-justify"><div class="line">शाखान्तच्छदमध्यस्थो ददर्श कपिकुञ्जरः।</div><div class="line" style="position:relative;">कृतार्थोऽहं कृतार्थोऽहं दृष्ट्वा जनकनन्दिनीम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0076"><div class="verse-block verse-justify"><div class="line">मयैव साधितं कार्यं रामस्य परमात्मनः।</div><div class="line" style="position:relative;">ततः किलकिलाशब्दो बभूवान्तःपुराद्बहिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0077"><div class="verse-block verse-justify"><div class="line">किमेतदिति सँल्लीनो वृक्षपत्रेषु मारुतिः।</div><div class="line" style="position:relative;">आयान्तं रावणं तत्र स्त्रीजनैः परिवारितम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0078"><div class="verse-block verse-justify"><div class="line">दशास्यं विंशतिभुजं नीलाञ्जनचयोपमम्।</div><div class="line" style="position:relative;">दृष्ट्वा विस्मयमापन्नः पत्रखण्डेष्वलीयत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0079"><div class="verse-block verse-justify"><div class="line">रावणो राघवेणाशु मरणं मे कथं भवेत्।</div><div class="line" style="position:relative;">सीतार्थमपि नाऽऽयाति रामः किं कारणं भवेत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0080"><div class="verse-block verse-justify"><div class="line">इत्येवं चिन्तयन्नित्यं राममेव सदा हृदि।</div><div class="line" style="position:relative;">तस्मिन् दिनेऽपररात्रौ रावणो राक्षसाधिपः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0081"><div class="verse-block verse-justify"><div class="line">स्वप्ने रामेण सन्दिष्टः कश्चिदागत्य वानरः।</div><div class="line" style="position:relative;">कामरूपधरः सूक्ष्मो वृक्षग्रस्थोऽनुपश्यति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0082"><div class="verse-block verse-justify"><div class="line">इति दृष्ट्वाऽद्भुतं स्वप्नं स्वात्मन्येवानुचिन्त्य सः।</div><div class="line" style="position:relative;">स्वप्नः कदाचित्सत्यः स्यादेवं तत्र करोम्यहम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0083"><div class="verse-block verse-justify"><div class="line">जानकीं वाक्शरैर्विद्ध्वा दुःखितां नितरामहम्।</div><div class="line" style="position:relative;">करोमि दृष्ट्वा रामाय निवेदयतु वानरः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0084"><div class="verse-block verse-justify"><div class="line">इत्येवं चिन्तयन् सीतासमीपमगमद्द्रुतम्।</div><div class="line" style="position:relative;">नूपुराणां किङ्किणीनां श्रुत्वा शिञ्जितमङ्गना॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0085"><div class="verse-block verse-justify"><div class="line">सीता भीता लीयमाना स्वात्मन्येव सुमध्यमा।</div><div class="line" style="position:relative;">अधोमुख्यश्रुनयना स्थिता रामार्पितान्तरा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0086"><div class="verse-block verse-justify"><div class="line">रावणोऽपि तदा सीतामालोक्याऽऽह सुमध्यमे।</div><div class="line" style="position:relative;">मां दृष्ट्वा किं वृथा सुभ्रु स्वात्मन्येव विलीयसे॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0087"><div class="verse-block verse-justify"><div class="line">रामो वनचराणां हि मध्ये तिष्ठति सानुजः।</div><div class="line" style="position:relative;">कदाचिद्दृश्यते कैश्चित्कदाचिन्नैव दृश्यते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0088"><div class="verse-block verse-justify"><div class="line">मया तु बहुधा लोकाः प्रेषितास्तस्य दर्शने।</div><div class="line" style="position:relative;">न पश्यन्ति प्रयत्नेन वीक्षमाणाः समन्ततः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0089"><div class="verse-block verse-justify"><div class="line">किं करिष्यसि रामेण निःस्पृहेण सदा त्वयि।</div><div class="line" style="position:relative;">त्वया सदाऽऽलिङ्गितोऽपि समीपस्थोऽपि सर्वदा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0090"><div class="verse-block verse-justify"><div class="line">हृदयेऽस्य न च स्नेहस्त्वयि रामस्य जायते।</div><div class="line" style="position:relative;">त्वत्कृतान् सर्वभोगांश्च त्वद्गुणानपि राघवः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0091"><div class="verse-block verse-justify"><div class="line">भुञ्जानोऽपि न जानाति कृतघ्नो निर्गुणोऽधमः।</div><div class="line" style="position:relative;">त्वमानीता मया साध्वी दुःखशोकसमाकुला॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0092"><div class="verse-block verse-justify"><div class="line">इदानीमपि नाऽऽयाति भक्तिहीनः कथं व्रजेत्।</div><div class="line" style="position:relative;">निःसत्त्वो निर्ममो मानी मूढः पण्डितमानवान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0093"><div class="verse-block verse-justify"><div class="line">नराधमं त्वद्विमुखं किं करिष्यसि भामिनि।</div><div class="line" style="position:relative;">त्वय्यतीव समासक्तं मां भजस्वासुरोत्तमम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0094"><div class="verse-block verse-justify"><div class="line">देवगन्धर्वनागानां यक्षकिन्नरयोषिताम्।</div><div class="line" style="position:relative;">भविष्यसि नियोक्त्री त्वं यदि मां प्रतिपद्यसे॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0095"><div class="verse-block verse-justify"><div class="line">रावणस्य वचः श्रुत्वा सीताऽमर्षसमन्विता।</div><div class="line" style="position:relative;">उवाचाधोमुखी भूत्वा निधाय तृणमन्तरे॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0096"><div class="verse-block verse-justify"><div class="line">राघवाद्बिभ्यता नूनं भिक्षुरूपं त्वया धृतम्।</div><div class="line" style="position:relative;">रहिते राघवाभ्यां त्वं शुनीव हविरध्वरे॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0097"><div class="verse-block verse-justify"><div class="line">हृतवानसि मां नीच तत्फलं प्राप्स्यसेऽचिरात्।</div><div class="line" style="position:relative;">यदा रामशराघातविदारितवपुर्भवान्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0098"><div class="verse-block verse-justify"><div class="line">ज्ञास्यसेऽमानुषं रामं गमिष्यसि यमान्तिकम्।</div><div class="line" style="position:relative;">समुद्रं शोषयित्वा वा शरैर्बद्ध्वाऽथ वारिधिम्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0099"><div class="verse-block verse-justify"><div class="line">हन्तुं त्वां समरे रामो लक्ष्मणेन समन्वितः।</div><div class="line" style="position:relative;">आगमिष्यत्यसन्देहो द्रक्ष्यसे राक्षसाधम॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0100"><div class="verse-block verse-justify"><div class="line">त्वां सपुत्रं सहबलं हत्वा नेष्यति मां पुरम्।</div><div class="line" style="position:relative;">श्रुत्वा रक्षःपतिः क्रुद्धो जानक्याः परुषाक्षरम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0101"><div class="verse-block verse-justify"><div class="line">वाक्यं क्रोधसमाविष्टः खड्गमुद्यम्य सत्वरः।</div><div class="line" style="position:relative;">हन्तुं जनकराजस्य तनयां ताम्रलोचनः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0102"><div class="verse-block verse-justify"><div class="line">मन्दोदरी निवार्याह पतिं पतिहिते रता।</div><div class="line" style="position:relative;">त्यजैनां मानुषीं दीनां दुःखितां कृपणां कृशाम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0103"><div class="verse-block verse-justify"><div class="line">देवगन्धर्वनागानां बह्व्यः सन्ति वराङ्गनाः।</div><div class="line" style="position:relative;">त्वामेव वरयन्त्युच्चैर्मदमत्तविलोचनाः॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0104"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीद् दशग्रीवो राक्षसीर्विकृताननाः।</div><div class="line">यथा मे वशगा सीता भविष्यति सकामना।</div><div class="line" style="position:relative;">तथा यतध्वं त्वरितं तर्जनादरणादिभिः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0105"><div class="verse-block verse-justify"><div class="line">द्विमासाभ्यन्तरे सीता यदि मे वशगा भवेत्।</div><div class="line" style="position:relative;">तदा सर्वसुखोपेता राज्यं भोक्ष्यति सा मया॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0106"><div class="verse-block verse-justify"><div class="line">यदि मासद्वयादूर्ध्वं मच्छय्यां नाभिनन्दति।</div><div class="line" style="position:relative;">तदा मे प्रातराशाय हत्वा कुरुत मानुषीम्॥४२॥</div></div></div>

<div class="verse-block-wrapper" id="sundara-kanda-b0108"><div class="verse-block verse-justify"><div class="line">तत्रैका जानकीमाह यौवनं ते वृथा गतम्।</div><div class="line" style="position:relative;">रावणेन समासाद्य सफलं तु भविष्यति॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0109"><div class="verse-block verse-justify"><div class="line">अपरा चाह कोपेन किं विलम्बेन जानकि।</div><div class="line" style="position:relative;">इदानीं छेद्यतामङ्गं विभज्य च पृथक् पृथक्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0110"><div class="verse-block verse-justify"><div class="line">अन्या तु खड्गमुद्यम्य जानकीं हन्तुमुद्यता।</div><div class="line" style="position:relative;">अन्या करालवदना विदार्यास्यमभीषयत्॥४६॥</div></div></div>
