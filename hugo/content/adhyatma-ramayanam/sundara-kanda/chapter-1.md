<div class="verse-block-wrapper" id="sundara-kanda-b0009"><div class="verse-block verse-justify"><div class="line">ऋजुग्रीवोर्ध्वदृष्टिः सन्नाकुञ्चितपदद्वयः।</div><div class="line" style="position:relative;">दक्षिणाभिमुखस्तूर्णं पुप्लुवेऽनिलविक्रमः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0010"><div class="verse-block verse-justify"><div class="line">आकाशात्त्वरितं देवैर्वीक्ष्यमाणो जगाम सः।</div><div class="line" style="position:relative;">दृष्ट्वाऽनिलसुतं देवा गच्छन्तं वायुवेगतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0011"><div class="verse-block verse-justify"><div class="line">परीक्षणार्थं सत्त्वस्य वानरस्येदमब्रुवन्।</div><div class="line" style="position:relative;">गच्छत्येष महासत्त्वो वानरो वायुविक्रमः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0012"><div class="verse-block verse-justify"><div class="line">लङ्कां प्रवेष्टुं शक्तो वा न वा जानीमहे बलम्।</div><div class="line" style="position:relative;">एवं विचार्य नागानां मातरं सुरसाभिधाम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0013"><div class="verse-block verse-justify"><div class="line">अब्रवीद् देवतावृन्दः कौतूहलसमन्वितः।</div><div class="line" style="position:relative;">गच्छ त्वं वानरेन्द्रस्य किञ्चिद् विघ्नं समाचर॥११॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0014"><div class="verse-block verse-justify"><div class="line">ज्ञात्वा तस्य बलं बुद्धिं पुनरेहि त्वरान्विता।</div><div class="line" style="position:relative;">इत्युक्ता सा ययौ शीघ्रं हनुमद्विघ्नकारणात्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0015"><div class="verse-block verse-justify"><div class="line">आवृत्य मार्गं पुरतः स्थित्वा वानरमब्रवीत्।</div><div class="line" style="position:relative;">एहि मे वदनं शीघ्रं प्रविशस्व महामते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0016"><div class="verse-block verse-justify"><div class="line">देवैस्त्वं कल्पितो भक्ष्यः क्षुधासम्पीडितात्मनः।</div><div class="line" style="position:relative;">तामाह हनुमान् मातरहं रामस्य शासनात्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0017"><div class="verse-block verse-justify"><div class="line">गच्छामि जानकीं द्रष्टुं पुनरागम्य सत्वरः।</div><div class="line" style="position:relative;">रामाय कुशलं तस्याः कथयित्वा त्वदाननम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0018"><div class="verse-block verse-justify"><div class="line">निवेक्ष्ये देहि मे मार्गं सुरसायै नमोऽस्तु ते।</div><div class="line" style="position:relative;">इत्युक्ता पुनरेवाह सुरसा क्षुधितािऽस्म्यहम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0019"><div class="verse-block verse-justify"><div class="line">प्रविश्य गच्छ मे वक्त्रं नो चेत्त्वां भक्षयाम्यहम्।</div><div class="line" style="position:relative;">इत्युक्तो हनुमानाह मुखं शीघ्रं विदारय॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0020"><div class="verse-block verse-justify"><div class="line">प्रविश्य वदनं तेऽद्य गच्छामि त्वरयान्वितः।</div><div class="line" style="position:relative;">इत्युक्त्वा योजनायामदेहो भूत्वा पुरः स्थितः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0021"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा हनूमतो रूपं सुरसा पञ्चयोजनम्।</div><div class="line" style="position:relative;">मुखं चकार हनुमान् द्विगुणं रूपमादधत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0022"><div class="verse-block verse-justify"><div class="line">ततश्चकार सुरसा योजनानां च विंशतिम्।</div><div class="line" style="position:relative;">वक्त्रं चकार हनुमान्स्त्रिंशद्योजनसम्मितम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0023"><div class="verse-block verse-justify"><div class="line">ततश्चकार सुरसा पञ्चाशद्योजनायतम्।</div><div class="line" style="position:relative;">वक्त्रं तदा हनूमांस्तु बभूवाङ्गुष्ठसन्निभः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0024"><div class="verse-block verse-justify"><div class="line">प्रविश्य वदनं तस्याः पुनरेत्य पुरः स्थितः।</div><div class="line" style="position:relative;">प्रविष्टो निर्गतोऽहं ते वदनं देवि ते नमः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0025"><div class="verse-block verse-justify"><div class="line">एवं वदन्तं दृष्ट्वा सा हनूमन्तमथाब्रवीत्।</div><div class="line" style="position:relative;">गच्छ साधय रामस्य कार्यं बुद्धिमतां वर॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0026"><div class="verse-block verse-justify"><div class="line">देवैः सम्प्रेषिताऽहं ते बलं जिज्ञासुभिः कपे।</div><div class="line" style="position:relative;">दृष्ट्वा सीतां पुनर्गत्वा रामं द्रक्ष्यसि गच्छ भोः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0027"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा सा ययौ देवलोकं वायुसुतः पुनः।</div><div class="line" style="position:relative;">जगाम वायुमार्गेण गरुत्मानिव पक्षिराट्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0028"><div class="verse-block verse-justify"><div class="line">समुद्रोऽप्याह मैनाकं मणिकाञ्चनपर्वतम्।</div><div class="line" style="position:relative;">गच्छत्येष महासत्त्वो हनुमान्मारुतात्मजः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0029"><div class="verse-block verse-justify"><div class="line">रामस्य कार्यसिद्ध्यर्थं तस्य त्वं सचिवो भव।</div><div class="line" style="position:relative;">सगरैर्वर्द्धितो यस्मात्पुराहं सागरोऽभवम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0030"><div class="verse-block verse-justify"><div class="line">तस्यान्वये बभूवासौ रामो दाशरथिः प्रभुः।</div><div class="line" style="position:relative;">तस्य कार्यार्थसिद्ध्यर्थं गच्छत्येष महाकपिः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0031"><div class="verse-block verse-justify"><div class="line">त्वमुत्तिष्ठ जलात्तूर्णं त्वयि विश्रम्य गच्छतु।</div><div class="line" style="position:relative;">स तथेति प्रादुरभूज्जलमध्यान्महोन्नतः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0032"><div class="verse-block verse-justify"><div class="line">नानामणिमयैः शृङ्गैस्तस्योपरि नराकृतिः।</div><div class="line" style="position:relative;">प्राह यान्तं हनूमन्तं मैनाकोऽहं महाकपे॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0033"><div class="verse-block verse-justify"><div class="line">समुद्रेण समादिष्टस्त्वद्विश्रामाय मारुते।</div><div class="line" style="position:relative;">आगच्छामृतकल्पानि जग्ध्वा पक्वफलानि मे॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0034"><div class="verse-block verse-justify"><div class="line">विश्रम्यात्र क्षणं पश्चाद्गमिष्यसि यथासुखम्।</div><div class="line" style="position:relative;">एवमुक्तोऽथ तं प्राह हनुमान्मारुतात्मजः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0035"><div class="verse-block verse-justify"><div class="line">गच्छतो रामकार्यार्थं भक्षणं मे कथं भवेत्।</div><div class="line" style="position:relative;">विश्रामो वा कथं मे स्याद्गन्तव्यं त्वरितं मया॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0036"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा स्पृष्टशिखरः कराग्रेण ययौ कपिः।</div><div class="line" style="position:relative;">किञ्चिद् दूरं गतस्यास्य छायां छायाग्रहोऽग्रहीत्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0037"><div class="verse-block verse-justify"><div class="line">सिंहिका नाम सा घोरा जलमध्ये स्थिता सदा।</div><div class="line" style="position:relative;">आकाशगामिनां छायामाक्रम्याऽऽकृष्य भक्षयेत्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0038"><div class="verse-block verse-justify"><div class="line">तया गृहीतो हनुमान्श्चिन्तयामास वीर्यवान्।</div><div class="line" style="position:relative;">केनेदं मे कृतं वेगरोधनं विघ्नकारिणा॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0039"><div class="verse-block verse-justify"><div class="line">दृश्यते नैव कोऽप्यत्र विस्मयो मे प्रजायते।</div><div class="line" style="position:relative;">एवं विचिन्त्य हनुमानधो दृष्टिं प्रसारयत्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0040"><div class="verse-block verse-justify"><div class="line">तत्र दृष्ट्वा महाकायां सिंहिकां घोररूपिणीम्।</div><div class="line" style="position:relative;">पपात सलिले तूर्णं पद्भ्यामेवाहनद्रुषा॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0041"><div class="verse-block verse-justify"><div class="line">पुनरुत्प्लुत्य हनुमान् दक्षिणाभिमुखो ययौ।</div><div class="line" style="position:relative;">ततो दक्षिणमासाद्य कूलं नानाफलद्रुमम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0042"><div class="verse-block verse-justify"><div class="line">नानापक्षिमृगाकीर्णं नानापुष्पलतावृतम्।</div><div class="line" style="position:relative;">ततो ददर्श नगरं त्रिकूटाचलमूर्धनि॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0043"><div class="verse-block verse-justify"><div class="line">प्राकारैर्बहुभिर्युक्तं परिखाभिश्च सर्वतः।</div><div class="line" style="position:relative;">प्रवेक्ष्यामि कथं लङ्कामिति चिन्तापरोऽभवत्॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0044"><div class="verse-block verse-justify"><div class="line">रात्रौ वेक्ष्यामि सूक्ष्मोऽहं लङ्कां रावणपालिताम्।</div><div class="line" style="position:relative;">एवं विचिन्त्य तत्रैव स्थित्वा लङ्कां जगाम सः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0045"><div class="verse-block verse-justify"><div class="line">धृत्वा सूक्ष्मं वपुर्द्वारं प्रविवेश प्रतापवान्।</div><div class="line" style="position:relative;">तत्र लङ्कापुरी साक्षाद्राक्षसीवेषधारिणी॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0046"><div class="verse-block verse-justify"><div class="line">प्रविशन्तं हनूमन्तं दृष्ट्वा लङ्का व्यतर्जयत्।</div><div class="line" style="position:relative;">कस्त्वं वानररूपेण मामनादृत्य लङ्किनीम्॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0047"><div class="verse-block verse-justify"><div class="line">प्रविश्य चोरवद्रात्रौ किं भवान् कर्तुमिच्छति।</div><div class="line" style="position:relative;">इत्युक्त्वा रोषताम्राक्षी पादेनाभिजघान तम्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0048"><div class="verse-block verse-justify"><div class="line">हनुमानपि तां वाममुष्टिनाऽवज्ञयाऽहनत्।</div><div class="line" style="position:relative;">तदैव पतिता भूमौ रक्तमुद्वमती भृशम्॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0049"><div class="verse-block verse-justify"><div class="line">उत्थाय प्राह सा लङ्का हनूमन्तं महाबलम्।</div><div class="line" style="position:relative;">हनूमन् गच्छ भद्रं ते जिता लङ्का त्वयाऽनघ॥४७॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0050"><div class="verse-block verse-justify"><div class="line">पुराहं ब्रह्मणा प्रोक्ता ह्यष्टाविंशतिपर्यये।</div><div class="line" style="position:relative;">त्रेतायुगे दाशरथी रामो नारायणोऽव्ययः॥४८॥</div></div></div>
<div class="verse-block-wrapper" id="sundara-kanda-b0051"><div class="verse-block verse-justify"><div class="line">जनिष्यते योगमाया सीता जनकवेश्मनि।</div><div class="line" style="position:relative;">भूभारहरणार्थाय प्रार्थितोऽयं मया क्वचित्॥४९॥</div></div></div>
