<div class="stotra-article"><h2 class="stotra-heading">षष्ठोऽध्यायः---आत्मसंयमयोगः</h2>
<p class="uvacha">श्रीभगवानुवाच</p>
<div class="verse-block-wrapper" id="gita-b0269"><div class="verse-block verse-justify"><div class="line">अनाश्रितः कर्मफलं कार्यं कर्म करोति यः।</div><div class="line" style="position:relative;">स सन्न्यासी च योगी च न निरग्निर्न चाक्रियः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0270"><div class="verse-block verse-justify"><div class="line">यं सन्न्यासमिति प्राहुर्योगं तं विद्धि पाण्डव।</div><div class="line" style="position:relative;">न ह्यसन्न्यस्तसङ्कल्पो योगी भवति कश्चन॥२॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0271"><div class="verse-block verse-justify"><div class="line">आरुरुक्षोर्मुनेर्योगं कर्म कारणमुच्यते।</div><div class="line" style="position:relative;">योगारूढस्य तस्यैव शमः कारणमुच्यते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0272"><div class="verse-block verse-justify"><div class="line">यदा हि नेन्द्रियार्थेषु न कर्मस्वनुषज्जते।</div><div class="line" style="position:relative;">सर्वसङ्कल्पसन्न्यासी योगारूढस्तदोच्यते॥४॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0273"><div class="verse-block verse-justify"><div class="line">उद्धरेदात्मनाऽऽत्मानं नाऽऽत्मानमवसादयेत्।</div><div class="line" style="position:relative;">आत्मैव ह्यात्मनो बन्धुरात्मैव रिपुरात्मनः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0274"><div class="verse-block verse-justify"><div class="line">बन्धुरात्माऽऽत्मनस्तस्य येनाऽऽत्मैवाऽऽत्मना जितः।</div><div class="line" style="position:relative;">अनात्मनस्तु शत्रुत्वे वर्तेताऽऽत्मैव शत्रुवत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0275"><div class="verse-block verse-justify"><div class="line">जितात्मनः प्रशान्तस्य परमात्मा समाहितः।</div><div class="line" style="position:relative;">शीतोष्णसुखदुःखेषु तथा मानापमानयोः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0276"><div class="verse-block verse-justify"><div class="line">ज्ञानविज्ञानतृप्तात्मा कूटस्थो विजितेन्द्रियः।</div><div class="line" style="position:relative;">युक्त इत्युच्यते योगी समलोष्टाश्मकाञ्चनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0277"><div class="verse-block verse-justify"><div class="line">सुहृन्मित्रार्युदासीनमध्यस्थद्वेष्यबन्धुषु।</div><div class="line" style="position:relative;">साधुष्वपि च पापेषु समबुद्धिर्विशिष्यते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0278"><div class="verse-block verse-justify"><div class="line">योगी युञ्जीत सततमात्मानं रहसि स्थितः।</div><div class="line" style="position:relative;">एकाकी यतचित्तात्मा निराशीरपरिग्रहः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0279"><div class="verse-block verse-justify"><div class="line">शुचौ देशे प्रतिष्ठाप्य स्थिरमासनमात्मनः।</div><div class="line" style="position:relative;">नात्युच्छ्रितं नातिनीचं चैलाजिनकुशोत्तरम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0280"><div class="verse-block verse-justify"><div class="line">तत्रैकाग्रं मनः कृत्वा यतचित्तेन्द्रियक्रियः।</div><div class="line" style="position:relative;">उपविश्याऽऽसने युञ्ज्याद् योगमात्मविशुद्धये॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0281"><div class="verse-block verse-justify"><div class="line">समं कायशिरोग्रीवं धारयन्नचलं स्थिरः।</div><div class="line" style="position:relative;">सम्प्रेक्ष्य नासिकाग्रं स्वं दिशश्चानवलोकयन्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0282"><div class="verse-block verse-justify"><div class="line">प्रशान्तात्मा विगतभीर्ब्रह्मचारिव्रते स्थितः।</div><div class="line" style="position:relative;">मनः संयम्य मच्चित्तो युक्त आसीत मत्परः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0283"><div class="verse-block verse-justify"><div class="line">युञ्जन्नेवं सदाऽऽत्मानं योगी नियतमानसः।</div><div class="line" style="position:relative;">शान्तिं निर्वाणपरमां मत्संस्थामधिगच्छति॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0284"><div class="verse-block verse-justify"><div class="line">नात्यश्नतस्तु योगोऽस्ति न चैकान्तमनश्नतः।</div><div class="line" style="position:relative;">न चाति स्वप्नशीलस्य जाग्रतो नैव चार्जुन॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0285"><div class="verse-block verse-justify"><div class="line">युक्ताहारविहारस्य युक्तचेष्टस्य कर्मसु।</div><div class="line" style="position:relative;">युक्तस्वप्नावबोधस्य योगो भवति दुःखहा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0286"><div class="verse-block verse-justify"><div class="line">यदा विनियतं चित्तमात्मन्येवावतिष्ठते।</div><div class="line" style="position:relative;">निःस्पृहः सर्वकामेभ्यो युक्त इत्युच्यते तदा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0287"><div class="verse-block verse-justify"><div class="line">यथा दीपो निवातस्थो नेङ्गते सोपमा स्मृता।</div><div class="line" style="position:relative;">योगिनो यतचित्तस्य युञ्जतो योगमात्मनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0288"><div class="verse-block verse-justify"><div class="line">यत्रोपरमते चित्तं निरुद्धं योगसेवया।</div><div class="line" style="position:relative;">यत्र चैवाऽऽत्मनाऽऽत्मानं पश्यन्नात्मनि तुष्यति॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0289"><div class="verse-block verse-justify"><div class="line">सुखमात्यन्तिकं यत् तद् बुद्धिग्राह्यमतीन्द्रियम्।</div><div class="line" style="position:relative;">वेत्ति यत्र न चैवायं स्थितश्चलति तत्त्वतः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0290"><div class="verse-block verse-justify"><div class="line">यं लब्ध्वा चापरं लाभं मन्यते नाधिकं ततः।</div><div class="line" style="position:relative;">यस्मिन् स्थितो न दुःखेन गुरुणाऽपि विचाल्यते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0291"><div class="verse-block verse-justify"><div class="line">तं विद्याद् दुःखसंयोगवियोगं योगसंज्ञितम्।</div><div class="line" style="position:relative;">स निश्चयेन योक्तव्यो योगोऽनिर्विण्णचेतसा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0292"><div class="verse-block verse-justify"><div class="line">सङ्कल्पप्रभवान् कामांस्त्यक्त्वा सर्वानशेषतः।</div><div class="line" style="position:relative;">मनसैवेन्द्रियग्रामं विनियम्य समन्ततः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0293"><div class="verse-block verse-justify"><div class="line">शनैः शनैरुपरमेद् बुद्‌ध्या धृतिगृहीतया।</div><div class="line" style="position:relative;">आत्मसंस्थं मनः कृत्वा न किञ्चिदपि चिन्तयेत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0294"><div class="verse-block verse-justify"><div class="line">यतो यतो निश्चरति मनश्चञ्चलमस्थिरम्।</div><div class="line" style="position:relative;">ततस्ततो नियम्यैतदात्मन्येव वशं नयेत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0295"><div class="verse-block verse-justify"><div class="line">प्रशान्तमनसं ह्येनं योगिनं सुखमुत्तमम्।</div><div class="line" style="position:relative;">उपैति शान्तरजसं ब्रह्मभूतमकल्मषम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0296"><div class="verse-block verse-justify"><div class="line">युञ्जन्नेवं सदाऽऽत्मानं योगी विगतकल्मषः।</div><div class="line" style="position:relative;">सुखेन ब्रह्मसंस्पर्शमत्यन्तं सुखमश्नुते॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0297"><div class="verse-block verse-justify"><div class="line">सर्वभूतस्थमात्मानं सर्वभूतानि चाऽऽत्मनि।</div><div class="line" style="position:relative;">ईक्षते योगयुक्तात्मा सर्वत्र समदर्शनः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0298"><div class="verse-block verse-justify"><div class="line">यो मां पश्यति सर्वत्र सर्वं च मयि पश्यति।</div><div class="line" style="position:relative;">तस्याहं न प्रणश्यामि स च मे न प्रणश्यति॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0299"><div class="verse-block verse-justify"><div class="line">सर्वभूतस्थितं यो मां भजत्येकत्वमास्थितः।</div><div class="line" style="position:relative;">सर्वथा वर्तमानोऽपि स योगी मयि वर्तते॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="gita-b0300"><div class="verse-block verse-justify"><div class="line">आत्मौपम्येन सर्वत्र समं पश्यति योऽर्जुन।</div><div class="line" style="position:relative;">सुखं वा यदि वा दुःखं स योगी परमो मतः॥३२॥</div></div></div>
