<div class="verse-block-wrapper" id="shravana-mahatmyam-b0162"><div class="verse-block verse-justify"><div class="line">सङ्कल्पयेन्मम प्रीत्यै धारणापारणाव्रतम्।</div><div class="line" style="position:relative;">एकस्मिन्धारणं कुर्यात्पारणं च तथापरे॥२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0163"><div class="verse-block verse-justify"><div class="line">उपवासो धारणे स्यात्पारणे भोजनं भवेत्।</div><div class="line" style="position:relative;">समाप्ते मासि चैवात्र कुर्यादुद्यापनं व्रती॥३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0164"><div class="verse-block verse-justify"><div class="line">समाप्ते श्रावणे मासि पुण्याहं कारयेत्पुरा।</div><div class="line" style="position:relative;">आचार्यं वरयेत्पश्चाद् ब्राह्मणांश्चैव मानद॥४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0165"><div class="verse-block verse-justify"><div class="line">पार्वतीशङ्करस्यापि प्रतिमां स्वर्णनिर्मिताम्।</div><div class="line" style="position:relative;">पूर्णकुम्भे तु संस्थाप्य पूजयेन्निशि भक्तितः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0166"><div class="verse-block verse-justify"><div class="line">रात्रौ जागरणं कुर्यात्पुराणश्रवणादिभिः।</div><div class="line" style="position:relative;">प्रातरग्निं समाधाय होमं कुर्याद्यथाविधि॥६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0167"><div class="verse-block verse-justify"><div class="line">त्र्यम्बकेणेति मन्त्रेण जुहुयाच्च तिलौदनम्।</div><div class="line" style="position:relative;">तथैव शिवगायत्र्या जुहुयाच्च घृतौदनम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0168"><div class="verse-block verse-justify"><div class="line">षडक्षरेण मन्त्रेण पायसं जुहुयात्ततः।</div><div class="line" style="position:relative;">पूर्णाहुतिं ततो हुत्वा होमशेषं समापयेत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0169"><div class="verse-block verse-justify"><div class="line">ब्राह्मणान्भोजयेत्पश्चादाचार्यं चैव पूजयेत्।</div><div class="line" style="position:relative;">एवं कृत्वा महाभाग ब्रह्महत्यादिपातकैः॥९॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0171"><div class="verse-block verse-justify"><div class="line">सङ्कल्पयेत्तु प्रतिपद्दिने प्रातर्वतं मुने।</div><div class="line" style="position:relative;">नारी वा पुरुषो वापि संयतात्मा जितेन्द्रियः॥११॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0173"><div class="verse-block verse-justify"><div class="line">ब्राह्मणान्पूजयेच्चैव वस्त्रालङ्करणादिभिः।</div><div class="line" style="position:relative;">भोजयेच्च यथाशक्त्या प्रणिपत्य विसर्जयेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0174"><div class="verse-block verse-justify"><div class="line">एवं मासोपवासस्तु मम प्रीतिकरो भवेत्।</div><div class="line" style="position:relative;">रुद्रवर्तिविधानं च सम्मितं लक्षसङ्ख्यया॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0175"><div class="verse-block verse-justify"><div class="line">शृणुष्वावहितो भूत्वा सर्वसिद्धिकरं नृणाम्।</div><div class="line" style="position:relative;">कार्पासतन्तुभिः कार्या एकादशभिरादरात्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0176"><div class="verse-block verse-justify"><div class="line">वर्तयस्ता रुद्रवर्तिसंज्ञाः प्रीतिकरा मम।</div><div class="line" style="position:relative;">श्रावणस्याद्यदिवसे सङ्कल्प्य विधिपूर्वकम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0177"><div class="verse-block verse-justify"><div class="line">देवदेवं महादेवं लक्षवर्तिभिरादरात्।</div><div class="line" style="position:relative;">नीराजयामि गौरीशं श्रावणे मासि भक्तितः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0178"><div class="verse-block verse-justify"><div class="line">पूजयित्वा प्रतिदिनं वर्तीनां तु सहस्रशः।</div><div class="line" style="position:relative;">नीराजयेदन्त्यदिने सहस्राण्येकसप्ततिः॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b0180"><div class="verse-block verse-justify"><div class="line">एकस्मिन्वा दिने रुद्रवर्तिलक्षं प्रदीपयेत्।</div><div class="line" style="position:relative;">सुघृतेनापि बहुना स्निग्धास्ता मम वल्लभाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b0181"><div class="verse-block verse-justify"><div class="line" style="position:relative;">पूजयित्वा तु विश्वेशं शृणुयाच्च कथां ततः॥२१॥</div></div></div>
