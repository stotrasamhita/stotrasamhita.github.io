<div class="verse-block-wrapper" id="shravana-mahatmyam-b1012"><div class="verse-block verse-justify"><div class="line">रक्ताक्षतैः पीतगन्धैर्द्रव्यैः सौगन्धिकैः शुभैः।</div><div class="line" style="position:relative;">पुष्टिकाजनकैर्द्रव्यै रेतोवृद्धिकरैः परैः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1013"><div class="verse-block verse-justify"><div class="line">नैवेद्यमर्पयेच्चैव ताम्बूलं मुखरोचकम्।</div><div class="line" style="position:relative;">ताम्बूले योजयेद् द्रव्यं चिक्कणं क्रमुकं शुभम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1014"><div class="verse-block verse-justify"><div class="line">खादिरं चूर्णकं जातित्वचं जातिफलं तथा।</div><div class="line" style="position:relative;">लवङ्गैलानारिकेलबीजस्य शकलं लघु॥५॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b1016"><div class="verse-block verse-justify"><div class="line">श्वेतवर्णानि पक्वानि जीर्णानि च दृढानि च।</div><div class="line" style="position:relative;">रसयुक्तानि देयानि प्रीतये शम्बरद्विषः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1017"><div class="verse-block verse-justify"><div class="line">माक्षीकमलसारेण निर्मिताभिश्च वर्तिभिः।</div><div class="line" style="position:relative;">नीराजयेच्चित्तभवं पुष्पाञ्जलिमथार्पयेत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1018"><div class="verse-block verse-justify"><div class="line">प्रार्थयेन्नामभिस्तस्य तानि ते कथयाम्यहम्।</div><div class="line" style="position:relative;">सर्वोपमानसौन्दर्यः प्रद्युम्नाख्यो हरेः सुतः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1019"><div class="verse-block verse-justify"><div class="line">मीनकेतनकन्दर्पकानङ्गा मन्मथस्तथा।</div><div class="line" style="position:relative;">मारः कामात्मसम्भूतो झषकेतुर्मनोभवः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1020"><div class="verse-block verse-justify"><div class="line">रतिपीनघनोत्तुङ्गस्तनयोः पत्रवल्लिका।</div><div class="line" style="position:relative;">यस्य वक्षसि कस्तूर्याः शोभते परिरम्भणात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1021"><div class="verse-block verse-justify"><div class="line">पुष्पधन्वञ्छम्बरारे कुसुमेषो रतेः पते।</div><div class="line" style="position:relative;">मकरध्वज पञ्चेषो मदन स्मर सुन्दर॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1022"><div class="verse-block verse-justify"><div class="line">देवानां कार्यसिद्धयर्थं शिवक्षिप्तहुताशन।</div><div class="line" style="position:relative;">परोपकारसीमानं ध्वनयंस्तेन कर्मणा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1023"><div class="verse-block verse-justify"><div class="line">निमित्तमात्रं विजये वसन्तस्य सहायता।</div><div class="line" style="position:relative;">त्वन्मनोरञ्जने शक्रस्तिष्ठत्येव दिवानिशि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1024"><div class="verse-block verse-justify"><div class="line">स्वपदभ्रंशने यस्मात्तपस्विभ्यो बिभेति सः।</div><div class="line" style="position:relative;">त्वदन्यः शम्भुना कोऽन्यो विरुध्येद् दृढमानसः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1025"><div class="verse-block verse-justify"><div class="line">परब्रह्मानन्दसमानन्ददस्त्वदृतेऽत्र कः।</div><div class="line" style="position:relative;">महामोहस्य सैन्येषु त्वादृशः कोऽस्ति वीर्यवान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1026"><div class="verse-block verse-justify"><div class="line">अनिरुद्धपतिः कृष्णात्मजो कृष्णात्मजो यश्च सुरप्रभुः।</div><div class="line" style="position:relative;">मलयाचलसम्भूतचन्दनागरुवासितः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1027"><div class="verse-block verse-justify"><div class="line">दक्षिणादिङ्मातरिश्वा सहायस्ते जगज्जये।</div><div class="line" style="position:relative;">शरत्सुधांशुसन्मित्र जगत्सर्जनकारण॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1028"><div class="verse-block verse-justify"><div class="line">नाथ त्वदस्त्रं परमममोघमतिदूरगम्।</div><div class="line" style="position:relative;">मर्मच्छिदामकरुणं रहितं प्रतिकारतः॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b1030"><div class="verse-block verse-justify"><div class="line">प्रवृत्तिर्मुख्यालङ्कारः सहायेन जगज्जये।</div><div class="line" style="position:relative;">सर्वे श्रेष्ठास्त्वया देवा उपहास्याः कृता विभो॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1031"><div class="verse-block verse-justify"><div class="line">ब्रह्मा कन्यालम्पटोऽभूद् वृन्दासक्तो हरिः स्मृतः।</div><div class="line" style="position:relative;">परदारकलङ्केन अस्पृष्टव्यः शिवो यतः॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b1033"><div class="verse-block verse-justify"><div class="line">द्विजराजो गुरोर्भार्यां बलादेवापहारवान्।</div><div class="line" style="position:relative;">विश्वामित्रस्तपोभ्रष्टः केनाकारि च भूयसा॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="shravana-mahatmyam-b1035"><div class="verse-block verse-justify"><div class="line">तस्मात्प्रसीद भगवन्कृतया पूजयानया।</div><div class="line" style="position:relative;">पूजितः श्रावणे शुक्लत्रयोदश्यां मनोभवः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="shravana-mahatmyam-b1036"><div class="verse-block verse-justify"><div class="line">प्रवृत्तिलम्पटस्यातिवीर्यं पुष्टिं ददात्यलम्।</div><div class="line" style="position:relative;">निवृत्तिमार्गनियतः स्वविकारं हरत्यपि॥२७॥</div></div></div>
