<div class="verse-block-wrapper" id="kartika-mahatmyam-b0868"><div class="verse-block verse-justify"><div class="line">कार्तिकस्य च माहात्म्यं महत्पुण्यफलप्रदम्।</div><div class="line" style="position:relative;">कदा धात्री समुत्पन्ना कथं सा ख्यातिमागता॥१॥</div></div></div>

<p class="uvacha">सूत उवाच</p>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0871"><div class="verse-block verse-justify"><div class="line">कथयामि द्विजश्रेष्ठ यथा चेयं हि पुण्यदा।</div><div class="line" style="position:relative;">ऊर्जशुक्लचतुर्दश्यां धात्रीपूजां समाचरेत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0872"><div class="verse-block verse-justify"><div class="line">आमर्दकीमहावृक्षः सर्वपापप्रणाशनः।</div><div class="line" style="position:relative;">वैकुण्ठाख्यचतुर्दश्यां धात्रीछायां गतो नरः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0873"><div class="verse-block verse-justify"><div class="line">पूजयेत् तत्र देवेशं राधया सहितं हरिम्।</div><div class="line" style="position:relative;">प्रदक्षिणां ततः कुर्याच्छतमष्टोत्तरं तथा॥५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0874"><div class="verse-block verse-justify"><div class="line">सुवर्णरजतैर्वाऽपि फलैरामलकैस्तथा।</div><div class="line" style="position:relative;">शतमष्टोत्तरं कुर्यादेकैकेन प्रदक्षिणाम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0875"><div class="verse-block verse-justify"><div class="line">साष्टाङ्गं प्रणतो भूत्वा प्रार्थयेत्परमेश्वरम्।</div><div class="line" style="position:relative;">धात्रीछायां समाश्रित्य शृणुयाच्च कथामिमाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0876"><div class="verse-block verse-justify"><div class="line">ब्राह्मणान्भोजयेत्पश्चाद् यथाशक्त्या च दक्षिणाम्।</div><div class="line" style="position:relative;">ब्राह्मणेषु च तुष्टेषु तुष्टो मोक्षप्रदो हरिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0877"><div class="verse-block verse-justify"><div class="line">अत्र ते कथयिष्यामि कथां पुण्यफलप्रदाम्।</div><div class="line" style="position:relative;">आमर्दकीफलं वक्तुं ब्रह्मा चापि न पार्यते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0878"><div class="verse-block verse-justify"><div class="line">एकार्णवे पुरा जाते नष्टे स्थावरजङ्गमे।</div><div class="line" style="position:relative;">नष्टे देवासुरगणे प्रणष्टोरगराक्षसे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0879"><div class="verse-block verse-justify"><div class="line">तत्र देवाधिदेवेशः परमात्मा सनातनः।</div><div class="line" style="position:relative;">जजाप ब्रह्म परममात्मनः परमाव्ययम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0880"><div class="verse-block verse-justify"><div class="line">ततोऽस्य ब्रह्म जपतो निरगाच्छ्वसितं पुरः।</div><div class="line" style="position:relative;">तद्दर्शनानुरागेण नेत्राभ्यामगमज्जलम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0881"><div class="verse-block verse-justify"><div class="line">प्रेमाश्रुभरनिर्भिन्नो भूमौ बिन्दुः पपात सः।</div><div class="line" style="position:relative;">तस्माद्बिन्दो समुत्पन्नः स्वयं धात्री नगो महान्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0882"><div class="verse-block verse-justify"><div class="line">शाखाप्रशाखाबहुलः फलभारेण पीडितः।</div><div class="line" style="position:relative;">सर्वेषामेव वृक्षाणामादिरोहः प्रकीर्तितः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0883"><div class="verse-block verse-justify"><div class="line">ब्रह्मा तमसृजत्पूर्वं तत्पश्चाच्चासृजत्प्रजाः।</div><div class="line" style="position:relative;">देवदानवगन्धर्वयक्षराक्षसपन्नगान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0884"><div class="verse-block verse-justify"><div class="line">असृजद्भगवान्देवो मानुषांश्च तथाऽमलान्।</div><div class="line" style="position:relative;">आजग्मुस्तत्र देवास्ते यत्र धात्री हरिप्रिया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0885"><div class="verse-block verse-justify"><div class="line">तां दृष्ट्वा ते महाभागाः परमं विस्मयं गताः।</div><div class="line" style="position:relative;">न जानीम इमं वृक्षं चिन्तयन्तो मुहुर्मुहुः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0886"><div class="verse-block verse-justify"><div class="line">एवं चिन्तयतां तेषां वागुवाचाशरीरिणी।</div><div class="line" style="position:relative;">आमर्दकी नगो ह्येष प्रवरो वैष्णवो यतः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0887"><div class="verse-block verse-justify"><div class="line">अस्य वै स्मरणादेव लभेद्गोदानजं फलम्।</div><div class="line" style="position:relative;">दर्शनाद्द्विगुणं पुण्यं त्रिगुणं भक्षणात्तथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0888"><div class="verse-block verse-justify"><div class="line">तस्मात् सर्वप्रयत्नेन सेव्या आमर्दकी सदा।</div><div class="line" style="position:relative;">सर्वपापहरा प्रोक्ता वैष्णवी पापनाशिनी॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0889"><div class="verse-block verse-justify"><div class="line">तस्या मूले स्थितो विष्णुस्तदूर्ध्वं च पितामहः।</div><div class="line" style="position:relative;">स्कन्धे च भगवान् रुद्रः संस्थितः परमेश्वरः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0890"><div class="verse-block verse-justify"><div class="line">शाखासु सवितारश्च प्रशाखासु च देवताः।</div><div class="line" style="position:relative;">पर्णेषु देवताः सन्ति पुष्पेषु मरुतस्तथा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0891"><div class="verse-block verse-justify"><div class="line">प्रजानां पतयः सर्वे फलेष्वेवं व्यवस्थिताः।</div><div class="line" style="position:relative;">सर्वदेवमयी ह्येषा धात्री वै कथिता मया॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0892"><div class="verse-block verse-justify"><div class="line">अतः सा पूजनीया च सर्वकामार्थसिद्धये।</div><div class="line">एकदा नारदो योगी ब्रह्मणः पुरतः स्थितः।</div><div class="line" style="position:relative;">नमस्कृत्वा जगन्नाथं पप्रच्छातीव विस्मितः॥२४॥</div></div></div>
<p class="uvacha">नारद उवाच</p>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0894"><div class="verse-block verse-justify"><div class="line">यथा प्रियं सुतुलसीकाननं सर्वदा हरेः।</div><div class="line" style="position:relative;">तथा धात्रीवनं मासे कार्तिके श्रीहरिप्रियम्॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0897"><div class="verse-block verse-justify"><div class="line">तीर्थानि मुनयो देवा यज्ञाः सर्वेऽपि कार्तिके।</div><div class="line" style="position:relative;">नित्यं धात्रीं समाश्रित्य तिष्ठन्त्यर्के तुलास्थिते॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0898"><div class="verse-block verse-justify"><div class="line">यत्किञ्चित्कुरुते पुण्यं धात्रीछायासु मानवः।</div><div class="line" style="position:relative;">तत्कोटिगुणितं भूयान्नात्र कार्या विचारणा॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0899"><div class="verse-block verse-justify"><div class="line" style="position:relative;">अत्रैवोदाहरन्तीममितिहासं पुरातनम्॥२९॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0901"><div class="verse-block verse-justify"><div class="line">भिक्षया चोदराग्निं स शमयामास नारद।</div><div class="line" style="position:relative;">कदाचिद्वणिजो वैश्यो ययाचे क्षुत्प्रपीडितः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0902"><div class="verse-block verse-justify"><div class="line">भिक्षाप्तचणकान्गृह्य धात्रीछायामगात्किल।</div><div class="line" style="position:relative;">तत्र तान्भक्षयामास कार्तिके मासि नारद॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0903"><div class="verse-block verse-justify"><div class="line">केचिदुर्वरितास्तेषु चणकास्तत्र नारद।</div><div class="line" style="position:relative;">वैश्येन तेन दत्ता हि क्षुत्क्षामाय द्विजातये॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0904"><div class="verse-block verse-justify"><div class="line">तेन पुण्यप्रभावेन राजाऽऽसीद्धनिकः क्षितौ।</div><div class="line" style="position:relative;">तस्माद्दानं प्रकर्तव्यं कार्तिके मासि सर्वदा॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0905"><div class="verse-block verse-justify"><div class="line">धात्रीवने मुनिश्रेष्ठ सर्वकामार्थसिद्धये।</div><div class="line">धात्रीछायां समाश्रित्य कार्तिके च हरेः कथाम्।</div><div class="line" style="position:relative;">यः शृणोति स पापेभ्यो मुच्यते द्विजसूनुवत्॥३५॥</div></div></div>
