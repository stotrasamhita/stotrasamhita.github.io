<div class="verse-block-wrapper" id="kartika-mahatmyam-b1935"><div class="verse-block verse-justify"><div class="line">शृणु षण्मुख वक्ष्यामि जागरस्य च लक्षणम्।</div><div class="line" style="position:relative;">तस्य विज्ञानमात्रेण दुर्लभो न जनार्दनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1936"><div class="verse-block verse-justify"><div class="line">गीतं वाद्यं च नृत्यं च पुराणपठनं तथा।</div><div class="line" style="position:relative;">धूपं दीपं च नैवेद्यं पुष्पगन्धानुलेपनम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1937"><div class="verse-block verse-justify"><div class="line">फलमर्घ्यं च श्रद्धा च दानमिन्द्रियसंयमम्।</div><div class="line" style="position:relative;">सत्यान्वितं विनिन्दं च मुदा युक्तं क्रियान्वितम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1938"><div class="verse-block verse-justify"><div class="line">साश्चर्यं चैव प्रोत्साहमालस्यादिविवर्जितम्।</div><div class="line" style="position:relative;">प्रदक्षिणादिसंयुक्तं नमस्कारपुरःसरम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1939"><div class="verse-block verse-justify"><div class="line">नीराजनसमायुक्तमनिर्विण्णेन चेतसा।</div><div class="line" style="position:relative;">यामे यामे महाभाग कुर्वन्नीराजनं हरेः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1940"><div class="verse-block verse-justify"><div class="line">एतैर्गुणैः समायुक्तं कुर्याज्जागरणं विभोः।</div><div class="line" style="position:relative;">एकाग्रमनसा यस्तु न पुनर्जायते भुवि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1941"><div class="verse-block verse-justify"><div class="line">य एवं कुरुते भक्त्या वित्तशाठ्यविवर्जितः।</div><div class="line" style="position:relative;">जागरं वासरे विष्णोर्लीयते परमात्मनि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1942"><div class="verse-block verse-justify"><div class="line">पुरुषसूक्तेन यो नित्यं कार्तिकेऽथार्चयेद्धरिम्।</div><div class="line" style="position:relative;">वर्षकोटिसहस्राणि पूजितस्तेन केशवः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1943"><div class="verse-block verse-justify"><div class="line">यथोक्तेन विधानेन पञ्चरात्रोदितेन वै।</div><div class="line" style="position:relative;">कार्तिके त्वर्चयेन्नित्यं मुक्तिभागी भवेन्नरः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1944"><div class="verse-block verse-justify"><div class="line">नमो नारायणायेति कार्तिके योऽर्चयेद्धरिम्।</div><div class="line" style="position:relative;">स मुक्तो नारकैर्दुःखैः पदं गच्छत्यनामयम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1945"><div class="verse-block verse-justify"><div class="line">हरेर्नामसहस्रं च गजराजस्य मोक्षणम्।</div><div class="line" style="position:relative;">कार्तिके पठते यस्तु पुनर्जन्म न विन्दति॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1946"><div class="verse-block verse-justify"><div class="line">युगकोटिसहस्राणि मन्वन्तरशतानि च।</div><div class="line" style="position:relative;">द्वादश्यां कार्तिके मासि जागरी वसते दिवि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1947"><div class="verse-block verse-justify"><div class="line">कुले तस्य च ये जाताः शतशोऽथ सहस्रशः।</div><div class="line" style="position:relative;">प्राप्नुवन्ति पदं विष्णोस्तस्मात् कुर्वीत जागरम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1948"><div class="verse-block verse-justify"><div class="line">कार्तिके पश्चिमे यामे स्तवं गानं करोति यः।</div><div class="line" style="position:relative;">श्वेतद्वीपे तु वसते पितृभिः सह सुव्रत॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1949"><div class="verse-block verse-justify"><div class="line">नैवेद्यदानं हरये कार्तिके दिनसङ्क्षये।</div><div class="line" style="position:relative;">युगानि वसते स्वर्गे तावन्ति मुनिसत्तमाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1950"><div class="verse-block verse-justify"><div class="line">अक्षयं मुनिशार्दूल मालतीकमलार्चनम्।</div><div class="line" style="position:relative;">अर्चयेद् देवदेवेशं स याति परमं पदम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1951"><div class="verse-block verse-justify"><div class="line">कार्तिके शुक्लपक्षे तु कृत्वा ह्येकादशीं नरः।</div><div class="line" style="position:relative;">प्रातर्दत्त्वा शुभान्कुम्भान्स याति मम मन्दिरम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1952"><div class="verse-block verse-justify"><div class="line">अत्रैव तु प्रकर्तव्यः प्रबोधस्तु हरेः खग।</div><div class="line" style="position:relative;">हतः शङ्खासुरो दैत्यो नभसः शुक्लपक्षके॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1953"><div class="verse-block verse-justify"><div class="line">एकादश्यां ततो विष्णुश्चातुर्मास्ये प्रसुप्तवान्।</div><div class="line" style="position:relative;">क्षीराम्भोधौ जागृतोऽसावेकादश्यां तु कार्तिके॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1954"><div class="verse-block verse-justify"><div class="line">अतः प्रबोधनं कार्यमेकादश्यां तु वैष्णवैः।</div><div class="line">उत्तिष्ठोत्तिष्ठ गोविन्द उत्तिष्ठ गरुडध्वज।</div><div class="line" style="position:relative;">उत्तिष्ठ कमलाकान्त त्रैलोक्यं मङ्गलं कुरु॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1955"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा शङ्खभेर्यादि प्रातःकाले तु वादयेत्।</div><div class="line" style="position:relative;">वीणावेणुमृदङ्गादि नृत्यगीतादि कारयेत्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1956"><div class="verse-block verse-justify"><div class="line">उत्थापयित्वा देवेशं पूजां तस्य विधाय च।</div><div class="line" style="position:relative;">सायङ्काले प्रकर्तव्यस्तुलस्युद्वाहजो विधिः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1957"><div class="verse-block verse-justify"><div class="line">सर्वदैकादशी पुण्या विशेषात्कार्तिकी स्मृता।</div><div class="line" style="position:relative;">यानि कानि च पापानि ब्रह्महत्यादिकानि च॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b1959"><div class="verse-block verse-justify"><div class="line">तस्मात् सर्वप्रयत्नेन कुर्यादेकादशीव्रतम्।</div><div class="line" style="position:relative;">न कुर्याद्यदि मोहेन उपवासं नराधमः॥३४॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b1977"><div class="verse-block verse-justify"><div class="line">धात्रीच्छायां गतो यस्तु द्वादश्यां पूजयेद्धरिम्।</div><div class="line" style="position:relative;">तत्रैव भोजनं यस्तु ब्राह्मणानां तु कारयेत्॥५०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1978"><div class="verse-block verse-justify"><div class="line">स्वयं च तत्र भुङ्क्ते यः सूपभक्ष्यादिकं तथा।</div><div class="line" style="position:relative;">न तस्य पुनरावृत्तिः कल्पकोटिशतैरपि॥५१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1979"><div class="verse-block verse-justify"><div class="line">एवं प्रातर्विधायाथ पूजां दामोदरस्य हि।</div><div class="line" style="position:relative;">रात्रौ पुनः प्रकर्तव्यं पूजाकर्म हरेर्द्विज॥५२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1980"><div class="verse-block verse-justify"><div class="line">तुलसीसन्निधौ कृत्वा पताकाध्वजशोभितम्।</div><div class="line" style="position:relative;">पुष्पमालासमाकीर्णं नानारत्नोपशोभितम्॥५३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1981"><div class="verse-block verse-justify"><div class="line">मुक्तादामभिराच्छन्नं कृत्वा मण्डपमुत्तमम्।</div><div class="line" style="position:relative;">पूजयेद्विष्णुमव्यग्रस्तद्गतैकाग्रमानसः॥५४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1982"><div class="verse-block verse-justify"><div class="line">पञ्चरात्रोक्तमार्गेण गन्धपुष्पाक्षतादिभिः।</div><div class="line" style="position:relative;">नवनीतं दधिक्षीरं तथैव च घनं घृतम्॥५५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1983"><div class="verse-block verse-justify"><div class="line">विविधैः खाद्यनैवेद्यैर्जलेन च सुगन्धिना।</div><div class="line" style="position:relative;">युक्तं निवेदयेद्विष्णोस्ताम्बूलं सलवङ्गकम्॥५६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1984"><div class="verse-block verse-justify"><div class="line">पुष्पाणि च विचित्राणि सुगन्धीनि बहूनि च।</div><div class="line" style="position:relative;">प्रोक्षयित्वा च विधिवदर्पयित्वा दलैः शुभैः॥५७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1985"><div class="verse-block verse-justify"><div class="line">तुलस्याश्चापि धात्र्याश्च फलैश्चापि प्रपूजयेत्।</div><div class="line" style="position:relative;">नीराजनं ततः कृत्वा मन्त्रपुष्पं समर्पयेत्॥५८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1986"><div class="verse-block verse-justify"><div class="line">अभिषेकं विना सर्वपूजां कृत्वा विधानतः।</div><div class="line" style="position:relative;">विष्णोः पूजां समाप्याथ ब्राह्मणानां प्रपूजनम्॥५९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1987"><div class="verse-block verse-justify"><div class="line">कुर्याद्भक्तियुतो विप्र दद्याच्चैव फलादिकम्।</div><div class="line" style="position:relative;">ताम्बूलं च ततो दत्त्वा दक्षिणां शक्तितोऽर्पयेत्॥६०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1988"><div class="verse-block verse-justify"><div class="line">ततो वृद्धान् पितॄन्मातॄन् पूजयित्वा विधानतः।</div><div class="line" style="position:relative;">ततः स्वयं स्वभार्याभिर्नैवेद्यं भक्षयेत्सुधीः॥६१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1989"><div class="verse-block verse-justify"><div class="line">इत्येवं तु विधानेन यः कुर्याद् द्वादशीव्रतम्।</div><div class="line" style="position:relative;">न तस्य लोकाः क्षीयन्ते कल्पकोटिशतैरपि॥६२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1990"><div class="verse-block verse-justify"><div class="line">पुत्रपौत्रैः परिवृतो भुक्त्वा भोगान्मनोहरान्।</div><div class="line" style="position:relative;">भोगान्ते च व्रजेन्मोक्षमतीतकुलसप्तकैः॥६३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1991"><div class="verse-block verse-justify"><div class="line">तस्मान्नारद माहात्म्यं द्वादश्याः कार्तिकस्य च।</div><div class="line" style="position:relative;">न मया शक्यते वक्तुं किमन्यैर्मनुजैरपि॥६४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1992"><div class="verse-block verse-justify"><div class="line">द्वादश्या ह्युत्तमं पुण्यं माहात्म्यं यः पठेन्नरः।</div><div class="line" style="position:relative;">शृणुयाद्वा मुनिश्रेष्ठ स याति परमां गतिम्॥६५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b1993"><div class="verse-block verse-justify"><div class="line">राजर्षिरम्बरीषोऽपि चकारैतद्व्रतं शुभम्।</div><div class="line" style="position:relative;">यथाविधि तपोनिष्ठस्तेन मोक्षमवाप्तवान्॥६६॥</div></div></div>
