<div class="verse-block-wrapper" id="kartika-mahatmyam-b0394"><div class="verse-block verse-justify"><div class="line">रताभून्न हि तस्यास्तु लज्जा लोकापवादतः।</div><div class="line" style="position:relative;">सुतबन्धुविहीना सा सदाभिक्षान्नभोजना॥९॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0396"><div class="verse-block verse-justify"><div class="line">कथायाः श्रवणं चैव न श्रुतं तु तया द्विज।</div><div class="line" style="position:relative;">एकदा ब्राह्मणः कश्चित्तीर्थयात्रापरायणः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0397"><div class="verse-block verse-justify"><div class="line">तस्या गृहं समागच्छद् विद्वान्वै कुत्सनामकः।</div><div class="line">अनाचाररतां तां तु दृष्ट्वा ब्रह्मर्षिसत्तमः।</div><div class="line" style="position:relative;">कोपेन रक्तचक्षुः संस्तामुवाचासतीं स्त्रियम्॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0401"><div class="verse-block verse-justify"><div class="line">जलबुद्बुदवद्देहो नाशमायाति निश्चितम्।</div><div class="line" style="position:relative;">अनित्यं देहमाश्रित्य नित्यं त्वं मन्यसे हृदि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0402"><div class="verse-block verse-justify"><div class="line">तस्मादन्तःस्थितं मोहं त्यज मूढे विचारतः।</div><div class="line" style="position:relative;">स्मर सर्वोत्तमं देवं कुरु श्रवणमादरात्॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0406"><div class="verse-block verse-justify"><div class="line">सर्वपापप्रशमनं सर्वोपद्रवनाशनम्।</div><div class="line" style="position:relative;">तत्रापि कार्तिके मासि दीयतां दीप उत्तमः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0407"><div class="verse-block verse-justify"><div class="line">दीपो हरेः प्रियकरः कार्तिके मासि निश्चितम्।</div><div class="line" style="position:relative;">महापातककृद् वाऽपि दीपदानात्प्रमुच्यते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0408"><div class="verse-block verse-justify"><div class="line">पुरा कश्चिद्द्विजवरो नाम्ना हरिकरो ह्यभूत्।</div><div class="line" style="position:relative;">अधर्मविषयासक्तः शश्वद्वेश्यारतो द्विजः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0409"><div class="verse-block verse-justify"><div class="line">पितृवित्तक्षयकरो वंशच्छेदे कुठारकः।</div><div class="line" style="position:relative;">कदाचित्तेन विधवे द्यूते पितृधनं महत्॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0413"><div class="verse-block verse-justify"><div class="line">महापातककृद् वाऽपि गतवानभयं हरिम्।</div><div class="line" style="position:relative;">तस्मात् त्वं कार्तिके मासि दीपदानं तथा कुरु॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0414"><div class="verse-block verse-justify"><div class="line">तथाऽन्यान्यपि दानानि कुरु भक्तिसमन्विता।</div><div class="line" style="position:relative;">इत्यादिश्याथ तां कुत्सो जगामाऽन्यगृहं द्विजः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0415"><div class="verse-block verse-justify"><div class="line">साऽपि कुत्सवचः श्रुत्वा पश्चात्तापेन संयुता।</div><div class="line" style="position:relative;">व्रतं तु कार्तिके मासि करिष्यामीति निश्चिता॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0416"><div class="verse-block verse-justify"><div class="line">पतङ्गोदयवेलायां कार्तिके स्नानमम्भसि।</div><div class="line" style="position:relative;">दीपदानं व्रतं चैव मासमेकं चकार सा॥३०॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0422"><div class="verse-block verse-justify"><div class="line">स्वस्यापि शक्तिराहित्ये परस्यापि प्रबोधनम्।</div><div class="line" style="position:relative;">यः कुर्याल्लभते सोऽपि नात्र कार्या विचारणा॥३६॥</div></div></div>

<div class="verse-block-wrapper" id="kartika-mahatmyam-b0489"><div class="verse-block verse-justify"><div class="line">कृष्णादिमासक्रमतः कार्तिकस्यादिमासतः।</div><div class="line" style="position:relative;">आकाशादीपदानं तु कुर्वन्तु ऋषिसत्तमाः॥१००॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0490"><div class="verse-block verse-justify"><div class="line">तुलायां तिलतैलेन सायं सन्ध्यासमागमे।</div><div class="line" style="position:relative;">आकाशदीपं यो दद्यान्मासमेकं निरन्तरम्॥१०१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0491"><div class="verse-block verse-justify"><div class="line">सश्रीकाय श्रीपतये श्रिया न स वियुज्यते।</div><div class="line" style="position:relative;">आकाशदीपवंशस्तु विंशद्धस्तोत्तमो भवेत्॥१०२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0492"><div class="verse-block verse-justify"><div class="line">मध्यमो नवहस्तः स्यात्कनिष्ठः पञ्चहस्तकः।</div><div class="line" style="position:relative;">यथा दूरस्थितैर्लोकैर्दृश्यते तत्तथाऽऽचरेत्॥१०३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0493"><div class="verse-block verse-justify"><div class="line">तथाऽभ्रादिकरण्डेषु दीपदानं विशिष्यते।</div><div class="line" style="position:relative;">वंशस्य नवमांशेन लम्बा कार्या पताकिका॥१०४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0494"><div class="verse-block verse-justify"><div class="line">मयूरपिच्छमुष्टिं वा कलशं चोपरि न्यसेत्।</div><div class="line" style="position:relative;">विष्णुप्रीतिकरो दीपः पित्रुद्धारस्य कारकः॥१०५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0495"><div class="verse-block verse-justify"><div class="line">एकादश्यास्तुलार्काद्वा दीपदानमतोऽपि वा।</div><div class="line" style="position:relative;">दामोदराय नभसि तुलायां लोलया सह॥१०६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0496"><div class="verse-block verse-justify"><div class="line">प्रदीपं ते प्रयच्छामि नमोऽनन्ताय वेधसे।</div><div class="line" style="position:relative;">आकाशदीपसदृशं पितुरुद्धारकं नहि॥१०७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0497"><div class="verse-block verse-justify"><div class="line">हेलिकस्य च द्वौ पुत्रौ तत्रैकस्तु पिशाचकः।</div><div class="line" style="position:relative;">व्योमदीपपुण्यदानान्मोक्षं प्राप सुदुर्लभम्॥१०८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0498"><div class="verse-block verse-justify"><div class="line">नमः पितृभ्यः प्रेतेभ्यो नमो धर्माय विष्णवे।</div><div class="line" style="position:relative;">नमो यमाय रुद्राय कान्तारपतये नमः॥१०९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0499"><div class="verse-block verse-justify"><div class="line">मन्त्रेणानेन ये मर्त्याः पितृभ्यः खे तु दीपकम्।</div><div class="line">प्रयच्छन्ति गता ये स्युर्नरके यान्ति तेऽपि वै।</div><div class="line" style="position:relative;">उत्तमां गतिमित्थं ते दीपदानं मयेरितम्॥११०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0500"><div class="verse-block verse-justify"><div class="line" style="position:relative;">लक्ष्मीसन्ततिसिद्ध्यर्थमारोग्याय प्रदीपयेत्॥१११॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0501"><div class="verse-block verse-justify"><div class="line">कार्तिके कृष्णपक्षे तु द्वादश्यादिषु पञ्चसु।</div><div class="line" style="position:relative;">तिथीषूक्तः पूर्वरात्रे नृणां नीराजनाविधिः॥११२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0502"><div class="verse-block verse-justify"><div class="line">ब्रह्मविष्णुशिवादीनां भवनेषु विशेषतः।</div><div class="line" style="position:relative;">कूटागारेषु चैत्येषु सभासु च नदीषु च॥११३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0503"><div class="verse-block verse-justify"><div class="line">प्राकारोद्यानवापीषु प्रतोलीनिष्कुटेषु च।</div><div class="line" style="position:relative;">मन्दुरासु विविक्तासु हस्तिशालासु चैव हि॥११४॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0504"><div class="verse-block verse-justify"><div class="line">प्रदोषसमये दीपान्दद्यादेवं मनोहरान्।</div><div class="line" style="position:relative;">कृतं यैः कार्तिके मासि दीपदानं विधानतः॥११५॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0505"><div class="verse-block verse-justify"><div class="line">दृश्यन्ते ये रत्नभाजस्तेऽत एव प्रकीर्तिताः।</div><div class="line" style="position:relative;">दीपदानासमर्थश्चेत्परदीपं तु रक्षयेत्॥११६॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0506"><div class="verse-block verse-justify"><div class="line">यो वेदाभ्यासिने दद्याद्दीपार्थं तैलमादरात्।</div><div class="line" style="position:relative;">को वा तस्य फलं वक्तुं भुवि तिष्ठति मानवः॥११७॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0507"><div class="verse-block verse-justify"><div class="line">दीपान्दद्याद्बहुविधान्कार्तिके विष्णुसन्निधौ।</div><div class="line" style="position:relative;">कार्तिके मासि सम्प्राप्ते गगने स्वच्छतारके॥११८॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0508"><div class="verse-block verse-justify"><div class="line">रात्रौ लक्ष्मीः समायाति द्रष्टुं भुवनकौतुकम्।</div><div class="line" style="position:relative;">यत्र यत्र च दीपान्सा पश्यत्यब्धिसमुद्भवा॥११९॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0509"><div class="verse-block verse-justify"><div class="line">तत्र तत्र रतिं कुर्यान्नान्धकारे कदाचन।</div><div class="line" style="position:relative;">तस्माद्दीपः स्थापनीयः कार्तिके मासि वै सदा॥१२०॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0510"><div class="verse-block verse-justify"><div class="line">लक्ष्मीरूपार्थिनां प्रोक्तं दीपदानं विशेषतः।</div><div class="line" style="position:relative;">देवालये नदीतीरे राजमार्गे विशेषतः॥१२१॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0511"><div class="verse-block verse-justify"><div class="line">निद्रास्थले दीपदाता तस्य श्रीः सर्वतोमुखी।</div><div class="line" style="position:relative;">दुर्बलस्याऽऽलयं वीक्ष्य दीपशून्यं तु यो ददेत्॥१२२॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0512"><div class="verse-block verse-justify"><div class="line">विप्रस्य वाऽन्यवर्णस्य विष्णुलोके महीयते।</div><div class="line" style="position:relative;">कीटकण्टकसङ्कीर्णे दुर्गमे विषमस्थले॥१२३॥</div></div></div>
<div class="verse-block-wrapper" id="kartika-mahatmyam-b0513"><div class="verse-block verse-justify"><div class="line">कुर्याद्यो दीपदानानि नरकं स न गच्छति।</div><div class="line" style="position:relative;">दद्याद्रात्रौ पञ्चनदे दीपं यो विधिपूर्वकम्॥१२४॥</div></div></div>
