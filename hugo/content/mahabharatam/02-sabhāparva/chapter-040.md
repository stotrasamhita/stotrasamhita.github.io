<div class="verse-block-wrapper" id="02-sabhāparva-b1783"><div class="verse-block verse-justify"><div class="line">नायमर्हति वार्ष्णेयस्तिष्ठत्स्विह महात्मसु।</div><div class="line" style="position:relative;">महीपतिषु कौरव्य राजवत्पार्थिवार्हणम्॥१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1784"><div class="verse-block verse-justify"><div class="line">नायं युक्तः समाचारः पाण्डवेषु महात्मसु।</div><div class="line" style="position:relative;">यत्कामाद्देवकीपुत्रं पाण्डवार्चितवानसि॥२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1785"><div class="verse-block verse-justify"><div class="line">बाला यूयं न जानीध्वं धर्मः सूक्ष्मो हि पाण्डवाः।</div><div class="line" style="position:relative;">अयं तत्राभ्यतिक्रान्तो ह्यापगेयोऽल्पदर्शनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1786"><div class="verse-block verse-justify"><div class="line">त्वादृशो धर्मयुक्तो हि कुर्वाणः प्रियकाम्यया।</div><div class="line" style="position:relative;">भवत्यभ्यधिकं भीष्मो लोकेष्ववमतः सताम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1787"><div class="verse-block verse-justify"><div class="line">कथं ह्यराजा दाशार्हो मध्ये सर्वमहीक्षिताम्।</div><div class="line" style="position:relative;">अर्हणामर्हति तथा यथा युष्माभिरर्चितः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1788"><div class="verse-block verse-justify"><div class="line">अथवा मन्यसे कृष्णं स्थविरं कुरुपुङ्गवः।</div><div class="line" style="position:relative;">वसुदेवे स्थिते वृद्धे कथमर्हति तत्सुतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1789"><div class="verse-block verse-justify"><div class="line">अथवा वासुदेवोऽपि प्रियकामोऽनुवृत्तवान्।</div><div class="line" style="position:relative;">द्रुपदे तिष्ठति कथं माधवोऽर्हति पूजनम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1790"><div class="verse-block verse-justify"><div class="line">आचार्यं मन्यसे कृष्णमथवा कुरुनन्दन।</div><div class="line" style="position:relative;">द्रोणे तिष्ठति वार्ष्णेयं कस्मादर्चितवानसि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1791"><div class="verse-block verse-justify"><div class="line">ऋत्विजं मन्यसे कृष्णमथवा कुरुनन्दन।</div><div class="line" style="position:relative;">द्वौपायने स्थिते वृद्धे कथं कृष्णोऽर्चितस्त्वया॥९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1792"><div class="verse-block verse-justify"><div class="line">भीष्मे शान्तनवे राजन्स्थिते पुरुषसत्तमे।</div><div class="line" style="position:relative;">स्वच्छन्दमृत्युके राजन्कथं कृष्णोऽर्चितस्त्वया॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1793"><div class="verse-block verse-justify"><div class="line">अश्वत्थाम्नि स्थिते वीरे सर्वशास्त्रविशारदे।</div><div class="line" style="position:relative;">कथं कृष्णस्त्वया राजन्नर्चितः कुरुनन्दन॥११॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1794"><div class="verse-block verse-justify"><div class="line">दुर्योधने च राजेन्द्रे स्थिते पुरुषसत्तमे।</div><div class="line" style="position:relative;">कृपे च भारताचार्ये कथं कृष्णस्त्वयाऽर्चितः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1795"><div class="verse-block verse-justify"><div class="line">द्रुमं कम्पुरुषाचार्यमतिक्रम्य तथाऽर्चितः।</div><div class="line" style="position:relative;">भीष्मके चैव दुर्धर्षे पाण्डुवत्कृतलक्षणे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1796"><div class="verse-block verse-justify"><div class="line">नृपे च रुक्मिणि श्रेष्ठे एकलव्ये तथैव च।</div><div class="line" style="position:relative;">शल्ये मद्राधिपे चैव कथं कृष्णस्त्वयार्चितः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1797"><div class="verse-block verse-justify"><div class="line">अयं च सर्वराज्ञां वै बलश्लाघी महाबलः।</div><div class="line" style="position:relative;">जामदग्न्यस्य दयितः शिष्यो विप्रस्य भारत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1798"><div class="verse-block verse-justify"><div class="line">येनात्मबलमाश्रित्य राजानो युधि निर्जिताः।</div><div class="line" style="position:relative;">तं च कर्णमतिक्रम्य कथं कृष्णस्त्वयार्चितः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1799"><div class="verse-block verse-justify"><div class="line">नैवर्त्विङ्नैव चाचार्यो न राजा मधुसूदनः।</div><div class="line" style="position:relative;">अर्चितश्च कुरुश्रेष्ठ किमन्यत्प्रियकाम्यया॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1800"><div class="verse-block verse-justify"><div class="line">अथवाऽभ्यर्चनीयोऽयं युष्माकं मधुसूदनः।</div><div class="line" style="position:relative;">किं राजभिरिहानीतैरवमानाय भारत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1801"><div class="verse-block verse-justify"><div class="line">वयं तु न भयादस्य कौन्तेयस्य महात्मनः।</div><div class="line" style="position:relative;">प्रयच्छामः करान्सर्वे न लोभान्न च सान्त्वनात्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1802"><div class="verse-block verse-justify"><div class="line">अस्य धर्मप्रवृत्तस्य पार्थिवत्वं चिकीर्षतः।</div><div class="line" style="position:relative;">करानस्मै प्रयच्छामः सोऽयमस्मान्न मन्यते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1803"><div class="verse-block verse-justify"><div class="line">किमन्यदवमनानाद्धे यदेनं राजसंसदि।</div><div class="line" style="position:relative;">अप्राप्तलक्षणं कृष्णमर्घ्येणार्चितवानसि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1804"><div class="verse-block verse-justify"><div class="line">अकस्माद्धर्मपुत्रस्य धर्मात्मेति यशो गतम्।</div><div class="line" style="position:relative;">को हि धर्मच्युते पूजामेवं युक्तां नियोजयेत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1805"><div class="verse-block verse-justify"><div class="line">योयं वृष्णिकुले जातो राजानं हतवान्पुरा।</div><div class="line" style="position:relative;">जरासन्धं महात्मानमन्यायेन दुरात्मवान्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1806"><div class="verse-block verse-justify"><div class="line">अद्य धर्मात्मता चैव व्यपकृष्टा युधिष्ठिरात्।</div><div class="line" style="position:relative;">दर्शितं कृपणत्वं च कृष्णेऽर्घ्यस्य निवेदनात्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1807"><div class="verse-block verse-justify"><div class="line">यदि भीताश्च कौन्तेयाः कृपणाश्च तपस्विनः।</div><div class="line" style="position:relative;">ननु त्वयाऽपि बोद्धव्यं यां पूजां माधवार्हसि॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1808"><div class="verse-block verse-justify"><div class="line">अथवा कृपणैरेतामुपनीतां जनार्दन।</div><div class="line" style="position:relative;">पूजामनर्हः कस्मात्त्वमभ्यनुज्ञातवानसि॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1809"><div class="verse-block verse-justify"><div class="line">अयुक्तामात्मनः पूजां त्वं पुनर्बहुमन्यसे।</div><div class="line" style="position:relative;">हविषः प्राप्य निष्यन्दं प्राशिता श्वेव निर्जने॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1810"><div class="verse-block verse-justify"><div class="line">न त्वं पार्थिवेन्द्राणामपमानः प्रयुज्यते।</div><div class="line" style="position:relative;">त्वामेव कुरवो व्यक्तं प्रलम्भन्ते जनार्दन॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1811"><div class="verse-block verse-justify"><div class="line">क्लीबे दारक्रिया यादृगन्धे वा रूपदर्शनम्।</div><div class="line" style="position:relative;">अराज्ञो राजवत्पूजा तथा ते मधुसूदन॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b1812"><div class="verse-block verse-justify"><div class="line">दृष्टो युधिष्ठिरो राजा दृष्टो भीष्मश्च यादृशः।</div><div class="line" style="position:relative;">वासुदेवोऽप्ययं दृष्टः सर्वमेतद्यथातथम्॥३०॥</div></div></div>
