<div class="verse-block-wrapper" id="02-sabhāparva-b2036"><div class="verse-block verse-justify"><div class="line">विष्णोः पुनर्महाभागः प्रादुर्भावो महात्मनः।</div><div class="line" style="position:relative;">दत्तात्रेय इति ख्यात ऋषिरासीन्महायशाः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2037"><div class="verse-block verse-justify"><div class="line">तेन नष्टेषु वेदेषु क्रियासु च मखेषु च।</div><div class="line" style="position:relative;">चातुर्वर्ण्ये च सङ्कीर्णे धर्मे शिथिलतां गते॥२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2038"><div class="verse-block verse-justify"><div class="line">अभवर्धति चाधर्मे सत्ये नष्टे स्थितेऽनृते।</div><div class="line" style="position:relative;">प्रजासु क्षीयमाणासु धर्मे चामूलतां पते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2039"><div class="verse-block verse-justify"><div class="line">सयज्ञाः सक्रिया वेदाः प्रत्यानीता हि तेन वै।</div><div class="line" style="position:relative;">चातुर्वर्ण्यमसङ्कीर्णं कृतं तेन महात्मना॥४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2040"><div class="verse-block verse-justify"><div class="line">स एव वै यदा प्रादाद्धैहयाधिपतेर्वरम्।</div><div class="line" style="position:relative;">तं हैहयानामधिपस्त्वर्जुनोऽभिप्रसादयन्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2041"><div class="verse-block verse-justify"><div class="line">वनं पर्यचरन्सम्यक्छुश्रूषुरनुसूयकः।</div><div class="line" style="position:relative;">निर्ममो निरहङ्कारो दीर्घकालमतोषयत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2042"><div class="verse-block verse-justify"><div class="line">आराध्य दत्तात्रेयं हि अगृङ्णात्स वरानिमान्।</div><div class="line" style="position:relative;">आप्तादाप्ततरान्विप्राद्विद्वान्विद्वन्निषेवितात्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2043"><div class="verse-block verse-justify"><div class="line">ऋतेऽमरत्वं विप्रेण दत्तात्रेयेण धीमता।</div><div class="line" style="position:relative;">वरैश्चतुर्भिः प्रवृत इमान्वव्रे वरान्नृपः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2044"><div class="verse-block verse-justify"><div class="line">श्रीमान्मनस्वी बलवान्सत्यवागनसूयकः।</div><div class="line" style="position:relative;">सहस्रबाहुर्भूयासमेषु मे प्रथमो वरः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2045"><div class="verse-block verse-justify"><div class="line">जरायुजाण्डजं सर्वं सर्वं चैव चराचरम्।</div><div class="line" style="position:relative;">शास्तुमिच्छामि धर्मेण द्वितीयस्त्वेष मे वरः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2046"><div class="verse-block verse-justify"><div class="line">पितृन्देवानृषीन्विप्रान्यजेयं विपुलैर्मखैः।</div><div class="line" style="position:relative;">अमित्रांश्च शितैर्बाणैस्तृतीयो व्रर एष मे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2047"><div class="verse-block verse-justify"><div class="line">यस्य नासीन्न भविता न चास्ति सदृशः पुमान्।</div><div class="line" style="position:relative;">इह वा दिवि वा लोके स मे हन्ता भवेदिति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2048"><div class="verse-block verse-justify"><div class="line">सोऽर्जुनः कृतवीर्यस्य वरः पुत्रोऽभवद्युधि।</div><div class="line" style="position:relative;">स सहस्रं सहस्राणां माहिष्मत्यामवर्धत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2049"><div class="verse-block verse-justify"><div class="line">स भूमिमखिलां जित्वा द्वीपांश्चापि समुद्रिणः।</div><div class="line" style="position:relative;">नभसीवाज्वलत्सूर्यः पुण्यैः कर्मभिर्जुनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2050"><div class="verse-block verse-justify"><div class="line">इन्द्रद्वीपं कशेरुं च कामद्वीपं गभस्तितम्।</div><div class="line" style="position:relative;">गन्धर्ववरुणद्वीपं सौहृष्टममितप्रभः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2051"><div class="verse-block verse-justify"><div class="line">पूर्वैरजितपूर्वांश्च द्वीपनजयदर्जुनः।</div><div class="line" style="position:relative;">इदं तु कार्तवीर्यस्य बभूवासदृशं जनैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2052"><div class="verse-block verse-justify"><div class="line">न पूर्वे नापरे तस्य गमिष्यन्ति गतिं नृपाः।</div><div class="line" style="position:relative;">यदर्णवे प्रयातस्य वस्त्रं न परिषिच्यते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2053"><div class="verse-block verse-justify"><div class="line">सौवर्णं सर्वमप्यासीद्विमानवरमुत्तमम्।</div><div class="line" style="position:relative;">चतुर्धा व्यभजद्राष्ट्रं तद्विभज्यान्वपालयत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2054"><div class="verse-block verse-justify"><div class="line">एकांशेनाहरत्सेनामेकांशेनावसद्गृहान्।</div><div class="line" style="position:relative;">यस्तु तस्य तृतीयांशो राज्ञोऽभूज्जनसङ्ग्रहे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2055"><div class="verse-block verse-justify"><div class="line">आप्तः परमकल्याणस्तेन यज्ञानकल्पयत्।</div><div class="line" style="position:relative;">ये दस्यवो ग्रामचरा अरम्ये च वसन्ति ये॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2056"><div class="verse-block verse-justify"><div class="line">चतुर्थेन तु सोंऽशेन तान्सर्वान्प्रत्यषेधयत्।</div><div class="line" style="position:relative;">द्वाराणि नापिधीयन्ते पुरेषु नगरेषु च॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2057"><div class="verse-block verse-justify"><div class="line">स एव राष्ट्रपालोऽभूत्स्रीपालोऽभवदर्जुनः।</div><div class="line" style="position:relative;">स ----- सः गोपालो विशाम्पते॥२२॥</div></div></div>
