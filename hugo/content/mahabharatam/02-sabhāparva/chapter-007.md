<div class="verse-block-wrapper" id="02-sabhāparva-b0319"><div class="verse-block verse-justify"><div class="line">शक्रस्य तु सभा दिव्या भास्वरा कर्मनिर्मिता।</div><div class="line" style="position:relative;">स्वयं शक्रेण कौरव्य निर्जितार्कसमप्रभा॥१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0320"><div class="verse-block verse-justify"><div class="line">विस्तीर्णा योजनशतं शतमध्यर्धमायता।</div><div class="line">वैहायसी कामगमा पञ्चयोजनमुच्छ्रिता।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0321"><div class="verse-block verse-justify"><div class="line">जराशोकक्लमापेता निरातङ्का शिवा शुभा।</div><div class="line" style="position:relative;">वेश्मासनवती रम्या दिव्यपादपशोभिता॥३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0322"><div class="verse-block verse-justify"><div class="line">तस्यां देवेश्वरः पार्थ सभायां परमासने।</div><div class="line">आस्ते शच्या महेन्द्राण्या श्रिया लक्ष्म्या च भारत।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0323"><div class="verse-block verse-justify"><div class="line">बिभ्रद्वपुरनिर्देश्यं किरीटी लोहिताङ्गदः।</div><div class="line" style="position:relative;">विरजोम्बरश्चित्रमाल्यो ह्रीकीर्तिद्युतिभिः सह॥५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0324"><div class="verse-block verse-justify"><div class="line">तस्यामुपासते नित्यं महात्मानं शतक्रतुम्।</div><div class="line" style="position:relative;">मरुतः सर्वशो राजन्सर्वे च गृहमेधिनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0325"><div class="verse-block verse-justify"><div class="line">सिद्धा देवर्षयश्चैव साध्या देवगणास्तथा।</div><div class="line" style="position:relative;">मरुत्त्वन्तश्च सहिता भास्वन्तो हेममालिनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0326"><div class="verse-block verse-justify"><div class="line">एते सानुचराः सर्वे दिव्यरूपाः स्वलङ्कृताः।</div><div class="line" style="position:relative;">उपासते महात्मान देवराजमरिन्दमम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0327"><div class="verse-block verse-justify"><div class="line">तथा देवर्षयः सर्वे पार्ते शक्रमुपासते।</div><div class="line" style="position:relative;">अमला धूतपाप्मानो दीप्यमाना इवाग्नयः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0328"><div class="verse-block verse-justify"><div class="line">तेजस्विनः सोमसुतो विशोका विगतज्वराः।</div><div class="line" style="position:relative;">पराशरः पर्वतश्च तथा सावर्णिगालवौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0329"><div class="verse-block verse-justify"><div class="line">`एकतश्च द्वितश्चैव त्रितश्चैव महामतिः'।</div><div class="line">शङ्खश्च लिखितश्चैव तथा गौरशिरा मुनिः।</div><div class="line" style="position:relative;">दुर्वासाः क्रोधनः श्येनस्तथा दीर्घतमा मुनिः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0330"><div class="verse-block verse-justify"><div class="line">पवित्रपाणिः सावर्णिर्याज्ञवल्क्योऽथ भालुकिः।</div><div class="line" style="position:relative;">उद्दालकः श्वेतकेतुस्ताण्डो भाण्डायनिस्तथा॥१२॥</div></div></div>
<p class="prose">हविष्मांश्च गरिष्ठश्च हरिश्चन्द्रश्च पार्थिवः ॥हृद्यश्चोदरशाण्डिल्यः पाराशर्यः कृषीवलः</p>
<div class="verse-block-wrapper" id="02-sabhāparva-b0332"><div class="verse-block verse-justify"><div class="line">वातस्कन्धो विशाखश्च विधाता काल एव च।</div><div class="line" style="position:relative;">करालदन्तस्त्वष्टा च विश्वकर्मा च तुम्बुरुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0333"><div class="verse-block verse-justify"><div class="line">अयोनिजा योनिजाश्च वायुभक्षा हुताशिनः।</div><div class="line" style="position:relative;">ईशानं सर्वलोकस्य वज्रिणं समुपासते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0334"><div class="verse-block verse-justify"><div class="line">सहदेवः सुनीथश्च वाल्मीकिश्च महातपाः।</div><div class="line" style="position:relative;">समीकः सत्यवाक्चैव प्रचेताः सत्यसङ्गरः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0335"><div class="verse-block verse-justify"><div class="line">मेधातिथिर्वामदेवः पुलस्त्यः पुलहः क्रतुः।</div><div class="line" style="position:relative;">मरुत्तश्च मरीचिश्च स्थाणुश्चात्र महातपाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0336"><div class="verse-block verse-justify"><div class="line">कक्षीवान्गौतमस्तार्क्ष्यस्तथा वैश्वानरो मुनिः।</div><div class="line" style="position:relative;">मुनिः कालकवृक्षीय आश्राव्योऽथ हिरण्मयः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0337"><div class="verse-block verse-justify"><div class="line">संवर्तो देवहव्यश्च विष्वक्सेनश्च वीर्यवान्।</div><div class="line" style="position:relative;">दिव्या आपस्तथौषध्यः श्रद्धा मेधा सरस्वती॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0338"><div class="verse-block verse-justify"><div class="line">अर्थो धर्मश्च कामश्च विद्युतश्चैव पाण्डव।</div><div class="line" style="position:relative;">जलवाहास्तथा मेघा वायवः स्तनयित्नवः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0339"><div class="verse-block verse-justify"><div class="line">प्राचीदिग्यज्ञवाहाश्च पावकाः सप्तविंशतिः।</div><div class="line" style="position:relative;">अग्नीषोमौ तथेन्द्राग्नी मित्रश्च सविताऽर्यमा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0340"><div class="verse-block verse-justify"><div class="line">भगो विश्वे साध्याश्च गुरुः शुक्रस्तथैव च।</div><div class="line" style="position:relative;">विश्वावसुश्चित्रसेनः सुमनस्तरुणस्तथा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0341"><div class="verse-block verse-justify"><div class="line">यज्ञाश्च दक्षिणाश्चैवं ग्रहास्तोभाश्च भारत।</div><div class="line" style="position:relative;">यज्ञवाहाश्च ये मन्त्राः सर्वे तत्र समासते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0342"><div class="verse-block verse-justify"><div class="line">तथैवाप्सरसो राजन् `रम्भोर्वश्यथ मेनका।</div><div class="line" style="position:relative;">घृताची पञ्चचूडा च विप्रचित्तिपुरोगमाः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0343"><div class="verse-block verse-justify"><div class="line">विद्याधराश्च राजेन्द्र' गन्धर्वाश्च मनोरमाः।</div><div class="line" style="position:relative;">नृत्यवादित्रगीतैश्च हास्यैश्च विविधैरपि॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0344"><div class="verse-block verse-justify"><div class="line">रमयन्ति स्म नृपते देवराजं शतक्रतुम्।</div><div class="line" style="position:relative;">स्तुतिभिर्मङ्गलैश्चैव वस्तुवन्तः कर्मभिस्तथा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0345"><div class="verse-block verse-justify"><div class="line">विक्रमैश्च महात्मानं बलवृत्रनिषूदनम्।</div><div class="line" style="position:relative;">ब्रह्मराजर्षयश्चैव सर्वे देवर्षयस्तथा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b0346"><div class="verse-block verse-justify"><div class="line">विमानैर्विविधैर्दिव्यैर्दीप्यमाना इवाग्नयः।</div><div class="line" style="position:relative;">स्रग्विणो भूषिताः सर्वे यान्ति चायान्ति चापरे॥२७॥</div></div></div>
<p class="prose">बृहस्पतिश्च शुक्रश्च नित्यमास्तां हि तत्र वै ॥एते चान्ये च बहवो महात्मानो यतव्रताः</p>
