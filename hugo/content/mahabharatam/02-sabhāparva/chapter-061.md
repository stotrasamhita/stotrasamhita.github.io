<div class="verse-block-wrapper" id="02-sabhāparva-b2547"><div class="verse-block verse-justify"><div class="line">शौरिणा पृथिवीपालास्त्रासिता भरतर्षभ।</div><div class="line" style="position:relative;">धनुषश्च प्रणादेन पाञ्चजन्यस्वनेन च॥२॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2548"><div class="verse-block verse-justify"><div class="line">मेघप्रख्यैरनेकैश्च दाक्षिणात्याभिसंवृतम्।</div><div class="line" style="position:relative;">रुक्मिणं त्रासयामास केशवो भरतर्षभ॥३॥</div></div></div>

<div class="verse-block-wrapper" id="02-sabhāparva-b2551"><div class="verse-block verse-justify"><div class="line">इन्द्रद्युम्नो हतः कोपाद्यवनश्च कशेरुकः।</div><div class="line" style="position:relative;">हतः सौभपतिश्चैव साल्वश्च कृतधन्वना॥६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2552"><div class="verse-block verse-justify"><div class="line">पर्वतानां सहस्रं च चक्रेण पुरुषोत्तमः।</div><div class="line" style="position:relative;">विभज्य पुण्डरीकाक्षो द्युमत्सेनमपोथयत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2553"><div class="verse-block verse-justify"><div class="line">महेन्द्रशिखरे चैव निमेषान्तरचारिणौ।</div><div class="line" style="position:relative;">जग्राह भरतश्रेष्ठ वानरावभितश्चरौ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2554"><div class="verse-block verse-justify"><div class="line">इरावत्यां महाभोजो वह्निसूर्यसमो बले।</div><div class="line" style="position:relative;">गोपतिस्तालकेतुश्च निहतौ शार्ङ्गधन्वना॥९॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2555"><div class="verse-block verse-justify"><div class="line">अक्षप्रपत्तने राजन्नवहेलनतत्परौ।</div><div class="line" style="position:relative;">उभौ तावपि कृष्णेन स्वराष्ट्रे विनिपातितौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2556"><div class="verse-block verse-justify"><div class="line">दग्धा वाराणसी तात केशवेन महात्मना।</div><div class="line" style="position:relative;">पाण्ड्यं पौण्ड्रं च मात्स्यं च कलिङ्गं च जनार्दनः॥११॥</div></div></div>

<div class="verse-block-wrapper" id="02-sabhāparva-b2559"><div class="verse-block verse-justify"><div class="line">अथ गाण्डीवधन्वानं क्रीडार्थं मधुसूदनः।</div><div class="line" style="position:relative;">जिगाय भरतश्रेष्ठ कुन्त्याश्च प्रमुखे विभुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2560"><div class="verse-block verse-justify"><div class="line">द्रौणिं कृपं च कर्णं च भीमसेनं सुयोधनम्।</div><div class="line" style="position:relative;">युद्धाय सहितान्त्राजञ्जिगाय भरतर्षभ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2561"><div class="verse-block verse-justify"><div class="line">बभ्रोश्च प्रियमन्विच्छन्नेष चक्रगदाधरः।</div><div class="line" style="position:relative;">वेणुदारिवृतां भार्यां प्रममाथ युधिष्ठिर॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2562"><div class="verse-block verse-justify"><div class="line">पर्याप्तां पृथिवीं सर्वां साश्वां सरथकुञ्जराम्।</div><div class="line" style="position:relative;">वेणुदारिवशे युक्तां जिगाय मधुसूदनः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2563"><div class="verse-block verse-justify"><div class="line">अवाप्य तपसा वीर्यं बलमोजश्च भारत।</div><div class="line" style="position:relative;">त्रासिताः सगणाः सर्वे बाणेन विबुधाधिपाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2564"><div class="verse-block verse-justify"><div class="line">वज्राशनिगदाबामैस्ताडयद्भिरनेकशः।</div><div class="line" style="position:relative;">तस्य नासीद्रणे मृत्युर्देवैरपि सवासवैः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="02-sabhāparva-b2565"><div class="verse-block verse-justify"><div class="line">सोऽभिभूतश्च कृष्णेन न हतश्च मगात्मना।</div><div class="line" style="position:relative;">छित्वा बाहुसहस्रं तु गोविन्देन महात्मना॥१९॥</div></div></div>
