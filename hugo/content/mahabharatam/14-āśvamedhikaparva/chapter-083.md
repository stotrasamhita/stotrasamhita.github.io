<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2535"><div class="verse-block verse-justify"><div class="line">स तु वाजी समुद्रान्तां पर्येत्य वसुधामिमाम्।</div><div class="line" style="position:relative;">निवृत्तोऽभिमुखो राजन्येन वारणसाह्वयम्॥१॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2536"><div class="verse-block verse-justify"><div class="line">अनुगच्छंश्च तुरगं निवृत्तोऽथ किरीटभृत्।</div><div class="line" style="position:relative;">यदृच्छया समापेदे पुरं राजगृहं तदा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2537"><div class="verse-block verse-justify"><div class="line">तमभ्याशगतं दृष्ट्वा सहदेवात्मजः प्रभो।</div><div class="line" style="position:relative;">क्षत्रधर्मे स्थितो वीरः समरायाजुहाव ह॥३॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2538"><div class="verse-block verse-justify"><div class="line">ततः पुरात्स निष्क्रम्य रथी धन्वी शरी तली।</div><div class="line" style="position:relative;">मेघसन्धिः पदातिं तं धनंजयमुपाद्रवत्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2539"><div class="verse-block verse-justify"><div class="line">आसाद्य च महातेजा मेघसन्धिर्धनंजयम्।</div><div class="line" style="position:relative;">बालभावान्महाराज प्रोवाचेदं न कौशलात्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2540"><div class="verse-block verse-justify"><div class="line">किमयं चार्यते वाजी स्इत्रीमध्य इव भारत।</div><div class="line" style="position:relative;">हयमेनं हरिष्यामि प्रयतस्व विमोक्षणे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2541"><div class="verse-block verse-justify"><div class="line">अदत्तानुनयो युद्धे यदि त्वं पितृभिर्मम।</div><div class="line" style="position:relative;">करिष्यामि तवातिथ्यं प्रहर प्रहरामि च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2542"><div class="verse-block verse-justify"><div class="line">इत्युक्तः प्रत्युवाचैनं प्रहसन्निव पाण्डवः।</div><div class="line" style="position:relative;">विघ्नकर्ता मया वार्य इति मे व्रतमाहितम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2543"><div class="verse-block verse-justify"><div class="line">भ्रात्रा ज्येष्ठेन नृपते तवापि विदितं ध्रुवम्।</div><div class="line" style="position:relative;">प्रहरस्व यथाशक्ति न मन्युर्विद्यते मम॥९॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2544"><div class="verse-block verse-justify"><div class="line">इत्युक्तः प्राहरत्पूर्वं पाण्डवं मगधेस्वरः।</div><div class="line" style="position:relative;">किरञ्शरसहस्राणि वर्षाणीव सहस्रदृक्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2545"><div class="verse-block verse-justify"><div class="line">ततो गाण्डीवभृच्छूरो गाण्डीवप्रहितैः शरैः।</div><div class="line" style="position:relative;">चकार मोघांस्तान्बाणान्सयत्नान्भरतर्षभ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2546"><div class="verse-block verse-justify"><div class="line">स मोघं तस्य बाणौघं कृत्वा वानरकेतनः।</div><div class="line" style="position:relative;">शरात्मुमोच ज्वलितान्दीप्तास्यानिव पन्नगान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2547"><div class="verse-block verse-justify"><div class="line">ध्वजे पताकादण्डेषु रथे यन्त्रे हयेषु च।</div><div class="line" style="position:relative;">अन्येषु च रथाङ्गेषु न शरीरे न सारथौ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2548"><div class="verse-block verse-justify"><div class="line">संरक्ष्यमाणः पार्थेन शरीरे सव्यसाचिना।</div><div class="line" style="position:relative;">मन्यमानः स्ववीर्यं तन्मागधः प्राहिणोच्छरान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2549"><div class="verse-block verse-justify"><div class="line">ततो गाण्डीवधन्वा तु मागधेन भृशाहतः।</div><div class="line" style="position:relative;">बभौ वसन्तसमये पलासः पुष्पितो यथा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2550"><div class="verse-block verse-justify"><div class="line">अवध्यमानः सोऽभ्यघ्नन्मागधः पाण्डवर्षभम्।</div><div class="line" style="position:relative;">तेन तस्थौ स कौरव्य लोकवीरस्य दर्शने॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2551"><div class="verse-block verse-justify"><div class="line">सव्यसाची तु संक्रुद्धो विकृष्य बलवद्धनुः।</div><div class="line" style="position:relative;">हयांश्चकार निर्जीवान्सारथेश्च शिरोऽहरत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2552"><div class="verse-block verse-justify"><div class="line">धनुश्चास्य महच्चित्रं क्षुरेण प्रचकर्त ह।</div><div class="line" style="position:relative;">हस्तावापं पताकां च ध्वजं चास्यन्यपातयत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2553"><div class="verse-block verse-justify"><div class="line">स राजा व्यथितो व्यश्वो विधनुर्हतसारथिः।</div><div class="line" style="position:relative;">गदामादाय कौन्तेयमभिद्रद्राव वेगवान्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2554"><div class="verse-block verse-justify"><div class="line">तस्यापतत एवाशु गदां हेमपरिष्कृताम्।</div><div class="line" style="position:relative;">शरैश्चकर्त बहुधा बहुभिर्गृध्रवाजितैः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2555"><div class="verse-block verse-justify"><div class="line">सा गदा शकलीभूता विशीर्णिमणिबन्धना।</div><div class="line" style="position:relative;">व्याली विमुच्यमानेन पपात धरणीतले॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2556"><div class="verse-block verse-justify"><div class="line">विरथं विधनुष्कं च गदया परिवर्जितम्।</div><div class="line" style="position:relative;">`नैच्छत्ताडयितुं धीमानर्जुनः समराग्रणीः॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2558"><div class="verse-block verse-justify"><div class="line">पर्याप्तः क्षत्रधर्मोऽयं दर्शितः पुत्र गम्यताम्।</div><div class="line" style="position:relative;">बह्वेतत्समरे कर्म तव बालस्य पार्थिव॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2559"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरस्य संदेशो न हन्तव्या नृपा इति।</div><div class="line" style="position:relative;">तेन जीवसि राजंस्त्वमपराद्धोऽपि मे रणे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2560"><div class="verse-block verse-justify"><div class="line">इति मत्वा तदाऽऽत्मानं प्रत्यादिष्टं स्म मागधः।</div><div class="line" style="position:relative;">तथ्यमित्यभिगम्यैनं प्राञ्जलिः प्रत्यपूजयत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2561"><div class="verse-block verse-justify"><div class="line">परिजितोस्मि भद्रं ते नाहं योद्धुमिहोत्सहे।</div><div class="line" style="position:relative;">यद्यत्कृत्यं मया तेऽद्य तद्ब्रूहि कृतमेव तु॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2562"><div class="verse-block verse-justify"><div class="line">तमर्जुनः समाश्वास्य पुनरेवेदमब्रवीत्।</div><div class="line" style="position:relative;">आगन्तव्यं परां चैत्रीमश्वमेधे नृपस्य नः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2563"><div class="verse-block verse-justify"><div class="line">इत्युक्तः स तथेत्युक्त्वा पूजयमास तं हयम्।</div><div class="line" style="position:relative;">फल्गुनं च युधिश्रेष्ठं विदिवत्सहदेवजः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="14-āśvamedhikaparva-b2564"><div class="verse-block verse-justify"><div class="line">ततो यथेष्टमगमत्पुनरेव स कौरवः।</div><div class="line" style="position:relative;">ततः समुद्रतीरेण वङ्गान्पुण्ड्रान्सकेरलान्॥२९॥</div></div></div>
