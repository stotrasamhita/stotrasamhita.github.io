<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1411"><div class="verse-block verse-justify"><div class="line">इदं ज्ञानमुपाश्रित्य मम साधर्म्यमागताः।</div><div class="line" style="position:relative;">सर्गेऽपि नोपजायन्ते प्रलये न व्यथन्ति च॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1412"><div class="verse-block verse-justify"><div class="line">मम योनिर्महद्ब्रह्म तस्मिन्गर्भं दधाम्यहम्।</div><div class="line" style="position:relative;">संभवः सर्वभूतानां ततो भवति भारत॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1413"><div class="verse-block verse-justify"><div class="line">सर्वयोनिषु कौन्तेय मूर्तयः संभवन्तिः याः।</div><div class="line" style="position:relative;">तासां ब्रह्म महद्योनिरहं बीजप्रदः पिता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1414"><div class="verse-block verse-justify"><div class="line">सत्वं रजस्तम इति गुणाः प्रकृतिसंभवाः।</div><div class="line" style="position:relative;">निबध्नन्ति महाबाहो देहे देहिनमव्ययम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1415"><div class="verse-block verse-justify"><div class="line">तत्र सत्त्वं निर्मलत्वात्प्रकाशकमनामयम्।</div><div class="line" style="position:relative;">सुखसङ्गेन बध्नाति ज्ञानसङ्गेन चानघ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1416"><div class="verse-block verse-justify"><div class="line">रजो रागात्मकं विद्धि तृष्णासङ्गसमुद्भवम्।</div><div class="line" style="position:relative;">तन्निबध्नाति कौन्तेय कर्मसङ्गेन देहिनम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1417"><div class="verse-block verse-justify"><div class="line">तमस्त्वज्ञानजं विद्धि मोहनं सर्वदेहिनाम्।</div><div class="line" style="position:relative;">प्रमादालस्यनिद्राभिस्तन्निबध्नाति भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1418"><div class="verse-block verse-justify"><div class="line">सत्वं सुखे संजयति रजः कर्मणि भारत।</div><div class="line" style="position:relative;">ज्ञानमावृत्य तु तमः प्रमादे संजयत्युत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1419"><div class="verse-block verse-justify"><div class="line">रजस्तमश्चाभिभूय सत्वं भवति भारत।</div><div class="line" style="position:relative;">रजः सत्वं तमश्चैव तमः सत्वं रजस्तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1420"><div class="verse-block verse-justify"><div class="line">सर्वद्वारेषु देहेऽस्मिन्प्राकाश उपजायते।</div><div class="line" style="position:relative;">ज्ञानं यदा तदा विद्याद्विवृद्धं सत्वमित्युत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1421"><div class="verse-block verse-justify"><div class="line">लोभः प्रवृत्तिरारम्भः कर्मणामशमः स्पृहा।</div><div class="line" style="position:relative;">रजस्येतानि जायन्ते विवृद्धे भरतर्षभ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1422"><div class="verse-block verse-justify"><div class="line">अप्रकाशोऽप्रवृत्तिश्च प्रमादो मोह एव च।</div><div class="line" style="position:relative;">तमस्येतानि जायन्ते विवृद्धे कुरुनन्दन॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1423"><div class="verse-block verse-justify"><div class="line">यदा सत्वे प्रवृद्धे तु प्रलयं याति देहभृत्।</div><div class="line" style="position:relative;">तदोत्तमविदां लोकानमलान्प्रतिपद्यते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1424"><div class="verse-block verse-justify"><div class="line">रजसि प्रलयं गत्वा कर्मसङ्गिषु जायते।</div><div class="line" style="position:relative;">तथा प्रलीनस्तमसि मूढयोनिषु जायते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1425"><div class="verse-block verse-justify"><div class="line">कर्मणः सुकृतस्याहुः सात्विकं निर्मलं फलम्।</div><div class="line" style="position:relative;">रजसस्तु फलं दुःखमज्ञानं तमसः फलम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1426"><div class="verse-block verse-justify"><div class="line">सत्वात्संजायते ज्ञानं रजसो लोभ एव च।</div><div class="line" style="position:relative;">प्रमादमोहौ तमसो भवतोऽज्ञानमेव च॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b1429"><div class="verse-block verse-justify"><div class="line">गुणानेतानतीत्य त्रीन्देही देहसमुद्भवान्।</div><div class="line">जन्ममृत्युजरादुःखैर्विमुक्तोऽमृतमश्रुते ॥अर्जुन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
