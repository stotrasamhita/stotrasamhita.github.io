<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4856"><div class="verse-block verse-justify"><div class="line">प्रियो हि पाण्डवो नित्यं भारद्वाजस्य धीमतः।</div><div class="line" style="position:relative;">आचार्यश्च रणे नित्यं प्रियः पार्थस्य सञ्जय॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4857"><div class="verse-block verse-justify"><div class="line">तावुभौ रथिनो सङ्ख्ये हृष्टौ सिंहाविवोत्कटौ।</div><div class="line">कथं समीयतुर्यत्तौ भारद्वाजधनञ्जयौ ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4858"><div class="verse-block verse-justify"><div class="line">न द्रोणः समरे पार्थं जानीते प्रियमात्मनः।</div><div class="line" style="position:relative;">क्षत्रधर्मं पुरस्कृत्य पार्थो वा गुरुमाहवे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4859"><div class="verse-block verse-justify"><div class="line">न क्षत्रिया रणे राजन्वर्जयन्ति परस्परम्।</div><div class="line" style="position:relative;">निर्मर्यादं हि युध्यन्ते पितृभिर्भ्रातृभिः सह॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4860"><div class="verse-block verse-justify"><div class="line">रणे भारत पार्थेन द्रोणो विद्धस्त्रिभिः शरैः।</div><div class="line" style="position:relative;">नाचिन्तयच्च तान्बाणान्पार्थचापच्युतान्युधि॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4861"><div class="verse-block verse-justify"><div class="line">शरवृष्ट्या पुनः पार्थश्छादयामास तं रणे।</div><div class="line" style="position:relative;">स प्रजज्वाल रोषेण गहनेऽग्निरिवोर्जितः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4862"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनं रणे द्रोणः शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">छादयामास राजेन्द्र नचिरादेव भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4863"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राजा सुशर्माणमचोदयत्।</div><div class="line" style="position:relative;">द्रोणस्य समरे राजन्पार्ष्णिग्रहणकारणात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4864"><div class="verse-block verse-justify"><div class="line">त्रिगर्तराडपि क्रुद्धो भृशमायम्य कार्मुकम्।</div><div class="line" style="position:relative;">छादयामास समरे पार्थं बाणैरयोमुखैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4865"><div class="verse-block verse-justify"><div class="line">ताभ्यां मुक्ताः शरा राजन्नन्तरिक्षे विरेजिरे।</div><div class="line" style="position:relative;">हंसा इव महाराज शरत्काले नभस्तले॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4866"><div class="verse-block verse-justify"><div class="line">ते शराः प्राप्य कौन्तेयं समन्ताद्विविशुः प्रभो।</div><div class="line" style="position:relative;">फलभारनतं यद्वत्स्वादुवृक्षं विहंगमाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4867"><div class="verse-block verse-justify"><div class="line">अर्जुनस्तु रणे नादं विनद्य रथिनां वरः।</div><div class="line" style="position:relative;">त्रिगर्तराजं समरे सपुत्रं विव्यधे शरैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4868"><div class="verse-block verse-justify"><div class="line">ते वध्यमानाः पार्थेन कालेनेव युगक्षये।</div><div class="line">पार्थमेवाभ्यवर्तन्त मरणे कृतनिश्चयाः।</div><div class="line" style="position:relative;">मुमुचुः शरवृष्टिं च पाण्डवस्य रथं प्रति॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4869"><div class="verse-block verse-justify"><div class="line">शरवृष्टिं ततस्तां तु शरवर्षैः समन्ततः।</div><div class="line" style="position:relative;">प्रतिजग्राह राजेन्द्र तोयवृष्टिमिवाचलः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4870"><div class="verse-block verse-justify"><div class="line">तत्राद्भुतमपश्याम बीभत्सोर्हस्तलाघवम्।</div><div class="line" style="position:relative;">विमुक्तां बहुभिर्योधैः शस्त्रवृष्टिं दुरासदाम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4871"><div class="verse-block verse-justify"><div class="line">यदेको वारयामास मारुतोऽभ्रगणानिव।</div><div class="line" style="position:relative;">कर्मणा तेन पार्थस्य तुतुषुर्देवदानवाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4872"><div class="verse-block verse-justify"><div class="line">अथ क्रुद्धो रणे पार्थस्त्रिगर्तान्प्रति भारत।</div><div class="line" style="position:relative;">मुमोचास्त्रं महाराज वायव्यं पृतनामुखे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4873"><div class="verse-block verse-justify"><div class="line">प्रादुरासीत्ततो वायुः क्षोभयाणो नभस्तलम्।</div><div class="line" style="position:relative;">पातयन्वै तरुगणान्विनिघ्नंश्चैव सैनिकान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4874"><div class="verse-block verse-justify"><div class="line">ततो द्रोणोऽभिवीक्ष्यैव वायव्यास्त्रं सुदारुणम्।</div><div class="line" style="position:relative;">शैलमन्यन्महाराज घोरमस्त्रं मुमोच ह॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4875"><div class="verse-block verse-justify"><div class="line">द्रोणेन युधि निर्मुक्ते तस्मिन्नस्त्रे नराधिप।</div><div class="line" style="position:relative;">प्रशशाम ततो वयुः प्रसन्नाश्च दिशो दश॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4876"><div class="verse-block verse-justify"><div class="line">ततः पाण्डुसुतो वीरस्त्रिगर्तस्य रथव्रजान्।</div><div class="line" style="position:relative;">निरुत्साहान्रणे चक्रे विमुखान्विपराक्रमान्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4877"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनश्चैव कृपश्च रथिनां वरः।</div><div class="line" style="position:relative;">अश्वत्थामा तथा शल्यः काम्भोजश्च सुदक्षिणः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4878"><div class="verse-block verse-justify"><div class="line">विन्दानुविन्दावावन्त्यौ बाह्लिकः सह बाह्लिकैः।</div><div class="line" style="position:relative;">महता रथवंशेन पार्थस्यावारयन्दिशः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4879"><div class="verse-block verse-justify"><div class="line">तथैव भगदत्तश्च श्रतायुश्च महाबलः।</div><div class="line" style="position:relative;">गजानीकेन भीमस्य ताववारयतां दिशः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4880"><div class="verse-block verse-justify"><div class="line">भूरिश्रवाः शलश्चैव सौबलश्च विशांपते।</div><div class="line" style="position:relative;">शरौघैर्विमलैस्तीक्ष्णैर्माद्रीपुत्राववारयन्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4881"><div class="verse-block verse-justify"><div class="line">भीष्मस्तु संहतः सङ्ख्ये धार्तराष्ट्रैः ससैनिकैः।</div><div class="line" style="position:relative;">युधिष्ठिरं समासाद्य सर्वतः पर्यवारयत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4882"><div class="verse-block verse-justify"><div class="line">आपतन्तं गजानीकं दृष्ट्वा पार्थो वृकोदरः।</div><div class="line" style="position:relative;">लोलिहन्सृक्किणी वीरो मृगराडिव कानने॥२९॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4884"><div class="verse-block verse-justify"><div class="line">तमुद्वीक्ष्य गदाहस्तं ततस्ते गजसादिनः।</div><div class="line" style="position:relative;">परिवव्रू रणे यत्ता भीमसेनं समन्ततः॥३१॥</div></div></div>
