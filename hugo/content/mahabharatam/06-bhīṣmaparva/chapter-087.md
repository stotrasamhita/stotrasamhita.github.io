<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4041"><div class="verse-block verse-justify"><div class="line">ततः शब्दो महानासीत्सेनयोरुभयोर्नृप।</div><div class="line" style="position:relative;">निर्गच्छमानयोः सङ्ख्ये यथा सागरयोरिव॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4042"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राजा चित्रसेनो विविंशतिः।</div><div class="line" style="position:relative;">भीष्मश्च रथिनां श्रेष्ठो भारद्वाजश्च वै द्विजः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4043"><div class="verse-block verse-justify"><div class="line">एकीभूताः सुसंयत्ताः कौरवाणां महाचमूम्।</div><div class="line" style="position:relative;">व्यूहाय विदधू राजन्पाण्डवान्प्रतिदंशितान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4044"><div class="verse-block verse-justify"><div class="line">कूर्मव्यूहं ततः कृत्वा पिता तव विशांपते।</div><div class="line" style="position:relative;">सागरप्रतिमं घोरं वाहनोर्मितरङ्गिणम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4045"><div class="verse-block verse-justify"><div class="line">अग्रतः सर्वसैन्यानां भीष्मः शान्तनवो ययौ।</div><div class="line" style="position:relative;">मालवैर्दाक्षिणात्यैश्च आवन्त्यैश्च समन्वितः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4046"><div class="verse-block verse-justify"><div class="line">ततोऽनन्तरमेवासीद्भारद्वाजः प्रतापवान।</div><div class="line" style="position:relative;">पुलिन्दैः पारदैश्चैव तथा क्षुद्रकमालवैः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4047"><div class="verse-block verse-justify"><div class="line">द्रोणादनन्तरं यत्तो भगदत्तः प्रतापवान्।</div><div class="line" style="position:relative;">मगधैश्च कलिङ्गैश्च पिशाचैश्च विशांपते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4048"><div class="verse-block verse-justify"><div class="line">प्राग्ज्योतिषादनु नृपः कौसल्योऽथ बृहद्बलः।</div><div class="line" style="position:relative;">मेकलैः कुरुविन्दैश्च त्रैपुरैश्च समन्विताः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4049"><div class="verse-block verse-justify"><div class="line">बृहद्बलादनु नृपस्त्रिगर्तः प्रस्थलाधिपः।</div><div class="line" style="position:relative;">काम्भोजैर्बहुभिः सार्धं यवनैश्च सहस्रशः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4050"><div class="verse-block verse-justify"><div class="line">द्रौणिस्तु रभसः शूरस्त्रैगर्तादनु भारत।</div><div class="line" style="position:relative;">प्रययौ सिंहनादेन नादयानो धरातलम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4051"><div class="verse-block verse-justify"><div class="line">तथा सर्वेण सैन्येन राजा दुर्योधनस्तदा।</div><div class="line" style="position:relative;">द्रौणेरनन्तरं प्रायात्सोदर्यैः परिवारितः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4052"><div class="verse-block verse-justify"><div class="line">दुर्योधनादनु ततः कृपः शारद्वतो ययौ।</div><div class="line" style="position:relative;">एवमेष महाव्यूहः प्रययौ सागरोपमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4053"><div class="verse-block verse-justify"><div class="line">रेजुस्तत्र पताकाश्च श्वेतच्छत्राणि भारत।</div><div class="line" style="position:relative;">अङ्गदान्यत्र चित्राणि महार्हाणि धनूंषि च॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4054"><div class="verse-block verse-justify"><div class="line">तं तु दृष्ट्वा महाव्यूहं तावकानां महारथः।</div><div class="line" style="position:relative;">युधिष्ठिरोऽब्रवीत्तूर्णं पार्षतं पृतनापतिम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4055"><div class="verse-block verse-justify"><div class="line">पश्य व्यूहं महेष्वास निर्मितं सागरोपमम्।</div><div class="line" style="position:relative;">प्रतिव्यूहं रणे शूर कुरु क्षिप्रं महारथ॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4056"><div class="verse-block verse-justify"><div class="line">ततः स पार्षतः क्रूरो व्यूहं चक्रे सुदारुणम्।</div><div class="line" style="position:relative;">श्रृङ्गाटकं महाराज परव्यूहविनाशनम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4057"><div class="verse-block verse-justify"><div class="line">श्रृङ्गाभ्यां भीमसेनश्च सात्यकिश्च महारथः।</div><div class="line" style="position:relative;">रथैरनेकसाहस्रैस्तथा हयपदातिभिः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4058"><div class="verse-block verse-justify"><div class="line">नाभावभून्नरश्रेष्ठः श्वेताश्वः कृष्णसारथिः।</div><div class="line" style="position:relative;">मध्ये युधिष्ठिरो राजा माद्रीपुत्रौ च पाण्डवौ॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4059"><div class="verse-block verse-justify"><div class="line">अथेतरे महेष्वासाः सहसैन्या नराधिपाः।</div><div class="line" style="position:relative;">व्यूहं तं पूरयामासुर्व्यूहशास्त्रविशारदाः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4060"><div class="verse-block verse-justify"><div class="line">एवमेतं महाव्यूहं व्यूह्य भारत पाण्डवाः।</div><div class="line" style="position:relative;">अतिष्ठन्समरे शूरा योद्धुकामा जयैषिणः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4061"><div class="verse-block verse-justify"><div class="line">भेरीशब्दैश्च विमलैर्विमिश्रैः शङ्खनिःस्वनैः।</div><div class="line" style="position:relative;">क्ष्वेडितास्फोटितोत्क्रुष्टैर्नादिताः सर्वतो दिशः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4062"><div class="verse-block verse-justify"><div class="line">ततः शूराः समासाद्य समरे ते परस्परम।</div><div class="line">नेत्रैरनिमिषै राजन्नवैक्षन्त परस्परम् ॥ 6-87-24a`मनोभिस्तेमनुष्येन्द्र युद्धं योधाः प्रचक्रिरे।</div><div class="line" style="position:relative;">'पुनराहूय तेऽन्योन्यं शरीरैरपि चक्रिरे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4063"><div class="verse-block verse-justify"><div class="line">ततः प्रववृते युद्धं घोररूपं भयावहम्।</div><div class="line" style="position:relative;">तावकानां परेषां च निघ्नतामितरेतरम्॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4066"><div class="verse-block verse-justify"><div class="line">गदाश्च विमलैः पट्टैः पिनद्धाः स्वर्णभूषिताः।</div><div class="line" style="position:relative;">पतन्त्यस्तत्र दृस्यन्ते गिरिशृङ्गोपमाः शुभाः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4067"><div class="verse-block verse-justify"><div class="line">निस्त्रिंशाश्च व्यदृश्यन्त विमलाम्बरसन्निभाः।</div><div class="line" style="position:relative;">आर्षभाणि च चर्माणि शतचन्द्राणि भारतत॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4068"><div class="verse-block verse-justify"><div class="line">अशोभन्त रणे राजन्पात्यमानानि सर्वशः।</div><div class="line" style="position:relative;">तेऽन्योन्यं समरे सेने युध्यमाने नराधिप॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4069"><div class="verse-block verse-justify"><div class="line">अशोभेतां यथा देवदैत्यसेने समुद्यते।</div><div class="line" style="position:relative;">अभ्यद्रवन्त समरे तेऽन्योन्यं वै समन्ततः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4070"><div class="verse-block verse-justify"><div class="line">रथास्तु रथिभिस्तूर्णं प्रेषिताः परमाहवे।</div><div class="line" style="position:relative;">युगैर्युगानि संश्लिष्य युयुधुः पार्थिवर्षभाः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4071"><div class="verse-block verse-justify"><div class="line">दन्तिनां युध्यमानानां संघर्षात्पावकोऽभवत्।</div><div class="line" style="position:relative;">दन्तेषु भरतश्रेष्ठ सधूमः सर्वतो दिशम्॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4073"><div class="verse-block verse-justify"><div class="line">पादाताश्चाप्यदृश्यन्त निघ्नन्तोऽथ परस्परम्।</div><div class="line" style="position:relative;">चित्ररूपधराः शूरा नखरप्रासयोधिनः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4074"><div class="verse-block verse-justify"><div class="line">अन्योन्यं ते समासाद्य कुरुपाण्डवसैनिकाः।</div><div class="line" style="position:relative;">अस्त्रैर्नानाविधैर्घोरै रणे निन्युर्यमक्षयम्॥३६॥</div></div></div>
