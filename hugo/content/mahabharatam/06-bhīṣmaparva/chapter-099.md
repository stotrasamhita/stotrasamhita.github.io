<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4705"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4706"><div class="verse-block verse-justify"><div class="line">ततः शान्तनवो भीष्मो निर्ययौ सह सेनया।</div><div class="line" style="position:relative;">व्यूहं चाव्यूहत महत्सर्वतोभद्रमात्मनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4707"><div class="verse-block verse-justify"><div class="line">कृपश्च कृतवर्मा च शैब्यश्चैव महारथः।</div><div class="line" style="position:relative;">शकुनिः सैन्धवश्चैव काम्भोजश्च सुदक्षिणः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4708"><div class="verse-block verse-justify"><div class="line">भीष्मेण सहिताः सर्वे पुत्रैश्च तव भारत।</div><div class="line" style="position:relative;">अग्रतः सर्वसैन्यानां व्यूहस्य प्रमुखे स्थिताः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4709"><div class="verse-block verse-justify"><div class="line">द्रोणो भूरिश्रवाः शल्यो भगदत्तश्च मारिष।</div><div class="line" style="position:relative;">दक्षिणं पक्षमाश्रित्य स्थिता व्यूहस्य दंशिताः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4710"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा सोमदत्तश्चावन्त्यौ च महारथौ।</div><div class="line" style="position:relative;">महत्या सेनया युक्ता वामं पक्षमपालयन्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4711"><div class="verse-block verse-justify"><div class="line">दुर्योधनो महाराज त्रिगर्तैः सर्वतो वृतः।</div><div class="line" style="position:relative;">व्यूहमध्ये स्थितो राजन्पाण्डवान्प्रति भारत॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4712"><div class="verse-block verse-justify"><div class="line">अलम्बुसो रथश्रेष्ठः श्रुतायुश्च महारथः।</div><div class="line" style="position:relative;">पृष्ठतः सर्वसैन्यानां स्थितौ व्यूहस्य दंशितौ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4713"><div class="verse-block verse-justify"><div class="line">एवं च तं तदा व्यूहं कृत्वा भारत तावकाः।</div><div class="line" style="position:relative;">सन्नद्धाः समदृश्यन्त प्रतपन्त इवाग्नयः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4714"><div class="verse-block verse-justify"><div class="line">ततो युधिष्ठिरो राजा भीमसेनश्च पाण्डवः।</div><div class="line" style="position:relative;">नकुलः सहदेवश्च माद्रीपुत्रावुभावपि॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4715"><div class="verse-block verse-justify"><div class="line">अग्रतः सर्वसैन्यानां स्थिता व्यूहस्य दंशिताः।</div><div class="line" style="position:relative;">धृष्टद्युम्नो विराटश्च सात्यकिश्च महारथः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4716"><div class="verse-block verse-justify"><div class="line">स्थिताः सैन्येन महता परानीकविनाशनाः।</div><div class="line" style="position:relative;">शिखण्डी विजयश्चैव राक्षसश्च घटोत्कचः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4717"><div class="verse-block verse-justify"><div class="line">चेकितानो महाबाहुः कुन्तिभोजश्च वीर्यवान्।</div><div class="line" style="position:relative;">स्थिता रणे महाराज महत्या सेनया वृताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4718"><div class="verse-block verse-justify"><div class="line">अभिमन्युर्महेष्वासो द्रुपदश्च महाबलः।</div><div class="line" style="position:relative;">युयुधानो महेष्वासो युधामन्युश्च वीर्यवान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4719"><div class="verse-block verse-justify"><div class="line">केकया भ्रातरश्चैव स्थिता युद्धाय दंशिताः।</div><div class="line" style="position:relative;">एवं तेऽपि महाव्यूहं प्रतिव्यूह्य सुदुर्जयम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4720"><div class="verse-block verse-justify"><div class="line">पाण्डवाः समरे शूराः स्थिता युद्धाय दंशिताः।</div><div class="line" style="position:relative;">तावकास्तु रणे यत्ताः सहसेना नराधिपाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4721"><div class="verse-block verse-justify"><div class="line">अभ्युद्ययू रणे पार्थान्भीष्मं कृत्वाऽग्रतो नृप।</div><div class="line" style="position:relative;">तथैव पाण्डवा राजन्भीमसेनपरोगमाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4722"><div class="verse-block verse-justify"><div class="line">भीष्मं योद्धुमभीप्सन्तः संग्रामे विजयैषिणः।</div><div class="line" style="position:relative;">क्ष्वेलाः किलकिलाः शङ्खान्क्रकचान्गोविषाणिकाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4723"><div class="verse-block verse-justify"><div class="line">भेरीमृदङ्गपणवान्नादयन्तश्च पुष्करान्।</div><div class="line" style="position:relative;">पाण्डवा अभ्यवर्तन्त नदन्तो भैरवान्रवान्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4724"><div class="verse-block verse-justify"><div class="line">भेरीमृद्गशङ्खानां दुन्दुभीनां च निःस्वनैः।</div><div class="line" style="position:relative;">उत्कृष्टसिंहनादैश्च वल्गितैश्च पृथग्विधैः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4725"><div class="verse-block verse-justify"><div class="line">वयं प्रतिनदन्तस्तानगच्छाम त्वरान्विताः।</div><div class="line" style="position:relative;">सहसैवाभिसंक्रद्धास्तदासीत्तुमुलं महत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4726"><div class="verse-block verse-justify"><div class="line">ततोऽन्योन्यं प्रधावन्तः संप्रहारं प्रचक्रिरे।</div><div class="line" style="position:relative;">ततः शब्देन महता प्रचकम्पे वसुंधरा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4727"><div class="verse-block verse-justify"><div class="line">पक्षिणश्च महाघोरं व्याहरन्तो विबभ्रमुः।</div><div class="line" style="position:relative;">सप्रभश्चोदितः सूर्यो निष्प्रभः समपद्यत॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4728"><div class="verse-block verse-justify"><div class="line">ववुश्च वातास्तुमुलाः शंसन्तः सुमहद्भयम्।</div><div class="line" style="position:relative;">घोराश्च घोरनिर्ह्रादाः शिवास्तत्र ववाशिरे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4729"><div class="verse-block verse-justify"><div class="line">वेदयन्त्यो महाराज महद्वैशसमागतम्।</div><div class="line" style="position:relative;">दिशः प्रज्वलिता राजन्पांसुवर्षं पपात च॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4730"><div class="verse-block verse-justify"><div class="line">रुधिरेण समुन्मिश्रमस्थिवर्षं पपात च।</div><div class="line" style="position:relative;">रुदतां वाहनानां च नेत्रेभ्यः प्रापतञ्जलम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4731"><div class="verse-block verse-justify"><div class="line">सुस्रुवुश्च शकृन्मूत्रं प्रध्यायन्तो विशांपते।</div><div class="line" style="position:relative;">अन्तर्हिता महानादाः श्रूयन्ते भरतर्षभ॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4732"><div class="verse-block verse-justify"><div class="line">रक्षसां पुरुषादानां नदतां भैरवान्रवान्।</div><div class="line" style="position:relative;">संपतन्तश्च दृश्यन्ते गोमायबलवायसाः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4733"><div class="verse-block verse-justify"><div class="line">श्वानश्च विविधैर्नादैर्भषन्तस्तत्र मारिष।</div><div class="line">ज्वलिताश्च महोल्का वै समाहत्य दिवाकरम्।</div><div class="line" style="position:relative;">निपेतुः सहसा भूमौ वेदयन्त्यो महद्भयम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4734"><div class="verse-block verse-justify"><div class="line">महान्त्यनीकानि महासमुच्छ्रयेततस्तयोः पाण्डवधार्तराष्ट्रयोः।</div><div class="line" style="position:relative;">चकम्पिरे शङ्खमृदङ्गनिःस्वनैःप्रकम्पितानीव वनानि वायुना॥३०॥</div></div></div>
