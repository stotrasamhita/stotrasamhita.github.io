<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4528"><div class="verse-block verse-justify"><div class="line">कुरूणां पाण्डवानां च क्षयं घोरं महामतिःक।</div><div class="line" style="position:relative;">स ततो निवारितवान्धृतराष्ट्रं जनेश्वरम्॥४॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4531"><div class="verse-block verse-justify"><div class="line">अधनस्य मृतं श्रेयो न च ज्ञातिवधाद्वनम्।</div><div class="line" style="position:relative;">किं नु प्राप्स्यामहे कृष्ण हत्वा ज्ञातीन्समागतान्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4532"><div class="verse-block verse-justify"><div class="line">दुर्योधनापराधेन शकुनेः सौबलस्य च।</div><div class="line" style="position:relative;">क्षत्रिया निधनं यान्ति कर्णदुर्मन्त्रितेन च॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4533"><div class="verse-block verse-justify"><div class="line">इदानीं च विजानामि सुकृतं मधुसूदन।</div><div class="line" style="position:relative;">कृतं राज्ञा महाबाहो याचता त सुयोधनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4534"><div class="verse-block verse-justify"><div class="line">राज्यार्धं पञ्च वा ग्रामान्नाकार्षीत्स च दुर्मतिःक।</div><div class="line" style="position:relative;">दृष्ट्वा हि क्षत्रियाञ्शूराञ्शयानान्धरणीतले॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4535"><div class="verse-block verse-justify"><div class="line">निन्दामि भृशमात्मानं धिगस्तु क्षत्रिजीविकाम्।</div><div class="line" style="position:relative;">अशक्तमिति मामेते ज्ञास्यन्ते क्षत्रिया रणे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4536"><div class="verse-block verse-justify"><div class="line">एतदर्थं मया युद्धं रोचितं मधुसूदन।</div><div class="line" style="position:relative;">संचोदय हयाञ्शीघ्रं धार्तराष्ट्रचमूं प्रति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4537"><div class="verse-block verse-justify"><div class="line">प्रतरिष्ये महापारं भुजाभ्यां समरोदधिम्।</div><div class="line">नायं क्लीबायितुं कालो विद्यते माधव क्वचित् ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4538"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु पार्थेन केशवः परवीरहा।</div><div class="line" style="position:relative;">चोदयामास तानश्वान्पाण्डुरान्वातरंहसः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4539"><div class="verse-block verse-justify"><div class="line">अथ शब्दो महानासीत्तव सैन्यस्य भारत।</div><div class="line" style="position:relative;">मारुतोद्धूतवेगस्य सागरस्येव पर्वणि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4540"><div class="verse-block verse-justify"><div class="line">अपराह्णे महाराज संग्रामः समपद्यत।</div><div class="line" style="position:relative;">पर्जन्यसमनिर्घोषो भीष्मस्य सह पाण्डवैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4541"><div class="verse-block verse-justify"><div class="line">ततो राजंस्तव सुता भीमसेनमुपाद्रवन्।</div><div class="line" style="position:relative;">परिवार्य रणे द्रोणं वसवो वासवं यथा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4542"><div class="verse-block verse-justify"><div class="line">ततः शान्तनवो भीष्मः कृपश्च रथिनां वरः।</div><div class="line" style="position:relative;">भगदत्तः सुशर्मा च धनंजयमुपाद्रवन्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4543"><div class="verse-block verse-justify"><div class="line">हार्दिक्यो बाह्लिकश्चैव सात्यकिं समभिद्रुतौ।</div><div class="line" style="position:relative;">अम्बष्ठकस्तु नृपतिरभिमन्युमवस्थितः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4544"><div class="verse-block verse-justify"><div class="line">शेषास्त्वन्ये महाराज शेषानेव महारथान्।</div><div class="line" style="position:relative;">ततः प्रववृते युद्धं घोररूपं भयावहम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4545"><div class="verse-block verse-justify"><div class="line">`भीमसेनस्तु संप्रेक्ष्य पुत्रांस्तव विशांपते'।</div><div class="line" style="position:relative;">प्रजज्वाल रणे क्रुद्धो हविषा हव्यवाडिव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4546"><div class="verse-block verse-justify"><div class="line">पुत्रास्तु तव कौन्तेयं छादयांचक्रिरे शरैः।</div><div class="line" style="position:relative;">प्रावृषीव महाराज जलदा इव पर्वतम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4547"><div class="verse-block verse-justify"><div class="line">स च्छाद्यमानो बहुधा पुत्रैस्तव विशांपते।</div><div class="line" style="position:relative;">सृक्विणी संलिहन्वीरः शार्दूल इव दर्पितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4548"><div class="verse-block verse-justify"><div class="line">व्यूढोरस्कस्ततो भीमः पोथयामास पार्थिवम्।</div><div class="line" style="position:relative;">क्षुरप्रेण सुतीक्ष्णेन सुमुक्तेन महारणे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4549"><div class="verse-block verse-justify"><div class="line">ताडयामास संक्रुद्धः सोऽभवद्व्यथितेन्द्रियः।</div><div class="line">अपरेण तु भल्लेन पीतेन निशितेन तु।</div><div class="line" style="position:relative;">अपातयत्कुण्डलितं सिंहः क्षुद्रमृगं यथा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4550"><div class="verse-block verse-justify"><div class="line">ततः सुनिशितान्बाणान्भीमसेनः शिलाशितान्।</div><div class="line" style="position:relative;">स सप्त संदधे हन्तुं पुत्रास्ते भरतर्षभ॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4552"><div class="verse-block verse-justify"><div class="line">अनाधृष्टिं कुण्डभेदिं विराजं दीप्तलोचनम्।</div><div class="line" style="position:relative;">दीर्घबाहुं सुबाहुं च तथैव मकरध्वजम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4553"><div class="verse-block verse-justify"><div class="line">प्रपतन्तिस्म वीरास्ते विरेजुर्भरतर्षभ।</div><div class="line" style="position:relative;">वसन्ते पुष्पशबलाः किंशुकाः पतिता इव॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4554"><div class="verse-block verse-justify"><div class="line">ततः प्रदुद्रुवुः शेषास्तव पुत्रा महाहवे।</div><div class="line" style="position:relative;">तं कालमिव मन्यन्तो भीमसेनं महाबलम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4555"><div class="verse-block verse-justify"><div class="line">द्रोणस्तु समरे वीरं निर्दहनक्तं सुतांस्तव।</div><div class="line" style="position:relative;">यथाद्रिं वारिधाराभिः समन्ताद्व्यकिरच्छरैः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4556"><div class="verse-block verse-justify"><div class="line">तत्राद्भुतमपश्याम कुन्तीपुत्रस्य पौरुषम्।</div><div class="line" style="position:relative;">द्रोणेन वार्यमाणोऽपि निजघ्ने यत्सुतांस्तव॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4559"><div class="verse-block verse-justify"><div class="line">पुत्रेषु तव वीरेषु चिक्रीडार्जुनपूर्वजः।</div><div class="line" style="position:relative;">मृगेष्विव महाराज चरन्व्याघ्रो महाबलः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4560"><div class="verse-block verse-justify"><div class="line">यथा हि पशुमध्यस्थो द्रावयेत पशून्वृकः।</div><div class="line" style="position:relative;">वृकोदरस्तव सुतांस्तथा व्यद्रावयद्रणे॥३६॥</div></div></div>
