<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0660"><div class="verse-block verse-justify"><div class="line">द्विधाभूत इवादित्य उदये प्रत्यदृश्यत।</div><div class="line" style="position:relative;">ज्वलन्त्या शिखया भूयो भानुमानुदितो रविः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0661"><div class="verse-block verse-justify"><div class="line">ववाशिरे च दीप्तायां दिशि गोमायुवायसाः।</div><div class="line" style="position:relative;">लिप्समानाः शरीराणि मांसशोणितभोजनाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0662"><div class="verse-block verse-justify"><div class="line">अहन्यहनि पार्थानां वृद्धः कुरुपितामहः।</div><div class="line" style="position:relative;">भरद्वाजात्मजश्चैव प्रातरुत्थाय संयतौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0663"><div class="verse-block verse-justify"><div class="line">जयोऽस्तु पाण्डुपुत्राणामित्यूचतुरिन्दमौ।</div><div class="line" style="position:relative;">युयुधाते तवार्थाय यथा स समयः कृतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0664"><div class="verse-block verse-justify"><div class="line">सर्वधर्मविशेषज्ञः पिता देवव्रतस्तव।</div><div class="line" style="position:relative;">समानीय महीपालानिदं वचनमब्रवीत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0665"><div class="verse-block verse-justify"><div class="line">इदं वः क्षत्रिया द्वारं स्वर्गायापावृतं महत्।</div><div class="line" style="position:relative;">गच्छध्वं तेन शक्रस्य ब्रह्मणः सहलोकताम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0666"><div class="verse-block verse-justify"><div class="line">एष वः शाश्वतः पन्थाः पूर्वैः पूर्वतरैः कृतः।</div><div class="line" style="position:relative;">संभावयध्वमात्मानमव्यग्रमनसो युधि॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0667"><div class="verse-block verse-justify"><div class="line">नाभागोऽथ ययातिश्च मान्धाता नहुषो नृगः।</div><div class="line" style="position:relative;">संसिद्धाः परमं स्थानं गताः कर्मभिरीदृशैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0668"><div class="verse-block verse-justify"><div class="line">अधर्मः क्षत्रियस्यैष यद्व्याधिमरणं गृहे।</div><div class="line" style="position:relative;">यदयोनिधनं याति सोऽस्य धर्मः सनातनः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0669"><div class="verse-block verse-justify"><div class="line">एवमुक्ता महीपाला भीष्मेण भरतर्षभ।</div><div class="line" style="position:relative;">निर्ययुः स्वान्यनीकानि शोभयन्तो रथोत्तमैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0670"><div class="verse-block verse-justify"><div class="line">स तु वैकर्तनः कर्णः सामात्यः सह बन्धुभिः।</div><div class="line" style="position:relative;">न्यासितः समरे शस्त्रं भीष्मेण भरतर्षभ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0671"><div class="verse-block verse-justify"><div class="line">अपेतकर्णाः पुत्रास्ते राजनश्चैव तावकाः।</div><div class="line" style="position:relative;">निर्ययुः सिंहनादेन नादयन्तो दिशो दश॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0672"><div class="verse-block verse-justify"><div class="line">श्वेतैश्छत्रैः पताकाभिर्ध्वजवारणवाजिभिः।</div><div class="line" style="position:relative;">तान्यनीकानि शोभन्ते गजैरथ पदातिभिः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0673"><div class="verse-block verse-justify"><div class="line">भेरीपणवशब्दैश्च दुन्दुभीनां च निःस्वनैः।</div><div class="line" style="position:relative;">रथनेमिनिनादैश्च बभूवाकुलिता मही॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0674"><div class="verse-block verse-justify"><div class="line">काञ्चनाङ्गदकेयूरैः कार्मुकैश्च महारथाः।</div><div class="line" style="position:relative;">भ्राजमाना व्यराजन्त साग्नयः पर्वता इव॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0675"><div class="verse-block verse-justify"><div class="line">तालेन महता भीष्मः पञ्चतारेण केतुना।</div><div class="line" style="position:relative;">विमलादित्यसंकाशस्तस्थौ कुरुचमूपरि॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0676"><div class="verse-block verse-justify"><div class="line">ये त्वदीया महेष्वासा राजानो भरतर्षभ।</div><div class="line" style="position:relative;">अवर्तन्त यथादेशं राजञ्शान्तनवस्य ते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0677"><div class="verse-block verse-justify"><div class="line">स तु गोवासनः शैब्यः सहितः सर्वराजभिः।</div><div class="line">ययौ मातङ्गराजेन राजार्हेण पताकिना।</div><div class="line" style="position:relative;">पद्मवर्णस्त्वनीकानां सर्वेषामग्रतः स्थितः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0678"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा ययौ यत्तः सिंहलाङ्गूलकेतुना।</div><div class="line" style="position:relative;">श्रुतायुधश्चित्रसेनः पुरुमित्रो विविंशतिः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0679"><div class="verse-block verse-justify"><div class="line">शल्यो भूरिश्रवाश्चैव विकर्णश्च महारथः।</div><div class="line" style="position:relative;">एते सप्त महेष्वासा द्रोणपुत्रपुरोगमाः॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0681"><div class="verse-block verse-justify"><div class="line">भ्राजमाना व्यरोचन्त जाम्बूनदमया ध्वजाः।</div><div class="line" style="position:relative;">जाम्बूनदमयी वेदी कमण्डलुविभूषिता॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0682"><div class="verse-block verse-justify"><div class="line">केतुराचार्यमुख्यस्य द्रोणस्य धनुषा सह।</div><div class="line" style="position:relative;">अनेकशतसाहस्रमनीकमनुकर्षतः॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0685"><div class="verse-block verse-justify"><div class="line">तदङ्गपतिनां गुप्तं कृपेण च मनस्विना।</div><div class="line" style="position:relative;">शारदाम्बुधरप्रख्यं प्राच्यानां सुमहद्बलम्॥२९॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0692"><div class="verse-block verse-justify"><div class="line">केतुमानपि मातङ्गं विचित्रपरमाङ्कुशम्।</div><div class="line" style="position:relative;">आस्थितः समरे राजन्मेघस्थ इव भानुमान्॥३६॥</div></div></div>
