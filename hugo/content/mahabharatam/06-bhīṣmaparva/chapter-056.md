<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2504"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2505"><div class="verse-block verse-justify"><div class="line">प्रभातायां च शर्वर्यां भीष्मः शान्तनस्तदा।</div><div class="line" style="position:relative;">अनीकान्यनुसंयाने व्यादिदेशाय भारत॥२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2506"><div class="verse-block verse-justify"><div class="line">गारुडं च महाव्यूहं चक्रे शान्तनवस्तदा।</div><div class="line">पुत्राणां ते जयाकाङ्क्षी भीष्मः कुरुपितामहः।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2507"><div class="verse-block verse-justify"><div class="line">गरुडस्य स्वयं तुण्डे पिता देवव्रतस्तव।</div><div class="line" style="position:relative;">चक्षुषी च भरज्वाजः कृतवर्मा च सात्वतः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2508"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा कृपश्चैव शिर आस्तां यशस्विनौ।</div><div class="line" style="position:relative;">त्रैगर्त्तैरथ कैकेयैर्वाटधानैश्च संयुगे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2509"><div class="verse-block verse-justify"><div class="line">भूरिश्रवाः शलः शल्यो भगदत्तश्च मारिष।</div><div class="line" style="position:relative;">मद्रकाः सिन्धुसौवीरास्तथा पाञ्चनदाश्च ये॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2510"><div class="verse-block verse-justify"><div class="line">जयद्रथेन सहिता ग्रीवायां सन्निवेशिताः।</div><div class="line" style="position:relative;">पृष्ठे दुर्योधनो राजा सोदर्यैः सानुगैर्वृतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2511"><div class="verse-block verse-justify"><div class="line">विन्दानुविन्दावावन्त्यौ काम्भोजश्च शकैः सह।</div><div class="line" style="position:relative;">पच्छमासन्महाराज शूरसेनाश्च सर्वशः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2512"><div class="verse-block verse-justify"><div class="line">मागधाश्च कलिङ्गाश्च दासेरकगणैः सह।</div><div class="line" style="position:relative;">दक्षिणं पक्षमासाद्य स्थिता व्यूहस्य दंशिताः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2513"><div class="verse-block verse-justify"><div class="line">काकरुशा विकुञ्चाश्च मुण्डाः कुण्डीवृषास्तथा।</div><div class="line" style="position:relative;">बृहद्वलेन सहिता वामं पार्श्वमवस्थिताः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2514"><div class="verse-block verse-justify"><div class="line">व्यूढं दृष्ट्वा तु तत्सैन्यं सव्यसाची परंतपः।</div><div class="line" style="position:relative;">धृष्टद्युम्नेन सहितः प्रत्यव्यूहत संयुगे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2515"><div class="verse-block verse-justify"><div class="line">अर्धचन्द्रेण व्यूहेन व्यूहन्तमतिदारुणम्।</div><div class="line" style="position:relative;">दक्षिणं शृङ्गमास्वय भिमसेनो व्यरोचत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2516"><div class="verse-block verse-justify"><div class="line">नानासंस्त्रौपसंषत्रैर्नानादेश्यैर्नृपैर्वृतः।</div><div class="line" style="position:relative;">तदन्येव विराटश्च द्रुपदश्च महारथः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2517"><div class="verse-block verse-justify"><div class="line">तदनन्तरमेवासीन्नीलो नीलायुधैः सह।</div><div class="line" style="position:relative;">नीलादनन्तरश्चैव धृष्टकेतुर्महाबलः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2518"><div class="verse-block verse-justify"><div class="line">चेदिकाशिकरूपेश्च पौरवैरपि संवृतः।</div><div class="line" style="position:relative;">धृष्टद्युम्नः शिखण्डी च पाञ्चालाश्च प्रभद्रकाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2519"><div class="verse-block verse-justify"><div class="line">मध्ये सैन्यस्य महतः स्थिता युद्धाय भारत।</div><div class="line">तत्रैव धर्मराजोऽपि गजानीकेन संवृतः।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2520"><div class="verse-block verse-justify"><div class="line">ततस्तु सात्यकी राजन्द्रौपद्याः पञ्च चात्मजाः।</div><div class="line" style="position:relative;">अभिमन्युस्ततः शूर इरावांश्च ततः परम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2521"><div class="verse-block verse-justify"><div class="line">भैमसेनिस्ततो राजन्केकयाश्च महारथाः।</div><div class="line" style="position:relative;">एते सर्वे महाराज वामं पार्श्वमुपाश्रिताः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2522"><div class="verse-block verse-justify"><div class="line">सर्वस्य जगतो गोप्ता गोप्ता यस्य जनार्दनः।</div><div class="line">`तत्रानुरथिनां श्रेष्ठो वामश्रृङ्गे व्यवस्थितः।</div><div class="line" style="position:relative;">'एवमेतं महाव्यूहं प्रत्यव्यूहन्त पाण्डवाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2523"><div class="verse-block verse-justify"><div class="line">वधार्थं तव पुत्रामां तत्पक्षे ये च सङ्गताः।</div><div class="line" style="position:relative;">ततः प्रववृते युद्धं व्यतिषक्तरथद्विपम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2524"><div class="verse-block verse-justify"><div class="line">तावकानां परेषां च निघ्नतामितरेतरम्।</div><div class="line" style="position:relative;">हयौघाश्च रथौघाश्व तत्र तत्र विशांपते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2525"><div class="verse-block verse-justify"><div class="line">संपतन्तो व्यदृश्यन्त निघ्नन्तस्ते परस्परम्।</div><div class="line" style="position:relative;">धावतां च रथोघानां निघ्नतां च पृथक् पृथक्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2526"><div class="verse-block verse-justify"><div class="line">बभूव तुमुलः शब्दो विमिश्रो दुन्दुभिस्वनैः।</div><div class="line">दिवस्पृङ्वरवीराणां निघ्नतामितरेतरम्।</div><div class="line" style="position:relative;">संप्रहारे सुतुमुले तव तेषां च भारत॥२३॥</div></div></div></div>
