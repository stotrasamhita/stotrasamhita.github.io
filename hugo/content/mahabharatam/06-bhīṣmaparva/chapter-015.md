<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0606"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0607"><div class="verse-block verse-justify"><div class="line">त्वद्युक्तोऽयमनुप्रश्नो महाराज यथार्हसि।</div><div class="line" style="position:relative;">न तु दुर्योधने दोषमिममासङ्क्तुमर्हसि॥२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0608"><div class="verse-block verse-justify"><div class="line">य आत्मनो दुश्चरितादशुभं प्राप्नुयान्नरः।</div><div class="line" style="position:relative;">एनसा तेन नान्यं स उपाशङ्कितुमर्हति॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0609"><div class="verse-block verse-justify"><div class="line">महाराज मनुष्येषु निन्द्यं यः सर्वमाचरेत्।</div><div class="line" style="position:relative;">स वध्यः सर्वलोकस्य निन्दितानि समाचरन्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0610"><div class="verse-block verse-justify"><div class="line">निकारो निकृतिप्रज्ञैः पाण्डवैस्त्वत्प्रतीक्षया।</div><div class="line" style="position:relative;">अनुभूतः सहामात्यैः क्षान्तश्च सुचिरं वने॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0611"><div class="verse-block verse-justify"><div class="line">हयानां च गजानां च राज्ञां चामिततेजसाम्।</div><div class="line" style="position:relative;">प्रत्यक्षं यन्मया दृष्टं दृष्टं योगबलेन च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0612"><div class="verse-block verse-justify"><div class="line">शृणु तत्पृथिवीपाल मा च शोके मनः कृथाः।</div><div class="line" style="position:relative;">दिष्टमेतत्पुरा नूनमिदमेव नराधिप॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0613"><div class="verse-block verse-justify"><div class="line">नमस्कृत्वा प्रवक्ष्यामि पाराशर्याय धीमते।</div><div class="line" style="position:relative;">यस्य प्रसादाद्दिव्यं तत्प्राप्तं ज्ञानमनुत्तमम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0614"><div class="verse-block verse-justify"><div class="line">दृष्टिश्चातीन्द्रिया राजन्दूराच्छ्रवणमेव च।</div><div class="line" style="position:relative;">परचित्तस्य विज्ञानमतीतानागतस्य च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0615"><div class="verse-block verse-justify"><div class="line">व्युत्थितोत्पत्तिविज्ञानमाकाशे च गतिः शुभा।</div><div class="line" style="position:relative;">अस्त्रैरसङ्गो युद्धेषु वरदानान्महात्मनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0616"><div class="verse-block verse-justify"><div class="line">श्रृणु मे विस्तरेणेदं विचित्रं परमाद्भुतम्।</div><div class="line" style="position:relative;">भरतानामभूद्युद्धं यथा तद्रोमहर्षणम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0617"><div class="verse-block verse-justify"><div class="line">तेष्वनीकेषु यत्तेषु व्यूढेषु च विधानतः।</div><div class="line" style="position:relative;">दुर्योधनो महाराज दुःशासनमथाब्रवीत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0618"><div class="verse-block verse-justify"><div class="line">दुःशासन रथास्तूर्णं युज्यन्तां भीष्मरक्षिणः।</div><div class="line" style="position:relative;">अनीकानि च सर्वाणि शीघ्रं त्वमनुचोदय॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0619"><div class="verse-block verse-justify"><div class="line">अयं स मामभिप्राप्तो वर्षपूगाभिचिन्तितः।</div><div class="line" style="position:relative;">पाण्डवानां ससैन्यानां कुरूणां च समागमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0620"><div class="verse-block verse-justify"><div class="line">नातः कार्यतमं मन्ये रणे भीष्मस्य रक्षणात्।</div><div class="line" style="position:relative;">हन्युद्गुप्तो ह्यसौ पार्थान्सोमकांश्च ससृञ्जयान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0621"><div class="verse-block verse-justify"><div class="line">अब्रवीच्च विशुद्धात्मा नाहं हन्यां शिखण्डिनाम्।</div><div class="line" style="position:relative;">श्रूयते स्त्री ह्यसौ पूर्वं तस्मादूर्त्यो रणे मम॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0622"><div class="verse-block verse-justify"><div class="line">तस्माद्भीष्मो रक्षितव्यो विशेषेणेति मे मतिः।</div><div class="line" style="position:relative;">शिखण्डिनो वधे यत्ताः सर्वे तिष्ठन्तु मामकाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0623"><div class="verse-block verse-justify"><div class="line">तथा प्राच्याः प्रतीच्याश्च दाक्षिणात्योत्तरापथाः।</div><div class="line" style="position:relative;">सर्वथाऽस्त्रेषु कुशलास्ते रक्षन्तु पितामहम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0624"><div class="verse-block verse-justify"><div class="line">अरक्ष्यमाणं हि वृको हन्यात्सिंहं महाबलम्।</div><div class="line" style="position:relative;">मा सिंहं जम्बुकेनेव घातयामः शिखण्डिना॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0625"><div class="verse-block verse-justify"><div class="line">वामं चक्रं युधामन्युरुत्तमौजाश्च दक्षिणम्।</div><div class="line">गोप्तारौ फाल्गुनं प्राप्तौ फाल्गुनोपि शिखण्डिनः।</div><div class="line" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b0626"><div class="verse-block verse-justify"><div class="line">संरक्ष्यमाणः पार्थेन भीष्मेण च विवर्जितः।</div><div class="line" style="position:relative;">यथा न हन्याद्गाङ्गेयं दुःशासन तथा कुरु॥२१॥</div></div></div></div>
