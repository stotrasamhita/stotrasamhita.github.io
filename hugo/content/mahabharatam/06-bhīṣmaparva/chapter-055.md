<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2458"><div class="verse-block verse-justify"><div class="line">ततोऽपराह्णभूयिष्ठे तस्मिन्नहनि भारत।</div><div class="line" style="position:relative;">रथनागाश्वपत्तीनां सादिनां च महाक्षये॥२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2459"><div class="verse-block verse-justify"><div class="line">द्रोणपुत्रेण शल्येन कृपेण च महात्मना।</div><div class="line" style="position:relative;">समसज्जत पाञ्चाल्यस्त्रिभिरैतैर्महारथैः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2460"><div class="verse-block verse-justify"><div class="line">स लोकविदितानश्वान्निजघान महाबलः।</div><div class="line" style="position:relative;">द्रौणेः पाञ्चालदायादः शितैर्दशभिराशुगैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2461"><div class="verse-block verse-justify"><div class="line">ततः शल्यरथं तूर्णमास्थाय हतवाहनः।</div><div class="line" style="position:relative;">द्रौणिः पाञ्चालदायादमभ्यवर्षदथेषुभिः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2462"><div class="verse-block verse-justify"><div class="line">धृष्टद्युम्नं तु संयुक्तं द्रौणिना वीक्ष्य भारत।</div><div class="line" style="position:relative;">सौभद्रोऽभ्यपतत्तूर्णं विकिरन्निशिताञ्शरान्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2463"><div class="verse-block verse-justify"><div class="line">स शल्यं प़ञ्चविंशत्या कृपं च नवभिः शरैः।</div><div class="line" style="position:relative;">अश्वत्थामानमष्टाभिर्विव्याध पुरुषर्षभः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2464"><div class="verse-block verse-justify"><div class="line">आर्जुनिं तु ततस्तूर्णं द्रौणिर्विव्याध पत्रिणा।</div><div class="line" style="position:relative;">शल्योऽथ दशभिश्चैव कृपश्च निशितैस्त्रिभिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2465"><div class="verse-block verse-justify"><div class="line">लक्ष्मणस्तव पौत्रस्तु सौभद्रं समवस्थितम्।</div><div class="line" style="position:relative;">अभ्यवर्तत संहृष्टस्ततो युद्धमवर्तत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2466"><div class="verse-block verse-justify"><div class="line">दौर्योधनिः सुसंक्रुद्धः सौभद्रं परवीरहा।</div><div class="line" style="position:relative;">विव्याध समरे राजंस्तदद्भुतमिवाभवत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2467"><div class="verse-block verse-justify"><div class="line">अभिमन्युः सुसंक्रुद्धो भ्रातरं भरतर्षभ।</div><div class="line" style="position:relative;">शरैः पञ्चाशतै राजन्क्षिप्रहस्तोऽभ्यविध्यत॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2468"><div class="verse-block verse-justify"><div class="line">लक्ष्मणोऽपि पुनस्तस्य धनुश्चिच्छेद पत्रिणा।</div><div class="line" style="position:relative;">मुष्टिदेशे महाराज ततस्ते चुक्रुशुर्जनाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2469"><div class="verse-block verse-justify"><div class="line">तद्विहाय धनुश्छिन्नं सौभद्रं परवीरहा।</div><div class="line" style="position:relative;">अन्यदादत्तवांश्चित्रं कार्मुकं वेगवत्तरम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2470"><div class="verse-block verse-justify"><div class="line">तौ तत्र समरे युक्तौ कृतप्रतिकृतैषिणौ।</div><div class="line" style="position:relative;">अन्योन्यं विशिस्वैस्तीक्ष्णैर्जघ्नतुः पुरुषर्षभौ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2471"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राज दृष्ट्वा पुत्रं महारथम्।</div><div class="line" style="position:relative;">पीडितं तव पौत्रेण प्रायात्तत्र प्रजेश्वरः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2472"><div class="verse-block verse-justify"><div class="line">सन्निवृत्ते तव सुते सर्व एव जनाधिपाः।</div><div class="line" style="position:relative;">आर्जुनिं रथवंशेन समन्तात्पर्यवारयन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2473"><div class="verse-block verse-justify"><div class="line">स तैः परिवृतः शूरैः शूरो युधि सुदुर्जयैः।</div><div class="line" style="position:relative;">न स्म प्रव्यथते राजन्कृष्णतुल्यपराक्रमः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2474"><div class="verse-block verse-justify"><div class="line">सौभद्रमथ संसक्तं दृष्ट्वा तत्र धनंजयः।</div><div class="line" style="position:relative;">अभिदुद्राव वेगेन त्रातुकामः स्वमात्मजम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2475"><div class="verse-block verse-justify"><div class="line">ततः सरथनागाश्वा भीष्मद्रोणपुरोगमाः।</div><div class="line" style="position:relative;">अभ्यवर्तन्त राजानः सहिताः सव्यसाचिनम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2476"><div class="verse-block verse-justify"><div class="line">उद्वूतं सहसा भौमं नागाश्वरथपत्तिभिः।</div><div class="line" style="position:relative;">दिवाकररथं प्राप्य रजस्तीव्रमदृश्यत॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2477"><div class="verse-block verse-justify"><div class="line">तानि नागसहस्राणि भूमिपालशतानि च।</div><div class="line" style="position:relative;">तस्य बाणपथं प्राप्य नाभ्यवर्तन्त सर्वशः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2478"><div class="verse-block verse-justify"><div class="line">प्रणेदुः सर्वभूतानि बभूवुस्तिमिरा दिशः।</div><div class="line" style="position:relative;">कुरूणां चानयस्तीव्रः समदृश्यत दारुणः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2479"><div class="verse-block verse-justify"><div class="line">नाप्यन्तरिक्षं न दिशो न भूमिर्न च भास्करः।</div><div class="line" style="position:relative;">प्रजज्ञे भरतश्रेष्ठ सस्त्रसङ्घैः किरीटिनः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2480"><div class="verse-block verse-justify"><div class="line">सादिता रथनागाश्च हताश्वा रथिनो रणे।</div><div class="line" style="position:relative;">विप्रद्रुतरथाः केचिद्दृश्यन्ते रथयूथपाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2481"><div class="verse-block verse-justify"><div class="line">विरथा रथिनश्चान्ये धावमानाः समन्ततः।</div><div class="line" style="position:relative;">तत्रतत्रैव दृश्यन्ते सायुधाः साङ्गदैर्भुजैः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2482"><div class="verse-block verse-justify"><div class="line">हयारोहा हयांस्त्यक्त्वा गजारोहाश्च दन्तिनः।</div><div class="line" style="position:relative;">अर्जुनस्य भयाद्राजन्समन्ताद्विप्रदुद्रुवुः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2483"><div class="verse-block verse-justify"><div class="line">रथेभ्यश्च गजेभ्यश्च हयेभ्यश्च नराधिपाः।</div><div class="line" style="position:relative;">पतिताः पात्यमानाश्च दृश्यन्तेऽर्जुनसायकैः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2484"><div class="verse-block verse-justify"><div class="line">सगदानुद्यतान्बाहून्सखङ्गांश्च विशांपते।</div><div class="line" style="position:relative;">सप्रासांश्च सतूणीरान्सशरान्सशरासनान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2485"><div class="verse-block verse-justify"><div class="line">साङ्कुशाकन्सपताकांश्च तत्रतत्रार्जुनो नृणाम्।</div><div class="line" style="position:relative;">निचकर्त शरैरुग्रै रौद्रं वपुरधारयत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2486"><div class="verse-block verse-justify"><div class="line">परिघाणां प्रदीप्तानां मुद्गराणां च मारिष।</div><div class="line" style="position:relative;">प्रासानां भिन्दिपालानां निस्त्रिशानां च संयुगे॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2487"><div class="verse-block verse-justify"><div class="line">परश्वथानां पीक्ष्णानां तोमराणां च भारत।</div><div class="line" style="position:relative;">वर्मणां चापविद्धानां काञ्चनानां च भूमिप॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2488"><div class="verse-block verse-justify"><div class="line">ध्वजानां चर्मणां चैव व्यजनानां च सर्वशः।</div><div class="line" style="position:relative;">छत्राणां हेमदण्डानां तोमराणां च भारत॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2489"><div class="verse-block verse-justify"><div class="line">प्रतोदानां च योक्राणां कशानां चैव मारिष।</div><div class="line" style="position:relative;">राशयः स्मात्र दृश्यन्ते विनिकीर्णा रणक्षितौ॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2490"><div class="verse-block verse-justify"><div class="line">नासीत्तत्र पुमान्कश्चित्तव सैन्यस्य भारत।</div><div class="line" style="position:relative;">योऽर्जुनं समरे शूरं प्रत्युद्यायात्कथंचन॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2491"><div class="verse-block verse-justify"><div class="line">यो यो हि समरे पार्थं प्रत्युद्याति विशांपते।</div><div class="line" style="position:relative;">स सङ्ख्ये विशिखैस्तीक्ष्णैः परलोकाय नीयते॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2492"><div class="verse-block verse-justify"><div class="line">तेषु विद्रवमाणेषु तव योधेषु सर्वशः।</div><div class="line" style="position:relative;">अर्जुनो वासुदेवश्च दध्मतुर्वारिजोत्तमौ॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2493"><div class="verse-block verse-justify"><div class="line">तत्पभग्नं बलं दृष्ट्वा पिता देवव्रतस्तव।</div><div class="line" style="position:relative;">अब्रवीत्समरे शूरं भारद्वाजं स्मयन्निव॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2494"><div class="verse-block verse-justify"><div class="line">एष पाण्डुसुतो वीर कृष्णेन सहितो बली।</div><div class="line" style="position:relative;">तथा करोति सैन्यानि यथा कुर्याद्धऩञ्जयः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2495"><div class="verse-block verse-justify"><div class="line">न ह्येष समरे शक्यो विजेतुं हि कथंचन।</div><div class="line" style="position:relative;">यथास्य दृश्यते रूप कालान्तकयमोपमम्॥३९॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2497"><div class="verse-block verse-justify"><div class="line">एष चास्तं गिरिश्रेष्ठं भानुमान्प्रतिपद्यते।</div><div class="line" style="position:relative;">चक्षूंषि सर्वलोकस्य संहरन्निव सर्वथा॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2498"><div class="verse-block verse-justify"><div class="line">तत्रावहारं संप्राप्तं मन्येऽहं पुरुषर्षभ।</div><div class="line" style="position:relative;">श्रान्ता भीताश्च नो योधा न योत्स्यन्ति कथंचन॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b2499"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा ततो भीष्मो द्रोणमाचार्यसत्तमम्।</div><div class="line" style="position:relative;">अवहारमथो चक्रे तावकानां महारथः॥४३॥</div></div></div>
