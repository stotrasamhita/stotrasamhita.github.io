<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4946"><div class="verse-block verse-justify"><div class="line">अर्जुनस्तान्नरव्याघ्रः सुशर्मानुचरान्नृपान्।</div><div class="line" style="position:relative;">अनयत्प्रेतराजस्य सदनं सायकैः शितैः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4947"><div class="verse-block verse-justify"><div class="line">सुशर्मापि ततो बाणैः पार्थं विव्याध संयुगे।</div><div class="line" style="position:relative;">वासुदेवं च सप्तत्या पार्थं च नवभिः पुनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4948"><div class="verse-block verse-justify"><div class="line">तं निवार्य शरौघेण सक्रुसूनुर्महारथः।</div><div class="line" style="position:relative;">सुशर्मणो रणे योधान्प्राहिणोद्यमसादनम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4949"><div class="verse-block verse-justify"><div class="line">ते वध्यमानाः पार्थेन कालेनेव युगक्षये।</div><div class="line" style="position:relative;">व्यद्रवन्त रणे राजन्भये जाते महारथाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4950"><div class="verse-block verse-justify"><div class="line">उत्सृज्य तुरगान्केचिद्रथान्केचिच्च मारिष।</div><div class="line" style="position:relative;">गजानन्ये समुत्सृज्य प्राद्रवन्त दिशो दश॥६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4951"><div class="verse-block verse-justify"><div class="line">अपरे तु तदादाय वाजिनागरथान्रणे।</div><div class="line" style="position:relative;">त्वरया परया युक्ताः प्राद्रवन्त विशांपते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4952"><div class="verse-block verse-justify"><div class="line">पादाताश्चापि शस्त्राणि समुत्सृज्य महारणे।</div><div class="line" style="position:relative;">निरपेक्षा व्यधावन्त तेनतेन स्म भारत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4953"><div class="verse-block verse-justify"><div class="line">वार्यमाणाः सुबहुशस्त्रैगर्तेन सुशर्मणा।</div><div class="line" style="position:relative;">तथान्यैः पार्थिवश्रेष्ठैर्न व्यतिष्ठन्त संयुगे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4954"><div class="verse-block verse-justify"><div class="line">तद्बलं प्रद्रुतं दृष्ट्वा पुत्रो दुर्योधनस्तव।</div><div class="line" style="position:relative;">पुरस्कृत्य रणे भीष्मं सर्वसैन्यपुरस्कृतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4955"><div class="verse-block verse-justify"><div class="line">सर्वोद्योगेन महता धनञ्जयमुपाद्रवत्।</div><div class="line" style="position:relative;">त्रिगर्ताधिपतेरर्थे जीवितस्य विशांपते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4956"><div class="verse-block verse-justify"><div class="line">स एकः समरे तस्थौ किरन्बहुविधाञ्शरान्।</div><div class="line" style="position:relative;">भ्रातृभिः सहितः सर्वैः शेषा हि प्रद्रुता नराः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4957"><div class="verse-block verse-justify"><div class="line">तथैव पाण्डवा राजन्सर्वोद्योगेन दंशिताः।</div><div class="line" style="position:relative;">प्रययुः फल्गुनार्थाय यत्र भीष्मो व्यतिष्ठत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4958"><div class="verse-block verse-justify"><div class="line">ज्ञायमाना रणे वीर्यं घोरं गाण्डीवधन्वनः।</div><div class="line" style="position:relative;">हाहाकारकृतोत्साहा भीष्मं जग्मुः समन्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4959"><div class="verse-block verse-justify"><div class="line">ततस्तालध्वजः शूरः पाण्डवानां वरूथिनीम्।</div><div class="line" style="position:relative;">छादयामास समरे शरैः सन्नतपर्वभिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4960"><div class="verse-block verse-justify"><div class="line">एकीभूतास्ततः सर्वे कुरवः सह पाण्डवैः।</div><div class="line" style="position:relative;">अयुध्यन्त महाराज मध्यं प्राप्ते दिवाकरे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4961"><div class="verse-block verse-justify"><div class="line">सात्यकिः कृतवर्माणं विद्ध्वा पञ्चभिराशुगैः।</div><div class="line" style="position:relative;">अतिष्ठदाहवे शूरः किरन्बाणान्सहस्रशः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4962"><div class="verse-block verse-justify"><div class="line">तथैव द्रुपदो राजा द्रोणं विद्ध्वा शितैः शरैः।</div><div class="line" style="position:relative;">पुनर्विव्याध सप्तत्या सारथिं चास्य पञ्चभिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4963"><div class="verse-block verse-justify"><div class="line">भीमसेनस्तु राजानं बाह्लीकं प्रपितामहम्।</div><div class="line" style="position:relative;">विद्ध्वा नदन्महानादं शार्दूल इव कानने॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4964"><div class="verse-block verse-justify"><div class="line">आर्जुनिश्चित्रसेनेन विद्धो बहुभिरशुगैः।</div><div class="line">अतिष्ठदाहवे शूरः किरन्बाणान्सहस्रशः।</div><div class="line" style="position:relative;">चित्रसेनं त्रिभिर्बाणैर्विव्याध समरे भृशम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4965"><div class="verse-block verse-justify"><div class="line">समागतौ तौ तु रणे महामात्रौ व्यरोचताम्।</div><div class="line" style="position:relative;">यथा दिवि महाघोरौ राजन्बुधशनैश्चरौ॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4966"><div class="verse-block verse-justify"><div class="line">तस्याश्वांश्चतुरो हत्वा सूतं च नवभिः शरैः।</div><div class="line" style="position:relative;">ननाद बलवान्नादं सौभद्रः परवीरहा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4967"><div class="verse-block verse-justify"><div class="line">हताश्वात्तु रथात्तूर्णं सोऽवप्लुत्य महारथः।</div><div class="line" style="position:relative;">आरुरोह रथं तूर्णं दुर्मुखस्य विशांपते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4968"><div class="verse-block verse-justify"><div class="line">द्रोणश्च द्रुपदं भित्त्वा शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">सारथिं चास्य विव्याध त्वरमाणः पराक्रमी॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4969"><div class="verse-block verse-justify"><div class="line">पीड्यमानस्ततो राजा द्रुपदो वाहिनीमुखे।</div><div class="line" style="position:relative;">अपायाज्जवनैरश्वैः पूर्ववैरमनुस्मरन्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4970"><div class="verse-block verse-justify"><div class="line">भीमसेनस्तु राजानं मुहूर्तादिव बाह्लिकम्।</div><div class="line" style="position:relative;">व्यश्वसूतरथं चक्रे सर्वसैन्यस्य पश्यतः॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4975"><div class="verse-block verse-justify"><div class="line">तस्यायसीं महाशक्तिं चिक्षेपाथ पितामहः।</div><div class="line" style="position:relative;">हेमचित्रां महावेगां नागकन्योपमां शुभाम्॥३०॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4977"><div class="verse-block verse-justify"><div class="line">अनासाद्य तु वार्ष्णेयं शक्तिः परमदारुणा।</div><div class="line" style="position:relative;">न्यपतद्धरणीपृष्ठे महोल्केव महाप्रभा॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4978"><div class="verse-block verse-justify"><div class="line">वार्ष्णेयस्तु ततो राजन्स्वां शक्तिं कनकप्रभाम्।</div><div class="line" style="position:relative;">वेगवद्गृह्य चिक्षेप पितामहरथं प्रति॥३३॥</div></div></div>
