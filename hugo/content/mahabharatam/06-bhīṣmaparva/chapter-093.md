<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4344"><div class="verse-block verse-justify"><div class="line">तालमात्राणि चापानि विकर्षन्तो महारथाः।</div><div class="line" style="position:relative;">तमेकमभ्यधावन्त नदन्तः सिंहसङ्घवत्॥४॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4347"><div class="verse-block verse-justify"><div class="line">व्यनदत्सुमहानादं जीमूत इव शारदः।</div><div class="line" style="position:relative;">दिशः खं विदिशश्चैव नादयन्भैरवस्वनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4348"><div class="verse-block verse-justify"><div class="line">राक्षसस्य तु तं शब्दं श्रुत्वा राजा युधिष्ठिरः।</div><div class="line" style="position:relative;">उवाच भरतश्रेष्ठ भीमसेनमरिन्दमम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4349"><div class="verse-block verse-justify"><div class="line">युध्यते राक्षसो नूनं धार्तराष्ट्रैर्महारथैः।</div><div class="line" style="position:relative;">यथाऽस्य श्रूयते शब्दो नदतो भैरवं स्वनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4350"><div class="verse-block verse-justify"><div class="line">अतिभारं च पश्यामि तस्मिन्राक्षसपुङ्गवे।</div><div class="line" style="position:relative;">पितामहश्च संक्रुद्धः पाञ्चालान्हन्तुमुद्यतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4351"><div class="verse-block verse-justify"><div class="line">तेषां च रक्षणार्थाय युध्यते फल्गुनः परैः।</div><div class="line" style="position:relative;">एतज्ज्ञात्वा महाबाहो कार्यद्वयमुपस्थितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4352"><div class="verse-block verse-justify"><div class="line">गच्छ रक्षस्व हैडिम्बं शंशयं परमं गतम्।</div><div class="line" style="position:relative;">भ्रातुर्वचनमाज्ञाय त्वरमाणो वृकोदरः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4353"><div class="verse-block verse-justify"><div class="line">प्रययौ सिंहनादेन त्रासयन्सर्वपार्थिवान्।</div><div class="line" style="position:relative;">वेगेन महता राजन्पर्वकाले यथोदधिःक॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4354"><div class="verse-block verse-justify"><div class="line">तमन्वगात्सत्यधृतिः सौचित्तिर्युद्धदुर्मदः।</div><div class="line" style="position:relative;">श्रेणिमान्वसुदानश्च पुत्रः काश्यस्य चाभिभूः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4355"><div class="verse-block verse-justify"><div class="line">अभिमन्युमुखाश्चैव द्रौपदेया महारथाः।</div><div class="line" style="position:relative;">क्षत्रदेवश्च विक्रन्तः क्षत्रधर्मा तथैव च॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4356"><div class="verse-block verse-justify"><div class="line">अनूपाधिपतिश्चैव नीलः स्वबलमास्थितः।</div><div class="line" style="position:relative;">महता रथवंशेन हैडिम्बं पर्यवारयन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4357"><div class="verse-block verse-justify"><div class="line">कुञ्चरैश्च सदा मत्तैः षट्सहस्रैः प्रहारिभिःक।</div><div class="line" style="position:relative;">अभ्यरक्षन्त सहिता राक्षसेन्द्रं घटोत्कचम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4358"><div class="verse-block verse-justify"><div class="line">सिंहनादेन महता नेमिघोषेण चैव ह।</div><div class="line" style="position:relative;">खुरशब्दनिपातैश्च कम्पयन्तो वसुंधराम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4359"><div class="verse-block verse-justify"><div class="line">तेषामापततां श्रुत्वा शब्दं तं तावकं बलम्।</div><div class="line" style="position:relative;">भीमसेनभयोद्विग्नं विवर्णवदनं तथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4360"><div class="verse-block verse-justify"><div class="line">परिवृत्य महाराज परित्यज्य घटोत्कचम्।</div><div class="line" style="position:relative;">ततः प्रववृते युद्धंक तत्र तेषां महात्मनाम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4361"><div class="verse-block verse-justify"><div class="line">तावकानां परेषां च संग्रामेष्वनिवर्तिनाम्।</div><div class="line" style="position:relative;">नानारूपाणि शस्त्राणि विसृजन्तो महारथाः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4362"><div class="verse-block verse-justify"><div class="line">अन्योन्यमभिधावन्तः संप्रहारं प्रचक्रिरे।</div><div class="line" style="position:relative;">व्यतिष्यक्तं महारौद्रं युद्धं भीरुभयावहम्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4365"><div class="verse-block verse-justify"><div class="line">धूम्रारुणं रजस्तीव्रं रणभूमिं समावृणोत्।</div><div class="line" style="position:relative;">नैव स्वे न परे राजन्समजानन्परस्परम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4366"><div class="verse-block verse-justify"><div class="line">पिता पुत्रं न जानीते पुत्रो वा पितरं तथा।</div><div class="line" style="position:relative;">निर्मर्यादे तथाभूते वैशसे रोमहर्षणे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4367"><div class="verse-block verse-justify"><div class="line">शस्त्राणां भरतश्रेष्ठ मनुष्याणां च र्जताम्।</div><div class="line" style="position:relative;">सुमहानभवच्छब्दः प्रेतानामिव भारत॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4368"><div class="verse-block verse-justify"><div class="line">गजवाजिमनुष्याणां शोणितान्रतरङ्गिणी।</div><div class="line" style="position:relative;">प्राववर्तत नदी तत्र केशशैवलशाद्वला॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4369"><div class="verse-block verse-justify"><div class="line">नराणां चैव कायेभ्यः शिरसां पततां रणे।</div><div class="line" style="position:relative;">शुश्रुवे सुमहाञ्छब्दः पततामश्मनामिव॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4370"><div class="verse-block verse-justify"><div class="line">विशिरस्कैर्मनुष्यैश्च छिन्नगात्रैश्च वारणैः।</div><div class="line" style="position:relative;">अश्वैः संभिन्नदेहैश्च संकीर्णाऽभूद्वसुंधरा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4371"><div class="verse-block verse-justify"><div class="line">नानाविधानि शस्त्राणि विसृजन्तो महारथाः।</div><div class="line" style="position:relative;">अन्योन्यमभिधावन्तः संप्रहारार्थमुद्यताः॥३१॥</div></div></div>

<div class="verse-block-wrapper" id="06-bhīṣmaparva-b4374"><div class="verse-block verse-justify"><div class="line">प्रेषिताश्च महामात्रैर्वारणाः परवारणैः।</div><div class="line" style="position:relative;">अभ्यघ्नन्त विषणाग्रैर्वारणानेव संयुगे॥३४॥</div></div></div>
