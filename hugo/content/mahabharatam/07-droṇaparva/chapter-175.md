<div class="verse-block-wrapper" id="07-droṇaparva-b8404"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नन्तरे राजञ्जटासुरसुतो बली।</div><div class="line" style="position:relative;">दुर्योधनमुपागम्य प्राह प्रहरतां वरः॥६॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b8406"><div class="verse-block verse-justify"><div class="line">जटासुरो मम पिता रक्षसां ग्रामणीः पुरा।</div><div class="line" style="position:relative;">प्रयुज्य कर्म रक्षोघ्नं क्षुद्रैः पार्थैर्निपातितः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8407"><div class="verse-block verse-justify"><div class="line">तस्यापचितिमिच्छामि शत्रुशोणितपूजया।</div><div class="line" style="position:relative;">शत्रुमांसैश्च राजेन्द्र मामनुज्ञातुमर्हसि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8408"><div class="verse-block verse-justify"><div class="line">तमब्रवीत्ततो राजा प्रीयमाणः पुनःपुनः।</div><div class="line" style="position:relative;">द्रोणकर्मादिभिः सार्धं पर्याप्तोऽहं द्विषद्वधे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8409"><div class="verse-block verse-justify"><div class="line">त्वं तु गच्छ मयाऽऽज्ञप्तो जहि युद्धे घटोत्कचम्।</div><div class="line" style="position:relative;">राक्षसं क्रूरकर्माणं रक्षोमानुषसम्भवम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8410"><div class="verse-block verse-justify"><div class="line">पाण्डवानां हितं नित्यं हस्त्यश्वरथघातिनम्।</div><div class="line" style="position:relative;">वैहायसगतं युद्धे प्रेषयेर्यमसादनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8411"><div class="verse-block verse-justify"><div class="line">तथेत्युक्त्वा महाकायः समाहूय घटोत्कचम्।</div><div class="line" style="position:relative;">जाटासुरिर्भैमसेनिं नानाशस्त्रैरवाकिरत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8412"><div class="verse-block verse-justify"><div class="line">अलम्बलं च कर्णं च कुरुसैन्यं च दुस्तरम्।</div><div class="line" style="position:relative;">हैडिम्बिः प्रममाथैको महाबातोम्बुदानिव॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8413"><div class="verse-block verse-justify"><div class="line">ततो मायाबलं दृष्ट्वा रक्षस्तूर्णमलम्बलः।</div><div class="line" style="position:relative;">घटोत्कचं शरव्रातैर्नानालिङ्गैः समार्पयत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8414"><div class="verse-block verse-justify"><div class="line">विद्धा च बहुभिर्बाणैर्भैमसेनिं महाबलः।</div><div class="line" style="position:relative;">व्यद्रावयच्छरव्रातैः पाण्डवानामनीकिनीम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8415"><div class="verse-block verse-justify"><div class="line">तेन विद्राव्यमाणानि पाण्डुसैन्यानि भारत।</div><div class="line" style="position:relative;">निशीथे विप्रकीर्यन्ते वातनुन्ना घना इव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8416"><div class="verse-block verse-justify"><div class="line">घटोत्कचशरैर्नुन्ना तथैव तव वाहिनी।</div><div class="line" style="position:relative;">निशीथे प्राद्रवद्राजन्नुत्सृज्योल्काः सहस्रशः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8417"><div class="verse-block verse-justify"><div class="line">अलम्बलस्ततः क्रुद्धो भैमसेनिं महामृधे।</div><div class="line" style="position:relative;">आजघ्ने दशभिर्बाणैस्तोत्रैरिव महाद्विपम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8418"><div class="verse-block verse-justify"><div class="line">तिxशस्तस्य संवाहं सूतं सर्वायुधानि च।</div><div class="line" style="position:relative;">घटोत्कचः प्रचिच्छेद प्रणदंश्चातिदारुणम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b8419"><div class="verse-block verse-justify"><div class="line">ततः कर्णं शरव्रातैः कुरूनन्यान्सहस्रशः।</div><div class="line" style="position:relative;">अलम्बलं चाभ्यवर्षन्मेघो मेरुमिवाचलम्॥२१॥</div></div></div>
