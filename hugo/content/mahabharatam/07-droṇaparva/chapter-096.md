<div class="verse-block-wrapper" id="07-droṇaparva-b3847"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3848"><div class="verse-block verse-justify"><div class="line">राजन्सङ्ग्राममाश्चर्यं शृणु कीर्तयतो मम।</div><div class="line" style="position:relative;">कुरूणां पाण्डवानां च यथा युद्धमवर्तत॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3849"><div class="verse-block verse-justify"><div class="line">भारद्वाजं समासाद्य व्यूहस्य प्रमुखे स्थितम्।</div><div class="line" style="position:relative;">अयोधयन्रणे पार्था द्रोणानीकं बिभित्सवः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3850"><div class="verse-block verse-justify"><div class="line">रक्षमाणः स्वथं व्यूहं द्रोणोऽपि सह सैनिकैः।</div><div class="line" style="position:relative;">अयोधयद्रणे पार्थान्प्रार्थयानो महद्यशः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3851"><div class="verse-block verse-justify"><div class="line">विन्दानुविन्दावावन्त्यौ विराटं दशभिः शरैः।</div><div class="line" style="position:relative;">अयोधयद्रणे पार्थान्प्रार्थयानो महद्यशः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3852"><div class="verse-block verse-justify"><div class="line">विन्दानुविन्दावावन्त्यौ विराटं दशभिः शरैः।</div><div class="line" style="position:relative;">पराक्रान्तौ पराक्रम्य योधयामास सानुगौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3853"><div class="verse-block verse-justify"><div class="line">तेषां युद्धं समभवद्दारुणं शोणितोदकम्।</div><div class="line" style="position:relative;">सिंहस्य द्विपमुख्याभ्यां प्रभिन्नाभ्यां यथा वने॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3854"><div class="verse-block verse-justify"><div class="line">बाह्लीकं रभसं युद्धे याज्ञसेनिर्महाबलः।</div><div class="line" style="position:relative;">आजघ्ने विशिखैस्तीक्ष्णैर्घोरैर्मर्मास्थिभेदिभिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3855"><div class="verse-block verse-justify"><div class="line">बाह्लीको याज्ञसेनिं तु हेमपुङ्खैः शिलाशितैः।</div><div class="line" style="position:relative;">आजघान भृशं क्रुद्धो नवभिर्नतपर्वभिः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3856"><div class="verse-block verse-justify"><div class="line">तद्युद्धमभवद्धोरं शरशक्तिसमाकुलम्।</div><div class="line" style="position:relative;">भीरूणां त्रासजननं शूराणां हर्षवर्धनम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3857"><div class="verse-block verse-justify"><div class="line">ताभ्यां तत्र शरैर्मुक्तैरन्तरिक्षं दिशस्तथा।</div><div class="line" style="position:relative;">अभवत्संवृतं सर्वं न प्राज्ञायत किञ्चन॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3858"><div class="verse-block verse-justify"><div class="line">शैब्यो गोवासनो युद्धे काश्यपुत्रं महारथम्।</div><div class="line" style="position:relative;">ससैन्यो योधयामास गजः प्रतिगजं यथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3859"><div class="verse-block verse-justify"><div class="line">बाह्लीकराजः सङ्क्रुद्धो द्रौपदेयान्महारथान्।</div><div class="line" style="position:relative;">मनः पञ्चेन्द्रियाणीव शुशुभे योधयन्रणे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3860"><div class="verse-block verse-justify"><div class="line">अयोधयंस्ते सुभृशं तं शरौघैः समन्ततः।</div><div class="line" style="position:relative;">इन्द्रियाणि यथा देहं शश्वद्देहवतां वर॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3861"><div class="verse-block verse-justify"><div class="line">वार्ष्णेयं सात्यकिं युद्धे पुत्रो दुःशासनस्तव।</div><div class="line" style="position:relative;">आजघ्ने सायकैस्तीक्ष्णैर्नवभिर्नतपर्वभिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3862"><div class="verse-block verse-justify"><div class="line">सोऽतिविद्धो बलवता महेष्वासेन धन्विना।</div><div class="line" style="position:relative;">ईषन्मूर्च्छां जगामाशु सात्यकिः सत्यविक्रमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3863"><div class="verse-block verse-justify"><div class="line">समाश्वस्तस्तु वार्ष्णेयस्तव पुत्रं महारथम्।</div><div class="line" style="position:relative;">विव्याध दशभिस्तूर्णं सायकैः कङ्कपत्रिभिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3864"><div class="verse-block verse-justify"><div class="line">तावन्योन्यं दृढं विद्धावन्योन्यशरपीडितौ।</div><div class="line" style="position:relative;">रजेतुः समरे राजन्पुष्पिताविव किंशुकौ॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3865"><div class="verse-block verse-justify"><div class="line">अलम्बुसस्तु सङ्क्रुद्धः कृन्तिभोजशरार्दितः।</div><div class="line" style="position:relative;">अशोभत भृशं लक्ष्म्या पुष्पाढ्य इव किंशुकः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3866"><div class="verse-block verse-justify"><div class="line">कुन्तिभोजं ततो रक्षो विद्ध्वा बहुभिरायसैः।</div><div class="line" style="position:relative;">अनदद्भैरवं नादं वाहिन्याः प्रमुखे तव॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3867"><div class="verse-block verse-justify"><div class="line">ततस्तौ समरे शूरौ योधयन्तौ परस्परम्।</div><div class="line" style="position:relative;">ददृशुः सर्वसैन्यानि शक्रजम्भौ यथा पुरा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3868"><div class="verse-block verse-justify"><div class="line">शकुनिं रभसं युद्धे कृतवैरं च भारत।</div><div class="line" style="position:relative;">माद्रीपुत्रौ च संरब्धौ शरैश्चार्दयतां भृशम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3869"><div class="verse-block verse-justify"><div class="line">तुमुलः स महान्राजन्प्रावर्तत जनक्षयः।</div><div class="line" style="position:relative;">त्वया सञ्जनितोऽत्यर्थं कर्णेन च विवर्धितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3870"><div class="verse-block verse-justify"><div class="line">रक्षितस्तव पुत्रैश्च क्रोधमूलो हुताशनः।</div><div class="line" style="position:relative;">य इमां पृथिवीं राजन्दग्धुं सर्वां समुद्यतः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3871"><div class="verse-block verse-justify"><div class="line">शकुनिः पाण्डुपुत्राभ्यां कृतः स विमुखः शरैः।</div><div class="line" style="position:relative;">न स्म जानाति कर्तव्यं युद्धे किञ्चित्पराक्रमम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3872"><div class="verse-block verse-justify"><div class="line">विमुखं चैनमालोक्य माद्रीपुत्रौ महारथौ।</div><div class="line" style="position:relative;">ववर्षतुः पुनर्बाणैर्यथा मेघौ महागिरिम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3873"><div class="verse-block verse-justify"><div class="line">स वध्यमानो बहुभिः शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">सम्प्रायाज्जवनैरश्वैर्द्रोणानीकाय सौबलः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3874"><div class="verse-block verse-justify"><div class="line">घटोत्कचस्तथा शूरं राक्षसं तमलामुधम्।</div><div class="line" style="position:relative;">अभ्ययाद्रभसं युद्धे वेगमास्थाय मध्यमम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3875"><div class="verse-block verse-justify"><div class="line">तयोर्युद्धं महाराज चित्ररूपमिवाभवत्।</div><div class="line" style="position:relative;">यादृशं हि पुरावृत्तं रामरावणयोर्मृधे॥२९॥</div></div></div>
