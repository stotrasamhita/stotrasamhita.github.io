<div class="verse-block-wrapper" id="07-droṇaparva-b0191"><div class="verse-block verse-justify"><div class="line">ततो वादित्रघोषेण शङ्खानां च महास्वनैः।</div><div class="line" style="position:relative;">प्रादुरासीत्कृते द्रोणे हर्षः सेनापतौ तदा॥८॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b0194"><div class="verse-block verse-justify"><div class="line">सैनापत्यं तु संप्राप्य भारद्वाजो महारथः।</div><div class="line" style="position:relative;">युयुत्सुर्व्यूह्य सैन्यानि प्रायात्तव सुतैः सह॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0195"><div class="verse-block verse-justify"><div class="line">सैन्धवश्च कलिङ्गश्च विकर्णश्च तवात्मजः।</div><div class="line" style="position:relative;">दक्षिणं पार्श्वमास्थाय समतिष्ठन्त दंशिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0196"><div class="verse-block verse-justify"><div class="line">प्रपक्षः शकुनिस्तेषां प्रवरैर्हयसादिभिः।</div><div class="line" style="position:relative;">ययौ गान्धारकैः सार्धं विमलप्रासयोधिभिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0197"><div class="verse-block verse-justify"><div class="line">कृपश्च कृतवर्मा च चित्रसेनो विविंशतिः।</div><div class="line" style="position:relative;">दुःशासनमुखा यत्ताः सव्यं पक्षमपालयन्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0198"><div class="verse-block verse-justify"><div class="line">तेषां प्रपक्षाः काम्भोजाः सुदक्षिणपुस्सराः।</div><div class="line" style="position:relative;">ययुरश्चैर्महावेगैः शकाश्च यवनैः सह॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0199"><div class="verse-block verse-justify"><div class="line">मद्रास्त्रिगर्ताः साम्बष्ठाः प्रतीच्योदीच्यमालवाः।</div><div class="line" style="position:relative;">शिबयः शूरसेनाश्च शूद्राश्च मलदैः सह॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0200"><div class="verse-block verse-justify"><div class="line">सौवीराः कितवाः प्राच्या दाक्षिणात्याश्च सर्वशः।</div><div class="line" style="position:relative;">नवात्मजं पुरस्कृत्य सूतपुत्रस्य पृष्ठतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0201"><div class="verse-block verse-justify"><div class="line">हर्षयन्तः स्वसैन्यानि ययुस्तव सुतैः सह।</div><div class="line" style="position:relative;">प्रवरः सर्वयोधानां बलेषु बलमादधम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0202"><div class="verse-block verse-justify"><div class="line">ययौ वैकर्तनः कर्णः प्रमुखे सर्वधन्विनाम्।</div><div class="line" style="position:relative;">तस्य दीप्तो महाकायः स्वान्यनीकानिहर्षयन्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0203"><div class="verse-block verse-justify"><div class="line">हस्तिकक्ष्यो गहाकेतुर्बभौ सूर्यसमद्युतिः।</div><div class="line" style="position:relative;">न भीष्मव्यसनं कश्चिदृष्ट्वा कर्णममन्यत॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0204"><div class="verse-block verse-justify"><div class="line">विशोकाश्चाभवन्सर्वे राजानः कुरुभिः सह।</div><div class="line" style="position:relative;">हृष्टाश्च बहवो योधास्तत्राजल्पन्त वेगतः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0205"><div class="verse-block verse-justify"><div class="line">न हि कर्णं रणे दृष्ट्वा युधि स्थास्यन्ति पाण्डवाः।</div><div class="line" style="position:relative;">कर्णो हि समरे शक्तो जेतुं देवान्सवासवान्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0206"><div class="verse-block verse-justify"><div class="line">किमु पाण्डुसुतान्युद्धे हीनवीर्यपराक्रमान्।</div><div class="line" style="position:relative;">भीष्मेण तु रणे पार्थाः पालिता बाहुशालिना॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0207"><div class="verse-block verse-justify"><div class="line">तांस्तु कर्णः शरैस्तीक्ष्णैर्नाशयिष्यति संयुगे।</div><div class="line" style="position:relative;">एवं ब्रुवन्तस्तेऽन्योन्यं हृष्टरूपा विशांपते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0208"><div class="verse-block verse-justify"><div class="line">राधेयं पूजयन्तश्च प्रशंसन्तश्च निर्ययुः।</div><div class="line" style="position:relative;">अस्माकं शकटव्यूहों द्रोणेन विहितोऽभवत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0209"><div class="verse-block verse-justify"><div class="line">परेषां क्रौञ्च एवासीद्व्यूहो राजन्महात्मनाम्।</div><div class="line" style="position:relative;">प्रीयमाणेन विहितो धर्मराजेन भारत॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0210"><div class="verse-block verse-justify"><div class="line">व्यूहप्रमुखतस्तेषां तस्थतुः पुरुषर्षभौ।</div><div class="line" style="position:relative;">वानरध्वजमुच्छ्रित्य विष्वक्सेनधनंजयौ॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0211"><div class="verse-block verse-justify"><div class="line">ककुदं सर्वसैन्यानां धाम सर्वधनुष्मताम्।</div><div class="line" style="position:relative;">आदित्यपथगः केतुः पार्थस्यामिततेजसः॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b0213"><div class="verse-block verse-justify"><div class="line">दीप्यन्दृश्येत हि तथा केतुः सर्वत्र धीमतः।</div><div class="line" style="position:relative;">योधानामर्जुनः श्रेष्ठो गाण्डीवं धनुषां वरम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0214"><div class="verse-block verse-justify"><div class="line">वासुदेवश्च भूतानां चक्राणां च सुदर्शनम्।</div><div class="line" style="position:relative;">चत्वार्येतानि तेजाञ्सि वहञ्श्वेतहयो रथः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0215"><div class="verse-block verse-justify"><div class="line">परेषामग्रतस्तस्थौ कालचक्रमिवोद्यतम्।</div><div class="line" style="position:relative;">एवं तौ सुमहात्मानौ बलसेनाग्रगावुभौ॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0216"><div class="verse-block verse-justify"><div class="line">तावकानां मुखे कर्णः परेषां च धनंजयः।</div><div class="line" style="position:relative;">ततो जयाभिसंरब्धौ परस्परवधैषिणौ॥३३॥</div></div></div>
