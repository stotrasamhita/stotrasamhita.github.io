<div class="verse-block-wrapper" id="07-droṇaparva-b2543"><div class="verse-block verse-justify"><div class="line">नारद उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2544"><div class="verse-block verse-justify"><div class="line">साङ्कृतिं रन्तिदेवं च मृतं सृञ्जय शुश्रुम।</div><div class="line" style="position:relative;">यस्य द्विशतसाहस्रा आसन्सूदा महात्मनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2545"><div class="verse-block verse-justify"><div class="line">गृहानभ्यागतान्विप्रानतिथीन्परिवेषकाः।</div><div class="line" style="position:relative;">पक्वंपक्वं दिवारात्रं वरान्नममृतोपमम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2546"><div class="verse-block verse-justify"><div class="line">न्यायेनाधिगतं वित्तं ब्राह्मणेभ्यो ह्यमन्यत।</div><div class="line" style="position:relative;">वेदानधीत्य धर्मेण यश्चक्रे द्विषतो वशे॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2547"><div class="verse-block verse-justify"><div class="line">उपस्थिताश्च पशवः स्वयं यं शंसितव्रतम्।</div><div class="line" style="position:relative;">बहवः स्वर्गमिच्छन्तो विधिवत्सत्रयाजिनम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2548"><div class="verse-block verse-justify"><div class="line">नदी महानसाद्यस्य प्रवृतक्ता चर्मराशितः।</div><div class="line" style="position:relative;">तस्माच्चर्मण्वती नाम ख्याता पुण्या सरिद्वरा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2549"><div class="verse-block verse-justify"><div class="line">ब्राह्मणेभ्यो ददन्निष्कान्सौवर्णान्स प्रभावतः।</div><div class="line" style="position:relative;">तुभ्यं निष्कं तुभ्यं निष्कमिति ह स्म प्रभाषते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2550"><div class="verse-block verse-justify"><div class="line">तुभ्यन्तुभ्यमिति प्रादान्निष्कान्निष्कान्सहस्रशः।</div><div class="line" style="position:relative;">तत पुनः समाश्वास्य निष्कानेव प्रयच्छति॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2551"><div class="verse-block verse-justify"><div class="line">अल्पं दत्तं मयाऽद्येति निष्ककोटिं सहस्रशः।</div><div class="line" style="position:relative;">एकाह्ना ताम्यति ततः क्वाद्य विप्रा इति ब्रुवन्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2552"><div class="verse-block verse-justify"><div class="line">द्विजपाणिवियोगेन दुःखं मे शाश्वतं महत्।</div><div class="line" style="position:relative;">भविष्यति न सन्देह एवं राजाऽददद्वसु॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2553"><div class="verse-block verse-justify"><div class="line">सहस्रशश्च सौवर्णान्वृषभान्गोशतानुगान्।</div><div class="line" style="position:relative;">साष्टं शतं सुवर्णानां निष्कमाहुर्धनं तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2554"><div class="verse-block verse-justify"><div class="line">अध्यर्धमासमददद्ब्राह्मणेभ्यः शतं समाः।</div><div class="line" style="position:relative;">अग्निहोत्रोपकरणं यज्ञोपकरणं च यत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2555"><div class="verse-block verse-justify"><div class="line">ऋषिभ्यः करकान्कुम्भान्स्थालीः पिठरमेव च।</div><div class="line" style="position:relative;">शयनासनायानानि प्रासादांश्च गृहाणि च॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2556"><div class="verse-block verse-justify"><div class="line">वृक्षांश्च विविधान्दद्यादन्नानि च धनानि च।</div><div class="line" style="position:relative;">सर्वं सौवर्णमेवासीद्रन्तिदेवस्य धीमतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2557"><div class="verse-block verse-justify"><div class="line">तत्रास्य गाथा गायन्ति ये पुराणविदो जनाः।</div><div class="line" style="position:relative;">रन्तिदेवस्य तां दृष्ट्वा समृद्धिमतिमानुषीम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2558"><div class="verse-block verse-justify"><div class="line">नैतादृशं दृष्टपूर्वं कुबेरसदनेष्वपि।</div><div class="line" style="position:relative;">धनं च पूर्यमाणं नः किं पुनर्मनुजेष्विति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2559"><div class="verse-block verse-justify"><div class="line">व्यक्तं वस्वोकसारेयमित्युचुस्तत्र विस्मिताः।</div><div class="line" style="position:relative;">साङ्कृते रन्तिदेवस्य यां रात्रिमतिथिर्वसेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2560"><div class="verse-block verse-justify"><div class="line">आलभ्यन्त तदा गावः सहस्राण्येकविंशतिः।</div><div class="line" style="position:relative;">तत्र स्म सूदाः क्रोशन्ति सुमृष्टमणिकुण्डलाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2561"><div class="verse-block verse-justify"><div class="line">सूपं भूयिष्ठमश्नीध्वं नाद्य मांसं यथा पुरा।</div><div class="line" style="position:relative;">रन्तिदेवस्य यत्किञ्चित्सौवर्णमभवत्तदा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2562"><div class="verse-block verse-justify"><div class="line">तत्सर्वं वितते यज्ञे ब्राह्मणेभ्यो ह्यमन्यत।</div><div class="line" style="position:relative;">प्रत्यक्षं तस्य हव्यानि प्रतिगृह्णन्ति देवताः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2563"><div class="verse-block verse-justify"><div class="line">कव्यानि पितरः काले सर्वकामान्द्विजोत्तमाः।</div><div class="line" style="position:relative;">स चेन्ममार सृञ्जय चतुर्भद्रतरस्त्वया॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2564"><div class="verse-block verse-justify"><div class="line">पुत्रात्पुण्डतरस्तुभ्यं मा पुत्रमनुतप्यथाः।</div><div class="line" style="position:relative;">अयज्वानमदक्षिण्यमभि श्वैत्येत्युदाहरत्॥२२॥</div></div></div></div>
