<div class="verse-block-wrapper" id="07-droṇaparva-b6090"><div class="verse-block verse-justify"><div class="line">अविचिन्त्याथ तान्बाणान्कर्णेनास्तान्वृकोदरः।</div><div class="line" style="position:relative;">रणे विव्याध राधेयं शरेणानतपर्वणा॥६॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b6092"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय कर्णो भारत दुर्मनाः।</div><div class="line" style="position:relative;">इषुभिश्छादयामास मीमसेनं परन्तपः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6093"><div class="verse-block verse-justify"><div class="line">तस्य भीमो हयान्हत्वा विनिहत्य च सारथिम्।</div><div class="line" style="position:relative;">प्रजहास महाहासं कृतप्रतिकृतं महत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6094"><div class="verse-block verse-justify"><div class="line">इषुभिः कार्मुकं चास्य चकर्त पुरुषर्षभः।</div><div class="line" style="position:relative;">तत्पपात महाराज स्वर्णपृष्ठं महास्वनम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6095"><div class="verse-block verse-justify"><div class="line">अवारोहद्रथात्तस्मादथ कर्णो महारथः।</div><div class="line" style="position:relative;">गदां गृहीत्वा समरे भीमाय प्राहिणोद्रुषा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6096"><div class="verse-block verse-justify"><div class="line">तामापतन्तीमालक्ष्य भीमसेनो महागदाम्।</div><div class="line" style="position:relative;">शरैरवारयद्राजन्सर्वसैन्यस्य पश्यतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6097"><div class="verse-block verse-justify"><div class="line">ततो बाणसहस्राणि प्रेषयामास पाण्डवः।</div><div class="line" style="position:relative;">सूतपुत्रवधाकाङ्क्षी त्वरमाणः पराक्रमी॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6098"><div class="verse-block verse-justify"><div class="line">तानिषूनिषुभिः कर्णो वारयित्वा महामृधे।</div><div class="line" style="position:relative;">कवचं भीमसेनस्य पाटयामास सायकैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6099"><div class="verse-block verse-justify"><div class="line">अथैनं पञ्चविंशत्या नाराचानां समार्पयत्।</div><div class="line" style="position:relative;">पश्यतां सर्वसैन्यानां तदद्भुतमिवाभवत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6100"><div class="verse-block verse-justify"><div class="line">ततो भीमो महाबाहुर्नवभिर्नतपर्वभिः।</div><div class="line" style="position:relative;">प्रेषयामास सङ्क्रुद्धः सूतपुत्रस्य मारिष॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6101"><div class="verse-block verse-justify"><div class="line">ते तस्य कवचं भित्त्वा तथा बाहुं च दक्षिणम्।</div><div class="line" style="position:relative;">अभ्ययुर्धरणीं तीक्ष्णा वल्मीकमिव पन्नगाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6102"><div class="verse-block verse-justify"><div class="line">स च्छाद्यमानो बाणौधैर्भीमसेनधनुश्च्युतैः।</div><div class="line" style="position:relative;">पुनरेवाभवत्कर्णो भीमसेनात्पराङ्मुखः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6103"><div class="verse-block verse-justify"><div class="line">तं पराङ्मुखमालोक्य पदातिं सूतनन्दनम्।</div><div class="line" style="position:relative;">कौन्तेयशरसञ्छन्नं राजा दुर्योधनोऽब्रवीत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6104"><div class="verse-block verse-justify"><div class="line">त्वरध्वं सर्वतो यत्ता राधेयस्य रथं प्रति।</div><div class="line" style="position:relative;">ततस्तव सुता राजञ्श्रुत्वा भ्रातुर्वचो द्रुतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6105"><div class="verse-block verse-justify"><div class="line">अभ्ययुः पाण्डवं युद्धे विसृजन्तः शिलीमुखान्।</div><div class="line" style="position:relative;">चित्रोपचित्रश्चित्राक्षश्चारुचित्रः शरासनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6106"><div class="verse-block verse-justify"><div class="line">चित्रायुधश्चित्रवर्मा समरे चित्रयोधिनः।</div><div class="line" style="position:relative;">तानापतत एवाशु भीमसेनो महारथः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6107"><div class="verse-block verse-justify"><div class="line">एकैकेन शरेणाजौ पातयामास ते सुतान्।</div><div class="line" style="position:relative;">ते हता न्यपतन्भूमौ वातरूग्णा इव द्रुमाः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b6108"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा विनिहतान्पुत्रांस्तव राजन्महारथान्।</div><div class="line" style="position:relative;">अश्रुपूर्णमुखः कर्णः क्षत्तुः सस्मार तद्वचः॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b6112"><div class="verse-block verse-justify"><div class="line">सूतपुत्रोऽपि कौन्तेयं शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">पञ्चाशता महाबाहुर्विव्याध भरतर्षभ॥२८॥</div></div></div>
