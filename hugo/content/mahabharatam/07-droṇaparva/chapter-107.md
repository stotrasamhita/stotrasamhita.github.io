<div class="verse-block-wrapper" id="07-droṇaparva-b4350"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4351"><div class="verse-block verse-justify"><div class="line">बृहत्क्षत्रमथायान्तं केकयं दृढविक्रमम्।</div><div class="line" style="position:relative;">क्षेमधूर्तिर्महाराज विव्याधोरसि मार्गणैः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4352"><div class="verse-block verse-justify"><div class="line">बृहत्क्षत्रस्तु तं राजा नवत्या नतपर्वणाम्।</div><div class="line" style="position:relative;">आजघ्ने त्वरितो राजन्द्रोणानीकबिभित्सया॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4353"><div class="verse-block verse-justify"><div class="line">क्षेमधूर्तिस्तु सङ््क्रुद्धः केकयस्य महात्मनः।</div><div class="line" style="position:relative;">धनुश्चिच्छेद भल्लेन पीतेन निशितेन ह॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4354"><div class="verse-block verse-justify"><div class="line">अथैनं छिन्नधन्वानं शरेणानतपर्वणा।</div><div class="line" style="position:relative;">विव्याध समरे तूर्णं प्रवरं सर्वधन्विनाम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4355"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय बृहत्क्षत्रो हसन्निव।</div><div class="line" style="position:relative;">व्यश्वसूतरथं चक्रे क्षमदूर्तिं महारथम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4356"><div class="verse-block verse-justify"><div class="line">ततोऽपरेण भल्लेन पीतेन निशितेन च।</div><div class="line" style="position:relative;">जहार नृपतेः कायाच्छिरोऽज्वलितकुण़्डलम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4357"><div class="verse-block verse-justify"><div class="line">तच्छिन्नं सहसा तस्य शिरः कुञ्चितमूर्धजम्।</div><div class="line" style="position:relative;">सकिरीटं महीं प्राप्य बभौ ज्योतिरिवाम्बरात्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4358"><div class="verse-block verse-justify"><div class="line">तं निहत्य रणे हृष्टो बृहत्क्षत्रो महारथः।</div><div class="line" style="position:relative;">सहसाऽभ्यपतत्सैन्यं तावकं पार्थकारणात्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4359"><div class="verse-block verse-justify"><div class="line">धृष्टकेतुं तथाऽऽयान्तं द्रोणहेतोः पराक्रमी।</div><div class="line" style="position:relative;">वीरधन्वा महेष्वासो वारयामास भारत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4360"><div class="verse-block verse-justify"><div class="line">तौ परस्परमासाद्य शरदंष्ट्रौ तरस्विनौ।</div><div class="line" style="position:relative;">शरैरनेकसाहस्रेरन्योन्यमभिजघ्नतुः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4361"><div class="verse-block verse-justify"><div class="line">तावुभौ नरशार्दूलौ युयुधाते परस्परम्।</div><div class="line" style="position:relative;">महावने तीव्रमदौ वारणाविव यूथपौ॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4362"><div class="verse-block verse-justify"><div class="line">गिरिगह्वरमासाद्य शार्दूलाविव रोषितौ।</div><div class="line" style="position:relative;">युयुधाते महावीर्यौ परस्परजिघांसया॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4363"><div class="verse-block verse-justify"><div class="line">तद्युद्धमासीत्तुमुलं प्रेक्षणीयं विशाम्पते।</div><div class="line" style="position:relative;">सिद्धचारणसङ्घानां विस्मयाद्भुतदर्शनम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4364"><div class="verse-block verse-justify"><div class="line">वीरधन्वा ततः क्रुद्धो धृष्टकेतोः शरासनम्।</div><div class="line" style="position:relative;">द्विधा चिच्छेद भल्लेन प्रहसन्निव भारत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4365"><div class="verse-block verse-justify"><div class="line">तदुत्सृज्य धनुच्छिन्नं चेदिराजो महारथः।</div><div class="line" style="position:relative;">शक्तिं जग्राह विपुलां हेमदण्डामयस्मयीम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4366"><div class="verse-block verse-justify"><div class="line">तां तु शक्तिं महावीर्यां दोर्भ्यामायस्य भारत।</div><div class="line" style="position:relative;">चिक्षेप सहसा यत्तो वीरधन्वरथं प्रति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4367"><div class="verse-block verse-justify"><div class="line">तया तु वीरघातिन्या शक्त्या त्वभिहतो भृशम्।</div><div class="line" style="position:relative;">निर्भिन्नहृदयस्तूर्णं निपपात रथान्महीम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4368"><div class="verse-block verse-justify"><div class="line">तस्मिन्विनिहते वीरे त्रैगर्तानां महारथे।</div><div class="line" style="position:relative;">बलं तेऽभज्यत विभो पाण्डवेयैः समन्ततः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4369"><div class="verse-block verse-justify"><div class="line">सहदेवे ततः षष्टिं सायकान्दुर्मुखोऽक्षिपत्।</div><div class="line" style="position:relative;">ननाद च महानादं तर्जयन्पाण्डवं रणे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4370"><div class="verse-block verse-justify"><div class="line">माद्रेयस्तु ततः क्रुद्धो दुर्मुखं च शितैः शरैः।</div><div class="line" style="position:relative;">भ्राता भ्रातरमायत्तो विव्याध प्रहसन्निव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4371"><div class="verse-block verse-justify"><div class="line">तं रणे रभसं दृष्ट्वा सहदेवं महाबलम्।</div><div class="line" style="position:relative;">दुर्मुखो नवभिर्बाणैस्ताडयामास भारत॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4372"><div class="verse-block verse-justify"><div class="line">दुर्मुखस्य तु भल्लेन च्छित्त्वा केतुं महाबलः।</div><div class="line" style="position:relative;">जघान चतुरो वाहांश्चतुर्भिर्निशितैः शरैः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4373"><div class="verse-block verse-justify"><div class="line">अथापरेण भल्लेन पीतेन निशितेन ह।</div><div class="line" style="position:relative;">चिच्छेद सारथेः कायाच्छिरो ज्वलितकुण्डलम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4374"><div class="verse-block verse-justify"><div class="line">क्षुरप्रेण च तीक्ष्णेन कौरव्यस्य महद्धनुः।</div><div class="line" style="position:relative;">सहदेवो रणे छित्त्वा तं च विव्याध पञ्चभिः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4375"><div class="verse-block verse-justify"><div class="line">हताश्वं तु रथं त्यक्त्वा दुर्मुखो विमनास्तदा।</div><div class="line" style="position:relative;">आरुरोह रथं राजन्निरमित्रस्य भारत॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4376"><div class="verse-block verse-justify"><div class="line">सहदेवस्ततः क्रुद्धो निरमित्रं महाहवे।</div><div class="line" style="position:relative;">जघान पृथुधारेण भल्लेन परवीरहा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4377"><div class="verse-block verse-justify"><div class="line">स पपात रथोपस्थान्निरमित्रो जनेश्वरः।</div><div class="line" style="position:relative;">त्रिगर्तराजस्य सुतो व्यथयंस्तव वाहिनीम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4378"><div class="verse-block verse-justify"><div class="line">तं तु हत्वा महाबाहुः सहदेवो व्यरोचत।</div><div class="line" style="position:relative;">यथा दाशरथी रामः खरं हत्वा महाबलम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4379"><div class="verse-block verse-justify"><div class="line">हाहाकारो महानासीत्त्रिगर्तानां जनेश्वर।</div><div class="line" style="position:relative;">राजपुत्रं हतं दृष्ट्वा निरमित्रं महारथम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4380"><div class="verse-block verse-justify"><div class="line">नकुलस्ते सुतं राजन्विकर्णं पृथुलोचनम्।</div><div class="line" style="position:relative;">मुहूर्ताज्जितवाँल्लोके तदद्भुतमिवाभवत्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4381"><div class="verse-block verse-justify"><div class="line">सात्यकिं व्याघ्रदत्तस्तु शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">चक्रेऽदृश्यं साश्वसूतं सध्वजं पृतनान्तरे॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4382"><div class="verse-block verse-justify"><div class="line">तान्निवार्य शराञ्शूरः शैनेयः कृतहस्तवत्।</div><div class="line" style="position:relative;">साश्वसूतध्वजं बाणैर्व्याघ्रदत्तमपातयत्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4383"><div class="verse-block verse-justify"><div class="line">कुमारे निहते तस्मिन्मागधस्य सुते प्रभो।</div><div class="line" style="position:relative;">मागधाः सर्वतो यत्ता युयुधानमुपाद्रवन्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4384"><div class="verse-block verse-justify"><div class="line">विसृजन्तः शरांश्चैव तोमरांश्च सहस्रशः।</div><div class="line" style="position:relative;">मिण्डिपालांस्तथा प्रासान्मुद्रान्मुसलानपि ॥अयोधयन्रणे शूराः सात्वतं युद्धदुर्मदम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4385"><div class="verse-block verse-justify"><div class="line">तांस्तु सर्वान्स बलवान्सात्यकिर्युद्धदुर्मदः।</div><div class="line" style="position:relative;">नातिकृच्छ्राद्धसन्नेव विजिग्ये पुरुषर्षभः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4386"><div class="verse-block verse-justify"><div class="line">मागधान्द्रवतो दृष्ट्वा हतशेषान्समन्ततः।</div><div class="line" style="position:relative;">बलं तेऽभज्यत विभो युयुधानशरार्दितम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4387"><div class="verse-block verse-justify"><div class="line">नाशयित्वा रणे सैन्यं त्वदीयं तु समन्ततः।</div><div class="line" style="position:relative;">विधुन्वानो धनुःश्रेष्ठं व्यभ्राजत महायशाः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4388"><div class="verse-block verse-justify"><div class="line">भज्यमानं बलं राजन्सात्वतेन महात्मना।</div><div class="line" style="position:relative;">नाभ्यवर्तत युद्धाय त्रासितं दीर्घबाहुना॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4389"><div class="verse-block verse-justify"><div class="line">ततो द्रोणो भृशं क्रुद्धः सहसोद्वृत्य चक्षुषी।</div><div class="line" style="position:relative;">सात्यकिं सत्यकर्माणं स्वयमेवाभिदुद्रुवे॥४०॥</div></div></div></div>
