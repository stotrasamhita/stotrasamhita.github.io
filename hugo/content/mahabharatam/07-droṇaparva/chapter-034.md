<div class="verse-block-wrapper" id="07-droṇaparva-b1585"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1586"><div class="verse-block verse-justify"><div class="line">स्थिरो भव महाराज शोकं धारय दुर्धरम्।</div><div class="line" style="position:relative;">महान्तं बन्धुनाशं ते कथयिष्यामि तच्छृशु॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1587"><div class="verse-block verse-justify"><div class="line">पद्मव्यूहो महाराज आचार्येणाभिकल्पितः।</div><div class="line" style="position:relative;">तत्र पद्मोपमाः सर्वे राजानो विनिवेशिताः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1588"><div class="verse-block verse-justify"><div class="line">कुमारा राजलोकस्य निक्षिप्ताः केसरोपामाः।</div><div class="line" style="position:relative;">कर्णिकास्थो महाराज तस्य दुर्योधनोऽभवत्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1589"><div class="verse-block verse-justify"><div class="line">कृताभिसमयाः सर्वे सुवर्णविकृतध्वजाः।</div><div class="line" style="position:relative;">रक्ताम्बरधराः सर्वे सर्वे रक्तविभूषणाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1590"><div class="verse-block verse-justify"><div class="line">सर्वे रक्तपताकाश्च सर्वे वै हेममालिनः।</div><div class="line" style="position:relative;">चन्दनागुरुदिग्धाङ्गाः स्रग्विणः सूक्ष्मवाससः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1591"><div class="verse-block verse-justify"><div class="line">सहिताः पर्यधावन्त कार्ष्णि प्रतियुयुत्सवः।</div><div class="line" style="position:relative;">तेषां दशसहस्राणि बभूवुर्दृढधन्विनाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1592"><div class="verse-block verse-justify"><div class="line">पौत्रं तव पुरस्कृत्य लक्ष्मणं प्रियदर्शनम्।</div><div class="line" style="position:relative;">अन्योन्यसमदुःखास्ते अन्योन्यसमसाहसाः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1593"><div class="verse-block verse-justify"><div class="line">अन्योन्यं स्पर्धमानाश्च अन्योन्यस्य हिते रताः।</div><div class="line" style="position:relative;">दुर्योधनस्तु राजेन्द्र सैन्यमध्ये व्यवस्थितः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1594"><div class="verse-block verse-justify"><div class="line">कर्णदुःशासनकृपैर्वृतो राजा महारथैः।</div><div class="line" style="position:relative;">देवराजोपमः श्रीमाञ्श्वेतच्छत्राभिसंवृतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1595"><div class="verse-block verse-justify"><div class="line">चामरव्यजनाक्षेपैरुदयन्निव भास्करः।</div><div class="line" style="position:relative;">प्रमुखे तस्य सैन्यस्य द्रोणोऽवस्थित नायकः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1596"><div class="verse-block verse-justify"><div class="line">सिन्धुराजस्तथाऽतिष्ठच्छ्रीमान्मेरुरिवाचलः।</div><div class="line" style="position:relative;">सिन्धुराजस्य पार्श्वस्था अश्वत्थामपुरोगमाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1597"><div class="verse-block verse-justify"><div class="line">सुतास्तव महाराज त्रिंशत्त्रिदशसन्निभाः।</div><div class="line" style="position:relative;">गान्धारराजः कितवः शल्यो भूरिश्रवास्तथा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1598"><div class="verse-block verse-justify"><div class="line">पार्श्वतः सिन्धुराजस्य व्यराजन्त महारथाः।</div><div class="line">ततः प्रववृते युद्धं तुमुलं लोमहर्षणम्।</div><div class="line" style="position:relative;">तावकानां परेषां च मृत्युं कृत्वा निवर्तनम्॥१४॥</div></div></div></div>
