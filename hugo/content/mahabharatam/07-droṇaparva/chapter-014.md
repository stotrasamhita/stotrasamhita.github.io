<div class="verse-block-wrapper" id="07-droṇaparva-b0544"><div class="verse-block verse-justify"><div class="line">शोणितोदां रथावर्तां हस्त्यश्वकृतरोधसम्।</div><div class="line" style="position:relative;">कवचोडुपसंयुक्तां मांसपङ्कसमाकुलाम्॥११॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b0547"><div class="verse-block verse-justify"><div class="line">उत्तमाङ्गैः पङ्कजिनीं निस्त्रिंशझषसङ्कुलाम्।</div><div class="line" style="position:relative;">रथनागहूदोपेतां नानाभारणभूषिताम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0548"><div class="verse-block verse-justify"><div class="line">महारथशतावर्तां भूमिरेणूर्मिमालिनीम्।</div><div class="line" style="position:relative;">महावीर्यवतां सङ्ख्ये सुतरां भीरुदुस्तराम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0549"><div class="verse-block verse-justify"><div class="line">शरीरशतसम्बाधां गृध्रकङ्कनिषेविताम्।</div><div class="line" style="position:relative;">महारथसहस्राणि नयन्तीं यमसादनम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0550"><div class="verse-block verse-justify"><div class="line">शूलव्यालसमाकीर्णां प्राणिवाजिनिषेविताम्।</div><div class="line" style="position:relative;">छिन्नक्षत्रमहाहंसां मुकुटाण्डजसेविताम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0551"><div class="verse-block verse-justify"><div class="line">चक्रकूर्माङ्गदानक्रां शरक्षुद्रझषाकुलाम्।</div><div class="line" style="position:relative;">बकगृध्रसृगालानां घोरसङ्घैर्निषेविताम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0552"><div class="verse-block verse-justify"><div class="line">निहतान्प्राणिनः सङ्ख्ये द्रोणेन बलिना रणे।</div><div class="line" style="position:relative;">वहन्तीं पितृलोकाय शतशो राजसत्तम॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0553"><div class="verse-block verse-justify"><div class="line">शऱीरशतसम्बाधां केशशैवलशाद्वलाम्।</div><div class="line" style="position:relative;">नदीं प्रावर्तयद्राजन्भीरूपणां भयवर्धिनीम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0554"><div class="verse-block verse-justify"><div class="line">तर्जयन्तमनीकानि तानि तानि महारथम्।</div><div class="line" style="position:relative;">सर्वतोऽभ्यद्रवन्द्रोणं युधिष्ठिरपुरोगमाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0555"><div class="verse-block verse-justify"><div class="line">तानभिद्रवतः शूरांस्तावका दृढविक्रमाः।</div><div class="line" style="position:relative;">सर्वतः प्रत्यगृह्णन्त तदभूद्रोमहर्षणम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0556"><div class="verse-block verse-justify"><div class="line">शतमायस्तु शकुनिः सहदेवं समाद्रवत्।</div><div class="line" style="position:relative;">सनियन्तृध्वजरथं विव्याघ निशितैः शरैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0557"><div class="verse-block verse-justify"><div class="line">तस्य माद्रीसुतः केतुं धनुः सूतं हयानपि।</div><div class="line">नातिक्रुद्धः शरैश्छित्त्वा षष्ठ्या विव्याध सौबलम्।</div><div class="line" style="position:relative;">`भित्त्वा च शरवर्षेण शकुनिं प्रत्यवारयत्'॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0558"><div class="verse-block verse-justify"><div class="line">गदां गृहीत्वा शकुनिः प्रचस्कन्द रथोत्तमात्।</div><div class="line" style="position:relative;">स तस्य गदया राजन्रथात्सूतमपातयत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0559"><div class="verse-block verse-justify"><div class="line">ततस्तौ विरथौ राजन्गदाहस्तौ महाबलौ।</div><div class="line" style="position:relative;">चिक्रीडतू रणे शूरौ सशृङ्गाविव पर्वतौ॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b0560"><div class="verse-block verse-justify"><div class="line">द्रोणः पाञ्चालदायादं विव्याध निशितैः शरैः।</div><div class="line">तयोस्तत्र महाराज बाणवर्षैः प्रकाशितम्।</div><div class="line" style="position:relative;">खद्योतैरिव चाकाशं प्रदोषे पुरुषर्षभ॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b0562"><div class="verse-block verse-justify"><div class="line">विविंशतिस्तु सहसा व्यश्वकेतुशरासनम्।</div><div class="line" style="position:relative;">भीमं चक्रे महाराज ततः सैन्यान्यपूजयन्॥२८॥</div></div></div>
