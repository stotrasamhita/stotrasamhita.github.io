<div class="verse-block-wrapper" id="07-droṇaparva-b4534"><div class="verse-block verse-justify"><div class="line">भारद्वाजो रणे विद्धो युयुधानेन सत्वरम्।</div><div class="line" style="position:relative;">सात्यकिं बहुभिर्बाणैर्यतमानमविध्यत॥९॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b4536"><div class="verse-block verse-justify"><div class="line">स वध्यमानः समरे भारद्वाजेन सात्यकिः।</div><div class="line" style="position:relative;">नान्वपद्यत कर्तव्यं किञ्चिदेव विशाम्पते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4537"><div class="verse-block verse-justify"><div class="line">विषण्णवदनश्चापि युयुधानोऽभवन्नृप।</div><div class="line" style="position:relative;">भारद्वाजं रणे दृष्ट्वा विसृजन्तं शिताञ्शरान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4538"><div class="verse-block verse-justify"><div class="line">तं तु सम्प्रेक्ष्य ते पुत्राः सैनिकाश्च विशाम्पते।</div><div class="line" style="position:relative;">प्रहृष्टमनसो भूत्वा सिंहवद्व्यनदन्मुहुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4539"><div class="verse-block verse-justify"><div class="line">तं श्रुत्वा निनदं घोरं पीड्यमानं च माधवम्।</div><div class="line" style="position:relative;">युधिष्ठिरोऽब्रवीद्राजा सर्वसैन्यानि भारत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4540"><div class="verse-block verse-justify"><div class="line">एष वृष्णिवरो वीरः सात्यकिः सत्यविक्रमः।</div><div class="line">ग्रस्यते युधि वीरेण भानुमानिव राहुणा।</div><div class="line" style="position:relative;">अभिद्रवत गच्छध्वं सात्यकिर्यत्र युध्यते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4541"><div class="verse-block verse-justify"><div class="line">अभिद्रवत गच्छद्वं सात्यकिर्यत्र युध्यते ॥धृष्टद्युम्नं च पाञ्चाल्यमिदमाह जनाधिपः।</div><div class="line" style="position:relative;">`द्रोणं वारय सुक्षित्रं सात्यकिं मावधीद्द्विजः'॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4542"><div class="verse-block verse-justify"><div class="line">अभिद्रव द्रुतं द्रोणं किमु तिष्ठसि पार्पत।</div><div class="line" style="position:relative;">न पश्यसि भयं द्रोणाद्धोरं नः समुपस्थितम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4543"><div class="verse-block verse-justify"><div class="line">असौ द्रोणो महेष्वासो युयुधानेन संयुगे।</div><div class="line" style="position:relative;">क्रीडते सूत्रबद्धेन पक्षिणा बालको यथा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4544"><div class="verse-block verse-justify"><div class="line">तत्रैव सर्वे गच्छन्तु भीमसेनपुरोगमाः।</div><div class="line" style="position:relative;">त्वयैव सहिताः सर्वे युयुधानरथं प्रति॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4545"><div class="verse-block verse-justify"><div class="line">पृष्ठतोऽनुगमिष्यामि त्वामहं सहसैनिकः।</div><div class="line">सात्यकिं मोक्षयस्वाद्य यमदंष्ट्रान्तरं गतम् ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4546"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा ततो राजा सर्वसैन्येन भारत।</div><div class="line" style="position:relative;">अभ्यद्रवद्रणे द्रोणं युयुधानस्य कारणात्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4547"><div class="verse-block verse-justify"><div class="line">तत्रारावो महानासीद्द्रोणमेकं युयुत्सताम्।</div><div class="line" style="position:relative;">पाण्डवानां च भद्रं ते सृञ्जयानां च सर्वशः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4548"><div class="verse-block verse-justify"><div class="line">ते समेत्य नरव्याघ्रा भारद्वाजं महारथम्।</div><div class="line" style="position:relative;">अभ्यवर्षञ्शरैस्तीक्ष्णैः कङ्कबर्हिणवाजितैः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4549"><div class="verse-block verse-justify"><div class="line">स्मयन्नेव तु तान्वीरान्द्रोणः प्रत्यग्रहीत्स्वयम्।</div><div class="line" style="position:relative;">अतिथीनागतान्यद्वत्सलिलेनासनेन च॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4550"><div class="verse-block verse-justify"><div class="line">तर्पितास्ते शरैस्तस्य भारद्वाजस्य धन्विनः।</div><div class="line" style="position:relative;">आतिथेयं गृहं प्राप्य तर्प्यन्तेऽतिथयो यथा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4551"><div class="verse-block verse-justify"><div class="line">भारद्वाजं च ते सर्वे न शेकुः प्रतिवीक्षितुम्।</div><div class="line" style="position:relative;">मध्यन्दिनमनुप्राप्तं सहस्रांशुमिव प्रभो॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4552"><div class="verse-block verse-justify"><div class="line">तांस्तु सर्वान्महेष्वासान्द्रोणः शस्त्रभृतां वरः।</div><div class="line" style="position:relative;">अतापयच्छरव्रातैर्गभस्तिभिरिवांशुमान्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4553"><div class="verse-block verse-justify"><div class="line">वध्यमाना महाराज पाण्डवाः सृञ्जयास्तथा।</div><div class="line" style="position:relative;">त्रातारं नाध्यगच्छन्त पङ्कमग्ना इव द्विपाः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4554"><div class="verse-block verse-justify"><div class="line">द्रोणस्य च व्यदृश्यन्त विसर्पन्तो महाशराः।</div><div class="line" style="position:relative;">गभस्तय इवार्कस्य प्रतपन्तः समन्ततः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4555"><div class="verse-block verse-justify"><div class="line">तस्मिन्द्रोणेन निहताः पाञ्चालाः पञ्चविंशतिः।</div><div class="line" style="position:relative;">महारथाः समाख्याता धृष्टद्युम्नस्य सम्मताः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4556"><div class="verse-block verse-justify"><div class="line">पाण्डूनां सर्वसैन्येषु पाञ्चालानां तथैव च।</div><div class="line" style="position:relative;">द्रोणं स्म ददृशुः शूरं विनिघ्नन्तं वरान्वरान्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4557"><div class="verse-block verse-justify"><div class="line">केकयानां शतं हत्वा विद्राव्य च समन्ततः।</div><div class="line" style="position:relative;">द्रोणस्तस्थौ महाराज व्यादितास्य इवान्तकः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4558"><div class="verse-block verse-justify"><div class="line">पाञ्चालान्सृञ्जयान्मत्स्यान्केकयांश्च नराधिप।</div><div class="line" style="position:relative;">द्रोणोऽजयन्महाबाहुः शतशोऽथ सहस्रशः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4559"><div class="verse-block verse-justify"><div class="line">तेषां समभवच्छब्दो विद्धानां द्रोणसायकैः।</div><div class="line" style="position:relative;">वनौकसामिवारण्ये व्याप्तानां धृमकेतुना॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4560"><div class="verse-block verse-justify"><div class="line">तत्र देवाः सगन्धर्वाः पितरश्चाब्रुवन्नृप।</div><div class="line" style="position:relative;">एते द्रवन्ति पाञ्चालाः पाण्डवाश्च ससैनिकाः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4561"><div class="verse-block verse-justify"><div class="line">तं तथा समरे द्रोणं निघ्नन्तं सौमकान्रणे।</div><div class="line" style="position:relative;">न चाप्यभिययुः केचिदपरे नैव विव्यधुः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b4562"><div class="verse-block verse-justify"><div class="line">वर्तमाने तथा रौद्रे तस्मिन्वीरवरक्षये।</div><div class="line" style="position:relative;">अशृणोत्सहसा पार्थः पाञ्चजन्यस्य निःस्वनम्॥३७॥</div></div></div>
