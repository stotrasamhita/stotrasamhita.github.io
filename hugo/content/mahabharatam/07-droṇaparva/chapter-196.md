<div class="verse-block-wrapper" id="07-droṇaparva-b9581"><div class="verse-block verse-justify"><div class="line">युद्धेष्वपि प्रवृत्तानां ध्रुवं जयपराजयौ।</div><div class="line" style="position:relative;">द्वयमेतद्भवेद्राजन्वधस्तत्र प्रशस्यते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9582"><div class="verse-block verse-justify"><div class="line">न्यायवृत्तो वधो यस्तु संङ्ग्रामे युध्यतो भवेत्।</div><div class="line" style="position:relative;">न स दुःखाय भवति तथा दृष्टो हि स द्विजैः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9583"><div class="verse-block verse-justify"><div class="line">गतः स वीरलोकाय पिता मम न संशयः।</div><div class="line" style="position:relative;">न शोच्यः पुरुषव्याघ्र यस्तदा निधनं गतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9584"><div class="verse-block verse-justify"><div class="line">यत्तु धर्मप्रवृतः सन्केशग्रहणमाप्तवान्।</div><div class="line" style="position:relative;">पश्यतां सर्वसैन्यानां तन्मे मर्माणि कृन्तति॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9585"><div class="verse-block verse-justify"><div class="line">मयि जीवति यत्तातः केशग्रहमवाप्तवान्।</div><div class="line" style="position:relative;">कथमन्ये करिष्यन्ति पुत्रेभ्यः पुत्रिणः स्पृहाम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9586"><div class="verse-block verse-justify"><div class="line">कामात्क्रोधादविज्ञानाद्वर्षाद्बाल्येन वा पुनः।</div><div class="line" style="position:relative;">विधर्मकाणि कुर्वन्ति तथा परिभवन्ति च॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9587"><div class="verse-block verse-justify"><div class="line">तदिदं पार्षतेनेह महदाधर्मिकं कृतम्।</div><div class="line" style="position:relative;">अवज्ञाय च मां नूनं नृशंसेन दुरात्मना॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9588"><div class="verse-block verse-justify"><div class="line">तस्यानुबन्धं द्रष्टाऽसौ धृष्टद्युम्नः सुदारुणम्।</div><div class="line" style="position:relative;">अकार्यं परमं कृत्वा मिथ्यावादी च पाण्डवः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9589"><div class="verse-block verse-justify"><div class="line">यो ह्यसौ छद्मनाऽऽचार्यं शस्त्रं सन्न्यासयत्तदा।</div><div class="line" style="position:relative;">तस्याद्य धर्मराजस्य भूमिः पास्यति शोणितम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9590"><div class="verse-block verse-justify"><div class="line">शपे सत्येन कौरव्य इष्टापूर्तेन चैव ह।</div><div class="line" style="position:relative;">अहत्वा सर्वपाञ्चालाञ्जीवेयं न कथञ्चन॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9591"><div class="verse-block verse-justify"><div class="line">सर्वोपायैर्यतिष्यामि पाञ्चालानामहं वधे।</div><div class="line" style="position:relative;">धृष्टद्युम्नं च समरे हन्ताऽहं पापकारिणम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9592"><div class="verse-block verse-justify"><div class="line">कर्मणा येन तेनेह मृदुना दारुणेन च।</div><div class="line" style="position:relative;">पाञ्चालानां वधं कृत्वा शान्तिं लब्धास्मि कौरव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9593"><div class="verse-block verse-justify"><div class="line">यदर्थं पुरुषव्याघ्र पुत्रानिच्छन्ति मानवाः।</div><div class="line" style="position:relative;">प्रेत्य चेह च सम्प्राप्तात्त्रायन्ते महतो भयात्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9594"><div class="verse-block verse-justify"><div class="line">पित्रा तु मम साऽवस्था प्राप्ता निर्बन्धुना यथा।</div><div class="line" style="position:relative;">मयि शैलप्रतीकाशे पुत्रे शिष्ये च जीवति॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9595"><div class="verse-block verse-justify"><div class="line">धिङ्ममास्त्राणि दिव्यानि धिग्बाहू धिक्पराक्रमम्।</div><div class="line" style="position:relative;">यं स्म द्रोणः सुतं प्राप्य केशग्रहमवाप्तवान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9596"><div class="verse-block verse-justify"><div class="line">स तथाहं करिष्यासि यथा भरतसत्तम।</div><div class="line" style="position:relative;">परलोकगतस्यापि भविष्याम्यनृणः पितुः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9597"><div class="verse-block verse-justify"><div class="line">आर्येण हि न वक्तव्या कदाचित्स्तुतिरात्मनः।</div><div class="line" style="position:relative;">पितुर्वधममृष्यंस्तु वक्ष्याम्यद्येह पौरुषम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9598"><div class="verse-block verse-justify"><div class="line">अद्य पश्यन्तु मे वीर्यं पाण्डवाः सजनार्दनाः।</div><div class="line" style="position:relative;">मृद्गतः सर्वसैन्यानि युगान्तमिव कुर्वतः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9599"><div class="verse-block verse-justify"><div class="line">न हि देवा न गन्धर्वा नासुरा न च राक्षसाः।</div><div class="line" style="position:relative;">अद्य शक्ता रणे जेतुं रथस्थं मां नरर्षभाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9600"><div class="verse-block verse-justify"><div class="line">मदन्यो नास्ति लोकेऽस्मिन्नर्जुनाद्वाऽस्त्रवित्क्वचित्।</div><div class="line">अहं हि ज्वलतां मध्ये मयूखानामिवांशुमान्।</div><div class="line" style="position:relative;">प्रयोक्ता देवसृष्टानामस्त्राणां पृतनागतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9601"><div class="verse-block verse-justify"><div class="line">भृशमिष्वसनादद्य मत्प्रयुक्ता महाहवे।</div><div class="line" style="position:relative;">दर्शयन्तः शरा वीर्यं प्रमथिष्यन्ति पाण्डवान्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9602"><div class="verse-block verse-justify"><div class="line">अद्य सर्वा दिशो राजन्धाराभिरिव सङ्कुलाः।</div><div class="line" style="position:relative;">आवृताः पत्रिभिस्तीक्ष्णैर्द्रष्टारो मामकैरिह॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9603"><div class="verse-block verse-justify"><div class="line">विकिरञ्छरजालानि सर्वतो भैरवस्वनान्।</div><div class="line" style="position:relative;">शत्रून्निपातयिष्यामि महावात इव द्रुमान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9604"><div class="verse-block verse-justify"><div class="line">न हि जानाति बीभत्सुस्तदस्त्रं न जनार्दनः।</div><div class="line" style="position:relative;">न भीमसेनो न यमौ न च राजा युधिष्ठिरः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9605"><div class="verse-block verse-justify"><div class="line">न पार्षतो दुरात्माऽसौ न शिखण्डी न सात्यकिः।</div><div class="line" style="position:relative;">यदिदं मयि कौरव्य सकल्यं सनिवर्तनम्॥३०॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b9607"><div class="verse-block verse-justify"><div class="line">तं स्वयं प्रतिगृह्याथ भगवान्स वरं ददौ।</div><div class="line" style="position:relative;">वव्रे पिता मे परममस्त्रं नारायणं ततः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9608"><div class="verse-block verse-justify"><div class="line">अथैनमब्रवीद्राजन्भागवान्देवसत्तमः।</div><div class="line" style="position:relative;">भविता त्वत्समो नान्यः कश्चिद्युधि नरः क्वचित्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9609"><div class="verse-block verse-justify"><div class="line">न त्विदं सहसा ब्रह्मन्प्रयोक्तव्यं कथञ्चन।</div><div class="line" style="position:relative;">न ह्येतदस्त्रमन्यत्र वधाच्छत्रोर्निवर्तते॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b9610"><div class="verse-block verse-justify"><div class="line">न चैतच्छक्यते जेतुं को न वध्येत वै प्रभो।</div><div class="line" style="position:relative;">अवध्यमपि हन्याद्वि तस्मान्नैतत्प्रयोजयेत्॥३५॥</div></div></div>
