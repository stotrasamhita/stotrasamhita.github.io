<div class="verse-block-wrapper" id="07-droṇaparva-b5709"><div class="verse-block verse-justify"><div class="line">तथा तु विप्रकुर्वाणं रथयूथपयूथपम्।</div><div class="line" style="position:relative;">भारद्वाजो महाराज भीमसेनं समभ्ययात्॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5711"><div class="verse-block verse-justify"><div class="line">तद्युद्धमासीत्सुमहद्धोरं देवासुरोपमम्।</div><div class="line" style="position:relative;">द्रोणस्य च महाराज भीमस्य च महात्मनः॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5713"><div class="verse-block verse-justify"><div class="line">ततो रथादवप्लुत्य वेगमास्थाय पाण्डवः।</div><div class="line" style="position:relative;">निमील्य नयने राजन्पदातिर्द्रोणमभ्ययात्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5714"><div class="verse-block verse-justify"><div class="line">अंसे शिरो भीमसेनः करौ कृत्वोरसि स्थिरौ।</div><div class="line" style="position:relative;">वेगमास्थाय बलवान्मनोनिलगरुत्मताम्॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5716"><div class="verse-block verse-justify"><div class="line">स वध्यमानः समरे रथं द्रोणस्य मारिष।</div><div class="line" style="position:relative;">ईषायां पाणिना गृह्य प्रचिक्षेप महाबलः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5717"><div class="verse-block verse-justify"><div class="line">द्रोणस्तु सत्वरो राजन्क्षिप्तो भीमेन संयुगे।</div><div class="line">`ददृशे तावकैर्योधैर्विस्मयोत्फुल्ललोचनैः।</div><div class="line" style="position:relative;">परिहृत्य रथं द्रोणश्चूर्णितं तं महीतले'॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5718"><div class="verse-block verse-justify"><div class="line">रथमन्यं समारुह्य व्यूहद्वारं ययौ पुनः।</div><div class="line" style="position:relative;">तमायान्तं तथा दृष्ट्वा भग्नोत्साहं गुरुं तदा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5719"><div class="verse-block verse-justify"><div class="line">गत्वा वेगात्पुनर्भीमो धुरं गृह्य रथस्य तु।</div><div class="line" style="position:relative;">तमप्यतिरथं भीमश्चिक्षेप भृशरोषितः॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5722"><div class="verse-block verse-justify"><div class="line">तस्मिन्क्षणे तस्य यन्ता तूर्णमश्वानचोदयत्।</div><div class="line" style="position:relative;">भीमसेनस्य कौरव्य तदद्भुतमिवाभवत्॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5725"><div class="verse-block verse-justify"><div class="line">भोजानीकं समासाद्य हार्दिक्येनाभिरक्षितम्।</div><div class="line" style="position:relative;">प्रमथ्य तरसा वीरस्तदप्यतिबलोऽभ्ययात्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5726"><div class="verse-block verse-justify"><div class="line">सन्त्रासयन्ननीकानि तलशब्देन पाण्डवः।</div><div class="line" style="position:relative;">अजयत्सर्वसैन्यानि शार्दूल इव गोवृषान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5727"><div class="verse-block verse-justify"><div class="line">भोजानीकमतिक्रम्य दरदानां च वाहिनीम्।</div><div class="line" style="position:relative;">तथा म्लेच्छगणानन्यान्बहून्युद्धविशारदान्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5728"><div class="verse-block verse-justify"><div class="line">सात्यकिं चैव सम्प्रेक्ष्य युध्यमानं महारथम्।</div><div class="line" style="position:relative;">रथेन यत्तः कौन्तेयो वेगेन प्रययौ तदा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5729"><div class="verse-block verse-justify"><div class="line">भीमसेनो महाराज द्रष्टुकामो धनञ्जयम्।</div><div class="line" style="position:relative;">अतीत्य समरे योधांस्तावकान्पाण्डुनन्दनः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5730"><div class="verse-block verse-justify"><div class="line">सोऽपश्यदर्जुनं तत्र युध्यमानं महारथम्।</div><div class="line" style="position:relative;">सैन्धवस्य वधार्थं हि पराक्रान्तं पराक्रमी॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5731"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा पुरुषव्याघ्रश्चुक्रोश महतो रवान्।</div><div class="line" style="position:relative;">प्रावृट््काले महाराज नर्दन्निव बलाहकः॥३३॥</div></div></div>
