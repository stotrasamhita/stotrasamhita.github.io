<div class="verse-block-wrapper" id="07-droṇaparva-b7952"><div class="verse-block verse-justify"><div class="line">छिन्नधन्वा महाराज सात्यकिः क्रोधमूर्च्छितः।</div><div class="line" style="position:relative;">प्रजहार महावेगां शक्तिं तस्य महोरसि॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b7954"><div class="verse-block verse-justify"><div class="line">तं तु दृष्ट्वा हतं शूरमश्वत्थामा महारथः।</div><div class="line" style="position:relative;">अभ्यधावत वेगेन शैनेयं प्रति संयुगे॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b7956"><div class="verse-block verse-justify"><div class="line">तमापतन्तं संरब्धं शैनेयस्य रथं प्रति।</div><div class="line" style="position:relative;">घटोत्कचोऽब्रवीद्राजन्नाहं मुक्त्वा महारथः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7957"><div class="verse-block verse-justify"><div class="line">तिष्ठतिष्ठ न मे जीवन्द्रोणपुत्र गमिष्यसि।</div><div class="line" style="position:relative;">एष त्वां निहनिष्यामि महिषं मृगराडिव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7958"><div class="verse-block verse-justify"><div class="line">युद्धश्रद्धामहं तेऽद्य विनेष्यामि रणाजिरे।</div><div class="line">इत्युक्त्वा क्रोधताम्राक्षो राक्षसः परवीरहा।</div><div class="line" style="position:relative;">द्रोणिमभ्यद्रवत्क्रुद्धो गजेन्द्रमिव केसरी॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7959"><div class="verse-block verse-justify"><div class="line">रथाक्षमात्रैरिषुभिरभ्यवर्षद्धटोत्कचः।</div><div class="line" style="position:relative;">रथिनामृषभं द्रौणिं धाराभिरिव तोयदः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7960"><div class="verse-block verse-justify"><div class="line">शरवृष्टिं तु तां प्राप्तां शरैराशीविषोपमैः।</div><div class="line" style="position:relative;">शातयामास समरे तरसा द्रौणिरुत्स्मयन्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7961"><div class="verse-block verse-justify"><div class="line">ततः शरशतैस्तीक्ष्णैर्मर्मभेदिभिराशुगैः।</div><div class="line" style="position:relative;">समाचिनोद्राक्षसेन्द्रं घटोत्कचमरिन्दमम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7962"><div class="verse-block verse-justify"><div class="line">स शरैराचितस्तेन राक्षसो रणमूर्धनि।</div><div class="line" style="position:relative;">व्यकाशत महाराज श्वाविच्छललतो यथा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7963"><div class="verse-block verse-justify"><div class="line">ततः क्रोधसमाविष्टो भैमसेनिः प्रतापवान्।</div><div class="line" style="position:relative;">शरैरवचकर्तोग्रैर्द्रौणिं वज्राशनिप्रभैः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7964"><div class="verse-block verse-justify"><div class="line">क्षुरप्रैरर्धचन्द्रैश्च नाराचैः सशिलीमुखैः।</div><div class="line" style="position:relative;">वराहकर्णैर्नालीकैर्विकर्णैश्चाभ्यवीवृषत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7965"><div class="verse-block verse-justify"><div class="line">तां शस्त्रवृष्टिमतुलां वज्राशनिसमस्वनाम्।</div><div class="line" style="position:relative;">पतन्तीमुपरि क्रुद्धो द्रौणिरव्यथितेन्द्रियः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7966"><div class="verse-block verse-justify"><div class="line">सुदुःसहां शरैर्घोरैर्दिव्यास्त्रप्रतिमन्त्रितैः।</div><div class="line" style="position:relative;">व्यधमत्सुमहातेजा महाभ्राणीव मारुतः॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b7978"><div class="verse-block verse-justify"><div class="line">स भित्त्वा हृदयं तस्य राक्षसस्य शरोत्तमः।</div><div class="line" style="position:relative;">विवेश वसुधामुग्रः सपुङ्खः पृथिवीपते॥३६॥</div></div></div>
