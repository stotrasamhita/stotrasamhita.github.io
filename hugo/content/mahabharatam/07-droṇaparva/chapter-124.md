<div class="verse-block-wrapper" id="07-droṇaparva-b5435"><div class="verse-block verse-justify"><div class="line">न सन्ति तस्मात्पुत्रा मे यथा सञ्जय भाषसे।</div><div class="line" style="position:relative;">एको वै बहुलाः सेनाः प्रामृद्गात्सत्यविक्रमः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5436"><div class="verse-block verse-justify"><div class="line">कथं च युध्यमानानामपक्रान्ती महात्मनाम्।</div><div class="line">एको बहूनां शैनेयस्तन्ममाचक्ष्व सञ्जय ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5437"><div class="verse-block verse-justify"><div class="line">राजन्सेनासमुद्योगो रथनागाश्वपत्तिमान्।</div><div class="line" style="position:relative;">अतुल्यस्तव सेनायां युगान्तसदृशोऽभवत्॥९॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5439"><div class="verse-block verse-justify"><div class="line">तत्रव देवास्त्वभाषन्त चारणाश्च समागताः।</div><div class="line" style="position:relative;">एतदन्ताः समूहा वै भविष्यन्ति महीतले॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5440"><div class="verse-block verse-justify"><div class="line">न च वै तादृशो व्यूह आसीत्कश्चिद्विशाम्पते।</div><div class="line" style="position:relative;">यादृग्जयद्रथवधे द्रोणेन विहितोऽभवत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5441"><div class="verse-block verse-justify"><div class="line">`उद्धृता पृथिवी नूनं युद्धहेतोः समागतैः।</div><div class="line" style="position:relative;">इति तत्र जनाश्चाहुर्दृष्टा तां जनसंसदम्'॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5442"><div class="verse-block verse-justify"><div class="line">चण्डवातविभिन्नानां समुद्राणामिव स्वनः।</div><div class="line" style="position:relative;">रणेऽभवद्बलौघानामन्योन्यमभिधावताम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5443"><div class="verse-block verse-justify"><div class="line">पार्थिवानां समेतानां बहून्यासन्नरोत्तम।</div><div class="line" style="position:relative;">त्वद्बले प्राण्डवानां च सहस्राणि शतानि च॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5444"><div class="verse-block verse-justify"><div class="line">संरब्धानां प्रवीराणां समरे दृढकर्मणाम्।</div><div class="line" style="position:relative;">तत्रासीत्सुमहाशब्दस्तुमुलो रोमहर्षणः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5445"><div class="verse-block verse-justify"><div class="line">`पाण़्डवानां कुरूणां च गर्जतामितरेतरम्।</div><div class="line" style="position:relative;">क्ष्वेलाः किलकिलाशब्दास्तत्रासन्वै सहस्रशः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5446"><div class="verse-block verse-justify"><div class="line">भेरीशब्दाश्च तुमुला बाणशब्दाश्च भारत।</div><div class="line" style="position:relative;">अन्योन्यं निघ्नतां चैव नराणां शुश्रुवे स्वनः'॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5447"><div class="verse-block verse-justify"><div class="line">अथाक्रन्दन्भीसमेनो धृष्टद्युम्नश्च मारिष।</div><div class="line" style="position:relative;">नकुलः सहदेवश्च धर्मराजश्च पाण्डवः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5448"><div class="verse-block verse-justify"><div class="line">आगच्छत प्रहरत द्रुतं विपरिधावत।</div><div class="line" style="position:relative;">प्रविष्टावरिसेनां हि वीरौ माधवपाण्डवौ॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5449"><div class="verse-block verse-justify"><div class="line">यथा सुखेन गच्छेतां जयद्रथवधं प्रति।</div><div class="line" style="position:relative;">तथा प्रकुरुत क्षिप्रमिति सैन्यान्यचोदयन्॥२१॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5451"><div class="verse-block verse-justify"><div class="line">`यत्र यातौ महात्मानौ तूर्णं परपुरञ्जयौ'।</div><div class="line">ते यूयं सहिता भूत्वा तूर्णमेव बलार्णवम्।</div><div class="line" style="position:relative;">क्षोभयध्वं महावेगाः पवनः सागरं यथा॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5453"><div class="verse-block verse-justify"><div class="line">इच्छन्तो निधनं युद्धे शस्त्रैरुत्तमतेजसः।</div><div class="line" style="position:relative;">स्वर्गेप्सवो मित्रकार्ये नाभ्यनन्दन्त जीवितम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5454"><div class="verse-block verse-justify"><div class="line">तथैव तावका राजन्प्रार्थयन्तो महद्यशः।</div><div class="line" style="position:relative;">आर्यां युद्धे मतिं कृत्वा युद्धायैवावतस्थिरे॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5455"><div class="verse-block verse-justify"><div class="line">तस्मिन्सुतुमुले युद्धे वर्तमाने भयावहे।</div><div class="line" style="position:relative;">जित्वा सर्वाणि सैन्यानि प्रायात्सात्यकिरर्जुनं॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5456"><div class="verse-block verse-justify"><div class="line">कवचानां प्रभास्तत्र सूर्यरश्मिविराजिताः।</div><div class="line">दृष्टीः सङ्ख्ये सैनिकानां प्रतिजघ्नुः समन्ततः।</div><div class="line" style="position:relative;">`ध्वजशस्त्रप्रतिहता लोकान्समवदीपयन्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5457"><div class="verse-block verse-justify"><div class="line">चण्डवातोद्धतान्मेघान्विकिरन्रश्मिमानिव।</div><div class="line" style="position:relative;">तथा तव महत्सैन्यं तद्व्यरोचत तापयन्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5458"><div class="verse-block verse-justify"><div class="line">सम्प्रहृष्टः स सहसा तव सैन्यार्णवं प्रति।</div><div class="line" style="position:relative;">लोलयन्सर्वतो गत्वा समुद्रं मकरो यथा॥२९॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5460"><div class="verse-block verse-justify"><div class="line">रुद्रस्याक्रीडसङ्काशः संहारः सर्वदेहिनाम्।</div><div class="line" style="position:relative;">ततः शब्दो महानासीत्प्रायाद्यत्र धनञ्चयः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5461"><div class="verse-block verse-justify"><div class="line">तत्र स्म कदनं घोरं वर्तते पाण्डुपूर्वज।</div><div class="line" style="position:relative;">अर्जुनस्य महाबाहोस्तावकानां च धन्विनाम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5462"><div class="verse-block verse-justify"><div class="line">मध्ये भारतसैन्यस्य माधवस्य महारणे।</div><div class="line" style="position:relative;">द्रोणस्यापि परैः सार्धं व्यूहद्वारे सुदारुणम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5463"><div class="verse-block verse-justify"><div class="line">एवमेष क्षयो वृत्तः पृथिव्यां पृथिवीपते।</div><div class="line" style="position:relative;">क्रुद्धेऽर्जुने तथा द्रोणे सात्वते च महारथे'॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b5464"><div class="verse-block verse-justify"><div class="line">तथा प्रयतमानानां पाण्डवानां महात्मनाम्।</div><div class="line" style="position:relative;">दुर्योधनो महाराजन्व्यगाहत महद्बलम्॥३५॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5466"><div class="verse-block verse-justify"><div class="line">तथा यातेषु सैन्येषु तथा कृच्छ्रगतः स्वयम्।</div><div class="line" style="position:relative;">कच्चिद्दुर्योधनः सूत नाकार्षीत्पृष्ठतो रणम्॥३७॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b5474"><div class="verse-block verse-justify"><div class="line">शतशश्चापरान्योधान्सद्विपांश्च रथान्रणे।</div><div class="line" style="position:relative;">शरैरवचकर्तोग्रैः क्रुद्धोऽन्तक इव प्रजाः॥४५॥</div></div></div>
