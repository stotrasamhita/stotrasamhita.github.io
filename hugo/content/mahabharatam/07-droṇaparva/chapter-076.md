<div class="verse-block-wrapper" id="07-droṇaparva-b2927"><div class="verse-block verse-justify"><div class="line">अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2928"><div class="verse-block verse-justify"><div class="line">षड्रथान्धार्तराष्ट्रस्य मन्यसे यान्बलाधिकान्।</div><div class="line" style="position:relative;">तेषां वीर्यं ममार्धेन न तुल्यमिति मे मतिः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2929"><div class="verse-block verse-justify"><div class="line">अस्त्रमस्त्रेण सर्वेषामेतेषां मधुसूदन।</div><div class="line" style="position:relative;">मया द्रक्ष्यसि निर्भिन्नं जयद्रथवधैषिणा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2930"><div class="verse-block verse-justify"><div class="line">द्रोणस्य मिषतश्चाहं सगणस्य विपश्चितः।</div><div class="line" style="position:relative;">मूर्धानं सिन्धुराजस्य पातयिष्यामि भूतले॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2931"><div class="verse-block verse-justify"><div class="line">यदि साध्याश्च रुद्राश्च वसवश्च सहाश्विनः।</div><div class="line" style="position:relative;">मरुतश्च सहेन्द्रेण विश्वेदेवाः सहेश्वराः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2932"><div class="verse-block verse-justify"><div class="line">पितरः सहगन्धर्वाः सुपर्णाः सागराद्रयः।</div><div class="line" style="position:relative;">द्यौर्वियत्पृथिवी चेयं दिशश्च सदिगीश्वराः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2933"><div class="verse-block verse-justify"><div class="line">ग्राम्यारण्यानि भूतानि स्थावराणि चराणि च।</div><div class="line" style="position:relative;">त्रातारः सिन्धुराजस्य भवन्ति मधुसूदन॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2934"><div class="verse-block verse-justify"><div class="line">तथापि बाणैर्निहतं श्वो द्रष्टासि रणे मया।</div><div class="line" style="position:relative;">सत्येन च शपे कृष्ण तथैवायुधमालभे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2935"><div class="verse-block verse-justify"><div class="line">यस्य गोप्ता महेष्वासो द्रोणः पापस्य दुर्मतेः।</div><div class="line" style="position:relative;">तमेव प्रथमं द्रोणमभियास्यामि केशव॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2936"><div class="verse-block verse-justify"><div class="line">तस्मिन्युद्धमिदं बद्धं मन्यते स सुयोधनः।</div><div class="line" style="position:relative;">तस्मात्तस्यैव सेनाग्रं भित्त्वा यास्यामि सैन्धवम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2937"><div class="verse-block verse-justify"><div class="line">द्रष्टासि श्वो महेष्वासान्नाराचैस्तिग्मतेजितैः।</div><div class="line" style="position:relative;">शृङ्गाणीव गिरेर्वज्रैर्दार्यमाणान्मया युधि॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2938"><div class="verse-block verse-justify"><div class="line">नरनागाश्वदेहेभ्यो विस्रविष्यति शोणितम्।</div><div class="line" style="position:relative;">पतद्ध्यः पतितेभ्यश्च विभिन्नेभ्यः शितैः शरैः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2939"><div class="verse-block verse-justify"><div class="line">गाण्डीवप्रेषिता बाणा मनोऽनिलसमा जवे।</div><div class="line" style="position:relative;">नृनागाश्वान्विदेहासून्कर्तारश्च सहस्रशः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2940"><div class="verse-block verse-justify"><div class="line">शक्राद्भीष्मात्कृपाद्द्रोणाद्देवाद्रुद्राच्च यन्मया।</div><div class="line" style="position:relative;">उपात्तमस्त्रं घोरं तद्दष्टारोऽत्र नरा युधि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2941"><div class="verse-block verse-justify"><div class="line">ब्राह्मेणास्त्रेण चास्त्राणि हन्यमानानि संयुगे।</div><div class="line" style="position:relative;">मया द्रष्टाऽसि सर्वेषां सैन्धवस्याभिरक्षिणाम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2942"><div class="verse-block verse-justify"><div class="line">शरवेगसमुत्कृत्तै राज्ञां केशव मूर्धभिः।</div><div class="line" style="position:relative;">आस्तीर्यमाणां पृथिवीं द्रष्टाऽसि श्वो मया युधि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2943"><div class="verse-block verse-justify"><div class="line">क्रव्यादांस्तर्पयिष्यामि द्रावयिष्यामि शात्रवान्।</div><div class="line" style="position:relative;">सुहृदो नन्दयिष्यामि प्रमथिष्यामि सैन्धवम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2944"><div class="verse-block verse-justify"><div class="line">बह्वागस्कृत्कुसंबन्धी पापदेशसमुद्भवः।</div><div class="line" style="position:relative;">मया सैन्धवको राजा हतः स्वाञ्शोचयिष्यति॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2945"><div class="verse-block verse-justify"><div class="line">सर्वे क्षीरान्नभोक्तारः पापाचारा रणाजिरे।</div><div class="line" style="position:relative;">मया सराजका बाणैर्हता नश्यन्ति सैन्धवाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2946"><div class="verse-block verse-justify"><div class="line">तथा प्रभाते कर्ताऽस्मि यथा कृष्ण सुयोधनः।</div><div class="line" style="position:relative;">नान्यं धनुर्धरं लोके मंस्यते मत्समं युधि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2947"><div class="verse-block verse-justify"><div class="line">`गाण्डीवं च धनुर्दिव्यं योद्धारं च धनञ्जम्।</div><div class="line" style="position:relative;">यन्तारं च हृषीकेशं कोऽतिवर्तेत संयुगे'॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2948"><div class="verse-block verse-justify"><div class="line">तव प्रसादाद्भगवन्किंनावाप्तं रणे मम।</div><div class="line" style="position:relative;">अविषह्यं हृषीकेश किं जानन्मां विगर्हसे॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2949"><div class="verse-block verse-justify"><div class="line">यथा लक्ष्म स्थिरं चन्द्रे समुद्रे च यथा जलम्।</div><div class="line" style="position:relative;">एवमेतां प्रतिज्ञां मे सत्यां विद्धि जनार्दन॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2950"><div class="verse-block verse-justify"><div class="line">मावमंस्था ममास्त्राणि मावमंस्थाश्च गाण्डिवम्।</div><div class="line" style="position:relative;">मावमंस्था बलं बाह्वोर्मावमंस्था धनञ्जयम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2951"><div class="verse-block verse-justify"><div class="line">गाण्डीवं च धनुर्दिव्यं योद्धा चाहं नरर्षभ।</div><div class="line" style="position:relative;">त्वं च यन्ता हृषीकेश किन्नु स्यादजितं मया॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2952"><div class="verse-block verse-justify"><div class="line">यथाभियाय सङ्ग्रमं न जितो वै जयामि च।</div><div class="line" style="position:relative;">तेन सत्येन सङ्ग्रामे हतं विद्धि जयद्रथम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2953"><div class="verse-block verse-justify"><div class="line">`त्वं च माधव सर्वं तत्तथा प्रतिविधास्यसि।</div><div class="line" style="position:relative;">यथा रिपूणां मिषतां प्रमथिष्यामि सैन्धवम्'॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2954"><div class="verse-block verse-justify"><div class="line">ध्रुवं वै ब्राह्मणे सत्यं ध्रुवा साधुषु सन्नतिः।</div><div class="line">श्रीर्ध्रुवाऽपि च दक्षेषु ध्रुवो नारायणे जयः ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2955"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा हृषीकेशं स्वयमात्मानमात्मना।</div><div class="line" style="position:relative;">सन्दिदेशार्जुनो नर्दन्वासविः केशवं प्रभुम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b2956"><div class="verse-block verse-justify"><div class="line">यथा प्रभातां रजनीं कल्पितः स्याद्रथोत्तमः।</div><div class="line" style="position:relative;">तथा कार्यं त्वया कृष्ण प्रतिज्ञा स्याद्यथा ध्रुवा॥३०॥</div></div></div></div>
