<div class="verse-block-wrapper" id="07-droṇaparva-b1725"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b1727"><div class="verse-block verse-justify"><div class="line">शृणु राजन्कुमारस्य रणे विक्रीडितं महत्।</div><div class="line" style="position:relative;">बिभित्सतो रथानीकं भारद्वाजेन रक्षितम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1728"><div class="verse-block verse-justify"><div class="line">मद्रेशं सादितं दृष्ट्वा सौभद्रेणाशुगै रणे।</div><div class="line" style="position:relative;">शल्यादवरजः क्रुद्ध क्रिन्बाणान्समभ्ययात्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1729"><div class="verse-block verse-justify"><div class="line">स विद्ध्वा दशभिर्बाणैः साश्वयन्तारमार्जुनिम्।</div><div class="line" style="position:relative;">उदक्रोशन्महाशब्दं तिष्ठतिष्ठेति चाब्रवीत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1730"><div class="verse-block verse-justify"><div class="line">तस्यार्जुनिः शिरोग्रीवं पाणिपादं धनुर्हयान्।</div><div class="line" style="position:relative;">छत्रं ध्वजं नियन्तारं त्रिवेणुं चाप्युपस्करम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1731"><div class="verse-block verse-justify"><div class="line">चक्रं युगं च तूणीरं ह्यनुकर्षं सायकैः।</div><div class="line" style="position:relative;">पताकां चक्रगोप्तारौ सर्वोपकरणानि च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1732"><div class="verse-block verse-justify"><div class="line">लघुहस्तः प्रचिच्छेद ददृशे तं न कश्चन।</div><div class="line" style="position:relative;">स पपात क्षितौ क्षीणः प्रविद्धाभरणाम्बरः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1733"><div class="verse-block verse-justify"><div class="line">वायुनेव महाचैत्यः समूलं चित्रवेदिकः।</div><div class="line" style="position:relative;">अनुगास्तस्य वित्रस्ताः प्राद्रवन्सर्वतो दिशः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1734"><div class="verse-block verse-justify"><div class="line">आर्जुनेः कर्म तद्दृष्ट्वा सम्प्रणेदुः समन्ततः।</div><div class="line" style="position:relative;">नादेन सर्वभूतानि साधुसाध्विति भारत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1735"><div class="verse-block verse-justify"><div class="line">शल्यभ्रातर्यथारुग्णे बहुशस्तस्य सैनिकाः।</div><div class="line" style="position:relative;">कुलाधिवासनामानि श्रावयन्तोऽर्जुनात्मजम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1736"><div class="verse-block verse-justify"><div class="line">अभ्यधावन्त सङ्क्रुद्धा विविधायुधपाणयः।</div><div class="line" style="position:relative;">रथैरश्वैर्गजैश्चान्ये पद्भिश्चान्ये बलोत्कटाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1737"><div class="verse-block verse-justify"><div class="line">बाणशब्देन महता रथनेमिस्वनेन च।</div><div class="line" style="position:relative;">हुंकारैः क्ष्वेडितोत्कृष्टैः सिंहनादैः सगर्जितैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1738"><div class="verse-block verse-justify"><div class="line">ज्यातलत्रस्वनैरन्ये गर्जन्तोऽर्जुननन्दनम्।</div><div class="line" style="position:relative;">ब्रुवन्तश्च न नो जीवन्मोक्ष्यसे जीवितादिति॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1739"><div class="verse-block verse-justify"><div class="line">तांस्तथा ब्रुवतो दृष्ट्वा सौभद्रः प्रहसन्निव।</div><div class="line" style="position:relative;">यो योस्मै प्राहरत्पूर्वं तं तं विव्याध पत्रिभिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1740"><div class="verse-block verse-justify"><div class="line">सन्दर्शयिष्यन्नस्त्राणि विचित्राणि लघूनि च।</div><div class="line" style="position:relative;">आर्जुनिः समरे शूरो मृदुपूर्वमयुध्यत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1741"><div class="verse-block verse-justify"><div class="line">वासुदेवादुपात्तं यदस्त्रं यच्च धनञ्जयात्।</div><div class="line" style="position:relative;">अदर्शयत तत्कार्ष्णिः कृष्णाभ्यामविशेषवत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1742"><div class="verse-block verse-justify"><div class="line">दूरमस्य गुरुं भारं साध्वसं च पुनः पुनः।</div><div class="line" style="position:relative;">सन्दधद्विसृजंश्चेषून्निर्विशेषमदृश्यत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1743"><div class="verse-block verse-justify"><div class="line">चापमण्डलमेवास्य विस्फुरद्दिक्ष्वदृश्यत।</div><div class="line" style="position:relative;">सुदीप्तस्य शरत्काले सवितुर्मण्डलं यथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1744"><div class="verse-block verse-justify"><div class="line">ज्याशब्दः शुश्रुवे तस्य तलशब्दश्च दारुणः।</div><div class="line" style="position:relative;">महाशनिमुचः काले पयोदस्येव निःस्वनः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1745"><div class="verse-block verse-justify"><div class="line">हीमानमर्षी सौभद्रो मानकृत्प्रियदर्शनः।</div><div class="line" style="position:relative;">सम्मिमानयिषुर्वीरानिष्वस्त्रैश्चाप्ययुध्यत॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b1746"><div class="verse-block verse-justify"><div class="line">मृदुर्भूत्वा महाराज दारुणः समपद्यत।</div><div class="line" style="position:relative;">वर्षाभ्यतीतो भगवाञ्शरदीव दिवाकरः॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b1748"><div class="verse-block verse-justify"><div class="line">क्षुरप्रैर्वत्सदन्तैश्च विपाठेश्च महायशाः।</div><div class="line" style="position:relative;">नाराचैरर्धचन्द्राभैर्भल्लैरञ्जलिकैरपि॥२४॥</div></div></div>
