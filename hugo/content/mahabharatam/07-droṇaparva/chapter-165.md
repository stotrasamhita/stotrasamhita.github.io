<div class="verse-block-wrapper" id="07-droṇaparva-b7861"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7862"><div class="verse-block verse-justify"><div class="line">प्रकाशिते तदा लोके रजसा तमसावृते।</div><div class="line" style="position:relative;">समाजग्मुरथो वीराः परस्परवधैषिणः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7863"><div class="verse-block verse-justify"><div class="line">ते समेत्य रणे राजञ्शस्त्रप्रासासिधारिणः।</div><div class="line" style="position:relative;">परस्परमुदैक्षन्त परस्परकृतागसः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7864"><div class="verse-block verse-justify"><div class="line">प्रदीपानां सहस्रैश्च दीप्यमानैः समन्ततः।</div><div class="line" style="position:relative;">रत्नाचितैः स्वर्णदण्डैर्गन्धतैलावसिञ्चितैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7865"><div class="verse-block verse-justify"><div class="line">देवगन्धर्वदीपाद्यैः प्रभाभिरधिकोज्ज्वलैः।</div><div class="line" style="position:relative;">विरराज तदा भूमिर्ग्रहैर्द्यौरिव भारत॥५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7866"><div class="verse-block verse-justify"><div class="line">उल्काशतैः प्रज्वलितै रणभूमिर्व्यराजत।</div><div class="line" style="position:relative;">दह्यमानेव लोकानामभावे च वसुन्धरा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7867"><div class="verse-block verse-justify"><div class="line">व्यदीप्यन्त दिशः सर्वाः प्रदीपैस्तैः समन्ततः।</div><div class="line" style="position:relative;">वर्षाप्रदोषे खद्योतैर्वृता वृक्षा इवाबभुः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7868"><div class="verse-block verse-justify"><div class="line">असज्जन्त ततो वीरा विरेष्वेव पृथक्पृथक्।</div><div class="line">नागा नागैः समाजग्मुस्तुरगा हयसादिभिः।</div><div class="line" style="position:relative;">रथा रथवरैरेव समजग्मुर्मुदा युताः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7869"><div class="verse-block verse-justify"><div class="line">तस्मिन्रात्रिमुखे घोरे तव पुत्रस्य शासनात्।</div><div class="line" style="position:relative;">चतुरङ्गस्य सैन्यस्य सम्पातश्च महारनभूत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7870"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनो महाराज कौरवाणामनीकिनीम्।</div><div class="line">व्यधमत्त्वरया युक्तः क्षपयन्सर्वपार्थिवान् ॥धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7871"><div class="verse-block verse-justify"><div class="line">तस्मिन्प्रविष्टे संरब्धे मम पुत्रस्य वाहिनीम्।</div><div class="line" style="position:relative;">अमृष्यमाणे दुर्धर्षे कथमासीन्मनो हि वः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7872"><div class="verse-block verse-justify"><div class="line">किमकुर्वत सैन्यानि प्रविष्टे परषीडने।</div><div class="line" style="position:relative;">दुर्योधनश्च किं कृत्यं प्राप्तकालममन्यत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7873"><div class="verse-block verse-justify"><div class="line">के चैनं समरे वीरं प्रत्युद्ययुररिन्दमाः।</div><div class="line" style="position:relative;">द्रोणं च के व्यरक्षन्त प्रविष्टे श्वेतवाहने॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7874"><div class="verse-block verse-justify"><div class="line">केऽरक्षन्दक्षिणं चक्रं के च द्रोणस्य सव्यतः।</div><div class="line">के पृष्ठतश्चाप्यभवन्वीरा वीरान्विनिघ्नतः।</div><div class="line" style="position:relative;">के पुरस्तादगच्छन्त निघ्नन्तः शास्त्रवान्रणे॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7875"><div class="verse-block verse-justify"><div class="line">यत्प्राविशन्महेष्वासः पाञ्चालनापराजिताः।</div><div class="line" style="position:relative;">नृत्यन्निव नरव्याघ्रो रथमार्गेषु वीर्यवान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7876"><div class="verse-block verse-justify"><div class="line">यो ददाह शरैर्द्रोणः पाञ्चालानां रथव्रजान्।</div><div class="line" style="position:relative;">धूमकेतुरिव क्रुद्धः कथं मृत्युमुपेयिवान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7877"><div class="verse-block verse-justify"><div class="line">अव्यग्रानेव हि परान्कथयस्यपराजितान्।</div><div class="line" style="position:relative;">हृष्टानुदीर्णान्सङ्ग्रामे न तथा सूत मामकान्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7878"><div class="verse-block verse-justify"><div class="line">हतांश्चैव विदीर्णांश्च विप्रकीर्णांश्च शंससि।</div><div class="line">रथिनो विरथांश्चैव कृतान्युद्वेषु मामकान् ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7879"><div class="verse-block verse-justify"><div class="line">द्रोणस्य मतमाज्ञाय योद्वुकामस्य तां निशाम्।</div><div class="line" style="position:relative;">दुर्योधनो महाराज वश्यान्भ्रातॄनुवाच ह॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7880"><div class="verse-block verse-justify"><div class="line">कर्णं च वृषसेनं च मद्रराजं च कौरव।</div><div class="line" style="position:relative;">दुर्धर्षं दीर्घबाहुं च ये च तेषां पदानुगाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7881"><div class="verse-block verse-justify"><div class="line">द्रोणं यत्ताः पराक्रान्ताः सर्वे रक्षन्तु पृष्ठतः।</div><div class="line" style="position:relative;">हार्दिक्यो दक्षिणं चक्रं शल्यश्चैवोत्तरं तथा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7882"><div class="verse-block verse-justify"><div class="line">त्रिगर्तानां च ये शूरा हतशिष्टा महारथाः।</div><div class="line">तांश्चैव पुरतः सर्वान्पुत्रस्ते समचोदयत् ॥दुर्योधन उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7883"><div class="verse-block verse-justify"><div class="line">आचार्यो हि सुसंयत्तो भृशं यत्ताश्च पाण्डवाः।</div><div class="line" style="position:relative;">तं रक्षत सुसंयत्ता निघ्नन्तं शास्त्रवान्रणे॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7884"><div class="verse-block verse-justify"><div class="line">द्रोणो हि बलवान्युद्धे क्षिप्रहस्तः प्रतापवान्।</div><div class="line" style="position:relative;">निर्जयेत्त्रिदशान्युद्धे किमु पार्थान्ससोमकान्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7885"><div class="verse-block verse-justify"><div class="line">ते यूयं सहिताः सर्वे भृशं यत्ता महारथाः।</div><div class="line" style="position:relative;">द्रोणं रक्षत पाञ्चालाद्धृष्टद्युम्नान्महारथात्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7886"><div class="verse-block verse-justify"><div class="line">पाण़्डवीयेषु सैन्येषु न तं पश्याम कञ्चन।</div><div class="line" style="position:relative;">यो योधयेद्रणे द्रोणं धृष्टद्युम्नादृते पुमान्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7887"><div class="verse-block verse-justify"><div class="line">तस्मात्सर्वात्मना मन्ये भारद्वाजस्य रक्षणम्।</div><div class="line" style="position:relative;">सुगुप्तः पाण्डवान्हन्यात्सृञ्जयांश्च ससोमकान्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7888"><div class="verse-block verse-justify"><div class="line">सृञ्जयेष्वथ सर्वेषु निहतेषु चमूमुखे।</div><div class="line" style="position:relative;">धृष्टद्युम्नं रमे द्रौणिर्हनिष्यति न संशयः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7889"><div class="verse-block verse-justify"><div class="line">तथाऽर्जुनं च राधेयो हनिष्यति महारथः।</div><div class="line">भीमसेनमहं चापि युद्धेजेष्यामि दीक्षितः।</div><div class="line" style="position:relative;">शेषांश्च पाण्डवान्योधाः प्रसभं हीनतेजसः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7890"><div class="verse-block verse-justify"><div class="line">सोयं मम जयो व्यक्तो दीर्घकालं भविष्यति।</div><div class="line" style="position:relative;">तस्माद्रक्षत सङ्ग्रामे द्रोणमेव महारथम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7891"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा भरतश्रेष्ठ पुत्रो दुर्याधनस्तव।</div><div class="line" style="position:relative;">व्यादिदेश तथा सैन्यं तस्मिंस्तमसि दारुणे॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7892"><div class="verse-block verse-justify"><div class="line">ततः प्रववृते युद्धं रात्रौ भरतसत्तम।</div><div class="line" style="position:relative;">उभयोः सेनयोर्घोरं परस्परजिगीषया॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7893"><div class="verse-block verse-justify"><div class="line">अर्जुनः कौरवं सैन्यमर्जुनं चापि कौरवाः।</div><div class="line" style="position:relative;">नानाशस्त्रसमावायैरन्योन्यं समपीडयन्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7894"><div class="verse-block verse-justify"><div class="line">द्रौणिः पाञ्चालराजं च भारद्वाजश्च सृञ्जयान्।</div><div class="line" style="position:relative;">छादयांचक्रिरे सङ्ख्ये शरैः सन्नतपर्वभिः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b7895"><div class="verse-block verse-justify"><div class="line">पाण्डुपाञ्चालसैन्यानां कौरवाणां च भारत।</div><div class="line" style="position:relative;">आसीन्निष्टानको घोरो निघ्नतामितरेतरम्॥३५॥</div></div></div>
