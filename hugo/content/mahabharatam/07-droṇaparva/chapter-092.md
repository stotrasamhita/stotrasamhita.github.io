<div class="verse-block-wrapper" id="07-droṇaparva-b3564"><div class="verse-block verse-justify"><div class="line">तं द्रोणः पञ्चविंशत्या मर्मभिद्भिरजिह्मगैः।</div><div class="line" style="position:relative;">अन्तेवासिनमाचार्यो महेष्वासं समार्पयत्॥८॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b3566"><div class="verse-block verse-justify"><div class="line">तस्याशु क्षिपतो भल्लान्भल्लैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">प्रत्यविध्यदमेयात्मा ब्रह्मास्त्रं समुदीरयन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3567"><div class="verse-block verse-justify"><div class="line">तदद्भुतमपश्याम द्रोणस्याचार्यकं युधि।</div><div class="line" style="position:relative;">यतमानो युवा नैनं प्रत्यविध्यद्यदर्जुनः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3568"><div class="verse-block verse-justify"><div class="line">क्षरन्निव महामेघो वारिधाराः सहस्रशः।</div><div class="line" style="position:relative;">द्रोणमेघः पार्थशैलं ववर्ष शरवृष्टिभिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3569"><div class="verse-block verse-justify"><div class="line">अर्जुनः शरवर्षं तच्छरवर्षेण वीर्यवान्।</div><div class="line" style="position:relative;">अवारयदसम्भ्रान्तो न त्वाचार्यमपीडयत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3570"><div class="verse-block verse-justify"><div class="line">द्रोणस्तु पञ्चविंशत्या श्वेतवाहनमार्दयत्।</div><div class="line" style="position:relative;">वासुदेवं च सप्तत्या बाह्वोरुरसि चाशुगैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3571"><div class="verse-block verse-justify"><div class="line">पार्थस्तु प्रहसन्धीमानाचार्यं स शरौघिणम्।</div><div class="line" style="position:relative;">विसृजन्तं शितान्बाणानवारयत तं युधि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3572"><div class="verse-block verse-justify"><div class="line">अथ तौ वध्यमानौ तु द्रोणेन रथसत्तमौ।</div><div class="line" style="position:relative;">अवर्जयेतां दुर्धर्षं युगान्ताग्निमिवोत्थितम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3573"><div class="verse-block verse-justify"><div class="line">वर्जयन्निशितान्बाणान्द्रोणचापविनिः सृतान्।</div><div class="line" style="position:relative;">किरीटमाली कौन्तेयो भोजानीकमथाविशत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3574"><div class="verse-block verse-justify"><div class="line">सोऽन्तरा कृतवर्माणं काम्भोजं च सुदक्षिणम्।</div><div class="line" style="position:relative;">अभ्ययाद्वर्जयन्द्रोणं मैनाकमिव पर्वतम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3575"><div class="verse-block verse-justify"><div class="line">ततो भोजो नरव्याघ्रो दुर्धर्षं कुरुसत्तमम्।</div><div class="line" style="position:relative;">अविध्यत्तूर्णमव्यग्रो दशभिः कङ्कपत्रिभिः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3576"><div class="verse-block verse-justify"><div class="line">तमर्जुनः शतेनाजौ राजन्विव्याध पत्रिणाम्।</div><div class="line" style="position:relative;">पुनश्चान्यैस्त्रिभिर्बाणैर्मोहयन्निव सात्वतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3577"><div class="verse-block verse-justify"><div class="line">भोजस्तु प्रहसन्पार्थं वासुदेवं च माधवम्।</div><div class="line" style="position:relative;">एकैकं पञ्चविंशत्या सायकानां समार्पयत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3578"><div class="verse-block verse-justify"><div class="line">तस्यार्जुनो धनुश्छित्त्वा वियाधैनं त्रिसप्तभिः।</div><div class="line" style="position:relative;">शरैरग्निशिखाकारैः क्रुद्धाशीविषसन्निभैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3579"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय कृतवर्मा महारथः।</div><div class="line" style="position:relative;">पञ्चभिः सायकैस्तूर्णं विव्याधोरसि भारत॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3580"><div class="verse-block verse-justify"><div class="line">पुनश्च निशितैर्बाणैः पार्थं विव्याध पञ्चभिः।</div><div class="line" style="position:relative;">तं पार्थो नवभिर्बाणैराजघान स्तनान्तरे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3581"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा विषक्तं कौन्तेयं कृतवर्मरथं प्रति।</div><div class="line" style="position:relative;">चिन्तयामास वार्ष्णेयो न नः कालात्ययो भवेत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3582"><div class="verse-block verse-justify"><div class="line">ततः कृष्णोऽब्रवीत्पार्थं कृतवर्मणि मा दयाम्।</div><div class="line" style="position:relative;">कुरु सम्बन्धकं हित्वा प्रमथ्यैनं विशातय॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3583"><div class="verse-block verse-justify"><div class="line">ततः स कृतवर्माणं महोयित्वाऽर्जुनः शरैः।</div><div class="line" style="position:relative;">अभ्यगाज्जवनैरश्वैः काम्भोजानामनीकिनीम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3584"><div class="verse-block verse-justify"><div class="line">अमर्षितस्तु हार्दिक्यः प्रविष्टे श्वेतवाहने।</div><div class="line" style="position:relative;">विधुन्वन्सशरं चापं पाञ्चाल्याभ्यां समागतः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3585"><div class="verse-block verse-justify"><div class="line">चक्ररक्षौ तु पाञ्चाल्यावर्जुनस्य पदानुगौ।</div><div class="line" style="position:relative;">पर्यवारयदायान्तौ कृतवर्मा रथेषुभिः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3586"><div class="verse-block verse-justify"><div class="line">तावविध्यत्ततो भोजः कृतवर्मा शितैः शरैः।</div><div class="line" style="position:relative;">त्रिभिरेव युधामन्युं चतुर्भिश्चोत्तमौजसम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3587"><div class="verse-block verse-justify"><div class="line">तावप्येनं विविधतुर्दशभिर्दशभिः शरैः।</div><div class="line">त्रिभिरेव युधामन्युरुत्तगौजास्त्रिभिस्तथा।</div><div class="line" style="position:relative;">सञ्चिच्छिदतुरप्यस्य ध्वजं कार्मुकमेव च॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3588"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय हार्दिक्यः क्रोधमूर्च्छितः।</div><div class="line" style="position:relative;">कृत्वा विधनुषौ वीरौ शरवर्षैरवाकिरत्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3589"><div class="verse-block verse-justify"><div class="line">तावन्ये धनुषी सज्ये कृत्वा भोजं विजघ्नतुः।</div><div class="line" style="position:relative;">तेनान्तरेषु बीभत्सुर्विवेशामित्रवाहिनीम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3590"><div class="verse-block verse-justify"><div class="line">न लेभाते तु तौ द्वारं वारितौ कृतवर्मणा।</div><div class="line" style="position:relative;">धार्तराष्ट्रेष्वनीकेषु यतमानौ नरर्षभौ॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="07-droṇaparva-b3591"><div class="verse-block verse-justify"><div class="line">अनीकान्यर्दयन्युद्धे त्वरितः श्वेतवाहनः।</div><div class="line" style="position:relative;">नावधीत्कृतवर्माणं प्राप्तमप्यरिसूदनः॥३५॥</div></div></div>

<div class="verse-block-wrapper" id="07-droṇaparva-b3593"><div class="verse-block verse-justify"><div class="line">स पार्थं त्रिभिरानर्च्छत्सप्तत्या च जनार्दनम्।</div><div class="line" style="position:relative;">क्षुरप्रेण सुतीक्ष्णेन पार्थकेतुमताडयत्॥३७॥</div></div></div>
