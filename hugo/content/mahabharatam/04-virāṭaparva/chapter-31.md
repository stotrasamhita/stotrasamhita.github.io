<div class="verse-block-wrapper" id="04-virāṭaparva-b1695"><div class="verse-block verse-justify"><div class="line">सर्वोपायैर्यतस्व त्वं यथा पश्यसि पाण्डवान्।</div><div class="line" style="position:relative;">दुर्ज्ञेयाः खलु शूरास्ते रक्ष्या नित्यं च दैवतैः॥९॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1697"><div class="verse-block verse-justify"><div class="line">तस्माद् यत्नश्च क्रियतां भूयश्च मृगयामहे।</div><div class="line" style="position:relative;">ब्राह्मणैश्चारकैः सिद्धैस्तापसैर्निपुणैरपि॥११॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1698"><div class="verse-block verse-justify"><div class="line">विविधैस्तत्परैः सम्यङ्निर्भीकैस्तज्ज्ञसम्मतैः।</div><div class="line" style="position:relative;">अन्वेष्टव्या मनुष्येन्द्र पाण्डवाश्छन्नचारिणः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1699"><div class="verse-block verse-justify"><div class="line">ततः शान्तनवो धीमान् भारतानां पितामहः।</div><div class="line" style="position:relative;">श्रुतवान् देशकालज्ञो नीतिमांश्च महामतिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1700"><div class="verse-block verse-justify"><div class="line">तस्मिन्नुपरते वाक्ये आचार्यस्य महात्मनः।</div><div class="line" style="position:relative;">अनन्तरमुवाचेदं वाक्यं हेत्वर्थसम्मितम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1701"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरे समायुक्तां धर्मज्ञे धर्मसंहिताम्।</div><div class="line" style="position:relative;">पाण्डवे नित्यमव्यग्रां गिरं भीष्मः समाददे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1702"><div class="verse-block verse-justify"><div class="line">असत्सु दुर्लभां नित्यं सतां चाभिमतां सदा।</div><div class="line" style="position:relative;">भीष्मस्त्वभ्यवदत् तत्र गिरं साधुभिरर्चिताम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1703"><div class="verse-block verse-justify"><div class="line">यथा नो ब्राह्मणोऽवादीदाचार्यः सर्वधर्मवित्।</div><div class="line" style="position:relative;">श्रुतवृत्तोपसम्पन्ना नाशं नाऽऽयान्ति पाण्डवाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1704"><div class="verse-block verse-justify"><div class="line">सर्वलक्षणसम्पन्नाः साधुवृत्तसमन्विताः।</div><div class="line" style="position:relative;">वृद्धानुशासने यत्ताः सत्यधर्मपरायणाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1705"><div class="verse-block verse-justify"><div class="line">समयं समयज्ञास्ते पालयन्तः शुभव्रताः।</div><div class="line" style="position:relative;">न विषीदन्ति ते पार्था उद्वहन्तः सतां धुरम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1706"><div class="verse-block verse-justify"><div class="line">तपसा चैव गुप्तास्ते स्ववीर्येण च पाण्डवाः।</div><div class="line" style="position:relative;">न नाशमभिगच्छेयुरिति मे नैष्ठिकी मतिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1707"><div class="verse-block verse-justify"><div class="line">क्षत्रधर्मरता नित्यं केशवानुगताः सदा।</div><div class="line" style="position:relative;">प्रवीरपुरुषास्ते वै महात्मानो महाबलाः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1708"><div class="verse-block verse-justify"><div class="line">तत्र बुद्धिं प्रवक्ष्यामि पाण्डवान्वेषणे शृणु।</div><div class="line" style="position:relative;">न तु नीतिः सुनीतस्य शक्यते वेदितुं परैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1709"><div class="verse-block verse-justify"><div class="line">यत् तु शक्यमिहास्माभिस्तान् वै सञ्चिन्त्य पाण्डवान्।</div><div class="line" style="position:relative;">बुद्ध्या प्रणेतुं तत् तेऽहं प्रवक्ष्यामि निबोध तत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1710"><div class="verse-block verse-justify"><div class="line">न त्वियं साधु वक्तव्या तस्य नीतिः कथञ्चन।</div><div class="line" style="position:relative;">वृद्धानुशासने तात तिष्ठतः सत्यशीलिनः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1711"><div class="verse-block verse-justify"><div class="line">अयुक्तं तु मया वक्तुं तुल्या मे कुरुपाण्डवाः।</div><div class="line">निवासं पाण्डुपुत्राणां सञ्चिन्त्य च वदाम्यहम्।</div><div class="line" style="position:relative;">बहुना किं प्रलापेन यतो धर्मस्ततो जयः॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1713"><div class="verse-block verse-justify"><div class="line">यत्र नाहं तथा मन्ये यथाऽन्ये मेनिरे जनाः।</div><div class="line" style="position:relative;">निवासं पाण्डुपुत्राणां शृणुष्वं मनुजाधिप॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1714"><div class="verse-block verse-justify"><div class="line">भ्रातृभिः सहितो वीरैः कृष्णया च महायशाः।</div><div class="line" style="position:relative;">किमर्थं स महाराजो नाऽऽत्मश्रेयो भविष्यति॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1718"><div class="verse-block verse-justify"><div class="line">यस्मिन् स राजा वसति च्छन्नः सत्त्वभृतां वरः।</div><div class="line" style="position:relative;">भविष्यन्ति नरास्तत्र रागमोहविवर्जिताः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1719"><div class="verse-block verse-justify"><div class="line">नाऽऽधयो हि महाराज न व्याधिः क्षत्रियर्षभ।</div><div class="line" style="position:relative;">पुरे जनपदे वाऽपि यत्र राजा युधिष्ठिरः॥३३॥</div></div></div>
