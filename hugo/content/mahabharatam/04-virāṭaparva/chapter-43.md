<div class="verse-block-wrapper" id="04-virāṭaparva-b2358"><div class="verse-block verse-justify"><div class="line">क्व नु ते पाण्डवाः शूराः सङ्ग्रामेष्वपराजिताः।</div><div class="line" style="position:relative;">येषामिमानि दीप्तानि श्रिया दीप्यन्ति भान्ति च॥३॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2360"><div class="verse-block verse-justify"><div class="line">धर्मशीलश्च धर्मात्मा धर्मवान् धर्मवित् सुधीः।</div><div class="line" style="position:relative;">धर्माध्यक्षो धर्मपुत्रो धर्मज्ञो धर्ममूर्तिमान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2361"><div class="verse-block verse-justify"><div class="line">धर्मनिष्ठो धर्मकर्ता धर्मगोप्ता सुधर्मकृत्।</div><div class="line" style="position:relative;">सत्यार्जवक्षमाधारो घृणी धर्मपरायणः॥६॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2364"><div class="verse-block verse-justify"><div class="line">द्रौपदी चापि पाञ्चाली स्त्रीरत्नमिति मे श्रुता।</div><div class="line" style="position:relative;">जिता चाक्षैस्तदा कृष्णा तानेवान्वगमद् वने॥९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2365"><div class="verse-block verse-justify"><div class="line" style="position:relative;">उत्सृज्य राज्यं धर्म्यं ते नः श्रूयन्ते वनं गताः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2366"><div class="verse-block verse-justify"><div class="line">पाण्डवान् यदि जानीषे क्व नु ते धर्मचारिणः।</div><div class="line" style="position:relative;">क्व नु वा निवसन्तीति सत्यं ब्रूहि बृहन्नले॥११॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2367"><div class="verse-block verse-justify"><div class="line">किमर्थमागतान्यत्र शस्त्रास्त्राणि महात्मनाम्।</div><div class="line" style="position:relative;">कथं ज्ञातानि भवता तथा मे ब्रूहि शोभने॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2369"><div class="verse-block verse-justify"><div class="line">ततः प्रहस्य बीभत्सुः कौन्तेयः श्वेतवाहनः।</div><div class="line" style="position:relative;">उवाच राजपुत्रं तमुत्तरं शृणु मे वचः॥१३॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2371"><div class="verse-block verse-justify"><div class="line">मा भैस्त्वं राजशार्दूल सर्वं ते वर्णयाम्यहम्।</div><div class="line" style="position:relative;">नात्र भेतव्यमद्यापि राजपुत्र यथा तथा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2372"><div class="verse-block verse-justify"><div class="line">वयं ते पाण्डवा नाम वनवासस्य पारगाः।</div><div class="line">अतीते द्वादशे वर्षे छन्नवासमिहोषिताः।</div><div class="line" style="position:relative;">तस्मादशङ्कितमनाः शृणुष्वावहितोत्तर॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2373"><div class="verse-block verse-justify"><div class="line">अहमस्म्यर्जुनो नाम कङ्को नाम युधिष्ठिरः।</div><div class="line">भीमसेनस्तु वललः पितुस्ते रसपाचकः।</div><div class="line" style="position:relative;">अश्वबन्धस्तु नकुलः सहदेवस्तु गोपतिः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2374"><div class="verse-block verse-justify"><div class="line">सैरन्ध्रीं द्रौपदीं विद्धि यदर्थे कीचको हतः।</div><div class="line" style="position:relative;">भीमसेनेन दुर्वृत्तः सहभ्रातृभिराहवे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2375"><div class="verse-block verse-justify"><div class="line">श्रुत्वैतद् वचनं जिष्णोर्विस्मयस्फारितेक्षणः।</div><div class="line" style="position:relative;">पश्यन्ननिमिषः पार्थं शनैर्वाचमुवाच ह॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2377"><div class="verse-block verse-justify"><div class="line">दश पार्थस्य नामानि श्रूयन्ते मे कथासु च।</div><div class="line" style="position:relative;">प्रब्रूयास्तानि यदि मे श्रद्दध्यां सर्वमेव ते॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2379"><div class="verse-block verse-justify"><div class="line">अहं तर्हि तवाऽऽचक्षे दश नामानि तानि मे।</div><div class="line">वैराटे शृणु तानि त्वं यानि पूर्वं श्रुतानि ते।</div><div class="line" style="position:relative;">ईशानो विदधे देवस्त्रिदिवस्येश्वरो दिवि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2380"><div class="verse-block verse-justify"><div class="line">अर्जुनः फल्गुनो जिष्णुः किरीटी श्वेतवाहनः।</div><div class="line">बीभत्सुर्विजयः पार्थः सव्यसाची धनञ्जयः।</div><div class="line" style="position:relative;">एतानि मम नामानि स्थापितानि सुरोत्तमैः॥२१॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2384"><div class="verse-block verse-justify"><div class="line">अर्जुनः फल्गुनः पार्थः किरीटी केन सारथे।</div><div class="line" style="position:relative;">धनञ्जयश्च केनासि शीघ्रं वद बृहन्नले॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2385"><div class="verse-block verse-justify"><div class="line">श्रुता मे तस्य वीरस्य केवला नामहेतवः।</div><div class="line">इतस्ततश्चलत्येतन्मनो मे चञ्चलं त्वयि।</div><div class="line" style="position:relative;">अर्जुनो वा भवान्नेति वद शीघ्रं बृहन्नले॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2387"><div class="verse-block verse-justify"><div class="line">सर्वाञ्जित्वा जनपदान् धनं चाऽऽच्छिद्य सर्वशः।</div><div class="line" style="position:relative;">मध्ये धनस्य तिष्ठन्तं तन्मामाहुर्धनञ्जयम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b2388"><div class="verse-block verse-justify"><div class="line">अभिप्रयामि सङ्ग्रामे यदाऽहं युद्धदुर्मदान्।</div><div class="line" style="position:relative;">अजित्वा न निवर्तेयं तेन वै विजयं विदुः॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b2392"><div class="verse-block verse-justify"><div class="line">न कुर्यां कर्म बीभत्सं युध्यमानः कदाचन।</div><div class="line" style="position:relative;">तेन देवमनुष्येषु बीभत्सुरिति मां विदुः॥३१॥</div></div></div>
