<div class="verse-block-wrapper" id="04-virāṭaparva-b1003"><div class="verse-block verse-justify"><div class="line">कीचकं मत्स्यराजेन कृतागसमनिन्दिता।</div><div class="line" style="position:relative;">नापराधानुरूपेण दण्डेन प्रतिपादितम्॥२॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1006"><div class="verse-block verse-justify"><div class="line">न राजन् राजवत् किञ्चित् समाचरसि कीचके।</div><div class="line" style="position:relative;">दस्यूनामिव ते धर्मो न सत्सु परिवर्तते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1007"><div class="verse-block verse-justify"><div class="line">न कीचकः स्वधर्मस्थो न च मत्स्यः कथञ्चन।</div><div class="line" style="position:relative;">सभासदोऽप्यधर्मज्ञा य इमं पर्युपासते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1008"><div class="verse-block verse-justify"><div class="line">न धर्मं कीचको वेत्ति राजभृत्यास्तथैव च।</div><div class="line" style="position:relative;">न राजा विनयं ब्रूते अमात्याश्च न जानते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1009"><div class="verse-block verse-justify"><div class="line">नोपालभे त्वां नृपते विराटं नृपसंसदि।</div><div class="line" style="position:relative;">नाहमेतेन युक्ता वै हन्तुं मात्स्य तवान्तिके॥८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1010"><div class="verse-block verse-justify"><div class="line">सभासदस्तु पश्यन्तु कीचकं धर्मलङ्घिनम्।</div><div class="line" style="position:relative;">विराटनृपते पश्य मामनाथामनागसम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1011"><div class="verse-block verse-justify"><div class="line" style="position:relative;">न साम फलते दुष्टे दुष्टे दण्डः प्रयुज्यते॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1012"><div class="verse-block verse-justify"><div class="line">अदण्ड्यान् दण्डयन् राजा दण्ड्यांश्चैवाप्यदण्डयन्।</div><div class="line" style="position:relative;">स राजा न भवेल्लोके राजशब्दस्य भाजनम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1013"><div class="verse-block verse-justify"><div class="line">दीनान्धकृपणाशक्तपङ्गुकुब्जजडादिकान्।</div><div class="line">अनाथबालवृद्धांश्च पुरुषान् वा स्त्रियोऽपि वा।</div><div class="line" style="position:relative;">दुष्टचोराभिभूतांश्च पालयेदवनीपतिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1014"><div class="verse-block verse-justify"><div class="line">अनाथानां च नाथः स्यादपितॄणां पिता नृपः।</div><div class="line">माता भवेदमातॄणामगुरूणां गुरुर्भवेत्।</div><div class="line" style="position:relative;">अगतीनां गती राजा नृणां राजा परायणम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1015"><div class="verse-block verse-justify"><div class="line">विशेषतः परैर्दुष्टैः परामृष्टं नरोत्तमः।</div><div class="line" style="position:relative;">स्त्रियं साध्वीमनाथां च पालयेत् स्वसुतामिव॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1016"><div class="verse-block verse-justify"><div class="line">त्वद्गृहावसतिं राजन्नेतावत् कालपर्ययम्।</div><div class="line" style="position:relative;">अधिकां त्वत्सुतायाश्च पश्य मां कीचकाहताम्॥१५॥</div></div></div>
<p class="uvacha">विराट उवाच</p>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1018"><div class="verse-block verse-justify"><div class="line">परोक्षं नाभिजानामि विग्रहं युवयोरहम्।</div><div class="line" style="position:relative;">अर्थतत्त्वमविज्ञाय किं स्यादकुशलं मम॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1021"><div class="verse-block verse-justify"><div class="line">ये च दद्युर्न याचेयुर्ब्रह्मण्याः सत्यवादिनः।</div><div class="line">येषां दुन्दुभिनिर्घोषो ज्याघोषः श्रूयते भृशम्।</div><div class="line" style="position:relative;">तेषां मां दयितां भार्यां सूतपुत्रः पदाऽवधीत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1022"><div class="verse-block verse-justify"><div class="line">तेजस्विनस्तथा क्षान्ता बलवन्तश्च मानिनः।</div><div class="line">महेष्वासा रणे शूरा गर्विता मानतत्पराः।</div><div class="line" style="position:relative;">तेषां मां मानिनीं भार्यां सूतपुत्रः पदाऽवधीत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1023"><div class="verse-block verse-justify"><div class="line">सर्वलोकमिमं हन्युर्यदि क्रुद्धा महाबलाः।</div><div class="line" style="position:relative;">तेषां मां दयितां भार्यां सूतपुत्रः पदाऽवधीत्॥२०॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1025"><div class="verse-block verse-justify"><div class="line">येषां न सदृशः कश्चिद् धनाद्यैर्भुवि मानवः।</div><div class="line" style="position:relative;">तेषां मां दयितां भार्यां सूतपुत्रः पदाऽवधीत्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1028"><div class="verse-block verse-justify"><div class="line">कथं ते सूतपुत्रेण वध्यमानां प्रियां सतीम्।</div><div class="line" style="position:relative;">मर्षयन्ति यथा क्लीबा बलवन्तोऽतितेजसः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1029"><div class="verse-block verse-justify"><div class="line">क्व नु तेषाममर्षश्च वीर्यं तेषां च तद्बलम्।</div><div class="line" style="position:relative;">न परीप्सन्ति ये भार्यां वध्यमानां दुरात्माना॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b1030"><div class="verse-block verse-justify"><div class="line">मयाऽपि शक्यं किं कर्तुं विराटे धर्मदूषणे।</div><div class="line" style="position:relative;">मां मर्षयति यः पश्यन्वध्यमानामनागसम्॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b1032"><div class="verse-block verse-justify"><div class="line">यत्र धर्मो ह्यधर्मेण सत्यं यत्रानृतेन च।</div><div class="line" style="position:relative;">हन्यते प्रेक्षमाणानां हतास्तत्र सभासदः॥२९॥</div></div></div>
