<div class="verse-block-wrapper" id="04-virāṭaparva-b0937"><div class="verse-block verse-justify"><div class="line">प्रतस्थे सा सुकेशान्ता त्वरमाणा पुनः पुनः।</div><div class="line" style="position:relative;">विलम्बमाना विवशा कीचकस्य निवेशनम्॥६॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0939"><div class="verse-block verse-justify"><div class="line">श्लक्ष्णं चोवाच वाक्यं स कीचकः काममूर्च्छितः।</div><div class="line" style="position:relative;">स्वागतं ते सुकेशान्ते सुव्युष्टा रजनी मम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0940"><div class="verse-block verse-justify"><div class="line">स्वामिनी त्वमनुप्राप्ता चिरस्य भवनं शुभे।</div><div class="line" style="position:relative;">कुरुष्व च मयि प्रीतिं वशं चोपानयस्व माम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0941"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्णीष्व मे भोगांस्त्वदर्थमुपकल्पितान्।</div><div class="line" style="position:relative;">सर्वरत्नमयीं मालां कुण्डले च हिरण्मये॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0942"><div class="verse-block verse-justify"><div class="line">वासांसि चन्दनं माल्यं धूपशुद्धां च वारुणीम्।</div><div class="line">प्रतिगृह्णीष्व भद्रं ते विहर त्वं यथेच्छसि।</div><div class="line" style="position:relative;">प्रीत्या मे कुरु पद्माक्षि प्रसादं प्रियदर्शने॥११॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0943"><div class="verse-block verse-justify"><div class="line">स्वास्तीर्णमस्ति शयनं सितसूक्ष्मोत्तरच्छदम्।</div><div class="line" style="position:relative;">अत्राऽऽरुह्य मया सार्धं पिबेमां वरवारुणीम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0944"><div class="verse-block verse-justify"><div class="line">भजस्व मां विशालाक्षि भर्ता ते सदृशोऽस्म्यहम्।</div><div class="line" style="position:relative;">उपसर्प वरारोहे मेरुमर्कप्रभा यथा॥१३॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0946"><div class="verse-block verse-justify"><div class="line">स मूढः कीचकस्तत्र प्राप्तां राजीवलोचनाम्।</div><div class="line" style="position:relative;">अब्रवीद् द्रौपदीं दृष्ट्वा दुरात्मा ह्यात्मसम्मतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0947"><div class="verse-block verse-justify"><div class="line">कीचकेनैवमुक्ता सा द्रौपदी वरवर्णिनी।</div><div class="line" style="position:relative;">अब्रवीत् तमनाचारं नेदृशं वक्तुमर्हसि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0948"><div class="verse-block verse-justify"><div class="line">नाहं शक्या त्वया स्प्रष्टुं श्वपचेनेव ब्राह्मणी।</div><div class="line" style="position:relative;">गन्तुमिच्छसि दुर्बुद्धे गतिं दुर्गतरान्तराम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0949"><div class="verse-block verse-justify"><div class="line">यत्र गच्छन्ति बहवः परदाराभिमर्शकाः।</div><div class="line" style="position:relative;">नराः सम्भिन्नमर्यादाः कीटवच्चाशुभाश्रयाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0950"><div class="verse-block verse-justify"><div class="line">अप्रैषीन्मां सुराहारीं सुदेष्णा त्वन्निवेशनम्।</div><div class="line" style="position:relative;">तस्यै नयिष्ये मदिरां भगिनी तृषिता तव॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0951"><div class="verse-block verse-justify"><div class="line">पिपासिता च कैकेयी तूर्णं मामादिशत् ततः।</div><div class="line" style="position:relative;">दीयतां मे सुरा शीघ्रं सूतपुत्र व्रजाम्यहम्॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0953"><div class="verse-block verse-justify"><div class="line">अन्या भद्रे हरिष्यन्ति राजपुत्र्याः सुरामिमाम्।</div><div class="line" style="position:relative;">किं त्वं यास्यसि कल्याणि मदर्थं त्वमिहागता॥२०॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0956"><div class="verse-block verse-justify"><div class="line">तां कीचकः प्रधावन्तीं केशपक्षे परामृशत्।</div><div class="line" style="position:relative;">पातयित्वा तु तां भूमौ सूतपुत्रः पदाऽवधीत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0957"><div class="verse-block verse-justify"><div class="line">सभायां पश्यतो राज्ञो विराटस्य महात्मनः।</div><div class="line" style="position:relative;">ब्राह्मणानां च वृद्धानां क्षत्रियाणां च पश्यताम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0958"><div class="verse-block verse-justify"><div class="line" style="position:relative;">तस्याः पादाभितप्ताया मुखाद् रुधिरमास्रवत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0959"><div class="verse-block verse-justify"><div class="line">ततो दिवाकरेणाऽऽशु राक्षसः सन्नियोजितः।</div><div class="line" style="position:relative;">स कीचकमपोवाह वातवेगेन भारत॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0960"><div class="verse-block verse-justify"><div class="line">स पपात तदा भूमौ रक्षोबलसमीरितः।</div><div class="line" style="position:relative;">विघूर्णमानो निश्चेष्टश्छिन्नमूल इव द्रुमः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0961"><div class="verse-block verse-justify"><div class="line">तां दृष्ट्वा तत्र ते सभ्या हाहाभूताः समन्ततः।</div><div class="line">न युक्तं सूतपुत्रेति कीचकेति च तेऽवदन्।</div><div class="line" style="position:relative;">किमियं वध्यते बाला कृपणा चाप्यबान्धवा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0962"><div class="verse-block verse-justify"><div class="line">तस्यामासन् हि ते पार्थाः सभायां भ्रातरस्तथा।</div><div class="line" style="position:relative;">अमृष्यमाणाः कृष्णायाः कीचकेन पदा वधम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0963"><div class="verse-block verse-justify"><div class="line">तां दृष्ट्वा भीमसेनस्य क्रोधादास्रमवर्तत।</div><div class="line">धूमोच्छ्वासः समभवन्नेत्रे चोच्छ्रितपक्ष्मणी।</div><div class="line" style="position:relative;">सस्वेदा भ्रुकुटी चोग्रा ललाटे समवर्तत॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0964"><div class="verse-block verse-justify"><div class="line">तस्य भीमो वधप्रेप्सुः कीचकस्य दुरात्मनः।</div><div class="line" style="position:relative;">दन्तैर्दन्तांस्तदा रोषान्निष्पिपेष महामनाः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0965"><div class="verse-block verse-justify"><div class="line">भूयः सञ्चरितः क्रुद्धः सहसोत्थाय चाऽऽसनात्।</div><div class="line" style="position:relative;">निरैक्षत द्रुमं दीर्घं राजानं चाप्यवैक्षत॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0966"><div class="verse-block verse-justify"><div class="line">वधमाकाङ्क्षमाणं तं कीचकस्य दुरात्मनः।</div><div class="line" style="position:relative;">आकारेणैव भीमं स प्रत्यपेधद् युधिष्ठिरः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0967"><div class="verse-block verse-justify"><div class="line">तस्य राजा शनैः संज्ञां कुन्तीपुत्रो युधिष्ठिरः।</div><div class="line" style="position:relative;">चकार भीमसेनस्य रोषाविष्टस्य धीमतः॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0983"><div class="verse-block verse-justify"><div class="line">त्वत्समक्षं नृपश्रेष्ठ निष्पिष्टा वसुधातले।</div><div class="line" style="position:relative;">अनागसं कृपार्हां मां स्त्रियं त्वं परिपालय॥४९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0984"><div class="verse-block verse-justify"><div class="line">रक्ष मां कीचकाद् भीतां धर्मं रक्ष नरेश्वर।</div><div class="line" style="position:relative;">मत्स्याधिप प्रजा रक्ष पिता पुत्रानिवौरसान्॥५०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0985"><div class="verse-block verse-justify"><div class="line">यस्त्वधर्मेण कार्याणि मोहात्मा कुरुते नृपः।</div><div class="line" style="position:relative;">अचिरात् तं दुरात्मानं वशे कुर्वन्ति शत्रवः॥५१॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0986"><div class="verse-block verse-justify"><div class="line">मत्स्यानां कुलजस्त्वं हि तेषां सत्यं परायणम्।</div><div class="line" style="position:relative;">त्वं किलैवंविधो जातः कुले धर्मपरायणे॥५२॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0987"><div class="verse-block verse-justify"><div class="line">अतस्त्वाऽहमभिक्रन्दे शरणार्थं नराधिप।</div><div class="line" style="position:relative;">त्राहि मामद्य राजेन्द्र कीचकात् पापपूरुषात्॥५३॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0988"><div class="verse-block verse-justify"><div class="line">अनाथामिह मां ज्ञात्वा कीचकः पुरुषाधमः।</div><div class="line" style="position:relative;">प्रहरत्येव नीचात्मा न तु धर्ममवेक्षते॥५४॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0989"><div class="verse-block verse-justify"><div class="line">अकार्याणामनारम्भात् कार्याणामनुपालनात्।</div><div class="line" style="position:relative;">प्रजासु ये सुवृत्तास्ते स्वर्गमायान्ति भूमिपाः॥५५॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0990"><div class="verse-block verse-justify"><div class="line">कार्याकार्यविशेषज्ञाः कामकारेण पार्थिवाः।</div><div class="line" style="position:relative;">प्रजासु किल्बिषं कृत्वा नरकं यान्त्यधोमुखाः॥५६॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0991"><div class="verse-block verse-justify"><div class="line">नैव यज्ञैर्न वा दानैर्न गुरोरुपसेवनात्।</div><div class="line" style="position:relative;">प्राप्नुवन्ति तथा धर्मं यथा कार्यानुपालनात्॥५७॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0992"><div class="verse-block verse-justify"><div class="line">अपि चेदं पुरा ब्रह्मा प्रोवाचेन्द्राय पृच्छते।</div><div class="line" style="position:relative;">द्वन्द्वं कार्यमकार्यं च लोके चाऽऽसीत् परं यथा॥५८॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0993"><div class="verse-block verse-justify"><div class="line">धर्माधर्मौ पुनर्द्वन्द्वं विनियुक्तमथापि वा।</div><div class="line" style="position:relative;">क्रियाणामक्रियाणां च प्रापणे पुण्यपापयोः॥५९॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0994"><div class="verse-block verse-justify"><div class="line">प्रजायां सृज्यमानायां पुरा ह्येतदुदाहृतम्।</div><div class="line" style="position:relative;">एतद् वो मानुषाः सम्यक् कार्यं द्वन्द्वत्रयं भुवि॥६०॥</div></div></div>
<div class="verse-block-wrapper" id="04-virāṭaparva-b0995"><div class="verse-block verse-justify"><div class="line">अस्मिन् सुनीते दुर्नीते लभते कर्मजं फलम्।</div><div class="line" style="position:relative;">कल्याणकारी कल्याणं पापकारी च पापकम्॥६१॥</div></div></div>

<div class="verse-block-wrapper" id="04-virāṭaparva-b0997"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा परं वाक्यं विससर्ज शतक्रतुम्।</div><div class="line" style="position:relative;">शक्रोऽप्यापृच्छ्य ब्रह्माणं देवराज्यमपालयत्॥६३॥</div></div></div>
