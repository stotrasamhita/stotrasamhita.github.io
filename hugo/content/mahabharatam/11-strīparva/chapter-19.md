<div class="verse-block-wrapper" id="11-strīparva-b0584"><div class="verse-block verse-justify"><div class="line">गान्धार्युवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0585"><div class="verse-block verse-justify"><div class="line">एष माधव पुत्रो मे विकर्णः पाज्ञसम्मतः।</div><div class="line" style="position:relative;">भूमौ विनिहतः शेते भीमेन शतधा कृतः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0586"><div class="verse-block verse-justify"><div class="line">गजमध्ये हतः शेते विकर्णो मधुसूदन।</div><div class="line" style="position:relative;">नीलमेघपरिक्षिप्तः शरदीव निशाकरः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0587"><div class="verse-block verse-justify"><div class="line">अस्य चापग्रहेणैव पाणिः कृतकिणो महान्।</div><div class="line" style="position:relative;">कथञ्चिच्छिद्यते गृध्रैरत्तुकामैस्तलत्रवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0588"><div class="verse-block verse-justify"><div class="line">अस्य भार्याऽऽमिषप्रेप्सून्गृध्रकाकांस्तपस्विनी।</div><div class="line" style="position:relative;">वारयत्यनिशं बाला न च शक्नोति माधव॥५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0589"><div class="verse-block verse-justify"><div class="line">युवा वृन्दारकसमो विकर्णः पुरुषर्षभ।</div><div class="line" style="position:relative;">सुखोषितः सुखार्हश्च शेते पांसुषु माधव॥६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0590"><div class="verse-block verse-justify"><div class="line">कर्णिनालीकनाराचैर्भिन्नमर्माणमाहवे।</div><div class="line" style="position:relative;">अद्यापि न जहात्येनं लक्ष्मीर्भरतसत्तमम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0591"><div class="verse-block verse-justify"><div class="line">एष सङ्ग्रामशूरणे प्रतिज्ञां पालयिष्यता।</div><div class="line" style="position:relative;">दुर्मुखोऽभिमुखः शेते हतोऽरिगणहा रणे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0592"><div class="verse-block verse-justify"><div class="line">तस्यैतद्वदनं कृष्ण श्वापदैरर्धभक्षितम्।</div><div class="line" style="position:relative;">विभात्यभ्यधिकं तात सप्तम्यामिव चन्द्रमाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0593"><div class="verse-block verse-justify"><div class="line">शूरस्य हि रणे कृष्ण पश्याननमथेदृशम्।</div><div class="line" style="position:relative;">स कथं निहतोऽमित्रैः पांसून्ग्रसति मे सुतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0594"><div class="verse-block verse-justify"><div class="line">यस्याहवमुखे सौम्य स्थाता नैवोपपद्यते।</div><div class="line" style="position:relative;">स कथं दुर्मुखोऽमित्रैर्हतो विबुधलोकजित्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0595"><div class="verse-block verse-justify"><div class="line">चित्रसेनं हतं भूमौ शयानं मधुसूदन।</div><div class="line" style="position:relative;">धार्तराष्ट्रमिमं पश्य प्रतिमानं धनुष्मताम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0596"><div class="verse-block verse-justify"><div class="line">तं चित्रमाल्याभरणं युवत्यः शोककर्शिताः।</div><div class="line" style="position:relative;">क्रव्यादसङ्घैः सहिता रुदन्त्यः पर्युपासते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0597"><div class="verse-block verse-justify"><div class="line">स्त्रीणां रुदितनिर्घोषः श्वापदानां च गर्जितम्।</div><div class="line" style="position:relative;">चित्ररूपमिदं कृष्ण विमृश्य प्रतिभाति मे॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0598"><div class="verse-block verse-justify"><div class="line">युवा वृन्दारको नित्यं प्रवरस्त्रीनिषेवितः।</div><div class="line" style="position:relative;">विविंशतिरसौ शेते ध्वस्तः पांसुषु माधव॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0599"><div class="verse-block verse-justify"><div class="line">शरसङ्कृत्तवर्माणं वीरं विशसने हतम्।</div><div class="line" style="position:relative;">परिवार्यासते गृध्राः परिविंशं विविंशतिम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0600"><div class="verse-block verse-justify"><div class="line">प्रविश्य समरे शूरः पाण्डवानामनीकिनीम्।</div><div class="line" style="position:relative;">स वीरशयने शेते परः सत्पुरुषोचिते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0601"><div class="verse-block verse-justify"><div class="line">स्मितोपपन्नं सुनसं सुभ्रु ताराधिपोपमम्।</div><div class="line" style="position:relative;">अतीव सौम्यवदनं कृष्ण पश्य विविंशतेः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0602"><div class="verse-block verse-justify"><div class="line">यं स्म तं पर्युपासन्ते बहुधा वरयोषितः।</div><div class="line" style="position:relative;">क्रीडन्तमिव गन्धर्वं देवकन्याः सहस्रशः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0603"><div class="verse-block verse-justify"><div class="line">हन्तारं परसैन्यानां शूरं समितिशोभनम्।</div><div class="line" style="position:relative;">निबर्हणममित्राणां दुःसहं विषहेत कः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0604"><div class="verse-block verse-justify"><div class="line">दुःसहस्यैतदाभाति शरीरं संवृतं शरैः।</div><div class="line" style="position:relative;">गिरिरात्मरुहैः फुल्लैः कर्णिकारैरिवाचितः॥२१॥</div></div></div>
