<div class="verse-block-wrapper" id="11-strīparva-b0409"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा वचनं तस्या भीमसेनोऽथ भीतवत्।</div><div class="line" style="position:relative;">गान्धारीं प्रत्युवाचेदं वचः सानुनयं तदा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0410"><div class="verse-block verse-justify"><div class="line">अधर्मो यदि वा धर्मस्त्रासात्तत्र मया कृतः।</div><div class="line" style="position:relative;">आत्मानं त्रातुकामेन तन्मे त्वं क्षन्तुमर्हसि॥३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0411"><div class="verse-block verse-justify"><div class="line">न हि युद्धेन पुत्रस्ते धर्मेण स महाबलः।</div><div class="line" style="position:relative;">दुःशक्यः केनचिद्धर्तुमतो विषममाचरम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0412"><div class="verse-block verse-justify"><div class="line">अधर्मेण जितः पूर्वं तेन चापि युधिष्ठिरः।</div><div class="line" style="position:relative;">निकृताश्च सदैव स्म तो विषममाचरम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0413"><div class="verse-block verse-justify"><div class="line">सैन्यस्यैकोऽवशिष्टोऽयं गदायुद्धे च वीर्यवान्।</div><div class="line" style="position:relative;">न त्यक्ष्यति हृतं राज्यमिति वै तत्कृतं मया॥६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0414"><div class="verse-block verse-justify"><div class="line">एकपत्नीं च पाश्चालीमेकवस्त्रां रजस्वलाम्।</div><div class="line" style="position:relative;">भवत्या विदितं सर्वमुक्तवान्यत्सुतस्तव॥७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0415"><div class="verse-block verse-justify"><div class="line">सुयोधनं त्वसंहृत्य न शक्या भूः ससागरा।</div><div class="line" style="position:relative;">केवला भोक्तुमस्माभिरतश्चैतत्कृतं मया॥८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0416"><div class="verse-block verse-justify"><div class="line">तच्चाप्यप्रियमस्माकं पुत्रस्ते समुपाचरत्।</div><div class="line" style="position:relative;">द्रौपद्या यत्सभामध्ये सव्यमूरुमदर्शयत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0417"><div class="verse-block verse-justify"><div class="line">तत्रैव वध्यः सोऽस्माकं दुराचारोऽप्ब ते सुतः।</div><div class="line" style="position:relative;">धर्मराजाज्ञया चैव स्थिताः स्म समये पुरा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0418"><div class="verse-block verse-justify"><div class="line">वैरमुद्दीपितं राज्ञि पुत्रेण तव यन्महत्।</div><div class="line" style="position:relative;">क्लेशिताश्च वने नित्यं तत एतत्कृतं मया॥११॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0419"><div class="verse-block verse-justify"><div class="line">वैरस्यास्य गताः पारं हत्वा दुर्योधनं रणे।</div><div class="line">राज्यं युधिष्ठिरे प्राप्ते वयं च गतमन्यवः ॥गान्धार्युवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0420"><div class="verse-block verse-justify"><div class="line">न तस्यैवं वधस्तात यं प्रशंससि मे सुतम्।</div><div class="line" style="position:relative;">कृतवांश्चापि तत्सर्वं यदिदं भाषसे मयि॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0421"><div class="verse-block verse-justify"><div class="line">हताश्वे नकुले यत्तु वृषसेनेन भारत।</div><div class="line" style="position:relative;">अपिबः शोणितं सङ्ख्ये दुःशासनशरीरजम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0422"><div class="verse-block verse-justify"><div class="line">सद्भिर्विगर्हितं घोरमनार्यजनसेवितम्।</div><div class="line">क्रूरं कर्माकृथास्तस्मात्तदयुक्तं वृकोदर ॥भीम उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0423"><div class="verse-block verse-justify"><div class="line">हताश्वं नकुलं दृष्ट्वा वृषसेनेन संयुगे।</div><div class="line" style="position:relative;">शत्रूणां तु प्रहृष्टानां त्रासः सञ्जनितो मया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0424"><div class="verse-block verse-justify"><div class="line">`स प्रतिज्ञामकरवं पिबाम्यसृगरेरिति'।</div><div class="line" style="position:relative;">रुधिरं नातिचक्राम दन्तोष्ठादम्ब मा शुचः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0425"><div class="verse-block verse-justify"><div class="line">[अन्यस्यापि न पातव्यं रुधिरं किंपुनः स्वकम्।</div><div class="line" style="position:relative;">यथैवात्मा तथा भ्राता विशेषो नास्ति कश्चन॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0426"><div class="verse-block verse-justify"><div class="line">वैवस्तितो हि तद्वेद यथावत्कुलनन्दिनि।</div><div class="line" style="position:relative;">मा कृथा हृदि तन्मातर्न तत्पीतं मयाऽनधे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0427"><div class="verse-block verse-justify"><div class="line">केशपक्षपरामर्शे द्रौपद्या द्यूतकारिते।</div><div class="line" style="position:relative;">क्रोधाद्यदब्रवं चाहं तच्च मे हृदि वर्तते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0428"><div class="verse-block verse-justify"><div class="line">क्षत्रधर्माच्च्युतो राज्ञि भवेयं शाश्वतीः समाः।</div><div class="line" style="position:relative;">प्रतिज्ञां तामनिस्तीर्य ततस्तत्कृतवानहम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0429"><div class="verse-block verse-justify"><div class="line">अनिगृह्य पुरा पुत्रानस्मस्वनपकारिषु।</div><div class="line">न ममार्हसि कल्याणि दोषेण परिशङ्कितुम् ॥गान्धार्युवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0430"><div class="verse-block verse-justify"><div class="line">वृद्धस्यास्य शतं पुत्रान्निघ्नंस्त्वमपराजितः।</div><div class="line" style="position:relative;">कस्मानाशेषयः कञ्चिद्येनाल्पमपराधितम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0431"><div class="verse-block verse-justify"><div class="line">सन्तानमावयोस्तात वृद्धयोर्हृतराज्ययोः।</div><div class="line" style="position:relative;">नाशेषयः कथं यष्टिमेकां वृद्धयुगस्य वै॥२४॥</div></div></div>
