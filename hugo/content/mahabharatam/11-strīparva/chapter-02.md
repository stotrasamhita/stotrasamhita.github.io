<div class="verse-block-wrapper" id="11-strīparva-b0046"><div class="verse-block verse-justify"><div class="line">वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0047"><div class="verse-block verse-justify"><div class="line">ततोऽमृतरसैर्वाक्यैर्ह्लादयन्पुरुषर्षभम्।</div><div class="line">वैचित्रवीर्यं विदुरो यदुवाच निबोध तत् ॥विदुर उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0048"><div class="verse-block verse-justify"><div class="line">उत्तिष्ठ राजन्किं शेषे धारयात्मानमात्मना।</div><div class="line" style="position:relative;">एषा वै सर्वसत्वानां लोकेश्वर परा गतिः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0049"><div class="verse-block verse-justify"><div class="line">सर्वे क्षयान्ता निचयाः पतनान्ताः समुच्छ्रयाः।</div><div class="line" style="position:relative;">संयोगा विप्रयोगान्ता मरणान्तं च जीवितम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0050"><div class="verse-block verse-justify"><div class="line">यदा शूरं च भीरुं च यमः कर्षति भारत।</div><div class="line" style="position:relative;">तत्किं न योत्स्यन्ति हि ते क्षत्रियाः क्षत्रियर्षभ॥५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0051"><div class="verse-block verse-justify"><div class="line">अयुध्यमानो म्रियते युध्यमानश्च जीवति।</div><div class="line" style="position:relative;">कालं प्राप्य महाराज न कश्चिदतिवर्तते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0052"><div class="verse-block verse-justify"><div class="line">अभावादीनि भूतानि भावमध्यानि भारत।</div><div class="line" style="position:relative;">अभावनिधनान्येव तत्र का परिदेवना॥७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0053"><div class="verse-block verse-justify"><div class="line">न शोचन्मृतमन्वेति न शोचन्म्रियते नरः।</div><div class="line" style="position:relative;">एवं सांसिद्धिके लोके किमर्थमनुशोचसि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0054"><div class="verse-block verse-justify"><div class="line">कालः कर्षति भूतानि सर्वाणि विविधान्युत।</div><div class="line" style="position:relative;">न कालस्य प्रियः कश्चिन्न द्वेष्यः कुरुसत्तम॥९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0055"><div class="verse-block verse-justify"><div class="line">यथा वायुस्तृणाग्राणि संवर्तयति सर्वशः।</div><div class="line" style="position:relative;">तथा कालवशं यान्ति भूतानि भरतर्षभ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0056"><div class="verse-block verse-justify"><div class="line">एकसार्थप्रयातानां सर्वेषां तत्र गामिनाम्।</div><div class="line" style="position:relative;">यस्य कालः स यात्यग्रे तत्र का परिदेवना॥११॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0057"><div class="verse-block verse-justify"><div class="line">न चाप्येतान्हतान्युद्धे राजञ्शोचितुमर्हसि।</div><div class="line" style="position:relative;">प्रमाणं यदि शास्त्राणि गतास्ते परमां गतिम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0058"><div class="verse-block verse-justify"><div class="line">सर्वे स्वाध्यायवन्तो हि सर्वे च चरितव्रताः।</div><div class="line" style="position:relative;">सर्वे चाभिमुखाः क्षीणास्तत्र का परिदेवना॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0059"><div class="verse-block verse-justify"><div class="line">अदर्शनादापतिताः पुनश्चादर्शनं गताः।</div><div class="line" style="position:relative;">नैते तव न तेषां त्वं तत्र का परिदेवना॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0060"><div class="verse-block verse-justify"><div class="line">हतो हि लभते स्वर्गं जित्वा च लभते यशः।</div><div class="line" style="position:relative;">उभयं नो बहुगुणं नास्ति निष्फलता रणे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0061"><div class="verse-block verse-justify"><div class="line">तेषां कामदुघाँल्लोकानिन्द्रः सङ्कल्पयिष्यति।</div><div class="line" style="position:relative;">इन्द्रस्यातिथयो ह्येते भवन्ति भरतर्षभ॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0062"><div class="verse-block verse-justify"><div class="line">न यज्ञैर्दक्षिणावद्भिर्न तपोभिर्न विद्यया।</div><div class="line" style="position:relative;">स्वर्गं यान्ति तथा मर्त्या यथा शूरा रणे हताः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0063"><div class="verse-block verse-justify"><div class="line">शरीराग्निषु शूराणां जुहुवुस्ते शराहुतीः।</div><div class="line" style="position:relative;">हूयमानाञ्शरांश्चैव सेहुस्तेजस्विनो मिथः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0064"><div class="verse-block verse-justify"><div class="line">एवं राजंस्तवाचक्षे स्वर्ग्यं पन्थानमुत्तमम्।</div><div class="line" style="position:relative;">न युद्धादधिकं किञ्चित्क्षत्रियस्येह विद्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0065"><div class="verse-block verse-justify"><div class="line">क्षत्रियास्ते महात्मानः शूराः समितिशोभनाः।</div><div class="line" style="position:relative;">आशिषः परमाः प्राप्ता न शोच्याः सर्व एव हि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0066"><div class="verse-block verse-justify"><div class="line">आत्मानमात्मनाऽऽश्वास्य मा शुचः पुरुषर्षभ।</div><div class="line" style="position:relative;">नाद्य शोकाभिभूतस्त्वं कायमुत्स्रष्टुमर्हसि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0067"><div class="verse-block verse-justify"><div class="line">मातापितृसहस्राणि पुत्रदारशतानि च।</div><div class="line" style="position:relative;">संसारेष्वनुभूतानि कस्य ते कस्य वा वयम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0068"><div class="verse-block verse-justify"><div class="line">शोकस्थानसहसाणि भयस्थानशतानि च।</div><div class="line" style="position:relative;">दिवसेदिवसे मूढमाविशन्ति न पण़्डितम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0069"><div class="verse-block verse-justify"><div class="line">न कालस्य प्रियः कश्चिन्न द्वेष्यः कुरुसत्तम।</div><div class="line" style="position:relative;">न मध्यस्थः क्वचित्कालः सर्वं कालः प्रकर्षति॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0070"><div class="verse-block verse-justify"><div class="line">कालः पचति भूतानि कालः संहरते प्रजाः।</div><div class="line" style="position:relative;">कालः सुप्तेषु जागर्ति कालो हि दुरतिक्रमः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0071"><div class="verse-block verse-justify"><div class="line">अनित्यं यौवनं रूपं जीवितं द्रव्यसञ्चयः।</div><div class="line" style="position:relative;">आरोग्यं प्रियसंवासो गृद्ध्येदेषु न पण्डितः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0072"><div class="verse-block verse-justify"><div class="line">न जानपदिकं दुःखमेकः शोचितुमर्हसि।</div><div class="line" style="position:relative;">अप्यभावेन युज्येत तच्चास्य न निवर्तते॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0073"><div class="verse-block verse-justify"><div class="line">अशोचन्प्रतिकुर्वीत यदि पश्येत्पराक्रमम्।</div><div class="line" style="position:relative;">भैषज्यमेतद्दुःखस्य यदेतन्नानुचिन्तयेत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0074"><div class="verse-block verse-justify"><div class="line">चिन्त्यमानं हि न व्येति भूयश्चापि प्रवर्धते ॥अनिष्टसम्प्रयोगाच्च विप्रयोगात्प्रियस्य च।</div><div class="line" style="position:relative;">॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0075"><div class="verse-block verse-justify"><div class="line">मानुषा मानसैर्दुःखैर्युज्यन्ते येऽल्पबुद्धयः ॥नार्थो न धर्मो न सुखं यदेतदनुशोचति।</div><div class="line" style="position:relative;">॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0076"><div class="verse-block verse-justify"><div class="line">तच्च नाप्नोति कार्यार्थं त्रिवर्गाच्चैव भ्रश्यते ॥अन्योन्यबाधनावस्थां प्राप्य वैषयिकीं नराः।</div><div class="line" style="position:relative;">॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0077"><div class="verse-block verse-justify"><div class="line">असन्तुष्टाः प्रमुह्यन्ति सन्तोषं यान्ति पण्डिताः ॥प्रज्ञया मानसं दुःखं हन्याच्छारीरमौषधैः।</div><div class="line" style="position:relative;">॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="11-strīparva-b0078"><div class="verse-block verse-justify"><div class="line">एतद्विज्ञानसामर्थ्यं न बालैः समतामियात् ॥शयानं चानुशेते हि तिष्ठन्तं चानुतिष्ठति।</div><div class="line" style="position:relative;">॥३३॥</div></div></div>
