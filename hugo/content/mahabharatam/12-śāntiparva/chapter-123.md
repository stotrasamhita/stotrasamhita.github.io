<div class="verse-block-wrapper" id="12-śāntiparva-b5088"><div class="verse-block verse-justify"><div class="line">तात धर्मार्थकामानां श्रोतुमिच्छामि निश्चयम्।</div><div class="line" style="position:relative;">लोकयात्रा हि कार्त्स्न्येन त्रिष्वेतेषु प्रतिष्ठिता॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5089"><div class="verse-block verse-justify"><div class="line">धर्मार्थकामाः किंमूलाः प्रभवः प्रलयश्च कः।</div><div class="line">अन्योन्यं चानुषज्जन्ते वर्तन्ते च पृथक्पृथक् ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5090"><div class="verse-block verse-justify"><div class="line">य एते स्युः सुमनसो लोकसंस्थार्थनिश्चये।</div><div class="line" style="position:relative;">कामप्रभवसंस्थासु सज्जन्ते च त्रयस्तदा॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5091"><div class="verse-block verse-justify"><div class="line">धर्ममूलोऽर्थ इत्युक्तः कामोऽर्थफलमुच्यते।</div><div class="line" style="position:relative;">संकल्पमूलास्ते सर्वे संकल्पो विषयात्मकः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5092"><div class="verse-block verse-justify"><div class="line">विषयाश्चैव कार्त्स्न्येन सर्व आहारसिद्धये।</div><div class="line" style="position:relative;">मूलमेतन्त्रिवर्गस्य निवृत्तिर्मोक्ष उच्यते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5093"><div class="verse-block verse-justify"><div class="line">धर्मः शरीरसंगुप्तिर्धर्मार्थश्चार्थ इष्यते।</div><div class="line" style="position:relative;">कामो रतिफलश्चात्र सर्वे रतिफलाः स्मृताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5094"><div class="verse-block verse-justify"><div class="line">सन्निकृष्टांश्चरेदेतान्न चैतान्मनसा त्यजेत्।</div><div class="line" style="position:relative;">विमुक्तस्तपसा सर्वान्धर्मादीन्कामनैष्ठिकान्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5095"><div class="verse-block verse-justify"><div class="line">श्रेष्ठबुद्धिस्त्रिवर्गस्य उदयं प्राप्नुयात्क्षणात्।</div><div class="line" style="position:relative;">[कर्मणा बुद्धिपूर्वेण भवत्यर्थो न वा पुनः ॥]॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5096"><div class="verse-block verse-justify"><div class="line">अर्थार्थमन्यद्भवति विपरीतमथापरम्।</div><div class="line">अनर्थार्थमवाप्यार्थमन्यत्राद्योपकारकम्।</div><div class="line" style="position:relative;">]बुद्ध्या बुद्ध इहार्थेन तदह्ना तु निकृष्टया॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5097"><div class="verse-block verse-justify"><div class="line">अपध्यानमलो धर्मो मलोऽर्थस्य निगूहनम्।</div><div class="line" style="position:relative;">संप्रमोहमलः कामो भूयस्तद्गुणवर्धितः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5098"><div class="verse-block verse-justify"><div class="line">अत्राप्युदाहरन्तीममितिहासं पुरातनम्।</div><div class="line" style="position:relative;">अरिष्टस्य च संवादं कामन्दस्य च भारत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5099"><div class="verse-block verse-justify"><div class="line">कामन्दमृषिमासीनमभिवाद्य नराधिपः।</div><div class="line" style="position:relative;">आङ्गोरिष्ठोऽथ पप्रच्छ कृत्वा समयमव्ययम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5100"><div class="verse-block verse-justify"><div class="line">यः पापं कुरुते राजा काममोहबलात्कृतः।</div><div class="line" style="position:relative;">प्रत्यासन्नस्य तस्यर्षे किं स्यात्पापप्रणाशनम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5101"><div class="verse-block verse-justify"><div class="line">अधर्मं धर्म इति च यो मोहादाचरेन्नरः।</div><div class="line">तं चापि प्रथितं लोके कथं राजा निवर्तयेत् ॥कामन्द उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5102"><div class="verse-block verse-justify"><div class="line">यो धर्मार्थौ परित्यज्य काममेवानुवर्तते।</div><div class="line" style="position:relative;">स धर्मार्थपरित्यागात्प्रज्ञानाशमिहार्च्छति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5103"><div class="verse-block verse-justify"><div class="line">प्रज्ञानाशात्मको मोहस्तथा धर्मार्थनाशकः।</div><div class="line" style="position:relative;">तस्मान्नास्तिकता चैव दुराचारश्च जायते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5104"><div class="verse-block verse-justify"><div class="line">दुराचारान्यदा राजा प्रदुष्टान्न नियच्छति।</div><div class="line" style="position:relative;">तस्मादुद्विजते लोकः सर्पाद्वेश्मगतादिव॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5105"><div class="verse-block verse-justify"><div class="line">तं प्रजा नानुरज्यन्ते न विप्रा न च साधवः।</div><div class="line" style="position:relative;">ततः संक्षयमाप्नोति तथा वध्यत्वमेव च॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5106"><div class="verse-block verse-justify"><div class="line">अपध्वस्तस्त्ववमतो दुःखं जीवति जीवितम्।</div><div class="line" style="position:relative;">जीवते यदपध्वस्तः शुद्धं मरणमेव तत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5107"><div class="verse-block verse-justify"><div class="line">अत्रैतदाहुराचार्याः पापस्य परिवर्तनम्।</div><div class="line" style="position:relative;">सेवितव्या त्रयी विद्या सत्कारो ब्राह्मणेषु च॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5108"><div class="verse-block verse-justify"><div class="line">महामना भवेद्धर्मे विवहेच्च महाकुले।</div><div class="line" style="position:relative;">ब्राह्मणांश्चापि सेवेत क्षमायुक्तान्मनस्विनः॥२२॥</div></div></div>
