<div class="verse-block-wrapper" id="12-śāntiparva-b10498"><div class="verse-block verse-justify"><div class="line">व्यास उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10499"><div class="verse-block verse-justify"><div class="line">--तीयमायुषो भागं गृहमेधी गृहे वसेत्।</div><div class="line" style="position:relative;">धर्मलब्धैर्युतो दारैरग्नीनाहृत्य सुव्रतः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10500"><div class="verse-block verse-justify"><div class="line">गृहस्थवृत्तयश्चैव चतस्रः कविभिः स्मृताः।</div><div class="line" style="position:relative;">कुमूलधान्यः प्रथमः कुम्भधान्यस्त्वनन्तरम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10501"><div class="verse-block verse-justify"><div class="line">अश्वस्तनोऽथ कापोतीमाश्रितो वृत्तिमाहरेत्।</div><div class="line" style="position:relative;">तेषां परः परो ज्यायान्धर्मतो लोकजित्तमः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10502"><div class="verse-block verse-justify"><div class="line">षट््कर्मावर्तयत्येकस्त्रिभिरन्यः प्रवर्तते।</div><div class="line" style="position:relative;">द्वाभ्यामेकश्चतुर्थस्तु ब्रह्मसत्रे व्यवस्थितः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10503"><div class="verse-block verse-justify"><div class="line">गृहमेधिव्रतान्यत्र महान्तीह प्रचक्षते।</div><div class="line" style="position:relative;">नात्मार्थे पाचयेदन्नं न वृथा घातयेत्पशून्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10504"><div class="verse-block verse-justify"><div class="line">प्राणी वा यदि वाऽप्राणी संस्कारं यजुषाऽर्हति।</div><div class="line" style="position:relative;">न दिवा प्रस्वपेज्जातु न पूर्वापररात्रयोः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10505"><div class="verse-block verse-justify"><div class="line">न भुञ्जीतान्तराकाले नानृतावाह्वयेत्स्त्रियम्।</div><div class="line" style="position:relative;">नास्यानश्नन्गृहे विप्रो वसेत्कश्चिदपूजितः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10506"><div class="verse-block verse-justify"><div class="line">तथास्यातिथयः पूज्या हव्यकव्यवहाः सदा।</div><div class="line" style="position:relative;">वेदविद्याव्रतस्नाताः श्रोत्रिया वेदपारगाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10507"><div class="verse-block verse-justify"><div class="line">स्वकर्मजीविनो दान्ताः क्रियावन्तस्तपस्विनः।</div><div class="line" style="position:relative;">तेषां हव्यं च कव्यं चाप्यर्हणार्थं विधीयते॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10508"><div class="verse-block verse-justify"><div class="line">नखरैः संप्रयातस्य स्वकर्मव्यापकस्य च।</div><div class="line" style="position:relative;">अपविद्धाग्निहोत्रस्य गुरोर्वाऽलीकचारिणः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10509"><div class="verse-block verse-justify"><div class="line">संविभागोऽत्र भूतानां सर्वेषामेव शिष्यते।</div><div class="line" style="position:relative;">तथैवापचमानेभ्यः प्रदेयं गृहमेधिना॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10510"><div class="verse-block verse-justify"><div class="line">विस्साशी भवेन्नित्यं नित्यं चामृतभोजनः।</div><div class="line" style="position:relative;">असुत --शेषं स्याद्भोजनं हविषा समम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10511"><div class="verse-block verse-justify"><div class="line">भृत्यशेष तु योऽश्नाति तमाहुर्विघसा शिनम्।</div><div class="line" style="position:relative;">विघसं भृत्यशेषं तु यज्ञशेषमथास्मृतय॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10512"><div class="verse-block verse-justify"><div class="line">स्वदारनिरतो दान्तो ह्यनसूयुर्जितेन्द्रियः।</div><div class="line" style="position:relative;">ऋत्विक्पुरोहिताचार्यर्मालुलातिथिनंश्रितैः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10513"><div class="verse-block verse-justify"><div class="line">वृद्धबालातुरैर्वैद्यैर्ज्ञातिसंबन्धिरान्धवैः।</div><div class="line" style="position:relative;">मातापितृभ्यां जामीभिर्भ्रात्रा पुत्रेण भावया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10514"><div class="verse-block verse-justify"><div class="line">दुहित्रा दासवर्गेण विवादं न समाचरत्।</div><div class="line" style="position:relative;">एतान्विमुच्य संवादान्सर्वपापैर्विमुच्यते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10515"><div class="verse-block verse-justify"><div class="line">एतैर्जितस्तु जयति सर्वाल्लोँकान्न संशयः।</div><div class="line" style="position:relative;">आचार्यो ब्रह्मलोकेशः प्राजापत्ये पिता प्रभुः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10516"><div class="verse-block verse-justify"><div class="line">अतिथिस्त्विन्द्रलोकस्य देवलोकस्य चर्त्विजः।</div><div class="line" style="position:relative;">जामयोऽप्सरसां लोके वैश्वदेवे तु ज्ञातयः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10517"><div class="verse-block verse-justify"><div class="line">संबन्धिबान्धवा दिक्षु पृथिव्यां मातृमातुलौ।</div><div class="line" style="position:relative;">बृद्धबावातुरकृशास्त्वाकाशे प्रभविष्णवः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10518"><div class="verse-block verse-justify"><div class="line">भ्राता ज्येष्ठः समः पित्रा भार्या पुत्रः स्वका तनुः।</div><div class="line" style="position:relative;">छाया स्वा दासवर्गश्च दुहिता कृपणं परम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10519"><div class="verse-block verse-justify"><div class="line">तस्मादेतैरधिक्षिप्तः सहेन्नित्यमसंज्वरः।</div><div class="line" style="position:relative;">गृहधर्मरतो विद्वान्धर्मनित्यो जितक्लमः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10520"><div class="verse-block verse-justify"><div class="line">न चार्थबद्धः कर्माणि धर्मं वा किंचिदाचरेत्।</div><div class="line" style="position:relative;">गृहस्थवृत्तयस्तिस्रस्तासां निःश्रेयसं परम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10521"><div class="verse-block verse-justify"><div class="line">परंपरं तथैवाहुश्चातुराश्रम्यमेव तत्।</div><div class="line" style="position:relative;">ये चोक्ता नियमास्तेषां सर्वं कार्यं बुभूषता॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10522"><div class="verse-block verse-justify"><div class="line">कुम्भधान्यैरुच्छशिलैः कापोतीं चास्थितास्तथा।</div><div class="line" style="position:relative;">यस्मिंश्चैते वसन्त्यर्हास्तद्राष्ट्रमभिवर्धते॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10523"><div class="verse-block verse-justify"><div class="line">दश पूर्वान्दश परान्पुनाति च पितामहान्।</div><div class="line" style="position:relative;">गृहस्थवृत्तीश्चाप्येता वर्तयेद्यो गतव्यथः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10524"><div class="verse-block verse-justify"><div class="line">स चक्रधरलोकानां सदृशीमाप्नुयाद्गतिम्।</div><div class="line" style="position:relative;">वितेन्द्रियाणामथवा गतिरेषा विधीयते॥२७॥</div></div></div>
