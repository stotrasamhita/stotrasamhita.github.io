<div class="verse-block-wrapper" id="12-śāntiparva-b2757"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2758"><div class="verse-block verse-justify"><div class="line">चातुराश्रम्यमुक्तं ते चातुर्वण्यं तथैव च।</div><div class="line">राष्ट्रस्य यत्कृत्यतमं तन्मे ब्रूहि पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2759"><div class="verse-block verse-justify"><div class="line">राष्ट्रस्य यत्कृत्यतमं राज्ञ एवाभिषेचनम्।</div><div class="line" style="position:relative;">अनिन्द्रमबलं राष्ट्रं दस्यवोऽभिभवन्त्युत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2760"><div class="verse-block verse-justify"><div class="line">अराजकेषु राष्ट्रेषु धर्मो न व्यवतिष्ठते।</div><div class="line" style="position:relative;">परस्परं च खादन्ति सर्वथा धिगराजकम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2761"><div class="verse-block verse-justify"><div class="line">इन्द्रमेव प्रणमते यद्राजानमिति श्रुतिः।</div><div class="line" style="position:relative;">यथैवेन्द्रस्तथा राजा संपूज्यो भूतिमिच्छता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2762"><div class="verse-block verse-justify"><div class="line">नाराजकेषु राष्ट्रेषु वस्तव्यमिति वैदिकम्।</div><div class="line" style="position:relative;">नाराजकेषु राष्ट्रेषु हव्यं वहति पावकः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2763"><div class="verse-block verse-justify"><div class="line">अथ चेदधिवेर्तेत राज्यार्थी बलवत्तरः।</div><div class="line" style="position:relative;">अराजकाणि राष्ट्राणि हतवीराणि वा पुनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2764"><div class="verse-block verse-justify"><div class="line">प्रत्युद्गम्याभिपूज्यः स्यादेतदत्र सुमन्त्रितम्।</div><div class="line" style="position:relative;">न हि राज्यात्पापतरमस्ति किंचिदराजकात्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2765"><div class="verse-block verse-justify"><div class="line">स चेत्समनुपश्येत समग्रं कुशलं भवेत्।</div><div class="line" style="position:relative;">बलवान्हि प्रकुपितः कुर्यान्निः शेषतामपि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2766"><div class="verse-block verse-justify"><div class="line">भूयांसं लभते क्लेशं या गौर्भवति दुर्दुहा।</div><div class="line" style="position:relative;">अथ या सुदुहा राजन्नैव तां वितुदन्त्यपि॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2767"><div class="verse-block verse-justify"><div class="line">यदतप्तं प्रणमते न तत्संतापयन्त्युत।</div><div class="line" style="position:relative;">यत्स्वयं नमते दारु न तत्संनामयन्त्यपि॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2768"><div class="verse-block verse-justify"><div class="line">एतयोपमया धीरः सन्नमेत बलीयसे।</div><div class="line" style="position:relative;">इन्द्राय स प्रणमते नमते यो बलीयसे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2769"><div class="verse-block verse-justify"><div class="line">तस्माद्राजैव कर्तव्यः सततं भूतिमिच्छता।</div><div class="line" style="position:relative;">न धनार्थो न दारार्थस्तेषां येषामराजकम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2770"><div class="verse-block verse-justify"><div class="line">प्रीयते हि हरन्पापः परवित्तमराजके।</div><div class="line" style="position:relative;">यदाऽस्य तद्धरन्त्यन्ये तदा राजानमिच्छति॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2771"><div class="verse-block verse-justify"><div class="line">पापा ह्यपि तदा क्षेमं न लभन्ते कदाचन।</div><div class="line" style="position:relative;">एकस्य हि द्वौ हरतो द्वयोश्च बहवोऽपरे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2772"><div class="verse-block verse-justify"><div class="line">अदासः क्रियते दासो ह्रियन्ते च बलात्स्त्रियः।</div><div class="line" style="position:relative;">एतस्मात्कारणाद्देवाः प्रजापालान्प्रचक्रिरे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2773"><div class="verse-block verse-justify"><div class="line">राजा चेन्न भवेल्लोके पृथिव्या दण्डधारकः।</div><div class="line" style="position:relative;">जले मत्स्यानिवाभक्ष्यन्दुर्बलं बलवत्तराः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2774"><div class="verse-block verse-justify"><div class="line">अराजकाः प्रजाः पूर्वं विनेशुरिति नः श्रुतम्।</div><div class="line" style="position:relative;">परस्परं भक्षयन्तो मत्स्या इव जले कृशान्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2775"><div class="verse-block verse-justify"><div class="line">समेत्य तास्ततश्चक्रुः समयानिति नः श्रुतम्।</div><div class="line" style="position:relative;">वाक्शूरो दण्डपरुषो यश्च स्यात्पारदारिकः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2776"><div class="verse-block verse-justify"><div class="line">यश्च नः समयं भिन्द्यात्त्याज्या नस्तादृशा इति।</div><div class="line">विश्वासार्थं च सर्वेषां वर्णानामविशेषतः।</div><div class="line" style="position:relative;">तास्तथा समयं कृत्वा समयेनावतस्थिरे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2777"><div class="verse-block verse-justify"><div class="line">सहितास्तास्तदा जग्मुरसुखार्ताः पितामहम्।</div><div class="line" style="position:relative;">अनीश्वरा विनश्यामो भगवन्नीश्वरं दिश॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2778"><div class="verse-block verse-justify"><div class="line">यं पूजयेम संभूय यश्च नः प्रतिपालयेत्।</div><div class="line">ताभ्यो मनुं व्यादिदेश मनुर्नाभिननन्द ताः ॥मनुरुवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2779"><div class="verse-block verse-justify"><div class="line">बिभेमि कर्मणः पापाद्राज्यं हि भृशदुष्करम्।</div><div class="line">विशेषतो मनुष्येषु मिथ्यावृत्तेषु नित्यदा ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2780"><div class="verse-block verse-justify"><div class="line">तमब्रुवन्प्रजा मा भैर्विधास्यामो धनं तव।</div><div class="line" style="position:relative;">पशूनामथ पञ्चांशं धरण्यस्य तथैव च॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2781"><div class="verse-block verse-justify"><div class="line">धान्यस्य दशमं भागं दास्यामः कोशवर्धनम्।</div><div class="line" style="position:relative;">कन्यां शुल्के चारुरूपां विवाहेषूद्यतासु च॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2782"><div class="verse-block verse-justify"><div class="line">मुख्येन शस्त्रपत्रेण ये मनुष्याः प्रधानतः।</div><div class="line" style="position:relative;">भवन्तं तेऽनुयास्यन्ति महेन्द्रमिव देवताः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2783"><div class="verse-block verse-justify"><div class="line">स त्वं जातबलो राजन्दुष्प्रधर्षः प्रतापवान्।</div><div class="line" style="position:relative;">सुखे धास्यसि नः सर्वान्कुबेर इव नैर्ऋतान्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2784"><div class="verse-block verse-justify"><div class="line">यं च धर्मं चरिष्यन्ति प्रजा राज्ञा सुरक्षिताः।</div><div class="line" style="position:relative;">चतुर्थं तस्य धर्मस्य त्वत्संस्थं नो भविष्यति॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2785"><div class="verse-block verse-justify"><div class="line">तेन धर्मेण महता सुखं लब्धेन भावितः।</div><div class="line" style="position:relative;">पाह्यस्मान्सर्वतो राजन्देवानिव शतक्रतुः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2786"><div class="verse-block verse-justify"><div class="line">विजयाय हि निर्याहि प्रतपत्रश्मिवानिव।</div><div class="line" style="position:relative;">मानं विधम शत्रूणां धर्मं जनय नः सदा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2787"><div class="verse-block verse-justify"><div class="line">स निर्ययौ महातेजा बलेन महता वृतः।</div><div class="line" style="position:relative;">महाभिजनसंपन्नस्तेजसा प्रज्वलन्निव॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2788"><div class="verse-block verse-justify"><div class="line">तस्य दृष्ट्वा महत्वं ते महेन्द्रस्येव देवताः।</div><div class="line">अपतत्रसिरे सर्वे स्वधर्मे च ददुर्मनः।</div><div class="line" style="position:relative;">`वर्णिनश्चाश्रमाश्चैव म्लेच्छाः सर्वे च दस्यवः ॥'॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2789"><div class="verse-block verse-justify"><div class="line">ततो महीं परिययौ पर्जन्य इव वृष्टिमान्।</div><div class="line" style="position:relative;">शमयन्सर्वतः पापान्स्वकर्मसु च योजयन्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2790"><div class="verse-block verse-justify"><div class="line">एवं ये भूतिमिच्छेयुः पृथिव्यां मानवाः क्वचित्।</div><div class="line" style="position:relative;">कुर्यू राजानमेवाग्रे प्रजानुग्रहकारणात्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2791"><div class="verse-block verse-justify"><div class="line">नमस्येरंश्च तं भक्त्या शिष्या इव गुरुं सदा।</div><div class="line" style="position:relative;">देवा इव च देवेन्द्रं नरा राजानमन्तिकात्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2792"><div class="verse-block verse-justify"><div class="line">सत्कृतं स्वजनेनेह परोऽपि बहुमन्यते।</div><div class="line" style="position:relative;">स्वजनेन त्ववज्ञातं परे परिभवन्त्युत॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2793"><div class="verse-block verse-justify"><div class="line">राज्ञः परैः परिभवः सर्वेषामसुखावहः।</div><div class="line" style="position:relative;">तस्माच्छत्रं च पत्रं च वासांस्याभरणानि च॥३७॥</div></div></div>
