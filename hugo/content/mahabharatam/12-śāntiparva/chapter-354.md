<div class="verse-block-wrapper" id="12-śāntiparva-b15260"><div class="verse-block verse-justify"><div class="line">नरनारायणावूचतुः।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15261"><div class="verse-block verse-justify"><div class="line">धन्योस्यनुगृहीतोसि यत्ते दृष्टः स्वयंप्रभुः।</div><div class="line" style="position:relative;">न हि तं दृष्टवान्कश्चित्पद्मयोनिरपि स्वयम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15262"><div class="verse-block verse-justify"><div class="line">अव्यक्तयोनिर्भगवान्दुर्दर्शः पुरुषोत्तमः।</div><div class="line" style="position:relative;">नारदैतद्धि नौ सत्यं वचनं समुदाहृतम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15263"><div class="verse-block verse-justify"><div class="line">नास्य भक्तात्प्रियतरो लोके कश्चन विद्यते।</div><div class="line" style="position:relative;">ततः स्वयं दर्शितवान्स्वमात्मानं द्विजोत्तम॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15264"><div class="verse-block verse-justify"><div class="line">तपो हि तप्यतस्तस्य यत्स्थानं परमात्मनः।</div><div class="line" style="position:relative;">न तत्संप्राप्नुते कश्चिदृते ह्यावां द्वितोत्तम॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15265"><div class="verse-block verse-justify"><div class="line">या हि सूर्यसहस्रस्य समस्तस्य भवेद्द्युतिः।</div><div class="line" style="position:relative;">स्थानस्य सा भवेत्तस्य स्वयं तेन विराजता॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15266"><div class="verse-block verse-justify"><div class="line">तस्मादुत्तिष्ठते विप्र देवाद्विश्वभुवः पतेः।</div><div class="line" style="position:relative;">क्षमा क्षमावतां श्रेष्ठ यया भूमिस्तु युज्यते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15267"><div class="verse-block verse-justify"><div class="line">तस्माच्चोत्तिष्ठते देवात्सर्वभूतहिताद्रसः।</div><div class="line" style="position:relative;">आपो हि तेन युज्यन्ते द्रवत्वं प्राप्नुवन्ति च॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15268"><div class="verse-block verse-justify"><div class="line">तस्मादेव समुद्भूतं तेजो रूपगुणात्मकम्।</div><div class="line" style="position:relative;">येन संयुज्यते सूर्यस्ततो लोके विराजते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15269"><div class="verse-block verse-justify"><div class="line">तस्माद्देवात्समुद्भूतः स्पर्शस्तु पुरुषोत्तमात्।</div><div class="line" style="position:relative;">येन स्म युज्यते वायुस्ततो लोकान्विवात्यसौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15270"><div class="verse-block verse-justify"><div class="line">तस्माच्चोत्तिष्ठते शब्दः सर्वलोकेश्वरात्प्रभोः।</div><div class="line" style="position:relative;">आकाशं युज्यते येन ततस्तिष्ठत्यसंवृतम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15271"><div class="verse-block verse-justify"><div class="line">तस्माच्चोत्तिष्ठते देवात्सर्वभूतगतं मनः।</div><div class="line" style="position:relative;">चन्द्रमा येन संयुक्तः प्रकाशगुणधारणः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15272"><div class="verse-block verse-justify"><div class="line">सद्भूतोत्पादकं नाम तत्स्थानं वेदसंज्ञितम्।</div><div class="line" style="position:relative;">विद्यासहायो यत्रास्ते भगवान्हव्यकव्यभुक्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15273"><div class="verse-block verse-justify"><div class="line">ये हि निष्कल्मषा लोके पुण्यपापविवर्जिताः।</div><div class="line" style="position:relative;">तेषां वै क्षेममध्वानं गच्छतां द्विजसत्तम॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15274"><div class="verse-block verse-justify"><div class="line">सर्वलोके तमोहन्ता आदित्यो द्वारमुच्यते।</div><div class="line" style="position:relative;">` ज्वालामाली महातेजा येनेदं धार्यते जगत् ॥'॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15275"><div class="verse-block verse-justify"><div class="line">आदित्यदग्धसर्वाङ्गा अदृश्याः केनचित्क्वचित्।</div><div class="line" style="position:relative;">परमाणुभूता भूत्वा तु तं देवं प्रविशन्त्युत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15276"><div class="verse-block verse-justify"><div class="line">तस्मादपि च निर्मुक्ता अनिरुद्धतनौ स्थिताः।</div><div class="line" style="position:relative;">मनोभूतास्ततो भूत्वा प्रद्युम्नं प्रविशन्त्युत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15277"><div class="verse-block verse-justify"><div class="line">प्रद्युम्नाच्चापि निर्मुक्ता जीवं संकर्षणं ततः।</div><div class="line" style="position:relative;">विशन्ति विप्रप्रवराः साङ्ख्या भागवतैः सह॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15278"><div class="verse-block verse-justify"><div class="line">ततस्त्रैगुण्यहीनास्ते परमात्मानमञ्जसा।</div><div class="line">प्रविशन्ति द्विजश्रेष्ठाः क्षेत्रज्ञं निर्गुणात्मकम्।</div><div class="line" style="position:relative;">सर्वावासं वासुदेवं क्षेत्रज्ञं विद्धि तत्त्वतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b15279"><div class="verse-block verse-justify"><div class="line">समाहितमनस्काश्च नियताः संयतेन्द्रियाः।</div><div class="line" style="position:relative;">एकान्तभावोपगता वासुदेवं विशन्ति ते॥२०॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b15281"><div class="verse-block verse-justify"><div class="line">ये तु तस्यैव देवस्य प्रादुर्भावाः सुरप्रियाः।</div><div class="line" style="position:relative;">भविष्यन्ति त्रिलोकस्थास्तेषां स्वस्तीत्यथो द्विज॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b15284"><div class="verse-block verse-justify"><div class="line">समागतो भगवता संकल्पं कृतवांस्तथा।</div><div class="line" style="position:relative;">सर्वं हि नौ संविदितं त्रैलोक्ये सचराचरे॥२५॥</div></div></div>
