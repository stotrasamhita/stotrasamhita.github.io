<div class="verse-block-wrapper" id="12-śāntiparva-b10760"><div class="verse-block verse-justify"><div class="line">व्यास उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10761"><div class="verse-block verse-justify"><div class="line">शरीराद्विप्रमुक्तं हि सूक्ष्मभूतं शरीरिणम्।</div><div class="line" style="position:relative;">कर्मभिः परिपश्यन्ति शास्त्रोक्तैः शास्त्रचेतसः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10762"><div class="verse-block verse-justify"><div class="line">यथा मरीच्यः सहिताश्चरन्तिगच्छन्ति तिष्ठन्ति च दृश्यमानाः।</div><div class="line" style="position:relative;">देहैर्विमुक्तानि वरन्ति लोकांस्तथैव सत्वान्यतिमानुषाणि॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10763"><div class="verse-block verse-justify"><div class="line">प्रतिरूपं यथैवाप्सु तावत्सूर्यस्य लक्ष्यते।</div><div class="line" style="position:relative;">सत्ववांस्तु तथा सत्वं प्रतिरूपं स पश्यति॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10764"><div class="verse-block verse-justify"><div class="line">तानि सूक्ष्माणि सत्वानि विमुक्तानि शरीरतः।</div><div class="line" style="position:relative;">तेन तत्वेन तत्वज्ञाः पश्यन्ति नियतेन्द्रियाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10765"><div class="verse-block verse-justify"><div class="line">स्वपतां जाग्रतां चैष सर्वेषामात्मचिन्तितम्।</div><div class="line" style="position:relative;">प्रधानाद्वैधयुक्तानां दह्यते कर्मजं रजः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10766"><div class="verse-block verse-justify"><div class="line">यथाऽहनि तथा रात्रौ यथा रात्रौ तथाऽहनि।</div><div class="line" style="position:relative;">वशे तिष्ठति सत्वात्मा सततं योगयोगिनाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10767"><div class="verse-block verse-justify"><div class="line">तेषां नित्यं सदा नित्यो भूतात्मा सततं गुणैः।</div><div class="line" style="position:relative;">सप्तभिस्त्वन्वितः सूक्ष्मैश्चरिष्णुरजरामरः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10768"><div class="verse-block verse-justify"><div class="line">मनोवुद्धिपराभूतः स्वदेहपरदेहवित्।</div><div class="line" style="position:relative;">स्वप्नेष्वपि भवत्येष विज्ञाता सुखदुःखयोः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10769"><div class="verse-block verse-justify"><div class="line">तत्रापि लभते दुःखं तत्रापि लभते सुखम्।</div><div class="line" style="position:relative;">कामं क्रोधं च तत्रापि कृत्वा व्यसनमृच्छति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10770"><div class="verse-block verse-justify"><div class="line">प्रीणितश्चापि भवति महतोऽर्थानवाप्य हि।</div><div class="line" style="position:relative;">करोति पुण्यं तत्रापि जाग्रन्निव च पश्यति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10771"><div class="verse-block verse-justify"><div class="line">सदोष्मान्तर्गतश्चापि गर्भत्वं समुपेयिवान्।</div><div class="line" style="position:relative;">दश मासान्वसन्कुक्षौ नैषोऽन्नमिव जीर्यते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10772"><div class="verse-block verse-justify"><div class="line">तमेतमतितेजोंशं भूतात्मानं हृदि स्थितम्।</div><div class="line" style="position:relative;">तमोरजोभ्यामाविष्टा नानुपश्यन्ति मूर्तिषु॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10773"><div class="verse-block verse-justify"><div class="line">योगशास्त्रपरा भूत्वा स्वमात्मानं परीप्सवः।</div><div class="line">`तमोरजोभ्यां निर्मुक्तास्तं प्रपश्यन्ति मूर्तिषु।</div><div class="line" style="position:relative;">'अनुच्छ्वासान्यमूर्तानि यानि वज्रोपमान्यपि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10774"><div class="verse-block verse-justify"><div class="line">पृथग्भूतेषु सृष्टेषु चतुर्ष्वाश्रमकर्मसु।</div><div class="line" style="position:relative;">समाधौ योगमेवैतच्छाण्डिल्यः सममब्रवीत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10775"><div class="verse-block verse-justify"><div class="line">विदित्वा सप्तसूक्ष्माणि षडङ्गं च महेश्वरम्।</div><div class="line" style="position:relative;">प्रधानविनियोगज्ञः परं ब्रह्मानुपश्यति॥१६॥</div></div></div></div>
