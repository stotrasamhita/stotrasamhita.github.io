<div class="stotra-article"><h2 class="stotra-heading">अध्यायः ३१</h2>
<div class="verse-block-wrapper" id="12-śāntiparva-b1280"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1281"><div class="verse-block verse-justify"><div class="line">तूष्णींभूतं तु राजानं शोचमानं युधिष्ठिरम्।</div><div class="line">तपस्वी धर्मतत्त्वज्ञः कृष्णद्वैपायनोऽब्रवीत् ॥व्यास उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1282"><div class="verse-block verse-justify"><div class="line">प्रजानां पालनं धर्मो राज्ञां राजीवलोचन।</div><div class="line" style="position:relative;">धर्मः प्रमाणं लोकस्य नित्यं धर्मोऽनुवर्त्यताम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1283"><div class="verse-block verse-justify"><div class="line">अनुतिष्ठस्व तद्राजन्पितृपैतामहं पदम्।</div><div class="line" style="position:relative;">ब्राह्मणेषु तु यो धर्मः स नित्यो वेदनिश्चितः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1284"><div class="verse-block verse-justify"><div class="line">तत्प्रमाणं प्रमाणानां शाश्वतं भरतर्षभ।</div><div class="line" style="position:relative;">तस्य धर्मस्य कृत्स्नस्य क्षत्रियः परिरक्षिता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1285"><div class="verse-block verse-justify"><div class="line">तथा यः प्रतिहन्त्यस्य शासनं विषये नरः।</div><div class="line" style="position:relative;">स बाहुभ्यां विनिग्राह्यो लोकयात्राविघातकः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1286"><div class="verse-block verse-justify"><div class="line">प्रमाणमप्रमाणं यः कुर्यान्मोहवशं गतः।</div><div class="line" style="position:relative;">भृत्यो वा यदि वा पुत्रस्तपस्वी वाऽथ कश्चन॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1287"><div class="verse-block verse-justify"><div class="line">पापान्सर्वैरुपायैस्तान्नियच्छेच्छातयीत वा।</div><div class="line" style="position:relative;">अतोऽन्यथा वर्तमानो राजा प्राप्नोति किल्विषम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1288"><div class="verse-block verse-justify"><div class="line">धर्मं विनश्यमानं हि यो न रक्षेत्स धर्महा।</div><div class="line" style="position:relative;">ते त्वया धर्महन्तारो निहताः सपदानुगाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1289"><div class="verse-block verse-justify"><div class="line">स्वधर्मे वर्तमानस्त्वं किंनु शोचसि पाण्डव।</div><div class="line">राजा हि हन्याद्दद्याच्च प्रजा रक्षेच्च धर्मतः ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1290"><div class="verse-block verse-justify"><div class="line">न तेऽतिशङ्के वचनं यद्ब्रवीषि तपोधन।</div><div class="line" style="position:relative;">अपरोक्षो हि ते धर्मः सर्वधर्मविदां वर॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1291"><div class="verse-block verse-justify"><div class="line">मया त्ववध्या बहवो घातिता राज्यकारणात्।</div><div class="line">तानि कर्माणि मे ब्रह्मन्दहन्ति च पचन्ति च ॥व्यास उवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1292"><div class="verse-block verse-justify"><div class="line">ईश्वरो वा भवेत्कर्ता पुरुषो वाऽपि भारत।</div><div class="line" style="position:relative;">हठो वा वर्तते लोके कर्मजं वा फलं स्मृतम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1293"><div class="verse-block verse-justify"><div class="line">ईश्वरेण नियुक्तो हि साध्वसाधु च भारत।</div><div class="line" style="position:relative;">कुरुते पुरुषः कर्म फलमीश्वरगामि तत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1294"><div class="verse-block verse-justify"><div class="line">यथाहि पुरुषश्छिन्द्याद्वृक्षं परशुना वने।</div><div class="line" style="position:relative;">छेत्तुरेव भवेत्पापं परशोर्न कथंचन॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1295"><div class="verse-block verse-justify"><div class="line">अथवा तदुपादानात्प्राप्नुयात्कर्मणः फलम्।</div><div class="line" style="position:relative;">दण्डशस्त्रकृतं पापं पुरुषे तन्न विद्यते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1296"><div class="verse-block verse-justify"><div class="line">न चैतदिष्टं कौन्तेय यदन्येन कृतं फलम्।</div><div class="line" style="position:relative;">प्राप्नुयादिति तस्माच्च ईश्वरे तन्निवेशय॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1297"><div class="verse-block verse-justify"><div class="line">अथापि पुरुषः कर्ता कर्मणोः शुभपापयोः।</div><div class="line" style="position:relative;">न परो विद्यते तस्मादेवमप्यशुभं कुतः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1298"><div class="verse-block verse-justify"><div class="line">न हि कश्चित्क्वचिद्राजन्दिष्टं प्रतिनिवर्तते।</div><div class="line" style="position:relative;">दण्डशस्त्रकृतं पापं पुरुषे तन्न विद्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1299"><div class="verse-block verse-justify"><div class="line">यदि वा मन्यसे राजन्हतमेकं प्रतिष्ठितम्।</div><div class="line" style="position:relative;">एवमप्यशुभं कर्म न भूतं न भविष्यति॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1300"><div class="verse-block verse-justify"><div class="line">अथाभिपत्तिर्लोकस्य कर्तव्या पुण्यपापयोः।</div><div class="line" style="position:relative;">अभिपन्नमिदं लोके राज्ञामुद्यतदण्डनम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1301"><div class="verse-block verse-justify"><div class="line">तथापि लोके कर्माणि समावर्तन्ति भारत।</div><div class="line" style="position:relative;">शुभाशुभफलं चैते प्राप्नुवन्तीति मे मतिः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1302"><div class="verse-block verse-justify"><div class="line">एवं पश्य शुभादेशं कर्मणस्तत्फलं ध्रुवम्।</div><div class="line" style="position:relative;">त्यज त्वं राजशार्दूल मैवं शोके मनः कृथा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1303"><div class="verse-block verse-justify"><div class="line">स्वधर्मे वर्तमानस्य सापवादेऽपि भारत।</div><div class="line" style="position:relative;">एवमात्मपरित्यागस्तव राजन्न शोभनः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1304"><div class="verse-block verse-justify"><div class="line">विहितानि हि कौन्तेय प्रायश्चित्तानि कर्मणाम्।</div><div class="line" style="position:relative;">शरीरवांस्तानि कुर्यादशरीरः पराभवेत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1305"><div class="verse-block verse-justify"><div class="line">तद्राजञ्जीवमानस्त्वं प्रायश्चित्तं करिष्यसि।</div><div class="line">प्रायश्चित्तमकृत्वा तु प्रेत्य तप्ताऽसि भारत ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1306"><div class="verse-block verse-justify"><div class="line">हताः पुत्राश्च पौत्राश्च भ्रातरः पितरस्तथा।</div><div class="line" style="position:relative;">श्वशुरा गुरवश्चैव मातुलाश्च पितामहाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1307"><div class="verse-block verse-justify"><div class="line">क्षत्रियाश्च महात्मानः संबन्धिसुहृदस्तथा।</div><div class="line" style="position:relative;">वयस्या भागिनेयाश्च ज्ञातयश्च पितामह॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1308"><div class="verse-block verse-justify"><div class="line">बहवश्च मनुष्येन्द्रा नानादेशसमागताः।</div><div class="line" style="position:relative;">घातिता राज्यलुब्धेन मयैकेन पितामह॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1309"><div class="verse-block verse-justify"><div class="line">तांस्तादृशानहं हत्वा धर्मनित्यान्महीक्षितः।</div><div class="line" style="position:relative;">असकृत्सोमपान्वीरान्क्रिं प्राप्स्यामि तपोधन॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1310"><div class="verse-block verse-justify"><div class="line">दह्याम्यनिशमद्यापि चिन्तयानः पुन पुनः।</div><div class="line" style="position:relative;">हीनां पार्थिवसिंहैस्तैः श्रीमद्भिः पृथिवीमिमाम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1311"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा ज्ञातिवधं घोरं हतांश्च शतशः परान्।</div><div class="line" style="position:relative;">कोटिशश्च नरानन्यान्परितप्ये पितामह॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1312"><div class="verse-block verse-justify"><div class="line">का नु तासां वरस्त्रीणामवस्थाऽद्य भविष्यति।</div><div class="line" style="position:relative;">विहीनानां तु तनयैः पतिभिर्भ्रातृभिस्तथा॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1313"><div class="verse-block verse-justify"><div class="line">अस्मानन्तकरान्घोरान्पाण्डवान्वृष्णिसंहतान्।</div><div class="line" style="position:relative;">आक्रोशन्त्यः कृशा दीनाः प्रपतिष्यन्ति भूतले॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1314"><div class="verse-block verse-justify"><div class="line">अपश्यन्त्यः पितॄन्भ्रातॄन्पतीन्पुत्रांश्च योषितः।</div><div class="line" style="position:relative;">त्यक्त्वा प्राणान्स्त्रियः सर्वागमिष्यन्ति यमक्षयम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1315"><div class="verse-block verse-justify"><div class="line">वत्सलत्वाद्द्विजश्रेष्ठ तत्र ये नास्ति संसयः।</div><div class="line" style="position:relative;">व्यक्तं सौक्ष्म्याच्च धर्मस्य प्राप्स्यामः स्त्रीवधं वयम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1316"><div class="verse-block verse-justify"><div class="line">ते वयं सुहृदो हत्वा कृत्वा पापमनन्तकम्।</div><div class="line" style="position:relative;">नकरे निपतिष्यामो ह्यधः शिरस एव ह॥३७॥</div></div></div>
