<div class="verse-block-wrapper" id="12-śāntiparva-b2712"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2713"><div class="verse-block verse-justify"><div class="line">श्रुता मे कथिताः पूर्वं चत्वारो मानवाश्रमाः।</div><div class="line">व्याख्यानमेषामाचक्ष्व पृच्छतो मे पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2714"><div class="verse-block verse-justify"><div class="line">विदिताः सर्व एवेह धर्मास्तव युधिष्ठिर।</div><div class="line" style="position:relative;">यथा मम महाबाहो विदिताः साधुसंमताः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2715"><div class="verse-block verse-justify"><div class="line">यत्तु लिङ्गान्तरगतं पृच्छसे मां युधिष्ठिर।</div><div class="line">धर्मं धर्मभृतां श्रेष्ठ तन्निबोध नराधिप।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2716"><div class="verse-block verse-justify"><div class="line">सर्वाण्येतानि कौन्तेय विद्यन्ते भरतर्षभ।</div><div class="line" style="position:relative;">साध्वाचारप्रवृत्तानां चातुराश्रम्यकर्मणाम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2717"><div class="verse-block verse-justify"><div class="line">अकामद्वेषसंयुक्तो दण़्डनीत्या युधिष्ठिर।</div><div class="line" style="position:relative;">समदर्शी च भूतेषु भैक्ष्याश्रमपदं भवेत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2718"><div class="verse-block verse-justify"><div class="line">वेत्ति दानं विसर्गं च विग्रहानुग्रहौ तथा।</div><div class="line" style="position:relative;">यथोक्तवृत्तो धीरश्च क्षमाश्रमपदं भवेत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2719"><div class="verse-block verse-justify"><div class="line">अर्हान्पूजयतो नित्यं संविभागेन पाण्डव।</div><div class="line" style="position:relative;">सर्वतस्तस्य कौन्तेय भैक्ष्याश्रमपदं भवेत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2720"><div class="verse-block verse-justify"><div class="line">ज्ञातिसंबन्धिमित्राणि व्यापन्नानि युधिष्ठिर।</div><div class="line" style="position:relative;">समभ्युद्धरमाणस्य दीक्षाश्रमपदं भवेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2721"><div class="verse-block verse-justify"><div class="line">लोकमुख्येषु सत्कारं लिङ्गिमुख्येषु चासकृत्।</div><div class="line" style="position:relative;">कुर्वतस्तस्य कौन्तेय वन्याश्रमपदं भवेत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2722"><div class="verse-block verse-justify"><div class="line">आह्निकं पितृयज्ञांश्च भूतयज्ञान्समानुषान्।</div><div class="line" style="position:relative;">कुर्वतः पार्थ विपुलान्वन्याश्रमपदं भवेत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2723"><div class="verse-block verse-justify"><div class="line">संविभागेन भूतानामतिथीनां तथाऽर्चनात्।</div><div class="line" style="position:relative;">देवयज्ञैश्च राजेन्द्र वन्याश्रमपदं भवेत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2724"><div class="verse-block verse-justify"><div class="line">मर्दनं परराष्ट्राणां शिष्टार्थं सत्यविक्रम।</div><div class="line" style="position:relative;">कुर्वतः पुरुषव्याघ्र वन्याश्रमपदं भवेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2725"><div class="verse-block verse-justify"><div class="line">पालनात्सर्वभूतानां स्वराष्ट्रपरिपालनात्।</div><div class="line" style="position:relative;">दीक्षा बहुविधा राजन्सत्याश्रमपदं भवेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2726"><div class="verse-block verse-justify"><div class="line">वेदाध्ययननित्यत्वं क्षमाऽथाचार्यपूजनम्।</div><div class="line" style="position:relative;">तथोपाध्यायशुश्रूषा ब्रह्माश्रमपदं भवेत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2727"><div class="verse-block verse-justify"><div class="line">आह्निकाञ्जपमानस्य देवान्पूजयतः सदा।</div><div class="line" style="position:relative;">धर्मेण पुरुषव्याघ्र धर्माश्रमपदं भवेत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2728"><div class="verse-block verse-justify"><div class="line">मृत्युर्वा रक्षणं वेति यस्य राज्ञो विनिश्चयः।</div><div class="line" style="position:relative;">प्राणद्यूते व्यवस्थाप्य ब्रह्माश्रमपदं भवेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2729"><div class="verse-block verse-justify"><div class="line">अजिह्नमशठं मार्गं वर्तमानस्य भारत।</div><div class="line" style="position:relative;">सर्वदा सर्वभूतेषु ब्रह्माश्रमपदं भवेत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2730"><div class="verse-block verse-justify"><div class="line">वानप्रस्थेषु विप्रेषु त्रैविद्येषु च भारत।</div><div class="line" style="position:relative;">प्रयच्छतोऽर्थान्विपुलान्वन्याश्रमपदं भवेत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2731"><div class="verse-block verse-justify"><div class="line">सर्वभूतेष्वनुक्रोशं कुर्वतस्तव भारत।</div><div class="line" style="position:relative;">आनृशंस्ये प्रवृत्तस्य नियतः पुण्यसंचयः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2732"><div class="verse-block verse-justify"><div class="line">बालवृद्धेषु कौन्तेय सर्वावस्थं युधिष्ठिर।</div><div class="line" style="position:relative;">अनुक्रोशक्रिया पार्थ धर्म एष सनातनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2733"><div class="verse-block verse-justify"><div class="line">बलात्कृतेषु भूतेषु परित्राणं कुरूद्वह।</div><div class="line" style="position:relative;">शरणागतेषु कौरव्य परं कारुण्यमाचर॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2734"><div class="verse-block verse-justify"><div class="line">चराचराणां भूतानां रक्षणं चापि सर्वशः।</div><div class="line" style="position:relative;">यथार्हपूजां च तथा कुर्वन्गार्हस्थ्यमावसेत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2735"><div class="verse-block verse-justify"><div class="line">ज्येष्ठानुज्येष्ठपत्नीनां भ्रातॄणां पुत्रनप्तृणाम्।</div><div class="line" style="position:relative;">निग्रहानुग्रहौ पार्थ गार्हस्थ्यममितं तपः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2736"><div class="verse-block verse-justify"><div class="line">साधूनामर्चनीयानां पूजासु विदितात्मनाम्।</div><div class="line" style="position:relative;">पालनं पुरुषव्याघ्र गृहाश्रमपदं भवेत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2737"><div class="verse-block verse-justify"><div class="line">आश्रमस्थानि भूतानि यस्य वेश्मनि भारत।</div><div class="line" style="position:relative;">भुञ्जते विपुलं भोज्यं तद्गार्हस्थ्यं युधिष्ठिर॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2738"><div class="verse-block verse-justify"><div class="line">यः स्थितः पुरुषो धर्मे धात्रा सृष्टे यथार्थवत्।</div><div class="line" style="position:relative;">आश्रमाणां हि सर्वेषां फलं प्राप्नोत्यनामयम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2739"><div class="verse-block verse-justify"><div class="line">यस्मिन्न नश्यन्ति गुणाः कौन्तेय पुरुषे सदा।</div><div class="line" style="position:relative;">आश्रमस्थं तमप्याहुर्नरश्रेष्ठं युधिष्ठिर॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2740"><div class="verse-block verse-justify"><div class="line">स्थानमानं कुलेमानं वयोमानं तथैव च।</div><div class="line" style="position:relative;">कुर्वन्वसति सर्वेषु ह्याश्रमेषु युधिष्ठिर॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2741"><div class="verse-block verse-justify"><div class="line">देशधर्मांश्च कौन्तेय कुलधर्मास्तथैव च।</div><div class="line" style="position:relative;">पालयन्पुरुषव्याघ्र राजा सर्वाश्रमी भवेत्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2742"><div class="verse-block verse-justify"><div class="line">काले विभूतिं भूतानामुपहारांस्तथैव च।</div><div class="line" style="position:relative;">अर्हयन्पुरुषव्याघ्र साधूनामाश्रमे वसेत्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2743"><div class="verse-block verse-justify"><div class="line">देशधर्मगतश्चापि यो धर्मं प्रत्यवेक्षते।</div><div class="line" style="position:relative;">सर्वलोकस्य कौन्तेय राजा भवति सोश्रमी॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2744"><div class="verse-block verse-justify"><div class="line">ये धर्मकुशला लोके धर्मं कुर्वन्ति भारत।</div><div class="line" style="position:relative;">पालिता यस्य विषये पादांशस्तस्य भूपतेः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2745"><div class="verse-block verse-justify"><div class="line">धर्मारामान्धर्मपरान्ये न रक्षन्ति मानवान्।</div><div class="line" style="position:relative;">पार्थिवाः पुरुषव्याघ्र तेषां पापं हरन्ति ते॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2746"><div class="verse-block verse-justify"><div class="line">ये च रक्षासहायाः स्युः पार्थिवानां युधिष्ठिर।</div><div class="line" style="position:relative;">ते चैवांशहराः सर्वे धर्मे परकृतेऽनघ॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2747"><div class="verse-block verse-justify"><div class="line">सर्वाश्रमपदेऽप्याहुर्गार्हस्थ्यं दीप्तनिर्णयम्।</div><div class="line" style="position:relative;">पावनं पुरुषव्याघ्र यद्वयं पर्युपास्महे॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2748"><div class="verse-block verse-justify"><div class="line">आत्मोपमस्तु भूतेषु यो वै भवति मानवः।</div><div class="line" style="position:relative;">न्यस्तदण्डो जितक्रोधः प्रेत्येह लभते सुखम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2749"><div class="verse-block verse-justify"><div class="line">धर्मोच्छ्रिता सत्यजला शीलयष्टिर्दमध्वजा।</div><div class="line" style="position:relative;">त्यागवाताध्वगा शीघ्रा नौस्तया सन्तरिष्यति॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2750"><div class="verse-block verse-justify"><div class="line">यदा सर्वत्र निर्मुक्तः कामो नास्य हृदि स्थितः।</div><div class="line" style="position:relative;">यदा सत्यान्वितो वृत्तैस्तदा ब्रह्म सम श्नुते॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2751"><div class="verse-block verse-justify"><div class="line">सुप्रसन्नस्तु भावेन योगेन च नराधिप।</div><div class="line" style="position:relative;">धर्मं पुरुषशार्दूल प्राप्स्यसे पालने रतः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2752"><div class="verse-block verse-justify"><div class="line">वेदाध्ययनशीलानां विप्राणां साधुकर्मणाम्।</div><div class="line" style="position:relative;">पालने यत्नमातिष्ठ सर्वलोकस्य चानघ॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2753"><div class="verse-block verse-justify"><div class="line">वने चरन्ति ये धर्ममाश्रमेषु च भारत।</div><div class="line" style="position:relative;">रक्षणात्तच्छतगुणं धर्मं प्राप्नोति पार्थिवः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2754"><div class="verse-block verse-justify"><div class="line">एष ते विविधो धर्मः पाण्डवश्रेष्ठ कीर्तितः।</div><div class="line" style="position:relative;">युधिष्ठिर त्वमेनं वै पूर्वं दृष्टं सनातनम्॥४३॥</div></div></div>
