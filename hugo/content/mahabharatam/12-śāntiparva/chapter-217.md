<div class="verse-block-wrapper" id="12-śāntiparva-b8928"><div class="verse-block verse-justify"><div class="line">गुरुरुवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8929"><div class="verse-block verse-justify"><div class="line">दुरन्तेष्विन्द्रियार्थेषु सक्ताः सीदन्ति जन्तवः।</div><div class="line" style="position:relative;">ये त्वसक्ता महात्मानस्ते यान्ति परमां गतिम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8930"><div class="verse-block verse-justify"><div class="line">जन्ममृत्युजरादुःखैर्व्याधिभिर्मानसक्लमैः।</div><div class="line" style="position:relative;">दृष्ट्वैव संततं लोकं घटेन्मोक्षाय बुद्धिमान्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8931"><div class="verse-block verse-justify"><div class="line">वाङ्भनोभ्यां शरीरेण शुचिः स्यादनहंकृतः।</div><div class="line" style="position:relative;">प्रशान्तो ज्ञानवान्भिक्षुर्निरपेक्षश्चरेत्सुखम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8932"><div class="verse-block verse-justify"><div class="line">`वशा मोक्षवतां पाशास्तासां रूपं प्रदर्शकम्।</div><div class="line" style="position:relative;">दुर्ग्रहं पश्यमानोऽपि मन्यते मोहितस्तदा॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8933"><div class="verse-block verse-justify"><div class="line">एवं पश्यन्तमात्मानमनुध्यातं हि बन्धुषु।</div><div class="line" style="position:relative;">अयथात्वेन जानामि भेदरूपेण संस्थितम् ॥'॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8934"><div class="verse-block verse-justify"><div class="line">अथवा मनसः सङ्गं पश्येद्भूतानुकम्पया।</div><div class="line" style="position:relative;">तत्राप्युपेक्षां कुर्वीत ज्ञात्वा कर्मफलं जगत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8935"><div class="verse-block verse-justify"><div class="line">यत्कृतं स्याच्छुभं कर्म पापं वा यदि वाऽश्नुते।</div><div class="line" style="position:relative;">तस्माच्छुभानि कर्माणि कुर्याद्वा बुद्धिकर्मभिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8936"><div class="verse-block verse-justify"><div class="line">अहिंसा सत्यवचनं सर्वभूतेषु चार्जवम्।</div><div class="line" style="position:relative;">क्षमा चैवाप्रमादश्च यस्यैते स सुखी भवेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8937"><div class="verse-block verse-justify"><div class="line">`अनक्षसाध्यं तद्ब्रह्म निर्मलं जगतः परम्।</div><div class="line" style="position:relative;">स्वात्मप्रकाशमग्राह्यमहेतुकमचञ्चलम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8938"><div class="verse-block verse-justify"><div class="line">विवेकज्ञानवाचिस्थो ह्याशुरूपेण संस्थितः।</div><div class="line" style="position:relative;">वैकारिकात्प्रदृश्येतै गैरिके मधुधारवत् ॥'॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8939"><div class="verse-block verse-justify"><div class="line">यश्चैनं परमं धर्मं सर्वभूतसुखावहम्।</div><div class="line" style="position:relative;">दुःखान्निः सरणं वेद तत्त्वज्ञः स सुखी भवेत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8940"><div class="verse-block verse-justify"><div class="line">तस्मात्समाहितं बुद्ध्या मनो भूतेषु धारयेत्।</div><div class="line" style="position:relative;">नापथ्यायेन्न स्पृहयेन्नाबद्धं चिन्तयेदसत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8941"><div class="verse-block verse-justify"><div class="line">अथामोघप्रयत्नेन मनो ज्ञाने निवेशयेत्।</div><div class="line" style="position:relative;">सुवाचोऽथ प्रयोगेण मनोज्ञं संप्रवर्तते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8942"><div class="verse-block verse-justify"><div class="line">विवेकयित्वा तद्वाक्यं धर्मसूक्ष्ममवेक्ष्य च।</div><div class="line" style="position:relative;">सत्यां वाचमहिंस्रां च वदेदनपवादिनीम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8943"><div class="verse-block verse-justify"><div class="line">कल्कापेतामपरुषामनृशंसामपैशुनीम्।</div><div class="line" style="position:relative;">ईदृगल्पं च वक्तव्यमविक्षिप्तेन चेतसा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8944"><div class="verse-block verse-justify"><div class="line">वाक्यबन्धेन संरागविहाराद्व्याहरेद्यदि।</div><div class="line" style="position:relative;">बुद्ध्याऽप्यनुगृहीतेन मनसा कर्म तामसम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8945"><div class="verse-block verse-justify"><div class="line">रजोभूतैर्हि करणैः कर्मणि प्रतिपद्यते।</div><div class="line">स दुःखं प्राप्य लोकेऽस्मिन्नरकायोपपद्यते।</div><div class="line" style="position:relative;">तस्मान्मनोवाक्शरीरैराचरेद्वैर्यमात्मनः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8946"><div class="verse-block verse-justify"><div class="line">प्रकीर्ण एव भारो हि यद्वद्धार्येत दस्युभिः।</div><div class="line">प्रतिलोमां दिशं बुद्ध्वा संसारमबुधास्तथा।</div><div class="line" style="position:relative;">`संसारमार्गमापन्नः प्रतिलोमं विवर्जयेत् ॥'॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8947"><div class="verse-block verse-justify"><div class="line">तामेव च यथा दस्यून्हत्वा गच्छेच्छिवां दिशम्।</div><div class="line" style="position:relative;">तथा रजस्तमः कर्माण्युत्सृज्य प्राप्नुयाच्छुभम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8948"><div class="verse-block verse-justify"><div class="line">निःसंदिग्धमनीहो वै मुक्तः सर्वपरिग्रहैः।</div><div class="line" style="position:relative;">विविक्तचारी लघ्वाशी तपस्वी नियतेन्द्रियः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8949"><div class="verse-block verse-justify"><div class="line">ज्ञानदग्धपरिक्लेशः प्रयोगरतिरात्मवान्।</div><div class="line" style="position:relative;">निष्प्रचारेण मनसा परं तदधिगच्छति॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b8952"><div class="verse-block verse-justify"><div class="line">विशेषहीनं गृह्णन्ति विशेषां कारणात्मिकाम्।</div><div class="line">अथवा न प्रभुस्तत्र परमात्मनि वर्तितुम्।</div><div class="line" style="position:relative;">आगामित्तत्त्वं योगात्मा योगतन्त्रमुपक्रमेत् ॥'॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b8954"><div class="verse-block verse-justify"><div class="line">ताभिः संयुक्तमनसो ब्रह्म तत्संप्रकाशते।</div><div class="line" style="position:relative;">शनैश्चापगते सत्वे ब्रह्मभूयाय कल्पते॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b8956"><div class="verse-block verse-justify"><div class="line">कणकुल्माषपिण्याकशाकयावकसक्तवः।</div><div class="line" style="position:relative;">तथा मूलफलं भैक्ष्यं पर्यायेणोपयोजयेत्॥२९॥</div></div></div>
