<div class="verse-block-wrapper" id="12-śāntiparva-b10281"><div class="verse-block verse-justify"><div class="line">ज्ञानेन यच्छेदात्मानं य इच्छेच्छान्तिमात्मनः।</div><div class="line" style="position:relative;">एतेषां चेदनुद्रष्टा पुरुषोऽपि सुदारुणः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10282"><div class="verse-block verse-justify"><div class="line">यदि वा सर्ववेदज्ञो यदि वाऽप्यनृचो द्विजः।</div><div class="line" style="position:relative;">यदि वा धार्मिको यज्वा यदि वा पापकृत्तमः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10283"><div class="verse-block verse-justify"><div class="line">यदि वा पुरुषव्याघ्रो यदि वैक्लव्यधारणः।</div><div class="line" style="position:relative;">तरत्येवं महादुर्गं जरामरणसागरम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10284"><div class="verse-block verse-justify"><div class="line">एवं ह्येतेन योगेन युञ्जानो ह्येवमन्ततः।</div><div class="line" style="position:relative;">अपि जिज्ञासमानोऽपि शब्दब्रह्माऽतिवर्तते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10285"><div class="verse-block verse-justify"><div class="line">धर्मोपस्थो ह्रीवरूथ उपायापायकूवरः।</div><div class="line" style="position:relative;">अपानाक्षः प्राणयुगः प्रज्ञायुर्जीववन्धनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10286"><div class="verse-block verse-justify"><div class="line">चेतनाबन्धुरश्चारुश्चाचारग्रहनेमिमान्।</div><div class="line" style="position:relative;">दर्शनस्पर्शनवहो घ्राणश्रवणवाहनः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10287"><div class="verse-block verse-justify"><div class="line">प्रज्ञानाभिः सर्वतन्त्रप्रतोदो ज्ञानसारथिः।</div><div class="line" style="position:relative;">क्षेत्रज्ञाधिष्ठितो धीरः श्रद्धादमपुरः सरः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10288"><div class="verse-block verse-justify"><div class="line">त्यागरश्म्यनुगः क्षेम्यः शौचगो ध्यानगोतरः।</div><div class="line" style="position:relative;">जीवयुक्तो रथो दिव्यो ब्रह्मलोके धिराजते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10289"><div class="verse-block verse-justify"><div class="line">अथ संत्वरमाणस्य रथमेवं युयुक्षतः।</div><div class="line" style="position:relative;">अक्षरं गन्तुमनसो विधिं वक्ष्यामि शीघ्रगम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10290"><div class="verse-block verse-justify"><div class="line">सप्त यो धारणाः कृत्स्ना वाग्यतः प्रतिपद्यते।</div><div class="line" style="position:relative;">पृष्ठतः पार्श्वतश्चान्यास्तावत्यस्ताः प्रधारणाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10291"><div class="verse-block verse-justify"><div class="line">क्रमशः पार्थिवं यच्च वायव्यं खं तथा पयः।</div><div class="line">ज्योतिषो यत्तदैश्वर्यमहंकारस्य बुद्धितः।</div><div class="line" style="position:relative;">अव्यक्तस्य तथैश्वर्यं क्रमशः प्रतिपद्यते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10292"><div class="verse-block verse-justify"><div class="line">विक्रमाश्चापि यस्यैते तथा युङ्क्ते स योगतः।</div><div class="line" style="position:relative;">तथाऽस्य योगयुक्तस्य सिद्धिमात्मनि पश्यतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10293"><div class="verse-block verse-justify"><div class="line">निर्मुच्यमानः सूक्ष्मत्वाद्रूपाणीमानि पश्यतः।</div><div class="line" style="position:relative;">शैशिरस्तु यथा धूमः सूक्ष्मः संश्रयते नभः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10294"><div class="verse-block verse-justify"><div class="line">तथा देहाद्विमुक्तस्य पूर्वरूपं भवत्युत।</div><div class="line" style="position:relative;">अथ धूमस्य विरमेद्द्वितीयं रूपदर्शनम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10295"><div class="verse-block verse-justify"><div class="line">जलरूपमिवाकाशे तत्रैवात्मनि पश्यति।</div><div class="line" style="position:relative;">अपां व्यतिक्रमे चास्य वह्निरूपं प्रकाशते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10296"><div class="verse-block verse-justify"><div class="line">तस्मिन्नुपरते चास्य वायव्यं सूक्ष्ममव्ययम्।</div><div class="line" style="position:relative;">रूपं प्रकाशते तस्य पीतवस्त्रवदव्ययम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10297"><div class="verse-block verse-justify"><div class="line">तस्मिन्नुपरते रुपमाकाशस्य प्रकाशते।</div><div class="line">तस्मिन्नुपरते चास्य बुद्धिरूपं प्रकाशते।</div><div class="line" style="position:relative;">ऊर्णारूपसवर्णस्य तस्य रूपं प्रकाशते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10298"><div class="verse-block verse-justify"><div class="line">अथ श्वेतां गतिं गत्वा सोहङ्कारे प्रकाशते।</div><div class="line" style="position:relative;">सुशुक्लं चेतसः सौक्ष्म्यमप्युक्तं ब्राह्मणस्य वै॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10299"><div class="verse-block verse-justify"><div class="line">एतेष्वपि हि जातेषु फलजातानि मे शृणु।</div><div class="line" style="position:relative;">जातस्य पार्थिवैश्वर्यैः सृष्टिरिष्टा विधीयते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10300"><div class="verse-block verse-justify"><div class="line">प्रजापतिरिवाक्षोभ्यः शरीरात्सृजते प्रजाः।</div><div class="line" style="position:relative;">अङ्गुल्यङ्गुष्ठमात्रेण हस्तपादेन वा तथा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10301"><div class="verse-block verse-justify"><div class="line">पृथिवीं कम्पयत्येको गुणो वायोरिति श्रुतिः।</div><div class="line">आकाशभूतश्चाकाशे सवर्णत्वात्प्रकाशते।</div><div class="line" style="position:relative;">वर्णतो गृह्यते चाप्सु नापः पिबति चाशया॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10302"><div class="verse-block verse-justify"><div class="line">न चास्य तेजसां रूपं दृश्यते शाम्यते तथा।</div><div class="line" style="position:relative;">अहंकारेऽस्य विजिते पञ्चैते स्युर्वशानुगाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10303"><div class="verse-block verse-justify"><div class="line">षण्णामात्मनि बुद्धौ च जितायां प्रभवत्यथ।</div><div class="line" style="position:relative;">निर्दोषा प्रतिभा ह्येनं कृत्स्ना समभिवर्तते॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b10305"><div class="verse-block verse-justify"><div class="line">तत्राव्यक्तमयीं विद्यां शृणु त्वं विस्तरेण मे।</div><div class="line" style="position:relative;">तथा व्यक्तमयं चैव संख्यापूर्वं निबोध मे॥३०॥</div></div></div>
