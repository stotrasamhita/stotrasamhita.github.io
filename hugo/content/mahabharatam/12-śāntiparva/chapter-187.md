<div class="verse-block-wrapper" id="12-śāntiparva-b7609"><div class="verse-block verse-justify"><div class="line">भरद्वाज उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7610"><div class="verse-block verse-justify"><div class="line">ब्राह्मणः केन भवति क्षत्रियो वा द्विजोत्तम।</div><div class="line">वैश्यः शूद्रश्च विप्रर्षे तद्ब्रूहि वदतां वर ॥भृगुरुवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7611"><div class="verse-block verse-justify"><div class="line">जातकर्मादिभिर्यस्तु संस्कारैः संस्कृतः शुचिः।</div><div class="line" style="position:relative;">वेदाध्ययनसंपन्नः षट््सु कर्मस्ववस्थितः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7612"><div class="verse-block verse-justify"><div class="line">शौचाचारस्थितः सम्यग्विघसाशी गुरुप्रियः।</div><div class="line" style="position:relative;">नित्यव्रती सत्यपरः स वै ब्राह्मण उच्यते॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7613"><div class="verse-block verse-justify"><div class="line">सत्यं दानमथाद्रोह आनृशंस्यं क्षमा धृणा।</div><div class="line" style="position:relative;">तपश्च दृश्यते यत्र स ब्राह्मण इति स्मृतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7614"><div class="verse-block verse-justify"><div class="line">क्षत्रजं सेवते कर्म देवाध्ययनसंगतः।</div><div class="line" style="position:relative;">दानादानरतिर्यस्तु स वै क्षत्रिय उच्यते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7615"><div class="verse-block verse-justify"><div class="line">कृपिगोरक्षवाणिज्यं यो विशत्यनिशं शुचिः।</div><div class="line" style="position:relative;">वेदाध्ययनसंपन्नः स वैश्य इति संज्ञितः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7616"><div class="verse-block verse-justify"><div class="line">सर्वभक्षरतिर्नित्यं सर्वकर्मकरोऽशुचिः।</div><div class="line" style="position:relative;">त्यक्तवेदस्त्वनाचारः स वै शूद्र इति स्मृतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7617"><div class="verse-block verse-justify"><div class="line">शूद्रे चैतद्भवेल्लक्ष्यं द्विजे तच्च न विद्यते।</div><div class="line" style="position:relative;">न वै शूद्रो भवेच्छ्रद्रो ब्राह्मणो न च ब्राह्मणः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7618"><div class="verse-block verse-justify"><div class="line">सर्वोपायैस्तु लोभस्य क्रोधस्य च विनिग्रहः।</div><div class="line" style="position:relative;">एतत्पवित्रं ज्ञातव्यं तथा चैवात्मसंयमः॥१०॥</div></div></div>
<p class="prose">वार्यौ सर्वात्मना तौ हि श्रेयोघातार्थमुच्छ्रितौ</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b7620"><div class="verse-block verse-justify"><div class="line">नित्यं क्रोधाच्छ्रियं रक्षेत्तपो रक्षेच्च मत्सरात्।</div><div class="line" style="position:relative;">विद्यां मानापमानाभ्यामात्मानं तु प्रमादतः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7621"><div class="verse-block verse-justify"><div class="line">यस्य सर्वे समारम्भा निराशाबन्धना द्विज।</div><div class="line" style="position:relative;">त्यागे यस्य हुतं सर्वं स त्यागी च स बुद्धिमान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7622"><div class="verse-block verse-justify"><div class="line">अहिंस्रः सर्वभूतानां मैत्रायणगतिश्चरेत्।</div><div class="line">परिग्रहान्परित्यज्य भवेद्बुद्ध्या जितेन्द्रियः।</div><div class="line" style="position:relative;">अचलं स्थानमातिष्ठेदिह चामुत्र चोभयोः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7623"><div class="verse-block verse-justify"><div class="line">तपोनित्येन दान्तेन मुनिना संयतात्मना।</div><div class="line" style="position:relative;">अजितं जेतुकामेन भाव्यं सङ्गेष्वसङ्गिना॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7624"><div class="verse-block verse-justify"><div class="line">इन्द्रियैर्गृह्यते यद्यत्तत्तद्व्यक्तमिति स्थितिः।</div><div class="line" style="position:relative;">अव्यक्तमिति विज्ञेयं लिङ्गग्राह्यमतीन्द्रियम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7625"><div class="verse-block verse-justify"><div class="line">अविस्रम्भे न गन्तव्यं विस्रम्भे धारयेन्मनः।</div><div class="line" style="position:relative;">मनः प्राणे निगृह्णीयात्प्राणं ब्रह्मणि धारयेत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7626"><div class="verse-block verse-justify"><div class="line">निर्वेदादेव निर्वायान्न च किंचिद्विचिन्तयेत्।</div><div class="line" style="position:relative;">सुखं वै ब्राह्मणो ब्रह्म स वै तेनाधिगच्छति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7627"><div class="verse-block verse-justify"><div class="line">शौचेन सततं युक्तः सदाचारसमन्वितः।</div><div class="line" style="position:relative;">सानुक्रोशश्च भूतेषु तद्द्विजातिषु लक्षणम्॥१८॥</div></div></div></div>
