<div class="verse-block-wrapper" id="12-śāntiparva-b12583"><div class="verse-block verse-justify"><div class="line">एवमेतन्महाराज येन जातः स एव सः।</div><div class="line" style="position:relative;">तपसस्त्वपकर्षेण जातिग्रहणतां गतः॥४॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b12586"><div class="verse-block verse-justify"><div class="line">मुखजा ब्राह्मणास्तात बाहुजाः क्षत्रियाः स्मृताः।</div><div class="line" style="position:relative;">ऊरुजा धनिनो राजन्पादजाः परिचारकाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12587"><div class="verse-block verse-justify"><div class="line">चतुर्णामेव वर्णानामागमः पुरुषर्षभ।</div><div class="line" style="position:relative;">अतोन्ये त्वतिरिक्ता ये ते वै संकरजाः स्मृताः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12588"><div class="verse-block verse-justify"><div class="line">क्षत्रियातिरथाम्बष्ठा उग्रा वैदेहकास्तथा।</div><div class="line" style="position:relative;">श्वपाकाः पुल्कसाः स्तेना निषादाः सूतमागधाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12589"><div class="verse-block verse-justify"><div class="line">अयोगाः कारणा व्रात्याश्चाण्डालाश्च नराधिप।</div><div class="line">एते चतुर्भ्यो वर्णेभ्यो जायन्ते वै परस्परात् ॥जनक उवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12590"><div class="verse-block verse-justify"><div class="line">ब्रह्मणैकेन जातानां नानात्वं गोत्रतः कथम्।</div><div class="line" style="position:relative;">बहूनीह हि लोके वै गोत्राणि मुनिसत्तम॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12591"><div class="verse-block verse-justify"><div class="line">यत्र तत्र कथं जाताः स्वयोनिं मुनयो गताः।</div><div class="line">शूद्रयोनौ समुत्पन्ना वियोनौ च तथा परे ॥पराशर उवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12592"><div class="verse-block verse-justify"><div class="line">राजन्नैतद्भवेद्ब्राह्ममपकृष्टेन जन्मना।</div><div class="line" style="position:relative;">महात्मनां समुत्पत्तिस्तपसा भावितात्मनाम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12593"><div class="verse-block verse-justify"><div class="line">उत्पाद्य पुत्रान्मुनयो नृपते यत्र तत्र ह।</div><div class="line" style="position:relative;">स्वेनैव तपसा तेषामृषित्वं विदधुः पुनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12594"><div class="verse-block verse-justify"><div class="line">पितामहश्च मे पूर्वमृश्यशृङ्गश्च काश्यपः।</div><div class="line" style="position:relative;">वेदस्ताण्ड्यः कृपश्चैव काक्षीवत्कमठादयः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12595"><div class="verse-block verse-justify"><div class="line">यवक्रीतश्च नृपते द्रोणश्च वदतांवरः।</div><div class="line" style="position:relative;">आयुर्मतङ्गो दत्तश्च द्रुमदो मात्स्य एव च॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12596"><div class="verse-block verse-justify"><div class="line">एते स्वां प्रकृतिं प्राप्ता वैदेह तपसो बलात्।</div><div class="line" style="position:relative;">प्रतिष्ठिता वेदविदो दमेन तपसैव हि॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b12598"><div class="verse-block verse-justify"><div class="line">कर्मतोऽन्यानि गोत्राणि समुत्पन्नानि पार्थिव।</div><div class="line">नामधेयानि तपसा तानि च ग्रहणं सताम् ॥जनक उवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12599"><div class="verse-block verse-justify"><div class="line">विशेषधर्मान्वर्णानां प्रब्रूहि भगवन्मम।</div><div class="line">ततः सामान्यधर्मांश्च सर्वत्र कुशलो ह्यसि ॥पराशर उवाच।</div><div class="line" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12600"><div class="verse-block verse-justify"><div class="line">प्रतिग्रहो याजनं च तथैवाध्यापनं नृप।</div><div class="line" style="position:relative;">विशेषधर्मा विप्राणां रक्षा क्षत्रस्य शोभना॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12601"><div class="verse-block verse-justify"><div class="line">कृषिश्च पाशुपाल्यं च वाणिज्यं च विशामपि।</div><div class="line" style="position:relative;">द्विजानां परिचर्या च शूद्रकर्म नराधिप॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12602"><div class="verse-block verse-justify"><div class="line">विशेषधर्मा नृपते वर्णानां परिकीर्तिताः।</div><div class="line" style="position:relative;">धर्मान्साधारणांस्तात विस्तरेण शृणुष्व मे॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12603"><div class="verse-block verse-justify"><div class="line">आनृशंस्यमर्हिसा चाप्रमादः संविभागिता।</div><div class="line" style="position:relative;">श्राद्धकर्मातिथेयं च सत्यमक्रोध एव च॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b12605"><div class="verse-block verse-justify"><div class="line">ब्राह्मणाः क्षत्रिया वैश्यास्रयो वर्णा द्विजातयः।</div><div class="line" style="position:relative;">अत्र तेषामधीकारो धर्मेषु द्विपदां वर॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12606"><div class="verse-block verse-justify"><div class="line">विकर्मावस्थिता वर्णाः पतन्ति नृपते त्रयः।</div><div class="line" style="position:relative;">उन्नमन्ति यथा सन्त आश्रित्येह स्वकर्मसु॥२७॥</div></div></div>
