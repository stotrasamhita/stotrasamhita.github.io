<div class="stotra-article"><h2 class="stotra-heading">अध्यायः १८३</h2>
<div class="verse-block-wrapper" id="12-śāntiparva-b7518"><div class="verse-block verse-justify"><div class="line">भरद्वाज उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7519"><div class="verse-block verse-justify"><div class="line">पार्थिवं धातुमाश्रित्य शारीरोऽग्निः कथं भवेत्।</div><div class="line">अवकाशविशेषेण कथं वर्तयतेऽनिलः ॥भृगुरुवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7520"><div class="verse-block verse-justify"><div class="line">वायोर्गतिमहं ब्रह्मन्कीर्तयिष्यामि तेऽनघ।</div><div class="line" style="position:relative;">प्राणिनामनिलो देहान्यथा चेष्टयते बली॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7521"><div class="verse-block verse-justify"><div class="line">श्रितो मूर्धानमग्निस्तु शरीरं परिपालयन्।</div><div class="line" style="position:relative;">प्राणो मूर्धनि चाग्नौ च वर्तमानो विचेष्टने॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7522"><div class="verse-block verse-justify"><div class="line">स जन्तुः सर्वभूतात्मा पुरुषः स सनातनः।</div><div class="line" style="position:relative;">मनो बुद्धिरहंकारो भूतानि विषयाश्च सः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7523"><div class="verse-block verse-justify"><div class="line">एवं त्विह स सर्वत्र प्राणेन परिपाल्यते।</div><div class="line" style="position:relative;">कोष्ठतस्तु समानेन स्वां स्वां गतिमुपाश्रितः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7524"><div class="verse-block verse-justify"><div class="line">वस्तिमूलं गुदं चैव पावकं समुपाश्रितः।</div><div class="line" style="position:relative;">वहन्मूत्रं पुरीषं चाप्यपानः परिवर्तते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7525"><div class="verse-block verse-justify"><div class="line">प्रयत्ने कर्मणि बले य एकस्त्रिषु वर्तते।</div><div class="line" style="position:relative;">उदान इति तं प्राहुरध्यात्मकुशला जनाः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7526"><div class="verse-block verse-justify"><div class="line">संधिष्वपि च सर्वेषु सन्निविष्टस्तथाऽनिलः।</div><div class="line" style="position:relative;">शरीरेषु मनुष्याणां व्यान इत्युपदिश्यते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7527"><div class="verse-block verse-justify"><div class="line">धातुष्वग्निस्तु विततः समानोऽग्निः समीरितः।</div><div class="line" style="position:relative;">रसान्धातूंश्च दोषांश्च वर्तयन्नवतिष्ठते॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7528"><div class="verse-block verse-justify"><div class="line">अपानप्राणयोर्मध्ये प्राणापानसमाहितः।</div><div class="line" style="position:relative;">समन्वितस्त्वधिष्ठानं सम्यक्पचति पावकः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7529"><div class="verse-block verse-justify"><div class="line">आस्यं हि पायुसंयुक्तमन्ते स्याद्गुदसंज्ञितम्।</div><div class="line" style="position:relative;">स्रोतस्तस्मात्प्रजायन्ते सर्वस्रोतांसि देहिनाम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7530"><div class="verse-block verse-justify"><div class="line">प्राणानां सन्निपाताच्च सन्निपातः प्रजायते।</div><div class="line" style="position:relative;">ऊष्मा सोग्निरिति ज्ञेयो योऽन्नं पचति देहिनाम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7531"><div class="verse-block verse-justify"><div class="line">अग्निवेगवहः प्राणो गुदान्ते प्रतिहन्यते।</div><div class="line" style="position:relative;">स ऊर्ध्वमागम्य पुनः समुत्क्षिपति पावकम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7532"><div class="verse-block verse-justify"><div class="line">पक्वाशयस्त्वधो नाभ्या ऊर्ध्वमामाशयः स्मृतः।</div><div class="line" style="position:relative;">नाभिमध्ये शरीरस्य सर्वे प्राणाः समाश्रिताः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7533"><div class="verse-block verse-justify"><div class="line">प्रसृता हृदयात्सर्वास्तिर्यगूर्ध्वमधस्तथा।</div><div class="line" style="position:relative;">वहन्त्यन्नरसान्नाड्यो दशप्राणप्रचोदिताः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7534"><div class="verse-block verse-justify"><div class="line">एष मार्गोऽथ योगानां येन गच्छन्ति तत्पदम्।</div><div class="line" style="position:relative;">जितक्लमासना धीरा सूर्धन्यात्मानमादधन्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b7535"><div class="verse-block verse-justify"><div class="line">एवं सर्वेषु विहितः प्राणापानेषु देहिनाम्।</div><div class="line">तस्मिन्योऽवस्थितो नित्यमग्निः स्थाल्यामिवाहितः।</div><div class="line" style="position:relative;">॥१८॥</div></div></div></div>
