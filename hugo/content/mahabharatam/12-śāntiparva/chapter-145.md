<div class="verse-block-wrapper" id="12-śāntiparva-b6125"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6126"><div class="verse-block verse-justify"><div class="line">सपत्न्या वचनं श्रुत्वा धर्मयुक्तिसमन्वितम्।</div><div class="line" style="position:relative;">हर्षेण महता युक्तो वाक्यं व्याकुललोचनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6127"><div class="verse-block verse-justify"><div class="line">तं वै शाकुनिकं दृष्ट्वा विधिदृष्टेन कर्मणा।</div><div class="line" style="position:relative;">स पक्षी पूजयामास यत्नात्तं पक्षिजीविनम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6128"><div class="verse-block verse-justify"><div class="line">उवाच स्वागतं तेऽद्य ब्रूहि किं करवाणि ते।</div><div class="line" style="position:relative;">सतांपश्च न कर्तव्यः स्वगृहे वर्तते भवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6129"><div class="verse-block verse-justify"><div class="line">तद्ब्रवीतु भवान्क्षिप्रं किं करोमि किमिच्छसि।</div><div class="line" style="position:relative;">प्रणयेन ब्रवीमि त्वां त्वं हि नः शरणागतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6130"><div class="verse-block verse-justify"><div class="line">अरावप्युचितं कार्यमातिथ्यं गृहमागते।</div><div class="line" style="position:relative;">छेत्तुमप्यागते छायां नोपसंहरते द्रुमः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6131"><div class="verse-block verse-justify"><div class="line">शरणागतस्य कर्तव्यमातिथ्यं हि प्रयत्नतः।</div><div class="line" style="position:relative;">पञ्चयज्ञप्रवृत्तेन गृहस्थेन विशेषतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6132"><div class="verse-block verse-justify"><div class="line">पञ्चयज्ञांस्तु यो मोहान्न करोति गृहाश्रमी।</div><div class="line" style="position:relative;">तस्य नायं न च परो लोको भवति धर्मतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6133"><div class="verse-block verse-justify"><div class="line">तद्ब्रूहि मां सुविस्रब्धो यत्त्वं वाचा वदिष्यसि।</div><div class="line" style="position:relative;">तत्करिष्याम्यहं सर्वं मा त्वं शोके मनः कृथाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6134"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा शकुनेर्लुब्धकोऽब्रवीत्।</div><div class="line" style="position:relative;">बाधते खलु मां शीतं संत्राणं हि विधीयताम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6135"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तनः पक्षी पर्णान्यास्तीर्य भूतले।</div><div class="line" style="position:relative;">यथा शुष्काणि यत्नेन ज्वलनार्थं द्रुतं ययौ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6136"><div class="verse-block verse-justify"><div class="line">स---वाऽङ्गारकर्मान्तं गृहीत्वाऽग्निमथागमत्।</div><div class="line" style="position:relative;">तथा शुष्केषु पर्णेषु पावकं सोऽप्यदीपयत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6137"><div class="verse-block verse-justify"><div class="line">स--प्तं महत्कृत्वा तमाह शरणागतम्।</div><div class="line" style="position:relative;">----- सुविस्रब्धः स्वगात्राण्यकुतोभयः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6138"><div class="verse-block verse-justify"><div class="line">-- तथोक्तस्तथेत्युक्त्वा लुब्धो गात्राण्यतापयत्।</div><div class="line" style="position:relative;">अग्निप्रत्यागतप्राणस्ततः प्राह विहंगमम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6139"><div class="verse-block verse-justify"><div class="line">हर्षेण महताऽऽविष्टो वाक्यं व्याकुललोचनः।</div><div class="line" style="position:relative;">तथेमं शकुनिं दृष्ट्वा विधिदृष्टेन कर्मणा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6140"><div class="verse-block verse-justify"><div class="line">दत्तमाहारमिच्छामि त्वया क्षुद्बाधते हि माम्।</div><div class="line" style="position:relative;">स तद्वचः प्रतिश्रुत्य वाक्यमाह विहंगमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6141"><div class="verse-block verse-justify"><div class="line">न मेऽस्ति विभवो येन नाशयेयं क्षुधां तव।</div><div class="line" style="position:relative;">उत्पन्नेन हि जीवामो वयं नित्यं वनौकसः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6142"><div class="verse-block verse-justify"><div class="line">संचयो नास्ति चास्माकं मुनीनामिव कानने।</div><div class="line" style="position:relative;">इत्युक्त्वा तं तदा तत्र विवर्णवदनोऽभवत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6143"><div class="verse-block verse-justify"><div class="line">कथं नु खलु कर्तव्यमिति चिन्तापरस्तदा।</div><div class="line" style="position:relative;">बभूव भरतश्रेष्ठ गर्हयन्वृत्तिमात्मनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6144"><div class="verse-block verse-justify"><div class="line">मुहूर्ताल्लब्धसंज्ञस्तु स पक्षी पक्षिघातिनम्।</div><div class="line" style="position:relative;">उवाच तर्पयिष्ये त्वां मुहूर्तं प्रतिपालय॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6145"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा शुष्कपर्णैस्तु समुज्ज्वाल्य हुताशनम्।</div><div class="line" style="position:relative;">हर्षेण महताऽऽविष्टः कपोतः पुनरब्रबीत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6146"><div class="verse-block verse-justify"><div class="line">ऋषीणां देवतानां च पितृणां च महात्मनाम्।</div><div class="line" style="position:relative;">युतः पूर्वं मया धर्मो महानतिथिपूजने॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6147"><div class="verse-block verse-justify"><div class="line">कुरुष्वानुग्रहं सौम्य सत्यमेतद्ब्रबीमि ते।</div><div class="line" style="position:relative;">निश्चिता खलु मे बुद्धिरतिथिप्रतिपूजने॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6148"><div class="verse-block verse-justify"><div class="line">ततः कृतप्रतिज्ञो वै स पक्षी प्रहसन्निव।</div><div class="line" style="position:relative;">तमग्निं त्रिः परिक्रम्य प्रविवेश महामतिः॥२४॥</div></div></div>
