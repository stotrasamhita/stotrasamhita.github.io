<div class="verse-block-wrapper" id="12-śāntiparva-b13432"><div class="verse-block verse-justify"><div class="line">नास्ति साङ्ख्यसमं ज्ञानं नास्ति योगसमं बलम्।</div><div class="line" style="position:relative;">तावुभावेकचर्यौ तावुभावनिधनौ स्मृतौ॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13433"><div class="verse-block verse-justify"><div class="line">पृथक्पृथक्प्रपश्यन्ति येऽप्यबुद्धिरता नराः।</div><div class="line" style="position:relative;">वयं तु राजन्पश्याम एकमेव तु निश्चयात्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13434"><div class="verse-block verse-justify"><div class="line">यदेव योगाः पश्यन्ति तत्साङ्ख्यैरपि दृश्यते।</div><div class="line" style="position:relative;">एकं साङ्ख्यं च योगं च यः पश्यति स तत्त्ववित्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13435"><div class="verse-block verse-justify"><div class="line">रुद्रप्रधानानपरान्विद्धि योगानरिंदम्।</div><div class="line" style="position:relative;">तेनैव चाथ देहेन विचरन्ति दिशो दश॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13436"><div class="verse-block verse-justify"><div class="line">यावद्धि प्रलयस्तात सूक्ष्मेणाष्टगुणेन ह।</div><div class="line" style="position:relative;">योगेन लोकान्विचरन्सुखं संन्यस्य चानघ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13437"><div class="verse-block verse-justify"><div class="line">तावदेवाष्टगुणिनं योगप्राहुर्मनीषिणः।</div><div class="line" style="position:relative;">सूक्ष्ममष्टगुणं प्राहुर्नेतरं नृपसत्तम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13438"><div class="verse-block verse-justify"><div class="line">द्विगुणं योगत्यं तु योगानां प्राहुरुत्तमम्।</div><div class="line" style="position:relative;">सगुणं निर्गुणं चैव यथाशास्त्रनिदर्शनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13439"><div class="verse-block verse-justify"><div class="line">धारणं चैव मनसः प्राणायामश्च पार्थिव।</div><div class="line" style="position:relative;">एकाग्रता च मनसः प्राणायामस्तथैव च॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13440"><div class="verse-block verse-justify"><div class="line">प्राणायामो हि सगुणो निर्गुणं धारयेन्मनः।</div><div class="line">यद्यदृश्यति मुञ्चन्वै प्राणान्मैथिलसत्तम।</div><div class="line" style="position:relative;">वाताधिक्यं भवत्येव तस्मात्तं न समाचरेत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13441"><div class="verse-block verse-justify"><div class="line">निशायाः प्रथमे यामे चोदना द्वादश स्मृताः।</div><div class="line" style="position:relative;">मध्ये स्वप्नात्परे यामे द्वादशैव तु चोदनाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13442"><div class="verse-block verse-justify"><div class="line">तदेवमुपशान्तेन दान्तेनैकान्तशीलिना।</div><div class="line" style="position:relative;">आत्मारामेण बुद्धेन योक्तव्योऽऽत्मा न संशयः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13443"><div class="verse-block verse-justify"><div class="line">पञ्चानामिन्द्रियाणां तु दोषानाक्षिप्य पञ्चधा।</div><div class="line" style="position:relative;">शब्दं रूपं तथा स्पर्शं रसं गन्धं तथैव च॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13444"><div class="verse-block verse-justify"><div class="line">प्रतिभामपवर्गं च प्रतिसंहृत्य मैथिल।</div><div class="line" style="position:relative;">इन्द्रियग्राममखिलं मनस्यभिनिवेश्य ह॥१५॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b13446"><div class="verse-block verse-justify"><div class="line">एवं हि परिसंख्याय ततो ध्यायन्ति केवलम्।</div><div class="line" style="position:relative;">विरजस्कमलं नित्यमनन्तं शुद्धमव्रणम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13447"><div class="verse-block verse-justify"><div class="line">तस्थुषं पुरुषं नित्यमभेद्यमजरामरम्।</div><div class="line" style="position:relative;">शाश्वतं चाव्ययं चैव ईशानं ब्रह्म चाख्यम्॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b13450"><div class="verse-block verse-justify"><div class="line">पाषाण इव मेघोत्थैर्यथा बिन्दुभिराहतः।</div><div class="line" style="position:relative;">नालं चालयितुं शक्यस्तथा युक्तस्य लक्षणम्॥२१॥</div></div></div>
