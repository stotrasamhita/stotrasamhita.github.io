<div class="verse-block-wrapper" id="12-śāntiparva-b9344"><div class="verse-block verse-justify"><div class="line">दममेव प्रशंसन्ति वृद्धाः श्रुतिसमाधयः।</div><div class="line" style="position:relative;">सर्वेषामेव वर्णानां ब्राह्मणस्य विशेषतः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9345"><div class="verse-block verse-justify"><div class="line">नादान्तस्य क्रियासिद्धिर्यथावदुपपद्यते।</div><div class="line" style="position:relative;">क्रिया तपश्च देवाश्च दमे सर्वं प्रतिष्ठितम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9346"><div class="verse-block verse-justify"><div class="line">दमस्तेजो वर्धयति पवित्रं दम उच्यते।</div><div class="line" style="position:relative;">विपाष्मा निर्भयो दान्तः पुरुषो विन्दते महत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9347"><div class="verse-block verse-justify"><div class="line">सुखं दान्तः प्रस्वपिति सुखं च प्रतिबुध्यते।</div><div class="line" style="position:relative;">सुखं लोके विपर्येति मनश्चास्य प्रसीदति॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9348"><div class="verse-block verse-justify"><div class="line">तेजो दमन ध्रियते तत्र तीक्ष्णोऽधिगच्छति।</div><div class="line" style="position:relative;">अमित्रांश्च बहून्नित्यं पृथगात्मनि पश्यति॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9349"><div class="verse-block verse-justify"><div class="line">क्रव्याद्भ्य इव भूतानामदान्तेभ्यः सदा भयम्।</div><div class="line" style="position:relative;">तेषां विप्रतिषेधार्थं राजा सृष्टः स्वयंभुवा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9350"><div class="verse-block verse-justify"><div class="line">आश्रमेषु च सर्वेषु दम एव विशिष्यते।</div><div class="line">12-225-8b`धर्मः संरक्ष्यतेतैस्तु यतस्ते धर्मसेतवः।</div><div class="line" style="position:relative;">' यच्च तेषु फलं धर्म्यं भूयो दान्ते तदुच्यते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9351"><div class="verse-block verse-justify"><div class="line">तेषां लिङ्गानि वक्ष्यानि येषां समुदयो दमः।</div><div class="line" style="position:relative;">अकार्पण्यमसंरम्भः संतोषः श्रद्दधानता॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9352"><div class="verse-block verse-justify"><div class="line">अक्रोध आर्जवं नित्यं नातिवादोऽभिमानिता।</div><div class="line" style="position:relative;">गुरुपूजाऽनसूया च दया भूतेष्वपैशुनम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9353"><div class="verse-block verse-justify"><div class="line">जनवादमृषावादस्तुतिनिन्दाविवर्जनम्।</div><div class="line" style="position:relative;">साधुकामांश्च स्पृहयेन्नायतिं प्रत्ययेषु च॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9354"><div class="verse-block verse-justify"><div class="line">अवैरकृत्सूपचारः समो निन्दाप्रशंसयोः।</div><div class="line" style="position:relative;">सुवृत्तः शीलसंपन्नः प्रसन्नात्माऽऽत्मवाञ्शुचिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9355"><div class="verse-block verse-justify"><div class="line">प्राप्य लोके च सत्कारं स्वर्गं वै प्रेत्य गच्छति।</div><div class="line" style="position:relative;">दुर्गमं सर्वभूतानां प्रापयन्मोदते सुखी॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9356"><div class="verse-block verse-justify"><div class="line">सर्वभूतहिते युक्तो न स्म यो द्विषते जनम्।</div><div class="line" style="position:relative;">महाह्रद इवाक्षोभ्यः प्राज्ञस्तृप्तः प्रसीदति॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9357"><div class="verse-block verse-justify"><div class="line">अभयं यस्य भूतेभ्यः सर्वेषामभयं यतः।</div><div class="line" style="position:relative;">नमस्यः सर्वभूतानां दान्तो भवति बुद्धिमान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b9358"><div class="verse-block verse-justify"><div class="line">न हृष्यति महत्यर्थे व्यसने च न शोचति।</div><div class="line" style="position:relative;">सदाऽपरिमितप्रज्ञः स दान्तो द्विज उच्यते॥१७॥</div></div></div>
