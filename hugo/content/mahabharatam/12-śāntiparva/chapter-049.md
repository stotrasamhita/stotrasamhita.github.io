<div class="verse-block-wrapper" id="12-śāntiparva-b2026"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2027"><div class="verse-block verse-justify"><div class="line">ततो रामस्य तत्कर्म श्रुत्वा राजा युधिष्ठिरः।</div><div class="line" style="position:relative;">विस्मयं परमं गत्वा प्रत्युवाच जनार्दनम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2028"><div class="verse-block verse-justify"><div class="line">अहो रामस्य वार्ष्णेय शक्रस्येव महात्मनः।</div><div class="line" style="position:relative;">विक्रमो वसुधा येन क्रोधान्निःक्षत्रिया कृता॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2029"><div class="verse-block verse-justify"><div class="line">गोभिः समुद्रेण तथा गोलाङ्गूलर्क्षवानरैः।</div><div class="line" style="position:relative;">गुप्ता रामभयोद्विग्नाः क्षत्रियाणां कुलोद्वहाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2030"><div class="verse-block verse-justify"><div class="line">अहो धन्यो नृलोकोऽयं समाग्याश्च नरा भुवि।</div><div class="line" style="position:relative;">यत्र कर्मेदृशं धर्म्यं द्विजाग्र्यैः कृतमच्युत॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2031"><div class="verse-block verse-justify"><div class="line">कथयन्तौ कथां तात तावच्युतयुधिष्ठिरौ।</div><div class="line" style="position:relative;">जग्मतुर्यत्र गाङ्गेयः शरतल्पगतः प्रभुः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2032"><div class="verse-block verse-justify"><div class="line">ततस्ते ददृशुर्भीष्मं शरप्रस्तरशायिनम्।</div><div class="line" style="position:relative;">स्वरश्मिमालासंवीतं सायंसूर्यसमप्रभम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2033"><div class="verse-block verse-justify"><div class="line">उपास्यमानं मुनिभिर्देवैरिव शतक्रतुम्।</div><div class="line" style="position:relative;">देशे परमधर्मिष्ठे नदीमोघवतीमनु॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2034"><div class="verse-block verse-justify"><div class="line">दूरादेव तमालोक्य कृष्णो राजा च धर्मजः।</div><div class="line" style="position:relative;">चत्वारः पाण्डवाश्चैव ते च शारद्वतादयः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2035"><div class="verse-block verse-justify"><div class="line">अवम्कन्द्याथ वाहेभ्यः संयम्य प्रचलं मनः।</div><div class="line" style="position:relative;">एकीकृत्येन्द्रियग्राममुपतम्थुर्महामुनीन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2036"><div class="verse-block verse-justify"><div class="line">अभिवाद्य तु गोविन्दः सात्यकिस्ते च पार्थिवाः।</div><div class="line" style="position:relative;">व्यासादीस्तानृपीन्पश्चाद्गाङ्गेयमुपतस्थिरे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2037"><div class="verse-block verse-justify"><div class="line">तपोवृद्धि ततः पृष्ट्वा गाङ्गेयं यदुपुङ्गवः।</div><div class="line" style="position:relative;">परिवार्य ततः सर्वे निपेदुः पुरुषर्षभाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2038"><div class="verse-block verse-justify"><div class="line">ततो निशाम्य गाङ्गेयं शाम्यमानमिवानलम्।</div><div class="line" style="position:relative;">किंचिद्दीनमना भीष्ममितिहोवाच केशवः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2039"><div class="verse-block verse-justify"><div class="line">कच्चिज्ज्ञानानि सर्वाणि प्रसन्नानि यथापुरम्।</div><div class="line" style="position:relative;">कच्चिन्न व्याकुला चैव बुद्धिस्ते वदतां वर॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2040"><div class="verse-block verse-justify"><div class="line">शराभिघातदुःखार्तं कच्चिद्गात्रं न दूयते।</div><div class="line" style="position:relative;">मानसादपि दुःखाद्धि शारीरं बलवत्तरम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2041"><div class="verse-block verse-justify"><div class="line">वरदानात्पितुः कामं छन्दमृत्युरसि प्रभो।</div><div class="line" style="position:relative;">शन्तनोर्धर्मनित्यस्य न त्वेतदिह कारणम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2042"><div class="verse-block verse-justify"><div class="line">सुमूक्ष्मोऽपि तु देहे वै शल्यो जनयते रुजम्।</div><div class="line" style="position:relative;">किंपुनः शरसंघातैश्चितस्य तव पार्थिव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2043"><div class="verse-block verse-justify"><div class="line">कामं नैतत्तवाख्येयं प्राणिनां प्रभवाप्ययौ।</div><div class="line" style="position:relative;">भवानुपदिशेच्छ्रेयो देवानामपि भारत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2044"><div class="verse-block verse-justify"><div class="line">यच्च भूतं भविष्यं च भवच्च पुरुषर्षभ।</div><div class="line" style="position:relative;">सर्वं तज्ज्ञानवृद्धस्य तव पाणाविवाहितम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2045"><div class="verse-block verse-justify"><div class="line">संसारस्येह भूतानां धर्मस्य च फलोदयः।</div><div class="line" style="position:relative;">विदितस्ते महाप्राज्ञ त्वं हि धर्ममयो निधिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2046"><div class="verse-block verse-justify"><div class="line">त्व, हि राज्ये स्थितं स्फीते समग्राङ्गमरोगिणम्।</div><div class="line" style="position:relative;">स्त्रीसहस्रैः परिवृतं पश्यामीवोर्ध्वरेतसम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2047"><div class="verse-block verse-justify"><div class="line">ऋते शान्तनवाद्भीष्मात्रिषु लोकेषु पार्थिवम्।</div><div class="line" style="position:relative;">सत्यधर्मान्महावीर्याच्छूराद्धर्मैकतत्परात्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2048"><div class="verse-block verse-justify"><div class="line">मृत्युमावार्य तपसा शरसंस्तरशायिनः।</div><div class="line" style="position:relative;">त्रिवर्गप्रभवं कंचिन्न च तातानुशुश्रुम॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2049"><div class="verse-block verse-justify"><div class="line">सत्ये तपसि दाने च यज्ञाधिकरणे तथा।</div><div class="line" style="position:relative;">धनुर्वेदे च वेदे च नित्यं चैवान्ववेक्षणे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2050"><div class="verse-block verse-justify"><div class="line">अनृशंसं शुचिं दान्तं सर्वभूतहिते रतम्।</div><div class="line" style="position:relative;">महारथं त्वत्सदृशं न कंचिदनुशुश्रुम॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2051"><div class="verse-block verse-justify"><div class="line">त्वं हि देवान्सगन्धर्वानसुरान्यक्षराक्षसान्।</div><div class="line" style="position:relative;">शक्तस्त्वेकरथेनैव विजेतुं नात्र संशयः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2052"><div class="verse-block verse-justify"><div class="line">स त्वं भीष्म महाबाहो वसूनां वासवोपमः।</div><div class="line" style="position:relative;">नित्यं विप्रैः समाख्यातो नवमोऽनवमो गुणैः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2053"><div class="verse-block verse-justify"><div class="line">अहं च त्वाऽभिजानामि स्वयं पुरुषसत्तम।</div><div class="line" style="position:relative;">त्रिदशेष्वपि विख्यातस्त्वं शक्त्या पुरुषोत्तमः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2054"><div class="verse-block verse-justify"><div class="line">मनुष्येषु मनुष्येन्द्र न दृष्टो न च मे श्रुतः।</div><div class="line" style="position:relative;">भवतो हि गुणैस्तुल्यः पृथिव्यां पुरुषः क्वचित्॥२९॥</div></div></div>
<p class="prose">त्वं हि सर्वगुणै राजन्देवानप्यतिरिच्यसे</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b2056"><div class="verse-block verse-justify"><div class="line">तपसा हि भवाञ्शक्तः स्रष्टुं लोकांश्चराचरान्।</div><div class="line" style="position:relative;">किंपुनश्चात्मनो लोकानुत्तमानुत्तमैर्गुणैः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2057"><div class="verse-block verse-justify"><div class="line">तदस्य तप्यमानस्य ज्ञातीनां संक्षयेन वै।</div><div class="line" style="position:relative;">ज्येष्ठस्य पाण्डुपुत्रस्य शोकं भीष्म व्यपानुद॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2058"><div class="verse-block verse-justify"><div class="line">ये हि धर्माः समाख्याताश्चातुर्वर्ण्यस्य भारत।</div><div class="line" style="position:relative;">चातुराश्रम्यसंयुक्ताः सर्वे ते विदितास्तव॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2059"><div class="verse-block verse-justify"><div class="line">चातुर्विद्ये च ये प्रोक्ताश्चातुर्होत्रे च भारत।</div><div class="line" style="position:relative;">योगे साङ्ख्ये च नियता ये च धर्माः सनातनाः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2060"><div class="verse-block verse-justify"><div class="line">चातुर्वर्ण्यस्य यश्चोक्तो धर्मो न स्म विरुध्यते।</div><div class="line" style="position:relative;">सेव्यमानः सवैयाख्यो गाङ्गेय विदितस्तव॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2061"><div class="verse-block verse-justify"><div class="line">प्रतिलोमप्रसूतानां म्लेच्छानां चैव यः स्मृतः।</div><div class="line">देशजातिकुलानां च जानीषे धर्मलक्षणम्।</div><div class="line" style="position:relative;">॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2062"><div class="verse-block verse-justify"><div class="line">वेदोक्तो यश्च शिष्टोक्तः सदैव विदितस्तव।</div><div class="line" style="position:relative;">`प्रवृत्तश्च निवृत्तश्च स चापि विदितस्तव ॥'॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2063"><div class="verse-block verse-justify"><div class="line">इतिहासपुराणार्थाः कार्त्स्न्येन विदितास्तव।</div><div class="line" style="position:relative;">धर्मशास्त्रं च सकलं नित्यं मनसि ते स्थितम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2064"><div class="verse-block verse-justify"><div class="line">ये च केचन लोकेऽस्मिन्नर्थाः संशयकारकाः।</div><div class="line" style="position:relative;">तेषां छेत्ता नास्ति लोके त्वदन्यः पुरुषर्षभः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2065"><div class="verse-block verse-justify"><div class="line">स पाण्डवेयस्य मनःसमुत्थितंनरेन्द्र शोकं व्यपकर्ष मेधया।</div><div class="line" style="position:relative;">भवद्विधा ह्युत्तमबुद्धिर्विस्तराविमुह्यमानस्य जनस्य शान्तये॥३९॥</div></div></div></div>
