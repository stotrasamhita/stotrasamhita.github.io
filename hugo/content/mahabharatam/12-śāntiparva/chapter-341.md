<div class="verse-block-wrapper" id="12-śāntiparva-b14447"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14448"><div class="verse-block verse-justify"><div class="line">इत्येवमुक्त्वा वचनं ब्रह्मर्षिः सुमहातपाः।</div><div class="line" style="position:relative;">प्रातिष्ठत शुकः सिद्धिं हित्वा दोषांश्चतुर्विधान्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14449"><div class="verse-block verse-justify"><div class="line">तमो ह्यष्टगुणं हित्वा जहौ पञ्चविधं रजः।</div><div class="line" style="position:relative;">ततः सत्वं जहौ धीमांस्तदद्भुतमिवाभवत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14450"><div class="verse-block verse-justify"><div class="line">ततस्तस्मिन्पदे नित्ये निर्गुणे लिङ्गवर्जिते।</div><div class="line" style="position:relative;">ब्रह्मणि प्रत्यतिष्ठत्स विधूमोऽग्निरिव ज्वलन्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14451"><div class="verse-block verse-justify"><div class="line">उत्कापाता दिशां दाहो भूमिकम्पस्तथैव च।</div><div class="line" style="position:relative;">प्रादुर्भूताः क्षणे तस्मिंस्तदद्भुतमिवाभवत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14452"><div class="verse-block verse-justify"><div class="line">द्रुमाः शाखाश्च मुमुचुः शिखराणि च पर्वताः।</div><div class="line" style="position:relative;">निर्घातशब्दैर्गुरुभिर्भूमिर्व्यादीर्यतेव ह॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14453"><div class="verse-block verse-justify"><div class="line">न बभासे सहस्रांशुर्न जज्वाल च पावकः।</div><div class="line" style="position:relative;">ह्रदाश्च सरितश्चैव चुक्षुभुः सागरास्तथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14454"><div class="verse-block verse-justify"><div class="line">ववर्ष वासवस्तोयं रसवच्च सुगन्धि च।</div><div class="line" style="position:relative;">ववौ समीरणश्चापि दिव्यगन्धवहः शुचिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14455"><div class="verse-block verse-justify"><div class="line">स शृङ्गेऽप्रतिमे दिव्ये हिमवन्मेरुसंनिभे।</div><div class="line" style="position:relative;">संश्लिष्टे स्वतेपीते द्वे रुक्मरूप्यमये शुभे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14456"><div class="verse-block verse-justify"><div class="line">शतयोजनविस्तारे तिर्यगूर्ध्वं च भारत।</div><div class="line" style="position:relative;">उदीचीं दिशमास्थाय रुचिरे संददर्श ह॥१०॥</div></div></div>
<p class="prose">सोऽविशङ्केन मनसा तथैवाभ्यपतच्छुकः</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b14458"><div class="verse-block verse-justify"><div class="line">ततः पर्वतशृङ्गे द्वे सहसैव द्विधाकृते।</div><div class="line" style="position:relative;">अदृश्येतां महाराज तदद्भुतमिवाभवत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14459"><div class="verse-block verse-justify"><div class="line">ततः पर्वतशृङ्गाभ्यां सहसैव विनिःसृतः।</div><div class="line" style="position:relative;">न च प्रतिजघानास्य स गतिं पर्वतोत्तमः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14460"><div class="verse-block verse-justify"><div class="line">ततो महानभूच्छब्दो दिवि सर्वदिवौकसाम्।</div><div class="line" style="position:relative;">गन्धर्वाणामृषीणां च ये च शैलनिवासिनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14461"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा शुकमतिक्रान्तं पर्वतं च द्विधा कृतम्।</div><div class="line" style="position:relative;">साधुसाध्विति तत्रासीन्नादः सर्वत्र भारत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14462"><div class="verse-block verse-justify"><div class="line">स पूज्यमानो देवैश्च गन्धर्वैर्ऋशिभिस्तथा।</div><div class="line" style="position:relative;">यक्षराक्षससङ्घैश्च विद्याधरगणैस्तथा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14463"><div class="verse-block verse-justify"><div class="line">दिव्यैः पुष्पैः समाकीर्णमन्तरिक्षं समन्ततः।</div><div class="line" style="position:relative;">आसीत्किल महाराज शुकाभिपतने तदा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14464"><div class="verse-block verse-justify"><div class="line">ततो मन्दाकिनीं रम्यामुपरिष्टादभिव्रजन्।</div><div class="line" style="position:relative;">शुको ददर्श धर्मात्मा पुष्पितद्रुमकाननाम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14465"><div class="verse-block verse-justify"><div class="line">तस्यां क्रीडन्त्यभिरताः स्नान्ति चैवाप्सरोगणाः।</div><div class="line" style="position:relative;">शून्याकारं निराकाराः शुकं दृष्ट्वा विवाससः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14466"><div class="verse-block verse-justify"><div class="line">तं प्रक्रामन्तमाज्ञाय पिता स्नेहसमन्वितः।</div><div class="line" style="position:relative;">उत्तमां गतिमास्थाय पृष्ठतोऽनुससार ह॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14467"><div class="verse-block verse-justify"><div class="line">शुकस्तु मारुतादूर्ध्वं गतिं कृत्वान्तरिक्षगाम्।</div><div class="line" style="position:relative;">दर्शयित्वा प्रभावं स्वं सर्वभूतोऽभवत्तदा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14468"><div class="verse-block verse-justify"><div class="line">महायोगगतिं त्वग्र्यां व्यासोत्थाय महातपाः।</div><div class="line" style="position:relative;">निमेषान्तरमात्रेण शुकाभिपतनं ययौ॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14469"><div class="verse-block verse-justify"><div class="line">स ददर्श द्विधा कृत्वा पर्वताग्रं शुकं गतम्।</div><div class="line" style="position:relative;">शशंसुर्ऋषयस्तत्र कर्म पुत्रस्य तत्तदा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14470"><div class="verse-block verse-justify"><div class="line">ततः शुकेति दीर्घेण शब्देनाक्रन्दितस्तदा।</div><div class="line" style="position:relative;">स्वयं पित्रा स्वरेणोच्चैस्त्रील्लोँकाननुनाद्य वै॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14471"><div class="verse-block verse-justify"><div class="line">शुकः सर्वगतो भूत्वा सर्वात्मा सर्वतोमुखः।</div><div class="line" style="position:relative;">प्रत्यभाषत धर्मात्मा भोःशब्देनानुनादयन्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14472"><div class="verse-block verse-justify"><div class="line">तत एकाक्षरं नादं भोरित्येव समीरयन्।</div><div class="line" style="position:relative;">प्रत्याहरञ्जगत्सर्वमुच्चैः स्थावरजङ्गमम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14473"><div class="verse-block verse-justify"><div class="line">ततःप्रभृति चाद्यापि शब्दानुच्चारितान्पृथक्।</div><div class="line" style="position:relative;">गिरिगह्वरपृष्ठेषु व्याहरन्ति शुकं प्रति॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14474"><div class="verse-block verse-justify"><div class="line">अन्तर्हितः प्रभावं तु दर्शयित्वा शुकस्तदा।</div><div class="line" style="position:relative;">गुणान्संत्यज्य शब्दादीन्पदमभ्यगमत्परम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14475"><div class="verse-block verse-justify"><div class="line">महिमानं तु तं दृष्ट्वा पुत्रस्यामिततेजसः।</div><div class="line" style="position:relative;">निषसाद गिरिप्रस्थे पुत्रमेवानुचिन्तयन्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14476"><div class="verse-block verse-justify"><div class="line">ततो मन्दाकिनीतीरे क्रीडन्तोऽऽप्सरसां गणाः।</div><div class="line" style="position:relative;">आसाद्य तमृषिं सर्वाः संभ्रान्ता गतचेतसः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14477"><div class="verse-block verse-justify"><div class="line">जले निलिल्यिरे काश्चित्काश्चिद्गुल्मान्प्रपेदिरे।</div><div class="line" style="position:relative;">वसनान्याददुः काश्चित्तं दृष्ट्वा मुनिसत्तमम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14478"><div class="verse-block verse-justify"><div class="line">तां मुक्ततां तु विज्ञाय मुनिः पुत्रस्य वै तदा।</div><div class="line" style="position:relative;">सक्ततामात्मनश्चैव प्रीतोऽभूद्बीडितश्च ह॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14479"><div class="verse-block verse-justify"><div class="line">तं देवगन्धर्ववृतो महर्षिगणपूजितः।</div><div class="line" style="position:relative;">पिनाकहस्तो भगवानभ्यागच्छत शंकरः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14480"><div class="verse-block verse-justify"><div class="line">तमुवाच महादेवः सान्त्वपूर्वमिदं वचः।</div><div class="line" style="position:relative;">पुत्रशोकाभिसंतप्तं कृष्णद्वैपायनं तदा॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14481"><div class="verse-block verse-justify"><div class="line">अग्नेर्भूमेरपां वायोरन्तरिक्षस्य चैव ह।</div><div class="line" style="position:relative;">वीर्येण सदृशः पुत्रः पुरा मत्तस्त्वया वृतः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14482"><div class="verse-block verse-justify"><div class="line">स तथालक्षणो जातस्तपसा तव संभृतः।</div><div class="line" style="position:relative;">मम चैव प्रसादेन ब्रह्मतेजोमयः शुचिः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14483"><div class="verse-block verse-justify"><div class="line">स गतिं परमां प्राप्तो दुष्प्रापामजितेन्द्रियैः।</div><div class="line" style="position:relative;">दैवतैरपि विप्रर्षे तं त्वं किमनुशोचसि॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14484"><div class="verse-block verse-justify"><div class="line">यावत्स्थास्यन्ति गिरयो यावत्स्थास्यन्ति सागराः।</div><div class="line" style="position:relative;">तावत्तवाक्षया कीर्तिः सपुत्रस्य भविष्यति॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14485"><div class="verse-block verse-justify"><div class="line">छायां स्वपुत्रसदृशीं सर्वतोऽनपगां सदा।</div><div class="line" style="position:relative;">द्रक्ष्यसे त्वं च लोकेऽस्मिन्मत्प्रसादान्महामुने॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14486"><div class="verse-block verse-justify"><div class="line">सोऽनुगीतो भगवता स्वयं रुद्रेण भारत।</div><div class="line" style="position:relative;">छायां पश्यन्परावृत्तः स मुनिः परया मुदा॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14487"><div class="verse-block verse-justify"><div class="line">इति जन्म गतिश्चैव शुकस्य भरतर्षभ।</div><div class="line" style="position:relative;">विस्तरेण समाख्याता यन्मां त्वं परिपृच्छसि॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b14488"><div class="verse-block verse-justify"><div class="line">एतदाचष्ट मे राजन्देवर्षिर्नारदः पुरा।</div><div class="line" style="position:relative;">व्यासश्चैव महायोगी संजल्पेषु पदेपदे॥४१॥</div></div></div>
