<div class="verse-block-wrapper" id="12-śāntiparva-b5370"><div class="verse-block verse-justify"><div class="line">हीने परमके धर्मे सर्वलोकातिलङ्घने।</div><div class="line" style="position:relative;">सर्वस्मिन्दस्युसाद्भूते पृथिव्यामुपजीवने॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5371"><div class="verse-block verse-justify"><div class="line">केन स्विद्ब्राह्मणो जीवेज्जघन्ये काल आगते।</div><div class="line">असंत्यजन्पुत्रपौत्राननुक्रोशात्पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5372"><div class="verse-block verse-justify"><div class="line">विज्ञानबलमास्थाय जीवितव्यं तथा गते।</div><div class="line" style="position:relative;">सर्वं साध्वर्थमेवेदमसाध्वर्थं न किंचन॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5373"><div class="verse-block verse-justify"><div class="line">असाधुभ्योऽर्थमादाय साधुभ्यो यः प्रयच्छति।</div><div class="line" style="position:relative;">आत्मानं संक्रमं कृत्वा कृच्छ्रधर्मकृदेव सः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5374"><div class="verse-block verse-justify"><div class="line">अरोषेणात्मनो राजन्राज्ये स्थितिमकोपयन्।</div><div class="line" style="position:relative;">अदत्तमप्याददीत् भ्रातुर्वित्तं ममेति वा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5375"><div class="verse-block verse-justify"><div class="line">विज्ञानबलपूतो यो वर्तते निन्दितेष्वपि।</div><div class="line" style="position:relative;">वृत्तिविज्ञानवान्धीरः कस्तं वक्तुमिहार्हति॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5376"><div class="verse-block verse-justify"><div class="line">येषां बहुकृता बुद्धिस्तेषामन्या न रोचते।</div><div class="line" style="position:relative;">यजसा ते प्रवर्तन्ते बलवन्तो युधिष्ठिर॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5377"><div class="verse-block verse-justify"><div class="line">यदैव प्रकृतं शास्त्रं जनस्तदनुवर्तते।</div><div class="line" style="position:relative;">यदैवमध्यासेवन्ते मेध्रावी वाऽप्यथोत्तरम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5378"><div class="verse-block verse-justify"><div class="line">ऋत्विक्पुरोहिताचार्यान्सत्कृतानभिसत्कृतान्।</div><div class="line" style="position:relative;">न ब्राह्मणान्घातयीत दोषान्प्राप्नोति घातयन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5379"><div class="verse-block verse-justify"><div class="line">एतत्प्रमाणं लोकस्य चक्षुरेतत्सातनम्।</div><div class="line" style="position:relative;">तत्प्रमाणोऽवगाहेत तेन तत्साध्वसाधु वा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5380"><div class="verse-block verse-justify"><div class="line">हवो ग्रामवास्तव्या दोषान्ब्रूयुः परस्परम्।</div><div class="line" style="position:relative;">न तेषां वचनाद्राजा सत्कुर्याद्धातयीत वा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5381"><div class="verse-block verse-justify"><div class="line">न वाच्यः परिवादो वै न श्रोतव्यः कथंचन।</div><div class="line" style="position:relative;">कर्णौ तत्र पिधातव्यौ गन्तव्यं वा ततोऽन्यतः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5382"><div class="verse-block verse-justify"><div class="line">न सतां शीलमेतद्वै परिवादो न पैशुनम्।</div><div class="line" style="position:relative;">गुणानामेव वक्तारः सन्तो नित्यं युधिष्ठिर॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5383"><div class="verse-block verse-justify"><div class="line">यथा समधुरौ दम्यौ सुदान्तौ साधुवाहिनौ।</div><div class="line" style="position:relative;">धुरमुद्यम्य वहतस्तथा वर्तेत वै नृपः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5384"><div class="verse-block verse-justify"><div class="line">यथायथाऽस्य बहवः सहायाः स्युस्तथा चरेत्।</div><div class="line" style="position:relative;">आचारमेव मन्यन्ते गरीयो धर्मलक्षणम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5385"><div class="verse-block verse-justify"><div class="line">अपरे नैवमिच्छन्ति ये शङ्खलिखिंतप्रियाः।</div><div class="line" style="position:relative;">अर्थे क्षीणेऽथवा लुब्धास्ते ब्रूयुर्वाक्यमीदृशम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5386"><div class="verse-block verse-justify"><div class="line">आर्षमष्यत्र पश्यन्ति विकर्मस्थस्य पातनम्।</div><div class="line" style="position:relative;">न चार्षात्सदृशं किंचित्प्रमाणं दृश्यते क्वचित्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5387"><div class="verse-block verse-justify"><div class="line">देवा ह्यपि विकर्मस्थं घातयन्ति नराधमम्।</div><div class="line" style="position:relative;">व्याजेन विन्दन्वित्तं हि धर्मतः परिहीयते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5388"><div class="verse-block verse-justify"><div class="line">सर्वतः सत्कृतः सद्भिर्भूतिप्रवरकारणैः।</div><div class="line" style="position:relative;">हृदयेनाभ्यनुज्ञातो यो धर्मस्तं व्यवस्यति॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5389"><div class="verse-block verse-justify"><div class="line">यश्चतुर्गुणसंपन्नं धर्मं वेद स धर्मवित्।</div><div class="line" style="position:relative;">अहेरिव हि धर्मस्य पदं दुःखं गवेषितुम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5390"><div class="verse-block verse-justify"><div class="line">यथा मृगस्य विद्धस्य मृगव्याधः पदं नयेत्।</div><div class="line" style="position:relative;">लक्षेद्रुधिरपातेन तथा धर्मपदं नयेत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b5391"><div class="verse-block verse-justify"><div class="line">यथा सम्यग्वितेन पथा गन्तव्यमप्युत।</div><div class="line" style="position:relative;">राजर्षीणां वृत्तमेतदेवं गच्छ युधिष्ठिर॥२३॥</div></div></div></div>
