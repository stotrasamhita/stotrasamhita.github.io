<div class="verse-block-wrapper" id="12-śāntiparva-b6494"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6495"><div class="verse-block verse-justify"><div class="line">पापस्य यदधिष्ठानं यतः पापं प्रवर्तते।</div><div class="line">एतदिच्छाम्यहं श्रोतुं तत्त्वेन भरतर्षभ ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6496"><div class="verse-block verse-justify"><div class="line">पापस्य यदधिष्ठानं तच्छृणुष्व नराधिप।</div><div class="line" style="position:relative;">एको लोभो महाग्राहो लोभात्पापं प्रवर्तते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6497"><div class="verse-block verse-justify"><div class="line">अतः पापमधर्मश्च तथा दुःखमनुत्तमम्।</div><div class="line" style="position:relative;">निकृत्या मूलमेतद्धि येन पापकृतो जनाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6498"><div class="verse-block verse-justify"><div class="line">लोभात्क्रोधः प्रभवति लोभात्कामः प्रवर्तते।</div><div class="line" style="position:relative;">लोभान्मोहश्च माया च मानस्तम्भः परासुता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6499"><div class="verse-block verse-justify"><div class="line">अक्षमा ह्रीपरित्यागः श्रीनाशो धर्मसंक्षयः।</div><div class="line" style="position:relative;">अभिध्याऽप्रख्यता चैव सर्वं लोभात्प्रवर्तते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6500"><div class="verse-block verse-justify"><div class="line">अत्यागश्च कुतर्कश्च विकर्मसु च याः क्रियः।</div><div class="line" style="position:relative;">कुलविद्यामदश्चैव रूपैश्वर्यमदस्तथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6501"><div class="verse-block verse-justify"><div class="line">सर्वभूतेष्वभिद्रोहः सर्वभूतेष्वसत्कृतिः।</div><div class="line" style="position:relative;">सर्वभूतेष्वविश्वासः सर्वभूतेष्वनार्जवम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6502"><div class="verse-block verse-justify"><div class="line">हरणं परवित्तानां परदाराभिमर्शनम्।</div><div class="line" style="position:relative;">वाग्वेगो मनसो वेगो निन्दावेगस्तथैव च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6503"><div class="verse-block verse-justify"><div class="line">उपस्थोदरयोर्वेगो मृत्युवेगश्च दारुणः।</div><div class="line" style="position:relative;">ईर्ष्यावेगश्च बलवान्मिथ्यावेगश्च दुर्जयः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6504"><div class="verse-block verse-justify"><div class="line">रसवेगश्च दुर्वार्यः श्रोत्रवेगश्च दुःसह।</div><div class="line">कुत्सा विकत्था मात्सर्यं पापं दुष्कर्मकारिता।</div><div class="line" style="position:relative;">साहसानां च सर्वेषामकार्याणां क्रियास्तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6505"><div class="verse-block verse-justify"><div class="line">आतौ बाल्ये च कौमारे यौवने चापि मानवाः।</div><div class="line" style="position:relative;">न त्यजन्त्यात्मकर्मैकं यन्न जीर्यति जीर्यतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6506"><div class="verse-block verse-justify"><div class="line">यो न पूरयितुं शक्यो लोभः प्रीत्या कथंचन।</div><div class="line" style="position:relative;">नित्यं गम्भीरतोयाभिरापगाभिरिवोदधिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6507"><div class="verse-block verse-justify"><div class="line">न प्रहृष्यति यो लोभैः कामैर्यश्च न तृप्यति।</div><div class="line" style="position:relative;">यो न देवैर्न गन्धर्वैर्नासुरैर्न महोरगैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6508"><div class="verse-block verse-justify"><div class="line">ज्ञायते नृप तत्त्वेन सर्वैर्भूतगणैस्तथा।</div><div class="line" style="position:relative;">स लोभः सह मोहेन विजेतव्यो जितात्मना॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6509"><div class="verse-block verse-justify"><div class="line">दम्भो द्रोहश्च निन्दा च पैशून्यं मत्सरस्तथा।</div><div class="line" style="position:relative;">भवन्त्येतानि कौरव्य लुब्धानामकृतात्मनाम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6510"><div class="verse-block verse-justify"><div class="line">सुमहान्त्यपि शास्त्राणि धारयन्तो बहुश्रुताः।</div><div class="line" style="position:relative;">छेत्तारः संशयानां च क्लिश्यन्तीहाल्पबुद्धयः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6511"><div class="verse-block verse-justify"><div class="line">द्वेपक्रोधप्रसक्ताश्च शिष्टाचारबहिष्कृताः।</div><div class="line" style="position:relative;">अन्तःक्षुरा वाङ्भधुराः कृपाश्छन्नास्तृणैरिव॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6512"><div class="verse-block verse-justify"><div class="line">धर्मवैतंसिकाः क्षुद्रा मुष्णन्ति ध्वजिनो जगत्।</div><div class="line">कुर्वते च बहून्मार्गांस्तान्हेतुबलमाश्रिताः।</div><div class="line" style="position:relative;">सर्वमार्गान्विलुम्पन्ति लोभज्ञानेष्ववस्थिताः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6513"><div class="verse-block verse-justify"><div class="line">धर्मस्य ह्रियमाणस्य लोभग्रस्तैर्दुरात्मभिः।</div><div class="line" style="position:relative;">याया विक्रियते संस्था ततः साऽपि प्रपद्यते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6514"><div class="verse-block verse-justify"><div class="line">दर्पः बोधो मदः स्वप्नो हर्षः शोकोऽभिमानिता।</div><div class="line" style="position:relative;">एत हि कौरव्य दृश्यन्ते लुब्धबुद्धिषु॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6515"><div class="verse-block verse-justify"><div class="line">एताना टान्बुद्धस्व नित्यं लोभसमन्वितान्।</div><div class="line" style="position:relative;">शिष्टांस्तु परिपृच्छेथा यान्वक्ष्यामि शुचिव्रतान्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6516"><div class="verse-block verse-justify"><div class="line">येष्वावृत्तिभयं नास्ति परलोकभयं न च।</div><div class="line" style="position:relative;">नामिपेषु प्रसङ्गोऽस्ति न प्रियेष्वप्रियेषु च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6517"><div class="verse-block verse-justify"><div class="line">शिष्टाचारः प्रियो येषु दमो येषु प्रतिष्ठितः।</div><div class="line" style="position:relative;">सुखं दुःखं समं येषां सत्यं येषां परायणम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6518"><div class="verse-block verse-justify"><div class="line">दातारो न ग्रहीतारो दयावन्तस्नथैव च।</div><div class="line" style="position:relative;">पितृदेवातिथेयाश्च नित्योद्युक्तास्तथैव च॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6519"><div class="verse-block verse-justify"><div class="line">सर्वोपकारिणो वीराः सर्वधर्मानुपालकाः।</div><div class="line" style="position:relative;">सर्वभूतहिताश्चैव सर्वदेयाश्च भारत॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6520"><div class="verse-block verse-justify"><div class="line">न ते चालयितुं शक्या धर्मव्याहारकारिणः।</div><div class="line" style="position:relative;">न तेषां भिद्यते वृत्तं यत्पुरा साधुभिः कृतम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6521"><div class="verse-block verse-justify"><div class="line">न त्रासिनो न चपला न रौद्राः सत्पथे स्थिताः।</div><div class="line" style="position:relative;">ते सेव्याः साधुभिर्नित्यमसाधूंश्च विवर्जयेत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6522"><div class="verse-block verse-justify"><div class="line">कामक्रोधव्यपेता ये निर्ममा निरहंकृताः।</div><div class="line" style="position:relative;">सुव्रताः स्थिरमर्यादास्तानुपास्व च पृच्छ च॥२९॥</div></div></div>
<p class="prose">न वागर्थं यशोर्थं वा धर्मस्तेपां युधिष्ठिर ॥अवश्यं कार्य इत्येव शरीरस्य क्रियास्तथा</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b6524"><div class="verse-block verse-justify"><div class="line">त भयं क्रोधचापल्ये न शोकस्तेषु विद्यते।</div><div class="line" style="position:relative;">न धर्मध्वजिनश्चैव न गुह्यं किंचिदास्थिताः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6525"><div class="verse-block verse-justify"><div class="line">येष्वलोभस्तथाऽमोहो ये च सत्यार्जवे स्थिताः।</div><div class="line" style="position:relative;">तेषु कौन्तेय रज्येथा येषां न भ्रश्यते पुनः॥३१॥</div></div></div>
