<div class="verse-block-wrapper" id="12-śāntiparva-b4774"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4775"><div class="verse-block verse-justify"><div class="line">व्याघ्रश्वोटजमूलस्थस्तृप्तः सुप्तो हतैर्मृगैः।</div><div class="line" style="position:relative;">नागभागात्तमुद्देशं मत्तो मेघ इवोत्थितः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4776"><div class="verse-block verse-justify"><div class="line">प्र--करटः प्रांशुः पझी विततकुम्भकः।</div><div class="line" style="position:relative;">सु--णो महाकायो मेघगम्भीरनिः स्वनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4777"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा कुञ्जरं मत्तमायान्तं बलगर्वितम्।</div><div class="line" style="position:relative;">व्या हस्तिभयान्त्रस्तस्तमृषिं शरणं ययौ॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4778"><div class="verse-block verse-justify"><div class="line">ततोऽनयत्कुञ्जरत्वं व्याघ्रं तमृषिसत्तमः।</div><div class="line" style="position:relative;">महमेघोपमं दृष्ट्वा स भीतो ह्यभवद्गजः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4779"><div class="verse-block verse-justify"><div class="line">ततः कमलषण़्डानि सल्लकीगहनानि च।</div><div class="line" style="position:relative;">व्य--रत्स मुदायुक्तः पझरेणुविभूषितः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4780"><div class="verse-block verse-justify"><div class="line">कदचिद्दममाणस्य हस्तिनः संमुखं तदा।</div><div class="line" style="position:relative;">ऋषेस्तस्योटस्थस्य कालोऽगच्छद्दिवानिशम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4781"><div class="verse-block verse-justify"><div class="line">अथाजगाम तं देशं केसरी केसरारुणः।</div><div class="line" style="position:relative;">गिन्विन्दरजो भीमः सिंहो नागकुलान्तकः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4782"><div class="verse-block verse-justify"><div class="line">तं--- सिंहमायान्तं नागः सिंहभयार्दितः।</div><div class="line" style="position:relative;">ऋषिं शरणमापेदे वेपमानो भयातुरः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4783"><div class="verse-block verse-justify"><div class="line">स त-ः सिंहतां नीतो गजेन्द्रो मुनिना तदा।</div><div class="line" style="position:relative;">तं च नागणयत्सिंहं तुल्यजातिसमन्वयात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4784"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा च सोऽभवत्सिंहो वन्यो हिंसन्नवाग्बलः।</div><div class="line" style="position:relative;">स चाश्रपेऽवसत्सिंहस्तस्मिन्नेव सुखी वने॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4785"><div class="verse-block verse-justify"><div class="line">न चान्ये क्षुद्रपशवस्तपोवनसमीपतः।</div><div class="line" style="position:relative;">व्यदृश्यन्त तदा त्रस्ता जीविताकाङ्क्षिणस्तथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4786"><div class="verse-block verse-justify"><div class="line">कदाचित्कालयोगेन सर्वप्राणिविहिंसकः।</div><div class="line" style="position:relative;">बलवान्क्षतजाहारो नानासत्वभयंकरः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4787"><div class="verse-block verse-justify"><div class="line">अष्टपादर्ध्वनयनः शरभो वनगोचरः।</div><div class="line" style="position:relative;">तं सिंहं हन्तुमागच्छन्मुनेस्तस्य निवेशने॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4788"><div class="verse-block verse-justify"><div class="line">`तं दृष्ट्वा शरभं यान्तं सिंहः परभयान्वितः।</div><div class="line" style="position:relative;">ऋषिं शरणमापेदे वेपमानः कृताञ्जलिः ॥'॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4789"><div class="verse-block verse-justify"><div class="line">सं मुनिः शरभं चक्रे बलोत्कटमरिंदम।</div><div class="line">ततः स शरभो वन्यो मुनेः शरभमग्रतः।</div><div class="line" style="position:relative;">दृष्ट्वा बलिनमत्युग्रं द्रुतं संप्राद्रवद्वनम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4790"><div class="verse-block verse-justify"><div class="line">स एवं शरभस्थाने न्यस्तो वै मुनिना तदा।</div><div class="line" style="position:relative;">मुनेः पार्श्वगतो नित्यं शरभः सुखमाप्तवान्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4791"><div class="verse-block verse-justify"><div class="line">ततः शरभसंत्रस्ताः सर्वे मृगगणा वनात्।</div><div class="line" style="position:relative;">दिशः संप्राद्रवत्राजन्भयाज्जीवितकाङ्क्षिणः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4792"><div class="verse-block verse-justify"><div class="line">शरभोऽप्यतिसंहृष्टो नित्यं प्राणिवधे रतः।</div><div class="line" style="position:relative;">फलमूलाशनं कर्तुं नैच्छत्स पिशिताशनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4793"><div class="verse-block verse-justify"><div class="line">ततः क्षुद्रसमाचारो बलेन च समन्वितः।</div><div class="line" style="position:relative;">इयेष तं मुनिं हन्तुमकृतज्ञः कृतान्वयः॥२०॥</div></div></div>
<p class="prose">`चिन्तयामास च तदा शरभः श्वानपूर्वकः</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b4795"><div class="verse-block verse-justify"><div class="line">अस्य प्रभावात्संप्राप्तो वाङ्भात्रेणैव केवलम्।</div><div class="line" style="position:relative;">शरभत्वं सुदुष्प्रापं सर्वभूतभयंकरम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4796"><div class="verse-block verse-justify"><div class="line">अन्येऽप्यत्र भयत्रस्ताः सन्ति सत्वा भयार्दिताः।</div><div class="line" style="position:relative;">मुनिमाश्रित्य जीवन्तो मृगाः पक्षिगणास्तथा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4797"><div class="verse-block verse-justify"><div class="line">तेषामपि कदाचिच्च शरभत्वं प्रयच्छति।</div><div class="line" style="position:relative;">सर्वसत्वोत्तमं लोके बलं यत्र प्रतिष्ठितम्॥२३॥</div></div></div>
<p class="prose">पक्षिणामप्ययं दद्यात्कदाचिद्गारुडं बलम्</p>
<div class="verse-block-wrapper" id="12-śāntiparva-b4799"><div class="verse-block verse-justify"><div class="line">यावदन्यस्य संप्रीतः कारुण्यं तु समाश्रितः।</div><div class="line" style="position:relative;">न ददाति बलं तुष्टः सत्वस्यान्यस्य कस्यचित्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4800"><div class="verse-block verse-justify"><div class="line">तावदेनमहं विप्रं वधिष्यामि च शीघ्रतः।</div><div class="line" style="position:relative;">स्थातुं मया शक्यमिह मुनिघातान्न संशयः ॥'॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4801"><div class="verse-block verse-justify"><div class="line">ततस्तेन तपःशक्त्या विदितो ज्ञानचक्षुषा।</div><div class="line" style="position:relative;">विज्ञाय च मुनिः प्राज्ञस्ततः शापं प्रयुक्तवान्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4802"><div class="verse-block verse-justify"><div class="line">`अहमग्निप्रभो नाम मुनिर्भृगुकुलान्वयः।</div><div class="line" style="position:relative;">मनसा निर्दहेयं च जगत्संधारयामि च॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4803"><div class="verse-block verse-justify"><div class="line">मम वश्यं जगत्सर्वं देवा यच्च चराचरम्।</div><div class="line">सन्ति देवाश्च ये भीताः स्वधर्मं न त्यजन्ति ये।</div><div class="line" style="position:relative;">स्वधर्माच्चलितान्सर्वान्वाङ्भात्रेणापि निर्दहे॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4804"><div class="verse-block verse-justify"><div class="line">किमङ्ग त्वं मया नीतः शरभत्वमनामयम्।</div><div class="line" style="position:relative;">क्रूरः स सर्वभूतेषु हीनश्चाशुचिरेव च ॥'॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4805"><div class="verse-block verse-justify"><div class="line">श्वा त्वं द्वीपित्वमापन्नो द्वीपी व्याघ्रत्वमागतः।</div><div class="line" style="position:relative;">व्याघ्रान्नागो मदपटुर्नागः सिंहत्वमागतः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4806"><div class="verse-block verse-justify"><div class="line">सिंहस्त्वं बलमापन्नो भूयः शरभतां गतः।</div><div class="line" style="position:relative;">मया स्नेहपरीतेन विसृष्टो न कुलान्वयः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4807"><div class="verse-block verse-justify"><div class="line">यस्मादेवमपापं मां पाप हिंसितुमिच्छसि।</div><div class="line" style="position:relative;">तस्मात्स्वयोनिमापन्नः पुनः श्वानो भविष्यसि॥३२॥</div></div></div>
