<div class="verse-block-wrapper" id="12-śāntiparva-b0673"><div class="verse-block verse-justify"><div class="line">कथयन्ति पुरावृत्तमितिहासमिमं जनाः।</div><div class="line" style="position:relative;">विदेहराज्ञः संवादं भार्यया सह भारत॥३॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b0675"><div class="verse-block verse-justify"><div class="line">धनान्यपत्यं मित्राणि रत्नानि विविधानि च।</div><div class="line" style="position:relative;">पन्थानं पावनं हित्वा जनको मौढ्यमास्थितः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0676"><div class="verse-block verse-justify"><div class="line">तं ददर्श प्रिया भार्या भैक्षवृत्तिमकिंचनम्।</div><div class="line" style="position:relative;">धानामुष्टिमुपासीनं निरीहं गतमत्सरम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0677"><div class="verse-block verse-justify"><div class="line">तमुवाच समामत्य भर्तारमकुतोभयम्।</div><div class="line" style="position:relative;">क्रुद्धा मनस्विनी भार्या विविक्ते हेतुमद्वचः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0678"><div class="verse-block verse-justify"><div class="line">कथमुत्सृज्य राज्यं स्वं धनधान्यसमन्वितम्।</div><div class="line" style="position:relative;">कापालीं वृत्तिमास्थाय धान्यमुष्टिमुपाससे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0679"><div class="verse-block verse-justify"><div class="line">प्रतिज्ञा तेऽन्यथा राजन्विचेष्टा चान्यथा तव।</div><div class="line" style="position:relative;">यद्राज्यं महदुत्सृज्य स्वल्पे लुभ्यसि पार्थिव॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0680"><div class="verse-block verse-justify"><div class="line">नैतेनातिथयो राजन्देवर्षिपितरस्तथा।</div><div class="line" style="position:relative;">अद्य शक्यास्त्वया भर्तुं मोघस्तेऽयं परिश्रमः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0681"><div class="verse-block verse-justify"><div class="line">देवतातिथिभिश्चैव पितृभिश्चैव पार्थिव।</div><div class="line" style="position:relative;">सर्वैरेतैः परित्यक्तः परिव्रजसि निष्क्रियः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0682"><div class="verse-block verse-justify"><div class="line">यस्त्वं त्रैविद्यवृद्धानां ब्राह्मणानां सहस्रशः।</div><div class="line" style="position:relative;">भर्ता भूत्वा च लोकस्य सोऽद्यान्यैर्भूतिमिच्छसि॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0683"><div class="verse-block verse-justify"><div class="line">श्रियं हित्वा प्रदीप्तां त्वं श्ववत्संप्रति वीक्ष्यसे।</div><div class="line" style="position:relative;">अपुत्रा जननी तेऽद्य कौसल्या चापतिस्त्वया॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0684"><div class="verse-block verse-justify"><div class="line">आश्रिता धर्मकामास्त्वां क्षत्रियाः पर्युपासते।</div><div class="line" style="position:relative;">त्वदाशामभिकाङ्क्षन्तः कृपणाः फलहेतुकाः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0685"><div class="verse-block verse-justify"><div class="line">तांश्च त्वं विफलान्कृत्वा कं नु लोकं गमिष्यसि।</div><div class="line" style="position:relative;">राजन्संशयिते मोक्षे परतन्त्रेषु देहिषु॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0686"><div class="verse-block verse-justify"><div class="line">नैव तेऽस्ति परो लोको नापरः पापकर्मणः।</div><div class="line" style="position:relative;">धर्म्यान्दारान्परित्यज्य यस्त्वमिच्छसि जीवितुम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0687"><div class="verse-block verse-justify"><div class="line">स्रजो गन्धानलंकारान्वासांसि विविधानि च।</div><div class="line" style="position:relative;">किमर्थमभिसंत्यज्य परिव्रजसि निष्क्रियः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0688"><div class="verse-block verse-justify"><div class="line">निपानं सर्वभूतानां भूत्वा त्वं पावनं महत।</div><div class="line" style="position:relative;">आढ्यो वनस्पतिर्भूत्वा सोन्यांस्त्वं पर्युपाससे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0689"><div class="verse-block verse-justify"><div class="line">खादन्ति हस्तिनं न्यासे क्रव्यादा बहवोऽप्युत।</div><div class="line" style="position:relative;">बहवः कृमयश्चैव किं पुनस्त्वामनर्थकम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0690"><div class="verse-block verse-justify"><div class="line">य इमां कुण्डिकां भिन्द्यान्त्रिविष्टब्धं च यो हरेत्।</div><div class="line" style="position:relative;">वासश्चापि हरेत्तस्मिन्कथं ते मानसं भवेत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b0691"><div class="verse-block verse-justify"><div class="line">यस्त्वं सर्वं समुत्सृज्य धानामुष्टिमनुग्रहः।</div><div class="line" style="position:relative;">यदनेन कृतं सर्वं किमिदं मम दीयते॥२१॥</div></div></div>
