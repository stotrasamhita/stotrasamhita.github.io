<div class="verse-block-wrapper" id="12-śāntiparva-b4119"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4120"><div class="verse-block verse-justify"><div class="line">अत्राप्युदाहरन्तीम-तिहासं पुरातनम्।</div><div class="line" style="position:relative;">प्रतर्दनो मैथिलश्च संग्रामं यत्र चक्रतुः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4121"><div class="verse-block verse-justify"><div class="line">यज्ञोपवीती संग्रामे जनको मिथिलाधिपः।</div><div class="line" style="position:relative;">योधानुद्धर्षयामास तन्निबोध युधिष्ठिर॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4122"><div class="verse-block verse-justify"><div class="line">जनको मैथिलो राजा महात्मा सर्वतत्त्ववित्।</div><div class="line" style="position:relative;">योधानां दर्शयामास स्वर्गं नरकमेव च॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4123"><div class="verse-block verse-justify"><div class="line">अभीरूणामिमे लोका भास्वन्तो हन्त पश्यत।</div><div class="line" style="position:relative;">पूर्णा गन्धर्वकन्याभिः सर्वकामदुहोऽक्षयाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4124"><div class="verse-block verse-justify"><div class="line">इमे पलायमानानां नरकाः प्रत्युपस्थिताः।</div><div class="line" style="position:relative;">अकीर्तिः शाश्वती चैव यतितव्यमनन्तरम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4125"><div class="verse-block verse-justify"><div class="line">तान्दृष्ट्वाऽरीन्विजयत भूत्वा संत्यागबुद्धयः।</div><div class="line" style="position:relative;">नरकस्याप्रतिष्ठस्य मा भूत वशवर्तिनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4126"><div class="verse-block verse-justify"><div class="line">त्यागमूलं हि शूराणां स्वर्गद्वारमनुत्तमम्।</div><div class="line" style="position:relative;">इत्युक्तास्ते नृपतिना योधाः परपुरंजय॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4127"><div class="verse-block verse-justify"><div class="line">अजयन्त रणे शत्रून्हर्षयन्तो नरेश्वरम्।</div><div class="line" style="position:relative;">तस्मात्त्यक्तात्मना नित्यं स्थातव्यं रणमूर्धनि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4128"><div class="verse-block verse-justify"><div class="line">गजानां रथिनो मध्ये रथानामनुसादिनः।</div><div class="line" style="position:relative;">सादिनामन्तरे स्थाप्यं पादातमपि दंशितम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4129"><div class="verse-block verse-justify"><div class="line">य एवं व्यूहते राजा स नित्यं जयते रिपून्।</div><div class="line" style="position:relative;">तस्मदितद्विधातव्यं नित्यमेव युधिष्ठिर॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4130"><div class="verse-block verse-justify"><div class="line">स्वर्गे सुकृतमिच्छन्तः सुयुद्धेनातिमन्यवः।</div><div class="line" style="position:relative;">क्षोभयेयुरनीकानि सागरं मकरा यथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4131"><div class="verse-block verse-justify"><div class="line">हर्षयेयुर्विषण्णांश्च व्यवस्थाप्य परस्परम्।</div><div class="line" style="position:relative;">तेषां च भूमिं रक्षेयुर्भग्नान्नात्यनुसारयेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4132"><div class="verse-block verse-justify"><div class="line">पुनरावर्तमानानां निराशानां च जीविते।</div><div class="line" style="position:relative;">वेगः सुदुःसहो राजंस्तस्मान्नात्यनुसारयेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4133"><div class="verse-block verse-justify"><div class="line">न हि प्रहर्तुमिच्छन्ति शूराः प्रद्रवतो भयात्।</div><div class="line" style="position:relative;">तस्मात्पलायमानानां कुर्यान्नात्यनुसारणम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4134"><div class="verse-block verse-justify"><div class="line">चराणामचरा ह्यन्नमदंष्ट्रा दंष्ट्रिणामपि।</div><div class="line" style="position:relative;">अपाणयः पाणिमतामन्नं शूरस्य कातराः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4135"><div class="verse-block verse-justify"><div class="line">समानपृष्ठोदरपाणिपादाःपश्चाच्छरं भीरवोऽनुव्रजन्ति।</div><div class="line" style="position:relative;">अतो भयार्ताः प्रणिपत्य भूयःकृत्वाञ्जलीनुपतिष्ठन्ति शूरान्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4136"><div class="verse-block verse-justify"><div class="line">शूरबाहुषु लोकोऽयं लम्बते पुत्रवत्सद।</div><div class="line" style="position:relative;">तस्मात्सर्वेषु लोकेषु शूरः संमानमर्हति॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b4137"><div class="verse-block verse-justify"><div class="line">न हि शौर्यात्परं किंचिन्त्रिषु लोकेषु विद्यते।</div><div class="line" style="position:relative;">शूरः सर्वं पालयति सर्वं शूरे प्रतिष्ठितम्॥१९॥</div></div></div></div>
