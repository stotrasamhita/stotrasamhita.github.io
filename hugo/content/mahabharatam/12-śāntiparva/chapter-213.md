<div class="verse-block-wrapper" id="12-śāntiparva-b8787"><div class="verse-block verse-justify"><div class="line">गुरुरुवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8788"><div class="verse-block verse-justify"><div class="line">चतुर्विधानि भूतानि स्थावराणि चराणि च।</div><div class="line">अव्यक्तप्रभवान्याहुरव्यक्तनिधनानि च।</div><div class="line" style="position:relative;">अव्यक्तलक्षणं विद्यादव्यक्तात्मात्मकं मनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8789"><div class="verse-block verse-justify"><div class="line">यथाऽश्वत्थकणीकायामन्तर्भूतो महाद्रुमः।</div><div class="line" style="position:relative;">निष्पन्नो दृश्यते व्यक्तमव्यक्तात्संभवस्तथा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8790"><div class="verse-block verse-justify"><div class="line">`आत्मानमनुसंयाति बुद्धिरव्यक्तजा तथा।</div><div class="line" style="position:relative;">तामन्वेति मनो यद्वल्लोहवर्मणि सन्निधौ ॥'॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8791"><div class="verse-block verse-justify"><div class="line">अभिद्रवत्ययस्कान्तमयोनिश्चेतनं यथा।</div><div class="line" style="position:relative;">स्वभावहेतुजा भावा यद्वदन्यदपीदृशम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8792"><div class="verse-block verse-justify"><div class="line">तद्वदव्यक्तजा भावाः कर्तुः कारणलक्षणाः।</div><div class="line" style="position:relative;">अचेतनाश्चेतयितुः कारणादभिसंगताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8793"><div class="verse-block verse-justify"><div class="line">न भूर्न खं द्यौर्भूतानि नर्षयो न सुरासुराः।</div><div class="line" style="position:relative;">नान्यदासीदृते जीवमासेदुर्न तु संहतिम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8794"><div class="verse-block verse-justify"><div class="line">सर्वं नित्यं सर्वगतं मनोहेतुत्वलक्षणम्।</div><div class="line" style="position:relative;">अज्ञानकर्म निर्दिष्टमेतत्कारणलक्षणम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8795"><div class="verse-block verse-justify"><div class="line">तत्कारणेन संयुक्तं कार्यसंग्रहकारकम्।</div><div class="line" style="position:relative;">येनैतद्वर्तते चक्रमनादिनिधनं महत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8796"><div class="verse-block verse-justify"><div class="line">`येन स्वभावसद्भावं हेतुभूता सकारणा।</div><div class="line" style="position:relative;">एवं प्राकृतविस्तारो ह्याश्रित्य पुरुषं परम् ॥'॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8797"><div class="verse-block verse-justify"><div class="line">अव्यक्तनाभं व्यक्तारं विकारपरिमण्डलम्।</div><div class="line" style="position:relative;">क्षेत्रज्ञाधिष्ठितं चक्रं स्निग्धाक्षं वर्तते ध्रुवम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8798"><div class="verse-block verse-justify"><div class="line">स्निग्धत्वात्तिलवत्सर्वं चक्रेऽस्मिन्पीड्यते जगत्।</div><div class="line" style="position:relative;">तिलपीडैरिवाक्रम्य भोगैरज्ञानसंभवैः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8799"><div class="verse-block verse-justify"><div class="line">`प्राणेनायं हि शान्ते तु विरोधात्प्रतिपालनम्।</div><div class="line" style="position:relative;">देहस्येषून्य आस्ते यः शुद्धोऽचिन्त्यः सनातनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8800"><div class="verse-block verse-justify"><div class="line">भ्रामयन्नेषतो याति कालचक्रसमन्वितः।</div><div class="line" style="position:relative;">भूतानि मोहयन्नित्यं चक्रस्य च रयं गतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8801"><div class="verse-block verse-justify"><div class="line">स्नेहद्रव्यसमायोगे क्षेत्रपाचं न वस्तुषु।</div><div class="line">तिलवत्पीडिते चक्रे ह्याधियन्त्रनिपीडिते।</div><div class="line" style="position:relative;">बहिश्चाधिष्ठिते यद्वज्ज्ञानिनां कर्मसंभवम्'॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8802"><div class="verse-block verse-justify"><div class="line">कर्म तत्कुरुते तर्षादहंकारपरिग्रहम्।</div><div class="line" style="position:relative;">कार्यकारणसंयोगे स हेतुरुपपादितः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8803"><div class="verse-block verse-justify"><div class="line">`यथाऽऽकर्ण्य च तच्छिष्यस्तत्वज्ञानमनुत्तमम्।</div><div class="line">'नात्येति कारणं कार्यं न कार्यं कारणं तथा।</div><div class="line" style="position:relative;">कार्याण्यमूनि करणे कालो भवति हेतुमान्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8804"><div class="verse-block verse-justify"><div class="line">हेतुयुक्ताः प्रकृतयो विकाराश्च परस्परम्।</div><div class="line" style="position:relative;">अन्योन्यमभिवर्तन्ते पुरुषाधिष्ठिताः सदा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8805"><div class="verse-block verse-justify"><div class="line">सत्वरजस्तामसैर्भावैश्च्युतो हेतुबलान्वितः।</div><div class="line" style="position:relative;">क्षेत्रज्ञमेवानयाति पांसुर्वातेरितो यथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8806"><div class="verse-block verse-justify"><div class="line">न च तैः स्पृश्यते भावैर्न ते तेन महात्मना।</div><div class="line" style="position:relative;">सरजस्कोऽरजस्कश्च स वै वायुर्भवेद्यथा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8807"><div class="verse-block verse-justify"><div class="line">तथैतदन्तरं विद्यात्सत्वक्षेत्रज्ञयोर्बुधः।</div><div class="line" style="position:relative;">अभ्यासात्स तथा युक्तो न गच्छेत्प्रकृतिं पुनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8808"><div class="verse-block verse-justify"><div class="line">संदेहमेतमुत्पन्नमच्छिनद्भगवानृषिः।</div><div class="line" style="position:relative;">तथा वार्तां समीक्षेत कृतलक्षणसंविदम्॥२२॥</div></div></div>
