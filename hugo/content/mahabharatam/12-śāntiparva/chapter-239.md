<div class="verse-block-wrapper" id="12-śāntiparva-b10181"><div class="verse-block verse-justify"><div class="line">व्यास उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10182"><div class="verse-block verse-justify"><div class="line">प्रत्याहारं तु वक्ष्यामि शर्वर्यादौ गतेऽहनि।</div><div class="line" style="position:relative;">यथेदं कुरुतेऽध्यात्मं सुसूक्ष्मं विश्वमीश्वरः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10183"><div class="verse-block verse-justify"><div class="line">दिवि सूर्यस्तथा सप्त दहन्ति शिखिनोऽर्चिषः।</div><div class="line" style="position:relative;">सर्वमेतत्तदाऽर्चिर्भिः पूर्णं जाज्वल्यते जगत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10184"><div class="verse-block verse-justify"><div class="line">पृथिव्यां यानि भूतानि जङ्गमानि ध्रुवाणि च।</div><div class="line" style="position:relative;">तान्येवाग्रे प्रलीयन्ते भूमित्वमुपयान्ति च॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10185"><div class="verse-block verse-justify"><div class="line">ततः प्रलीने सर्वास्मिन्स्थावरे जङ्गमे तथा।</div><div class="line" style="position:relative;">अकाष्ठा निस्तृणा भूमिर्दृश्यते कूर्मपृष्ठवत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10186"><div class="verse-block verse-justify"><div class="line">भूमेरपि गुणं गन्धमाप आददते यदा।</div><div class="line" style="position:relative;">आत्तगन्धा तदा भूमिः प्रलयत्वाय कल्पते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10187"><div class="verse-block verse-justify"><div class="line">आपस्तत्र प्रतिष्ठन्ते ऊर्मिमत्यो महास्वनाः।</div><div class="line" style="position:relative;">सर्वमेवेदमापूर्य तिष्ठन्ति च चरन्ति च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10188"><div class="verse-block verse-justify"><div class="line">अपामपि गुणांस्तात ज्योतिराददते यदा।</div><div class="line" style="position:relative;">आपस्तदा त्वात्तगुणा ज्योतिः षूपरमन्ति वै॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10189"><div class="verse-block verse-justify"><div class="line">यदाऽऽदित्यं स्थितं मध्ये गूहन्ति शिखिनोऽर्चिषः।</div><div class="line" style="position:relative;">सर्वमेवेदमर्चिर्भिः पूर्णं जाज्वल्यते नमः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10190"><div class="verse-block verse-justify"><div class="line">ज्योतिषोऽपि गुणं रूपं वायुराददते यदा।</div><div class="line" style="position:relative;">प्रशाम्यति ततो ज्योतिर्वायुर्दोधूयते महान्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10191"><div class="verse-block verse-justify"><div class="line">ततस्तु मूलमासाद्य वायुः संभवमात्मनः।</div><div class="line" style="position:relative;">अधश्चोर्ध्वं च तिर्यक्च दोधवीति दिशो दश॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10192"><div class="verse-block verse-justify"><div class="line">वायोरपि गुणं स्पर्शमाकाशं ग्रसते यदा।</div><div class="line" style="position:relative;">प्रशाम्यति तदा वायुः खं तु तिष्ठति नानदन्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10193"><div class="verse-block verse-justify"><div class="line">अरूपमरसस्पर्शमगन्धं न च मूर्तिमत्।</div><div class="line" style="position:relative;">सर्वलोकप्रणुदितं स्वं तु तिष्ठति नानदत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10194"><div class="verse-block verse-justify"><div class="line">आकाशस्य गुणं शब्दमभिव्यक्तात्मकं मनः।</div><div class="line">`ग्रसते च यदा सोऽपि शाम्यति प्रतिसंचरे।</div><div class="line" style="position:relative;">'मनसो व्यक्तमव्यक्तं ब्राह्मः संप्रतिसंचरः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10195"><div class="verse-block verse-justify"><div class="line">तदाऽऽत्मगुणमाविश्य मनो ग्रसति चन्द्रमाः।</div><div class="line" style="position:relative;">मनस्युपरते चात्मा चन्द्रमस्युपतिष्ठते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10196"><div class="verse-block verse-justify"><div class="line">तं तु कालेन महता संकल्पं कुरुते वशे।</div><div class="line" style="position:relative;">चित्तं ग्रसति संकल्पस्तच्च ज्ञानमनुत्तमम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10197"><div class="verse-block verse-justify"><div class="line">कालो गिरति विज्ञानं कालं बलमिति श्रुतिः।</div><div class="line" style="position:relative;">बलं कालो ग्रसति तु तं विद्वान्कुरुते वशे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10198"><div class="verse-block verse-justify"><div class="line">[आकाशस्य तदा घोषं तं विद्वान्कुरुतेऽऽत्मनि।</div><div class="line">]तदव्यक्तं परं ब्रह्म तच्छाश्वतमनुत्तमम्।</div><div class="line" style="position:relative;">एवं सर्वाणि भूतानि ब्रह्मैव प्रतिसंहरेत्॥१८॥</div></div></div>
