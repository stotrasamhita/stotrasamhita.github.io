<div class="verse-block-wrapper" id="12-śāntiparva-b1686"><div class="verse-block verse-justify"><div class="line">ततो विसर्जयामास सर्वास्ताः प्रकृतीर्नृपः।</div><div class="line" style="position:relative;">विविशुश्चाभ्यनुज्ञाता यथास्वानि गृहाणि ते॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1687"><div class="verse-block verse-justify"><div class="line">ततो युधिष्ठिरो राजा भीमं भीमपराक्रमम्।</div><div class="line" style="position:relative;">सान्त्वयन्नब्रवीच्छ्रीमानर्जुनं यमजौ तथा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1688"><div class="verse-block verse-justify"><div class="line">शत्रुभिर्विविधैः शस्त्रैः क्षतदेहा महारणे।</div><div class="line" style="position:relative;">श्रान्ता भवन्तः सुभृशं तापिताः शोकमन्युभिः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1689"><div class="verse-block verse-justify"><div class="line">अरण्ये दुःखवसतिर्मत्कृते भरतर्षभाः।</div><div class="line" style="position:relative;">भवद्भिरनुभूता हि यथा कापुरुषैस्तथा॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1690"><div class="verse-block verse-justify"><div class="line">यथासुखं यथाजोषं जयोऽयमनुभूयताम्।</div><div class="line" style="position:relative;">विश्रान्ताँल्लब्धविश्वासाञ्श्वः समेताऽस्मि वः पुनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1691"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनगृहं प्रासादैरुपशोभितम्।</div><div class="line" style="position:relative;">बहुरत्नसमाकीर्णं दासीदाससमाकुलम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1692"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्राभ्यनुज्ञातं भ्रात्रा दत्तं वृकोदरः।</div><div class="line" style="position:relative;">प्रतिपेदे महाबाहुर्मन्दिरं मघवानिव॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1693"><div class="verse-block verse-justify"><div class="line">यथा दुर्योधनगृहं तथा दुःशासनस्य तु।</div><div class="line" style="position:relative;">प्रासादभालासंयुक्तं हेमतोरणभूषितम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1694"><div class="verse-block verse-justify"><div class="line">दासीदाससुसंपूर्णं प्रभूतधनधान्यवत्।</div><div class="line" style="position:relative;">प्रतिपेदे महाबाहुरर्जुनो राजशासनात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1695"><div class="verse-block verse-justify"><div class="line">दुर्मर्षणस्य भवनं दुःशासनगृहाद्वरम्।</div><div class="line" style="position:relative;">कुबेरभवनप्रख्यं मणिहेमविभूषितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1696"><div class="verse-block verse-justify"><div class="line">नकुलाय वरार्हाय कर्शिताय महावने।</div><div class="line" style="position:relative;">ददौ प्रीतो महाराज धर्मपुत्रो युधिष्ठिरः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1697"><div class="verse-block verse-justify"><div class="line">दुर्मुखस्य च वेश्माग्र्यं श्रीमत्कनकभूषणम्।</div><div class="line" style="position:relative;">पूर्णपझदलाक्षीणां स्त्रीणां शयनसंकुलम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1698"><div class="verse-block verse-justify"><div class="line">प्रददौ सहदेवाय संततं प्रियकारिणे।</div><div class="line" style="position:relative;">मुमुदे तच्च लब्ध्वाऽसौ कैलासं धनदो यथा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1699"><div class="verse-block verse-justify"><div class="line">युयुत्सुर्विदुरश्चैव सञ्जयश्च विशांपते।</div><div class="line" style="position:relative;">सुधर्मा चैव धौम्यश्च यथा स्वाञ्जग्मुरालयान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1700"><div class="verse-block verse-justify"><div class="line">सह सात्यकिना शौरिरर्जुनस्य निवेशनम्।</div><div class="line" style="position:relative;">विवेश पुरुषव्याघ्रो व्याघ्रो गिरिगुहामिव॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b1701"><div class="verse-block verse-justify"><div class="line">तत्र भक्ष्यान्नपानैस्ते मुदिताः सुसुखोषिताः।</div><div class="line" style="position:relative;">सुखप्रबद्धा राजानमुपतस्थुर्युधिष्ठिरम्॥१७॥</div></div></div></div>
