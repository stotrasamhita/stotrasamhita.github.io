<div class="verse-block-wrapper" id="12-śāntiparva-b3884"><div class="verse-block verse-justify"><div class="line">यद्वॄत्तमुपजीवन्ति प्रकृतिस्थस्य मानवाः।</div><div class="line" style="position:relative;">तदेव विषमस्थस्य स्वजनोऽपि न मृष्यते॥४॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b3886"><div class="verse-block verse-justify"><div class="line">सद्वॄत्ताचरितां वृत्तिं क्षत्रियो नानुवर्तते।</div><div class="line" style="position:relative;">जितानामजितानां च क्षत्रधर्मादपैति सः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3887"><div class="verse-block verse-justify"><div class="line">द्विषन्तं कृतकल्याणं गृहीत्वा नृपतिं रणे।</div><div class="line" style="position:relative;">यो न नयते द्वेषात्क्षत्रधर्मादपैति सः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3888"><div class="verse-block verse-justify"><div class="line">शक्तः आत्सुसुखो राजा कुर्यात्तारणमापदि।</div><div class="line" style="position:relative;">प्रियो अति भूतानां न च विभ्रश्यते श्रियः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3889"><div class="verse-block verse-justify"><div class="line">अप्रियं यस्य कुर्वीत भूयस्तस्य प्रियं चरेत्।</div><div class="line" style="position:relative;">अचिरेण प्रियः स स्याद्योऽप्रिये प्रियमाचरेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3890"><div class="verse-block verse-justify"><div class="line">मृषावादं परिहरेत्कुर्यात्प्रियमयाचितः।</div><div class="line" style="position:relative;">न कामान्न च संरम्भान्न द्वेषाद्धर्ममुत्सृजेत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3891"><div class="verse-block verse-justify"><div class="line">`अमाययैव वर्तेत न च सत्यं त्यजेद्बुधः।</div><div class="line" style="position:relative;">दमं धर्मं च शीलं च क्षत्रधर्मं प्रजाहितम् ॥'॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3892"><div class="verse-block verse-justify"><div class="line">नापत्रपेत प्रश्नेषु नाभिभाविगिरं सृजेत्।</div><div class="line" style="position:relative;">न त्वरेत न चासूयेत्तथा संगृह्यते परः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3893"><div class="verse-block verse-justify"><div class="line">प्रिये नातिभृशं हृष्येदप्रिये न च संज्वरेत्।</div><div class="line" style="position:relative;">न तप्येदर्थकृच्छ्रेषु प्रजाहितमनुस्मरन्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3894"><div class="verse-block verse-justify"><div class="line">यः प्रियं कुरुते नित्यं गुणतो वसुधाधिपः।</div><div class="line" style="position:relative;">तस्य कर्माणि सिद्ध्यन्ति न च संत्यज्यते श्रिया॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3895"><div class="verse-block verse-justify"><div class="line">निवृत्तं प्रतिकूलेभ्यो वर्तमानमनुप्रिये।</div><div class="line" style="position:relative;">भक्तं भजेत नृपतिस्तद्वै वृत्तं सतामिह॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3896"><div class="verse-block verse-justify"><div class="line">अप्रकीर्णेन्द्रियग्राममत्यन्तानुगतं शुचिम्।</div><div class="line" style="position:relative;">शक्तं चैवानुरक्तं च युञ्ज्यान्महति कर्मणि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3897"><div class="verse-block verse-justify"><div class="line">`श्रेयसो लक्षणं चैतद्विक्रमो यत्र दृश्यते।</div><div class="line">कीर्तिप्रधानो यश्च स्यात्समये यश्च तिष्ठति।</div><div class="line" style="position:relative;">समर्थान्पूजयेद्यश्च न च स्पर्धेत यश्चतैः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3898"><div class="verse-block verse-justify"><div class="line">एवमेतैर्गुणैर्युक्तो योऽनुरज्यति भूमिपम्।</div><div class="line" style="position:relative;">भर्तुरर्थेष्वप्रमत्तं नियुञ्ज्यादर्थकर्मणि॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3899"><div class="verse-block verse-justify"><div class="line">मूढमैन्द्रियकं लुब्धमनार्यचरितं शठम्।</div><div class="line" style="position:relative;">अनतीतोपधं हिंस्रं दुर्बुद्धिमबहुश्रुतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3900"><div class="verse-block verse-justify"><div class="line">त्यक्तोपात्तं मद्यरतं द्यूतस्त्रीमृगयापरम्।</div><div class="line" style="position:relative;">कार्ये महति युञ्जानो हीयते नृपतिः श्रिया॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3901"><div class="verse-block verse-justify"><div class="line">रक्षितात्मा च यो राजा रक्ष्यान्यश्चानुरक्षति।</div><div class="line" style="position:relative;">प्रजाश्च तस्य वर्धन्ते सुखं च महदश्नुते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3902"><div class="verse-block verse-justify"><div class="line">ये केचिद्भूमिपतयः सर्वांस्तानन्ववेक्षयेत्।</div><div class="line" style="position:relative;">सुहृद्भिरनभिख्यातैस्तेन राजा न रिष्यते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3903"><div class="verse-block verse-justify"><div class="line">अपकृत्य बलस्थस्व दूरस्थोऽस्मीति नाश्वसेत्।</div><div class="line" style="position:relative;">श्येनाभिपतनैरेते निपतन्ति प्रमाद्यतः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3904"><div class="verse-block verse-justify"><div class="line">दृढमूलस्त्वदुष्टात्मा विदित्वा बलमात्मनः।</div><div class="line" style="position:relative;">अबलानभियुञ्जीत न तु ये बलवत्तराः॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b3907"><div class="verse-block verse-justify"><div class="line">रक्षाधिकरणं युद्धं तथा धर्मानुशासनम्।</div><div class="line" style="position:relative;">मन्त्रचिन्ता सुखं काले पञ्चभिर्वर्धते मही॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3908"><div class="verse-block verse-justify"><div class="line">एतानि यस्य गुप्तानि स राजा राजसत्तम।</div><div class="line" style="position:relative;">सततं वर्तमानोऽत्र राजा भुङ््क्ते महीमिमाम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3909"><div class="verse-block verse-justify"><div class="line">नैतान्येकेन शक्यानि सातत्येनानुवीक्षितुम्।</div><div class="line" style="position:relative;">एतेष्वाप्तान्प्रतिष्ठाप्य राजा भुङ्क्ते चिरं महीम्॥२९॥</div></div></div>
