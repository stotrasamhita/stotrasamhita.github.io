<div class="verse-block-wrapper" id="12-śāntiparva-b10861"><div class="verse-block verse-justify"><div class="line">प्रियान्पुत्रान्वयस्यांश्च भ्रातॄन्मातॄः पितृनपि।</div><div class="line" style="position:relative;">अपध्यास्यन्ति यद्येवं मृतास्तेभ्यो विभेम्यहम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10862"><div class="verse-block verse-justify"><div class="line">कृपणाश्रुपरिक्लेदो दहेन्मां शाश्वतीः समाः।</div><div class="line" style="position:relative;">तेभ्योऽहं बलवद्भीता शरणं त्वामुपागता॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10863"><div class="verse-block verse-justify"><div class="line">यमस्य भवने देव पात्यन्ते पापकर्मिणः।</div><div class="line" style="position:relative;">प्रसादये त्वां वरद प्रसादं कुरु मे प्रभो॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10864"><div class="verse-block verse-justify"><div class="line">एतदिच्छाम्यहं कामं त्वत्तो लोकपितामह।</div><div class="line">इच्छेयं त्वत्प्रसादाच्च तपस्तप्तुं महेश्वर ॥पितामह उवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10865"><div class="verse-block verse-justify"><div class="line">त्वं हिं संहारबुद्ध्या मे चिन्तिता रुषितेन च।</div><div class="line" style="position:relative;">तस्मात्संहर सर्वास्त्वं प्रजा मा च विचारय॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10866"><div class="verse-block verse-justify"><div class="line">एतदेवमवश्यं हि भविता नैतदन्यथा।</div><div class="line" style="position:relative;">क्रियतामनवद्याङ्गि यथोक्तं मद्वचोऽनघे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10867"><div class="verse-block verse-justify"><div class="line">एवमुक्ता महाबाहो मृत्युः परपुरंजय।</div><div class="line" style="position:relative;">न व्याजहार तस्थौ च प्रह्वा भगवदुन्मुखी॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10868"><div class="verse-block verse-justify"><div class="line">पुनः पुनरथोक्ता सा गतसत्त्वेव भामिनी।</div><div class="line" style="position:relative;">तूष्णीमासीत्ततो देवो लोकानामीश्वरेश्वरः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10869"><div class="verse-block verse-justify"><div class="line">प्रससाद किल ब्रह्मा स्वयमेवात्मनाऽऽत्मनि।</div><div class="line" style="position:relative;">स्मयमानश्च लोकेशो लोकान्सर्वानवैक्षत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10870"><div class="verse-block verse-justify"><div class="line">निवृत्तरोषे तस्मिंस्तु भगवत्यपराजिते।</div><div class="line" style="position:relative;">सा कन्याऽथ जगामास्य समीपादिति नःश्रुतम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10871"><div class="verse-block verse-justify"><div class="line">अपसृत्याप्रतिश्रुत्य प्रजासंहरणं तदा।</div><div class="line" style="position:relative;">त्वरमाणेव राजेन्द्र मृत्युर्धेनुकमभ्यगात्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10872"><div class="verse-block verse-justify"><div class="line">सा तत्र परमं देवी तपोऽचरत दुश्चरम्।</div><div class="line" style="position:relative;">समा ह्येकपदे तस्थौ दशपद्मानि पञ्च च॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10873"><div class="verse-block verse-justify"><div class="line">तां तथा कुर्वतीं तत्र तपः परमदुश्चरम्।</div><div class="line" style="position:relative;">पुनरेव महातेजा ब्रह्मा वचनमब्रवीत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10874"><div class="verse-block verse-justify"><div class="line">कुरुष्व मे वचो मृत्यो तदनादृत्य सत्वरा।</div><div class="line" style="position:relative;">तथैवैकपदे तात पुनरन्यानि सप्त सा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10875"><div class="verse-block verse-justify"><div class="line">तस्थौ पद्मानि षट् चैव पञ्च द्वे चैव मानद।</div><div class="line" style="position:relative;">भूयः पद्मायुतं तात मृगैः सह चचार सा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10876"><div class="verse-block verse-justify"><div class="line">द्वे चायुते नरश्रेष्ठ वाय्वाहारा महामते।</div><div class="line" style="position:relative;">पुनरेव ततो राजन्मौनमातिष्ठदुत्तमम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10877"><div class="verse-block verse-justify"><div class="line">अप्सु वर्षसहस्राणि सप्त चैकं च पार्थिव।</div><div class="line" style="position:relative;">ततो जगाम सा कन्या कौशिकीं नृपसत्तम॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10878"><div class="verse-block verse-justify"><div class="line">तत्र वायुजलाहारा चचार नियमं पुनः।</div><div class="line" style="position:relative;">ततो ययौ महाभागा गङ्गां मेरुं च केवलम्॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b10880"><div class="verse-block verse-justify"><div class="line">तत्राङ्गुष्ठेन राजेन्द्र निखर्वमचरत्तपः।</div><div class="line" style="position:relative;">तस्थौ पितामहं चैव तोषयामास यत्नतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10881"><div class="verse-block verse-justify"><div class="line">ततस्तामब्रवीत्तत्र लोकानां प्रपितामहः।</div><div class="line" style="position:relative;">किमिदं वर्तसे पुत्रि क्रियतां मम तद्वचः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10882"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीत्पुनर्मृत्युर्भगवन्तं पितामहम्।</div><div class="line" style="position:relative;">न हरेयं प्रजा देव पुनस्त्वाऽहं प्रसादये॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10883"><div class="verse-block verse-justify"><div class="line">तामधर्मभयाद्भीतां पुनरेव प्रयाचतीम्।</div><div class="line" style="position:relative;">तदाऽब्रवीद्देवदेवो निगृह्येदं वचस्ततः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10884"><div class="verse-block verse-justify"><div class="line">अधर्मो नास्ति ते मृत्यो संयच्छेमाः प्रजाः शुभे।</div><div class="line" style="position:relative;">मयाऽप्युक्तं मृषा भद्रे भविता नेह किंचन॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10885"><div class="verse-block verse-justify"><div class="line">धर्मः सनातनश्च त्वामिहैवानुप्रवेक्ष्यति।</div><div class="line" style="position:relative;">अहं च विबुधाश्चैव त्वद्धिते निरताः सदा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10886"><div class="verse-block verse-justify"><div class="line">इमन्यं च ते कामं ददानि मनसेप्सितम्।</div><div class="line" style="position:relative;">न त्वां दोषेण यास्यन्ति व्याधिसंपीडिताः प्रजाः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10887"><div class="verse-block verse-justify"><div class="line">पुरुषेषु च रूपेण पुरुषस्त्वं भविष्यसि।</div><div class="line" style="position:relative;">स्त्रीषु स्त्रीरूपिणी चैव तृतीयेषु नपुंसकम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10888"><div class="verse-block verse-justify"><div class="line">सैवमुक्ता महाराज कृताञ्जलिरुवाच ह।</div><div class="line" style="position:relative;">पुनरेव महात्मानं नेति देवेशमव्ययम्॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b10890"><div class="verse-block verse-justify"><div class="line">`त्वं हि शक्ता च युक्ता च पूर्वोत्पन्ना च भामिनि।</div><div class="line" style="position:relative;">अनुशिष्टा च निर्दोषा तस्मात्त्वं कुरु मे मतम्'॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10891"><div class="verse-block verse-justify"><div class="line">यानश्रुबिन्दून्पतितानपश्यंये पाणिभ्यां धारितास्ते पुरस्तात्।</div><div class="line" style="position:relative;">ते व्याधयो मानवान्घोररूपाःप्राप्ते काले पीडयिष्यन्ति मृत्यो॥३६॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b10893"><div class="verse-block verse-justify"><div class="line">एवं धर्मं पालयिष्यस्यथो त्वंन चात्मानं मञ्जयिष्यस्यधर्मे।</div><div class="line" style="position:relative;">तस्मात्कामं रोचयाभ्यागतं त्वंसा त्वं साधो संहरस्वेह जन्तून्॥३८॥</div></div></div>
