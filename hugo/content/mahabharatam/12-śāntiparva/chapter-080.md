<div class="verse-block-wrapper" id="12-śāntiparva-b3353"><div class="verse-block verse-justify"><div class="line">यदप्यल्पतरं कर्म तदप्येकेन दुष्करम्।</div><div class="line" style="position:relative;">पुरुषेणासहायेन किमु राज्यं पितामह॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3354"><div class="verse-block verse-justify"><div class="line">किंशीलः किंसमाचारो राज्ञो यः सचिवो भवेत्।</div><div class="line">कीदृशे विश्वसेद्राजा कीदृशे न च विश्वसेत् ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3355"><div class="verse-block verse-justify"><div class="line">चतुर्विधानि मित्राणि राज्ञां राजन्भवन्त्युत।</div><div class="line" style="position:relative;">सहार्थो भजः निश्च सहजः कृत्रिमस्तथा॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3356"><div class="verse-block verse-justify"><div class="line">धर्मात्मा पञ्चमं मित्रं स तु नैकस्य न द्वयोः।</div><div class="line" style="position:relative;">यतो धर्मस्ततो वा स्यान्मध्यस्थो वा ततो भवेत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3357"><div class="verse-block verse-justify"><div class="line">यो यस्यार्थो न रोचेत न तं तस्य प्रकाशयेत।</div><div class="line">`मित्राणां प्रकृतिर्नास्ति त्वमित्राणां च भारत।</div><div class="line" style="position:relative;">उपकाराद्भवेन्मित्रमपकाराद्भवेदरिः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3358"><div class="verse-block verse-justify"><div class="line">यस्यैव हि मनुष्यस्य नरो मरणमृच्छति।</div><div class="line" style="position:relative;">तस्य पर्यागते काले पुनर्जीवितुमिच्छति॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3359"><div class="verse-block verse-justify"><div class="line">धर्माधर्मेण राजानश्चरन्ति विजिगीषवः।</div><div class="line">चतुर्णां मध्यमौ श्रेष्ठौ नित्यं शङ्क्यौ तथाऽपरौ।</div><div class="line" style="position:relative;">सर्वे नित्यं शङ्कितव्याः प्रत्यक्षं कार्यमात्मनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3360"><div class="verse-block verse-justify"><div class="line">न हि राज्ञा प्रमादो वै कर्तव्यो मित्ररक्षणे।</div><div class="line" style="position:relative;">प्रमादिनं हि राजानं लोकाः परिभवन्त्युत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3361"><div class="verse-block verse-justify"><div class="line">असाधुः साधुतामेति साधुर्भवति दारुणः।</div><div class="line" style="position:relative;">अरिश्च मित्रं भवति मित्रं चापि प्रदुष्यति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3362"><div class="verse-block verse-justify"><div class="line">अनित्यचित्तः पुरुषस्तस्मिन्को जातु विश्वसेत्।</div><div class="line" style="position:relative;">तस्मात्प्रधानं यत्कार्यं प्रत्यक्षं तत्समाचरेत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3363"><div class="verse-block verse-justify"><div class="line">एकान्तेन हि विश्वासः कृत्स्नो कर्मार्थनाशकः।</div><div class="line" style="position:relative;">अविश्वासश्च सर्वत्र मृत्युर्नापि विशिष्यते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3364"><div class="verse-block verse-justify"><div class="line">अकालमृत्युर्विश्वासोऽविश्वसन्हि विपद्यते।</div><div class="line" style="position:relative;">यस्मिन्करोति विश्वासमिच्छतस्तस्य जीवति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3365"><div class="verse-block verse-justify"><div class="line">तस्माद्विश्वसितव्यं च शङ्कितव्यं च केषुचित्।</div><div class="line" style="position:relative;">एषा नीतिगतिस्तात लक्ष्मीश्चैषा सनातनी॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3366"><div class="verse-block verse-justify"><div class="line">यं मन्येत ममाभावादिममर्थागमः स्पृशेत्।</div><div class="line" style="position:relative;">नित्यं तस्माच्छङ्कितव्यममित्रं तं विदुर्बुधाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3367"><div class="verse-block verse-justify"><div class="line">यस्य क्षेत्रादप्युदकं क्षेत्रमन्यस्य गच्छति।</div><div class="line" style="position:relative;">न तत्रानिच्छतस्तस्य भिद्येरन्सर्वसेतवः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3368"><div class="verse-block verse-justify"><div class="line">तथैवात्युदकाद्भीतस्तस्य भेदनमिच्छति।</div><div class="line" style="position:relative;">यमेवंलक्षणं विद्यात्तममित्रं विदुर्बुधाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3369"><div class="verse-block verse-justify"><div class="line">यस्तु वृद्ध्या न तप्येत क्षये दीनतरो भवेत्।</div><div class="line" style="position:relative;">एतदुत्तममित्रस्य निमित्तमभिचक्षते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3370"><div class="verse-block verse-justify"><div class="line">यन्मन्येत ममाभावादस्याभावो भवेदिति।</div><div class="line" style="position:relative;">तस्मिन्कुर्वीत विश्वासं यथा पितरी वै तथा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3371"><div class="verse-block verse-justify"><div class="line">तं शक्त्या वर्तमानं च सर्वतः परिबृंहयेत्।</div><div class="line" style="position:relative;">नित्यं क्षताद्वारयति यो धर्मेष्वपि कर्मसु॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3372"><div class="verse-block verse-justify"><div class="line">क्षताद्भीतं विजानीयादुत्तमं मित्रलक्षणम्।</div><div class="line" style="position:relative;">ये यस्य क्षयमिच्छन्ति ते तस्य रिपवः स्मृताः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3373"><div class="verse-block verse-justify"><div class="line">व्यसनान्नित्यभीतो यः समृद्ध्या यो न दुष्यति।</div><div class="line" style="position:relative;">यत्स्यादेवंविधं मित्रं तदात्मसममुच्यते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3374"><div class="verse-block verse-justify"><div class="line">रूपवर्णस्वरोपेतस्तितिक्षुरनसूयकः।</div><div class="line" style="position:relative;">कुलीनः शीलसंपन्नः स ते स्यात्प्रत्यनन्तरः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3375"><div class="verse-block verse-justify"><div class="line">मेधावी स्मृतिमान्दक्षः प्रकृत्या चानृशंस्यवान्।</div><div class="line" style="position:relative;">यो मानितोऽमानितो वा न सन्तुष्येत्कथंचन॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3376"><div class="verse-block verse-justify"><div class="line">ऋत्विग्वा यदि वाऽऽचार्यः सखा वाऽत्यंतसत्कृतः।</div><div class="line" style="position:relative;">गृहे वसेदमात्यस्ते स स्यात्परमपूजितः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3377"><div class="verse-block verse-justify"><div class="line">संविद्याः परमं मित्रं प्रकृतिं चार्थधर्मयोः।</div><div class="line" style="position:relative;">विश्वासस्ते भवेत्तत्र यथा पितरि वै तथा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3378"><div class="verse-block verse-justify"><div class="line">नैव द्वौ न त्रयः कार्या न मृष्येरन्परस्परम्।</div><div class="line" style="position:relative;">एकार्थे हेतुभूतानां भेदो भवति सर्वदा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3379"><div class="verse-block verse-justify"><div class="line">कीर्तिप्रधानो यस्त स्याद्यश्च स्यात्समये स्थितः।</div><div class="line" style="position:relative;">समर्थान्यश्च न द्वेष्टि नानर्थान्कुरुते च यः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3380"><div class="verse-block verse-justify"><div class="line">यो न कामाद्भयाल्लोभात्क्रोधाद्वा धर्ममुत्सृजेत्।</div><div class="line" style="position:relative;">दक्षः पर्याप्तवचनः स ते स्यात्प्रत्यनन्तरः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3381"><div class="verse-block verse-justify"><div class="line">कुलीनः शीलसंपन्नस्तितिक्षुरविकत्थनः।</div><div class="line" style="position:relative;">शूरश्चार्यश्च विद्वांश्च प्रतिपत्तिविशारदः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3382"><div class="verse-block verse-justify"><div class="line">एते ह्यमात्याः कर्तव्याः सर्वकर्मस्ववस्थिताः।</div><div class="line" style="position:relative;">पूजिताः संबिभक्ताश्च सुसहायाः स्वनुष्ठिताः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3383"><div class="verse-block verse-justify"><div class="line">कृत्स्नप्रेते विनिक्षिप्ताः प्रतिरूपेषु कर्मसु।</div><div class="line" style="position:relative;">युक्ता महत्सु कार्येषु श्रेयांस्युत्पादयन्त्युत॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3384"><div class="verse-block verse-justify"><div class="line">एते कर्माणि कुर्वन्ति स्पर्धमाना मिथः सदा।</div><div class="line" style="position:relative;">अनुतिष्ठन्ति चैवार्थमाचक्षाणाः परस्परम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3385"><div class="verse-block verse-justify"><div class="line">ज्ञातिभ्यो बिभियाश्चैव मृत्योरिव यतस्तदा।</div><div class="line" style="position:relative;">उपराजेव राजर्धि ज्ञातिर्न सहते सदा॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3386"><div class="verse-block verse-justify"><div class="line">ऋजोर्मृदोर्वदान्यस्य ह्रीमतः सत्यवादिनः।</div><div class="line" style="position:relative;">नान्यो ज्ञातेर्महाबाहो विनाशमभिनन्दति॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3387"><div class="verse-block verse-justify"><div class="line">अज्ञातयोऽप्यसुखदा ज्ञातयोऽपि सुखावहाः।</div><div class="line" style="position:relative;">अज्ञातिमन्तं पुरुषं परे चाभिभवन्त्युत॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3388"><div class="verse-block verse-justify"><div class="line">निकृतस्य नरैरन्यैर्ज्ञातिरेव परायणम्।</div><div class="line" style="position:relative;">नान्यो निकारं सहते ज्ञातिर्ज्ञातेः कदाचन॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3389"><div class="verse-block verse-justify"><div class="line">आत्मानमेव जानाति निकृतं बान्धवैपरि।</div><div class="line" style="position:relative;">तेषु सन्ति गुणाश्चैव नैर्गुण्यं चैव लक्ष्यते॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3390"><div class="verse-block verse-justify"><div class="line">नाज्ञातिरनुगृह्णाति नाज्ञातिर्वृद्धिमश्नुते।</div><div class="line" style="position:relative;">उभयं ज्ञातिवर्गेषु दृश्यते साध्वसाधु च॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3391"><div class="verse-block verse-justify"><div class="line">संमानयेत्पूजयेच्च वाचा नित्यं च कर्मणा।</div><div class="line" style="position:relative;">कुर्याच्च प्रियमेतेभ्यो नाप्रियं किंचिद चरेत्॥४०॥</div></div></div>
