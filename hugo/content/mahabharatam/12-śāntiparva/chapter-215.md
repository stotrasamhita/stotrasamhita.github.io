<div class="verse-block-wrapper" id="12-śāntiparva-b8866"><div class="verse-block verse-justify"><div class="line">तस्य मायापिनद्धाङ्गा ज्ञान भ्रष्टा निराशिषः।</div><div class="line" style="position:relative;">मानवा ज्ञानसंमोहात्ततः कामं प्रयान्ति वै॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8867"><div class="verse-block verse-justify"><div class="line">कामात्क्रोधमवाप्याथ लोभमोहौ च मानवाः।</div><div class="line" style="position:relative;">मानदर्पावहंकारमहंकारात्ततः क्रियाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8868"><div class="verse-block verse-justify"><div class="line">क्रियाभिः स्नेहसंबन्धः स्नेहाच्छोकमनन्तरम्।</div><div class="line" style="position:relative;">अथ दुःखसमारम्भा जराजन्मकृतक्षणाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8869"><div class="verse-block verse-justify"><div class="line">जन्मतो गर्भवासं तु शुक्रशोणितसंभवम्।</div><div class="line" style="position:relative;">पुरीषमूत्रविक्लेदं शोणितप्रभवाविलम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8870"><div class="verse-block verse-justify"><div class="line">तृष्णाभिभूतस्तैर्वद्धस्तानेवाभिपरिप्लवन्।</div><div class="line">` तथा नरकगर्तस्थस्तृष्णारज्जुभिराचितः।</div><div class="line" style="position:relative;">पुण्यपापप्रणुन्नाङ्गो जायते स यथा कृमिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8871"><div class="verse-block verse-justify"><div class="line">मशकैर्मत्कुणैर्दष्टस्तथा चित्रवधार्दितः।</div><div class="line" style="position:relative;">नानाव्याधिभिराकीर्णः कथंचिद्यौवनं गतः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8872"><div class="verse-block verse-justify"><div class="line">कूर्मोत्सृजति भूयश्च रज्जुः स्वस्वमुखेप्सया।</div><div class="line" style="position:relative;">योषितं नरकं गृह्य जन्मकर्मवशानुगः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8873"><div class="verse-block verse-justify"><div class="line">पुरक्षेत्रनिमित्तं यद्दुःखं वक्तुं न शक्यते।</div><div class="line" style="position:relative;">कस्तत्र निन्दकश्चैव नरके पच्यते भृशम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8874"><div class="verse-block verse-justify"><div class="line">वार्धक्यमनुलङ्घेत तत्र कर्मारभेत्पुनः।</div><div class="line" style="position:relative;">भगवान्संस्तुतः पश्चात्किं प्रवक्ष्यामि ते भृशम्'॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8875"><div class="verse-block verse-justify"><div class="line">संसारतन्त्रवाहिन्यस्तत्र बुद्ध्येत योषितः।</div><div class="line">प्रकृत्याः क्षेत्रभूतास्ता नराः क्षेत्रज्ञलक्षणाः।</div><div class="line" style="position:relative;">तस्मादेवाविशेषेण नरोऽतीयाद्विशेषतः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8876"><div class="verse-block verse-justify"><div class="line">कृत्या ह्येता घोररूपा मोहयन्त्यविचक्षणान्।</div><div class="line" style="position:relative;">रजस्यन्तर्हिता मूर्तिरिन्द्रियाणां सनातनी॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8877"><div class="verse-block verse-justify"><div class="line">तस्मात्तर्षात्मकाद्रागाद्बीजाज्जायन्ति जन्तवः।</div><div class="line">स्वदेहजानस्वसंज्ञान्यद्वदङ्गात्कृमींस्त्यजेत्।</div><div class="line" style="position:relative;">स्वसंज्ञानस्वकांस्तद्वत्सुतसंज्ञान्कृमींस्त्यजेत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8878"><div class="verse-block verse-justify"><div class="line">शुक्रतो रसतश्चैव देहाज्जायन्ति जन्तवः।</div><div class="line" style="position:relative;">स्वभावात्कर्मयोगाद्वा तानुपेक्षेत बुद्धिमान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8879"><div class="verse-block verse-justify"><div class="line">रजस्तमसि पर्यस्तं सत्वं च रजसि स्थितम्।</div><div class="line" style="position:relative;">ज्ञानाधिष्ठानमज्ञानं बुद्ध्यंहंकारलक्षणम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b8880"><div class="verse-block verse-justify"><div class="line">तद्बीजं देहिनामाद्दुस्तद्बीजं जीवसंज्ञितम्।</div><div class="line" style="position:relative;">कर्मणा कालयुक्तेन संसारपरिवर्तनम्॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b8882"><div class="verse-block verse-justify"><div class="line">कर्मणा बीजभूतेन चोद्यते यद्यदिन्द्रियम्।</div><div class="line" style="position:relative;">जायते तदहंकाराद्रागयुक्तेन चेतसा॥२०॥</div></div></div>
