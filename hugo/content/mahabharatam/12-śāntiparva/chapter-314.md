<div class="verse-block-wrapper" id="12-śāntiparva-b13259"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13260"><div class="verse-block verse-justify"><div class="line">मृगयां विचरन्कश्चिद्विजने जनकात्मजः।</div><div class="line" style="position:relative;">वने ददर्श विप्रेन्द्रमृषिं वंशधरं भृगोः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13261"><div class="verse-block verse-justify"><div class="line">तमासीनमुपासीनः प्रणम्य शिरसा मुनिम्।</div><div class="line" style="position:relative;">पश्चादनुमतस्तेन पप्रच्छ वसुमानिदम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13262"><div class="verse-block verse-justify"><div class="line">भगवन्किमिदं श्रेयः प्रेत्य चापीह वा भवेत्।</div><div class="line" style="position:relative;">पुरुषस्याध्रुवे देहे कामस्य वशवर्तिनः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13263"><div class="verse-block verse-justify"><div class="line">सत्कृत्य परिपृष्टः सन्सुमहात्मा महातपाः।</div><div class="line">निजगाद ततस्तस्मै श्रेयस्करमिदं वचः ॥ऋषिरुवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13264"><div class="verse-block verse-justify"><div class="line">मनसः प्रतिकूलानि प्रेत्य चेह नचेच्छसि।</div><div class="line" style="position:relative;">भूतानां प्रतिकूलेभ्यो निवर्तस्य यतेन्द्रियः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13265"><div class="verse-block verse-justify"><div class="line">धर्मः सदा हितः पुंसां धर्मश्चैवाश्रयः सताम्।</div><div class="line" style="position:relative;">धर्माल्लोकास्त्रयस्तात प्रवृत्ताः सचराचराः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13266"><div class="verse-block verse-justify"><div class="line">स्वादुकामुक कामानां वैतृष्ण्यं किं न गच्छसि।</div><div class="line" style="position:relative;">मधु पश्यसि दुर्बुद्धे प्रपातं नानुपश्यसि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13267"><div class="verse-block verse-justify"><div class="line">यथा ज्ञाने परिचयः कर्तव्यस्तत्फलार्थिना।</div><div class="line" style="position:relative;">तथा धर्मे परिचयः कर्तव्यस्तत्फलार्थिना॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13268"><div class="verse-block verse-justify"><div class="line">असता धर्मकामेन विशुद्धं कर्म दुष्करम्।</div><div class="line" style="position:relative;">सता तु धर्मकामेन सुकरं कर्म दुष्करम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13269"><div class="verse-block verse-justify"><div class="line">वने ग्राम्यसुखाचारो यथाग्राम्यस्तथैव सः।</div><div class="line" style="position:relative;">ग्रामे वनसुखाचारो यथा वनचरस्तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13270"><div class="verse-block verse-justify"><div class="line">मनोवाक्कर्मगे धर्मे कुरु श्रद्धां समाहितः।</div><div class="line" style="position:relative;">निवृत्तौ वा प्रवृत्तौ वा संप्रधार्य गुणागुणान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13271"><div class="verse-block verse-justify"><div class="line">नित्यं च बहु दातव्यं साधुभ्यश्चानसूयता।</div><div class="line" style="position:relative;">प्रार्थितं ब्राह्मणेभ्यश्च सत्कृतं देशकालयोः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13272"><div class="verse-block verse-justify"><div class="line">शुभेन विधिना लब्धमर्हाय प्रतिपादयेत्।</div><div class="line" style="position:relative;">क्रोधमुत्सृज्य दत्त्वाऽथ नानुतप्येन्न कीर्तयेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13273"><div class="verse-block verse-justify"><div class="line">अनृशंसः शुचिर्दान्तः सत्यवागार्जवे स्थितः।</div><div class="line" style="position:relative;">योनिकर्मविशुद्धश्च पात्रं स्याद्वेदविद्द्विजः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13274"><div class="verse-block verse-justify"><div class="line">संस्कृता चैकपत्नी च जात्या योनिरिहेष्यते।</div><div class="line" style="position:relative;">ऋग्यजुःसामगो विद्वान्षट््कर्मा पात्रमुच्यते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13275"><div class="verse-block verse-justify"><div class="line">स एव धर्मः सोऽधर्मस्तं तं प्रति नरं भवेत्।</div><div class="line" style="position:relative;">पात्राकर्मविशेषेण देशकालाववेक्ष्य च॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13276"><div class="verse-block verse-justify"><div class="line">लीलयाऽल्पं यथा गात्रात्प्रमृज्यात्तु रजः पुमान्।</div><div class="line" style="position:relative;">बहुयत्नेन च महत्पापनिर्हरणं तथा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13277"><div class="verse-block verse-justify"><div class="line">विरिक्तस्य यथा सम्यग्घृतं भवति भेषजम्।</div><div class="line" style="position:relative;">तथा निर्हृतदोषस्य प्रेत्य धर्मः सुखावहः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13278"><div class="verse-block verse-justify"><div class="line">मानसं सर्वभूतेषु वर्तते वै शुभाशुभम्।</div><div class="line" style="position:relative;">अशुभेभ्यः समाक्षिप्य शुभेष्वेवावधारय॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13279"><div class="verse-block verse-justify"><div class="line">सर्वं सर्वेणा सर्वत्र क्रियमाणं च पूजय।</div><div class="line" style="position:relative;">स्वधर्मे यत्र रागस्ते कामं धर्मो विधीयताम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13280"><div class="verse-block verse-justify"><div class="line">अधृतात्मन्धृतौ तिष्ठ दुर्बद्धे बुद्धिमान्भव।</div><div class="line" style="position:relative;">अप्रशान्तः प्रशाम्य त्वमप्राज्ञः प्राज्ञवच्चर॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13281"><div class="verse-block verse-justify"><div class="line">तेजसा शक्यते प्राप्तुमुपायः सहचारिणा।</div><div class="line" style="position:relative;">इह च प्रेत्य च श्रेयस्तस्य मूलं धृतिः परा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13282"><div class="verse-block verse-justify"><div class="line">राजर्षिरधृतिः स्वर्गात्पतितो हि महाभिषः।</div><div class="line" style="position:relative;">ययातिः क्षीणपुण्योपि धृत्या लोकानवाप्तवान्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13283"><div class="verse-block verse-justify"><div class="line">तपस्विनां धर्मवतां विदुषां चोपसेवनात्।</div><div class="line">प्राप्स्यसे विपुलां बुद्धिं तथा श्रेयोऽभिपत्स्यसे ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13284"><div class="verse-block verse-justify"><div class="line">स तु स्वभावसंपन्नस्तच्छ्रुत्वा मुनिभाषितम्।</div><div class="line" style="position:relative;">विनिवर्त्य मनः कामाद्धर्मे बुद्धिं चकार ह॥२६॥</div></div></div></div>
