<div class="verse-block-wrapper" id="12-śāntiparva-b12184"><div class="verse-block verse-justify"><div class="line">सर्वज्ञानं परं बुद्ध्या यन्मां त्वमनुपृच्छसि।</div><div class="line" style="position:relative;">तद्व्याख्यास्यामि ते तात तस्य व्याख्यामिमां शृणु॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12185"><div class="verse-block verse-justify"><div class="line">पृथिवी वायुराकाशमापो ज्योतिश्च पञ्चमम्।</div><div class="line" style="position:relative;">महाभूतानि भूतानां सर्वेषां प्रभवाप्ययौ॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12186"><div class="verse-block verse-justify"><div class="line">स तेषां गुणसंघातः शरीरं भरतर्षभ।</div><div class="line" style="position:relative;">सततं हि प्रलीयन्ते गुणास्ते प्रभवन्ति च॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12187"><div class="verse-block verse-justify"><div class="line">ततः सृष्टानि भूतानि तानि यान्ति पुनः पुनः।</div><div class="line" style="position:relative;">महाभूतानि भूतेभ्य ऊर्मयः सागरे यथा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12188"><div class="verse-block verse-justify"><div class="line">प्रसारयित्वेहाङ्गानि कूर्मः संहरते यथा।</div><div class="line" style="position:relative;">तद्वद्भूतानि भूतानामल्पीयांसि स्थवीयसाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12189"><div class="verse-block verse-justify"><div class="line">आकाशात्खलु यो घोषः संघातस्तु महीगुणः।</div><div class="line" style="position:relative;">वायोः प्राणो रसस्त्वद्भ्यो रूपं तेजस उच्यते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12190"><div class="verse-block verse-justify"><div class="line">इत्येतन्मयमेवैतत्सर्वं स्थावरजङ्गमम्।</div><div class="line" style="position:relative;">प्रलये च तमभ्येति तस्मादुद्दिश्यते पुनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12191"><div class="verse-block verse-justify"><div class="line">महाभूतानि पञ्चैव सर्वभूतेषु भूतकृत्।</div><div class="line" style="position:relative;">विषयान्कल्पयामास यस्मिन्यदनुपश्यति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12192"><div class="verse-block verse-justify"><div class="line">शब्दश्रोत्रे तथा खानि त्रयमाकाशयोनिजम्।</div><div class="line" style="position:relative;">रसः स्नेहश्च जिह्वा च अपामेते गुणाः स्मृताः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12193"><div class="verse-block verse-justify"><div class="line">रूपं चक्षुर्विपाकश्च त्रिविधं ज्योतिरूच्यते।</div><div class="line" style="position:relative;">घ्रेयं घ्राणं शरीरं च एते भूमिगुणाः स्मृताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12194"><div class="verse-block verse-justify"><div class="line">प्राणः स्पर्शश्च चेष्टा च वायोरेते गुणाः स्मृताः।</div><div class="line" style="position:relative;">इति सर्वगुणा राजन्व्याख्याताः पाञ्चभौतिकाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12195"><div class="verse-block verse-justify"><div class="line">सत्त्वं रजस्तमः कालः कर्म बुद्धिश्च भारत।</div><div class="line" style="position:relative;">मनः षष्ठानि चैतेषु ईश्वरः समकल्पयत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12196"><div class="verse-block verse-justify"><div class="line">यदूर्ध्वपादतलयोरवाड्यूर्ध्नश्च पश्यसि।</div><div class="line" style="position:relative;">एतस्मिन्नेव कृत्स्नेयं वर्तते बुद्धिरन्तरे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12197"><div class="verse-block verse-justify"><div class="line">इन्द्रियाणि नरे पञ्च षष्ठं तु मन उच्यते।</div><div class="line" style="position:relative;">सप्तमीं बुद्धिमेवाहुः क्षेत्रज्ञः पुनरष्टमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12198"><div class="verse-block verse-justify"><div class="line">इन्द्रियाणि च कर्ता च विचेतव्यानि भागशः।</div><div class="line" style="position:relative;">तमः सत्वं रजस्तैव तेऽपि भावास्तदाश्रयाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12199"><div class="verse-block verse-justify"><div class="line">चक्षुरालोचनायैव संशयं कुरुते मनः।</div><div class="line" style="position:relative;">बुद्धिरध्यवसानाय साक्षी क्षेत्रज्ञ उच्यते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12200"><div class="verse-block verse-justify"><div class="line">तमः सत्वं रजश्चेति कालः कर्म च भारत।</div><div class="line">गुणैर्नेनीयते बुद्धिर्बुद्धिरेवेन्द्रियाणि च।</div><div class="line" style="position:relative;">मनः षष्ठानि सर्वाणि बुद्ध्यभावे कुतो गुणाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12201"><div class="verse-block verse-justify"><div class="line">येन पश्यति तच्चक्षुः शृण्वती श्रोत्रमुच्यते।</div><div class="line" style="position:relative;">जिघ्रती भवति घ्राणं रसती रसना रसान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12202"><div class="verse-block verse-justify"><div class="line">स्पर्शनं स्पर्शती स्पर्शान्बुद्धिर्विक्रियतेऽसकृत्।</div><div class="line" style="position:relative;">यदा प्रार्थयते किंचित्तदा भवति सा मनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12203"><div class="verse-block verse-justify"><div class="line">अधिष्ठानानि बुद्ध्या हि पृथगेतानि पञ्चधा।</div><div class="line" style="position:relative;">इन्द्रियाणीति तान्याहुस्तेषु दुष्टेषु दुष्यति॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12204"><div class="verse-block verse-justify"><div class="line">पुरुषे तिष्ठती बुद्धिस्त्रिषु भावेषु वर्तते।</div><div class="line" style="position:relative;">कदाचिल्लभते प्रीतिं कदाचिदपि शोचति॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12205"><div class="verse-block verse-justify"><div class="line">न सुखेन न दुःखेन कदाचिदपि वर्तते।</div><div class="line" style="position:relative;">सेयं भावात्मिका भावांस्त्रीनेतान्परिवर्तते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12206"><div class="verse-block verse-justify"><div class="line">सरितां सागरो भर्ता यथा वेलामिवोर्मिमान्।</div><div class="line" style="position:relative;">इति भावगता बुद्धिर्भावे मनसि वर्तते॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12207"><div class="verse-block verse-justify"><div class="line">प्रवर्तमानं तु रजस्तद्भावेनानुर्तते।</div><div class="line" style="position:relative;">प्रहर्षः प्रीतिरानन्दः सुखं संशान्तचित्तता॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12208"><div class="verse-block verse-justify"><div class="line">कथंचिदुपपद्यन्ते पुरुषे सात्विका गुणाः।</div><div class="line" style="position:relative;">पिरदाहस्तथा शोकः संतापोऽपूर्तिरक्षमा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12209"><div class="verse-block verse-justify"><div class="line">लिङ्गानि रजसस्तानि दृश्यन्ते हेत्वहेतुभिः।</div><div class="line" style="position:relative;">अविद्या रागमोहौ च प्रमादः स्तब्धता भयम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12210"><div class="verse-block verse-justify"><div class="line">असमृद्धिस्तथा दैन्यं प्रमोहः स्वप्नतन्द्रिता।</div><div class="line" style="position:relative;">कथंचिदुपवर्तन्ते विविधास्तामसा गुणाः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12211"><div class="verse-block verse-justify"><div class="line">तत्र यत्प्रीतिसंयुक्तं काये मनसि वा भवेत्।</div><div class="line" style="position:relative;">वर्तते सात्विको भाव इत्युपेक्षेत तत्तथा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12212"><div class="verse-block verse-justify"><div class="line">अथ यद्दुःखसंयुक्तमप्रीतिकरमात्मनः।</div><div class="line" style="position:relative;">प्रवृत्तं रज इत्येव तदसंरभ्य चिन्तयेत्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12213"><div class="verse-block verse-justify"><div class="line">अथ यन्मोहसंयुक्तं काये मनसि वा भवेत्।</div><div class="line" style="position:relative;">अप्रतर्क्यमविज्ञेयं तमस्तदुपधारयेत्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12214"><div class="verse-block verse-justify"><div class="line">इति बुद्धिगतीः सर्वा व्याख्याता यावतीरिह।</div><div class="line" style="position:relative;">एतद्बुद्ध्वा भवेद्बुद्धः किमन्यद्बुद्धलक्षणम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12215"><div class="verse-block verse-justify"><div class="line">सत्वक्षेत्रज्ञयोरेतदन्तरं विद्धि सूक्ष्मयोः।</div><div class="line" style="position:relative;">सृजतेऽत्र गुणानेक एको न सृजते गुणान्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12216"><div class="verse-block verse-justify"><div class="line">पृथग्भूतौ प्रकृत्या तु संप्रयुक्तौ च सर्वदा।</div><div class="line" style="position:relative;">यथा मत्स्योऽद्भिरन्यः स्यात्संप्रयुक्तो भवेत्तथा॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12217"><div class="verse-block verse-justify"><div class="line">न गुणा विदुरात्मानं स गुणान्वेद सर्वतः।</div><div class="line" style="position:relative;">परिद्रष्टा गुणानां तु संस्रष्टा मन्यते यथा॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12218"><div class="verse-block verse-justify"><div class="line">आश्रयो नास्ति सत्वस्य गुणसर्गेण चेतना।</div><div class="line" style="position:relative;">सत्वमस्य सृजन्त्यन्ये गुणान्वेद कदाचन॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12219"><div class="verse-block verse-justify"><div class="line">सृजते हि गुणान्सत्वं क्षेत्रज्ञः परिपश्यति।</div><div class="line" style="position:relative;">संप्रयोगस्तयोरेष सत्वक्षेत्रज्ञयोर्ध्रुवः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12220"><div class="verse-block verse-justify"><div class="line">इन्द्रियैस्तु प्रदीपार्थं क्रियते बुद्धिरन्तरा।</div><div class="line" style="position:relative;">निश्चक्षुर्भिरजानद्भिरिन्द्रियाणि प्रदीपवत्॥३९॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b12225"><div class="verse-block verse-justify"><div class="line">ताम्येयुः प्रच्युताः पृथ्वीं मोहपूर्णां नदीं नराः।</div><div class="line" style="position:relative;">यथा गाधमविद्वांसो बुद्धियोगमयं तथा॥४४॥</div></div></div>
