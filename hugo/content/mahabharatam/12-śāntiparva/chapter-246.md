<div class="verse-block-wrapper" id="12-śāntiparva-b10411"><div class="verse-block verse-justify"><div class="line">कामं क्रोधं त्त लोभं च भयं स्वप्नं च पञ्चमम्।</div><div class="line" style="position:relative;">क्रोधं शमेन जयति कामं संकल्पवर्जनात्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10412"><div class="verse-block verse-justify"><div class="line">सत्त्वसंसेवनाद्धीरो निद्रामुच्छेत्तुमर्हति।</div><div class="line" style="position:relative;">धृत्या शिश्नोदरं रक्षेत्पाणिपादं च चक्षुषा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10413"><div class="verse-block verse-justify"><div class="line">चक्षुःश्रोत्रे च मनसा मनो वाचं च कर्मणा।</div><div class="line" style="position:relative;">अप्रमादाद्भयं जह्याल्लोभं प्राज्ञोपसेवनात्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10414"><div class="verse-block verse-justify"><div class="line">एवमेतान्योगदोषाञ्चयेन्नित्यमतन्द्रितः।</div><div class="line" style="position:relative;">अग्नींश्च ब्राह्मणांश्चार्चेद्देवताः प्रणमेत च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10415"><div class="verse-block verse-justify"><div class="line">वर्जयेदुशतीं वाचं हिंसायुक्तां मनोनुदाम्।</div><div class="line" style="position:relative;">ब्रह्म तेजोमयं शुक्रं यस्य सर्वमिदं ततम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10416"><div class="verse-block verse-justify"><div class="line">एतस्य सूत्रभूतस्य द्वयं स्थावरजङ्गमम्।</div><div class="line" style="position:relative;">ध्यानमध्ययनं दानं सत्यं ह्रीरार्जवं क्षमा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10417"><div class="verse-block verse-justify"><div class="line">शोचमाहारसंशुद्धिरिन्द्रियाणां च निग्रहः।</div><div class="line" style="position:relative;">एतैर्विवर्धते तेजः पाप्मानं चापकर्षति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10418"><div class="verse-block verse-justify"><div class="line">सिद्ध्यन्ति चास्य सर्वार्था विज्ञानं च प्रवर्धते।</div><div class="line" style="position:relative;">समः सर्वेषु भूतेषु लब्धालब्धेन वर्तयेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10419"><div class="verse-block verse-justify"><div class="line">धूतपाप्मा तु तेजस्वी लघ्वाहारो जितेन्द्रियः।</div><div class="line" style="position:relative;">कामक्रोधौ वशे कृत्वा निनीषेद्ब्रह्मणः पदम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10420"><div class="verse-block verse-justify"><div class="line">मनसश्चेन्द्रियाणां च कृत्वैकाग्र्यं समाहितः।</div><div class="line" style="position:relative;">पूर्वरात्रेऽपरात्रे च धारयेन्मन आत्मनि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10421"><div class="verse-block verse-justify"><div class="line">जन्तोः पञ्चेन्द्रियस्यास्य यदेकं छिद्रमिन्द्रियम्।</div><div class="line" style="position:relative;">ततोऽस्य स्रवते प्रज्ञा दृतेः पादादिवोदकम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10422"><div class="verse-block verse-justify"><div class="line">मनस्तु पूर्वमादद्यात्कुमीनमिव मत्स्यहा।</div><div class="line" style="position:relative;">ततः श्रोत्रं ततश्चक्षुर्जिह्वा घ्राणं च योगवित्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10423"><div class="verse-block verse-justify"><div class="line">तत एतानि संयम्य मनसि स्थापयेद्यतिः।</div><div class="line" style="position:relative;">तथैवापो ह्यसंकल्पान्मनो ह्यात्मनि धारयेत्॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b10425"><div class="verse-block verse-justify"><div class="line">प्रसीदन्ति च संस्थाय तदा ब्रह्म प्रकाशते।</div><div class="line" style="position:relative;">विधूम इव सप्तार्चिरादित्य इव दीप्तिमान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b10426"><div class="verse-block verse-justify"><div class="line">वैद्युतोऽग्निरिवाकाशे दृश्यतेऽऽत्मा तथाऽऽत्मनि।</div><div class="line" style="position:relative;">सर्वस्तत्र स सर्वत्र व्यापकत्वाच्च दृश्यते॥२१॥</div></div></div>
