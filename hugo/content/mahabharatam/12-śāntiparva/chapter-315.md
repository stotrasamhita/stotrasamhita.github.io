<div class="verse-block-wrapper" id="12-śāntiparva-b13287"><div class="verse-block verse-justify"><div class="line">धर्माधर्मविमुक्तं यद्विमुक्तं सर्वसंशयात्।</div><div class="line" style="position:relative;">जन्ममृत्युविमुक्तं च विमुक्तं पुण्यपापयोः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13288"><div class="verse-block verse-justify"><div class="line">यच्छिवं नित्यमभयं नित्यमक्षरमव्ययम्।</div><div class="line">शुचि नित्यमनायासं तद्भवान्वक्तुमर्हति ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13289"><div class="verse-block verse-justify"><div class="line">अत्र ते वर्तयिष्यामि इतिहासं पुरातनम्।</div><div class="line" style="position:relative;">याज्ञवल्क्यस्य संवादं जनकस्य च भारत॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13290"><div class="verse-block verse-justify"><div class="line">याज्ञवल्क्यमृषिश्रेष्ठं दैवरातिर्महायशाः।</div><div class="line">पप्रच्छ जनको राजा प्रश्नं प्रश्नविदांवरः ॥जनक उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13291"><div class="verse-block verse-justify"><div class="line">कतीन्द्रियाणि विप्रर्षे कति प्रकृतयः स्मृताः।</div><div class="line" style="position:relative;">किमव्यक्तं परं ब्रह्म तस्माच्च परतस्तु किम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13292"><div class="verse-block verse-justify"><div class="line">प्रभवं चाप्ययं चैव कालसङ्ख्यां तथैव च।</div><div class="line" style="position:relative;">वक्तुमर्हसि विप्रेन्द्र त्वदनुग्रहकाङ्क्षिणः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13293"><div class="verse-block verse-justify"><div class="line">अज्ञानात्परिपृच्छामि त्वं हि ज्ञानमयो निधिः।</div><div class="line">तदहं श्रोतुमिच्छामि सर्वमेतदसंशयम् ॥याज्ञवल्क्य उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13294"><div class="verse-block verse-justify"><div class="line">श्रूयतामवनीपाल यदेतदनुपृच्छसि।</div><div class="line" style="position:relative;">योगानां परमं ज्ञानं साङ्ख्यानां च विशेषतः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13295"><div class="verse-block verse-justify"><div class="line">त तवाविदितं किंचिन्मां तु जिज्ञासते भवान्।</div><div class="line" style="position:relative;">पृष्टेन चापि वक्तव्यमेष धर्मः सनातनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13296"><div class="verse-block verse-justify"><div class="line">अष्टौ प्रकृतयः प्रोक्ता विकाराश्चापि षोडश।</div><div class="line" style="position:relative;">आसां तु सप्त व्यक्तानि प्राहुरध्यात्मचिन्तकाः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13297"><div class="verse-block verse-justify"><div class="line">अव्यक्तं च महांश्चैव तथाऽहंकार एव च।</div><div class="line" style="position:relative;">पृथिवी वायुराकाशमापो ज्योतिश्च पञ्चमम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13298"><div class="verse-block verse-justify"><div class="line">एताः प्रकृतयस्त्वष्टौ विकारानपि मे शृणु।</div><div class="line" style="position:relative;">श्रोत्रं त्वक्चैव चक्षुश्च जिह्वा घ्राणं च पञ्चमम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13299"><div class="verse-block verse-justify"><div class="line">शब्दः स्पर्शश्च रूपं च रसो गन्धस्तथैव च।</div><div class="line" style="position:relative;">वाक्च हस्तौ च पादौ च पायुर्मेढ्रं तथैव च॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13300"><div class="verse-block verse-justify"><div class="line">एते विशेषा राजेन्द्रा महाभूतेषु पञ्चसु।</div><div class="line" style="position:relative;">बुद्धीन्द्रियाण्यथैतानि सविशेषाणि मैथिल॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13301"><div class="verse-block verse-justify"><div class="line">मनः षोडशकं प्राहुरध्यात्मगतिचिन्तकाः।</div><div class="line" style="position:relative;">त्वं चैवान्ये च विद्वांसस्तत्त्वबुद्धिविशारदाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13302"><div class="verse-block verse-justify"><div class="line">अव्यक्ताच्च महानात्मा समुत्पद्यति पार्थिव।</div><div class="line" style="position:relative;">प्रथमं सर्गमित्येतदाहुः प्राधानिकं बुधाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13303"><div class="verse-block verse-justify"><div class="line">महतश्चाप्यहंकार उत्पद्यति नराधिप।</div><div class="line" style="position:relative;">द्वितीयं सर्गमित्याहुरेतद्बुद्ध्यात्मकं स्मृतम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13304"><div class="verse-block verse-justify"><div class="line">अहंकाराच्च संभूतं मनो भूतगुणात्मकम्।</div><div class="line" style="position:relative;">तृतीयः सर्ग इत्येष आहंकारिक उच्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13305"><div class="verse-block verse-justify"><div class="line">मनसस्तु समुद्भूता महाभूता नराधिप।</div><div class="line" style="position:relative;">चतुर्थं सर्गमित्येतन्मानसं चिन्तनात्मकम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b13306"><div class="verse-block verse-justify"><div class="line">शब्दः स्पर्शश्च रूपं च रसो गन्धस्तथैव च।</div><div class="line" style="position:relative;">पञ्चमं सर्गमित्याहुर्भौतिकं भूतचिन्तकाः॥२१॥</div></div></div>
