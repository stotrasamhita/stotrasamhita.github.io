<div class="verse-block-wrapper" id="12-śāntiparva-b6021"><div class="verse-block verse-justify"><div class="line">नैतच्छ्रुत्वागमादेव तव धर्मानुशासनम्।</div><div class="line" style="position:relative;">प्रज्ञासमभिहारोऽयं कविभिः संभृतं मधु॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6022"><div class="verse-block verse-justify"><div class="line">बाह्याः प्रतिविधातव्याः प्रज्ञा राज्ञा ततस्ततः।</div><div class="line" style="position:relative;">बहुशाखेन धर्मेण यत्रैषा संप्रसिध्यते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6023"><div class="verse-block verse-justify"><div class="line">बुद्धिं संजनयेद्राज्ञां धर्ममाचरतां सदा।</div><div class="line" style="position:relative;">जयो भवति कौरव्य सदा तद्वृद्धिरेव च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6024"><div class="verse-block verse-justify"><div class="line">बुद्धिश्रेष्ठा हि राजानो जयन्ति विजयैषिणः।</div><div class="line" style="position:relative;">धर्मः प्रतिविधातव्यो बुद्ध्या राज्ञा ततस्ततः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6025"><div class="verse-block verse-justify"><div class="line">नैकशाखेन धर्मेण राज्ञो धर्मो विधीयते।</div><div class="line" style="position:relative;">दुर्बलस्य कुतः प्रज्ञा पुरस्तादनुदाहृता॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6026"><div class="verse-block verse-justify"><div class="line">अद्वैधज्ञः प्रतिद्वैधे संशयं प्राप्नुमर्हति।</div><div class="line" style="position:relative;">बुद्धिद्वैधं विधातव्यं पुरस्तादेव भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6027"><div class="verse-block verse-justify"><div class="line">पार्श्वतः कारणं राज्ञो विषूच्यस्त्वापगा इव।</div><div class="line" style="position:relative;">जनास्तूच्चरितं धर्मं विजान्त्यन्यथाऽन्यथा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6028"><div class="verse-block verse-justify"><div class="line">सम्यग्विज्ञानिनः केचिन्मिथ्याविज्ञानिनः परे।</div><div class="line" style="position:relative;">तद्वै यथायथं बुद्ध्वा ज्ञानमाददते सताम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6029"><div class="verse-block verse-justify"><div class="line">परिमुष्णन्ति शास्त्राणि धर्मस्य परिपन्थिनः।</div><div class="line" style="position:relative;">वैषम्यमर्थविद्यानां निरर्थाः ख्यापयन्ति ते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6030"><div class="verse-block verse-justify"><div class="line">आजिजीविषवो विद्यां यशः कामौ समन्तनः।</div><div class="line" style="position:relative;">ते सर्वे नृप पापिष्ठा धर्मस्य परिपन्थिनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6031"><div class="verse-block verse-justify"><div class="line">अपक्वमतयो मन्दा न जानन्ति यथातथम्।</div><div class="line" style="position:relative;">तथा ह्यशास्त्रकुशलाः सर्वत्रायुक्तिनिष्ठिताः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6032"><div class="verse-block verse-justify"><div class="line">परिमुष्णन्ति शास्त्राणि शास्त्रदोषानुदर्शिनः।</div><div class="line" style="position:relative;">विज्ञानमथ विद्यानां न सम्यगिति मे मतिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6033"><div class="verse-block verse-justify"><div class="line">निन्दया परविद्यानां स्वविद्यां ख्यापयन्ति च।</div><div class="line" style="position:relative;">वागास्तिक्यानुनीताश्च दुग्धविद्याफला इव॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6034"><div class="verse-block verse-justify"><div class="line">तान्विद्यावणिजो विद्धि राक्षसानिव भारत।</div><div class="line" style="position:relative;">व्याजेन कृत्स्नो विहितो धर्मस्ते परिहास्यते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6035"><div class="verse-block verse-justify"><div class="line">न धर्मवचनं वाचा नैव बुद्ध्येति नः श्रुतम्।</div><div class="line" style="position:relative;">इति बार्हस्पत्यविज्ञानं प्रोवाच मघवा स्वयम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6036"><div class="verse-block verse-justify"><div class="line">न त्यव वचनं किंचिदनिमित्तादिहोच्यते।</div><div class="line" style="position:relative;">स्वविनीतेन शास्त्रेण ह्यविद्यः स्यादथापरः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6037"><div class="verse-block verse-justify"><div class="line">लोकयात्रामिहैके तु धर्मं प्राहुर्मनीषिणः।</div><div class="line" style="position:relative;">समुद्दिष्टं सतां धर्मं स्वयमूहेत् पण्डितः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6038"><div class="verse-block verse-justify"><div class="line">अमर्षाच्छास्त्रसंमोहादनिमित्तादिहोच्यते।</div><div class="line" style="position:relative;">शास्त्रं प्राज्ञस्य वदतः समूहे यात्यदर्शनम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6039"><div class="verse-block verse-justify"><div class="line">आगमागतया बुद्ध्या वचनेन प्रशस्यते।</div><div class="line" style="position:relative;">अज्ञानाज्ज्ञानहेतुत्वाद्वचनं साधु मन्यते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6040"><div class="verse-block verse-justify"><div class="line">अनुपागतमेवेदं शास्त्रमेवमपार्थकम्।</div><div class="line" style="position:relative;">दैतेयानुशना प्राह संशयच्छेदनं पुरा॥२३॥</div></div></div>
