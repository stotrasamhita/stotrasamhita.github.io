<div class="verse-block-wrapper" id="12-śāntiparva-b3436"><div class="verse-block verse-justify"><div class="line">स काकं पञ्जरे बद्ध्वा विषयं क्षेमदर्शिनः।</div><div class="line" style="position:relative;">सर्वं पर्यचरद्युक्तः प्रवृत्त्यर्थी पुनः पुनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3437"><div class="verse-block verse-justify"><div class="line">अधीये वायसीं विद्यां शंसन्ति मम वायसाः।</div><div class="line" style="position:relative;">अनागतमतीतं च यच्च संप्रति वर्तते॥९॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b3439"><div class="verse-block verse-justify"><div class="line">स बुद्ध्वा तस्य राष्ट्रस्य व्यवसायं हि सर्वशः।</div><div class="line" style="position:relative;">राजयुक्तापचारांश्च सर्वान्बुद्ध्वा ततस्ततः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3440"><div class="verse-block verse-justify"><div class="line">ततः स काकमादाय राजानं द्रष्टुमागमत्।</div><div class="line" style="position:relative;">सर्वज्ञोऽस्मीति वचनं ब्रुवाणः संशितव्रतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3441"><div class="verse-block verse-justify"><div class="line">स स्म कौसल्यमागम्य राजामात्यमलंकृतम्।</div><div class="line" style="position:relative;">प्राह काकस्य वचनादमुत्रेदं त्वया कृतम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3442"><div class="verse-block verse-justify"><div class="line">असौ चासौ च जानीते राजकेशस्त्वया हृतः।</div><div class="line" style="position:relative;">एवमाख्याति काकोऽयं तच्छीघ्रमनुगम्यताम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3443"><div class="verse-block verse-justify"><div class="line">तथाऽन्यानपि स प्राह राजकोशहरांस्तदा।</div><div class="line" style="position:relative;">न चास्य वचनं किंचिदनृतं श्रूयते क्वचित्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3444"><div class="verse-block verse-justify"><div class="line">तेन विप्रकृताः सर्वे राजयुक्ताः कुरूद्वह।</div><div class="line" style="position:relative;">तमतिक्रम्य सुप्तं तु निशि काकमपोथयन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3445"><div class="verse-block verse-justify"><div class="line">वायसं तु विनिर्भिन्नं दृष्ट्वा वाणेन पञ्जरे।</div><div class="line" style="position:relative;">पूर्वाह्णे ब्राह्मणो वाक्यं क्षेमदर्शिनमब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3446"><div class="verse-block verse-justify"><div class="line">राजंस्त्वामभयं याचे प्रभुं प्राणधनेश्वरम्।</div><div class="line" style="position:relative;">अनुज्ञातस्त्वया ब्रूयां वचनं भवतो हितम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3447"><div class="verse-block verse-justify"><div class="line">मित्रार्थमभिसंतप्तो भक्त्या सर्वात्मनाऽऽगतः।</div><div class="line" style="position:relative;">ह्रियन्ते हि महार्थाश्च पुरुषे विक्रमत्यपि॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3448"><div class="verse-block verse-justify"><div class="line">संबुबोधयिषुर्मित्रं सदश्वमिव सारथिः।</div><div class="line" style="position:relative;">अतिमन्युप्रसक्तो हि प्रसह्य हितकारणात्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3449"><div class="verse-block verse-justify"><div class="line">तथाविधस्य सुहृदा क्षन्तव्यं संविजानता।</div><div class="line" style="position:relative;">ऐश्वर्यमिच्छता नित्यं पुरुषेण बुभूषता॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3450"><div class="verse-block verse-justify"><div class="line">तं राजा प्रत्युवाचेदं यत्किंचिन्मां भवान्वदेत्।</div><div class="line" style="position:relative;">कस्मादहं न क्षमेयमाकाङ्क्षन्नात्मनो हितम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3451"><div class="verse-block verse-justify"><div class="line">ब्राह्मण प्रतिजाने ते प्रब्रूहि यदिहेच्छसि।</div><div class="line">करिष्यामि हि ते वाक्यं यन्मां विप्र प्रवक्ष्यसि ॥मुनिरुवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3452"><div class="verse-block verse-justify"><div class="line">विद्वान्नयानपायांश्च भयाख्यातॄन्भयानि च।</div><div class="line" style="position:relative;">भक्त्या वृत्तिं समाख्यातुं भवतोऽन्तिकमागतः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3453"><div class="verse-block verse-justify"><div class="line">प्रागेवोक्तं तु दोषोऽयमाचार्यैर्नृपसेवनम्।</div><div class="line" style="position:relative;">अगतेः कुगतिर्ह्येषा या राज्ञा सहजीविका॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3454"><div class="verse-block verse-justify"><div class="line">आशीविषैश्च तस्याहुः संगमं यस्य राजभिः।</div><div class="line" style="position:relative;">बहुमित्रांश्च राजानो बह्वभित्रास्तथैव च॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3455"><div class="verse-block verse-justify"><div class="line">तेभ्यः सर्वेभ्य एवाहुर्भयं राजोपजीविनाम्।</div><div class="line" style="position:relative;">तथाऽस्य राजतो राजन्मुहुर्तादागतं भयम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3456"><div class="verse-block verse-justify"><div class="line">नैकान्तेनाप्रमादो हि शक्यः कर्तुं महीपतौ।</div><div class="line" style="position:relative;">न तु प्रमादः कर्तव्यः कथंचिद्भूतिमिच्छता॥२८॥</div></div></div>
