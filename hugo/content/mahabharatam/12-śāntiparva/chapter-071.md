<div class="verse-block-wrapper" id="12-śāntiparva-b3006"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3007"><div class="verse-block verse-justify"><div class="line">कथं राजा प्रजा रक्षन्नाधिबन्धेन युज्यते।</div><div class="line">धर्मे च नापराध्नोति तन्मे ब्रूहि पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3008"><div class="verse-block verse-justify"><div class="line">समासेनैव ते राजन्धर्मान्वक्ष्यामि शाश्वतान्।</div><div class="line" style="position:relative;">विस्तरेणैव धर्माणां न जात्वन्तमवाप्नुयात्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3009"><div class="verse-block verse-justify"><div class="line">धर्मनिष्ठाञ्श्रुतवतो देवव्रतसमाहितान्।</div><div class="line" style="position:relative;">अर्चितान्वासयेथास्त्वं गृहे गुणवतो द्विजान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3010"><div class="verse-block verse-justify"><div class="line">प्रत्युत्थायोपसंगृह्य चरणावभिवाद्य च।</div><div class="line" style="position:relative;">अथ सर्वाणि कुर्वीथाः कार्याणि सपुरोहितः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3011"><div class="verse-block verse-justify"><div class="line">धर्मकार्याणि निर्वर्त्य मङ्गलानि प्रयुज्य च।</div><div class="line" style="position:relative;">ब्राह्मणान्वाचयेथास्त्वमर्थसिद्धिजयाशिषः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3012"><div class="verse-block verse-justify"><div class="line">आर्जवेन च संपन्नो धृत्या बुद्ध्या च भारत।</div><div class="line" style="position:relative;">धर्मार्थौ प्रतिगृह्णीयात्कामक्रोधौ च वर्जयेत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3013"><div class="verse-block verse-justify"><div class="line">कामक्रोधौ पुरस्कृत्य योऽर्थं राजाऽनुतिष्ठति।</div><div class="line" style="position:relative;">न स धर्मं न चाप्यर्थं प्रतिगृह्णाति बालिशः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3014"><div class="verse-block verse-justify"><div class="line">मा स्म लुब्धांश्च मूर्खांश्च कामार्थे च प्रयूयुजः।</div><div class="line" style="position:relative;">अलुब्धान्बुद्धिसंपन्नान्सर्वकर्मसु योजयेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3015"><div class="verse-block verse-justify"><div class="line">मूर्खो ह्यधिकृतोऽर्थेषु कार्याणामविशारदः।</div><div class="line" style="position:relative;">प्रजाः क्लिश्नात्ययोगेन कामक्रोधसमन्वितः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3016"><div class="verse-block verse-justify"><div class="line">बलिषष्ठेन शुल्केन दण्डेनाथापराधिनाम्।</div><div class="line" style="position:relative;">शास्त्रानीतेन लिप्सेथा वेतनेन धनागमम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3017"><div class="verse-block verse-justify"><div class="line">दापयित्वा करं धर्म्यं राष्ट्रं नीत्या यथाविधि।</div><div class="line" style="position:relative;">तथैतं कल्पयेद्राजा योगक्षेममतन्द्रितः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3018"><div class="verse-block verse-justify"><div class="line">गोपायितारं दातारं धर्मनित्यमतन्द्रितम्।</div><div class="line" style="position:relative;">अकामद्वेषसंयुक्तमनुरज्यन्ति मानवाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3019"><div class="verse-block verse-justify"><div class="line">तस्माद्धर्मेण लाभेन लिप्सेथास्त्वं धनागमम्।</div><div class="line" style="position:relative;">धर्मार्थावध्रुवौ तस्य यो न शास्त्रपरो भवेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3020"><div class="verse-block verse-justify"><div class="line">अपशास्त्रधनो राजा संचयं नाधिगच्छति।</div><div class="line" style="position:relative;">अस्थाने चास्य तद्वित्तं सर्वमेव विनश्यति॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3021"><div class="verse-block verse-justify"><div class="line">अर्थमूलोऽपि हिंसां च कुरुते स्वयमात्मनः।</div><div class="line" style="position:relative;">करैरशास्त्रदृष्टैर्हि मोहात्संपीडयन्प्रजाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3022"><div class="verse-block verse-justify"><div class="line">ऊधश्छिन्द्यात्तु यो धेन्वाः क्षीरार्थी न लभेत्पयः।</div><div class="line" style="position:relative;">एवं राष्ट्रमयोगेन पीडितं न विवर्धते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3023"><div class="verse-block verse-justify"><div class="line">`यवसोदकमादाय सान्त्वेन विनयेन च।</div><div class="line">'यो हि दोग्ध्रीमुपास्ते च स नित्यं विन्दते पयः।</div><div class="line" style="position:relative;">एवं राष्ट्रमुपायेन भुञ्जानो लभते फलम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3024"><div class="verse-block verse-justify"><div class="line">अथ राष्ट्रमुपायेन भुज्यमानं सुरक्षितम्।</div><div class="line" style="position:relative;">जनयत्यतुलां नित्यं कोशवृद्धिं युधिष्ठिर॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3025"><div class="verse-block verse-justify"><div class="line">दोग्ध्री धान्यं हिरण्यं च मही राजा सुरक्षिता।</div><div class="line" style="position:relative;">नित्यं स्वेभ्यः परेभ्यश्च तृप्ता माता यथा पयः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3026"><div class="verse-block verse-justify"><div class="line">मालाकारोपमो राजन्भव माऽऽङ्गारिकोपमः।</div><div class="line" style="position:relative;">तथायुक्तश्चिरं राज्यं भोक्तुं शक्ष्यसि पालयन्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3027"><div class="verse-block verse-justify"><div class="line">परचक्राभियानेन यदि ते स्याद्धनक्षयः।</div><div class="line" style="position:relative;">अथ साम्नैव लिप्सेथा धनमब्राह्मणेषु यत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3028"><div class="verse-block verse-justify"><div class="line">मा स्म ते ब्राह्मणं दृष्ट्वा धनस्थं प्रचलेन्मनः।</div><div class="line" style="position:relative;">अन्त्यायामप्यवस्थायां किमु स्फीतस्य भारत॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3029"><div class="verse-block verse-justify"><div class="line">धनानि तेभ्यो दद्यास्त्वं यथाशक्ति यथार्हतः।</div><div class="line" style="position:relative;">सान्त्वयन्परिरक्षंश्च स्वर्गमाप्स्यसि दुर्जयम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3030"><div class="verse-block verse-justify"><div class="line">एवं धर्म्येण वृत्तेन प्रजास्त्वं परिपालय।</div><div class="line" style="position:relative;">स्वर्ग्यं पुण्यं यशो नित्यं प्राप्स्यसे कुरुनन्दन॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3031"><div class="verse-block verse-justify"><div class="line">धर्मेण व्यवहारेण प्रजाः पालय पाण्डव।</div><div class="line" style="position:relative;">युधिष्ठिर यथायुक्तो नाधिबन्धेन योक्ष्यसे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3032"><div class="verse-block verse-justify"><div class="line">एष एव परो धर्मो यद्राजा रक्षति प्रजाः।</div><div class="line" style="position:relative;">भूतानां हि यदा धर्मो रक्षणं परमा दया॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3033"><div class="verse-block verse-justify"><div class="line">तस्मादेवं परं धर्मं मन्यन्ते धर्मकोविदाः।</div><div class="line" style="position:relative;">यो राजा रक्षणे युक्तो भूतेषु कुरुते दयाम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3034"><div class="verse-block verse-justify"><div class="line">यदह्ना कुरुते पापमरक्षन्भयतः प्रजाः।</div><div class="line" style="position:relative;">राजा वर्षसहस्रेण तस्यान्तमधिगच्छति॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3035"><div class="verse-block verse-justify"><div class="line">यदह्ना कुरुते धर्मं प्रजा धर्मेण पालयन्।</div><div class="line" style="position:relative;">दशवर्षसहस्राणि तस्य भुङ्क्ते फलं दिवि॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3036"><div class="verse-block verse-justify"><div class="line">स्विष्टिः स्वधीतिः सुतपा लोकाञ्जयति यावतः।</div><div class="line" style="position:relative;">क्षणेन तानवाप्नोति प्रजा धर्मेण पालयन्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3037"><div class="verse-block verse-justify"><div class="line">एवं धर्मं प्रयत्नेन कौन्तेय परिपालय।</div><div class="line" style="position:relative;">ततः पुण्यफलं लब्ध्वा नानुबन्धेन योक्ष्यसे॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b3038"><div class="verse-block verse-justify"><div class="line">स्वर्गलोके सुमहतीं श्रियं प्राप्स्यसि पाण्डव।</div><div class="line" style="position:relative;">असंभवश्च धर्माणामीदृशानामराजसु॥३३॥</div></div></div>
