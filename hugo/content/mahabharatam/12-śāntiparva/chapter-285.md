<div class="verse-block-wrapper" id="12-śāntiparva-b11687"><div class="verse-block verse-justify"><div class="line">लोकसंभावितैर्दुःखं यत्प्राप्तं कुरुसत्तम।</div><div class="line" style="position:relative;">प्राप्य जातिं मनुष्येषु देवैरपि पितामह॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11688"><div class="verse-block verse-justify"><div class="line">कदा वयं करिष्यामः संन्यासं दुःखभेषजम्।</div><div class="line" style="position:relative;">दुःखमेतच्छरीराणां धारणं कुरुसत्तम॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11689"><div class="verse-block verse-justify"><div class="line">विमुक्ताः सप्तदशभिर्हेतुभूतैश्च पञ्चभिः।</div><div class="line" style="position:relative;">इन्द्रियार्थैर्गुणैश्चैव अष्टाभिश्च पितामह॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11690"><div class="verse-block verse-justify"><div class="line">न गच्छन्ति पुनर्भावं मुनयः संशितव्रताः।</div><div class="line">कदा वयं गमिष्यामो राज्यं हित्वा परंतप ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11691"><div class="verse-block verse-justify"><div class="line">नास्त्यनन्तं महाराज सर्वं सङ्ख्यानगोचरम्।</div><div class="line" style="position:relative;">पुनर्भावोपि संख्यातो नास्ति किंचिदिहाचलम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11692"><div class="verse-block verse-justify"><div class="line">न चापि गम्यते राजन्नैष दोषः प्रसङ्गतः।</div><div class="line" style="position:relative;">उद्योगादेव धर्मज्ञाः कालेनैव गमिष्यथ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11693"><div class="verse-block verse-justify"><div class="line">नेशेऽयं सततं देही नृपते पुण्यपापयोः।</div><div class="line" style="position:relative;">तत एव समुत्थेन तमसा रुध्यतेऽपि च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11694"><div class="verse-block verse-justify"><div class="line">यथाञ्जनमयो वायुः पुनर्मानःशिलं रजः।</div><div class="line" style="position:relative;">अनुप्रविश्य तद्वर्णो दृश्यते रञ्जयन्दिशः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11695"><div class="verse-block verse-justify"><div class="line">तथा कर्मफलैर्देही रञ्जितस्तमसा वृतः।</div><div class="line" style="position:relative;">विवर्णो वर्णमाश्रित्य देहेषु परिवर्तते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11696"><div class="verse-block verse-justify"><div class="line">ज्ञानेन हि यदा जन्तुरज्ञानप्रभवं तमः।</div><div class="line" style="position:relative;">व्यपोहति तदा ब्रह्म प्रकाशेत सनातनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11697"><div class="verse-block verse-justify"><div class="line">अयत्नसाध्यं मुनयो वदन्तिचे चापि मुक्तास्तदुपासितव्याः।</div><div class="line" style="position:relative;">त्वया च लोकेन च सामरेणतस्मान्न शाम्यन्ति महर्षिसङ्घाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11698"><div class="verse-block verse-justify"><div class="line">अस्मिन्नर्थे पुरा गीतं शृणुष्वैकमना नृप।</div><div class="line" style="position:relative;">यथा दैत्येन वृत्रेण भ्रष्टैश्वर्येण चेष्टितम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11699"><div class="verse-block verse-justify"><div class="line">निर्जितेनासहायेन हृतराज्येन भारत।</div><div class="line" style="position:relative;">अशोचता शत्रुमध्ये बुद्धिमास्थाय केवलाम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11700"><div class="verse-block verse-justify"><div class="line">भ्रष्टैश्वर्यं पुरा वृत्रमुशना वाक्यमब्रवीत्।</div><div class="line">कच्चित्पराजितस्याद्य न व्यथा तेऽस्ति दानव ॥वृत्र उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11701"><div class="verse-block verse-justify"><div class="line">सत्येन तपसा चैव विदित्वा संक्षयं ह्यहम्।</div><div class="line" style="position:relative;">न शोचामि न हृष्यामि भूतानामागतिं गतिम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11702"><div class="verse-block verse-justify"><div class="line">कालसंचोदिता जीवा मज्जन्ति नरकेऽवशाः।</div><div class="line" style="position:relative;">परिहृष्टानि सर्वाणि दिव्यान्याहुर्मनीषिणः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11703"><div class="verse-block verse-justify"><div class="line">क्षपयित्वा तु तं कालं गणितं कालचोदिताः।</div><div class="line" style="position:relative;">सावशेषेण कालेन संधावन्ति पुनःपुनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11704"><div class="verse-block verse-justify"><div class="line">तिर्यग्योनिसहस्राणि गत्वा नरकमेव च।</div><div class="line" style="position:relative;">निर्गच्छन्त्यवशा जीवाः कालबन्धनबन्धनाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11705"><div class="verse-block verse-justify"><div class="line">एवं संसरमाणानि ह्यहं भूतानि दृष्टवान्।</div><div class="line" style="position:relative;">यथा कर्म तथा लाभ इति शास्त्रनिदर्शनम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11706"><div class="verse-block verse-justify"><div class="line">तिर्यग्गच्छन्ति नरकं मानुष्यं दैवमेव च।</div><div class="line" style="position:relative;">सुखदुःखे प्रिये द्वेष्ये चरित्वा पूर्वमेव च॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11707"><div class="verse-block verse-justify"><div class="line">कृतान्तविधिसंयुक्तः सर्वो लोकः प्रपद्यते।</div><div class="line" style="position:relative;">गतं गच्छन्ति चाध्वानं सर्वभूतानि सर्वदा॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b11709"><div class="verse-block verse-justify"><div class="line">प्रत्यक्षमेतद्भवतस्तथाऽन्येषां मनीषिणाम्।</div><div class="line" style="position:relative;">मया यज्जयलुब्धेन पुरा तप्तं महत्तपः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b11710"><div class="verse-block verse-justify"><div class="line">गन्धानादाय भूतानां रसांश्च विविधानपि।</div><div class="line" style="position:relative;">अवर्धं त्रीन्समाक्रम्य लोकान्वै स्वेन तेजसा॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="12-śāntiparva-b11714"><div class="verse-block verse-justify"><div class="line">वैकुण्ठः पुरुषोऽनन्तः शुक्लो विष्णुः सनातनः।</div><div class="line" style="position:relative;">मुञ्जकेशो हरिश्मश्रुः सर्वभूतपितामहः॥३०॥</div></div></div>
