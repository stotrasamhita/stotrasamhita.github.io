<div class="verse-block-wrapper" id="12-śāntiparva-b2153"><div class="verse-block verse-justify"><div class="line">धर्मात्मनि महावीर्ये सत्यसन्धे जितात्मनि।</div><div class="line" style="position:relative;">देवव्रते महाभागे शरतल्पगतेऽच्युते॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2154"><div class="verse-block verse-justify"><div class="line">शयाने वीरशयने भीष्मे शन्तनुनन्दने।</div><div class="line" style="position:relative;">गाङ्गेये पुरुषव्याघ्रे पाण्डवैः पर्युपासिते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2155"><div class="verse-block verse-justify"><div class="line">काः कथाः समवर्तन्त तस्मिन्वीरसमागमे।</div><div class="line">हतेषु सर्वसैन्येषु तन्मे शंस महामुने ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2156"><div class="verse-block verse-justify"><div class="line">शरतल्पगते भीष्मे कौरवाणां पितामहे।</div><div class="line" style="position:relative;">आजग्मुर्ऋषयः सिद्धा नारदप्रमुखा नृप॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2157"><div class="verse-block verse-justify"><div class="line">हतशिष्टाश्च राजानो युधिष्ठिरपुरोगमाः।</div><div class="line" style="position:relative;">धृतराष्ट्रश्च कृष्णश्च भीमार्जुनयमास्तथा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2158"><div class="verse-block verse-justify"><div class="line">तेऽभिगम्य महात्मानो भरतानां पितामहम्।</div><div class="line" style="position:relative;">अन्वशोचन्त गाङ्गेयमादित्यं पतितं यथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2159"><div class="verse-block verse-justify"><div class="line">मुहूर्तमिव च ध्यात्वा नारदो देवदर्शनः।</div><div class="line" style="position:relative;">उवाच पाण्डवान्सर्वान्हतशिष्टांश्च पार्थिवान्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2160"><div class="verse-block verse-justify"><div class="line">प्राप्तकालं समाचक्षे भीष्मोऽयमनुयुज्यताम्।</div><div class="line" style="position:relative;">अस्तमेति हि गाङ्गेयो भानुमानिव भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2161"><div class="verse-block verse-justify"><div class="line">अयं प्राणानुत्सिसृक्षुस्तं सर्वेऽभ्यनुपृच्छत।</div><div class="line" style="position:relative;">कृत्स्नान्हि विविधान्धर्मांश्चातुर्वर्ण्यस्य वेत्त्ययम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2162"><div class="verse-block verse-justify"><div class="line">एष वृद्धः पराँल्लोकान्संप्राप्नोति तनुं त्यजन्।</div><div class="line">तं शीघ्रमनुयुञ्जीध्वं संशयान्मनसि स्थितान् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2163"><div class="verse-block verse-justify"><div class="line">एवमुक्ते नारदेन भीष्ममीयुर्नराधिपाः।</div><div class="line" style="position:relative;">प्रष्टुं चाशक्रुवन्तस्ते वीक्षांचक्रुः परस्परम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2164"><div class="verse-block verse-justify"><div class="line">अथोवाच हृषीकेशं पाण्डुपुत्रो युधिष्ठिरः।</div><div class="line" style="position:relative;">नान्यस्त्वद्देवकीपुत्र शक्तः प्रष्टुं पितामहम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2165"><div class="verse-block verse-justify"><div class="line">प्रव्याहर यदुश्रेष्ठ त्वमग्रे मधुसूदन।</div><div class="line" style="position:relative;">त्वं हि नस्तात सर्वेषां सर्वधर्मविदुत्तमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2166"><div class="verse-block verse-justify"><div class="line">एवमुक्तः पाण्डवेन भगवान्केशवस्तदा।</div><div class="line" style="position:relative;">अभिगम्य दुराधर्षं प्रव्याहारयदच्युतः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2167"><div class="verse-block verse-justify"><div class="line">कच्चित्सुखेन रजनी व्युष्टा ते राजसत्तम।</div><div class="line" style="position:relative;">विस्पष्टलक्षणा बुद्धिः कच्चिच्चोपस्थिता तव॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2168"><div class="verse-block verse-justify"><div class="line">कच्चिज्ज्ञानानि सर्वाणि प्रतिभान्ति च तेऽनघ।</div><div class="line">न ग्लायते च हृदयं न च ते व्याकुलं मनः ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2169"><div class="verse-block verse-justify"><div class="line">दाहो मोहः श्रमश्चैव क्लमो ग्लानिस्तथा रुजा।</div><div class="line" style="position:relative;">तव प्रसादाद्वार्ष्णेय सद्यो व्यपगतानि मे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2170"><div class="verse-block verse-justify"><div class="line">यच्च भूतं भविष्यच्च भवच्च परमद्युते।</div><div class="line" style="position:relative;">तत्सर्वमनुपश्यामि पाणौ फलमिवाहितम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2171"><div class="verse-block verse-justify"><div class="line">वेदोक्ताश्चैव ये धर्मा वेदान्ताधिगताश्च ये।</div><div class="line" style="position:relative;">तान्सर्वान्संप्रपश्यामि वरदानात्तवाच्युत॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2172"><div class="verse-block verse-justify"><div class="line">शिष्टैश्च धर्मो यः प्रोक्तः स च मे हृदि वर्तते।</div><div class="line" style="position:relative;">देशजातिकुलानां च धर्मज्ञोऽस्मि जनार्दन॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2173"><div class="verse-block verse-justify"><div class="line">चतुर्ष्वाश्रमधर्मेषु योऽर्थः स च हृदि स्थितः।</div><div class="line" style="position:relative;">राजधर्मांश्च सकलानवगच्छामि केशव॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2174"><div class="verse-block verse-justify"><div class="line">यच्च यत्र च वक्तव्यं तद्वक्ष्यामि जनार्दन।</div><div class="line" style="position:relative;">तव प्रसादाद्धि शुभा मनो मे बुद्धिराविशत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2175"><div class="verse-block verse-justify"><div class="line">युवेवास्मि समावृत्तस्त्वदनुध्यानबृंहितः।</div><div class="line" style="position:relative;">वक्तुं श्रेयः समर्थोऽस्मि त्वत्प्रसादाज्जनार्दन॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2176"><div class="verse-block verse-justify"><div class="line">स्वयं किमर्थं तु भवाञ्श्रेयो न प्राह पाण्डवम्।</div><div class="line">किं ते विवक्षितं चात्र तदाशु वद माधव ॥वासुदेव उवाच।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2177"><div class="verse-block verse-justify"><div class="line">यशसः श्रेयसश्चैव मूल मां विद्धि कौरव।</div><div class="line" style="position:relative;">मत्तः सर्वेऽभिनिर्वृत्ता भावाः सदसदात्मकाः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2178"><div class="verse-block verse-justify"><div class="line">शीतांशुश्चन्द्र इत्युक्ते लोके को विस्मयिष्यति।</div><div class="line" style="position:relative;">तथैव यशसा पूर्णे मयि को विस्मयिष्यति॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2179"><div class="verse-block verse-justify"><div class="line">आधेयं तु मया भूयो यशस्तव महाद्युते।</div><div class="line" style="position:relative;">ततो मे विपुला बुद्धिस्त्वयि भीष्म समाहिता॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2180"><div class="verse-block verse-justify"><div class="line">यावद्धि पृथिवीपाल पृथ्वीयं स्थास्यति ध्रुवा।</div><div class="line" style="position:relative;">तावत्तवाक्षया कीर्तिर्लोकाननुचरिष्यति॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2181"><div class="verse-block verse-justify"><div class="line">यच्च त्वं वक्ष्यसे भीष्म पाण्डवायानुपृच्छते।</div><div class="line" style="position:relative;">वेदप्रवाद इव ते स्थास्यते वसुधातले॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2182"><div class="verse-block verse-justify"><div class="line">यश्चैतेन प्रमाणेन योक्ष्यत्यात्मानमात्मना।</div><div class="line" style="position:relative;">स फलं सर्वपुण्यानां प्रेत्य चानुभविष्यति॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2183"><div class="verse-block verse-justify"><div class="line">एतस्मात्कारणाद्भीष्म मतिर्दिव्या मया हि ते।</div><div class="line" style="position:relative;">दत्ता यशो विप्रथयेत्कथं भूयस्तवेति ह॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2184"><div class="verse-block verse-justify"><div class="line">यावद्धि प्रथते लोके पुरुषस्य यशो भुवि।</div><div class="line" style="position:relative;">तावत्तस्याक्षया कीर्तिर्भवतीति विनिश्चिता॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2185"><div class="verse-block verse-justify"><div class="line">राजानो हतशिष्टास्त्वां राजन्नभित आसते।</div><div class="line" style="position:relative;">धर्माननुयुयुक्षन्तस्तेभ्यः प्रब्रूहि भारत॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2186"><div class="verse-block verse-justify"><div class="line">भवान्हि वयसा वृद्धः श्रुताचारसमन्वितः।</div><div class="line" style="position:relative;">कुशलो राजधर्माणां सर्वेषामपराश्च ये॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2187"><div class="verse-block verse-justify"><div class="line">जन्मप्रभृति ते किंचिद्वॄजिनं न ददर्श ह।</div><div class="line" style="position:relative;">ज्ञातारं सर्वधर्माणां त्वां विदुः सर्वपार्थिवाः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2188"><div class="verse-block verse-justify"><div class="line">तेभ्यः पितेव पुत्रेभ्यो राजन्ब्रूहि परं नयम्।</div><div class="line" style="position:relative;">ऋषयश्चैव देवाश्च त्वया नित्यमुपासिताः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b2189"><div class="verse-block verse-justify"><div class="line">तस्माद्वक्तव्यमेवेदं त्वयाऽवश्यमशेषतः।</div><div class="line" style="position:relative;">धर्मं शुश्रूषमाणेभ्यः पृष्टेन न सता पुनः॥३८॥</div></div></div>
