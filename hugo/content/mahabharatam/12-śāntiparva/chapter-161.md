<div class="verse-block-wrapper" id="12-śāntiparva-b6630"><div class="verse-block verse-justify"><div class="line">यतः प्रभवति क्रोधः कामो वा भरतर्षभ।</div><div class="line" style="position:relative;">शोकमोहौ विधित्सा च परासुत्वं तथा मदः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6631"><div class="verse-block verse-justify"><div class="line">लोभो मात्सर्यमीर्ष्या च कुत्साऽसूया कृपा भयम्।</div><div class="line">एतत्सर्वं महाप्राज्ञ याथातथ्येन मे वद ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6632"><div class="verse-block verse-justify"><div class="line">त्रयोदशैतेऽतिबलाः शत्रवः प्राणिनां स्मृताः।</div><div class="line" style="position:relative;">उपासते महाराज समन्तात्पुरुषानिह॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6633"><div class="verse-block verse-justify"><div class="line">एते प्रमत्तं पुरुषमप्रमत्तास्तुदन्ति च।</div><div class="line" style="position:relative;">वृका इव विलुम्पन्ति दृष्ट्वेव पुरुषेतरान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6634"><div class="verse-block verse-justify"><div class="line">एभ्यः प्रवर्तते दुःखमेभ्यः पापं प्रवर्तते।</div><div class="line" style="position:relative;">इति मर्त्यो विजानीयात्सततं पुरुषर्षभ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6635"><div class="verse-block verse-justify"><div class="line">एतेषामुदयं स्थानं क्षयं च पृथिवीपते।</div><div class="line">हन्त ते कथयिष्यामि क्रोधस्योत्पत्तिमादितः।</div><div class="line" style="position:relative;">यथातत्त्वं क्षितिपते तन्मे निगदतः शृणु॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6636"><div class="verse-block verse-justify"><div class="line">लोभात्क्रोधः प्रभवति परदोषैरुदीर्यते।</div><div class="line" style="position:relative;">क्षमया तिष्ठते राजन्क्षमया विनिवर्तते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6637"><div class="verse-block verse-justify"><div class="line">संकल्पाज्जायते कामः सेव्यमानो विवर्धते।</div><div class="line" style="position:relative;">यदा प्राज्ञो विरमते तदा सद्यः प्रणश्यति॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6638"><div class="verse-block verse-justify"><div class="line">[परामूया क्रोधलोभावन्तरा प्रतिमुच्यते।</div><div class="line">दयया सर्वभूतानां निर्वेदाद्विनिवर्तते।</div><div class="line" style="position:relative;">]अवद्यदर्शनादेति तत्त्वज्ञानाच्च नश्यति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6639"><div class="verse-block verse-justify"><div class="line">अज्ञानप्रभवो मोहः पापाभ्यासात्प्रवर्तते।</div><div class="line" style="position:relative;">यदा प्राज्ञेषु रमते तदा सद्यः प्रणश्यति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6640"><div class="verse-block verse-justify"><div class="line">विरुद्धानीह शास्त्राणि ये पश्यन्ति कुरूद्वह।</div><div class="line" style="position:relative;">विधित्सा जायते तेषां तत्त्वज्ञानान्निवर्तते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6641"><div class="verse-block verse-justify"><div class="line">प्रीतेः शोकः प्रभवति वियोगात्तस्य देहिनः।</div><div class="line" style="position:relative;">यदा निरर्थकं वेत्ति तदा सद्यः प्रणश्यति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6642"><div class="verse-block verse-justify"><div class="line">परासुता क्रोधलोभादभ्यासाच्च प्रवर्तते।</div><div class="line" style="position:relative;">दयया सर्वभूतानां निर्वेदात्सा निवर्तते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6643"><div class="verse-block verse-justify"><div class="line">सत्यत्यागात्तु मात्सर्यमहितानां च सेवया।</div><div class="line" style="position:relative;">एतत्तु क्षीयते तात साधूनामुपसेवनात्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6644"><div class="verse-block verse-justify"><div class="line">कुलाञ्ज्ञानात्तथैश्वर्यान्मदो भवति देहिनाम्।</div><div class="line" style="position:relative;">एभिरेव तु विज्ञातैर्मदः सद्यः प्रणश्यति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6645"><div class="verse-block verse-justify"><div class="line">ईर्ष्या कामात्प्रभवति संहर्षाच्चैव जायते।</div><div class="line" style="position:relative;">इतरेषां तु सत्वानां प्रज्ञया सा प्रणश्यति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6646"><div class="verse-block verse-justify"><div class="line">विभ्रमाल्लोकबाह्यानां द्वेष्यैर्वाक्यैरसंमतैः।</div><div class="line" style="position:relative;">कुत्सा संजायते राजँल्लोकान्प्रेक्ष्याभिशाम्यति॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6647"><div class="verse-block verse-justify"><div class="line">प्रतिकर्तुं न शक्ता ये बलस्थायापकारिणे।</div><div class="line" style="position:relative;">असूया जायते तीव्रा कारुण्याद्विनिवर्तते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6648"><div class="verse-block verse-justify"><div class="line">कृपणान्सततं दृष्ट्वा ततः संजायते कृपा।</div><div class="line" style="position:relative;">धर्मनिष्ठां यदा वेत्ति तदा शाम्यति सा कृपा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b6649"><div class="verse-block verse-justify"><div class="line">अज्ञानप्रभवो लोभो भूतानां दृश्यते सदा।</div><div class="line" style="position:relative;">अस्थिरत्वं च भोगानां दृष्ट्वा ज्ञात्वा निवर्तते॥२१॥</div></div></div>
