<div class="verse-block-wrapper" id="12-śāntiparva-b12363"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12364"><div class="verse-block verse-justify"><div class="line">तिष्ठते मे सदा तात कौतूहलमिदं हृदि।</div><div class="line" style="position:relative;">तदहं श्रोतुमिच्छामि त्वत्तः कुरुपितामह॥२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12365"><div class="verse-block verse-justify"><div class="line">कथं देवर्षिरुशना सदा काव्यो महामतिः।</div><div class="line" style="position:relative;">असुराणां प्रियकारः सुराणामप्रिये रतः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12366"><div class="verse-block verse-justify"><div class="line">वर्धयामास तेजश्च किमर्थममितौजसाम्।</div><div class="line" style="position:relative;">नित्यं वैरनिबद्धाश्च दानवाः सुरसत्तमैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12367"><div class="verse-block verse-justify"><div class="line">कथं चाप्युशना प्राप शुक्रत्वममरद्युतिः।</div><div class="line" style="position:relative;">ऋद्धिं च स कथं प्राप्तः सर्वमेतद्ब्रवीहि मे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12368"><div class="verse-block verse-justify"><div class="line">न याति च स तेजस्वी मध्येन नभसः कथम्।</div><div class="line">एतदिच्छामि विज्ञातुं निखिलेन पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12369"><div class="verse-block verse-justify"><div class="line">शृणु राजन्नवहितः सर्वमेतद्यथातथम्।</div><div class="line" style="position:relative;">यथामति यथा चैतच्छ्रुतपूर्वं मयाऽनघ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12370"><div class="verse-block verse-justify"><div class="line">एष भार्गवदायादो मुनिर्मान्यो दृढव्रतः।</div><div class="line" style="position:relative;">सुराणां विप्रियकरो निमित्ते कारणात्मके॥८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12371"><div class="verse-block verse-justify"><div class="line">इन्द्रोऽथ धनदो राजा यक्षरक्षोधिपः सदा।</div><div class="line" style="position:relative;">प्रभविष्णुश्च कोशस्य जगतश्च तथा प्रभुः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12372"><div class="verse-block verse-justify"><div class="line">तस्यात्मानमथाविश्य योगसिद्धो महामुनिः।</div><div class="line" style="position:relative;">रुद्ध्वा धनपतिं देवं योगेन हृतवान्वसु॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12373"><div class="verse-block verse-justify"><div class="line">हृते धने ततः शर्म न लेभे धनदस्तथा।</div><div class="line" style="position:relative;">आपन्नमन्युः संविग्नः सोभ्यगात्सुरसत्तमम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12374"><div class="verse-block verse-justify"><div class="line">निवेदयामास तदा शिवायामिततेजसे।</div><div class="line" style="position:relative;">देवश्रेष्ठाय रुद्राय सौम्याय बहुरूपिणे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12375"><div class="verse-block verse-justify"><div class="line">योगात्मकेनोशनसा रुद्ध्वा मम हृतं वसु।</div><div class="line" style="position:relative;">योगेनात्मगतं कृत्वा निःसृतश्च महातपाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12376"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा ततः क्रुद्धो महायोगी महेश्वरः।</div><div class="line" style="position:relative;">संरक्तनयनो राजञ्शूलमादाय तस्थिवान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12377"><div class="verse-block verse-justify"><div class="line">क्वासौ क्वासाविति प्राह गृहीत्वा परमायुधम्।</div><div class="line" style="position:relative;">उशना दूरतस्तस्य ह्यभूज्ज्ञात्वा चिकीर्षितम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12378"><div class="verse-block verse-justify"><div class="line">स महायोगिनो बुद्ध्वा तं रोषं वै महात्मनः।</div><div class="line" style="position:relative;">गतिमागमनं वेत्ति स्थानं चैव ततः प्रभुः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12379"><div class="verse-block verse-justify"><div class="line">संचिन्त्येग्रेण तपसा महात्मानं महेश्वरम्।</div><div class="line" style="position:relative;">उशना योगसिद्धात्मा शूलाग्रे प्रत्यदृश्यत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12380"><div class="verse-block verse-justify"><div class="line">विज्ञातरूपः स तदा तपः सिद्धोऽथ धन्विना।</div><div class="line" style="position:relative;">ज्ञात्वा शूलं च देवेशः पाणिना समनामयत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12381"><div class="verse-block verse-justify"><div class="line">आनतेनाथ शूलेन पाणिनामिततेजसा।</div><div class="line" style="position:relative;">पिनाकमिति चोवाच शूलमुग्रायुधः प्रभुः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12382"><div class="verse-block verse-justify"><div class="line">पाणिमध्यगतं दृष्ट्वा भार्गवं तमुमापतिः।</div><div class="line" style="position:relative;">आस्यं विवृत्य भ्रकुटिं पाणिं संप्राक्षिपच्छनैः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12383"><div class="verse-block verse-justify"><div class="line">स तु प्रविष्ट उशना कोष्ठं माहेश्वरं प्रभुः।</div><div class="line">व्यचरच्चापि तत्रासौ महात्मा भृगुनन्दनः ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12384"><div class="verse-block verse-justify"><div class="line">किमर्थं व्यचरद्राजन्नुशना तस्य धीमतः।</div><div class="line">जठरे देवदेवस्य किंचाकार्षीन्महाद्युतिः ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12385"><div class="verse-block verse-justify"><div class="line">पुरा सोऽन्तर्जलगतः स्थाणुभूतो महाव्रतः।</div><div class="line" style="position:relative;">वर्षाणामभवद्राजन्प्रयुतान्यर्बुदानि च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12386"><div class="verse-block verse-justify"><div class="line">उदतिष्ठत्तपस्तप्त्वा दुश्चरं च महाह्रदात्।</div><div class="line" style="position:relative;">ततो देवातिदेवस्तं ब्रह्मा वै समसर्पत॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12387"><div class="verse-block verse-justify"><div class="line">तपोवृद्धिमपृच्छच्च कुशलं चैवमव्ययः।</div><div class="line" style="position:relative;">तपः सुचीर्णमिति च प्रोवाच वृषभध्वजः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12388"><div class="verse-block verse-justify"><div class="line">तत्संयोगेन वृद्धिं चाप्यपश्यत्स तु शंकरः।</div><div class="line" style="position:relative;">महामतिरचिन्त्यात्मा सत्यधर्मरतः सदा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12389"><div class="verse-block verse-justify"><div class="line">स तेनाढ्यो महायोगी तपसा च धनेन च।</div><div class="line" style="position:relative;">व्यराजत महाराज त्रिषु लोकेषु वीर्यवान्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12390"><div class="verse-block verse-justify"><div class="line">ततः पिनाकी योगात्मा ध्यानयोगं समाविशत्।</div><div class="line" style="position:relative;">उशना तु समुद्विग्नो निलिल्ये जठरे ततः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12391"><div class="verse-block verse-justify"><div class="line">तुष्टाव च महायोगी देवं तत्रस्थ एव च।</div><div class="line" style="position:relative;">निःसारं काङ्क्षमाणः स तेन स्म प्रतिहन्यते॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12392"><div class="verse-block verse-justify"><div class="line">उशना तु तथोवाच जठरस्थो महामुनिः।</div><div class="line" style="position:relative;">प्रसादं मे कुरुष्वेति पुनः पुनररिंदम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12393"><div class="verse-block verse-justify"><div class="line">तमुवाच महादेवो गच्छ शिश्नेन मोक्षणम्।</div><div class="line" style="position:relative;">इति सर्वाणि स्रोतांसि रुद्ध्वा त्रिदशपुङ्गवः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12394"><div class="verse-block verse-justify"><div class="line">अपश्यमानस्तद्द्वारं सर्वतः पिहितो मुनिः।</div><div class="line" style="position:relative;">पर्यक्रामद्दह्यमान इतश्चेतश्च तेजसा॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12395"><div class="verse-block verse-justify"><div class="line">स वै निष्क्रम्य शिश्नेन शुक्रत्वमभिपेदिवान्।</div><div class="line" style="position:relative;">कार्येण तेन नभसो नाध्यगच्छत मध्यतः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12396"><div class="verse-block verse-justify"><div class="line">`तत एव च देवेषु अप्रविष्टो महामुनिः।</div><div class="line" style="position:relative;">पौरोहित्यं च दैत्यानां शक्रतेजोविवृद्धये ॥'॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12397"><div class="verse-block verse-justify"><div class="line">विनिष्क्रान्तं तु तं दृष्ट्वा ज्वलन्तमिव तेजसा।</div><div class="line" style="position:relative;">भवो रोषसमाविष्टः शूलोद्यतकरः स्थितः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12398"><div class="verse-block verse-justify"><div class="line">अवारयत तं देवी क्रुद्धं पशुपतिं पतिम्।</div><div class="line">पुत्रत्वमगमद्देव्या वारिते शंकरे च सः ॥देव्युवाच।</div><div class="line" style="position:relative;">॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12399"><div class="verse-block verse-justify"><div class="line">हिंसनीयस्त्वया नैव मम पुत्रत्वमागतः।</div><div class="line" style="position:relative;">न हि देवोदरात्कश्चिन्निःसृतो नाशमर्हति॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12400"><div class="verse-block verse-justify"><div class="line">ततः प्रीतो भवो देव्याः प्रहसंश्चेदमब्रवीत्।</div><div class="line" style="position:relative;">गच्छत्वेष यथाकाममिति राजन्पुनः पुनः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="12-śāntiparva-b12401"><div class="verse-block verse-justify"><div class="line">ततः प्रणम्य वरदं देवं देवीमुमां तथा।</div><div class="line" style="position:relative;">उशना प्राप तद्धीमान्गतिमिष्टां महामुनिः॥३९॥</div></div></div>
