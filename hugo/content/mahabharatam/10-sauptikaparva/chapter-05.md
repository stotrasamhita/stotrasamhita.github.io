<div class="verse-block-wrapper" id="10-sauptikaparva-b0190"><div class="verse-block verse-justify"><div class="line">तथैव तावन्मेधावी विनयं यो न शिक्षते।</div><div class="line" style="position:relative;">न च किञ्चन जानाति सोऽपि धर्मार्थनिश्चयम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="10-sauptikaparva-b0192"><div class="verse-block verse-justify"><div class="line">मुहूर्तमपि तं प्राज्ञः पण्डितं पर्युपास्य हि।</div><div class="line" style="position:relative;">क्षिप्रं धर्मान्विजानाति जिह्वा सूपरसानिव॥५॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0193"><div class="verse-block verse-justify"><div class="line">शुश्रूषुस्त्वेव मेधावी पुरुषो नियतेन्द्रियः।</div><div class="line" style="position:relative;">जानीयादागमान्सर्वान्ग्राह्यं च न विरोधयेत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0194"><div class="verse-block verse-justify"><div class="line">अनयस्त्ववमानी यो दुरात्मा पापपूरुषः।</div><div class="line" style="position:relative;">दिष्टमुत्सृज्य कल्याणं करोति बहुपातकम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0195"><div class="verse-block verse-justify"><div class="line">नाथवन्तं तु सुहृदः प्रतिषेधन्ति पातकात्।</div><div class="line" style="position:relative;">निवर्तते तु लक्ष्मीवान्नालक्ष्मीवान्निवर्तते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0196"><div class="verse-block verse-justify"><div class="line">यथा ह्युच्चावचैर्वाक्यैः क्षिप्तचित्तो नियम्यते।</div><div class="line" style="position:relative;">तथैव सुहृदा शक्यो नशक्यस्त्ववसीदति॥९॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0197"><div class="verse-block verse-justify"><div class="line">तथैव सुहृदोऽप्राज्ञान्कुर्वाणान्कर्म पापकम्।</div><div class="line" style="position:relative;">प्राज्ञाः सम्प्रतिषेधन्ति यथाशक्ति पुनःपुनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0198"><div class="verse-block verse-justify"><div class="line">स कल्याणे मनः कृत्वा नियम्यात्मानमात्मना।</div><div class="line" style="position:relative;">कुरु मे वचनं तात येन पश्चान्न तप्स्यसे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0199"><div class="verse-block verse-justify"><div class="line">न वधः पूज्यते लोके सुप्तानामिह धर्मतः।</div><div class="line" style="position:relative;">तथैव न्यस्तशस्त्राणां विमुक्तरथवाजिनाम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0200"><div class="verse-block verse-justify"><div class="line">ये व ब्रूयुस्तवास्मीति ये च स्युः शरणागताः।</div><div class="line" style="position:relative;">विमुक्तमूर्धजा ये च ये चापि हतवाहनाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0201"><div class="verse-block verse-justify"><div class="line">अद्य स्वप्स्यन्ति पाञ्चाला विमुक्तकवचा विभो।</div><div class="line" style="position:relative;">विश्वस्ता रजनीं सर्वे प्रेता इव विचेतसः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0202"><div class="verse-block verse-justify"><div class="line">यस्तेषां तदवस्थानां द्रुह्येत पुरुषोऽनृजुः।</div><div class="line" style="position:relative;">व्यक्तं स नरके मज्जेदगाधे विपुलेऽप्लुवे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0203"><div class="verse-block verse-justify"><div class="line">सर्वास्त्रविदुषां लोके श्रेष्ठस्त्वमसि विश्रुतः।</div><div class="line" style="position:relative;">न च ते जातु लोकेऽस्मिन्सुसूक्ष्ममपि किल्बिषम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0204"><div class="verse-block verse-justify"><div class="line">त्वं पुनः सूर्यसङ्काशः श्वोभूत उदिते रवौ।</div><div class="line" style="position:relative;">प्रकाशे सर्वभूतानां विजेता युधि शात्रवान्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0205"><div class="verse-block verse-justify"><div class="line">असम्भावितरूपं हि त्वयि कर्म विगर्हितम्।</div><div class="line">शुक्ले रक्तमिव न्यस्तं भवेदिति मतिर्मम ॥अश्वत्थामोवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0206"><div class="verse-block verse-justify"><div class="line">एवमेव यथाऽऽत्थ त्वमनुशाससि मातुल।</div><div class="line" style="position:relative;">तैस्तु पूर्वमयं सेतुः समन्ताद्विह्वलीकृतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0207"><div class="verse-block verse-justify"><div class="line">प्रत्यक्षं भूमिपालानां भवतां चापि सन्निधौ।</div><div class="line" style="position:relative;">न्यस्तशस्त्रो मम पिता धृष्टद्युम्नेन पातितः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0208"><div class="verse-block verse-justify"><div class="line">कर्णश्च पतिते चक्रे उत्थास्यन् रथिनां वरः।</div><div class="line" style="position:relative;">उत्तमे व्यसने मग्नो हतो गाण्डीवधन्वना॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0209"><div class="verse-block verse-justify"><div class="line">तथा शान्तनवो भीष्मो न्यस्तशस्त्रो निरायुधः।</div><div class="line" style="position:relative;">शिखण्डिनं पुरस्कृत्य हतो गाण्डीवधन्वना॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0210"><div class="verse-block verse-justify"><div class="line">भूरिश्रवा महेष्वासस्तथा प्रायगतो रणे।</div><div class="line" style="position:relative;">क्रोशतां बूमिपालानां युयुधानेन पातितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0211"><div class="verse-block verse-justify"><div class="line">दुर्योधनश्च भीमेन समेत्य गदया मृधे।</div><div class="line" style="position:relative;">पश्यतां भूमिपालानामधर्मेण निपातितः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0212"><div class="verse-block verse-justify"><div class="line">एकाकी बहुभिस्तत्र परिवार्य महारथैः।</div><div class="line" style="position:relative;">अधर्मेण नरव्याघ्रो भीमसेनेन पातितः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0213"><div class="verse-block verse-justify"><div class="line">विलापो भग्नसक्थस्य यो मे राज्ञः परिश्रुतः।</div><div class="line" style="position:relative;">वादिकानां कथयतां स मे मर्माणि कृन्तति॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="10-sauptikaparva-b0222"><div class="verse-block verse-justify"><div class="line">हत्वा शतसहस्राणि योधानां निशितैः शरैः।</div><div class="line" style="position:relative;">न्यस्तशस्त्रो मम पिता धृष्टद्युम्नेन पातितः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="10-sauptikaparva-b0223"><div class="verse-block verse-justify"><div class="line">तं तथैव वधिष्यामि न्यस्तवर्माणमद्य वै।</div><div class="line" style="position:relative;">पुत्रं पाञ्चालराजस्य पापं पापेन कर्मणा॥३६॥</div></div></div>
