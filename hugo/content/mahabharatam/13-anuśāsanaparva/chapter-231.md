<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9531"><div class="verse-block verse-justify"><div class="line">धर्मकार्ये यदाऽश्रद्धा पापकर्मणि हर्षणम्।</div><div class="line" style="position:relative;">एवमाद्यशुभं कर्म मनसा पापमुच्यते॥९॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9533"><div class="verse-block verse-justify"><div class="line">अगम्यागमनं चैव परदारनिषेवणम्।</div><div class="line" style="position:relative;">वधबन्धपरिक्लेशैः परप्राणोपतापनम्॥११॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9535"><div class="verse-block verse-justify"><div class="line">दर्पात्स्तम्भाभिमानाच्च परेषामुपतापनम्।</div><div class="line" style="position:relative;">अकार्याणां च करणमशौचं पानसेवनम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9536"><div class="verse-block verse-justify"><div class="line">दौःशील्यं पापसम्पर्के साहाय्यं पापकर्मणि।</div><div class="line">अधर्म्यमयशस्यं च कार्यं तस्य निषेवणम्।</div><div class="line" style="position:relative;">एवमाद्यशुभं चान्यच्छारीरं पापमुच्यते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9537"><div class="verse-block verse-justify"><div class="line">मानसाद्वाङ्मयं पापं विशिष्टमिति लक्ष्यते।</div><div class="line" style="position:relative;">वाङ्मयादपि वै पापाच्छारीरं गण्यते बहु॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9538"><div class="verse-block verse-justify"><div class="line">एवं पापयुतं कर्म त्रिविधं पातयेन्नरम्।</div><div class="line" style="position:relative;">परापकारजननमत्यन्तं पातकं स्मृतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9539"><div class="verse-block verse-justify"><div class="line">त्रिविधं तत्कृतं पापं कर्तारं पापकं नयेत्।</div><div class="line" style="position:relative;">पातकं चापि यत्कर्म कर्मणा बुद्धिपूर्वकम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9540"><div class="verse-block verse-justify"><div class="line">सापदेशमवश्यं तत्कर्तव्यमिति तत्कृतम्।</div><div class="line" style="position:relative;">कथञ्चित्तत्कृतमपि कर्ता तेन स लिप्यते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9541"><div class="verse-block verse-justify"><div class="line">अवश्यं पापदेशेन प्रतिहृन्येत कारणम् ॥उमोवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9542"><div class="verse-block verse-justify"><div class="line">भगवन्पापकं कर्म यथा कृत्वा न लिप्यते ॥महेश्वर उवाच।</div><div class="line" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9543"><div class="verse-block verse-justify"><div class="line">यो नरोऽनपराधी च स्वात्मप्राणस्य रक्षणात्।</div><div class="line">शत्रुमुद्यतशस्त्रं वा पूर्वं तेन हतोपि वा।</div><div class="line" style="position:relative;">प्रतिहन्यान्नरो हिंस्यान्न स पापेन लिप्यते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9544"><div class="verse-block verse-justify"><div class="line">चोरादधिकसंत्रस्तस्तत्प्रतीकारचेष्टया।</div><div class="line" style="position:relative;">यः प्रजघ्नन्नरो हन्यान्न स पापेन लिप्यते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9545"><div class="verse-block verse-justify"><div class="line">ग्रामार्थं भर्तृपिण्डार्थं दीनानुग्रहकारणात्।</div><div class="line" style="position:relative;">वधबन्धपरिक्लेशान्कुर्वन्पापात्प्रमुच्यते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9546"><div class="verse-block verse-justify"><div class="line">दुर्भिक्षे चात्मवृत्त्यर्थमेकायतनगस्तथा।</div><div class="line" style="position:relative;">अकार्यं वाऽप्यभक्ष्यं वा कृत्वा पपान्न लिप्यते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9547"><div class="verse-block verse-justify"><div class="line">विधिरेष गृहस्थानां प्रायेणैवोपदिश्यते।</div><div class="line" style="position:relative;">अवाच्यं वाऽप्यकार्यं वा देशकालवशेन तु॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9548"><div class="verse-block verse-justify"><div class="line">बुद्धिपूर्व नरः कुर्वस्तत्प्रयोजनमात्रया।</div><div class="line" style="position:relative;">किञ्चिद्वा लिप्यते पापैरथवा न च लिप्यते॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9549"><div class="verse-block verse-justify"><div class="line">एवं देवि विजानीहि नास्ति तत्र विचारणा ॥उमोवाच।</div><div class="line" style="position:relative;">॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b9550"><div class="verse-block verse-justify"><div class="line">भगवन्पानदोषांश्च पेयापेयत्वकारणम्।</div><div class="line">एतदिच्छाम्यहं श्रोतुं तन्मे वद महेश्वर ॥महेश्वर उवाच।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
