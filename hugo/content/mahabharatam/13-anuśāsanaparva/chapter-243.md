<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10030"><div class="verse-block verse-justify"><div class="line">केचिन्मोक्षं प्रशंसन्ति केचिद्यज्ञफलं द्विजाः।</div><div class="line" style="position:relative;">वानप्रस्थं पुनः केचिद्गार्हस्थ्यं केचिदाश्रमम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10031"><div class="verse-block verse-justify"><div class="line">राजधर्माश्रयं केचित्केचित्स्वाध्यायमेव च।</div><div class="line" style="position:relative;">ब्रह्मचर्याश्रमं केचित्केचिद्वाक्संयमाश्रयम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10032"><div class="verse-block verse-justify"><div class="line">मातरं पितरं केचित्सेवमाना दिवं गताः।</div><div class="line" style="position:relative;">अहिंसया परः स्वर्गे सत्येन च महीयते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10033"><div class="verse-block verse-justify"><div class="line">आहवेऽभिमुखाः केचिन्निहतास्त्रिदिवं गताः।</div><div class="line" style="position:relative;">केचिदुञ्छवृत्ते सिद्धाः स्वर्गमार्गं समाश्रिताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10034"><div class="verse-block verse-justify"><div class="line">आर्जवेनापरे युक्ता महतां पूजते रताः।</div><div class="line" style="position:relative;">ऋजवो नाकपृष्ठे तु शुद्धात्मानः प्रतिष्ठिताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10035"><div class="verse-block verse-justify"><div class="line">एवं बहुविधैर्लोके धर्मद्वारैः सुसंवृतैः।</div><div class="line" style="position:relative;">ममापि मतिराविद्धा मेघलेखेव वायुना॥८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10036"><div class="verse-block verse-justify"><div class="line">एतस्मिन्संशयस्थाने संशयच्छेदकारि यत्।</div><div class="line">वचनं ब्रूहि देवेश निश्चयज्ञानसंज्ञितम् ॥नारद उवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10037"><div class="verse-block verse-justify"><div class="line">एवं पृष्टः स्वया देव्या महादेवः पिनाकधृक्।</div><div class="line">प्रोवाच मधुरं वाक्यं सूक्ष्ममध्यात्मसंश्रितम् ॥महेश्वर उवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10038"><div class="verse-block verse-justify"><div class="line">न्यायतस्त्वं महाभागे श्रोतुकामाऽसि निश्चयम्।</div><div class="line" style="position:relative;">एतदेव विशिष्टं ते यत्त्वं पृच्छसि मां प्रिये॥११॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10039"><div class="verse-block verse-justify"><div class="line">सर्वत्र विहितो धर्मः स्वर्गलोकफलाश्रितः।</div><div class="line" style="position:relative;">बहुद्वारस्य धर्मस्य नेहास्ति विफलाः क्रियाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10040"><div class="verse-block verse-justify"><div class="line">यस्मिन्यस्मिंश्च विषये योयो याति विनिश्चयम्।</div><div class="line" style="position:relative;">तं तमेवाभिजानाति नान्यं धर्मं शुचिस्मिते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10041"><div class="verse-block verse-justify"><div class="line">शृणु देवि समासेन मोक्षद्वारसमनुत्तमम्।</div><div class="line" style="position:relative;">एतद्धि सर्वधर्माणां विशिष्टं शुभमव्ययम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10042"><div class="verse-block verse-justify"><div class="line">नास्ति मोक्षात्परं देवि मोक्ष एव परा गतिः।</div><div class="line" style="position:relative;">सुखमात्यन्तिकं श्रेष्ठमनिवृत्तं च तद्विदुः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10043"><div class="verse-block verse-justify"><div class="line">नात्र देवि जरा मृत्युः शोको वा दुःखमेव वा।</div><div class="line" style="position:relative;">अनुत्तममचिन्त्यं च तद्देवि परमं सुखम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10044"><div class="verse-block verse-justify"><div class="line">ज्ञानानामुत्तमं ज्ञानं मोक्षज्ञानं विदुर्बुधाः।</div><div class="line" style="position:relative;">ऋषिभिर्देवसङ्घैश्च प्रोच्यते परमं पदम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10045"><div class="verse-block verse-justify"><div class="line">नित्यमक्षरमक्षोभ्यमजेयं शाश्वतं शिवम्।</div><div class="line" style="position:relative;">विशन्ति तत्पदं प्राज्ञाः स्पृहणीयं सुरोत्तमैः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10046"><div class="verse-block verse-justify"><div class="line">दुःखादिश्च दुरन्तश्च संसारोयं प्रकीर्तितः।</div><div class="line" style="position:relative;">शोकव्याधिजरादोषैर्मरणेन च संयुतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10047"><div class="verse-block verse-justify"><div class="line">यथा ज्योतिर्गणा व्योम्नि विवर्तन्ते पुनःपुनः।</div><div class="line" style="position:relative;">तस्य मोक्षस्य मार्गोऽयं श्रुयतां शुभलक्षणे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10048"><div class="verse-block verse-justify"><div class="line">ब्रह्मादिस्थावरान्तश्च संसारो यः प्रकीर्तितः।</div><div class="line" style="position:relative;">संसारे प्राणिनः सर्वे निवर्तन्ते यथा पुनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10049"><div class="verse-block verse-justify"><div class="line">तत्र संसारचक्रस्य मोक्षो ज्ञानेन दृश्यते।</div><div class="line" style="position:relative;">अध्यात्मतत्वविज्ञानं ज्ञानमित्यभिधीयते॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10050"><div class="verse-block verse-justify"><div class="line">ज्ञानस्य ग्रहणोपायमाचारं ज्ञानिनस्तथा।</div><div class="line" style="position:relative;">यथावत्सम्प्रवक्ष्यामि तत्त्वमेकमनाः शृणु॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10051"><div class="verse-block verse-justify"><div class="line">ब्राह्मणः क्षत्रियो वाऽपि भूत्वा पूर्वं गृहे स्थितः।</div><div class="line" style="position:relative;">आनृण्यं सर्वतः प्राप्य ततस्तान्संत्यजेद्गृहान्॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10054"><div class="verse-block verse-justify"><div class="line">गृह्णीयादप्युपाध्यायान्मोक्षज्ञानमनिन्दितः।</div><div class="line" style="position:relative;">द्विविधं च पुनर्मोक्षं साङ्ख्ययोगमिति स्मृतिः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10055"><div class="verse-block verse-justify"><div class="line">पञ्चविंशतिविज्ञानं साङ्ख्यमित्यभिधीयते।</div><div class="line">ऐश्वर्यं देवसारूप्यं योगशास्त्रस्य निर्णयः।</div><div class="line" style="position:relative;">तयोरन्यतरं ज्ञानं शृणुयाच्छिष्यतां गतः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10056"><div class="verse-block verse-justify"><div class="line">नाकालो नाप्यकाषायी नाप्यसंवत्सरोषितः।</div><div class="line">नासाङ्ख्ययोगो नाश्राद्धं गुरुणा स्नेहपूर्वकम्।</div><div class="line" style="position:relative;">समः शीतोष्णहर्षादीन्विषहेत स वै मुनिः॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b10058"><div class="verse-block verse-justify"><div class="line">कुण्डिकाचमसं शिक्यं छत्रं यष्टिमुपानहौ।</div><div class="line" style="position:relative;">चेलमित्येव नैतेषु स्थापयेत्साम्यमात्मनः॥३०॥</div></div></div>
