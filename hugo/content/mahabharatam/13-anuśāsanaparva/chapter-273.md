<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11373"><div class="verse-block verse-justify"><div class="line">जनार्दनेनानुगतो विदुरेण च धीमता।</div><div class="line" style="position:relative;">युयुत्सुना च कौरव्यो युयुधानेन चाभिभो॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11374"><div class="verse-block verse-justify"><div class="line">महता राजभोगेन पारिबर्हेण संवृतः।</div><div class="line" style="position:relative;">स्तूयमानो महातेजा भीष्मिस्याग्नीननुव्रजन्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11375"><div class="verse-block verse-justify"><div class="line">निश्चक्राम पुरात्तस्माद्यथा देवपतिस्तथा।</div><div class="line" style="position:relative;">आससाद कुरुक्षेत्रे ततः शान्तनवं नृपः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11376"><div class="verse-block verse-justify"><div class="line">उपास्यमानं व्यासेन पाराशर्येण धीमता।</div><div class="line" style="position:relative;">नारदेन च राजर्षे देवलेनासितेन च॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11377"><div class="verse-block verse-justify"><div class="line">हतशिष्टैर्नृपैस्चान्यैर्नानादेशसमागतैः।</div><div class="line" style="position:relative;">रक्षिभिश्च महात्मानं रक्ष्यमाणं समन्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11378"><div class="verse-block verse-justify"><div class="line">शयानं वीरशयने ददर्श नृपतिस्ततः।</div><div class="line" style="position:relative;">ततो रथादवारोहद्धातृभिः सह धर्मराट्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11379"><div class="verse-block verse-justify"><div class="line">अभिवाद्याथ कौन्तेयः पितामहमरिंदमम्।</div><div class="line" style="position:relative;">द्वैपायनादीन्विप्रांश्च तैश्च प्रत्यभिनन्दितः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11380"><div class="verse-block verse-justify"><div class="line">ऋत्विग्भिर्ब्रह्मकल्पैश्च भ्रातृभिः सह धर्मजः।</div><div class="line" style="position:relative;">आसाद्य शरतल्पस्थमृषिभिः परिवारितम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11381"><div class="verse-block verse-justify"><div class="line">अब्रवीद्भरतश्रेष्ठं धर्मराजो युधिष्ठिरः।</div><div class="line" style="position:relative;">भ्रातृभिः सह कौरव्यः शयानं निम्नगासुतम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11382"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरोऽहं नृपते नमस्ते जाह्नवीसुत।</div><div class="line" style="position:relative;">शृणोषि चेन्महाबाहो ब्रूहि किं करवाणि ते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11383"><div class="verse-block verse-justify"><div class="line">प्राप्तोस्मि समये राजन्नग्नीनादाय ते विभो।</div><div class="line" style="position:relative;">आचार्यान्ब्राह्मणांश्चैव ऋत्विजो भ्रातरश्च मे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11384"><div class="verse-block verse-justify"><div class="line">पुत्रश्च ते महातेजा धृतराष्ट्रो जनेश्वरः।</div><div class="line" style="position:relative;">उपस्थितः सहामात्यो वासुदेवश्च वीर्यवान्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11385"><div class="verse-block verse-justify"><div class="line">हतशिष्टाश्च राजानः सर्वे च कुरुजाङ्गलाः।</div><div class="line" style="position:relative;">तान्पश्य नरशार्दूल समुन्मीलय लोचने॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11386"><div class="verse-block verse-justify"><div class="line">यच्चेह किञ्चित्कर्तव्यं तत्सर्वं प्रापितं मया।</div><div class="line">यथोक्तं भवता काले सर्वमेव च तत्कृतम् ॥वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11387"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु गाङ्गेयः कुन्तीपुत्रेण धीमता।</div><div class="line" style="position:relative;">ददर्श भारतान्सर्वान्स्थितान्सम्परिवार्य ह॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11388"><div class="verse-block verse-justify"><div class="line">ततश्चलवपुर्भीष्मः प्रगृह्य विपुलं भुजम्।</div><div class="line" style="position:relative;">ओघमेघस्वरो वाग्मी काले वचनमब्रवीत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11389"><div class="verse-block verse-justify"><div class="line">दिष्ट्या प्राप्तोसि कौन्तेय सहामात्यो युधिष्ठिर।</div><div class="line" style="position:relative;">परिवृत्तो हि भगवान्सहस्रांशुर्दिवाकरः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11390"><div class="verse-block verse-justify"><div class="line">अष्टपञ्चाशतं रात्र्यः शयानस्याद्य मे गताः।</div><div class="line" style="position:relative;">शरेषु निशिताग्नेषु यथा वर्षशतं तथा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11391"><div class="verse-block verse-justify"><div class="line">माघोऽयं समनुप्राप्तो मासः पुण्यो युधिष्ठिर।</div><div class="line" style="position:relative;">त्रिभागशेषः पक्षोऽयं शुक्लो भवितुमर्हति॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11392"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा तु गाङ्गेयो धर्मपुत्रं युधिष्ठिरम्।</div><div class="line">धृतराष्ट्रमथामन्त्र्य काले वचनमब्रवीत् ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11393"><div class="verse-block verse-justify"><div class="line">राजन्विदितधर्मोसि सुनिर्णीतार्थसंशयः।</div><div class="line" style="position:relative;">बहुश्रुता हि ते विप्रा बहवः पर्युपासिताः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11394"><div class="verse-block verse-justify"><div class="line">वेद शास्त्राणि सूक्ष्माणि धर्मांश्च मनुजेश्वर।</div><div class="line" style="position:relative;">वेदांश्च चतुरः सर्वान्षडङ्गैरुपबृंहितान्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11395"><div class="verse-block verse-justify"><div class="line">न शोचितव्यं कौरव्य भवितव्यं हि तत्तथा।</div><div class="line" style="position:relative;">श्रुतं देवरहस्यं ते कृष्णद्वैपायनादपि॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11396"><div class="verse-block verse-justify"><div class="line">यथा पाण्डोः सुता राजंस्तथैव तव धर्मतः।</div><div class="line" style="position:relative;">तान्पालय स्थितो धर्मे गुरुशुश्रूषणे रतान्॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b11398"><div class="verse-block verse-justify"><div class="line">तव पुत्रा दुरात्मानः क्रोधलोभपरायणाः।</div><div class="line">ईर्ष्याभिभूता दुर्वृत्तास्तान्न शोचितुमर्हसि ॥वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥३५॥</div></div></div>
