<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8378"><div class="verse-block verse-justify"><div class="line">हन्त ते कथयिष्यामि शृणु देवि समाहिता।</div><div class="line" style="position:relative;">आवासार्थं पुरा देवि शुद्धान्वेषी शुचिस्मिते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8379"><div class="verse-block verse-justify"><div class="line">नाध्यगच्छं चिरं कालं देशं शुचितमं शुभे।</div><div class="line" style="position:relative;">एष मेऽभिनिवेशोऽभूत्तस्मिन्काले प्रजापतिः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8380"><div class="verse-block verse-justify"><div class="line">आकुलः सुमहाघोरः प्रादुरासीत्समन्ततः।</div><div class="line" style="position:relative;">सम्भूता भूतसृष्टिश्च घोरा लोकभयावहा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8381"><div class="verse-block verse-justify"><div class="line">नानावर्णा विरूपाश्च तीक्ष्णदंष्ट्राः प्रहारिणः।</div><div class="line">पिशाचरक्षोवदनाः प्राणिनां प्राणहारिणः।</div><div class="line" style="position:relative;">इतश्चरन्ति निघ्नन्तः प्राणिनो भृशमेव च॥८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8382"><div class="verse-block verse-justify"><div class="line">एवं लोके प्राणिहीने क्षयं याते पितामहः।</div><div class="line" style="position:relative;">चिन्तयंस्तत्प्रतीकारं मां च शक्तं हि निग्रहे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8383"><div class="verse-block verse-justify"><div class="line">एवं ज्ञात्वा ततो ब्रह्मा तस्मिन्कर्मण्योजयत्।</div><div class="line" style="position:relative;">तच्च प्राणिहितार्थं तु मयाऽप्यनुमतं प्रिये॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8384"><div class="verse-block verse-justify"><div class="line">तस्मात्संरक्षिता देवि भूतेभ्यः प्राणिनो भयात्।</div><div class="line">अस्माच्छ्मशानान्मेध्यं तु नास्ति किञ्चिदनिन्दिते।</div><div class="line" style="position:relative;">निःसम्पातान्मनुष्याणां तस्माच्छुचितमं स्मृतं॥११॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8385"><div class="verse-block verse-justify"><div class="line">भूतसृष्टिं च तां चाहं श्मशाने संन्यवेशयम्।</div><div class="line" style="position:relative;">तत्रस्थः सर्वभूतानां विनिहन्मि प्रिये भयम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8386"><div class="verse-block verse-justify"><div class="line">न च भूतगणेनाहं विना वसितुमुत्सहे।</div><div class="line" style="position:relative;">तस्मान्मे सन्निवासाय श्मशाने रोचते मनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8387"><div class="verse-block verse-justify"><div class="line">मेध्यकामैर्द्विजैनित्यं मेध्यमित्यभिधीयते।</div><div class="line" style="position:relative;">आचरद्भिर्व्रतं नित्यं मोक्षकामैश्च सेव्यते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8388"><div class="verse-block verse-justify"><div class="line">स्थानं मे तत्र विहितं वीरस्थानमिति प्रिये।</div><div class="line" style="position:relative;">कपालशतसम्पूर्णमभिरूपं भयानकम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8389"><div class="verse-block verse-justify"><div class="line">मध्याह्ने सन्ध्ययोस्तत्र नक्षत्रे रुद्रदेवते।</div><div class="line" style="position:relative;">आयुष्कामैरशुद्धैर्वा न गन्तव्यमिति स्थितिः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8390"><div class="verse-block verse-justify"><div class="line">मदन्येन न शक्यं हि निहन्तुं भूतजं भयम्।</div><div class="line" style="position:relative;">तत्रस्थोऽहं प्रजाः सर्वाः पालयामि दिनेदिने॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8391"><div class="verse-block"><div class="line">मन्नियोगाद्भूतसङ्घा न च घ्नन्तीह कञ्चन</div><div class="line pada-even">तांस्तु लोकहितार्ताय श्मशाने रमयान्महम्।</div><div class="line">एतत्ते सर्वमाख्यातं किं भूयः श्रोतुमिच्छसि ॥उमोवाच</div><div class="line pada-even" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8392"><div class="verse-block verse-justify"><div class="line">भगवन्देवदेवेश त्रिनेत्र वृषभध्वज।</div><div class="line" style="position:relative;">पिङ्गलं विकृतं भाति रूपं ते तु भयानकम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8393"><div class="verse-block verse-justify"><div class="line">भस्मदिग्धं विरूपाक्षं तीक्ष्णदंष्ट्रं जटाकुलम्।</div><div class="line" style="position:relative;">व्याघ्रोदरत्वक्संवीतं कपिलश्मश्रुसंततम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8394"><div class="verse-block verse-justify"><div class="line">रौद्रं भयानकं घोरं शूलपट्टससंयुतम्।</div><div class="line">किमर्थं त्वीदृशं रूपं तन्मे शंसितुमर्हसि ॥महेश्वर उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8395"><div class="verse-block verse-justify"><div class="line">तदहं कथयिष्यामि शृणु तत्त्वं समाहिता।</div><div class="line" style="position:relative;">द्विविधो लौकिको भावः शीतमुष्णमिति प्रिये॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8398"><div class="verse-block verse-justify"><div class="line">रौद्राकृतिं विरूपाक्षं शूलपट्टससंयुतम्।</div><div class="line" style="position:relative;">आग्नेयमिति मे रूपं देवि लोकहिते रतम्॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8402"><div class="verse-block verse-justify"><div class="line">तदहं ते प्रवक्ष्यामि शृणु कल्याणि कारणम्।</div><div class="line">पुराऽहं कारणाद्देवि कोपयुक्तः शुचिस्मिते।</div><div class="line" style="position:relative;">दक्षयज्ञवधार्याय भूतसङ्घैः समावृतः॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="13-anuśāsanaparva-b8406"><div class="verse-block"><div class="line">तदाप्रभृति सोमं वै शिरसा धारयाम्यम्</div><div class="line pada-even">एवं मे पापहानिस्तु भवेदिति मतिर्मम।</div><div class="line">तदाप्रभृति वै सोमो मूर्ध्नि संदृश्यते सदा ॥नारद उवाच</div><div class="line pada-even" style="position:relative;">॥३२॥</div></div></div>
