<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4409"><div class="verse-block verse-justify"><div class="line">दह्यमानाय विप्राय यः प्रयच्छत्युपानहौ।</div><div class="line">यत्फलं तस्य भवति तन्मे ब्रूहि पितामह ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4410"><div class="verse-block verse-justify"><div class="line">उपानहौ प्रयच्छेद्यो ब्राह्मणेभ्यः समाहितः।</div><div class="line" style="position:relative;">मर्दते कण्टकान्सर्वान्विषमान्निस्तरत्यपि॥२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4411"><div class="verse-block verse-justify"><div class="line">स शत्रूणामुपरि च सन्तिष्ठति युधिष्ठिर।</div><div class="line" style="position:relative;">यानं चाश्वतरीयुक्तं तस्य शुभ्रं विशाम्पते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4412"><div class="verse-block verse-justify"><div class="line">उपतिष्ठति कौन्तेय रौप्यकाञ्चनभूषितम्।</div><div class="line">शकटं दम्यसंयुक्तं दत्तं भवति चैव हि ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4413"><div class="verse-block verse-justify"><div class="line">यत्फलं तिलदाने च भूमिदाने च कीर्तितम्।</div><div class="line">गोदाने चान्नदाने च भूयस्तद्ब्रूहि कौरव ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4414"><div class="verse-block verse-justify"><div class="line">शृणुष्व मम कौन्तैय तिलदानस्य यत्फलम्।</div><div class="line" style="position:relative;">निशम्य च यथान्यायं प्रयच्छ कुरुसत्तम॥६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4415"><div class="verse-block verse-justify"><div class="line">पितॄणां प्रथमं भोज्यं तिलाः सृष्टाः स्वयंभुवा।</div><div class="line" style="position:relative;">तिलदानेन वै तस्मात्पितृपक्षः प्रमोदते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4416"><div class="verse-block verse-justify"><div class="line">माघमासे तिलान्यस्तु ब्राह्मणेभ्यः प्रयच्छति।</div><div class="line" style="position:relative;">सर्वसत्वसमाकीर्णं नरकं स न पश्यति॥८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4417"><div class="verse-block verse-justify"><div class="line">सर्वसत्रैश्च यजते यस्तिलैर्यजते पितॄन्।</div><div class="line" style="position:relative;">न चाकामेन दातव्यं तिलैः श्राद्धं कदाचन॥९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4418"><div class="verse-block verse-justify"><div class="line">महर्षेः कश्यपस्यैते गात्रेभ्यः प्रसृतास्तिलाः।</div><div class="line" style="position:relative;">ततो दिव्यं गता भावं प्रदानेषु तिलाः प्रभो॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4419"><div class="verse-block verse-justify"><div class="line">पौष्टिका रूपदाश्चैव तथा पापविनाशनाः।</div><div class="line" style="position:relative;">तस्मात्सर्वप्रदानेभ्यस्तिलदानं विशिष्यते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4420"><div class="verse-block verse-justify"><div class="line">आपस्तम्बश्च मेधावी शङ्खश्च लिखितस्तथा।</div><div class="line" style="position:relative;">महर्षिर्गौतमश्चापि तिलदानैर्दिवं गताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4421"><div class="verse-block verse-justify"><div class="line">तिलहोमरता विप्राः सर्वे संयतमैथुनाः।</div><div class="line" style="position:relative;">समा गव्येन हविषा प्रवृत्तिषु च संस्थिताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4422"><div class="verse-block verse-justify"><div class="line">सर्वेषामिति दानानां तिलदानं विशिष्यते।</div><div class="line" style="position:relative;">अक्षयं सर्वदानानां तिलदानमिहोच्यते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4423"><div class="verse-block verse-justify"><div class="line">उच्छिन्ने तु पुरा हव्ये कुशिकर्षिः परन्तपः।</div><div class="line" style="position:relative;">तिलैरग्नित्रयं हुत्वा प्राप्तवान्गतिमुत्तमाम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4424"><div class="verse-block verse-justify"><div class="line">इति प्रोक्तं कुरुश्रेष्ठ तिलदानमनुत्तमम्।</div><div class="line" style="position:relative;">विधानं येन विधिना तिलानामिह शस्यते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4425"><div class="verse-block verse-justify"><div class="line">अत ऊर्ध्वं निबोधेदं देवानां यष्टुमिच्छताम्।</div><div class="line" style="position:relative;">समागमे महाराज ब्रह्मणा वै स्वयंभुवा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4426"><div class="verse-block verse-justify"><div class="line">देवाः समेत्य ब्रह्माणं भूमिभागे यियश्रवः।</div><div class="line">शुभं देशमयाचन्त यजेम इति पार्थिव ॥देवा ऊचुः।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4427"><div class="verse-block verse-justify"><div class="line">भगवंस्त्वं प्रभुर्भूमेः सर्वस्य त्रिदिवस्य च।</div><div class="line" style="position:relative;">यजेम हि महाभाग यज्ञं भवदनुज्ञया॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4428"><div class="verse-block"><div class="line">नाननुज्ञातभूमिर्हि यज्ञस्य फलमश्नुते</div><div class="line pada-even">त्वं हि सर्वस्य जगतः स्थावरस्य चरस्य च।</div><div class="line">प्रभुर्भवसि तस्मात्त्वं समनुज्ञातुमर्हसि ॥ब्रह्मोवाच</div><div class="line pada-even" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4429"><div class="verse-block verse-justify"><div class="line">ददानि मेदिनीभागं भवद्भ्योऽहं सुरर्षभाः।</div><div class="line">यस्मिन्देशे करिष्यध्वं यज्ञान्काश्यपनन्दनाः ॥दैवा ऊचुः।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4430"><div class="verse-block verse-justify"><div class="line">भगवन्कृतकामाः स्म यक्ष्महे स्वाप्तदक्षिणैः।</div><div class="line" style="position:relative;">इमं तु देशं मुनयः पर्युपासन्ति नित्यदा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4431"><div class="verse-block verse-justify"><div class="line">ततोऽगस्त्यश्च कण्वश्च भृगुरत्रिर्वृषाकपिः।</div><div class="line" style="position:relative;">असितो देवलश्चैव देवयज्ञमुपागमन्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4432"><div class="verse-block verse-justify"><div class="line">ततो देवा महात्मान ईजिरे यज्ञमच्युतम्।</div><div class="line" style="position:relative;">तथा समापयामासुर्यथाकालं सुरर्षभाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4433"><div class="verse-block verse-justify"><div class="line">त इष्टयज्ञास्त्रिदशा हिमवत्यचलोत्तमे।</div><div class="line">षष्ठमंशं क्रतोस्तस्य भूमिदानं प्रचक्रिरे।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4434"><div class="verse-block verse-justify"><div class="line">प्रादेशमात्रं भूमेस्तु यो दद्यादनुपस्कृतम्।</div><div class="line" style="position:relative;">न सीदति स कृच्छ्रेषु न च दुर्गाण्यवाप्नुते॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4435"><div class="verse-block verse-justify"><div class="line">शीतवातातपसहां यागभूमिं सुसंस्कृताम्।</div><div class="line" style="position:relative;">प्रदाय सुरलोकस्थः पुण्यान्तेऽपि न चाल्यते॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4436"><div class="verse-block verse-justify"><div class="line">मुदितो वसति प्राज्ञः शक्रेण सह पार्थिव।</div><div class="line" style="position:relative;">पतिश्रयप्रदानाच्च सोऽपि स्वर्गे महीयते॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4437"><div class="verse-block verse-justify"><div class="line">अध्यापककुले जातः श्रोत्रियो नियतेन्द्रियः।</div><div class="line" style="position:relative;">गृहे यस्य वसेत्तुष्टः प्रधानं लोकमश्नुते॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4438"><div class="verse-block verse-justify"><div class="line">तथा गवार्थे शरणं शीतवर्षसहं दृढम्।</div><div class="line" style="position:relative;">आसप्तमं तारयति कुलं भरतसत्तम॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4439"><div class="verse-block verse-justify"><div class="line">क्षेत्रभूमिं ददल्लोके शुभां श्रियमवाप्नुयात्।</div><div class="line" style="position:relative;">रत्नभूमिं प्रदद्यात्तु कुलवंशं प्रवर्धयेत्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4440"><div class="verse-block verse-justify"><div class="line">न चोषरां न निर्दग्धां महीं दद्यात्कथञ्चन।</div><div class="line" style="position:relative;">न श्मशानपरीतां च न च पापनिषेविताम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4441"><div class="verse-block verse-justify"><div class="line">पारक्ये भूमिदेशे तु पितॄणां निर्वपेत्तु यः।</div><div class="line" style="position:relative;">तद्भूमिं वाऽपि पितृभिः श्राद्धकर्म विहन्यते॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4442"><div class="verse-block verse-justify"><div class="line">तस्मात्क्रीत्वा महीं दद्यात्स्वल्पामपि विचक्षणः।</div><div class="line" style="position:relative;">पिण्डः पितृभ्यो दत्तो वै तस्यां भवति शाश्वतः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4443"><div class="verse-block verse-justify"><div class="line">अटवी पर्वताश्चैव नद्यस्तीर्थानि यानि च।</div><div class="line" style="position:relative;">सर्वाण्यस्वामिकान्याद्दुर्न हि तत्र परिग्रहः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4444"><div class="verse-block verse-justify"><div class="line">इत्येतद्भूमिदानस्य फलमुक्तं विशाम्पते।</div><div class="line" style="position:relative;">अतः परं तु गोदानं कीर्तयिष्यामि तेऽनघ॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4445"><div class="verse-block verse-justify"><div class="line">गावोऽधिकास्तपस्विभ्यो यस्मात्सर्वेभ्य एव च।</div><div class="line" style="position:relative;">तस्मान्महेश्वरो देवस्तपस्ताभिः सहास्थितः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4446"><div class="verse-block verse-justify"><div class="line">ब्राह्मे लोके वसन्त्येताः सोमेन सह भारत।</div><div class="line" style="position:relative;">यां तां ब्रह्मर्षयः सिद्धाः प्रार्थयन्ति परां गतिम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4447"><div class="verse-block verse-justify"><div class="line">पयसा हविषा दध्ना शकृता चाथ चर्मणा।</div><div class="line" style="position:relative;">अस्थिभिश्चोपकुर्वन्ति शृङ्गैर्वालैश्च भारत॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4448"><div class="verse-block verse-justify"><div class="line">नासां शीतातपौ स्यातां सदैताः कर्म कुर्वते।</div><div class="line" style="position:relative;">न वर्षविषयं वाऽपि दुःखमासां भवत्युत॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4449"><div class="verse-block verse-justify"><div class="line">ब्राह्मणैः सहिता यान्ति तस्मात्पारमकं पदम्।</div><div class="line" style="position:relative;">एकं गोब्राह्मणं तस्मात्प्रवदन्ति मनीषिणः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4450"><div class="verse-block verse-justify"><div class="line">रन्तिदेवस्य यज्ञे ताः पशुत्वेनोपकल्पिताः।</div><div class="line">अतश्चर्मण्वती राजन्गोचर्मभ्यः प्रवर्तिता।</div><div class="line" style="position:relative;">पशुत्वाच्च विनिर्मुक्ताः प्रदानायोपकल्पिताः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4451"><div class="verse-block verse-justify"><div class="line">ता इमा विप्रमुख्येभ्यो यो ददाति महीपते।</div><div class="line" style="position:relative;">निस्तरेदापदं कृच्छ्रां विषमस्थोऽपि पार्थिव॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="13-anuśāsanaparva-b4452"><div class="verse-block verse-justify"><div class="line">गवां सहस्रदः प्रेत्य नरकं न प्रपद्यते।</div><div class="line" style="position:relative;">सर्वत्र विजयं चापि लभते मनुजाधिप॥४४॥</div></div></div>
