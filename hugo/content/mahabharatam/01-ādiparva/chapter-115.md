<div class="verse-block-wrapper" id="01-ādiparva-b5225"><div class="verse-block verse-justify"><div class="line">`ततः सुप्तजनप्रायेऽर्धरात्रे भगवानृषिः।</div><div class="line" style="position:relative;">दीप्यमानेषु दीपेषु शरणं प्रविवेश ह॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5226"><div class="verse-block verse-justify"><div class="line">ततोऽम्बिकायां प्रथमं नियुक्तः सत्यवागृषिः।</div><div class="line" style="position:relative;">जगाम तस्याः शयनं विपुले तपसि स्थितः॥६॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5228"><div class="verse-block verse-justify"><div class="line">विरूपो हि जटी चापि दुर्वर्णः परुषः कृशः।</div><div class="line" style="position:relative;">सुगन्धेतरगन्धश्च सर्वथा दुष्प्रधर्षणः ॥'॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5229"><div class="verse-block verse-justify"><div class="line">तस्य कृष्णस्य कपिलां जटां दीप्ते च लोचने।</div><div class="line" style="position:relative;">बब्रूणि चैव श्मश्रूमि दृष्ट्वा देवी न्यमीलयत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5230"><div class="verse-block verse-justify"><div class="line">संभूव तया सार्धं मातुः प्रियचिकीर्षया।</div><div class="line" style="position:relative;">भयात्काशिसुता तं तु नाशक्नोदभिवीक्षितुम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5231"><div class="verse-block verse-justify"><div class="line">ततो निष्क्रान्तमागम्य माता पुत्रमुवाच ह।</div><div class="line" style="position:relative;">अप्यस्यां गुणवान्पुत्र राजपुत्रो भविष्यति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5232"><div class="verse-block verse-justify"><div class="line">निशम्य तद्वचो मातुर्व्यासः सत्यवतीसुतः।</div><div class="line" style="position:relative;">`प्रोवाचातीन्द्रियज्ञानो विधिना संप्रचोदितः ॥'॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5233"><div class="verse-block verse-justify"><div class="line">नागायुतसमप्राणो विद्वान्राजर्षिसत्तमः।</div><div class="line" style="position:relative;">महाभागो महावीर्यो महाबुद्धिर्भविष्यति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5234"><div class="verse-block verse-justify"><div class="line">तस्य चापि शतं पुत्रा भविष्यन्ति महात्मनः।</div><div class="line" style="position:relative;">किंतु मातुः स वैगुण्यादन्ध एव भविष्यति॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5235"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा माता पुत्रमथाऽब्रवीत्।</div><div class="line" style="position:relative;">नान्धः कुरूणां नृपतिरनुरूपस्तपोधन॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5236"><div class="verse-block verse-justify"><div class="line">ज्ञातिवंशस्य गोप्तारं पितॄणां वंशवर्धनम्।</div><div class="line" style="position:relative;">`अपरस्यामपि पुनर्मम शोकविनाशनम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5237"><div class="verse-block verse-justify"><div class="line">तस्मादवरजं पुत्रं जनयान्यं नराधिपम्।</div><div class="line" style="position:relative;">भ्रातुर्भार्याऽवरा चेयं रूपयौवनशालिनी॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5238"><div class="verse-block verse-justify"><div class="line">अस्यामुत्पादयाऽपत्यं मन्नियोगाद्गुणाधिकम्।</div><div class="line" style="position:relative;">'द्वितीयं कुरुवंशस्य राजानं दातुमर्हसि॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5239"><div class="verse-block verse-justify"><div class="line">स तथेति प्रतिज्ञाय निश्चक्राम महायशाः।</div><div class="line" style="position:relative;">साऽपि कालेन कौसल्या सुषुवेऽन्धं तमात्मजम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5240"><div class="verse-block verse-justify"><div class="line">पुनरेव तु सा देवी परिभाष्य स्नुषां ततः।</div><div class="line" style="position:relative;">ऋषिमावाहयत्सत्या यथापूर्वमरिन्दम॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5241"><div class="verse-block verse-justify"><div class="line">`अम्बालिकां समाहूय तस्यां सत्यवती सुतम्।</div><div class="line" style="position:relative;">भूयो नियोजयामास सन्तानाय कुलस्य वै॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5242"><div class="verse-block verse-justify"><div class="line">विषण्णाम्बालिका साध्वी निषण्णा शयनोत्तमे।</div><div class="line" style="position:relative;">कोन्वेष्यतीति ध्यायन्ती नियतां संप्रतीक्षते'॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5243"><div class="verse-block verse-justify"><div class="line">ततस्तेनैव विधिना महर्षिस्तामपद्यत।</div><div class="line" style="position:relative;">अम्बालिकामथाऽभ्यागादृषिं दृष्ट्वा च साऽपि तम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5244"><div class="verse-block verse-justify"><div class="line">विवर्णा पाण्डुसंकाशा समपद्यत भारत।</div><div class="line" style="position:relative;">तां भीतां पाण्डुसंकाशां विषण्णां प्रेक्ष्य भारत॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5245"><div class="verse-block verse-justify"><div class="line">व्यासः सत्यवतीपुत्र इदं वचनमब्रवीत्।</div><div class="line" style="position:relative;">यस्मात्पाण्डुत्वमापन्ना विरूपं प्रेक्ष्य मामिह॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5246"><div class="verse-block verse-justify"><div class="line">तस्मादेष सुतस्ते वै पाण्डुरेव भविष्यति।</div><div class="line" style="position:relative;">नाम चास्यैतदेवेह भविष्यति शुभानने॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5247"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा स निराक्रामद्भगवानृषिसत्तमः।</div><div class="line" style="position:relative;">ततो निष्क्रान्तमालोक्य सत्या पुत्रमथाऽब्रवीत्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5248"><div class="verse-block verse-justify"><div class="line">`कुमारो ब्रूहि मे पुत्र अप्यत्र भविता शुभः।</div><div class="line" style="position:relative;">'शशंस स पुनर्मात्रे तस्य बालस्य पाण्डुताम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5249"><div class="verse-block verse-justify"><div class="line">`भविष्यति सुविक्रान्तः कुमारो दिक्षु विश्रुतः।</div><div class="line" style="position:relative;">पाण्डुत्वं वर्णतस्तस्य मातृदोषाद्भविष्यति॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5250"><div class="verse-block verse-justify"><div class="line">तस्य पुत्रा महेष्वासा भविष्यन्तीह पञ्च वै।</div><div class="line" style="position:relative;">इत्युक्त्वा मातरं तत्र सोऽभिवाद्य जगाम ह ॥'॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5251"><div class="verse-block verse-justify"><div class="line">तं माता पुनरेवान्यमेकं पुत्रमयाचत।</div><div class="line" style="position:relative;">तथेति च महर्षिस्तां मातरं प्रत्यभाषत॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5252"><div class="verse-block verse-justify"><div class="line">ततः कुमारं सा देवी प्राप्तकालमजीजनत्।</div><div class="line" style="position:relative;">पाण्डुलक्षणसंपन्नं दीप्यमानमिव श्रिया॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5253"><div class="verse-block verse-justify"><div class="line">यस्य पुत्रा महेष्वासा जज्ञिरे पञ्च पाण्डवाः।</div><div class="line" style="position:relative;">`तयोर्जन्मक्रियाः सर्वा यथावदनुपूर्वशः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5254"><div class="verse-block verse-justify"><div class="line">कारयामास वै भीष्मो ब्राह्मणैर्वेदपारगैः।</div><div class="line" style="position:relative;">अन्धं दृष्ट्वाऽम्बिकापुत्रं जातं सत्यवती सुतम्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5255"><div class="verse-block verse-justify"><div class="line">कौसल्यार्थे समाहूय पुत्रमन्यमयाचत।</div><div class="line" style="position:relative;">अन्धोयमन्यमिच्छामि कौसल्यातनयं शुभम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5256"><div class="verse-block verse-justify"><div class="line">एवमुक्तो महर्षिस्तां मातरं प्रत्यभाषत।</div><div class="line" style="position:relative;">नियता यदि कौसल्या भविष्यति पुनःशुभा॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5257"><div class="verse-block verse-justify"><div class="line">भविष्यति कुमारोऽस्या धर्मशास्त्रार्थतत्ववित्।</div><div class="line" style="position:relative;">तां समाधाय वै भूयः स्नुषां सत्यवती पुनः ॥'॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5258"><div class="verse-block verse-justify"><div class="line">ऋतुकाले ततो ज्येष्ठां वधूं तस्मै न्ययोजयत्।</div><div class="line" style="position:relative;">सा तु रूपं च गन्धं च महर्षेः प्रविचिन्त्य तं॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5259"><div class="verse-block verse-justify"><div class="line">नाकरोद्वचनं देव्या भयात्सुरसुतोपमा।</div><div class="line" style="position:relative;">ततःस्वैर्भूषणैर्दासीं भूषयित्वाऽप्सरोपमाम्॥३९॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5262"><div class="verse-block verse-justify"><div class="line">कामोपभोगेन रहस्तस्यां तुष्टिमगादृषिः।</div><div class="line" style="position:relative;">तया सहोषितो राजन्महर्षिः संशितव्रतः॥४२॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5266"><div class="verse-block verse-justify"><div class="line">कृष्णद्वैपायनोऽप्येतत्सत्यवत्यै न्यवेदयत्।</div><div class="line" style="position:relative;">प्रलम्भमात्मनश्चैव शूद्रायाः पुत्रजन्म च॥४६॥</div></div></div>
