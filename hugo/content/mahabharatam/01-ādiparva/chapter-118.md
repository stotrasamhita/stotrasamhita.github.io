<div class="verse-block-wrapper" id="01-ādiparva-b5313"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5314"><div class="verse-block verse-justify"><div class="line">`धृतराष्ट्रे च पाण्डौ च विदुरे च महात्मनि।</div><div class="line">'एषु त्रिषु कुमारेषु जातेषु कुरुजाङ्गलम्।</div><div class="line" style="position:relative;">कुरवोऽथ कुरुक्षेत्रं त्रयमेतदवर्धत॥२॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5316"><div class="verse-block verse-justify"><div class="line">वाहनानि प्रहृष्टानि मुदिता मृगपक्षिणः।</div><div class="line" style="position:relative;">गन्धवन्ति च माल्यानि रसवन्ति फलानि च॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5317"><div class="verse-block verse-justify"><div class="line">वणिग्भिश्चान्वकीर्यन्त नगराण्यथ शिल्पिभिः।</div><div class="line" style="position:relative;">शूराश्च कृतविद्याश्च सन्तश्च सुखिनोऽभवन्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5318"><div class="verse-block verse-justify"><div class="line">नाभवन्दस्यवः केचिन्नाधर्मरुचयो जनाः।</div><div class="line" style="position:relative;">प्रदेशेष्वपि राष्ट्राणां कृतं युगमवर्तत॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5319"><div class="verse-block verse-justify"><div class="line">धर्मक्रिया यज्ञशीलाः सत्यव्रतपरायणाः।</div><div class="line" style="position:relative;">अन्योन्यप्रीतिसंयुक्ता व्यवर्धन्त प्रजास्तदा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5320"><div class="verse-block verse-justify"><div class="line">मानक्रोधविहीनाश्च नरा लोभविवर्जिताः।</div><div class="line" style="position:relative;">अन्योन्यमभ्यनन्दन्त धर्मोत्तरमवर्तत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5321"><div class="verse-block verse-justify"><div class="line">तन्महोदधिवत्पूर्णं नगरं वै व्यरोचत।</div><div class="line" style="position:relative;">द्वारतोरणनिर्यूहैर्युक्तमभ्रचयोपमैः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5322"><div class="verse-block verse-justify"><div class="line">प्रसादशतसंबाधं महेन्द्रपुरसन्निभम्।</div><div class="line">नदीषु वनखण्डेषु वापीपल्वलसानुषु।</div><div class="line" style="position:relative;">काननेषु च रम्येषु विजह्रुर्मुदिता जनाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5323"><div class="verse-block verse-justify"><div class="line">उत्तरैः कुरुभिः सार्धं दक्षिणाः कुरवस्तथा।</div><div class="line" style="position:relative;">विस्पर्धमाना व्यचरंस्तथा देवर्षिचारणैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5324"><div class="verse-block verse-justify"><div class="line">नाभवत्कृपणः कश्चिन्नाभवन्विधवाः स्त्रियः।</div><div class="line" style="position:relative;">तस्मिञ्जनपदे रम्ये कुरुभिर्बहुलीकृते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5325"><div class="verse-block verse-justify"><div class="line">कूपारामसभावाप्यो ब्राह्मणावसथास्तथा।</div><div class="line" style="position:relative;">बभूवुः सर्वर्द्धियुतास्तस्मिन्राष्ट्रे सदोत्सवाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5326"><div class="verse-block verse-justify"><div class="line">भीष्मेण धर्मतो राजन्सर्वतः परिरक्षिते।</div><div class="line" style="position:relative;">बभूव रमणीयश्च चैत्ययूपशताङ्कितः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5327"><div class="verse-block verse-justify"><div class="line">स देशः परराष्ट्राणि विमृज्याभिप्रवर्धितः।</div><div class="line" style="position:relative;">भीष्मेण विहितं राष्ट्रे धर्मचक्रमवर्तत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5328"><div class="verse-block verse-justify"><div class="line">क्रियमाणेषु कृत्येषु कुमाराणां महात्मनाम्।</div><div class="line" style="position:relative;">पौरजानपदाः सर्वे बभूवुः परमोत्सुकाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5329"><div class="verse-block verse-justify"><div class="line">गृहेषु कुरुमुख्यानां पौराणां च नराधिप।</div><div class="line" style="position:relative;">दीयतां भुज्यतां चेति वाचोऽश्रूयन्त सर्वशः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5330"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्रश्च पाण्डुश्च विदुरश्च महामतिः।</div><div class="line" style="position:relative;">जन्मप्रभृति भीष्मेण पुत्रवत्परिपालिताः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5331"><div class="verse-block verse-justify"><div class="line">संस्कारैः संस्कृतास्ते तु व्रताध्ययनसंयुताः।</div><div class="line" style="position:relative;">श्रमव्यायामकुशलाः समपद्यन्त यौवनम्॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5333"><div class="verse-block verse-justify"><div class="line">इतिहासपुराणेषु नानाशिक्षासु बोधिताः।</div><div class="line" style="position:relative;">वेदवेदाङ्गतत्त्वज्ञाः सर्वत्र कृतनिश्चयाः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5334"><div class="verse-block verse-justify"><div class="line">`वैदिकाध्ययने युक्तो नीतिशास्त्रेषु पारगः।</div><div class="line" style="position:relative;">भीष्मेण राजा कौरव्यो धृतराष्ट्रोऽभिषेचितः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5335"><div class="verse-block verse-justify"><div class="line">धनुर्वेदेऽश्वपृष्ठे च गदायुद्धेऽसिचर्मणि।</div><div class="line" style="position:relative;">तथैव गजशिक्षायामस्त्रेषु विविधेषु च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5336"><div class="verse-block verse-justify"><div class="line">अर्थधर्मप्रधानासु विद्यासु विविधासु च।</div><div class="line" style="position:relative;">गतः पारं यदा पाण्डुस्तदा सेनापतिः कृतः ॥'॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5337"><div class="verse-block verse-justify"><div class="line">पाण्डुर्धनुषि विक्रान्तो नरेष्वभ्यदिकोऽभवत्।</div><div class="line" style="position:relative;">अन्येभ्यो बलवानासीद्धृतराष्ट्रो महीपतिः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5338"><div class="verse-block verse-justify"><div class="line">अमात्यो मनुजेन्द्रस्य बाल एव यशस्विनः।</div><div class="line" style="position:relative;">भीष्मेण सर्वधर्माणां प्रणेता विदुरः कृतः॥२६॥</div></div></div>
