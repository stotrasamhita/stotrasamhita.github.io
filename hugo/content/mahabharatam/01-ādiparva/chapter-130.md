<div class="verse-block-wrapper" id="01-ādiparva-b5852"><div class="verse-block verse-justify"><div class="line">जनमेजय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5853"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्रस्य पुत्राणामादितः कथितं त्वया।</div><div class="line" style="position:relative;">ऋषेः प्रसादात्तु शतं न च कन्या प्रकीर्तिता॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5854"><div class="verse-block verse-justify"><div class="line">वैश्यापुत्रो युयुत्सुश्च कन्या चैका शताधिका।</div><div class="line" style="position:relative;">गान्धारराजदुहिता शतपुत्रेति चानघ॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5855"><div class="verse-block verse-justify"><div class="line">उक्ता महर्षिणा तेन व्यासेनामिततेजसा।</div><div class="line" style="position:relative;">कथं त्विदानीं भगवन्कन्यां त्वं तु ब्रवीषि मे॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5856"><div class="verse-block verse-justify"><div class="line">यदि भागशतं पेशी कृता तेन महर्षिणा।</div><div class="line" style="position:relative;">न प्रजास्यति चेद्भूयः सौबलेयी कथंचन॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5857"><div class="verse-block verse-justify"><div class="line">कथं तु संभवस्तस्या दुःशलाया वदस्व मे।</div><div class="line">यथावदिह विप्रर्षे परं मेऽत्र कुतूहलम् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5858"><div class="verse-block verse-justify"><div class="line">साध्वयं प्रश्न उद्दिष्टः पाण्डवेय ब्रवीमि ते।</div><div class="line" style="position:relative;">तां मांसपेशीं भगवान्स्वयमेव महातपाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5859"><div class="verse-block verse-justify"><div class="line">शीताभिरद्भिरासिच्य भागं भागमकल्पयत्।</div><div class="line" style="position:relative;">यो यथा कल्पितो भागस्तंत धात्र्या तथा नृप॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5860"><div class="verse-block verse-justify"><div class="line">घृतपूर्णेषु कुण्डेषु एकैकं प्राक्षिपत्तदा।</div><div class="line" style="position:relative;">एतस्मिन्नन्तरे साध्वी गान्धारी सुदृढव्रता॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5861"><div class="verse-block verse-justify"><div class="line">दुहितुः स्नेहसंयोगमनुध्याय वराङ्गना।</div><div class="line">`नाब्रवीत्तमृषिं किंचिद्गौरवाच्च यशस्विनी।</div><div class="line" style="position:relative;">'मनसा चिन्तयद्देवी एतत्पुत्रशतं मम॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5862"><div class="verse-block verse-justify"><div class="line">भविष्यति न संदेहो न ब्रवीत्यन्यथा मुनिः।</div><div class="line" style="position:relative;">ममेयं परमा तुष्टिर्दुहिता मे भवेद्यदि॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5863"><div class="verse-block verse-justify"><div class="line">एका शताधिका बाला भविष्यति कनीयसी।</div><div class="line" style="position:relative;">ततो दौहित्रजाल्लोकादबाह्योऽसौ पतिर्मम॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5864"><div class="verse-block verse-justify"><div class="line">अधिका किल नारीणां प्रीतिर्जामातृजा भवेत्।</div><div class="line" style="position:relative;">यदि नाम ममापि स्याद्दुहितैका शताधिका॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5865"><div class="verse-block verse-justify"><div class="line">कृतकृत्या भवेयं वै पुत्रदौहित्रसंवृता।</div><div class="line" style="position:relative;">यदि सत्यं तपस्तप्तं दत्तं वाऽप्यथवा हुतम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5866"><div class="verse-block verse-justify"><div class="line">गुरवस्तोषिता वापि तथाऽस्तु दुहिता मम।</div><div class="line" style="position:relative;">एतस्मिन्नेव काले तु कृष्णद्वैपायनः स्वयम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5867"><div class="verse-block verse-justify"><div class="line">व्यभजत्स तदा पेशीं भगवानृषिसत्तमः।</div><div class="line" style="position:relative;">`गण्यमानेषु कुण्डेषु शते पूर्णे महात्मना॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5868"><div class="verse-block verse-justify"><div class="line">अभवच्चापरं खण्डं वामहस्ते तदा किल।</div><div class="line">'गणयित्वा शतं पूर्णमंशानामाह सौबलीम् ॥व्यास उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5869"><div class="verse-block verse-justify"><div class="line">पूर्णं पुत्रशतं त्वेतन्न मिथ्या वागुदाहृता।</div><div class="line" style="position:relative;">दैवयोगाच्च भागैकः परिशिष्टः शतात्परः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5870"><div class="verse-block verse-justify"><div class="line">एषा ते सुभगा कन्या भविष्यति यतेप्सिता।</div><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">ततोऽन्यं घृतकुम्भं च समानाय्य महातपाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5871"><div class="verse-block verse-justify"><div class="line">तं चापि प्राक्षिपत्तत्र कन्याभागं तपोधनः।</div><div class="line" style="position:relative;">`संभूता चैव कालेन सर्वेषां च यवीयसी॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5872"><div class="verse-block verse-justify"><div class="line">ऐतत्ते कथितं राजन्दुःशलाजन्म भारत।</div><div class="line" style="position:relative;">ब्रूहि राजेन्द्र किं भूयो वर्तयिष्यामि तेऽनघ॥२१॥</div></div></div></div>
