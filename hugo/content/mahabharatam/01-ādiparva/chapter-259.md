<div class="verse-block-wrapper" id="01-ādiparva-b11411"><div class="verse-block verse-justify"><div class="line">स तप्यमानः पुत्रार्थे लपितामिदमब्रवीत्।</div><div class="line" style="position:relative;">कथं नु शक्ताः शरणे लपिते मम पुत्रकाः॥३॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b11413"><div class="verse-block verse-justify"><div class="line">कथं त्वशक्ता त्राणाय माता तेषां तपस्विनी।</div><div class="line" style="position:relative;">भविष्यति हि शोकार्ता पुत्रत्राणमपश्यती॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11414"><div class="verse-block verse-justify"><div class="line">कथमुड्डीयनेऽशक्तान्पतने च ममात्मजान्।</div><div class="line" style="position:relative;">सन्तप्यमाना बहुधा वाशमाना प्रधावती॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11415"><div class="verse-block verse-justify"><div class="line">जरितारिः कथं पुत्रः सारिसृक्कः कथं च मे।</div><div class="line" style="position:relative;">स्तम्बमित्रः कथं द्रोणः कथं सा च तपस्विनी॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11416"><div class="verse-block verse-justify"><div class="line">लालप्यमानं तमृषइं मन्दपालं तथा वने।</div><div class="line" style="position:relative;">लपिता प्रत्युवाचेदं सासूयमिव भारत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11417"><div class="verse-block verse-justify"><div class="line">न ते पुत्रेष्ववेक्षाऽस्ति यानृषीनुक्तवानसि।</div><div class="line" style="position:relative;">तेजस्विनो वीर्यवन्तो न तेषां ज्वलनाद्भयम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11418"><div class="verse-block verse-justify"><div class="line">त्वयाऽग्नौ ते परीताश्च स्वयं हि मम सन्निधौ।</div><div class="line" style="position:relative;">श्रुतं तथा चेति ज्वलनेन महात्मना॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11419"><div class="verse-block verse-justify"><div class="line">पलो न तां वाचमुक्त्वा मिथ्या करिष्यति।</div><div class="line" style="position:relative;">न्धुकृत्ये न तेन ते स्वस्थ मानसम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11420"><div class="verse-block verse-justify"><div class="line">तामेव तु ममामित्रां चिन्तयन्परितप्यसे।</div><div class="line" style="position:relative;">ध्रुवं मयि न ते स्नेहो यथा तपयं पुराऽभवत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11421"><div class="verse-block verse-justify"><div class="line">नहि पक्षवता न्याय्यं निः हेन सुहृज्जने।</div><div class="line" style="position:relative;">पीड्यमान उपद्रष्टुं शक्तेना मा कथंचन॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11422"><div class="verse-block verse-justify"><div class="line">गच्छ त्वं जरितामेव यदर्थं परितप्यसे।</div><div class="line">चरिष्याम्यहमप्येका यथा पुरुषाश्रिता ॥मन्दपाल उवाच।</div><div class="line" style="position:relative;">॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11423"><div class="verse-block verse-justify"><div class="line">नाहमेवं चरे लोके यथा त्वमभिमन्यसे।</div><div class="line" style="position:relative;">अपत्यहेतोर्विचरे तच्च कृच्छ्रगतं मम॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11424"><div class="verse-block verse-justify"><div class="line">भूतं हित्वा च भाव्यर्थे योऽवलम्बेत्स मन्दधीः।</div><div class="line" style="position:relative;">अवमन्येत तं लोको यथेच्छसि तथा कुरु॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11425"><div class="verse-block verse-justify"><div class="line">एष हि प्रज्वलन्नग्निर्लेलिहानी महीरुहान्।</div><div class="line">आविग्ने हृदि सन्तापं जनयत्यशिवं मम ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b11427"><div class="verse-block verse-justify"><div class="line">तस्माद्देशादतिक्रान्ते ज्वलने जरिता पुनः।</div><div class="line" style="position:relative;">जगाम पुत्रकानेन जरिता पुत्रगृद्धिनी॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b11429"><div class="verse-block verse-justify"><div class="line">अश्रूणि मुमुचे तेषां दर्शनात्सा पुनःपुनः।</div><div class="line" style="position:relative;">`न श्रद्धेयं ततस्तेषांर्शनं वै पुनःपुनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11430"><div class="verse-block verse-justify"><div class="line">इति मत्वाऽब्रवीद्वाकजरिता पुत्रगृद्धिनी।</div><div class="line" style="position:relative;">'एकाकशश्च पुत्रांस्तन्त्र्शमानान्वपद्यत॥२२॥</div></div></div>
