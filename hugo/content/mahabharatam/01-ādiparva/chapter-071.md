<div class="verse-block-wrapper" id="01-ādiparva-b3284"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3285"><div class="verse-block verse-justify"><div class="line">समावृतव्रतं तं तु विसृष्टं गुरुणा कचम्।</div><div class="line" style="position:relative;">प्रस्थितं त्रिदशावासं देवयान्यब्रवीदिदम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3286"><div class="verse-block verse-justify"><div class="line">ऋषेरङ्गिरसः पौत्र वृत्तेनाभिजनेन च।</div><div class="line" style="position:relative;">भ्राजसे विद्यया चैव तपसा च दमेन च॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3287"><div class="verse-block verse-justify"><div class="line">ऋषिर्यथाङ्गिरा मान्यः पितुर्मम महायशाः।</div><div class="line" style="position:relative;">तथा प्रान्यश्च पूज्यश्च मम भूयो बृहस्पतिः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3288"><div class="verse-block verse-justify"><div class="line">एवं ज्ञात्वा विजानीहि यद्ब्रवीमि तपोधन।</div><div class="line" style="position:relative;">व्रतस्थे नियमोपेते यथा वर्ताम्यहं त्वयि॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3289"><div class="verse-block verse-justify"><div class="line">स समावृतविद्यो मां भक्तां भजितुमर्हसि।</div><div class="line">गृहाण पाणिं विधिवन्मम मन्त्रपुरस्कृतम् ॥कच उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3290"><div class="verse-block verse-justify"><div class="line">पूज्यो मान्यश्च भगवान्यथा तव पिता मम।</div><div class="line" style="position:relative;">तथा त्वमनवद्याङ्गि पूजनीयतरा मम॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3291"><div class="verse-block verse-justify"><div class="line">प्राणेभ्योऽपि प्रियतरा भार्गवस्य महात्मनः।</div><div class="line" style="position:relative;">त्वं भत्रे धर्मतः पूज्या गुरुपुत्री सदा मम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3292"><div class="verse-block verse-justify"><div class="line">यथा मम गुरुर्नित्यं मान्यः शुक्रः पिता तव।</div><div class="line">देवयानि तथैव त्वं नैवं मां वक्तुमर्हसि ॥देवयान्युवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3293"><div class="verse-block verse-justify"><div class="line">गुरुपुत्रस्य पुत्रो वै न त्वं पुत्रश्च मे पितुः।</div><div class="line" style="position:relative;">तस्मात्पूज्यश्च मान्यश्च ममापि त्वं द्विजोत्तम॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3294"><div class="verse-block verse-justify"><div class="line">असुरैर्हन्यमाने च कच त्वयि पुनःपुनः।</div><div class="line" style="position:relative;">तदाप्रभृति या प्रीतिस्तां त्वमद्य स्मरस्व मे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3295"><div class="verse-block verse-justify"><div class="line">सौहार्दे चानुरागे च वेत्थ मे भक्तिमुत्तमाम्।</div><div class="line">न मामर्हसि धर्मज्ञ त्यक्तुं भक्तामनागसम् ॥कच उवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3296"><div class="verse-block verse-justify"><div class="line">अनियोज्ये नियोक्तुं मां देवयानि न चार्हसि।</div><div class="line" style="position:relative;">प्रसीद सुभ्रु त्वं मह्यं गुरोर्गुरुतरा शुभे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3297"><div class="verse-block verse-justify"><div class="line">यत्रोषितं विशालाक्षि त्वया चन्द्रनिभानने।</div><div class="line" style="position:relative;">तत्राहमुषितो भद्रे कुक्षौ काव्यस्य भामिनि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3298"><div class="verse-block verse-justify"><div class="line">भगिनी धर्मतो मे त्वं मैवं वोचः सुमध्यमे।</div><div class="line" style="position:relative;">सुखमस्म्युषितो भद्रे न मन्युर्विद्यते मम॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3299"><div class="verse-block"><div class="line">आपृच्छे त्वां गमिष्यामि शिवमाशंस मे पथि</div><div class="line pada-even">अविरोधेन धर्मस्य स्मर्तव्योऽस्मि कथान्तरे।</div><div class="line">अप्रमत्तोत्थिता नित्यमाराधय गुरुं मम ॥देवयान्युवाच</div><div class="line pada-even" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3300"><div class="verse-block verse-justify"><div class="line">यदि मां धर्मकामार्थे प्रत्याख्यास्यसि याचितः।</div><div class="line">ततः कच न ते विद्या सिद्धिमेषा गमिष्यति ॥कच उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3301"><div class="verse-block verse-justify"><div class="line">गुरुपुत्रीति कृत्वाऽहं प्रत्याचक्षे न दोषतः।</div><div class="line" style="position:relative;">गुरुणा चाननुज्ञातः काममेवं शपस्व माम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3302"><div class="verse-block verse-justify"><div class="line">आर्षं धर्मं ब्रुवाणोऽहं देवयानि यथा त्वया।</div><div class="line" style="position:relative;">शप्तो ह्यनर्हः शापस्य कामतोऽद्य न धर्मतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3303"><div class="verse-block verse-justify"><div class="line">तस्माद्भवत्या यः कामो न तथा स भविष्यति।</div><div class="line" style="position:relative;">ऋषिपुत्रो न ते कश्चिज्जातु पाणिं ग्रहीष्यति॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3304"><div class="verse-block verse-justify"><div class="line">फलिष्यति न ते विद्या यत्त्वं मामात्थ तत्तथा।</div><div class="line">अध्यापयिष्यामि तु यं तस्य विद्या फलिष्यति ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3305"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा द्विजश्रेष्ठो देवयानीं कचस्तदा।</div><div class="line" style="position:relative;">त्रिदशेशालयं शीघ्रं जगाम द्विजसत्तमः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3306"><div class="verse-block verse-justify"><div class="line">तमागतमभिप्रेक्ष्य देवा इन्द्रपुरोगमाः।</div><div class="line">बृहस्पतिं सभाज्येदं कचं वचनमब्रुवन् ॥देवा ऊचुः।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
