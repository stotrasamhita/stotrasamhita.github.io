<div class="stotra-article"><h2 class="stotra-heading">अध्यायः १५०</h2>
<div class="verse-block-wrapper" id="01-ādiparva-b6958"><div class="verse-block verse-justify"><div class="line">जनमेजय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6959"><div class="verse-block verse-justify"><div class="line">द्रुपदस्यापि विप्रर्षे श्रोतुमिच्छामि संभवम्।</div><div class="line" style="position:relative;">कथं चापि समुत्पन्नः कथमस्त्राण्यवाप्तवान् ॥'॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6960"><div class="verse-block verse-justify"><div class="line">एतदिच्छामि भगवंस्त्वत्तः श्रोतुं द्विजोत्तम।</div><div class="line">कौतूहलं जन्मसु मे कीर्त्यमानेष्वनेकशः ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6961"><div class="verse-block verse-justify"><div class="line">राजा बभूव पाञ्चालः पुत्रार्थी पुत्रकारणात्।</div><div class="line" style="position:relative;">वनं गतो महाराजस्तपस्तेपे सुदारुणम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6962"><div class="verse-block verse-justify"><div class="line">आराधयन्प्रयत्नेन महर्षीन्संशितव्रतान्।</div><div class="line" style="position:relative;">तस्य संतप्यमानस्य वने मृगगणायुते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6963"><div class="verse-block verse-justify"><div class="line">कालस्तु सुमहान्राजन्नत्ययात्सुतकारणात्।</div><div class="line" style="position:relative;">स तु राजा महातेजास्तपस्तीव्रं समाददे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6964"><div class="verse-block verse-justify"><div class="line">कंचित्कालं वायुभक्षो निराहारस्तथैव च।</div><div class="line" style="position:relative;">तथैव तु महाबाहोर्वर्तमानस्य भारत॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6965"><div class="verse-block verse-justify"><div class="line">कालस्तस्य महाराज यातो वै नृपसत्तम।</div><div class="line" style="position:relative;">ततो नातिचिरात्काले वसन्ते कामदीपने॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6966"><div class="verse-block verse-justify"><div class="line">फुल्लाशोकवने चैव प्राणिनां सुमनोहरे।</div><div class="line" style="position:relative;">नद्यास्तीरं ततो गत्वा गङ्गायाः पद्मलोचनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6967"><div class="verse-block verse-justify"><div class="line">नियमस्थश्च राजासीत्तदा भरतसत्तम।</div><div class="line" style="position:relative;">ततो नातिचिरात्काले वनं तन्मनुजेश्वर॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6968"><div class="verse-block verse-justify"><div class="line">संप्राप्ता ह्यप्सरा राजन्मेनकेत्यभिविश्रुता।</div><div class="line" style="position:relative;">पुष्पद्रुमान्सज्जमाना राज्ञो दर्शनमागमत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6969"><div class="verse-block verse-justify"><div class="line">न ददर्श तु सा राजंस्तत्र स्थानगतं नृपम्।</div><div class="line" style="position:relative;">दृष्ट्वा चाप्सरसं तां तु शुक्रं राज्ञोऽपतद्भुवि॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6970"><div class="verse-block verse-justify"><div class="line">ततः स राजा राजेन्द्र लज्जया नृपतिः स्वयम्।</div><div class="line" style="position:relative;">पद्भ्यामाक्रमतायुष्मंस्ततस्तु द्रुपदोऽभवत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6971"><div class="verse-block verse-justify"><div class="line">ततस्तु तपसा तस्य राजर्षेर्भावितात्मनः।</div><div class="line" style="position:relative;">पुत्रः समभवच्छीघ्रं पदोस्तस्य क्रमेण तु॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6972"><div class="verse-block verse-justify"><div class="line">तेनास्य ऋषयः सर्वे समागम्य तपोधनाः।</div><div class="line" style="position:relative;">नाम चुक्रुर्हि विद्वांसो द्रुपदोऽस्त्विति भारत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6973"><div class="verse-block verse-justify"><div class="line">स तस्यैवाश्रमे राजन्भरद्वाजस्य भारत।</div><div class="line" style="position:relative;">ववृधे सुमुखं तत्र कामैः सर्वैर्नृपोत्तम॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6974"><div class="verse-block verse-justify"><div class="line">पाञ्चालोऽपि हि राजेन्द्र स्वराज्यं गतवान्प्रभुः।</div><div class="line" style="position:relative;">भरद्वाजस्य विद्यार्थं सुतं दत्वा महात्मनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6975"><div class="verse-block verse-justify"><div class="line">स कुमारस्ततो राजन्द्रोणेन सहितो वने।</div><div class="line" style="position:relative;">वेदांश्चाधिजगे साङ्गान्धनुर्वेदांश्च भारत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6976"><div class="verse-block verse-justify"><div class="line">परया स मुदा युक्तो विचचार वने सुखम्।</div><div class="line" style="position:relative;">तस्यैवं वर्तमानस्य वने वनचरैः सह॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6977"><div class="verse-block verse-justify"><div class="line">कालेनातिचिराद्राजन्पिता स्वर्गमुपेयिवान्।</div><div class="line" style="position:relative;">स समागम्य पाञ्चालैः पाञ्चालेष्वभिषेचितः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6978"><div class="verse-block verse-justify"><div class="line">प्राप्तश्च राज्यं राजेन्द्र सुहृदां प्रीतिवर्धनः।</div><div class="line" style="position:relative;">राज्यं ररक्ष धर्मेण यथा चेन्द्रस्त्रिविष्टपम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6979"><div class="verse-block verse-justify"><div class="line">एतन्मया ते राजेन्द्र यथावत्परिकीर्तितम्।</div><div class="line" style="position:relative;">द्रुपदस्य च राजर्षेर्धृष्टद्युम्नस्य जन्म च॥२२॥</div></div></div></div>
