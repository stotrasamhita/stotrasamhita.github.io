<div class="verse-block-wrapper" id="01-ādiparva-b11137"><div class="verse-block verse-justify"><div class="line">तमब्रवीद्भमकेतुः प्रतिगृह्य जलेश्वरम्।</div><div class="line" style="position:relative;">चतुर्थं लोकपालानां देवदेवं सनातनम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11138"><div class="verse-block verse-justify"><div class="line">सोमेन राज्ञा यद्दत्तं धनुश्चैवेषुधी च ते।</div><div class="line" style="position:relative;">तत्प्रयच्छोभयं शीघ्रं रथं च कपिलक्षणम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11139"><div class="verse-block verse-justify"><div class="line">कार्यं च सुमहत्पार्थो गाण्डीवेन करिष्यति।</div><div class="line" style="position:relative;">चक्रेण वासुदेवश्च तन्ममाद्य प्रदीयताम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11140"><div class="verse-block verse-justify"><div class="line">ददानीत्येव वरुणः पावकं प्रत्यभाषत।</div><div class="line" style="position:relative;">तदद्भुतं महावीर्यं यशःकीर्तिविवर्धनम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11141"><div class="verse-block verse-justify"><div class="line">सर्वशस्त्रैरनाधृष्यं सर्वशस्त्रप्रमाथि च।</div><div class="line" style="position:relative;">सर्वायुधमहामात्रं परसैन्यप्रधर्षणम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11142"><div class="verse-block verse-justify"><div class="line">एकं शतसहस्रेण संमितं राष्ट्रवर्धनम्।</div><div class="line" style="position:relative;">चित्रमुच्चावचैर्वर्णैः शोभितं श्लक्ष्णमव्रणम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11143"><div class="verse-block verse-justify"><div class="line">देवदानवगन्धर्वैः पूजितं शाश्वतीः समाः।</div><div class="line" style="position:relative;">प्रादाच्चैव धनूरत्नमक्षय्यौ च महेषुधी॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11144"><div class="verse-block verse-justify"><div class="line">रथं च दिव्याश्वयुजं कपिप्रवरकेतनम्।</div><div class="line" style="position:relative;">उपेतं राजतैरश्वैर्गान्धर्वैर्हेममालिभिः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11145"><div class="verse-block verse-justify"><div class="line">पाण्डुराभ्रप्रतीकाशैर्मनोवायुसमैर्जवे।</div><div class="line" style="position:relative;">सर्वोपकरणैर्युक्तमजय्यं देवदानवैः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11146"><div class="verse-block verse-justify"><div class="line">भावुमन्तं महाघोषं सर्वरत्नमनोरमम्।</div><div class="line" style="position:relative;">ससर्ज यं सुतपसा भौमनो भुवनप्रभुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11147"><div class="verse-block verse-justify"><div class="line">प्रजापतिरनिर्देश्यं यस्य रूपं रवेरिव।</div><div class="line" style="position:relative;">यं स्म सोमः समारुह्य दानवानजयत्प्रभुः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11148"><div class="verse-block verse-justify"><div class="line">नवमेघप्रतीकाशं ज्ललन्तमिव च श्रिया।</div><div class="line" style="position:relative;">आश्रितौ तं रथश्रेष्ठं शक्रायुधसमावुभौ॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11149"><div class="verse-block verse-justify"><div class="line">तापनीया सुरुचिरा ध्वजयष्टिरनुत्तमा।</div><div class="line" style="position:relative;">तस्यां तु वानरो दिव्यः सिंहशार्दूलकेतनः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11150"><div class="verse-block verse-justify"><div class="line">`हनूमान्नाम तेजस्वी कामरूपी महाबलः।</div><div class="line">'दिधक्षन्निव तत्र स्म संस्थितो मूर्ध्न्यशोभत।</div><div class="line" style="position:relative;">ध्वजे भूतानि तत्रासन्विविधानि महान्ति च॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11151"><div class="verse-block verse-justify"><div class="line">नादेन रिपुसैन्यानां येषां संज्ञा प्रणश्यति।</div><div class="line" style="position:relative;">स तं नानापताकाभिः शोभितं रथसत्तमम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11152"><div class="verse-block verse-justify"><div class="line">प्रदक्षिणमुपावृत्य दैवतेभ्यः प्रणम्य च।</div><div class="line" style="position:relative;">सन्नद्धः कवची खड्गी बद्धगोधाङ्गुलित्रकः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11153"><div class="verse-block verse-justify"><div class="line">आरुरोह तदा पार्थो विमानं सुकृती यथा।</div><div class="line" style="position:relative;">तच्च दिव्यं धनुः श्रेष्ठं ब्रह्मणा निर्मितं पुरा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11154"><div class="verse-block verse-justify"><div class="line">गाण्डीवमुपसंगृह्य बभूव मुदितोऽर्जुनः।</div><div class="line" style="position:relative;">हुताशनं पुरस्कृत्य ततस्तदपि वीर्यवान्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11155"><div class="verse-block verse-justify"><div class="line">जग्राह बलमास्थाय ज्यया च युयुजे धनुः।</div><div class="line" style="position:relative;">मौर्व्यां तु योज्यमानायां बलिना पाण्डवेन ह॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11156"><div class="verse-block verse-justify"><div class="line">येऽशृष्वन्कूजितं यत्र तेषां वै व्यथितं मनः।</div><div class="line" style="position:relative;">लब्ध्वा रथं धनुश्चैव तथाऽक्षय्ये महेषुधी॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11157"><div class="verse-block verse-justify"><div class="line">बभूव कल्यः कौन्तेयः प्रहृष्टः साह्यकर्मणि।</div><div class="line" style="position:relative;">वज्रनाभं ततश्चक्रं ददौ कृष्णाय पावकः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11158"><div class="verse-block verse-justify"><div class="line">आग्नेयमस्त्रं दयितं स च कल्योऽभवत्तदा।</div><div class="line" style="position:relative;">अब्रवीत्पावकश्चैवमेतेन मधुसूदन॥२५॥</div></div></div>
