<div class="verse-block-wrapper" id="01-ādiparva-b10191"><div class="verse-block verse-justify"><div class="line">नारद उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10192"><div class="verse-block verse-justify"><div class="line">उत्सवे वृत्तमात्रे तु त्रैलोक्याकाङ्क्षिणावुभौ।</div><div class="line" style="position:relative;">मन्त्रयित्वा ततः सेनां तावज्ञापयतां तदा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10193"><div class="verse-block verse-justify"><div class="line">सुहृद्भिरप्यनुज्ञातौ दैत्यैर्वृद्धैश्च मन्त्रिभिः।</div><div class="line" style="position:relative;">कृत्वा प्रास्थानिकं रात्रौ मघासु ययतुस्तदा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10194"><div class="verse-block verse-justify"><div class="line">गदापिट्टशधारिण्या शूलमुद्गरहस्तया।</div><div class="line" style="position:relative;">प्रस्थितौ सह वर्मिण्या महत्या दैत्यसेनया॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10195"><div class="verse-block verse-justify"><div class="line">मङ्गलैः स्तुतिभिश्चापि विजयप्रतिसंहितैः।</div><div class="line" style="position:relative;">चारणैः स्तूयमानौ तौ जग्मतुः परया मुदा॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10196"><div class="verse-block verse-justify"><div class="line">तावन्तरिक्षमुत्प्लुत्य दैत्यौ कामगमावुभौ।</div><div class="line" style="position:relative;">देवानामेव भवनं जग्मतुर्युद्दुर्मदौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10197"><div class="verse-block verse-justify"><div class="line">तयोरागमनं ज्ञात्वा वरदानं च तत्प्रभोः।</div><div class="line" style="position:relative;">हित्वा त्रिविष्टपं जग्मुर्ब्रह्मलोकं ततः सुराः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10198"><div class="verse-block verse-justify"><div class="line">ताविन्द्रलोकं निर्जित्य यक्षरक्षोगणांस्तदा।</div><div class="line" style="position:relative;">खेचराण्यपि भूतानि जघ्नतुस्तीव्रविक्रमौ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10199"><div class="verse-block verse-justify"><div class="line">अन्तर्भूमिगतान्नागाञ्जित्वा तौ च महारथौ।</div><div class="line" style="position:relative;">समुद्रवासिनीः सर्वा म्लेच्छजातीर्विजिग्यतुः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10200"><div class="verse-block verse-justify"><div class="line">ततः सर्वां महीं जेतुमारब्धावुग्रशासनौ।</div><div class="line" style="position:relative;">सैनिकांश्च समाहूय सुतीक्ष्णं वाक्यमूचतुः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10201"><div class="verse-block verse-justify"><div class="line">राजर्षयो महायज्ञैर्हव्यकव्यैर्द्विजातयः।</div><div class="line" style="position:relative;">तेजो बलं च देवानां वर्धन्ति श्रियं तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10202"><div class="verse-block verse-justify"><div class="line">तेषामेवं प्रवृत्तानां सर्वेषामसुरद्विषाम्।</div><div class="line" style="position:relative;">संभूय सर्वैरस्माभिः कार्यः सर्वात्मना वधः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10203"><div class="verse-block verse-justify"><div class="line">एवं सर्वान्समादिश्य पूर्वतीरे महोदधेः।</div><div class="line" style="position:relative;">क्रूरां मतिं समास्थाय जग्मतुः सर्वतोमुखौ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10204"><div class="verse-block verse-justify"><div class="line">यज्ञैर्यजन्ति ये केचिद्याजयन्ति च ये द्विजाः।</div><div class="line" style="position:relative;">तान्सर्वान्प्रसभं हत्वा बलिनौ जग्मतुस्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10205"><div class="verse-block verse-justify"><div class="line">आश्रमेष्वग्निहोत्राणि मुनीनां भावितात्मनाम्।</div><div class="line" style="position:relative;">गृहीत्वा प्रक्षिपन्त्यप्सु विश्रब्धं सैनिकास्तयोः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10206"><div class="verse-block verse-justify"><div class="line">तपोधनैश्च ये क्रुद्धैः शापा उक्ता महात्मभिः।</div><div class="line" style="position:relative;">नाक्रामन्त तयोस्तेऽपि वरदाननिराकृताः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10207"><div class="verse-block verse-justify"><div class="line">नाक्रामन्त यदा शापा बाणा मुक्ताः शिलास्विव।</div><div class="line" style="position:relative;">नियमान्संपरित्यज्य व्यद्रवन्त द्विजातयः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10208"><div class="verse-block verse-justify"><div class="line">पृथिव्यां ये तपःसिद्धा दान्ताः शमपरायणाः।</div><div class="line" style="position:relative;">तयोर्भयाद्दुद्रुवुस्ते वैनतेयादिवोरगाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10209"><div class="verse-block verse-justify"><div class="line">मथितैराश्रमैर्भग्नैर्विकीर्णकलशस्रुवैः।</div><div class="line" style="position:relative;">शून्यमासीज्जगत्सर्वं कालेनेव हतं तदा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10210"><div class="verse-block verse-justify"><div class="line">ततो राजन्नदृश्यद्भिर्ऋषिभिश्च महासुरौ।</div><div class="line" style="position:relative;">उभौ विनिश्चयं कृत्वा विकुर्वाते वधैषिणौ॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10211"><div class="verse-block verse-justify"><div class="line">प्रभिन्नकरटौ मत्तौ भूत्वा कुञ्जररूपिणौ।</div><div class="line" style="position:relative;">संलीनमपि दुर्गेषु निन्यतुर्यमसादनम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10212"><div class="verse-block verse-justify"><div class="line">सिंहौ भूत्वा पुनर्व्याघ्रौ पुनश्चान्तर्हितावुभौ।</div><div class="line" style="position:relative;">तैस्तैरुपायैस्तौ क्रूरावृषीन्दृष्ट्वा निजघ्नतुः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10213"><div class="verse-block verse-justify"><div class="line">निवृत्तयज्ञस्वाध्याया प्रनष्टनृपतिद्विजा।</div><div class="line" style="position:relative;">उत्सन्नोत्सवयज्ञा च बभूव वसुधा तदा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10214"><div class="verse-block verse-justify"><div class="line">हाहाभूता भयार्ता च निवृत्तविपणापणा।</div><div class="line" style="position:relative;">निवृत्तदेवकार्या च पुण्योद्वाहविवर्जिता॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10215"><div class="verse-block verse-justify"><div class="line">निवृत्तकृषिगोरक्षा विध्वस्तनगराश्रमा।</div><div class="line" style="position:relative;">अस्थिकङ्कालसंकीर्णा भूर्बभूवोग्रदर्शना॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10216"><div class="verse-block verse-justify"><div class="line">निवृत्तपितृकार्यं च निर्वषट्कारमङ्गलम्।</div><div class="line" style="position:relative;">जगत्प्रतिभयाकारं दुष्प्रेक्ष्यमभवत्तदा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10217"><div class="verse-block verse-justify"><div class="line">चन्द्रादित्यौ ग्रहास्तारा नक्षत्राणि दिवौकसः।</div><div class="line" style="position:relative;">जग्मुर्विषादं तत्कर्म दृष्ट्वा सुन्दोपसुन्दयोः॥२७॥</div></div></div>
