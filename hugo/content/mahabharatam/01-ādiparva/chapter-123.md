<div class="verse-block-wrapper" id="01-ādiparva-b5493"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्राभ्यनुज्ञातः स्वबाहुविजितं धनम्।</div><div class="line" style="position:relative;">भीष्माय सत्यवत्यै च मात्रे चोपजहार सः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5494"><div class="verse-block verse-justify"><div class="line">विदुराय च वै पाण्डुः प्रेषयामास तद्धनम्।</div><div class="line" style="position:relative;">सुहृदश्चापि धर्मात्मा धनेन समतर्पयत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5495"><div class="verse-block verse-justify"><div class="line">ततः सत्यवतीं भीष्मं कौसल्यां च यशस्विनीम्।</div><div class="line" style="position:relative;">शुभैः पाण्डुर्जितैरर्थैस्तोषयामास भारत॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5496"><div class="verse-block verse-justify"><div class="line">ननन्द माता कौसल्या तमप्रतिमतेजसम्।</div><div class="line" style="position:relative;">जयन्तमिव पौलोमी परिष्वज्य नर्षभम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5497"><div class="verse-block verse-justify"><div class="line">तस्य वीरस्य विक्रान्तैः सहस्रशतदक्षिणैः।</div><div class="line" style="position:relative;">अश्वमेधशतैरीजे धृतराष्ट्रो महामखैः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5498"><div class="verse-block verse-justify"><div class="line">संप्रयुक्तस्तु कुन्त्या च माद्र्या च भरतर्षभ।</div><div class="line" style="position:relative;">जिततन्द्रिस्तदा पाण्डुर्बभूव वनगोचरः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5499"><div class="verse-block verse-justify"><div class="line">हित्वा प्रासादनिलयं शुभानि शयनानि च।</div><div class="line" style="position:relative;">अरण्यनित्यः सततं बभूव मृगयापरः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5500"><div class="verse-block verse-justify"><div class="line">स चरन्दक्षिणं पार्श्वं रम्यं हिमवतो गिरः।</div><div class="line" style="position:relative;">उवास गिरिपृष्ठेषु महाशालवनेषु च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5501"><div class="verse-block verse-justify"><div class="line">रराज कुन्त्या माद्र्या च पाण्डुः सह वने चरन्।</div><div class="line" style="position:relative;">करेण्वोरिव मध्यस्थः श्रीमान्पौरन्दरो गजः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5502"><div class="verse-block verse-justify"><div class="line">भारतं सह भार्याभ्यां खङ्गबाणधनुर्धरम्।</div><div class="line">विचित्रकवचं वीरं परमास्त्रविदं नृपम्।</div><div class="line" style="position:relative;">देवोऽयमित्यमन्यन्त चरन्तं वनवासिनः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5503"><div class="verse-block verse-justify"><div class="line">तस्य कामांश्च भोगांश्च नरा नित्यमतन्द्रिताः।</div><div class="line" style="position:relative;">उपजह्रुर्वनान्तेषु धृतराष्ट्रेण चोदिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5504"><div class="verse-block verse-justify"><div class="line">तदासाद्य महारण्यं मृगव्यालनिषेवितम्।</div><div class="line" style="position:relative;">तत्र मैथुनकालस्थं ददर्श मृगयूथपम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5505"><div class="verse-block verse-justify"><div class="line">ततस्तं च मृगीं चैव रुक्मपुङ्खैः सुपत्रिभिः।</div><div class="line" style="position:relative;">निर्बिभेद शरैस्तीक्ष्णैः पाण्डुः पञ्चभिराशुगैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5506"><div class="verse-block verse-justify"><div class="line">स च राजन्महातेजा ऋषिपुत्रस्तपोधनः।</div><div class="line" style="position:relative;">भार्यया सह तेजस्वी मृगरूपेण सङ्गतः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5507"><div class="verse-block verse-justify"><div class="line">स संयुक्तस्तया मृग्या मानुषीं वाचमीरयन्।</div><div class="line">क्षणेन पतितो भूमौ विललापातुरो मृगः ॥मृग उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5508"><div class="verse-block verse-justify"><div class="line">काममन्युवशं प्राप्ता बुद्ध्यन्तरगता अपि।</div><div class="line" style="position:relative;">वर्जयन्ति नृशंसानि पापेष्वपि रता नराः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5509"><div class="verse-block verse-justify"><div class="line">न विधिं ग्रसते प्रज्ञा प्रज्ञां तु ग्रसते विधिः।</div><div class="line" style="position:relative;">विधिपर्यागतानर्थान्प्रज्ञावान्प्रतिपद्यते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5510"><div class="verse-block verse-justify"><div class="line">शस्वद्धर्मात्मनां मुख्ये कुले जातस्य भारत।</div><div class="line">कामलोभाभिभूतस्य कथं ते चलिता मतिः ॥पाण्डुरुवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5511"><div class="verse-block verse-justify"><div class="line">शत्रूणां या वधे वृत्तिः सा मृगाणां वधे स्मृता।</div><div class="line" style="position:relative;">राज्ञां मृग न मां मोहात्त्वं गर्हयितुमर्हसि॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5512"><div class="verse-block verse-justify"><div class="line">अच्छद्मनाऽमायया च मृगाणां वध इष्यते।</div><div class="line" style="position:relative;">स एव धर्मो राज्ञां तु तद्धि त्वं किं नु गर्हसे॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5513"><div class="verse-block verse-justify"><div class="line">अगस्त्यः सत्रमासीनश्चकार मृगयामृषिः।</div><div class="line" style="position:relative;">आरण्यान्सर्वदैवत्यान्मृगान्प्रोक्ष्य महावने॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5514"><div class="verse-block verse-justify"><div class="line">प्रमाणदृष्टधर्मेण कथमस्मान्विगर्हसे।</div><div class="line" style="position:relative;">अगस्त्यस्याभिचारेण युष्माकं विहितो वधः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5515"><div class="verse-block verse-justify"><div class="line">न रिपून्वै समुद्दिश्य विमुञ्चन्ति नराः शरान्।</div><div class="line" style="position:relative;">रन्ध्र एषां विशेषेण वधकालः प्रशस्यते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5516"><div class="verse-block verse-justify"><div class="line">प्रमत्तमप्रमत्तं वा विवृतं घ्नन्ति चौजसा।</div><div class="line">उपायैर्विविधैस्तीक्ष्णैः कस्मान्मृग विगर्हसे ॥मृग उवाच।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5517"><div class="verse-block verse-justify"><div class="line">नाहं घ्न्तं मृगन्राजन्विगर्हे चात्मकारणात्।</div><div class="line" style="position:relative;">मैथुनं तु प्रतीक्ष्यं मे त्वयेहाद्यानृशंस्यतः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5518"><div class="verse-block verse-justify"><div class="line">सर्वभूतहिते काले सर्वभूतेप्सिते तथा।</div><div class="line" style="position:relative;">को हि विद्वान्मृगं हन्याच्चरन्तं मैथुनं वने॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5519"><div class="verse-block verse-justify"><div class="line">अस्यां मृग्यां च राजेन्द्र हर्षान्मैथुनमाचरम्।</div><div class="line" style="position:relative;">पुरुषार्थफलं कर्तुं तत्त्वया विफलीकृतम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5520"><div class="verse-block verse-justify"><div class="line">पौरवाणां महाराज तेषामक्लिष्टकर्मणाम्।</div><div class="line" style="position:relative;">वंशे जातस्य कौरव्य नानुरूपमिदं तव॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5521"><div class="verse-block verse-justify"><div class="line">नृशंसं कर्म सुमहत्सर्वलोकविगर्हितम्।</div><div class="line" style="position:relative;">अस्वर्ग्यमयशस्यं चाप्यधर्मिष्ठं च भारत॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5522"><div class="verse-block verse-justify"><div class="line">स्त्रीभोगानां विशेषज्ञः शास्त्रधर्मार्थतत्त्ववित्।</div><div class="line" style="position:relative;">नार्हस्त्वं सुरसङ्काश कर्तुमस्वर्ग्यमीदृशम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5523"><div class="verse-block verse-justify"><div class="line">त्वया नृशंसकर्तारः पापाचाराश्च मानवाः।</div><div class="line" style="position:relative;">निग्राह्याः पार्थिवश्रेष्ठ त्रिवर्गपरिवर्जिताः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5524"><div class="verse-block verse-justify"><div class="line">किं कृतं ते नरश्रेष्ठ मामिहानागसं घ्नतः।</div><div class="line" style="position:relative;">मुनिं मूलफलाहारं मृगवेषधरं नृप॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5525"><div class="verse-block verse-justify"><div class="line">वसमानमरण्येषु नित्यं शमपरायणम्।</div><div class="line" style="position:relative;">त्वयाऽहं हिंसितो यस्मात्तस्मात्त्वामप्यनागसः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5526"><div class="verse-block verse-justify"><div class="line">द्वयोर्नृशंसकर्तारमवशं काममोहितम्।</div><div class="line" style="position:relative;">जीवितान्तकरो भावो मैथुने समुपैष्यति॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5527"><div class="verse-block verse-justify"><div class="line">अहं हि किंदमो नाम तपसा भावितो मुनिः।</div><div class="line" style="position:relative;">व्यपत्रपन्मनुष्याणां मृग्यां मैथुनमाचरम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5528"><div class="verse-block verse-justify"><div class="line">मृगो भत्वा मृगैः सार्धं चरामि गहने वने।</div><div class="line" style="position:relative;">न तु ते ब्रह्महत्येयं भविष्यत्यविजानतः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5529"><div class="verse-block verse-justify"><div class="line">मृगरूपधरं हत्वा मामेवं काममोहितम्।</div><div class="line" style="position:relative;">अस्य तु त्वं फलं मूढ प्राप्स्यसीदृशमेव हि॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5530"><div class="verse-block verse-justify"><div class="line">प्रियया सह संवासं प्राप्य कामविमोहितः।</div><div class="line" style="position:relative;">त्वमप्यस्यामवस्थायां प्रेतलोकं गमिष्यसि॥३९॥</div></div></div>
