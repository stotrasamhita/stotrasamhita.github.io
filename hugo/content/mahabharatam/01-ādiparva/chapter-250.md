<div class="verse-block-wrapper" id="01-ādiparva-b11111"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11112"><div class="verse-block verse-justify"><div class="line">स तु नैराश्यमापन्नः सदा ग्लानिसमन्वितः।</div><div class="line" style="position:relative;">पितामहमुपागच्छत्संक्रुद्धो हव्यवाहनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11113"><div class="verse-block verse-justify"><div class="line">तच्च सर्वं यथान्यायं ब्रह्मणे संन्यवेदयत्।</div><div class="line" style="position:relative;">उवाच चैवनं भगवान्मुहूर्तं स विचिन्त्य तु॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11114"><div class="verse-block verse-justify"><div class="line">उपायः परिदृष्टो मे यथा त्वं धक्ष्यसेऽनघ।</div><div class="line" style="position:relative;">कालं च कंचित्क्षमतां ततस्तद्धक्ष्यते भवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11115"><div class="verse-block verse-justify"><div class="line">भविष्यतः सहायौ ते नरनारायणौ तदा।</div><div class="line" style="position:relative;">ताभ्यां त्वं सहितो दावं धक्ष्यसे हव्यवाहन॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11116"><div class="verse-block verse-justify"><div class="line">एवमस्त्विति तं वह्निर्ब्रह्माणं प्रत्यभाषत।</div><div class="line" style="position:relative;">संभूतौ तौ विदित्वा तु नरनारायणावृषी॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11117"><div class="verse-block verse-justify"><div class="line">कालस्य महतो राजंस्तस्य वाक्यं स्वयंभुवः।</div><div class="line" style="position:relative;">अनुस्मृत्य जगामाथ पुनरेव पितामहम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11118"><div class="verse-block verse-justify"><div class="line">अब्रवीच्च तदा ब्रह्मा यथा त्वं धक्ष्यसेऽनल।</div><div class="line" style="position:relative;">खाण्डवं दावमद्यैव मिषतोऽस्य शचीपतेः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11119"><div class="verse-block verse-justify"><div class="line">नरनारायणौ यौ तौ पूर्वदेवौ विभावसो।</div><div class="line" style="position:relative;">संप्राप्तौ मानुषे लोके कार्यार्थं हि दिवौकसाम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11120"><div class="verse-block verse-justify"><div class="line">अर्जुनं वासुदेवं च यौ तौ लोकोऽभिमन्यते।</div><div class="line" style="position:relative;">तावेतौ सहितावेहि खाण्डवस्य समीपतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11121"><div class="verse-block verse-justify"><div class="line">तौ त्वं याचस्व साहाय्ये दाहार्थं खाण्डवस्य च।</div><div class="line" style="position:relative;">ततो धक्ष्यसि तं दावं रक्षितं त्रिदशैरपि॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11122"><div class="verse-block verse-justify"><div class="line">तौ तु सत्वानि सर्वाणि यत्नतो वारयिष्यतः।</div><div class="line" style="position:relative;">देवराजं च सहितौ तत्र मे नास्ति संशयः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11123"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा तु वचनं त्वरितो हव्यवाहनः।</div><div class="line" style="position:relative;">कृष्णपार्थावुपागम्य यमर्थं त्वभ्यभाषत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11124"><div class="verse-block verse-justify"><div class="line">तं ते कथितवानस्मि पूर्वमेव नृपोत्तम।</div><div class="line" style="position:relative;">तच्छ्रुत्वा वचनं त्वग्नेर्बीभत्सुर्जातवेदसम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11125"><div class="verse-block verse-justify"><div class="line">अब्रवीन्नृपशार्दूल तत्कालसदृशं वचः।</div><div class="line">दिधक्षुं खाण्डवं दावमकामस्य शतक्रतोः ॥अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11126"><div class="verse-block verse-justify"><div class="line">उत्तमास्त्राणि मे सन्ति दिव्यानि च बहूनि च।</div><div class="line" style="position:relative;">यैरहं शक्नुयां योद्धुमपि वज्रधरान्बहून्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11127"><div class="verse-block verse-justify"><div class="line">धनुर्मे नास्ति भगवन्बाहुवीर्येण संमितम्।</div><div class="line" style="position:relative;">कुर्वतः समरे यत्नं वेगं यद्विषहेन्मम॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11128"><div class="verse-block verse-justify"><div class="line">शरैश्च मेऽर्थो बहुभिरक्षयैः क्षिप्रमस्यतः।</div><div class="line" style="position:relative;">न हि वोढुं रथः शक्तः शरान्मम यथेप्सितान्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11129"><div class="verse-block verse-justify"><div class="line">अश्वांश्च दिव्यानिच्छेयं पाण्डुरान्वातरंहसः।</div><div class="line" style="position:relative;">रथं च मेघनिर्घोषं सूर्यप्रतिमतेजसम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b11130"><div class="verse-block verse-justify"><div class="line">तथा कृष्णस्य वीर्येण नायुधं विद्यते समम्।</div><div class="line" style="position:relative;">येन नागान्पिशाचांश्च निहन्यान्माधवो रणे॥२०॥</div></div></div>
