<div class="verse-block-wrapper" id="01-ādiparva-b10508"><div class="verse-block verse-justify"><div class="line">ततः कतिपयाहस्य तस्मिन्रैवतके गिरौ।</div><div class="line" style="position:relative;">वृष्ण्यन्धकानामभवदुत्सवो नृपसत्तम॥२॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b10510"><div class="verse-block verse-justify"><div class="line">प्रसादै रत्नचित्रैश्च गिरेस्तस्य समन्ततः।</div><div class="line" style="position:relative;">स देशः शोभितो राजन्कल्पवृक्षैश्च सर्वशः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10511"><div class="verse-block verse-justify"><div class="line">वादित्राणि च तत्रान्ये वादकाः समवादयन्।</div><div class="line" style="position:relative;">ननृतुर्नर्तकाश्चैव जगुर्गेयानि गायनाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10512"><div class="verse-block verse-justify"><div class="line">अलङ्कृताः कुमाराश्च वृष्णीनां सुमहौजसाम्।</div><div class="line" style="position:relative;">यानैर्हाटकचित्रैश्च चञ्चूर्यन्ते स्म सर्वशः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10513"><div class="verse-block verse-justify"><div class="line">पौराश्च पादचारेण यानैरुच्चावचैस्तथा।</div><div class="line" style="position:relative;">सदाराः सानुयात्राश्च शतशोऽथ सहस्रशः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10514"><div class="verse-block verse-justify"><div class="line">ततो हलधरः क्षीबो रेवतीसहितः प्रभुः।</div><div class="line" style="position:relative;">अनुगम्यमानो गन्धर्वैरचरत्रत्र भारत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10515"><div class="verse-block verse-justify"><div class="line">तथैव राजा वृष्णीनामुग्रसेनः प्रतापवान्।</div><div class="line" style="position:relative;">अनुगीयमानो गन्धर्वैः स्त्रीसहस्रसहायवान्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10516"><div class="verse-block verse-justify"><div class="line">रौक्मिणेयश्च साम्बश्च क्षीबौ समरदुर्मदौ।</div><div class="line" style="position:relative;">दिव्यमाल्याम्बरधरौ विजह्वातेऽमराविव॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10517"><div class="verse-block verse-justify"><div class="line">अक्रूरः सारणश्चैव गदो बभ्रुर्विदूरथः।</div><div class="line" style="position:relative;">निशठश्चारुदेष्णश्च पृथुर्विपृथुरेव च॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10518"><div class="verse-block verse-justify"><div class="line">सत्यकः सात्यकिश्चैव भङ्गकारमहारवौ।</div><div class="line" style="position:relative;">हार्दिक्य उद्धवश्चैव ये चान्ये नानुकीर्तिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10519"><div class="verse-block verse-justify"><div class="line">एते परिवृताः स्त्रीभिर्गन्धर्वैश्च पृथक्पृथक्।</div><div class="line" style="position:relative;">तमुत्सवं रैवतके शोभयाञ्चक्रिरे तदा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10520"><div class="verse-block verse-justify"><div class="line">`वासदेवो ययौ तत्र सह स्त्रीभिर्मुदान्वितः।</div><div class="line" style="position:relative;">दत्त्वा दानं द्विजातिभ्यः परिव्राजमपश्यत ॥'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10521"><div class="verse-block verse-justify"><div class="line">चित्रकौतूहले तस्मिन्वर्तमाने महाद्भुते।</div><div class="line" style="position:relative;">वासुदेवश्च पार्थश्च सहितौ परिजग्मतुः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10522"><div class="verse-block verse-justify"><div class="line">तत्र चङ्क्रममाणौ तौ वसुदेवसुतां शुभाम्।</div><div class="line" style="position:relative;">अलङ्कृतां सखीमध्ये भद्रां ददृशतुस्तदा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10523"><div class="verse-block verse-justify"><div class="line">दृष्ट्वैव तामर्जुनस्य कन्दर्पः समजायत।</div><div class="line" style="position:relative;">तं तदैकाग्रमनसं कृष्णः पार्थमलक्षयत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10524"><div class="verse-block verse-justify"><div class="line">अब्रवीत्पुरुषव्याघ्रः प्रसहन्निव भारत।</div><div class="line" style="position:relative;">वनेचरस्य किमिदं कामेनालोड्यते मनः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10525"><div class="verse-block"><div class="line">ममैषा भगिनी पार्थ सारणस्य सहोदरी</div><div class="line pada-even">सुभद्रा नाम भद्रं ते पितुर्मे दयिता सुता।</div><div class="line">यदि ते वर्तते बुद्धिर्वक्ष्यामि पितरं स्वयम् ॥अर्जुन उवाच</div><div class="line pada-even" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10526"><div class="verse-block verse-justify"><div class="line">दुहिता वसुदेवस्य वासुदेवस्य च स्वसा।</div><div class="line" style="position:relative;">रूपेण चैषा संपन्ना कमिवैषा न मोहयेत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10527"><div class="verse-block verse-justify"><div class="line">कृतमेव तु कल्याणं सर्वं मम भवेद्ध्रुवम्।</div><div class="line" style="position:relative;">यदि स्यान्मम वार्ष्णेयी महिषीयं स्वसा तव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10528"><div class="verse-block verse-justify"><div class="line">प्राप्तौ तु क उपायः स्यात्तं व्रवीहि जनार्दन।</div><div class="line">आस्थास्यामि तदा सर्वं यदि शक्यं नरेण तत् ॥वासुदेव उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10529"><div class="verse-block verse-justify"><div class="line">स्वयं वरः क्षत्रियाणां विवाहः पुरुषर्षभ।</div><div class="line" style="position:relative;">स च संशयितः पार्थ स्वभावस्यानिमित्ततः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10530"><div class="verse-block verse-justify"><div class="line">प्रसह्य हरणं चापि क्षत्रियाणां प्रशस्यते।</div><div class="line" style="position:relative;">विवाहहेतुः शूराणामिति धर्मविदो विदुः॥२४॥</div></div></div>
