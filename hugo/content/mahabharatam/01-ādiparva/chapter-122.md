<div class="verse-block-wrapper" id="01-ādiparva-b5450"><div class="verse-block verse-justify"><div class="line">तं भीष्मः प्रत्युवाचेदं मद्रराजं कुरूद्वहः।</div><div class="line" style="position:relative;">आगतं मां विजानीहि कन्यार्थिनमरिन्दम॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5451"><div class="verse-block verse-justify"><div class="line">श्रूयते भवतः साध्वी स्वसा माद्री यशस्विनी।</div><div class="line" style="position:relative;">तामहं वरयिष्यामि पाण्डोरर्थे यशस्विनीम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5452"><div class="verse-block verse-justify"><div class="line">युक्तरूपो हि संबन्धे त्वं नो राजन्वयं तव।</div><div class="line" style="position:relative;">एतत्संचिन्त्य मद्रेश गृहाणास्मान्यथाविधि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5453"><div class="verse-block verse-justify"><div class="line">तमेवंवादिनं भीष्मं प्रत्यभाषत मद्रपः।</div><div class="line" style="position:relative;">न हि मेऽन्यो वरस्त्वत्तः श्रेयानिति मतिर्मम॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5454"><div class="verse-block verse-justify"><div class="line">पूर्वैः प्रवर्तितं किंचित्कुलेऽस्मिन्नृपसत्तमैः।</div><div class="line" style="position:relative;">साधु वा यदि वाऽसाधु तन्नातिक्रान्तुमुत्सहे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5455"><div class="verse-block verse-justify"><div class="line">व्यक्तं तद्भवतश्चापि विदितं नात्र संशयः।</div><div class="line" style="position:relative;">न च युक्तं तथा वक्तुं भवान्देहीति सत्तम॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5456"><div class="verse-block verse-justify"><div class="line">कुलधर्मः स नो वीर प्रमाणं परमं च तत्।</div><div class="line" style="position:relative;">तेन त्वां न ब्रवीम्येतदसंदिग्धं वचोऽरिहन्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5457"><div class="verse-block verse-justify"><div class="line">तं भीष्मः प्रत्युवाचेदं मद्रराजं जनाधिपः।</div><div class="line" style="position:relative;">धर्म एष परो राजन्स्वयमुक्तः स्वयंभुवा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5458"><div class="verse-block verse-justify"><div class="line">नात्र कश्चन दोषोऽस्ति पूर्वैर्विधिरयं कृतः।</div><div class="line" style="position:relative;">विदितेयं च ते शल्य मर्यादा साधुसंमता॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b5460"><div class="verse-block verse-justify"><div class="line">गजानश्वान्रथांश्चैव वासांस्याभरणानि च।</div><div class="line" style="position:relative;">मणिमुक्ताप्रवालं च गाङ्गेयो व्यसृजच्छुभम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5461"><div class="verse-block verse-justify"><div class="line">तत्प्रगृह्य धनं सर्वं शल्यः संप्रतीमानसः।</div><div class="line" style="position:relative;">ददौ तां समलङ्कृत्य स्वसारं कौरवर्षभे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5462"><div class="verse-block verse-justify"><div class="line">स तां माद्रीमुपादाय भीष्मः सागरगासुतः।</div><div class="line" style="position:relative;">आजगाम पुरीं धीमान्प्रविष्टो गजसाह्वयम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5463"><div class="verse-block verse-justify"><div class="line">तत इष्टेऽहनि प्राप्ते मुहूर्ते साधुसंमते।</div><div class="line">`विवाहं कारायामास भीष्मः पाण्डोर्महात्मनः।</div><div class="line" style="position:relative;">'जग्राह विधिवत्पाणिं माद्र्याः पाण्डुर्नराधिपः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5464"><div class="verse-block verse-justify"><div class="line">ततो विवाहे निर्वृत्ते स राजा कुरुनन्दनः।</div><div class="line" style="position:relative;">स्थापयामास तां भार्यां शुभे वेश्मनि भामिनीं॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5465"><div class="verse-block verse-justify"><div class="line">स ताभ्यां व्यचरत्सार्धं भार्याभ्यां राजसत्तमः।</div><div class="line" style="position:relative;">कुन्त्या माद्र्या च राजेन्द्रो यथाकामं यथासुखम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5466"><div class="verse-block verse-justify"><div class="line">ततः स कौरवो राजा विहृत्य त्रिदशा निशाः।</div><div class="line" style="position:relative;">जिगीषया महीं पाण्डुर्निरक्रामत्पुरात्प्रभो॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5467"><div class="verse-block verse-justify"><div class="line">स भीष्मप्रमुखान्वृद्धानभिवाद्य प्रणम्य च।</div><div class="line">धृतराष्ट्रं च कौरव्यं तथान्यान्कुरुसत्तमान्।</div><div class="line" style="position:relative;">आमन्त्र्य प्रययौ राजा तैश्चैवाप्यनुमोजितः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5468"><div class="verse-block verse-justify"><div class="line">मङ्गलाचारयुक्ताभिराशीर्भिरभिनन्दितः।</div><div class="line" style="position:relative;">गजवाजिरथौघेन बलेन महताऽगमत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5469"><div class="verse-block verse-justify"><div class="line">स राजा देवगर्भाभो विजिगीषुर्वसुन्धराम्।</div><div class="line" style="position:relative;">हृष्टपुष्टबलैः प्रायात्पाण्डुः शत्रूननेकशः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5470"><div class="verse-block verse-justify"><div class="line">पूर्वमागस्कृतो गत्वा दशार्णाः समरे जिताः।</div><div class="line" style="position:relative;">पाण्डुना नरसिंहेन कौरवाणां यशोभृता॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5471"><div class="verse-block verse-justify"><div class="line">ततः सेनामुपादाय पाण्डुर्नानाविधध्वजाम्।</div><div class="line" style="position:relative;">प्रभूतहस्त्यश्वयुतां पदातिरथसंकुलाम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5472"><div class="verse-block verse-justify"><div class="line">आगस्कारी महीपानां बहूनां बलदर्पितः।</div><div class="line" style="position:relative;">गोप्ता मगधराष्ट्रस्य दीर्घो राजगृहे हतः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5473"><div class="verse-block verse-justify"><div class="line">ततः कोशं समादाय वाहनानि च भूरिशः।</div><div class="line" style="position:relative;">पाण्डुना मिथिलां गत्वा विदेहाः समरे जिताः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5474"><div class="verse-block verse-justify"><div class="line">तथा काशिषु सुह्मेषु पुण्ड्रेषु च नरर्षभ।</div><div class="line" style="position:relative;">स्वबाहुबलवीर्येण कुरूणामकरोद्यशः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5475"><div class="verse-block verse-justify"><div class="line">तं शरौघमहाज्वालं शस्त्रार्चिषमरिन्दमम्।</div><div class="line" style="position:relative;">पाण्डुपावकमासाद्य व्यवह्यन्त नराधिपाः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b5476"><div class="verse-block verse-justify"><div class="line">ते ससेनाः ससेनेन विध्वंसितबला नृपाः।</div><div class="line" style="position:relative;">पाण्डुना वशगाः कृत्वा कुरुकर्मसु योजिताः॥३२॥</div></div></div>
