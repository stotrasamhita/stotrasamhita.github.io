<div class="verse-block-wrapper" id="01-ādiparva-b2166"><div class="verse-block verse-justify"><div class="line">सौतिरुवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2167"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा ततः श्रीमान्मन्त्रिभिश्चानुमीदितः।</div><div class="line" style="position:relative;">आरुरोह प्रतिज्ञां स सर्पसत्राय पार्थिवः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2168"><div class="verse-block verse-justify"><div class="line">ब्रह्मन्भरतशार्दूलो राजा पारिक्षितस्तदा।</div><div class="line" style="position:relative;">पुरोहितमथाहूय ऋत्विजो वसुधाधिपः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2169"><div class="verse-block verse-justify"><div class="line">अब्रवीद्वाक्यसंपन्नः कार्यसंपत्करं वचः।</div><div class="line" style="position:relative;">यो मे हिंसितवांस्तातं तक्षकः स दुरात्मवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2170"><div class="verse-block verse-justify"><div class="line">प्रतिकुर्यां यथा तस्य तद्भवन्तो ब्रुवन्तु मे।</div><div class="line" style="position:relative;">अपि तत्कर्म विदितं भवतां येन पन्नगम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2171"><div class="verse-block verse-justify"><div class="line">तक्षकं संप्रदीप्तेऽग्नौ प्रक्षिपेयं सबान्धवम्।</div><div class="line">यथा तेन पिता मह्यं पूर्वं दग्धो विषाग्निना ॥तथाऽहमपि तं पापं दग्धुमिच्छामि पन्नगम् ॥ऋत्विज ऊचुः।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2172"><div class="verse-block verse-justify"><div class="line">अस्ति राजन्महात्सत्रं त्वदर्थं देवनिर्मितम्।</div><div class="line" style="position:relative;">सर्वसत्रमिति ख्यातं पुराणे परिपठ्यते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2173"><div class="verse-block verse-justify"><div class="line">आहर्ता तस्य सत्रस्य त्वन्नान्योऽस्ति नराधिप।</div><div class="line" style="position:relative;">इति पौराणिकाः प्राहुरस्माकं चास्ति स क्रतुः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2174"><div class="verse-block verse-justify"><div class="line">एवमुक्तः स राजर्षिर्मेने दग्धं हि तक्षकम्।</div><div class="line" style="position:relative;">हुताशनमुखे दीप्ते प्रविष्टमिति सत्तम॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2175"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीन्मन्त्रविदस्तान्राजा ब्राह्मणांस्तदा।</div><div class="line">आहरिष्यामि तत्सत्रं संभाराः संभ्रियन्तु मे ॥सौतिरुवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2176"><div class="verse-block verse-justify"><div class="line">ततस्त ऋत्विजस्तस्य शास्त्रतो द्विजसत्तम।</div><div class="line" style="position:relative;">तं देशं मापयामासुर्यज्ञायतनकारणात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2177"><div class="verse-block verse-justify"><div class="line">यथावद्वेदविद्वांसः सर्वे बुद्धेः परंगताः।</div><div class="line" style="position:relative;">ऋद्ध्या परमया युक्तमिष्टं द्विजगणैर्युतम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2178"><div class="verse-block verse-justify"><div class="line">प्रभूतधनधान्याढ्यमृत्विग्भिः सुनिषेवितम्।</div><div class="line" style="position:relative;">निर्माय चापि विधिवद्यज्ञायतनमीप्सितम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2179"><div class="verse-block verse-justify"><div class="line">राजानं दीक्षयामासुः सर्पसत्राप्तये तदा।</div><div class="line" style="position:relative;">इदं चासीत्तत्र पूर्वं सर्पसत्रे भविष्यति॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2180"><div class="verse-block verse-justify"><div class="line">निमित्तं महदुत्पन्नं यज्ञविघ्नकरं तदा।</div><div class="line" style="position:relative;">यज्ञस्यायतने तस्मिन्क्रियमाणे वचोऽब्रवीत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2181"><div class="verse-block verse-justify"><div class="line">स्थपतिर्बुद्धिसंपन्नो वास्तुविद्याविशारदः।</div><div class="line" style="position:relative;">इत्यब्रवीत्सूत्रधारः सूतः पौराणिकस्तदा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2182"><div class="verse-block verse-justify"><div class="line">यस्मिन्देशे च काले च मापनेयं प्रवर्तिता।</div><div class="line" style="position:relative;">ब्राह्मणं कारणं कृत्वा नायं संस्थास्यते क्रतुः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2183"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा तु राजासौ प्राग्दीक्षाकालमब्रवीत्।</div><div class="line" style="position:relative;">क्षत्तारं न हि मे कश्चिदज्ञातः प्रविशेदिति॥१८॥</div></div></div></div>
