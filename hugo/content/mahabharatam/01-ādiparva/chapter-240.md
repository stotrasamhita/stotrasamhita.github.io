<div class="verse-block-wrapper" id="01-ādiparva-b10539"><div class="verse-block verse-justify"><div class="line">कृष्णस्य मतमास्थाय प्रययौ भरतर्षभः।</div><div class="line" style="position:relative;">द्वारकाया उपवने तस्थौ वै कार्यसाधनः॥३॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b10541"><div class="verse-block verse-justify"><div class="line">चिन्तयानस्ततो भद्रामुपविष्टः शिलातले।</div><div class="line" style="position:relative;">रमणीये वनोद्देशे बहुपादपसंवृते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10542"><div class="verse-block verse-justify"><div class="line">सालतालाश्वकर्णैश्च बकुलैरर्जुनैस्तथा।</div><div class="line" style="position:relative;">चम्पकाशोकपुन्नागैः केतकैः पाटलैस्तथा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10543"><div class="verse-block verse-justify"><div class="line">कर्णिकारैरशोकैश्च अङ्कोलैरतिमुक्तकैः।</div><div class="line" style="position:relative;">एवमादिभिरन्यैश्च संवृते स शिलातले॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10544"><div class="verse-block verse-justify"><div class="line">पुनःपुनश्चिन्तयानः सुभद्रां भद्रभाषिणीम्।</div><div class="line" style="position:relative;">यदृच्छया चोपपन्नान्वृष्णिवीरान्ददर्श सः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10545"><div class="verse-block verse-justify"><div class="line">बलदेवं च हार्दिक्यं साम्बं सारणमेव च।</div><div class="line" style="position:relative;">प्रद्युम्नं च गदं चैव चारुदेष्णं विदूरथम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10546"><div class="verse-block verse-justify"><div class="line">भानुं च हरितं चैव विपृथुं पृथुमेव च।</div><div class="line" style="position:relative;">तथान्यांश्च बहून्पश्यन्हृदि शोकमधारयत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10547"><div class="verse-block verse-justify"><div class="line">ततस्ते सहिताः सर्वे यतिं दृष्ट्वा समुत्सुकाः।</div><div class="line" style="position:relative;">वृष्णयो विनयोपेताः परिवार्योपतस्थिरे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10548"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनः प्रीतमनाः स्वागतं व्याजहार सः।</div><div class="line" style="position:relative;">आस्यतामास्यतां सर्वै रमणीये शिलातले॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10549"><div class="verse-block verse-justify"><div class="line">इत्येवमुक्ता यतिना प्रीतास्ते यादवर्षभाः।</div><div class="line" style="position:relative;">उपोपविविशुः सर्वे ते स्वागतमिति ब्रुवन्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10550"><div class="verse-block verse-justify"><div class="line">परितः सन्निविष्टेषु वृष्णिवीरेषु पाण्डवः।</div><div class="line" style="position:relative;">आकारं गूहमानस्तु कुशलप्रश्नमब्रवीत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10551"><div class="verse-block verse-justify"><div class="line">सर्वत्र कुशलं चोक्त्वा बलदेवोऽब्रवीदिदम्।</div><div class="line" style="position:relative;">प्रसादं कुरु मे विप्र कुतस्त्वं चागतो ह्यसि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10552"><div class="verse-block verse-justify"><div class="line">त्वया दृष्टानि पुण्यानि वद त्वं वदतांवर।</div><div class="line">पर्वतांश्चैव तीर्थानि वनान्यायतनानि च ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10553"><div class="verse-block verse-justify"><div class="line">तीर्थानां दर्शनं चैव पर्वतानां च भारत।</div><div class="line" style="position:relative;">आपगानां वनानां च कथयामास ताः कथाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10554"><div class="verse-block verse-justify"><div class="line">श्रुत्वा धर्मकथाः पुण्या वृष्णिवीरा मुदान्विताः।</div><div class="line" style="position:relative;">अपूजयंस्तदा भिक्षुं कथान्ते जनमेजय॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10555"><div class="verse-block verse-justify"><div class="line">ततस्तु यादवाः सर्वे मन्त्रयन्ति स्म भारत।</div><div class="line" style="position:relative;">अयं देशातिथिः श्रीमान्यतिलिङ्गधरो द्विजः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10556"><div class="verse-block verse-justify"><div class="line">आवासं कमुपाश्रित्य वसेत निरुपद्रवः।</div><div class="line" style="position:relative;">इत्येवं प्रब्रुवन्तस्तु रौहिणेयं च यादवाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10557"><div class="verse-block verse-justify"><div class="line">ददृशुः कृष्णमायान्तं सर्वे यादवनन्दनम्।</div><div class="line" style="position:relative;">एहि केशव तातेति रौहिणेयो वचोऽब्रवीत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10558"><div class="verse-block"><div class="line">यतिलिङ्गधरो विद्वान्देशातिथिरयं द्विजः</div><div class="line pada-even">वर्षमासनिवासार्थमागतो नः पुरं प्रति।</div><div class="line">स्थाने यस्मिन्निवसतु तन्मे ब्रूहि जनार्दन ॥श्रीकृष्ण उवाच</div><div class="line pada-even" style="position:relative;">॥२२॥</div></div></div>
