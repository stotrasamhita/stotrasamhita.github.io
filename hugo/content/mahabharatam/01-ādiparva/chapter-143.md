<div class="verse-block-wrapper" id="01-ādiparva-b6609"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6610"><div class="verse-block verse-justify"><div class="line">ततो धनञ्जयं द्रोणः स्मयमानोऽभ्यभाषत।</div><div class="line" style="position:relative;">त्वयेदानीं प्रहर्तव्यमेतल्लक्ष्यं विलोक्यताम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6611"><div class="verse-block verse-justify"><div class="line">मद्वाक्यसमकालं ते मोक्तव्योऽत्र भवेच्छरः।</div><div class="line" style="position:relative;">वितत्य कार्मुकं पुत्र तिष्ठ तावन्मुहूर्तकम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6612"><div class="verse-block verse-justify"><div class="line">एवमुक्तः सव्यसाची मण्डलीकृतकार्मुकः।</div><div class="line" style="position:relative;">तस्थौ भासं समुद्दिश्य गुरुवाक्यप्रचोदितः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6613"><div class="verse-block verse-justify"><div class="line">मुहूर्तादिव तं द्रोणस्तथैव समभाषत।</div><div class="line" style="position:relative;">पश्यस्येनं स्थितं भासं द्रुमं मामपि चार्जुन॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6614"><div class="verse-block verse-justify"><div class="line">पश्याम्येकं भासमिति द्रोणं पार्थोऽभ्यभाषत।</div><div class="line" style="position:relative;">न तु वृक्षं भवन्तं वा पश्यामीति च भारत॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6615"><div class="verse-block verse-justify"><div class="line">ततः प्रीतमना द्रोणो मुहूर्तादिव तं पुनः।</div><div class="line" style="position:relative;">प्रत्यभाषत दुर्धर्षः पाण्डवानां महारथम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6616"><div class="verse-block verse-justify"><div class="line">भासं पश्यसि यद्येनं तथा ब्रूहि पुनर्वचः।</div><div class="line" style="position:relative;">शिरः पश्यामि भासस्य न गात्रमिति सोऽब्रवीत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6617"><div class="verse-block verse-justify"><div class="line">अर्जुनेनैवमुक्तस्तु द्रोणो हृष्टतनूरुहः।</div><div class="line" style="position:relative;">मुञ्चस्वेत्यब्रवीत्पार्थं स मुमोचाविचारयन्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6618"><div class="verse-block verse-justify"><div class="line">ततस्तस्य गस्थस्य क्षुरेण निशितेन च।</div><div class="line" style="position:relative;">शिर उत्कृत्य तरसा पातयामास पाण्डवः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6619"><div class="verse-block verse-justify"><div class="line">तस्मिन्कर्मणि संसिद्धे पर्यष्वजत पाण्डवम्।</div><div class="line" style="position:relative;">मेने च द्रुपदं सङ्ख्ये सानुबन्धं पराजितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6620"><div class="verse-block verse-justify"><div class="line">कस्य चित्त्वथ कालस्य सशिष्योऽङ्गिरसां वरः।</div><div class="line" style="position:relative;">जगाम गङ्गामभितो मज्जितुं भरतर्षभ॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6621"><div class="verse-block verse-justify"><div class="line">अवगाढमथो द्रोणं सलिले सलिलेचरः।</div><div class="line" style="position:relative;">ग्राहो जग्राह बलवाञ्जङ्घान्ते कालचोदितः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6622"><div class="verse-block verse-justify"><div class="line">स समर्थोऽपि मोक्षाय शिष्यान्सर्वानचोदयत्।</div><div class="line" style="position:relative;">ग्राहं हत्वा तु मोक्ष्यध्वं मामिति त्वरयन्निव॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6623"><div class="verse-block verse-justify"><div class="line">तद्वाक्यसमकालं तु बीभत्सुर्निशितैः शरैः।</div><div class="line" style="position:relative;">अवार्यैः पञ्चभिर्ग्राहं मग्नमम्भस्यताडयत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6624"><div class="verse-block verse-justify"><div class="line">इतरे त्वथ संमूढास्तत्रपत्र प्रपेदिरे।</div><div class="line" style="position:relative;">तं तु दृष्ट्वा क्रियोपेतं द्रोणोऽमन्यत पाण्डवम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6625"><div class="verse-block verse-justify"><div class="line">विशिष्टं सर्वशिष्येभ्यः प्रीतिमांश्चाभवत्तदा।</div><div class="line" style="position:relative;">स पार्थबाणैर्बहुधा खण्डशः परिकल्पितः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6626"><div class="verse-block verse-justify"><div class="line">ग्राहः पञ्चत्वमापेदे जङ्घां त्यक्त्वा महात्मनः।</div><div class="line" style="position:relative;">`सर्वक्रियाभ्यनुज्ञानात्तथा शिष्यान्समानयत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6627"><div class="verse-block verse-justify"><div class="line">दुर्योधनं चित्रसेनं दुःशासनविविंशती।</div><div class="line" style="position:relative;">अर्जुनं च समानीय ह्यश्वत्थामानमेव च॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6628"><div class="verse-block verse-justify"><div class="line">शिशुकं मृण्मयं कृत्वा द्रोणो गङ्गाजले ततः।</div><div class="line" style="position:relative;">शिष्याणां पश्यतां चैव क्षिपति स्म महाभुजः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6629"><div class="verse-block verse-justify"><div class="line">चक्षुषी वाससा चैव बद्ध्वा प्रादाच्छरासनम्।</div><div class="line" style="position:relative;">शिशुकं विद्ध्यतेमं वै जलस्थं बद्धचक्षुषः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6630"><div class="verse-block verse-justify"><div class="line">तत्क्षणेनैव बीभत्सुरावापैर्दशभिर्वशी।</div><div class="line" style="position:relative;">पञ्चकैरनुविव्याध मग्नं शिशुकमम्भसि॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6631"><div class="verse-block verse-justify"><div class="line">ताः स दृष्ट्वा क्रियाः सर्वा द्रोणोऽमन्यत पाण्डवम्।</div><div class="line">विशिष्टं सर्वशिष्येभ्यः प्रीतिमांश्चाभवत्तदा।</div><div class="line" style="position:relative;">'तथाब्रवीन्महात्मानं भारद्वाजो महारथम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6632"><div class="verse-block verse-justify"><div class="line">गृहाणेदं महाबाहो विशिष्टमतिदुर्धरम्।</div><div class="line" style="position:relative;">अस्त्रं ब्रह्मशिरो नाम सप्रयोगनिवर्तनम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6633"><div class="verse-block verse-justify"><div class="line">न च ते मानुषेष्वेतत्प्रयोक्तव्यं कथंचन।</div><div class="line" style="position:relative;">जगद्विनिर्दहेदेतदल्पतेजसि पातितम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6634"><div class="verse-block verse-justify"><div class="line">असामान्यमिदं तात लोकेष्वस्त्रं निगद्यते।</div><div class="line" style="position:relative;">तद्धारयेथाः प्रयतः शृणु चेदं वचो मम॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6635"><div class="verse-block verse-justify"><div class="line">बाधेतामानुषः शत्रुर्यदि त्वां वीर कश्चन।</div><div class="line" style="position:relative;">तद्वधाय प्रयुञ्जीथास्तदस्त्रमिदमाहवे॥२७॥</div></div></div>
