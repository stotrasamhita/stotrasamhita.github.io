<div class="verse-block-wrapper" id="01-ādiparva-b2612"><div class="verse-block verse-justify"><div class="line">स चेदिविषयं रम्यं वसुः पौरवनन्दनः।</div><div class="line" style="position:relative;">इन्द्रोपदेशाज्जग्राह रमणीयं महीपतिः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2613"><div class="verse-block verse-justify"><div class="line">तमाश्रमे न्यस्तशस्त्रं निवसन्तं तपोनिधिम्।</div><div class="line" style="position:relative;">देवाः शक्रपुरोगा वै राजानमुपतस्थिरे॥४॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b2615"><div class="verse-block verse-justify"><div class="line">न संकीर्येत धर्मोऽयं पृथिव्यां पृथिवीपते।</div><div class="line">त्वया हि धर्मो विधृतः कृत्स्नं धारयते जगत् ॥इन्द्र उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2616"><div class="verse-block verse-justify"><div class="line">`देवानहं पालयिता पालय त्वं हि मानुषान्।</div><div class="line">'लोके धर्मं पालय त्वं नित्ययुक्तः समाहितः।</div><div class="line" style="position:relative;">धर्मयुक्तस्ततो लोकान्पुण्यान्प्राप्स्यसि शाश्वतान्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2617"><div class="verse-block verse-justify"><div class="line">दिविष्ठस्य भुविष्ठस्त्वं सखाभूतो मम प्रियः।</div><div class="line" style="position:relative;">ऊधः पृथिव्या यो देशस्तमावस नराधिप॥८॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b2619"><div class="verse-block verse-justify"><div class="line">अर्थवानेष देशो हि धनरत्नादिभिर्युतः।</div><div class="line" style="position:relative;">वसुपूर्णा च वसुधा वस चेदिषु चेदिप॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2620"><div class="verse-block verse-justify"><div class="line">धर्मशीला जनपदाः सुसंतोषाश्च साधवः।</div><div class="line" style="position:relative;">न च मिथ्या प्रलापोऽत्र स्वैरेष्वपि कुतोऽन्यथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2621"><div class="verse-block verse-justify"><div class="line">न च पित्रा विभज्यन्ते पुत्रा गुरुहिते रताः।</div><div class="line" style="position:relative;">युञ्जते धुरि नो गाश्च कृशान्संधुक्षयन्ति च॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2622"><div class="verse-block verse-justify"><div class="line">सर्वे वर्णाः स्वधर्मस्थाः सदा चेदिषु मानद।</div><div class="line" style="position:relative;">न तेऽस्त्यविदितं किंचित्त्रिषु लोकेषु यद्भवेत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2623"><div class="verse-block verse-justify"><div class="line">दैवोपभोग्यं दिव्यं त्वामाकाशे स्फाटिकं महत्।</div><div class="line" style="position:relative;">आकाशगं त्वां मद्दत्तं विमानमुपपत्स्यते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2624"><div class="verse-block verse-justify"><div class="line">त्वमेकः सर्वमर्त्येषु विमानवरमास्थितः।</div><div class="line" style="position:relative;">चरिष्यस्युपरिस्थो हि देवो विग्रहवानिव॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2625"><div class="verse-block verse-justify"><div class="line">ददामि ते वैजयन्तीं मालामम्लानपङ्कजाम्।</div><div class="line" style="position:relative;">धारयिष्यति सङ्ग्रामे या त्वां शस्त्रैरविक्षतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2626"><div class="verse-block verse-justify"><div class="line">लक्षणं चैतदेवेह भविता ते नराधिप।</div><div class="line" style="position:relative;">इन्द्रमालेति विख्यातं धन्यमप्रतिमं महत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2627"><div class="verse-block verse-justify"><div class="line">यष्टिं च वैणवीं तस्मै ददौ वृत्रनिषूदनः।</div><div class="line" style="position:relative;">इष्टप्रदानमुद्दिश्य शिष्टानां प्रतिपालिनीम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2628"><div class="verse-block verse-justify"><div class="line">`एवं संसान्त्व्य नृपतिं तपसः संन्यवर्तयत्।</div><div class="line" style="position:relative;">प्रययौ दैवतैः सार्धं कृत्वा कार्यं दिवौकसाम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2629"><div class="verse-block verse-justify"><div class="line">ततस्तु राजा चेदीनामिन्द्राभरणभूषितः।</div><div class="line" style="position:relative;">इन्द्रदत्तं विमानं तदास्थाय प्रययौ पुरीम् ॥'॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2630"><div class="verse-block verse-justify"><div class="line">तस्याः शक्रस्य पूजार्थं भूमौ भूमिपतिस्तदा।</div><div class="line" style="position:relative;">प्रवेशं कारयामास सर्वोत्सववरं तदा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2631"><div class="verse-block verse-justify"><div class="line">`मार्गशीर्षे महाराज पूर्वपक्षे महामखम्।</div><div class="line" style="position:relative;">ततःप्रभृति चाद्यापि यष्टेः क्षितिपसत्तमैः ॥'॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2632"><div class="verse-block verse-justify"><div class="line">प्रवेशः क्रियते राजन्यथा तेन प्रवर्तितः।</div><div class="line" style="position:relative;">अपरेद्युस्ततस्तस्याः क्रियतेऽत्युच्छ्रयो नृपैः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2633"><div class="verse-block verse-justify"><div class="line">अलङ्कृताया पिटकैर्गन्धमाल्यैश्च भूषणैः।</div><div class="line" style="position:relative;">`माल्यदामपरिक्षिप्तां द्वात्रिंशत्किष्कुसंमिताम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2634"><div class="verse-block verse-justify"><div class="line">उद्धृत्य पिटके चापि द्वादशारत्निकोच्छ्रये।</div><div class="line" style="position:relative;">महारजनवासांसि परिक्षिप्य ध्वजोत्तमम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2635"><div class="verse-block verse-justify"><div class="line">वासोभिरन्नपानैश्च पूजितैर्ब्राह्मणर्षभैः।</div><div class="line" style="position:relative;">पुण्याहं वाचयित्वाथ ध्वज उच्छ्रियते तदा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2636"><div class="verse-block verse-justify"><div class="line">शङ्खभेरीमृदङ्गैश्च संनादः क्रियते तदा'।</div><div class="line" style="position:relative;">भगवान्पूज्यते चात्र यष्टिरूपेण वासवः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2637"><div class="verse-block verse-justify"><div class="line">स्वयमेव गृहीतेन वसोः प्रीत्या महात्मनः।</div><div class="line" style="position:relative;">`माणिभद्रादयो यक्षाः पूज्यन्ते दैवतैः सह॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2638"><div class="verse-block verse-justify"><div class="line">नानाविधानि दानानि दत्त्वार्थिभ्यः सुहृज्जनैः।</div><div class="line" style="position:relative;">अलङ्कृत्वा माल्यदामैर्वस्त्रैर्नानाविधैस्तथा॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2639"><div class="verse-block verse-justify"><div class="line">दृतिभिः सजलैः सर्वैः क्रीडित्वा नृपशासनात्।</div><div class="line" style="position:relative;">सभाजयित्वा राजानं कृत्वा नर्माश्रयाः कथाः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2640"><div class="verse-block verse-justify"><div class="line">रमन्ते नागराः सर्वे तथा जानपदैः सह।</div><div class="line" style="position:relative;">सूताश्च मागधाश्चैव रमन्ते नटनर्तकाः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2641"><div class="verse-block verse-justify"><div class="line">प्रीत्या तु नृपशार्दूल सर्वे चक्रुर्महोत्सवम्।</div><div class="line" style="position:relative;">सान्तःपुरः सहामात्यः सर्वाभरणभूषितः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2642"><div class="verse-block verse-justify"><div class="line">महारजनवासांसि वसित्वा चेदिराट् तदा।</div><div class="line" style="position:relative;">जातिहिङ्गुलकेनाक्तः सदारो मुमुदे तदा॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b2646"><div class="verse-block verse-justify"><div class="line">ये पूजयिष्यन्ति च मुदा यथा चेदिपतिर्नृपः।</div><div class="line" style="position:relative;">कारयिष्यन्ति च मुदा यथा चेदिपतिर्नृपः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2647"><div class="verse-block verse-justify"><div class="line">तेषां श्रीर्विजयश्चैव सराष्ट्राणां भविष्यति।</div><div class="line" style="position:relative;">तथा स्फीतो जनपदो मुदितश्च भविष्यति॥३७॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b2653"><div class="verse-block verse-justify"><div class="line">नानाराज्येषु च सुतान्स सम्राडभ्यषेचयत्।</div><div class="line" style="position:relative;">महारथो मागधानां विश्रुतो यो बृहद्रथः॥४३॥</div></div></div>
