<div class="verse-block-wrapper" id="01-ādiparva-b8609"><div class="verse-block verse-justify"><div class="line">उवाच चैनं कल्याण्या वाचा मधुरयोत्थिम्।</div><div class="line" style="position:relative;">मा भैर्मनुजशार्दूल भद्रमस्तु तवानघ॥९॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b8611"><div class="verse-block verse-justify"><div class="line">वारिणा च सुशीतेन शिरस्तस्याभ्यषेचयत्।</div><div class="line" style="position:relative;">अस्पृशन्मुकुटं राज्ञः पुण्डरीकसुगन्धिना॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8612"><div class="verse-block verse-justify"><div class="line">ततः प्रत्यागतप्राणस्तद्बलं बलवान्नृपः।</div><div class="line" style="position:relative;">सर्वं विसर्जयामास तमेकं सचिवं विना॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b8614"><div class="verse-block verse-justify"><div class="line">ततस्तस्मिन् गिरिवरे शुचिर्भूत्वा कृताञ्जलिः।</div><div class="line" style="position:relative;">आरिराधयिषुः सूर्यं तस्थावूर्ध्वमुखः क्षितौ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8615"><div class="verse-block verse-justify"><div class="line">जगाम मनसा चैव वसिष्ठमृषिसत्तमम्।</div><div class="line" style="position:relative;">पुरोहितममित्रघ्नं तदा संवरणो नृपः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8616"><div class="verse-block verse-justify"><div class="line">नक्तन्दिनमथैकत्र स्थिते तस्मिञ्जनाधिपे।</div><div class="line" style="position:relative;">अथाजगाम विप्रर्षिस्तदा द्वादशमेऽहनि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8617"><div class="verse-block verse-justify"><div class="line">स विदित्वैव नृपतिं तपत्या हृतमानसम्।</div><div class="line" style="position:relative;">दिव्येन विधिना ज्ञात्वा भावितात्मा महानृपिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8618"><div class="verse-block verse-justify"><div class="line">तथा तु नियतात्मानं तं नृपं मुनिसत्तमः।</div><div class="line" style="position:relative;">आबभाषे स धर्मात्मा तस्यैवार्थचिकीर्षया॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8619"><div class="verse-block verse-justify"><div class="line">स तस्य मनुजेन्द्रस्य पश्यतो भगवानृषिः।</div><div class="line" style="position:relative;">ऊर्ध्वमाचक्रमे द्रष्टुं भास्करं भास्करद्युतिः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8620"><div class="verse-block verse-justify"><div class="line">`योजनानां तु नियुतं क्षणाद्गत्वा तपोधनः।</div><div class="line">'सहस्रांशुं ततो विप्रः कृताञ्जलिरुपस्थितः।</div><div class="line" style="position:relative;">वसिष्ठोहमिति प्रीत्या स चात्मानं न्यवेदयत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8621"><div class="verse-block verse-justify"><div class="line">तमुवाच महातेजा विवस्वान्मुनिसत्तमम्।</div><div class="line" style="position:relative;">महर्षे स्वागतं तेऽस्तु कथयस्व यथेप्सितम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8622"><div class="verse-block verse-justify"><div class="line">यदिच्छसि महाभाग मत्तः प्रवदतां वर।</div><div class="line" style="position:relative;">तत्ते दद्यामभिप्रेतं यद्यपि स्यात्सुदुर्लभम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8623"><div class="verse-block verse-justify"><div class="line">एवमुक्तः स तेनर्षिर्वसिष्ठः संस्तुवन्गिरा।</div><div class="line" style="position:relative;">प्रणिपत्य विवस्वन्तं भानुमन्तमथाब्रवीत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8624"><div class="verse-block verse-justify"><div class="line">`योजनानां चतुष्षष्टिं निमेषात्त्रिशतं तथा।</div><div class="line">अश्वैर्गच्छति नित्यं यस्तत्पार्श्वस्थोऽब्रवीदिदम् ॥वसिष्ठ उवाच।</div><div class="line" style="position:relative;">॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8625"><div class="verse-block verse-justify"><div class="line">अजाय लोकत्रयपावनायभूतात्मने गोपतये वृषाय।</div><div class="line" style="position:relative;">सूर्याय सर्गप्रलयालयायनमो महाकारुणिकोत्तमाय॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8626"><div class="verse-block verse-justify"><div class="line">विवस्वते ज्ञानभृतेऽन्तरात्मनेजगत्प्रदीपाय जगद्धितैषिणे।</div><div class="line" style="position:relative;">स्वयंभुवे दीप्तसहस्रचक्षुषेसुरोत्तमायामिततेजसे नमः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8627"><div class="verse-block verse-justify"><div class="line">नमः सवित्रे जगदेकचक्षुषेजगत्प्रसूतिस्थितिनाशहेतवे।</div><div class="line">त्रयीमयाय त्रिगुणात्मधारिणेविरिञ्चनारायणशङ्करात्मने ॥सूर्य उवाच।</div><div class="line" style="position:relative;">॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8628"><div class="verse-block verse-justify"><div class="line">संस्तुतो वरदः सोऽहं वरं वरय सुव्रत।</div><div class="line">स्तुतिस्त्वयोक्ता भक्तानां जप्येयं वग्दोस्म्यहम्' ॥वसिष्ठ उवाच।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8629"><div class="verse-block verse-justify"><div class="line">यैषा ते तपती नाम सावित्र्यवरजा सुता।</div><div class="line" style="position:relative;">तां त्वां संवरणस्यार्थे वरयामि विभावसो॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8630"><div class="verse-block verse-justify"><div class="line">स हि राजा बृहत्कीर्तिर्धर्मार्थविदुदारधीः।</div><div class="line">युक्तः संवरणो भर्ता दुहितुस्ते विहङ्गम ॥गन्धर्व उवाच।</div><div class="line" style="position:relative;">॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8631"><div class="verse-block verse-justify"><div class="line">इत्युक्तः स तदा तेन ददानीत्येव निश्चितः।</div><div class="line" style="position:relative;">प्रत्यभाषत तं विप्रं प्रतीनन्द्य दिवाकरः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8632"><div class="verse-block verse-justify"><div class="line">वरः संवरणो राज्ञां त्वमृषीणां वरो मुने।</div><div class="line" style="position:relative;">तपती योषितां श्रेष्ठा किमन्यत्रापवर्जनात्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8633"><div class="verse-block verse-justify"><div class="line">ततः सर्वानवद्याङ्गीं तपतीं तपनः स्वयम्।</div><div class="line" style="position:relative;">ददौ संवरणस्यार्थे वसिष्ठाय महात्मने॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8634"><div class="verse-block verse-justify"><div class="line">प्रतिजग्राह तां कन्यां महर्षिस्तपतीं तदा।</div><div class="line" style="position:relative;">वसिष्ठोऽथ विसृष्टस्तु पुनरेवाजगाम ह॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8635"><div class="verse-block verse-justify"><div class="line">यत्र विख्यातकीर्तिः स कुरूणामृषभोऽभवत्।</div><div class="line" style="position:relative;">स राजा मन्मथाविष्टस्तद्गतेनान्तरात्मना॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8636"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा च देवकन्यां तां तपतीं चारुहासिनीम्।</div><div class="line" style="position:relative;">वसिष्ठेन सहायान्तीं संहृष्टोऽभ्यधिकं बभौ॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8637"><div class="verse-block verse-justify"><div class="line">रुरुचे साऽधिकं सुभ्रूरापतन्ती नभस्तलात्।</div><div class="line" style="position:relative;">सौदामनीव विभ्रष्टा द्योतयन्ती दिशस्त्विषा॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8638"><div class="verse-block verse-justify"><div class="line">कृच्छ्राद्द्वादशरात्रे तु तस्य राज्ञः समाहिते।</div><div class="line" style="position:relative;">आजगाम विशुद्धात्मा वसिष्ठो भगवानृषिः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8639"><div class="verse-block verse-justify"><div class="line">तपसाऽऽराध्य वरदं देवं गोपतिमीश्वरम्।</div><div class="line" style="position:relative;">लेभे संवरणो भार्यां वसिष्ठस्यैव तेजसा॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8640"><div class="verse-block verse-justify"><div class="line">ततस्तस्मिन्गिरिश्रेष्ठे देवगन्धर्वसेविते।</div><div class="line" style="position:relative;">जग्राह विधिवत्पाणिं तपत्याः स नरर्षभः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8641"><div class="verse-block verse-justify"><div class="line">वसिष्ठेनाभ्यनुज्ञातस्तस्मिन्नेव धराधरे।</div><div class="line" style="position:relative;">सोऽकामयत राजर्षिर्विहर्तुं सह भार्यया॥४१॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b8645"><div class="verse-block verse-justify"><div class="line">तस्य राज्ञः पुरे तस्मिन्समा द्वादश सत्तम।</div><div class="line" style="position:relative;">न ववर्ष सहस्राक्षो राष्ट्रे चैवास्य भारत॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8646"><div class="verse-block verse-justify"><div class="line">ततस्तस्यामनावृष्ट्यां प्रवृत्तायामरिन्दम।</div><div class="line" style="position:relative;">प्रजाः क्षयमुपाजग्मुः सर्वाः सस्थाणुजङ्गमाः॥४६॥</div></div></div>
