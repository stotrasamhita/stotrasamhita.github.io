<div class="stotra-article"><h2 class="stotra-heading">अध्यायः १५७</h2>
<div class="verse-block-wrapper" id="01-ādiparva-b7276"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7277"><div class="verse-block verse-justify"><div class="line">पाण्डवास्तु रथान्युक्तान्सदश्वैरनिलोपमैः।</div><div class="line" style="position:relative;">आरोहमाणा भीष्मस्य पादौ जगृहुरार्तवत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7278"><div class="verse-block verse-justify"><div class="line">राज्ञश्च धृतराष्ट्रस्य द्रोणस्य च महात्मनः।</div><div class="line" style="position:relative;">अन्येषां चैव वृद्धानां कृपस्य विदुरस्य च॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7279"><div class="verse-block verse-justify"><div class="line">एवं सर्वान्कुरून्वृद्धानभिवाद्य यतव्रताः।</div><div class="line" style="position:relative;">समालिङ्ग्य समानान्वै बालैश्चाप्यभिवादिताः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7280"><div class="verse-block verse-justify"><div class="line">सर्वा मातॄस्तथाऽऽपृच्छ्य कृत्वा चैव प्रदक्षिणम्।</div><div class="line" style="position:relative;">प्रकृतीरपि सर्वाश्च प्रययुर्वारणावतम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7281"><div class="verse-block verse-justify"><div class="line">विदुरश्च महाप्राज्ञस्तथाऽन्ये कुरुपुङ्गवाः।</div><div class="line" style="position:relative;">पौराश्च पुरुषव्याघ्रानन्वीयुः शोककर्शिताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7282"><div class="verse-block verse-justify"><div class="line">तत्र केचिद्ब्रुवन्ति स्म ब्राह्मणा निर्भयास्तदा।</div><div class="line" style="position:relative;">दीनान्दृष्ट्वा पाण्डुसुतानतीव भृशदुःखिताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7283"><div class="verse-block verse-justify"><div class="line">विषमं पश्यते राजा सर्वथा स सुमन्दधीः।</div><div class="line" style="position:relative;">कौरव्यो धृतराष्ट्रस्तु न च धर्मं प्रपश्यति॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7284"><div class="verse-block verse-justify"><div class="line">न हि पापमपापात्मा रोचयिष्यति पाण्डवः।</div><div class="line" style="position:relative;">भीमो वा बलिनां श्रेष्ठः कौन्तेयो वा धनञ्जयः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7285"><div class="verse-block verse-justify"><div class="line">कुत एव महात्मानौ माद्रीपुत्रौ करिष्यतः।</div><div class="line" style="position:relative;">तान्राज्यं पितृतः प्राप्तान्धृतराष्ट्रो न मृष्यति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7286"><div class="verse-block verse-justify"><div class="line">अधर्म्यमिदमत्यन्तं कथं भीष्मोऽनुमन्यते।</div><div class="line" style="position:relative;">विवास्यमानानस्थाने नगरे योऽभिमन्यते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7287"><div class="verse-block verse-justify"><div class="line">पितेव हि नृपोऽस्माकमभूच्छान्तनवः पुरा।</div><div class="line" style="position:relative;">विचित्रवीर्यो राजर्षिः पाण्डुश्च कुरुनन्दनः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7288"><div class="verse-block verse-justify"><div class="line">स तस्मिन्पुरुषव्याघ्रे देवभावं गते सति।</div><div class="line" style="position:relative;">राजपुत्रानिमान्बालान्धृतराष्ट्रो न मृष्यति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7289"><div class="verse-block verse-justify"><div class="line">वयमेतदनिच्छन्तः सर्व एव पुरोत्तमात्।</div><div class="line">गृहान्विहाय गच्छामो यत्र गन्ता युधिष्ठिरः ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7290"><div class="verse-block verse-justify"><div class="line">तांस्तथा वादिनः पौरान्दुःकितान्दुःखकर्शितः।</div><div class="line" style="position:relative;">उवाच मनसा ध्यात्वा धर्मराजो युधिष्ठिरः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7291"><div class="verse-block verse-justify"><div class="line">पिता मान्यो गुरुः श्रेष्ठो यदाह पृथिवीपतिः।</div><div class="line" style="position:relative;">अशङ्कमानैस्तत्कार्यमस्माभिरिति नो व्रतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7292"><div class="verse-block verse-justify"><div class="line">भवन्तः सुहृदोऽस्माकं यात कृत्वा प्रदक्षिणम्।</div><div class="line" style="position:relative;">प्रतिनन्द्य तथाऽऽशीर्भिर्निवर्तध्वं यथागृहम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7293"><div class="verse-block verse-justify"><div class="line">यदा तु कार्यमस्माकं भवद्भिरुपपत्स्यते।</div><div class="line" style="position:relative;">तदा करिष्यथास्माकं प्रियाणि च हितानि च॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7294"><div class="verse-block verse-justify"><div class="line">एवमुक्तास्तदा पौराः कृत्वा चापि प्रदक्षिणम्।</div><div class="line" style="position:relative;">आशीर्भिश्चाभिनन्द्यैताञ्जग्मुर्नगरमेव हि॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7295"><div class="verse-block verse-justify"><div class="line">पौरेषु विनिवृत्तेषु विदुरः सर्वधर्मवित्।</div><div class="line" style="position:relative;">बोधयन्पाण्डवश्रेष्ठमिदंवचनमब्रवीत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7296"><div class="verse-block verse-justify"><div class="line">प्राज्ञः प्राज्ञं प्रलापज्ञः प्रलापज्ञमिदं वचः।</div><div class="line">यो जानाति परप्रज्ञां नीतिशास्त्रानुसारिणीम्।</div><div class="line" style="position:relative;">विज्ञायेह तथा कुर्यादापदं निस्तरेद्यथा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7297"><div class="verse-block verse-justify"><div class="line">अलोहं निशितं शस्त्रं शरीपरिकर्तनम्।</div><div class="line" style="position:relative;">यो वेत्ति न तु तं घ्नन्ति प्रतिघातविदं द्विषः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7298"><div class="verse-block verse-justify"><div class="line">कक्षघ्नः शिशिरघ्नश्च महाकक्षे बिलौकसः।</div><div class="line" style="position:relative;">न दहेदिति चात्मानं यो रक्षति स जीवति॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7299"><div class="verse-block verse-justify"><div class="line">नाचक्षुर्वेत्ति पन्थानं नाचक्षुर्विन्दते दिशः।</div><div class="line" style="position:relative;">नाधृतिर्भूतिमाप्नोति बुध्यस्वैवं प्रबोधितः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7300"><div class="verse-block verse-justify"><div class="line">अनाप्तैर्दत्तमादत्ते नरः शस्त्रमलोहजम्।</div><div class="line" style="position:relative;">श्वाविच्छरणमासाद्य प्रमुच्येत हुताशनात्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7301"><div class="verse-block verse-justify"><div class="line">चरन्मार्गान्विजानाति नक्षत्रैर्विन्दते दिशः।</div><div class="line" style="position:relative;">आत्मना चात्मनः पञ्च पीडयन्नानुपीड्यते॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7302"><div class="verse-block verse-justify"><div class="line">एवमुक्तः प्रत्युवाच धर्मराजो युधिष्ठिरः।</div><div class="line" style="position:relative;">विदुरं विदुषां श्रेष्ठं ज्ञातमित्येव पाण्डवः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7303"><div class="verse-block verse-justify"><div class="line">अनुशिक्ष्यानुगम्यैतान्कृत्वा चैव प्रदक्षिणम्।</div><div class="line" style="position:relative;">पाण्डवानभ्यनुज्ञाय विदुरः प्रययौ गृहान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7304"><div class="verse-block verse-justify"><div class="line">निवृत्ते विदुरे चापि भीष्मे पौरजने तथा।</div><div class="line" style="position:relative;">अजातशत्रुमासाद्य कुन्ती वचनमब्रवीत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7305"><div class="verse-block verse-justify"><div class="line">क्षत्ता यदब्रवीद्वाक्यं जनमध्येऽब्रुवन्निव।</div><div class="line" style="position:relative;">त्वया च स तथेत्युक्तो जानीमो न च तद्वयम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7306"><div class="verse-block verse-justify"><div class="line">यदीदं शक्यमस्माभिर्ज्ञातुं नैव च दोषवत्।</div><div class="line">श्रोतुमिच्छामि तत्सर्वं संवादं तव तस्य च ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7307"><div class="verse-block verse-justify"><div class="line">विषादग्नेश्च बोद्धव्यमिति मां विदुरोऽब्रवीत्।</div><div class="line">पन्थानो वेदितव्याश्च नक्षत्रैश्च तथा दिशः।</div><div class="line" style="position:relative;">`कुड्याश्चविदिताःकार्याःस्याच्छुद्धिरितिचाब्रवीत्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7308"><div class="verse-block verse-justify"><div class="line">जितेन्द्रियश्च वसुधां प्राप्स्यतीति च मेऽब्रवीत्।</div><div class="line">विज्ञातमिति तत्सर्वं प्रत्युक्तो विदुरो मया ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥३३॥</div></div></div>
