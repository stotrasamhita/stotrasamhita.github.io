<div class="verse-block-wrapper" id="01-ādiparva-b9012"><div class="verse-block verse-justify"><div class="line">अवाप्य धृष्टद्युम्नं हि न स द्रोणमचिन्तयत्।</div><div class="line" style="position:relative;">स तु वैरप्रसङ्गाच्च भीष्माद्भयमचिन्तयत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9013"><div class="verse-block verse-justify"><div class="line">कन्यादानात्तु शरणं सोऽमन्यत महीपतिः।</div><div class="line" style="position:relative;">'यज्ञसेनस्य कामस्तु पाण्डवाय किरीटिने॥११॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b9015"><div class="verse-block verse-justify"><div class="line">सोऽन्वेषमाणः कौन्तेयान्पाञ्चालो जनमेजय।</div><div class="line" style="position:relative;">दृढं धनुरथानम्यं कारयामास भारत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9016"><div class="verse-block verse-justify"><div class="line">`वैयाघ्रपद्यस्योग्रं वै सृञ्जयस्य महीपतिः।</div><div class="line" style="position:relative;">तद्धनुः किन्धुरं नाम देवदत्तमुपानयत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9017"><div class="verse-block verse-justify"><div class="line">आयसी तस्य च ज्याऽऽसीत्प्रतिबद्धा महाबला।</div><div class="line" style="position:relative;">न तु ज्यां प्रसहेदन्यस्तद्धनुःप्रवरं महत्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9018"><div class="verse-block verse-justify"><div class="line">शङ्करेण वरं दत्तं प्रीतेन च महात्मना।</div><div class="line" style="position:relative;">तन्निष्फलं स्यान्न तु मे इति प्रामाण्यमागतः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9019"><div class="verse-block verse-justify"><div class="line">मया कर्तव्यमधुना दुष्करं लक्ष्यवेधनम्।</div><div class="line" style="position:relative;">इति निश्चित्य मनसा कारितं लक्ष्यमुत्तमम् ॥'॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9020"><div class="verse-block verse-justify"><div class="line">यन्त्रं वैहायसं चापि कारयामास कृत्रिमम्।</div><div class="line">तेन यन्त्रेण सहितं राजँल्लक्ष्यं च काञ्चनम् ॥द्रुपद उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9021"><div class="verse-block verse-justify"><div class="line">इदं सज्यं धनुः कृत्वा सज्जैरेभिश्च सायकैः।</div><div class="line">अतीत्य लक्ष्य यो वेद्धा स लब्धा मत्सुतामिति ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9022"><div class="verse-block verse-justify"><div class="line">इति स द्रुपदो राजा स्वयंवरमघोषयत्।</div><div class="line" style="position:relative;">तच्छ्रुत्वा पार्थिवाः सर्वे समीयुस्तत्र भारत॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9023"><div class="verse-block verse-justify"><div class="line">ऋषयश्च महात्मानः स्वयंवरदिदृक्षवः।</div><div class="line" style="position:relative;">दुर्योधनपुरोगाश्च सकर्णाः कुरवो नृप॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9024"><div class="verse-block verse-justify"><div class="line">`यादवा वासुदेवेन सार्धमन्धकवृष्णयः।</div><div class="line" style="position:relative;">'ततोऽर्चिता राजगुणा द्रुपदेन महात्मना॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9025"><div class="verse-block verse-justify"><div class="line">उपोपविष्टा मञ्चेषु द्रष्टुकामाः स्वयंवरम्।</div><div class="line" style="position:relative;">`ब्राह्मणाश्च महाभागा देशेभ्यः समुपागमन्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9026"><div class="verse-block verse-justify"><div class="line">ब्राह्मणैरेव सहिताः पाण्डवाः समुपाविशन्।</div><div class="line" style="position:relative;">त्रयस्त्रिंशत्सुराः सर्वे विमानैर्व्योम्न्यवस्थिताः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9027"><div class="verse-block verse-justify"><div class="line">ततः पौरजनाः सर्वे सागरोद्धूतनिःस्वनाः।</div><div class="line" style="position:relative;">शिंशुमारशिरः प्राप्य न्यविशंस्ते स्म पार्थिवाः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9028"><div class="verse-block verse-justify"><div class="line">प्रागुत्तरेण नगराद्भूमिभागे समे शुभे।</div><div class="line" style="position:relative;">समाजवाटः शुशुभे भवनैः सर्वतो वृतः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9029"><div class="verse-block verse-justify"><div class="line">प्राकारपरिखोपेतो द्वारतोरणमण्डितः।</div><div class="line" style="position:relative;">वितानेन विचित्रेण सर्वतः समलङ्कृतः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9030"><div class="verse-block verse-justify"><div class="line">तूर्यौघशतसङ्कीर्णः परार्ध्यागुरुधूपितः।</div><div class="line" style="position:relative;">चन्दनोदकसिक्तश्च माल्यदामोपशोभितः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9031"><div class="verse-block verse-justify"><div class="line">कैलासशिखरप्रख्यैर्नभस्तलविलेखिभिः।</div><div class="line" style="position:relative;">सर्वतः संवृतः शुभ्रैः प्रासादैः सुकृतोच्छ्रयै॥२९॥</div></div></div>
