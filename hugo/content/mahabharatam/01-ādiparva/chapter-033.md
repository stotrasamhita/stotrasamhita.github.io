<div class="verse-block-wrapper" id="01-ādiparva-b1599"><div class="verse-block verse-justify"><div class="line">स चक्रं क्षुरपर्यन्तमपश्यदमृतान्तिके।</div><div class="line" style="position:relative;">परिभ्रमन्तमनिशं तीक्ष्णधारमयस्मयम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b1601"><div class="verse-block verse-justify"><div class="line">तस्यान्तरं स दृष्ट्वै पर्यवर्तत खेचरः।</div><div class="line" style="position:relative;">अरान्तरेणाभ्यपतत्संक्षिप्याङ्गं क्षणेन ह॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1602"><div class="verse-block verse-justify"><div class="line">अधश्चक्रस्य चैवात्र दीप्तानलसमद्व्युती।</div><div class="line" style="position:relative;">विद्युज्जिह्वौ महावीर्यौ दीप्तास्यौ दीप्तलोचनौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1603"><div class="verse-block verse-justify"><div class="line">चक्षुर्विषौ महाघोरौ नित्यं क्रुद्धौ तरस्विनौ।</div><div class="line" style="position:relative;">अमृतस्यैव रक्षार्थं ददर्श भुजगोत्तमौ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1604"><div class="verse-block verse-justify"><div class="line">सदा संरब्धनयनौ सदा चानिमिषेक्षणौ।</div><div class="line" style="position:relative;">तयोरेकोऽपि यं पश्येत्स तूर्णं भस्मसाद्भवेत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1605"><div class="verse-block verse-justify"><div class="line">`तौ दृष्ट्वा सहसा खेदं जगाम विनतात्मजः।</div><div class="line" style="position:relative;">कथमेतौ महावीर्यौ जेतव्यौ हरिभोजिनौ॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1606"><div class="verse-block verse-justify"><div class="line">इति संचिन्त्य गरुडस्तयोस्तूर्णं निराकरः।</div><div class="line">'तयोश्चक्षूंषि रजसा सुपर्णः सहसाऽऽवृणोत्।</div><div class="line" style="position:relative;">ताभ्यामदृष्टरूपोऽसौ सर्वतः समताडयत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1607"><div class="verse-block verse-justify"><div class="line">तयोरङ्गे समाक्रम्य वैनतेयोऽन्तरिक्षगः।</div><div class="line" style="position:relative;">आच्छिनत्तरसा मध्ये सोममभ्यद्रवत्ततः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1608"><div class="verse-block verse-justify"><div class="line">समुत्पाट्यामृतं तत्र वैनतेयस्ततो बली।</div><div class="line" style="position:relative;">उत्पपात जवेनैव यन्त्रमुन्मथ्य वीर्यवान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1609"><div class="verse-block verse-justify"><div class="line">अपीत्वैवामृतं पक्षी परिगृह्याशु निःसृतः।</div><div class="line" style="position:relative;">आगच्छदपरिश्रान्त आवार्यार्कप्रभां ततः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1610"><div class="verse-block verse-justify"><div class="line">विष्णुना च तदाकाशे वैनतेयः समेयिवान्।</div><div class="line" style="position:relative;">तस्य नारायणस्तुष्टस्तेनालौल्येन कर्मणा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1611"><div class="verse-block verse-justify"><div class="line">तमुवाचाव्ययो देवो वरदोऽस्मीति खेचरम्।</div><div class="line" style="position:relative;">स वव्रे तव तिष्ठेयमुपरीत्यन्तरिक्षगः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1612"><div class="verse-block verse-justify"><div class="line">उवाच चैनं भूयोऽपि नारायणमिदं वचः।</div><div class="line">अजरश्चामरश्च स्याममृतेन विनाऽप्यहम् ॥सौतिरुवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1613"><div class="verse-block verse-justify"><div class="line">एवमस्त्विति तं विष्णुरुवाच विनतासुतम्।</div><div class="line" style="position:relative;">प्रतिगृह्य वनौ तौ च गरुडो विष्णुमब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1614"><div class="verse-block verse-justify"><div class="line">भवतेपि वरं दद्यां वृणोतु भगवानपि।</div><div class="line" style="position:relative;">तं वव्रे वाहनं विष्णुर्नरुत्मन्तं महाबलम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1615"><div class="verse-block verse-justify"><div class="line">ध्वजं च चक्रे भगवानुपरि स्थास्यसीति तम्।</div><div class="line" style="position:relative;">एवमस्त्विति तं देवमुक्त्वा नारायणं खगः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1616"><div class="verse-block verse-justify"><div class="line">वव्राज तरसा वेगाद्वायुं स्पर्धन्महाजवः।</div><div class="line" style="position:relative;">तं व्रजन्तं खगश्रेष्ठं वज्रेणेन्द्रोऽभ्यताडयत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1617"><div class="verse-block verse-justify"><div class="line">हरन्तममृतं रोषाद्गरुडं पक्षिणां वरम्।</div><div class="line" style="position:relative;">तमुवाचेन्द्रमाक्रन्दे गरुडः पततां वरः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1618"><div class="verse-block verse-justify"><div class="line">प्रहसञ्श्लक्ष्णया वाचा तथा वज्रसमाहतः।</div><div class="line" style="position:relative;">ऋषेर्मानं करिष्यामि वज्रं यस्यास्थिसंभवम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1619"><div class="verse-block verse-justify"><div class="line">वज्रस्य च करिष्यामि तवैव च शतक्रतो।</div><div class="line" style="position:relative;">एतत्पत्रं त्यजाम्येकं यस्यान्तं नोपलप्स्यसे॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1620"><div class="verse-block verse-justify"><div class="line">न च वज्रनिपातेन रुजा मेऽस्तीह काचन।</div><div class="line" style="position:relative;">एवमुक्त्वा ततः पुत्रमुत्ससर्ज स पक्षिराट्॥२४॥</div></div></div>
