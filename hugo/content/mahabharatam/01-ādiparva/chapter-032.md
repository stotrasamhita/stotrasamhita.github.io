<div class="verse-block-wrapper" id="01-ādiparva-b1570"><div class="verse-block verse-justify"><div class="line">सौतिरुवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1571"><div class="verse-block verse-justify"><div class="line">`ततस्तस्माद्गिरिवरात्समुदीर्णमहाबलः।</div><div class="line" style="position:relative;">'गरुडः पक्षिराट् तूर्णं संप्राप्तो विबुधान्प्रति॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1572"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वातिबलं चैव प्राकम्पन्त सुरास्ततः।</div><div class="line" style="position:relative;">परस्परं च प्रत्यघ्नन्सर्वप्रहरणान्युत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1573"><div class="verse-block verse-justify"><div class="line">तत्र चासीदमेयात्मा विद्युदग्निसमप्रभः।</div><div class="line" style="position:relative;">भौमनः सुमहावीर्यः सोमस्य परिरक्षिता॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1574"><div class="verse-block verse-justify"><div class="line">स तेन पतगेन्द्रेण पक्षतुण्डनखैः क्षतः।</div><div class="line" style="position:relative;">मुहूर्तमतुलं युद्धं कृत्वा विनिहतो युधि॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1575"><div class="verse-block verse-justify"><div class="line">रजश्चोद्धूय सुमहत्पक्षवातेन खेचरः।</div><div class="line" style="position:relative;">कृत्वा लोकान्निरालोकांस्तेन देवानवाकिरत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1576"><div class="verse-block verse-justify"><div class="line">तेनावकीर्णा रजसा देवा मोहमुपागमन्।</div><div class="line" style="position:relative;">न चैवं ददृशुश्छन्ना रजसाऽमृतरक्षिणः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1577"><div class="verse-block verse-justify"><div class="line">एवं संलोडयामास गरुडस्त्रिदिवालयम्।</div><div class="line" style="position:relative;">पक्षतुण्डप्रहारैस्तु देवान्स विददार ह॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1578"><div class="verse-block verse-justify"><div class="line">ततो देवः सहस्राक्षस्तूर्णं वायुमचोदयत्।</div><div class="line">विक्षिपेमां रजोवृष्टिं तवेदं कर्म मारुत ॥सौतिरुवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1579"><div class="verse-block verse-justify"><div class="line">अथ वायुरपोवाह तद्रजस्तरसा बली।</div><div class="line" style="position:relative;">ततो वितिमिरे जाते देवाः शकुनिमार्दयन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1580"><div class="verse-block verse-justify"><div class="line">ननादोच्चैः स बलवान्महामेघ इवाम्बरे।</div><div class="line" style="position:relative;">वध्यमानः सुरगणैः सर्वभूतानि भीषयन्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1581"><div class="verse-block verse-justify"><div class="line">उत्पपात महावीर्यः पक्षिराट् परवीरहा।</div><div class="line" style="position:relative;">समुत्पत्यान्तरिक्षस्थं देवानामुपरि स्थितम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1582"><div class="verse-block verse-justify"><div class="line">वर्मिमो विबुधाः सर्वे नानाशस्त्रैरवाकिरन्।</div><div class="line" style="position:relative;">पट्टिशैः परिधैः शूलैर्गदाभिश्च सवासवाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1583"><div class="verse-block verse-justify"><div class="line">क्षुरप्रैर्ज्वलितैश्चापि चक्रैरादित्यरूपिभिः।</div><div class="line" style="position:relative;">नानाशस्त्रविसर्गैस्तैर्वध्यमानः समन्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1584"><div class="verse-block verse-justify"><div class="line">कुर्वन्सुतुमुलं युद्धं पक्षिराण्ण व्यकम्पत।</div><div class="line">निर्दहन्निव चाकाशे वैनतेयः प्रतापवान्।</div><div class="line" style="position:relative;">पक्षाभ्यामुरसा चैव समन्ताद्व्यक्षिपत्सुरान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1585"><div class="verse-block verse-justify"><div class="line">ते विक्षिप्तास्ततो देवा दुद्रुवुर्गरुडार्दिताः।</div><div class="line" style="position:relative;">नखतुण्डक्षताश्चैव सुस्रुवुः शोणितं बहु॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1586"><div class="verse-block verse-justify"><div class="line">साध्याः प्राचीं सगन्धर्वा वसवो दक्षिणां दिशम्।</div><div class="line" style="position:relative;">प्रजग्मुः सहिता रुद्राः पतगेन्द्रप्रधर्षिताः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1587"><div class="verse-block verse-justify"><div class="line">दिशं प्रतीचीमादित्या नासत्यावुत्तरां दिशम्।</div><div class="line" style="position:relative;">मुहुर्मुहुः प्रेक्षमाणा युध्यमानं महौजसः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1588"><div class="verse-block verse-justify"><div class="line">अश्वक्रन्देन वीरेण रेणुकेन च पक्षिराट्।</div><div class="line" style="position:relative;">क्रथनेन च शूरेण तपनेन च खेचरः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1589"><div class="verse-block verse-justify"><div class="line">उलूकश्वसनाभ्यां च निमेषेण च पक्षिराट्।</div><div class="line" style="position:relative;">प्ररुजेन च संग्रामं चकार पुलिनेन च॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1590"><div class="verse-block verse-justify"><div class="line">तान्पक्षनखतुण्डाग्रैरभिनद्विनतासुतः।</div><div class="line" style="position:relative;">युगान्तकाले संक्रुद्धः पिनाकीव परंतप॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1591"><div class="verse-block verse-justify"><div class="line">महाबला महोत्साहास्तेन ते बहुधा क्षताः।</div><div class="line" style="position:relative;">रेजुरभ्रघनप्रख्या रुधिरौघप्रवर्षिणः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1592"><div class="verse-block verse-justify"><div class="line">तान्कृत्वा पतगश्रेष्ठः सर्वानुत्क्रान्तजीवितान्।</div><div class="line" style="position:relative;">अतिक्रान्तोऽमृतस्यार्थे सर्वतोऽग्निमपश्यत॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1593"><div class="verse-block verse-justify"><div class="line">आवृण्वानं महाज्वालमर्चिर्भिः सर्वतोऽम्बरम्।</div><div class="line" style="position:relative;">दहन्तमिव तीक्ष्णांशुं चण्डवायुसमीरितम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1594"><div class="verse-block verse-justify"><div class="line">`नभः स्पृशन्तं ज्वालाभिः सर्वभूतभयंकरम्।</div><div class="line">'ततो नवत्या नवतीर्मुखानांकृत्वा महात्मा गरुडस्तरस्वी।</div><div class="line" style="position:relative;">नदीः समापीय मुखैस्ततस्तैःसुशीघ्रमागम्य पुनर्जवेन॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b1595"><div class="verse-block verse-justify"><div class="line">ज्वलन्तमग्निं तममित्रतापनःसमास्तरत्पत्ररथो नदीभिः।</div><div class="line" style="position:relative;">ततः प्रचक्रे वपुरन्यदल्पंप्रवेष्टुकामोऽग्निमभिप्रशाम्य॥२६॥</div></div></div></div>
