<div class="verse-block-wrapper" id="01-ādiparva-b8418"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8419"><div class="verse-block verse-justify"><div class="line">गते भगवति व्यासे पाण्डवा हृष्टमानसाः।</div><div class="line" style="position:relative;">`ते प्रयाता नरव्याघ्रा मात्रा सह परन्तपाः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8420"><div class="verse-block verse-justify"><div class="line">ब्राह्मणान्गच्छतो पश्यन्पाञ्चालान्सगणान्पथि।</div><div class="line" style="position:relative;">अथ ते ब्राह्मणा ऊचुः पाण्डवान्ब्रह्मचारिणः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8421"><div class="verse-block verse-justify"><div class="line">क्व भवन्तो गमिष्यन्ति कुतो वाऽऽगच्छथेति ह।</div><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">प्रयातानेकचक्रायाः सोदर्यान्देवदर्शिनः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8422"><div class="verse-block verse-justify"><div class="line">भवन्तो नोऽभिजानन्तु सहितान्ब्रह्मचारिणः।</div><div class="line" style="position:relative;">गच्छतो नस्तु पाञ्चालान्द्रुपदस्य पुरं प्रति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8423"><div class="verse-block verse-justify"><div class="line">इच्छामो भवतो ज्ञातुं परं कौतूहलं हि नः ॥ब्राह्मणा ऊचुः।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8424"><div class="verse-block verse-justify"><div class="line">एते सार्धं प्रयाताः स्मो वयमप्यत्र गामिनः।</div><div class="line" style="position:relative;">तत्राप्यद्भुतसङ्काश उत्सवो भविता महान्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8425"><div class="verse-block verse-justify"><div class="line">ततस्तु यज्ञसेनस्य द्रुपदस्य महात्मनः।</div><div class="line" style="position:relative;">यासावयोनिजा कन्या स्थास्यते सा स्वयंवरे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8426"><div class="verse-block verse-justify"><div class="line">दर्शनीयाऽनवद्याङ्गी सुकुमारी यशस्विनी।</div><div class="line" style="position:relative;">धृष्टद्युम्नस्य भगिनी द्रोणशत्रोः प्रतापिनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8427"><div class="verse-block verse-justify"><div class="line">जातो यः पावकाच्छूरः सशरः सशरासनः।</div><div class="line" style="position:relative;">सुसमिद्धान्महाभागः सोमकानां महारथः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8428"><div class="verse-block verse-justify"><div class="line">यस्मिन्संजायमाने हि वागुवाचाशरीरिणी।</div><div class="line" style="position:relative;">एष मृत्युश्च शिष्यश्च भारद्वाजस्य जायते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8429"><div class="verse-block verse-justify"><div class="line">स्वसा तस्य तु वेद्याश्च जाता तस्मिन्महामखे।</div><div class="line" style="position:relative;">स्त्रीरत्नमसितापाङ्गी श्यामा नीलोत्पलद्युतिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8430"><div class="verse-block verse-justify"><div class="line">तां यज्ञसेनस्य सुतां द्रौपदीं परमां स्त्रियम्।</div><div class="line" style="position:relative;">गच्छामस्तत्र वै द्रष्टुं तं चैवास्याः स्वयंवरम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8431"><div class="verse-block verse-justify"><div class="line">राजानो राजपुत्राश्च यज्वानो भूरिदक्षिणाः।</div><div class="line" style="position:relative;">स्वाध्यायवन्तः शुचयो महात्मानो धृतव्रताः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8432"><div class="verse-block verse-justify"><div class="line">तरुणा दर्शनीयाश्च बलवन्तो दुरासदाः।</div><div class="line" style="position:relative;">महारथाः कृतास्त्राश्च समेष्यन्तीह भूमिपाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8433"><div class="verse-block verse-justify"><div class="line">ते तत्र विविधं दानं विजयार्थं नरेश्वराः।</div><div class="line" style="position:relative;">प्रदास्यन्ति धनं गाश्च भक्ष्यभोज्यानि सर्वशः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8434"><div class="verse-block verse-justify"><div class="line">प्रतिलप्स्यामहे सर्वं दृष्ट्वा कृष्णां स्वयंवरे।</div><div class="line" style="position:relative;">यं च सा क्षत्रियं रङ्गे कुमारी वरयिष्यति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8435"><div class="verse-block verse-justify"><div class="line">तदा वैतालिकाश्चैव नर्तकाः सूतमागधाः।</div><div class="line" style="position:relative;">निबोधकाश्च देशेभ्यः समेष्यन्ति महाबलाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8436"><div class="verse-block verse-justify"><div class="line">एतत्कौतूहलं तत्र दृष्ट्वा वै प्रतिगृह्य च।</div><div class="line" style="position:relative;">सहास्माभिर्महात्मानो मात्रा सह निवत्स्यथ॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8437"><div class="verse-block verse-justify"><div class="line">दर्शनीयांश्च वः सर्वानेकरूपानवस्थितान्।</div><div class="line" style="position:relative;">समीक्ष्य कृष्मा वरयेत्संगत्यान्यतमं पतिम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8438"><div class="verse-block verse-justify"><div class="line">अयमेकश्च वो भ्राता दर्शनीयो महाभुजः।</div><div class="line">नियुध्यमानो विजयेत्संगत्य द्रविणं महत् ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8439"><div class="verse-block verse-justify"><div class="line">परमं भो गमिष्यामो द्रष्टुं तत्र स्वयंवरम्।</div><div class="line" style="position:relative;">द्रौपदीं यज्ञसेनस्य कन्यां तस्यास्तथोत्सवम् ॥'॥२२॥</div></div></div></div>
