<div class="verse-block-wrapper" id="01-ādiparva-b9136"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9137"><div class="verse-block verse-justify"><div class="line">यदा निवृत्ता राजानो धनुषः सज्यकर्मणः।</div><div class="line" style="position:relative;">अथोदतिष्ठद्विप्राणां मध्याज्जिष्णुरुदारधीः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9138"><div class="verse-block verse-justify"><div class="line">उदक्रोशन्विप्रमुख्या विधुन्वन्तोऽजिनानि च।</div><div class="line" style="position:relative;">दृष्ट्वा संप्रस्थितं पार्थमिन्द्रकेतुसमप्रभम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9139"><div class="verse-block verse-justify"><div class="line">केचिदासन्विमनसः केचिदासन्मुदान्विताः।</div><div class="line" style="position:relative;">आहुः परस्परं केचिन्निपुणा बुद्धिजीविनः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9140"><div class="verse-block verse-justify"><div class="line">यत्कर्णशल्यप्रमुखैः क्षत्रियैर्लोकविश्रुतैः।</div><div class="line" style="position:relative;">नानतं बलवद्भिर्हि धनुर्वेदपरायणैः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9141"><div class="verse-block verse-justify"><div class="line">तत्कथं त्वकृतास्त्रेण प्राणतो दुर्बलीयसा।</div><div class="line" style="position:relative;">वटुमात्रेण शक्यं हि सज्यं कर्तुं धनुर्द्विजाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9142"><div class="verse-block verse-justify"><div class="line">अवहास्या भविष्यन्ति ब्राह्मणाः सर्वराजसु।</div><div class="line" style="position:relative;">कर्मण्यस्मिन्नसंसिद्धे चापलादपरीक्षिते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9143"><div class="verse-block verse-justify"><div class="line">यद्येष दर्पाद्धर्षाद्वाप्यथ ब्राह्मणचापलात्।</div><div class="line" style="position:relative;">प्रस्थितो धनुरायन्तुं वार्यतां साधु मा गमत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9144"><div class="verse-block verse-justify"><div class="line">नावहास्या भविष्यामो न च लाघवमास्थिताः।</div><div class="line" style="position:relative;">न च विद्विष्टतां लोके गमिष्यामो महीक्षिताम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9145"><div class="verse-block verse-justify"><div class="line">केचिदाहुर्युवा श्रीमान्नागराजकरोपमः।</div><div class="line" style="position:relative;">पीनस्कन्धोरुबाहुश्च धैर्येण हिमवानिव॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9146"><div class="verse-block verse-justify"><div class="line">सिंहखेलगतिः श्रीमान्मत्तनागेन्द्रविक्रमः।</div><div class="line" style="position:relative;">संभाव्यमस्मिन्कर्मेदमुत्साहाच्चानुमीयते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9147"><div class="verse-block verse-justify"><div class="line">शक्तिरस्य महोत्साहा न ह्यशक्तः स्वयं व्रजेत्।</div><div class="line" style="position:relative;">न च तद्विद्यते किंचित्कर्म लोकेषु यद्भवेत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9148"><div class="verse-block verse-justify"><div class="line">ब्राह्मणानामसाध्यं च नृषु संस्थानचारिषु।</div><div class="line" style="position:relative;">अब्भक्षा वायुभक्षाश्च फलाहारा दृढव्रताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9149"><div class="verse-block verse-justify"><div class="line">दुर्बला अपि विप्रा हि बलीयांसः स्वेतजसा।</div><div class="line" style="position:relative;">ब्राह्मणो नावमन्तव्यः सदसद्वा समाचरन्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9150"><div class="verse-block verse-justify"><div class="line">सुखं दुःखं महद्ध्रस्वं कर्म यत्समुपागतम्।</div><div class="line" style="position:relative;">जामदग्न्येन रामेण निर्जिताः क्षत्रिया युधि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9151"><div class="verse-block verse-justify"><div class="line">पीतः समुद्रोऽगस्त्येन अगाधो ब्रह्मतेजसा।</div><div class="line" style="position:relative;">तस्माद्ब्रुवन्तु सर्वेऽत्र वटुरेष धनुर्महान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9152"><div class="verse-block verse-justify"><div class="line">आरोपयतु शीघ्रं वै तथेत्यूचुर्द्विजर्षभाः।</div><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">एवं तेषां विलपतां विप्राणां विविधा गिरः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9153"><div class="verse-block verse-justify"><div class="line">अर्जुनो धनुषोऽभ्याशे तस्थौ गिरिरिवाचलः।</div><div class="line" style="position:relative;">`अर्जुनः पाण्डवश्रेष्ठो धृष्टद्युम्नमथाब्रवीत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9154"><div class="verse-block verse-justify"><div class="line">एतद्धनुर्ब्राह्मणानां सज्यं कर्तुमलं तु किम्।</div><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">तस्य तद्वचनं श्रुत्वा धृष्टद्युम्नोऽब्रवीद्वचः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9155"><div class="verse-block verse-justify"><div class="line">ब्राह्मणो वाथ राजन्यो वैश्यो वा शूद्र एव वा।</div><div class="line" style="position:relative;">एतेषां यो धनुःश्रेष्ठं सज्यं कुर्याद्द्विजोत्तम॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9156"><div class="verse-block verse-justify"><div class="line">तस्मै प्रदेया भगिनी सत्यमुक्तं मया वचः ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9157"><div class="verse-block verse-justify"><div class="line">ततः पश्चान्महातेजाः पाण्डवो रणदुर्जयः।</div><div class="line" style="position:relative;">'स तद्धनुः परिक्रम्य प्रदक्षिणमथाकरोत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9158"><div class="verse-block verse-justify"><div class="line">प्रणम्य शिरसा देवमीनं वरदं प्रभुम्।</div><div class="line" style="position:relative;">कृष्णं च मनसा कृत्वा जगृहे चार्जुनो धनुः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9159"><div class="verse-block verse-justify"><div class="line">यत्पार्थिवै रुक्मसुनीथवक्रैराधेयदुर्योधनशल्यसाल्वैः।</div><div class="line" style="position:relative;">तदा धनुर्वेदपरैर्नृसिंहैःकृतं न सज्यं महतोऽपि यत्नात्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9160"><div class="verse-block verse-justify"><div class="line">तदर्जुनो वीर्यवतां सदर्प-स्तदैन्द्रिरिन्द्रावरजप्रभावः।</div><div class="line" style="position:relative;">सज्यं च चक्रे निमिषान्तरेणशरांश्च जग्राह दशार्दसङ्ख्यान्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9161"><div class="verse-block verse-justify"><div class="line">विव्याध लक्ष्यं निपपात तच्चछिद्रेण भूमौ सहसातिविद्धम्।</div><div class="line" style="position:relative;">ततोऽन्तरिक्षे च बभूव नादःसमाजमध्ये च महान्निनादः॥२६॥</div></div></div>
<p class="prose">पुष्पाणि दिव्यानि ववर्ष देवःपार्थस्य मूर्ध्नि द्विषतां निहन्तुः</p>
<div class="verse-block-wrapper" id="01-ādiparva-b9163"><div class="verse-block verse-justify"><div class="line">चेलानि विव्यधुस्तत्र ब्राह्मणाश्च सहस्रशः।</div><div class="line">विलक्षितास्ततश्चक्रुर्हाहाकारांश्च सर्वशः।</div><div class="line" style="position:relative;">न्यपतंश्चात्र नभसः समन्तात्पुष्पवृष्टयः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9164"><div class="verse-block verse-justify"><div class="line">शताङ्गानि च तूर्याणि वादकाः समवादयन्।</div><div class="line" style="position:relative;">सूतमागधसङ्घाश्चाप्यस्तुवंस्तत्र सुस्वराः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9165"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा द्रुपदः प्रीतो बभूव रिपुसूदनः।</div><div class="line" style="position:relative;">सह सैन्यैश्च पार्थस्य साहाय्यार्थमियेष सः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9166"><div class="verse-block verse-justify"><div class="line">तस्मिंस्तु शब्दे महति प्रवृद्धेयुधिष्ठिरो धर्मभृतां वरिष्ठः।</div><div class="line" style="position:relative;">आवासमेवोपजगाम शीघ्रंसार्धं यमाभ्यां पुरुषोत्तमाभ्याम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9167"><div class="verse-block verse-justify"><div class="line">विद्धं तु लक्ष्यं प्रसमीक्ष्य कृष्णापार्थं च शक्रप्रतिमं निरीक्ष्य।</div><div class="line" style="position:relative;">`स्वभ्यस्तरूपापि नवेव नित्यंविनापि हासं हसतीव कन्या॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9168"><div class="verse-block verse-justify"><div class="line">मदादृतेऽपि स्खलतीव भावै-र्वाचा विना व्याहरतीव दृष्ट्या।</div><div class="line" style="position:relative;">आदाय शुक्लं वरमाल्यदामजगाम कुन्तीसुतमुत्स्मयन्ती॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9169"><div class="verse-block verse-justify"><div class="line">गत्वा च पश्चात्प्रसमीक्ष्य कृष्णापार्थस्य वक्षस्यविशङ्कमाना।</div><div class="line" style="position:relative;">क्षिप्त्वा स्रजं पार्थिववीरमध्येवराय वव्रे द्विजसङ्घमध्ये॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9170"><div class="verse-block verse-justify"><div class="line">शचीव देवेन्द्रमथाग्निदेवंस्वाहेव लक्ष्मीश्च यथा मुकुन्दम्।</div><div class="line" style="position:relative;">उषेव सूर्यं मदनं रतीवमहेश्वरं पर्वतराजपुत्री ॥'॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9171"><div class="verse-block verse-justify"><div class="line">स तामुपादाय विजित्य रङ्गेद्विजातिभिस्तैरभिपूज्यमानः।</div><div class="line" style="position:relative;">रङ्गान्निरक्रामदचिन्त्यकर्मापत्न्या तया चाप्यनुगम्यमानः॥३५॥</div></div></div></div>
