<div class="verse-block-wrapper" id="01-ādiparva-b2380"><div class="verse-block verse-justify"><div class="line">सौतिरुवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2381"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु सर्पसत्राय दीक्षितं जनमेजयम्।</div><div class="line" style="position:relative;">अभ्यगच्छदृषिर्विद्वान्कृष्णद्वैपायनस्तदा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2382"><div class="verse-block verse-justify"><div class="line">जनयामास यं काली शक्तेः पुत्रात्पराशरात्।</div><div class="line" style="position:relative;">कन्यैव यमुनाद्वीपे पाण्डवानां पितामहम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2383"><div class="verse-block verse-justify"><div class="line">जातमात्रश्च यः सद्य इष्ट्या देहमवीवृधत्।</div><div class="line" style="position:relative;">वेदांश्चाधिजगे साङ्गन्सेतिहासान्महायशाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2384"><div class="verse-block verse-justify"><div class="line">यं नाति तपसा कश्चिन्न वेदाध्ययनेन च।</div><div class="line" style="position:relative;">न व्रतैर्नोपवासैश्च न प्रसूत्या न मन्युना॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2385"><div class="verse-block verse-justify"><div class="line">विव्यासैकं चतुर्धा यो वेदं वेदविदां वरः।</div><div class="line" style="position:relative;">परावरज्ञो ब्रह्मर्षिः कविः सत्यव्रतः शुचिः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2386"><div class="verse-block verse-justify"><div class="line">यः पाण्डुं धृतराष्ट्रं च विदुरं चाप्यजीजनत्।</div><div class="line" style="position:relative;">शन्तनोः संततिं तन्वन्पुण्यकीर्तिर्महायशाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2387"><div class="verse-block verse-justify"><div class="line">जनमेजयस्य राजर्षेः स महात्मा सदस्तथा।</div><div class="line" style="position:relative;">विवेश सहितः शिष्यैर्वेदवेदाङ्गपारगैः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2388"><div class="verse-block verse-justify"><div class="line">तत्र राजानमासीनं ददर्श जनमेजयम्।</div><div class="line" style="position:relative;">वृतं सदस्यैर्बहुभिर्देवैरिव पुरन्दरम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2389"><div class="verse-block verse-justify"><div class="line">तथा मूर्धाभिषिक्तैश्च नानाजनपदेश्वरैः।</div><div class="line" style="position:relative;">ऋत्विग्भिर्ब्रह्मकल्पैश्च कुशलैर्यज्ञसंस्तरे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2390"><div class="verse-block verse-justify"><div class="line">जनमेजयस्तु राजर्षिर्दृष्ट्वा तमृषिमागतम्।</div><div class="line" style="position:relative;">सगणोऽभ्युद्ययौ तूर्णं प्रीत्या भरतसत्तमः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2391"><div class="verse-block verse-justify"><div class="line">काञ्चनं विष्टरं तस्मै सदस्यानुमतः प्रभुः।</div><div class="line" style="position:relative;">आसनं कल्पयामास यथा शक्रो बृहस्पतेः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2392"><div class="verse-block verse-justify"><div class="line">तत्रोपविष्टं वरदं देवर्षिगणपूजितम्।</div><div class="line" style="position:relative;">पूजयामास राजेन्द्रः शास्त्रदृष्टेन कर्मणा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2393"><div class="verse-block verse-justify"><div class="line">पाद्यमाचमनीयं च अर्घ्यं गां च विधानतः।</div><div class="line" style="position:relative;">पितामहाय कृष्णाय तदर्हाय न्यवेदयत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2394"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्य तु तां पूजां पाम्डवाज्जनमेजयात्।</div><div class="line" style="position:relative;">गां चैव समनुज्ञाय व्यासः प्रीतोऽभवत्तदा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2395"><div class="verse-block verse-justify"><div class="line">तथा च पूजयित्वा तं प्रणयात्प्रतितामहम्।</div><div class="line" style="position:relative;">उपोपविश्य प्रीतात्मा पर्यपृच्छदनामयम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2396"><div class="verse-block verse-justify"><div class="line">भगवानापि तं दृष्ट्वा कुशलं प्रतिवेद्य च।</div><div class="line" style="position:relative;">सदस्यैः पूजितः सर्वैः सदस्यान्प्रत्यपूजयत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2397"><div class="verse-block verse-justify"><div class="line">ततस्तु सहितः सर्वैः सदस्यैर्जनमेजयः।</div><div class="line">इदं पश्चाद्द्विजश्रेष्ठं पर्यपृच्छत्कृताञ्जलिः ॥जनमेजय उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2398"><div class="verse-block verse-justify"><div class="line">कुरूणां पाण्डवानां च भवान्प्रत्यक्षदर्शिवान्।</div><div class="line" style="position:relative;">तेषां चरितमिच्छामि कथ्यमानं त्वया द्विज॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2399"><div class="verse-block verse-justify"><div class="line">कथं समभवद्भेदस्तेषामक्लिष्टकर्मणाम्।</div><div class="line" style="position:relative;">तच्च युद्धं कथं वृत्तं भूतान्तकरणं महत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2400"><div class="verse-block"><div class="line">पितामहानां सर्वेषां दैवेनाविष्टचेतसाम्</div><div class="line pada-even">कार्त्स्न्येनैतन्ममाचक्ष्व यथा वृत्तं द्विजोत्तम।</div><div class="line">`इच्छामि तत्त्वतः श्रोतुं भगवन्कुशलो ह्यसि' ॥सौतिरुवाच</div><div class="line pada-even" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2401"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा कृष्णद्वैपायनस्तदा।</div><div class="line">शशास शिष्यमासीनं वैशंपायनमन्तिके ॥व्यास उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2402"><div class="verse-block verse-justify"><div class="line">कुरूणां पाण्डवानां च यथा भेदोऽभवत्पुरा।</div><div class="line" style="position:relative;">तदस्मै सर्वमाचक्ष्व यन्मत्तः श्रुतवानसि॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2403"><div class="verse-block verse-justify"><div class="line">गुरोर्वचनमाज्ञाय स तु विप्रर्षभस्तदा।</div><div class="line" style="position:relative;">आचचक्षे ततः सर्वमितिहासं पुरातनम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b2404"><div class="verse-block verse-justify"><div class="line">राज्ञे तस्मै सदस्येभ्यः पार्थिवेभ्यश्च सर्वशः।</div><div class="line" style="position:relative;">भेदं सर्वविनाशं च कुरुपाण्डवयोस्तदा॥२५॥</div></div></div></div>
