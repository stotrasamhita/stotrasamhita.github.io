<div class="verse-block-wrapper" id="01-ādiparva-b8904"><div class="verse-block verse-justify"><div class="line">और्व उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8905"><div class="verse-block verse-justify"><div class="line">उक्तवानस्मि यां क्रोधात्प्रतिज्ञां पितरस्तदा।</div><div class="line" style="position:relative;">सर्वलोकविनाशाय न सा मे वितथा भवेत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8906"><div class="verse-block verse-justify"><div class="line">वृथारोषप्रतिज्ञो वै नाहं जीवितुमुत्सहे।</div><div class="line" style="position:relative;">अनिस्तीर्णो हि मां रोषो दहेदग्निरिवारणिम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8907"><div class="verse-block verse-justify"><div class="line">यो हि कारणतः क्रोधं संजातं क्षन्तुमर्हति।</div><div class="line" style="position:relative;">नालं स मनुजः सम्यक् त्रिवर्गं परिरक्षितुम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8908"><div class="verse-block verse-justify"><div class="line">अशिष्टानां नियन्ता हि शिष्टानां परिरक्षिता।</div><div class="line" style="position:relative;">स्थाने रोषः प्रयुक्तः स्यान्नृपैः सर्वजिगीषुभिः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8909"><div class="verse-block verse-justify"><div class="line">अश्रौषमहमूरुस्थो गर्भशय्यागतस्तदा।</div><div class="line" style="position:relative;">आरावं मातृवर्गस्य भृगूणां क्षत्रियैर्वधे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8910"><div class="verse-block verse-justify"><div class="line">सामरैर्हि यदा लोके भृगूणां क्षत्रियाधमैः।</div><div class="line" style="position:relative;">आगर्भोत्सादनं क्षान्तं तदा मां मन्युराविशत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8911"><div class="verse-block verse-justify"><div class="line">प्रकीर्णकेशाः किल मे मातरः पितरस्तथा।</div><div class="line" style="position:relative;">भयात्सर्वेषु लोकेषु नाधिजग्मुः परायणम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8912"><div class="verse-block verse-justify"><div class="line">तान्भृगूणां यदा दारान्कश्चिन्नाभ्युपपद्यत।</div><div class="line" style="position:relative;">माता तदा दधारेयमूरुणैकेन मां शुभा॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8913"><div class="verse-block verse-justify"><div class="line">प्रतिषेद्धा हि पापस्य यदा लोकेषु विद्यते।</div><div class="line" style="position:relative;">तदा सर्वेषु लोकेषु पापकृन्नोपपद्यते॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8914"><div class="verse-block verse-justify"><div class="line">यदा तु प्रतिषेद्धारं पापो न लभते क्वचित्।</div><div class="line" style="position:relative;">तिष्ठन्ति बहवो लोकास्तदा पापेषु कर्मसु॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8915"><div class="verse-block verse-justify"><div class="line">जानन्नपि च यः पापं शक्तिमान्न नियच्छति।</div><div class="line" style="position:relative;">ईशः सन्सोऽपि तेनैव कर्मणा संप्रयुज्यते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8916"><div class="verse-block verse-justify"><div class="line">राजभिश्चेश्वरैश्चैव यदि वै पितरो मम।</div><div class="line" style="position:relative;">शक्तैर्न शकितास्त्रातुमिष्टं मत्वेह जीवितम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8917"><div class="verse-block verse-justify"><div class="line">अत एषामहं क्रुद्धो लोकानामीश्वरो ह्यहम्।</div><div class="line" style="position:relative;">भवतां च वचो नालमहं समभिवर्तितुम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8918"><div class="verse-block verse-justify"><div class="line">ममापि चेद्भवेदेवमीश्वरस्य सतो महत्।</div><div class="line" style="position:relative;">उपेक्षमाणस्य पुनर्लोकानां किल्बिषाद्भयम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8919"><div class="verse-block verse-justify"><div class="line">यश्चायं मन्युजो मेऽग्निर्लोकानादातुमिच्छति।</div><div class="line" style="position:relative;">दहेदेष च मामेव निगृहीतः स्वतेजसा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8920"><div class="verse-block verse-justify"><div class="line">भवतां च विजानामि सर्वलोकहितेप्सुताम्।</div><div class="line">तस्माद्विधद्ध्वं यच्छ्रेयो लोकानां मम चेश्वराः ॥पितर ऊचुः।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8921"><div class="verse-block verse-justify"><div class="line">य एष मन्युजस्तेऽग्निर्लोकानादातुमिच्छति।</div><div class="line" style="position:relative;">अप्सु तं मुञ्च भद्रं ते लोका ह्यप्सु प्रतिष्ठिताः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8922"><div class="verse-block verse-justify"><div class="line">आपोमयाः सर्वरसाः सर्वमापोमयं जगत्।</div><div class="line" style="position:relative;">तस्मादप्सु विमुञ्चेमं क्रोधाग्निं द्विजसत्तम॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8923"><div class="verse-block verse-justify"><div class="line">अयं तिष्ठतु ते विप्र यदीच्छसि महोदधौ।</div><div class="line" style="position:relative;">मन्युजोऽग्निर्दहन्नापो लोका ह्यापोमयाः स्मृताः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8924"><div class="verse-block verse-justify"><div class="line">एवं प्रतिज्ञा सत्येयं तवानघ भविष्यति।</div><div class="line">न चैवं सामरा लोका गमिष्यन्ति पराभवम् ॥वसिष्ठ उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8925"><div class="verse-block verse-justify"><div class="line">ततस्तं क्रोधजं तात और्वोऽग्निं वरुणालये।</div><div class="line" style="position:relative;">उत्ससर्ज स चैवाप उपयुङ्क्ते महोदधौ॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8926"><div class="verse-block verse-justify"><div class="line">महद्धयशिरो भूत्वा यत्तद्वेदविदो विदुः।</div><div class="line" style="position:relative;">तमग्निमुद्हिरद्वक्त्रात्पिबत्यापो महोदधौ॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8927"><div class="verse-block verse-justify"><div class="line">तस्मात्त्वमपि भद्रं ते न लोकान्हन्तुमर्हसि।</div><div class="line" style="position:relative;">पराशरं पराँल्लोकाञ्जानञ्ज्ञानवतां वर॥२४॥</div></div></div></div>
