<div class="verse-block-wrapper" id="01-ādiparva-b7746"><div class="verse-block verse-justify"><div class="line">सा तानेवापतत्तूर्णं भगिनी तस्य रक्षसः।</div><div class="line" style="position:relative;">अब्रुवाणा हिडिम्बा तु राक्षसी पाण्डवान्प्रति॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7747"><div class="verse-block verse-justify"><div class="line">अभिवाद्य ततः कुन्तीं धर्मराजं च पाण्डवम्।</div><div class="line" style="position:relative;">अभिपूज्य ततः सर्वान्भीमसेनमभाषत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7748"><div class="verse-block verse-justify"><div class="line">अहं ते दर्शादेव मन्मथस्य वशं गता।</div><div class="line" style="position:relative;">क्रूरं भ्रातृवचो हित्वा सा त्वामेवानिरुन्धती॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7749"><div class="verse-block verse-justify"><div class="line">राक्षसे रौद्रसङ्काशे तवापश्यं विचेष्टितम्।</div><div class="line">अहं शुश्रूषुरिच्छेयं तव गात्रं निषेवितुम् ॥'भीमसेन उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7750"><div class="verse-block verse-justify"><div class="line">स्मरन्ति वैरं रक्षांसि मायामाश्रित्य मोहिनीम्।</div><div class="line">हिडिम्बे व्रज पन्थानं त्वमिमं भ्रातृसेवितम् ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7751"><div class="verse-block verse-justify"><div class="line">क्रुद्धोऽपि पुरुषव्याघ्र भीम मा स्म स्त्रियं वधीः।</div><div class="line" style="position:relative;">शरीरगुप्त्यभ्यधिकं धर्मं गोपाय पाण्डव॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7752"><div class="verse-block verse-justify"><div class="line">वधाभिप्रायमायान्तमवधीस्त्वं महाबलम्।</div><div class="line">रक्षसस्तस्य भगिनी किं नः क्रुद्धा करिष्यति ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7753"><div class="verse-block verse-justify"><div class="line">हिडिम्बा तु ततः कुन्तीमभिवाद्य कृताञ्जलिः।</div><div class="line" style="position:relative;">युधिष्ठिरं तु कौन्तेयमिदं वचनमब्रवीत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7754"><div class="verse-block verse-justify"><div class="line">आर्ये जानासि यद्दुःखमिह स्त्रीणामनङ्गजम्।</div><div class="line" style="position:relative;">तदिदं मामनुप्राप्तं भीमसेनकृते शुभे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7755"><div class="verse-block verse-justify"><div class="line">सोढं तत्परमं दुःखं मया कालप्रतीक्षया।</div><div class="line" style="position:relative;">सोऽयमभ्यागतः कालो भविता मे सुखोदयः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7756"><div class="verse-block verse-justify"><div class="line">मया ह्युत्सृज्य सुहृदः स्वधर्मं स्वजनं तथा।</div><div class="line" style="position:relative;">वृतोऽयं पुरुषव्याघ्रस्तव पुत्रः पतिः शुभे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7757"><div class="verse-block verse-justify"><div class="line">वीरेणाऽहं तथाऽनेन त्वया चापि यशस्विनी।</div><div class="line" style="position:relative;">प्रत्याख्याता न जीवामि सत्यमेतद्ब्रवीमि ते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7758"><div class="verse-block verse-justify"><div class="line">यदर्हसि कृपां कर्तुं मयि त्वं वरवर्णिनि।</div><div class="line" style="position:relative;">मत्वा मूढेति तन्मां त्वं भक्ता वाऽनुगतेति वा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7759"><div class="verse-block verse-justify"><div class="line">भर्त्राऽनेन महाभागे संयोजय सुतेन ह।</div><div class="line">समुपादाय गच्छेयं यथेष्टं देवरूपिणम्।</div><div class="line" style="position:relative;">पुनश्चैवानयिष्यामि विस्रम्भं कुरु मे शुभे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7760"><div class="verse-block verse-justify"><div class="line">`अहं हि समये लप्स्ये प्राग्भ्रातुरपर्वजनात्।</div><div class="line" style="position:relative;">ततः सोऽभ्यपतद्रात्रौ भीमसेनजिघांसया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7761"><div class="verse-block verse-justify"><div class="line">यथायथा विक्रमते यथारिमधितिष्ठति।</div><div class="line" style="position:relative;">तथातथा समासाद्य पाण्डवं काममोहिता॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7762"><div class="verse-block verse-justify"><div class="line">न यातुधान्यहं त्वार्ये न चास्मि रजनीचरी।</div><div class="line" style="position:relative;">ईशा रक्षस्स्वसा ह्यस्मि राज्ञि सालकटङ्कटी॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7763"><div class="verse-block verse-justify"><div class="line">पुत्रेण तव संयुक्ता युवतिर्देववर्णिनी।</div><div class="line" style="position:relative;">सर्वान्वोऽहमुपस्थास्ये पुरस्कृत्य वृकोदरम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7764"><div class="verse-block verse-justify"><div class="line">अप्रमत्ता प्रमत्तेषु शुश्रूषुरसकृत्त्वहम्।</div><div class="line" style="position:relative;">'वृजिने तारयिष्यामि दासीवच्च नरर्षभाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7765"><div class="verse-block verse-justify"><div class="line">पृष्ठेन वो वहिष्यामि विमानं सुकृतानिव।</div><div class="line" style="position:relative;">यूयं प्रसादं कुरुत भीमसेनो भजेत माम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7766"><div class="verse-block verse-justify"><div class="line">`एवं ब्रुवन्ती ह तथा प्रत्याख्याता क्रियां प्रति।</div><div class="line" style="position:relative;">भूम्यां दुष्कृतिनो लोकान्गमिष्येऽहं न संशयः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7767"><div class="verse-block verse-justify"><div class="line">अहं हि मनसा ध्यात्वा सर्वं वेत्स्यामि सर्वदा।</div><div class="line" style="position:relative;">'आपन्निस्तरणे प्राणान्धारयिष्ये न केनचित्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7768"><div class="verse-block verse-justify"><div class="line">सर्वमावृत्य कर्तव्यं धर्मं समनुपश्यता।</div><div class="line" style="position:relative;">आपत्सु यो धारयति स वै धर्मविदुत्तमः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7769"><div class="verse-block verse-justify"><div class="line">व्यसनं ह्येव धर्मस्य धर्मिणामापदुच्यते।</div><div class="line" style="position:relative;">पुम्यात्प्राणान्धारयति पुण्यं वै प्राणधारणम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7770"><div class="verse-block verse-justify"><div class="line">येन केनाचरेद्धर्मं तस्मिन्गर्हा न विद्यते।</div><div class="line" style="position:relative;">`महतोऽत्र स्त्रियं कामाद्वाधितां त्राहि मामपि॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7771"><div class="verse-block verse-justify"><div class="line">धर्मार्थकाममोक्षेषु दयां कुर्वन्ति साधवः।</div><div class="line" style="position:relative;">तत्तु धर्ममिति प्राहुर्मुनयो धर्मवत्सलाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7772"><div class="verse-block verse-justify"><div class="line">दिव्यज्ञानेन जानामि व्यतीतानागतानहम्।</div><div class="line" style="position:relative;">तस्माद्वक्ष्यामि वः श्रेय आसन्नं सर उत्तमम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7773"><div class="verse-block verse-justify"><div class="line">अद्यासाद्य सरः स्नात्वा विश्रम्य च वनस्पतौ।</div><div class="line" style="position:relative;">श्वः प्रभाते महद्भूतं प्रादुर्भूतं जगत्पतिम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7774"><div class="verse-block verse-justify"><div class="line">व्यासं कमलपत्राक्षं दृष्ट्वा शोकं विहास्यथ।</div><div class="line" style="position:relative;">धार्तराष्ट्राद्विवासं च दहनं वारणावते॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7775"><div class="verse-block verse-justify"><div class="line">त्राणं च विदुरात्तुभ्यं विदितं ज्ञानचक्षुषा।</div><div class="line" style="position:relative;">आवासे शालिहोत्रस्य स वो वासं विधास्यति॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7776"><div class="verse-block verse-justify"><div class="line">वर्षवातातपसहो ह्ययं पुण्यो वनस्पतिः।</div><div class="line" style="position:relative;">पीतमात्रे तु पानीये क्षुत्पिपासे विनश्यतः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7777"><div class="verse-block verse-justify"><div class="line">तपसा शालिहोत्रेण सरो वृक्षश्च निर्मितः।</div><div class="line" style="position:relative;">कादम्बाः सारसा हंसाः कुरर्यः कुररैः सह॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7778"><div class="verse-block verse-justify"><div class="line">रुवन्ति मधुरं गीतं गान्धर्वस्वनमिश्रितम्।</div><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">तस्यास्तद्वचनं श्रुत्वा कुन्ती वचनमब्रवीत्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7779"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरं महाप्राज्ञं सर्वधर्मविशारदम्।</div><div class="line">कुन्त्युवाच।</div><div class="line" style="position:relative;">त्वं हि धर्मभृतां श्रेष्ठो मयोक्तं शृणु भारत॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7780"><div class="verse-block verse-justify"><div class="line">राक्षस्येषा हि वाक्येन धर्मं वदति साधु वै।</div><div class="line" style="position:relative;">भावेन दुष्टा भीमं वै किं करिष्यति राक्षसी॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7781"><div class="verse-block verse-justify"><div class="line">भजतां पाण्डवं वीरमपत्यार्थं यदीच्छसि।</div><div class="line">' युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">एवमेतद्यथाऽऽत्थ त्वं हिडिम्बे नात्र संशयः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7782"><div class="verse-block verse-justify"><div class="line">स्थातव्यं तु त्वया धर्मे यथा ब्रूयां सुमध्यमे।</div><div class="line" style="position:relative;">`नित्यं कृताह्निका स्नाता कृतशौचा सुरूपिणी ॥'॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7783"><div class="verse-block verse-justify"><div class="line">स्नातं कृताह्निकं भद्रे कृतकौतुकमङ्गलम्।</div><div class="line" style="position:relative;">भीमसेनं भजेथास्त्वमुदिते वै दिवाकरे॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7784"><div class="verse-block verse-justify"><div class="line">अहस्सु विहरानेन यथाकामं मनोजवा।</div><div class="line" style="position:relative;">अयं त्वानयितव्यस्ते भीमसेनः सदा निशि॥४०॥</div></div></div>
