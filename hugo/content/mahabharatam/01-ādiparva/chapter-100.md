<div class="verse-block-wrapper" id="01-ādiparva-b4477"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4478"><div class="verse-block verse-justify"><div class="line">अथान्तरिक्षे दुष्यन्तं वागुवाचाशरीरिणी।</div><div class="line" style="position:relative;">ऋत्विक्पुरोहिताचार्यैर्मन्त्रिभिश्चाभिसंवृतम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4479"><div class="verse-block verse-justify"><div class="line">माता भस्त्रा पितुः पुत्रो यस्माज्जातः स एव सः।</div><div class="line" style="position:relative;">भरस्व पुत्रं दौष्यन्तिं सत्यमाह शकुन्तला॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4480"><div class="verse-block verse-justify"><div class="line">`सर्वेभ्यो ह्यङ्गमङ्गेभ्यः साक्षादुत्पद्यते सुतः।</div><div class="line" style="position:relative;">आत्मा चैव सुतो नाम तेनैव तव पौरव॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4481"><div class="verse-block verse-justify"><div class="line">आहितं ह्यात्मनाऽऽत्मानं परिरक्ष इमं सुतम्।</div><div class="line" style="position:relative;">अनन्यां त्वं प्रतीक्षस्व मावमंस्थाः शकुन्तलाम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4482"><div class="verse-block verse-justify"><div class="line">स्त्रियः पवित्रमतुलमेतद्दुष्यन्त धर्मतः।</div><div class="line" style="position:relative;">मासि मासि रजो ह्यासां दुरितान्यपकर्षति॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4483"><div class="verse-block verse-justify"><div class="line">ततः सर्वाणि भूतानि व्याजह्रस्तं समन्ततः।</div><div class="line">देवा ऊचुः।</div><div class="line" style="position:relative;">आहितस्त्वत्तनोरेष मावमंस्थाः शकुन्तलाम्'॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4484"><div class="verse-block verse-justify"><div class="line">रेतोधाः पुत्र उन्नयति नरदेव यमक्षयात्।</div><div class="line" style="position:relative;">त्वं चास्य धाता गर्भस्य सत्यमाह शकुन्तला॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4485"><div class="verse-block verse-justify"><div class="line">`पतिर्जायां प्रविशति स तस्यां जायते पुनः।</div><div class="line" style="position:relative;">अन्योन्यप्रकृतिर्ह्येषा मावमंस्थाः शकुन्तलाम् ॥'॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4486"><div class="verse-block verse-justify"><div class="line">जाया जनयते पुत्रमात्मनोऽङ्गाद्द्विधा कृतम्।</div><div class="line" style="position:relative;">तस्माद्भरस्व दुष्यन्त पुत्रं शाकुन्तलं नृप॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4487"><div class="verse-block verse-justify"><div class="line">सुभूतिरेषा न त्याज्या जीवितं जीवयात्मजम्।</div><div class="line" style="position:relative;">शाकुन्तलं महात्मानं दुष्यन्त भर पौरवम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4488"><div class="verse-block verse-justify"><div class="line">भर्तव्योऽयं त्वया यस्मादस्माकं वचनादपि।</div><div class="line" style="position:relative;">तस्माद्भवत्वयं नाम्ना भरतो नाम ते सुतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4489"><div class="verse-block verse-justify"><div class="line">`भरताद्भारती कीर्तिर्येनेदं भारतं कुलम्।</div><div class="line">अपरे ये च पूर्वे च भारता इति तेऽभवन् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4490"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा ततो देवा ऋषयश्च तपोधनाः।</div><div class="line" style="position:relative;">पतिव्रतेति संहृष्टाः पुष्पवृष्टिं ववर्षिरे ॥'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4491"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा पौरवो वाक्यं व्याहृतं वे दिवौकसाम्।</div><div class="line" style="position:relative;">`सिंहासनात्समुत्थाय प्रणम्य च दिवौकसः ॥'॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4492"><div class="verse-block verse-justify"><div class="line">पुरोहितममात्यांश्च संप्रहृष्टोऽब्रवीदिदम्।</div><div class="line" style="position:relative;">शृण्वन्त्वेतद्भवन्तोऽपि देवदूतस्य भाषितम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4493"><div class="verse-block verse-justify"><div class="line">`शृण्वन्तु देवतानां च महर्षीणां च भाषितम्'।</div><div class="line" style="position:relative;">अहमप्येवमेवैनं जानामि सुतमात्मजम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4494"><div class="verse-block verse-justify"><div class="line">यद्यहं वचनादस्या गृह्णीयामिममात्मजम्।</div><div class="line">भवेद्धि शङ्का लोकस्य नैवं शुद्धो भवेदयम् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4495"><div class="verse-block verse-justify"><div class="line">तां विशोध्य तदा राजा देवैः सह महर्षिभिः।</div><div class="line" style="position:relative;">हृष्टः प्रमुदितश्चापि प्रतिजग्राह तं सुतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4496"><div class="verse-block verse-justify"><div class="line">ततस्तस्य तदा राजा पितृकर्माणि सर्वशः।</div><div class="line">कारयामास मुदितः प्रीतिमानात्मजस्य ह।</div><div class="line" style="position:relative;">मूर्ध्नि चैनं समाघ्राय सस्नेहं परिषस्वजे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4497"><div class="verse-block verse-justify"><div class="line">सभाज्यमानो विप्रैश्च स्तूयमानश्च बन्दिभिः।</div><div class="line" style="position:relative;">मुदं स परमां लेभे पुत्रस्पर्शनजां नृपः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4498"><div class="verse-block verse-justify"><div class="line">स्वां चैव भार्यां धर्मज्ञः पूजयामास धर्मतः।</div><div class="line" style="position:relative;">अब्रवीच्चैव तां राजा सान्त्वपूर्वमिदं वचः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4499"><div class="verse-block verse-justify"><div class="line">लोकस्यायं परोक्षस्तु संबन्धो नौ पुराऽभवत्।</div><div class="line" style="position:relative;">कृतो लोकसमक्षोऽद्य संबन्धो वै पुनः कृतः॥२३॥</div></div></div>
<p class="prose">तस्मादेतन्मया तस्य तन्निमित्तं प्रभाषितम्</p>
<div class="verse-block-wrapper" id="01-ādiparva-b4501"><div class="verse-block verse-justify"><div class="line">शङ्केत वाऽयं लोकोऽथ स्त्रीभावान्मयि संगतम्।</div><div class="line" style="position:relative;">तस्मादेतन्मया चापि तच्छुद्ध्यर्थं विचारितम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4502"><div class="verse-block verse-justify"><div class="line">`ब्राह्मणाः क्षत्रिया वैश्याः शूद्राश्चैव पृथग्विधाः।</div><div class="line" style="position:relative;">त्वां देवि वूजयिष्यन्ति निर्विशङ्कां पतिव्रताम्'॥२५॥</div></div></div>
<p class="prose">पुत्रश्चायं वृतो राज्ये त्वमग्रमहिषी भव</p>
<div class="verse-block-wrapper" id="01-ādiparva-b4504"><div class="verse-block verse-justify"><div class="line">यच्च कोपनयात्यर्थं त्वयोक्तोऽस्म्यप्रियं प्रिये।</div><div class="line" style="position:relative;">प्रणयिन्या विशालाक्षितत्क्षान्तं ते मया शुभे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4505"><div class="verse-block verse-justify"><div class="line">`अनृतं वाप्यनिष्टं वा दुरुक्तं वातिदुष्कृतम्।</div><div class="line">त्वयाप्येवं विशालाक्षि क्षन्तव्यं मम दुर्वचः।</div><div class="line" style="position:relative;">क्षान्त्या पतिकृते नार्यः पातिव्रत्यं व्रजन्ति ताः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4506"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा तु राजर्षिस्तामनिन्दितगामिनीम्।</div><div class="line" style="position:relative;">अन्तःपुरं प्रवेश्याथ दुष्यन्तो महिषीं प्रियाम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4507"><div class="verse-block verse-justify"><div class="line">वासोभिरन्नपानैश्च पूजयित्वा तु भारत।</div><div class="line" style="position:relative;">`स मातरमुपस्थाय रथन्तर्यामभाषत॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4508"><div class="verse-block verse-justify"><div class="line">मम पुत्रो वने जातस्तव शोकप्रणशनः।</div><div class="line" style="position:relative;">ऋणादद्य विमुक्तोऽहं तव पौत्रेण शोभने॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4509"><div class="verse-block verse-justify"><div class="line">विश्वामित्रसुता चेयं कण्वेन च विवर्धिता।</div><div class="line" style="position:relative;">स्नुषा तव महाभागे प्रसीदस्व शकुन्तलाम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4510"><div class="verse-block verse-justify"><div class="line">पुत्रस्य वचनं श्रुत्वा पौत्रं सा परिषस्वजे।</div><div class="line" style="position:relative;">पादयोः पतितां तत्र रथन्तर्या शकुन्तलाम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4511"><div class="verse-block verse-justify"><div class="line">परिष्वज्य च बाहुभ्यां हर्षादश्रुण्यवर्तयत्।</div><div class="line" style="position:relative;">उवाच वचनं सत्यं लक्षये लक्षणानि च॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4512"><div class="verse-block verse-justify"><div class="line">तव पुत्रो विशालाक्षि चक्रवर्ती भविष्यति।</div><div class="line" style="position:relative;">तव भर्ता विशालाक्षि त्रैलोक्यविजयी भवेत्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4513"><div class="verse-block verse-justify"><div class="line">दिव्यान्भोगाननुप्राप्ता भव त्वं वरवर्णिनि।</div><div class="line" style="position:relative;">एवमुक्ता रथ्तर्या परं हर्षमवाप सा॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4514"><div class="verse-block verse-justify"><div class="line">शकुन्तलां तदा राजा शास्त्रोक्तेनैव कर्मणा।</div><div class="line" style="position:relative;">ततोऽग्रमहिषीं कृत्वा सर्वाभरणभूषिताम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4515"><div class="verse-block verse-justify"><div class="line">ब्राह्मणेभ्यो धनं दत्त्वा सैनिकानां च भूपतिः।</div><div class="line" style="position:relative;">दौष्यन्तिं च ततो राजा पुत्रं शाकुन्तलं तदा'॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4516"><div class="verse-block verse-justify"><div class="line">भरतं नामतः कृत्वा यौवराज्येऽभ्यषेचयत्।</div><div class="line" style="position:relative;">`भरते भारमावेश्य कृतकृत्योऽभवन्नृपः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b4517"><div class="verse-block verse-justify"><div class="line">ततो वर्षशतं पूर्णं राज्यं कृत्वा नराधिपः।</div><div class="line" style="position:relative;">गत्वा वनानि दुष्यन्तः स्वर्गलोकमुपेयिवान् ॥'॥३९॥</div></div></div></div>
