<div class="verse-block-wrapper" id="01-ādiparva-b6769"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6770"><div class="verse-block verse-justify"><div class="line">ततः स्रस्तोत्तरपटः सप्रस्वेदः सवेपथुः।</div><div class="line" style="position:relative;">विवेशाधिरथो रङ्गं यष्टिप्राणो ह्वयन्निव॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6771"><div class="verse-block verse-justify"><div class="line">तमालोक्य धनुस्त्यक्त्वा पितृगौरवयन्त्रितः।</div><div class="line" style="position:relative;">कर्णोऽभिषेकार्द्रशिराः शिरसा समवन्दत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6772"><div class="verse-block verse-justify"><div class="line">ततः पादाववच्छाद्य पटान्तेन ससंभ्रमः।</div><div class="line" style="position:relative;">पुत्रेति परिपूर्णार्थमब्रवीद्रथसारथिम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6773"><div class="verse-block verse-justify"><div class="line">परिष्वज्य च तस्याथ मूर्धानं स्नेहविक्लवः।</div><div class="line" style="position:relative;">अङ्गराज्याभिषेकार्द्रमश्रुभिः सिषिचे पुनः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6774"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा सूतपुत्रोऽयमिति संचिन्त्य पाण्डवः।</div><div class="line" style="position:relative;">भीमसेनस्तदा वाक्यमब्रवीत्प्रहसन्निव॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6775"><div class="verse-block verse-justify"><div class="line">न त्वमर्हसि पार्थेन सूतपुत्र रणे वधम्।</div><div class="line" style="position:relative;">कुलस्य सदृशस्तूर्णं प्रतोदो गृह्यतां त्वया॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6776"><div class="verse-block verse-justify"><div class="line">अङ्गराज्यं च नार्हस्त्वमुपभोक्तुं नराधम।</div><div class="line">श्वा हुताशसमीपस्थं पुरोडाशमिवाध्वरे ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6777"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्ततः कर्णः किंचित्प्रस्फुरिताधरः।</div><div class="line" style="position:relative;">गगनस्थं विनिःश्वस्य दिवाकरमुदैक्षत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6778"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनः कोपादुत्पपात महाबलः।</div><div class="line" style="position:relative;">भ्रातृपद्मवनात्तस्मान्मदोत्कट इव द्विपः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6779"><div class="verse-block verse-justify"><div class="line">सोऽब्रवीद्भीमकर्माणं भीमसेनमवस्थितम्।</div><div class="line" style="position:relative;">वृकोदर न युक्तं ते वचनं वक्तुमीदृशम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6780"><div class="verse-block verse-justify"><div class="line">क्षत्रियाणां बलं ज्यष्ठं योक्तव्यं क्षत्रबन्धुना।</div><div class="line" style="position:relative;">शूराणां च नदीनां च प्रभवो दुर्विभावनः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6781"><div class="verse-block verse-justify"><div class="line">सलिलादुत्थितो वह्निर्येन व्याप्तं चराचरम्।</div><div class="line" style="position:relative;">दधीचस्यास्थितो वज्रं कृतं दानवसूदनम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6782"><div class="verse-block verse-justify"><div class="line">आग्नेयः कृत्तिकापुत्रो रौद्रो गाङ्गेय इत्यपि।</div><div class="line" style="position:relative;">श्रूयते भगवान्देवः सर्वगुह्यमयो गुहः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6783"><div class="verse-block verse-justify"><div class="line">क्षत्रियेभ्यश्च ये जाता ब्राह्मणास्ते च ते श्रुताः।</div><div class="line" style="position:relative;">विश्वामित्रप्रभृतयः प्राप्ता ब्रह्मत्वमव्ययम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6784"><div class="verse-block verse-justify"><div class="line">आचार्यः कलशाज्जातो द्रोणः शस्त्रभृतां वरः।</div><div class="line" style="position:relative;">गौतमस्यान्ववाये च शरस्तम्बाच्च गौतमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6785"><div class="verse-block verse-justify"><div class="line">भवतां च यथा जन्म तदप्यागमितं मया।</div><div class="line">सकुण्डलं सकवचं सर्वलक्षणलक्षितम्।</div><div class="line" style="position:relative;">कथमादित्यसदृशं मृगी व्याघ्रं जनिष्यति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6786"><div class="verse-block verse-justify"><div class="line">पृथिवीराज्यमर्होऽयं नाङ्गराज्यं नरेश्वरः।</div><div class="line" style="position:relative;">अनेन बाहुवीर्येण मया चाज्ञानुवर्तिना॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6787"><div class="verse-block verse-justify"><div class="line">यस्य वा मनुजस्येदं न क्षान्तं मद्विचेष्टितम्।</div><div class="line" style="position:relative;">रथमारुह्य पद्भ्यां स विनामयतु कार्मुकम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6788"><div class="verse-block verse-justify"><div class="line">ततः सर्वस्य रङ्गस्य हाहाकारो महानभूत्।</div><div class="line" style="position:relative;">साधुवादानुसंबद्धः सूर्यश्चास्तमुपागमत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6789"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनः कर्णमालम्ब्याग्रकरे नृपः।</div><div class="line" style="position:relative;">दीपिकाभिः कृतालोकस्तस्माद्रङ्गाद्विनिर्ययौ॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6790"><div class="verse-block verse-justify"><div class="line">पाण्डवाश्च सहद्रोणाः सकृपाश्च विशांपते।</div><div class="line" style="position:relative;">भीष्मेण सहिताः सर्वे ययुः स्वं स्वं निवेशनम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6791"><div class="verse-block verse-justify"><div class="line">अर्जुनेति जनः कश्चित्कश्चित्कर्णेति भारत।</div><div class="line" style="position:relative;">कश्चिद्दुर्योधनेत्येवं ब्रुवन्तः प्रस्थितास्तदा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6792"><div class="verse-block verse-justify"><div class="line">कुन्त्याश्च प्रत्यभिज्ञाय दिव्यलक्षणसूचितम्।</div><div class="line" style="position:relative;">पुत्रमङ्गेश्वरं स्नेहाच्छन्ना प्रीतिरजायत॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b6793"><div class="verse-block verse-justify"><div class="line">दुर्योधनस्यापि तदा कर्णमासाद्य पार्थिव।</div><div class="line" style="position:relative;">भयमर्जुनसंजातं क्षिप्रमन्तरधीयत॥२५॥</div></div></div>
