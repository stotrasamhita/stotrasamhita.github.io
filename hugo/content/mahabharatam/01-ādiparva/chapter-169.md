<div class="verse-block-wrapper" id="01-ādiparva-b7842"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7843"><div class="verse-block verse-justify"><div class="line">गते भगवति व्यासे पाण्डवा विगतज्वराः।</div><div class="line" style="position:relative;">ऊषुस्तत्र च षण्मासान्वटवृक्षे यथासुखम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7844"><div class="verse-block verse-justify"><div class="line">शाकमूलफलाहारास्तपः कुर्वन्ति पाण्डवाः।</div><div class="line" style="position:relative;">अनुज्ञाता महाराज ततः कमलपालिका॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7845"><div class="verse-block verse-justify"><div class="line">रमयन्ती सदा भीमं तत्रतत्र मनोजवा।</div><div class="line" style="position:relative;">दिव्याभरणवस्त्रा हि दिव्यस्रगनुलेपना॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7846"><div class="verse-block verse-justify"><div class="line">एवं भ्रातॄन्सप्त मासान्हिडिम्बाऽवासयद्वने।</div><div class="line" style="position:relative;">पाण्डवान्भीमसेनार्थे राक्षसी कामरूपिणी॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7847"><div class="verse-block verse-justify"><div class="line">सुखं स विहरन्भीमस्तत्कालं पर्यणामयत्।</div><div class="line" style="position:relative;">ततोऽलभत सा गर्भं राक्षसी कामरूपिणी॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7848"><div class="verse-block verse-justify"><div class="line">अतृप्ता भीमसेनेन सप्तमासोपसंगता।</div><div class="line" style="position:relative;">'प्रजज्ञे राक्षसी पुत्रं भीमसेनान्महाबलात्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7849"><div class="verse-block verse-justify"><div class="line">विरूपाक्षं महावक्त्रं शङ्कुकर्णं विभीषणम्।</div><div class="line" style="position:relative;">भीमरूपं सुताम्राक्षं तीक्ष्णदंष्ट्रं महारथम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7850"><div class="verse-block verse-justify"><div class="line">महेष्वासं महावीर्यं महासत्वं महाजवम्।</div><div class="line" style="position:relative;">महाकायं महाकालं महाग्रीवं महाभुजम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7851"><div class="verse-block verse-justify"><div class="line">अमानुषं मानुषजं भीमवेगमरिन्दमम्।</div><div class="line" style="position:relative;">पिशाचकानतीत्यान्यान्बभूवाति स मानुषान्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7852"><div class="verse-block verse-justify"><div class="line">बालोऽपि विक्रमं प्राप्तो मानुषेषु विशांपते।</div><div class="line" style="position:relative;">सर्वास्त्रेषु वरो वीरः प्रकाममभवद्बली॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7853"><div class="verse-block verse-justify"><div class="line">सद्यो हि गर्भं राक्षस्यो लभन्ते प्रसवन्ति च।</div><div class="line" style="position:relative;">कामरूपधराश्चैव भवन्ति बहुरूपिकाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7854"><div class="verse-block verse-justify"><div class="line">प्रणम्य विकचः पादावगृह्णात्स पितुस्तदा।</div><div class="line" style="position:relative;">मातुश्च परमेष्वासस्तौ च नामास्य चक्रतुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7855"><div class="verse-block verse-justify"><div class="line">घटोहास्योत्कच इति माता तं प्रत्यभाषत।</div><div class="line" style="position:relative;">अभवत्तेन नामास्य घटोत्कच इति स्म ह॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7856"><div class="verse-block verse-justify"><div class="line">अनुरक्तश्च तानासीत्पाण्डवान्स घटोत्कचः।</div><div class="line" style="position:relative;">तेषां च दयितो नित्यमात्मनित्यो बभूव ह॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7857"><div class="verse-block verse-justify"><div class="line">घटोत्कचो महाकायः पाण्डवान्पृथया सह।</div><div class="line" style="position:relative;">अभिवाद्य यथान्यायमब्रवीच्च प्रभाष्य तान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7858"><div class="verse-block verse-justify"><div class="line">किं करोम्यहमार्याणां निःशङ्कं वदतानघाः।</div><div class="line" style="position:relative;">तं ब्रुवन्तं भैमसेनिं कुन्ती वचनमब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7859"><div class="verse-block verse-justify"><div class="line">त्वं कुरूणां कुले जातः साक्षाद्भीमसमो ह्यसि।</div><div class="line">ज्येष्ठः पुत्रोसि पञ्चानां साहाय्यं कुरु पुत्रक ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7860"><div class="verse-block verse-justify"><div class="line">पृथयाप्येवमुक्तस्तु प्रणम्यैव वचोऽब्रवीत्।</div><div class="line">यथा हि रावणो लोके इन्द्रजिच्च महाबलः।</div><div class="line" style="position:relative;">वर्ष्मवीर्यसमो लोके विशिष्टश्चाभवं नृषु॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7861"><div class="verse-block verse-justify"><div class="line">कृत्यकाल उपस्थास्ये पितॄनिति घटोत्कचः।</div><div class="line" style="position:relative;">आमन्त्र्य रक्षसां श्रेष्ठः प्रतस्थे चोत्तरां दिशम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7862"><div class="verse-block verse-justify"><div class="line">स हि सृष्टो भगवता शक्तिहेतोर्महात्मना।</div><div class="line">वर्णस्याप्रतिवीर्यस्य प्रतियोद्धा महारथः ॥`भीम उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7863"><div class="verse-block verse-justify"><div class="line">सह वासो मया जीर्णस्त्वया कमलपालिके।</div><div class="line">पुनर्द्रक्ष्यसि राज्यस्थानित्यभाषत तां तदा ॥हिडिम्बोवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b7864"><div class="verse-block verse-justify"><div class="line">पदा मां संस्मरेः कान्त रिरंसू रहसि प्रभो।</div><div class="line" style="position:relative;">तदा तव वशं भूय आगन्तास्म्याशु भारत॥२३॥</div></div></div>
