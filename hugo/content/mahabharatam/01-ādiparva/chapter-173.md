<div class="verse-block-wrapper" id="01-ādiparva-b8004"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8005"><div class="verse-block verse-justify"><div class="line">तयोर्दुःखितयोर्वाक्यमतिमात्रं निशम्य तु।</div><div class="line" style="position:relative;">ततो दुःखपरीताङ्गी कन्या तावभ्यभाषत॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8006"><div class="verse-block verse-justify"><div class="line">किमेवं भृशदुःखार्तौ रोरूयेतामनाथवत्।</div><div class="line" style="position:relative;">ममापि श्रूयतां वाक्यं श्रुत्वा च क्रियतां क्षमम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8007"><div class="verse-block verse-justify"><div class="line">धर्मतोऽहं परित्याज्या युवयोर्नात्र संशयः।</div><div class="line" style="position:relative;">त्यक्तव्यां मां परित्यज्य त्राहि सर्वं मयैकया॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8008"><div class="verse-block verse-justify"><div class="line">इत्यर्थमिष्यतेऽपत्यं तारयिष्यति मामिति।</div><div class="line" style="position:relative;">अस्मिन्नुपस्थिते काले तरध्वं प्लववन्मया॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8009"><div class="verse-block verse-justify"><div class="line">इह वा तारयेद्दुर्गादुत वा प्रेत्य भारत।</div><div class="line" style="position:relative;">सर्वथा तारयेत्पुत्रः पुत्र इत्युच्यते बुधैः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8010"><div class="verse-block verse-justify"><div class="line">आकाङ्क्षन्ते च दौहित्रान्मयि नित्यं पितामहाः।</div><div class="line" style="position:relative;">तत्स्वयं वै परित्रास्ये रक्षन्ती जीवितं पितुः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8011"><div class="verse-block verse-justify"><div class="line">भ्राता च मम बालोऽयं गते लोकममुं त्वयि।</div><div class="line" style="position:relative;">अचिरेणैव कालेन विनश्येत न संशयः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8012"><div class="verse-block verse-justify"><div class="line">तातेपि हि गते स्वर्गं विनष्टे च ममानुजे।</div><div class="line" style="position:relative;">पिण्डः पितॄणां व्युच्छिद्येत्तत्तेषां विप्रियं भवेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8013"><div class="verse-block verse-justify"><div class="line">पित्रा त्यक्ता तथा मात्रा भ्रात्रा चाहमसंशयम्।</div><div class="line" style="position:relative;">दुःखाद्दुःखतरं प्राप्य म्रियेयमतथोचिताम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8014"><div class="verse-block verse-justify"><div class="line">त्वयि त्वरोगे निर्मुक्तो माता भ्राता च मे शिशुः।</div><div class="line" style="position:relative;">सन्तानश्चैव पिण्डश्च प्रतिष्ठास्यत्यसंशयम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8015"><div class="verse-block verse-justify"><div class="line">आत्मा पुत्रः सखी भार्या कृच्छ्रं तु दुहिता किल।</div><div class="line" style="position:relative;">स कृच्छ्रान्मोचयात्मानं मां च धर्मे नियोजया॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8016"><div class="verse-block verse-justify"><div class="line">अनाथा कृपणा बाला यत्र क्वचन गामिनी।</div><div class="line" style="position:relative;">भविष्यामि त्वया तात विहीना कृपणा सदा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8017"><div class="verse-block verse-justify"><div class="line">अथवाहं करिष्यामि कुलस्यास्य विमोचनम्।</div><div class="line" style="position:relative;">फलसंस्था भविष्यामि कृत्वा कर्म सुदुष्करम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8018"><div class="verse-block verse-justify"><div class="line">अथवा यास्यसे तत्र त्यक्त्वा मां द्विजसत्तम।</div><div class="line" style="position:relative;">पीडिताऽहं भविष्यामि तदवेक्षस्व मामपि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8019"><div class="verse-block verse-justify"><div class="line">तदस्मदर्थं धर्मार्थं प्रसवार्थं च सत्तम।</div><div class="line" style="position:relative;">आत्मानं परिरक्षस्व त्यक्तव्यां मां च संत्यज॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8020"><div class="verse-block verse-justify"><div class="line">अवश्यकरणीये च मा त्वां कालोत्यगादयम्।</div><div class="line" style="position:relative;">किं त्वतः परमं दुःखं यद्वयं स्वर्गते त्वयि॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8021"><div class="verse-block verse-justify"><div class="line">याचमानाः परादन्नं परिधावेमहि श्ववत्।</div><div class="line">त्वयि त्वरोगे निर्मुक्ते क्लेशादस्मात्सबान्धवे।</div><div class="line" style="position:relative;">अमृतेव सती लोके भविष्यामि सुखान्विता॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8022"><div class="verse-block verse-justify"><div class="line">इतः प्रदाने देवाश्च पितरश्चेति नः श्रुतम्।</div><div class="line" style="position:relative;">त्वया दत्तेन तोयेन भविष्यति हिताय वै॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8023"><div class="verse-block verse-justify"><div class="line">`इत्येतदुभयं तात निशाम्य तव यद्धितम्।</div><div class="line" style="position:relative;">तद्व्यवस्य तथाम्बाया हितं स्वस्य सुतस्य च॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8024"><div class="verse-block verse-justify"><div class="line">मातापित्रोश्च पुत्रास्तु भवितारो गुणान्विताः।</div><div class="line">न तु पुत्रस्य पितरो पुनर्जातु भविष्यतः ॥'वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8025"><div class="verse-block verse-justify"><div class="line">एवं बहुविधं तस्या निशम्य परिदेवितम्।</div><div class="line" style="position:relative;">पिता माता च सा चैव कन्या प्ररुरुदुस्त्रयः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8026"><div class="verse-block verse-justify"><div class="line">ततः प्ररुदितान्सर्वान्निशम्याथ सुतस्तदा।</div><div class="line" style="position:relative;">उत्फुल्लनयनो बालः कलमव्यक्तमब्रवीत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8027"><div class="verse-block verse-justify"><div class="line">मा पिता रुद मा मातर्मा स्वसस्त्विति चाब्रवीत्।</div><div class="line" style="position:relative;">प्रहसन्निव सर्वांस्तानेकैकमनुसर्पति॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8028"><div class="verse-block verse-justify"><div class="line">ततः स तृणमादाय प्रहृष्टः पुनरब्रवीत्।</div><div class="line">अनेनाहं हनिष्यामि राक्षसं पुरुषादकम् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b8029"><div class="verse-block verse-justify"><div class="line">तथापि तेषां दुःखेन परीतानां निशम्य तत्।</div><div class="line" style="position:relative;">बालस्य वाक्यमव्यक्तं हर्षः समभवन्महान्॥२६॥</div></div></div>
