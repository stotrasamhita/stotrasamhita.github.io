<div class="verse-block-wrapper" id="01-ādiparva-b9782"><div class="verse-block verse-justify"><div class="line">`वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9783"><div class="verse-block verse-justify"><div class="line">दुर्योधनेनैवमुक्तः कर्णेन च विशांपते।</div><div class="line">पुत्रं च सूतपुत्रं च धृतराष्ट्रोऽब्रवीदिदम् ॥ 'धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9784"><div class="verse-block verse-justify"><div class="line">अहमप्येवमेवैतच्चिकीर्षामि यथा युवाम्।</div><div class="line" style="position:relative;">विवेक्तुं नाहमिच्छामि त्वाकारं विदुरं प्रति॥३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9785"><div class="verse-block verse-justify"><div class="line">ततस्तेषां गुणानेव कीर्तयामि विशेषतः।</div><div class="line" style="position:relative;">नावबुध्येत विदुरो ममाभिप्रायमिङ्गितैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9786"><div class="verse-block verse-justify"><div class="line">यच्च त्वं मन्यसे प्राप्तं तद्ब्रवीहि सुयोधन।</div><div class="line">राधेय मन्यसे यच्च प्राप्तकालं वदाशु मे ॥दुर्योधन उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9787"><div class="verse-block verse-justify"><div class="line">अद्य तान्कुशलैर्विप्रैः सुगुप्तैराप्तकारिभिः।</div><div class="line" style="position:relative;">कुन्तीपुत्रान्भेदयामो माद्रीपुत्रौ च पाण्डवौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9788"><div class="verse-block verse-justify"><div class="line">अथवा द्रुपदो राजा महद्भिर्वित्तसंचयैः।</div><div class="line" style="position:relative;">पुत्राश्चास्य प्रलोभ्यन्ताममात्याश्चैव सर्वशः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9789"><div class="verse-block verse-justify"><div class="line">परित्यजेद्यथा राजा कुन्तीपुत्रं युधिष्ठिरम्।</div><div class="line" style="position:relative;">अथ तत्रैव वा तेषां निवासं रोचयन्तु ते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9790"><div class="verse-block verse-justify"><div class="line">इहैषां दोषवद्वासं वर्णयन्तु पृथक्पृथक्।</div><div class="line" style="position:relative;">ते भिद्यमानास्तत्रैव मनः कुर्वन्तु पाण्डवाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9791"><div class="verse-block verse-justify"><div class="line">अथवा कुशळाः केचिदुपायनिपुणा नराः।</div><div class="line" style="position:relative;">इतरेतरतः पार्थान्भेदयन्त्वनुरागतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9792"><div class="verse-block verse-justify"><div class="line">व्युत्थापयन्तु वा कृष्णां बहुत्वात्सुकरं हि तत्।</div><div class="line" style="position:relative;">अथवा पाण्डवांस्तस्यां भेदयन्तु ततश्च ताम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9793"><div class="verse-block verse-justify"><div class="line">भीमसेनस्य वा राजन्नुपायकुशलैर्नरैः।</div><div class="line" style="position:relative;">मृत्युर्विधीयतां छन्नैः स हि तेषां बलाधिकः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9794"><div class="verse-block verse-justify"><div class="line">तमाश्रित्य हि कौन्तेयः पुरा चास्मान्न मन्यते।</div><div class="line" style="position:relative;">सहि तीक्ष्णश्च शूरश्च तेषां चैव परायणम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9795"><div class="verse-block verse-justify"><div class="line">तस्मिंस्त्वभिहते राजन्हतोत्साहा हतौजसः।</div><div class="line" style="position:relative;">यतिष्यन्ते न राज्याय स हि तेषां व्यपाश्रयः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9796"><div class="verse-block verse-justify"><div class="line">अजेयो ह्यर्जुनः सङ्ख्ये पृष्ठगोपे वृकोदरे।</div><div class="line" style="position:relative;">तमृते फाल्गुनो युद्धे राधेयस्य न पादभाक्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9797"><div class="verse-block verse-justify"><div class="line">ते जानानास्तु दौर्बल्यं भीमसेनमृते महत्।</div><div class="line" style="position:relative;">अस्मान्बलवतो ज्ञात्वा न यतिष्यन्ति दुर्बलाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9798"><div class="verse-block verse-justify"><div class="line">इहागतेषु वा तेषु निदेशवशवर्तिषु।</div><div class="line" style="position:relative;">प्रवर्तिष्यामहे राजन्यथाशास्त्रं निबर्हणम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9799"><div class="verse-block verse-justify"><div class="line">`दर्पं वा वदतां तेषां केचिदत्र मनस्विनः।</div><div class="line" style="position:relative;">द्रुपदस्यात्मजा राजन्प्रभिद्यन्ते ततः परैः ॥'॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9800"><div class="verse-block verse-justify"><div class="line">अथवा दर्शनीयाभिः प्रमदाभिर्विलोभ्यताम्।</div><div class="line" style="position:relative;">एकैकस्तत्र कौन्तेयस्ततः कृष्णा विरज्यताम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9801"><div class="verse-block verse-justify"><div class="line">प्रेष्यतां चैव राधेयस्तेषामागमनाय वै।</div><div class="line" style="position:relative;">तैस्तैः प्रकारैः सन्नीय पात्यन्तामाप्तकारिभिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9802"><div class="verse-block verse-justify"><div class="line">एतेषामप्युपायानां यस्ते निर्दोषवान्मतः।</div><div class="line" style="position:relative;">तस्य यप्रोगमातिष्ठ पुरा कालोऽतिवर्तते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9803"><div class="verse-block verse-justify"><div class="line">यावद्ध्यकृतविश्वासा द्रुपदे पार्थिवर्षभे।</div><div class="line" style="position:relative;">तावदेव हि ते शक्या न शक्यास्तु ततः परम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b9804"><div class="verse-block verse-justify"><div class="line">एषा मम मतिस्तात निग्रहाय प्रवर्तते।</div><div class="line" style="position:relative;">साध्वी वा यदि वाऽसाध्वी किं वा राधेय मन्यसे॥२३॥</div></div></div></div>
