<div class="verse-block-wrapper" id="01-ādiparva-b10467"><div class="verse-block verse-justify"><div class="line">प्राप्तुमं तां चिन्तयामास कोऽत्रोपायो भवेदिति।</div><div class="line" style="position:relative;">वेषवैकृत्यमापन्नः परिव्राजकरूपधृत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10468"><div class="verse-block verse-justify"><div class="line">कुकुरान्धकवृष्णीनामज्ञातो वेषधारणात्।</div><div class="line" style="position:relative;">भ्रममाणश्चरन्भैक्षं परिव्राजकवेषवान्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10469"><div class="verse-block verse-justify"><div class="line">येनकेनाप्युपायेन प्रविश्य च गृहं महत्।</div><div class="line" style="position:relative;">दृष्ट्वा सुभद्रां कृष्णस्य भगिनीमेकसुन्दरीम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10470"><div class="verse-block verse-justify"><div class="line">वासुदेवमतं ज्ञात्वा करिष्यामि हितं शुभम्।</div><div class="line" style="position:relative;">एवं विनिश्चयं कृत्वा दीक्षितो वै तदाऽभवत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10471"><div class="verse-block verse-justify"><div class="line">त्रिदण्डी मुण्डितः कुण्डी अक्षमालाङ्गुलीयकः।</div><div class="line" style="position:relative;">योगभारं वहन्पार्थो वटवृक्षस्य कोटरम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10472"><div class="verse-block verse-justify"><div class="line">प्रविशंश्चैव बीभत्सुर्वृष्टिं वर्षति वासवे।</div><div class="line" style="position:relative;">चिन्तयामास देवेशं केशवं क्लेशनाशनम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10473"><div class="verse-block verse-justify"><div class="line">केशवश्चिन्तितं ज्ञात्वा दिव्यज्ञानेन दृष्टवान्।</div><div class="line" style="position:relative;">शयानः शयने दिव्ये सत्यभामासहायवान्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10474"><div class="verse-block verse-justify"><div class="line">केशवः सहसा राजञ्जहाय च ननन्द च।</div><div class="line" style="position:relative;">पुनः पुनः सत्यभामा चाब्रवीत्पुरुषोत्तमम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10475"><div class="verse-block verse-justify"><div class="line">भगवंश्चिन्तयाविष्टः शयने शयितः सुखम्।</div><div class="line" style="position:relative;">भवान्बहुप्रकारेण जहास च पुनः पुनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10476"><div class="verse-block verse-justify"><div class="line">श्रोतव्यं यदि वा कृष्ण प्रसादो यदि ते मयि।</div><div class="line">वक्तुमर्हसि देवेश तच्छ्रोतुं कामयाम्यहम् ॥श्रीभगवानुवाच।</div><div class="line" style="position:relative;">॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10477"><div class="verse-block verse-justify"><div class="line">पितृष्वसुर्यः पुत्रो मे भीमसेनानुजोऽर्जुनः।</div><div class="line" style="position:relative;">तीर्थयात्रां गतः पार्थः कारणात्समयात्तदा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10478"><div class="verse-block verse-justify"><div class="line">तीर्थयात्रासमाप्तौ तु निवृत्तो निशि भारतः।</div><div class="line" style="position:relative;">सुभध्रां चिन्तयामास रूपेणाप्रतिमां भुवि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10479"><div class="verse-block verse-justify"><div class="line">चिन्तयेन्नेव तां भद्रां यतिरूपधरोऽर्जुनः।</div><div class="line" style="position:relative;">यतिरूपप्रतिच्छन्नो द्वारकां प्राप्य माधवीम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10480"><div class="verse-block verse-justify"><div class="line">येनकेनाप्युपायेन दृष्ट्वा तु वरवर्णिनीम्।</div><div class="line" style="position:relative;">वासुदेवमतं ज्ञात्वा प्रयतिष्ये मनोरथम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10481"><div class="verse-block verse-justify"><div class="line">एवं व्यवसितः पार्थो यतिलिङ्गेन पाण्डवः।</div><div class="line" style="position:relative;">छायायां वटवृक्षस्य वृष्टिं वर्षति वासवे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10482"><div class="verse-block verse-justify"><div class="line">योगभारं वहन्नेव मानसं दुःखमाप्तवान्।</div><div class="line" style="position:relative;">एतदर्थं विजानीहि हसन्तं मां मुदा प्रिये॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10483"><div class="verse-block verse-justify"><div class="line">भ्रातरं तव पश्येति सत्यभामां व्यसर्जयत्।</div><div class="line" style="position:relative;">तत उत्थाय शयनात्प्रस्थितो मधुसूदनः ॥'॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10484"><div class="verse-block verse-justify"><div class="line">प्रभासदेशं संप्राप्तं बीभत्सुमपराजितम्।</div><div class="line" style="position:relative;">तीर्थान्यनुचरन्तं तं शुश्राव मधुसूदनः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10485"><div class="verse-block verse-justify"><div class="line">चाराणां चैव वचनादेकाकी स जनार्दनः।</div><div class="line" style="position:relative;">तत्राभ्यगच्छत्कौन्तेयं महात्मातं स माधवः॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b10488"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनं वासुदेवस्तां चर्यां पर्यपृच्छत।</div><div class="line" style="position:relative;">किमर्थं पाण्डवैतानि तीर्थान्यनुचरस्युत॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10489"><div class="verse-block verse-justify"><div class="line">ततोऽर्जनो यथावृत्तं सर्वमाख्यातवांस्तदा।</div><div class="line" style="position:relative;">श्रुत्वोवाच च वार्ष्णेय एवमेतदिति प्रभुः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10490"><div class="verse-block verse-justify"><div class="line">तौ विहृत्य यथाकामं प्रभासे कृष्णपाण्डवौ।</div><div class="line" style="position:relative;">महीधरं रैवतकं वासायैवाभिजग्मतुः॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="01-ādiparva-b10492"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्यार्जुनः सर्वमुपभुज्य च पाण्डवः।</div><div class="line" style="position:relative;">सहैव वासुदेवेन दृष्टवान्नटनर्तकान्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10493"><div class="verse-block verse-justify"><div class="line">अभ्यनुज्ञाय तान्सर्वानर्चयित्वा च पाण्डवः।</div><div class="line" style="position:relative;">सत्कृतं शनं दिव्यमभ्यगच्छन्महामतिः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b10494"><div class="verse-block verse-justify"><div class="line">ततस्तत्र महाबाहुः शयानः शयने शुभे।</div><div class="line">तीर्थानां पल्वलानां च पर्वतानां च दर्शनम्।</div><div class="line" style="position:relative;">आपगानां वनानां च कथयामास सात्वते॥३१॥</div></div></div>
