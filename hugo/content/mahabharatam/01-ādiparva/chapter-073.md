<div class="verse-block-wrapper" id="01-ādiparva-b3369"><div class="verse-block verse-justify"><div class="line">यः समुत्पतितं क्रोधमक्रोधेन निरस्यति।</div><div class="line" style="position:relative;">देवयानि विजानीहि तेन सर्वमिदं जितम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3370"><div class="verse-block verse-justify"><div class="line">यः समुत्पतितं क्रोधं क्षमयेह निरस्यति।</div><div class="line" style="position:relative;">यथोरगस्त्वचं जीर्णां स वै पुरुष उच्यते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3371"><div class="verse-block verse-justify"><div class="line">यः संधारयते मन्युं योऽतिवादांस्तितिक्षते।</div><div class="line" style="position:relative;">यश्च तप्तो न तपति दृढं सोऽर्थस्य भाजनम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3372"><div class="verse-block verse-justify"><div class="line">यो यजेदपरिश्रान्तो मासिमासि शतं समाः।</div><div class="line" style="position:relative;">न क्रुद्ध्येद्यश्च सर्वस्य तयोरक्रोधनोऽधिकः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3373"><div class="verse-block verse-justify"><div class="line">`तस्मादक्रोधनः श्रेष्ठः कामक्रोधौ विगर्हितौ।</div><div class="line" style="position:relative;">क्रुद्धस्य निष्फलान्येव दानयज्ञतपांसि च॥८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3374"><div class="verse-block verse-justify"><div class="line">तस्मादक्रोधने यज्ञतपोदानफलं महत्।</div><div class="line" style="position:relative;">भवेदसंशयं भद्रे नेतरस्मिन्कदाचन॥९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3375"><div class="verse-block verse-justify"><div class="line">न यतिर्न तपस्वी च न यज्वा न च धर्मभाक्।</div><div class="line" style="position:relative;">क्रोधस्य यो वशं गच्छेत्तस्य लोकद्वयं न च॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3376"><div class="verse-block verse-justify"><div class="line">पुत्रो भृत्यः सुहृद्भ्राता भार्या धर्मश्च सत्यता।</div><div class="line" style="position:relative;">तस्यैतान्यपयास्यन्ति क्रोधशीलस्य निश्चितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3377"><div class="verse-block verse-justify"><div class="line">यत्कुमाराः कुमार्यश्च वैरं कुर्युरचेतसः।</div><div class="line">न तत्प्राज्ञोऽनुकुर्वीत न विदुस्ते बलाबलम् ॥देवयान्युवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3378"><div class="verse-block verse-justify"><div class="line">वेदाहं तात बालाऽपि धर्माणां यदिहान्तरम्।</div><div class="line" style="position:relative;">अक्रोधे चातिवादे च वेद चापि बलाबलम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3379"><div class="verse-block verse-justify"><div class="line">`स्ववृत्तिमननुष्ठाय धर्ममुत्सृज्य तत्त्वतः।</div><div class="line" style="position:relative;">'शिष्यस्याशिष्यवृत्तेस्तु न क्षन्तव्यं बुभूषता॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3380"><div class="verse-block verse-justify"><div class="line">`प्रेष्यः शिष्यः स्ववृत्तिं हि विसृज्य विफलं गतः।</div><div class="line" style="position:relative;">'तस्मात्संकीर्णवृत्तेषु वासो मम न रोचते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3381"><div class="verse-block verse-justify"><div class="line">पुमांसो ये हि निन्दन्ति वृत्तेनाभिजनेन च।</div><div class="line" style="position:relative;">न तेषु निवसेत्प्राज्ञः श्रेयोऽर्थी पापबुद्धिषु॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3382"><div class="verse-block verse-justify"><div class="line">ये त्वेनमभिजानन्ति वृत्तेनाभिजनेन वा।</div><div class="line" style="position:relative;">तेषु साधुषु वस्तव्यं स वासः श्रेष्ठ उच्यते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3383"><div class="verse-block verse-justify"><div class="line">`सुयन्त्रितपरा नित्यं विहीनाश्च धनैर्वराः।</div><div class="line" style="position:relative;">दुर्वृत्ताः पापकर्माणश्चण्डाला धनिनोपि च॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3384"><div class="verse-block verse-justify"><div class="line">नैव जात्या हि चण्डालाः स्वकर्मविहितैर्विना।</div><div class="line" style="position:relative;">धनाभिजनविद्यासु सक्ताश्चण्डालधर्मिणः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3385"><div class="verse-block verse-justify"><div class="line">अकारणाश्च द्वेष्यन्ति परिवादं वदन्ति ते।</div><div class="line" style="position:relative;">साधोस्तत्र न वासोस्ति पापिभिः पापतां व्रजेत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3386"><div class="verse-block verse-justify"><div class="line">सुकृते दुष्कृते वापि यत्र सज्जति यो नरः।</div><div class="line" style="position:relative;">ध्रुवं रतिर्भवेत्तस्य तस्माद्द्वेषं न रोचयेत् ॥'॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3387"><div class="verse-block verse-justify"><div class="line">वाग्दुरुक्तं महाघोरं दुहितुर्वृषपर्वणः।</div><div class="line" style="position:relative;">मम मथ्नाति हृदयमग्निकाम इवारणिम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3388"><div class="verse-block verse-justify"><div class="line">न ह्यतो दुष्करतरं मन्ये लोकेष्वपि त्रिषु।</div><div class="line" style="position:relative;">यः सपत्नश्रियं दीप्तां हीनश्रीः पर्युपासते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="01-ādiparva-b3389"><div class="verse-block verse-justify"><div class="line">मरणं शोभनं तस्य इति विद्वज्जना विदुः।</div><div class="line" style="position:relative;">`अवमानमवाप्नोति शनैर्नीचसमागमात्॥२४॥</div></div></div>
