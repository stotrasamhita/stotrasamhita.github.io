<div class="verse-block-wrapper" id="09-śalyaparva-b0498"><div class="verse-block verse-justify"><div class="line">तथा च रथिनं क्रुद्धं विकिरन्तं शरान्बहून्।</div><div class="line" style="position:relative;">नागा जघ्नुर्महाराज परिवार्य समन्ततः॥१०॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0500"><div class="verse-block verse-justify"><div class="line">पादातानवमृद्गन्तो रथवारणवाजिनः।</div><div class="line" style="position:relative;">रणमध्ये व्यदृश्यन्त कुर्वन्तो महदाकुलम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0501"><div class="verse-block verse-justify"><div class="line">हयाश्च पर्यधावन्त चामरैरुपशोभिताः।</div><div class="line" style="position:relative;">हंसा हिमवतः प्रस्थे पिबन्त इव मेदिनीम्॥१३॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0503"><div class="verse-block verse-justify"><div class="line">वाजिनां खुरशब्देन रथनेमिस्वनेन च।</div><div class="line" style="position:relative;">पत्तीनां चापि शब्देन नागानां बृंहितेन च॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0504"><div class="verse-block verse-justify"><div class="line">वादित्राणां च घोषेण शङ्खानां निनदेन च।</div><div class="line" style="position:relative;">अभवन्नादिता भूमिर्निर्घातैरिव भारत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0505"><div class="verse-block verse-justify"><div class="line">धनुषां कूजमानानां शस्त्रौघानां च पात्यताम्।</div><div class="line" style="position:relative;">कवचानां प्रभाभिश्च न प्राज्ञायत किञ्चन॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0506"><div class="verse-block verse-justify"><div class="line">बहवो बाहवश्छिन्ना नागराजकरोपमाः।</div><div class="line" style="position:relative;">उद्वेष्टन्ते विचेष्टन्ते वेगं कुर्वन्ति दारुणम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0507"><div class="verse-block verse-justify"><div class="line">शिरसां च महाराज पततां धरणीतले।</div><div class="line" style="position:relative;">च्युतानामिव तालेभ्यः फलानां श्रूयते स्वनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0508"><div class="verse-block verse-justify"><div class="line">शिरोभिः पतितैर्भाति रुधिरार्द्रैर्वसुन्धरा।</div><div class="line" style="position:relative;">तपनीयनिभैः काले नलिनैरिव भारत॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0509"><div class="verse-block verse-justify"><div class="line">उद्वृत्तनयनैस्तैस्तु गतसत्वैः सुविक्षतैः।</div><div class="line" style="position:relative;">व्यभ्राजत मही राजन्पुण्डरीकैरिवावृता॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0510"><div class="verse-block verse-justify"><div class="line">बाहुभिश्चन्दनादिग्धैः सकेयूरैर्महारधनैः।</div><div class="line" style="position:relative;">पतितैर्भाति राजेन्द्र महाशक्रध्वजैरिव॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0512"><div class="verse-block verse-justify"><div class="line">कबन्धंशतसङ्कीर्णं छत्रचामरसङ्कुलम्।</div><div class="line" style="position:relative;">सेनावनं तच्छुशुभे वनं पुष्पाचितं यथा॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0514"><div class="verse-block verse-justify"><div class="line">मातङ्गाश्चाप्यदृश्यन्त शरतोमरपीडिताः।</div><div class="line" style="position:relative;">पतन्तस्तत्र तवैव छिन्नाभ्रासदृशा रणे॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0519"><div class="verse-block verse-justify"><div class="line">भुजनक्रा धनुःस्रोता हस्तिशैला हयोपला।</div><div class="line" style="position:relative;">मेदोमज्जाकर्दमिनी छत्रहंसा गदोडुपा॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0520"><div class="verse-block verse-justify"><div class="line">कवचोष्णीषसञ्छन्ना पताकारुचिरद्रुमा।</div><div class="line" style="position:relative;">चक्रचक्रावलीजुष्टा त्रिवेणूरगसंवृता॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0522"><div class="verse-block verse-justify"><div class="line">तां नदीं परलोकाय वहन्तीमतिभैरवाम्।</div><div class="line" style="position:relative;">तेरुर्वाहननौभिस्ते शूराः परिघबाहवः॥३४॥</div></div></div>
