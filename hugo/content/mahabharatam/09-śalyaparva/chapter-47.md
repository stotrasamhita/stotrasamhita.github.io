<div class="verse-block-wrapper" id="09-śalyaparva-b2640"><div class="verse-block verse-justify"><div class="line">ऋतवश्च ग्रहाश्चैव ज्योतींषि च विशाम्पते।</div><div class="line" style="position:relative;">मूर्तिमत्यश्च सरितो वेदाश्चैव सनातनाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2641"><div class="verse-block verse-justify"><div class="line">समुद्राश्च हदाश्चैव तीर्थानि विविधानि च।</div><div class="line" style="position:relative;">पृथिवी द्यौर्दिशश्चैव पादपाश्च जनाधिप॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2642"><div class="verse-block verse-justify"><div class="line">अदितिर्देवमाता च हीः श्रीः स्वाहा सरस्वती।</div><div class="line" style="position:relative;">उमा शची सिनीवाली तथैवानुमतिः कुहूः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2643"><div class="verse-block verse-justify"><div class="line">राका च धिषणा चैव पत्न्यश्चान्या दिवौकसाम्।</div><div class="line" style="position:relative;">हिमवांश्चैव विन्ध्यश्च मेरुश्चानेकशृङ्गवान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2644"><div class="verse-block verse-justify"><div class="line">ऐरावतः सानुचरः कलाः काष्ठास्तथैव च।</div><div class="line" style="position:relative;">मासार्धमासा ऋतवस्तथा रात्र्यहनी नृप॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2645"><div class="verse-block verse-justify"><div class="line">उच्चैः श्रवा हयश्रेष्ठो नागराजश्च वासुकिः।</div><div class="line" style="position:relative;">अरुणो गरुडश्चैव वृक्षाश्चौषधिभिः सह॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2646"><div class="verse-block verse-justify"><div class="line">धर्मश्च भगवान्देवः समाजग्मुर्हि सङ्गताः।</div><div class="line" style="position:relative;">कालो यमश्च मृत्युश्च यमस्यानुचराश्च ये॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2647"><div class="verse-block verse-justify"><div class="line">बहुलत्वाच्च नोक्ता ये विविधा देवतागणाः।</div><div class="line" style="position:relative;">ते कुमाराभिषेकार्षं समाजग्मुस्ततस्ततः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2648"><div class="verse-block verse-justify"><div class="line">जगृहुस्ते तदा राजन्सर्वं एव दिवौकसः।</div><div class="line" style="position:relative;">आभिषेचनिकं भाण्डं मङ्गलानि च सर्वशः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2649"><div class="verse-block verse-justify"><div class="line">दिव्यसम्भारसंयुक्तैः कलशैः काञ्चनैर्नृप।</div><div class="line" style="position:relative;">सारस्वताभिः पुण्याभिरद्भिस्ताभिरलङ्कृतम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2650"><div class="verse-block verse-justify"><div class="line">अभ्यषिञ्चन्कुमारं वै सम्प्रहृष्टा दिवौकसः।</div><div class="line" style="position:relative;">सैनापत्ये महात्मानमसुराणां भयङ्करम्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b2652"><div class="verse-block verse-justify"><div class="line">कश्यपश्च महातेजा ये चान्ये सोककीर्तिताः।</div><div class="line" style="position:relative;">तस्मै ब्रह्मा ददौ प्रीतो बलिनो वातरंहसः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2653"><div class="verse-block verse-justify"><div class="line">कामवीर्यधरान्सिद्धान्महापारिषदान्प्रभुः।</div><div class="line" style="position:relative;">नन्दिसेनं लोहिताक्षं घण्टाकार्णं च सम्मतम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2654"><div class="verse-block verse-justify"><div class="line">चतुर्थमस्यानुचरं ख्यातं कुमुदमालिनम्।</div><div class="line" style="position:relative;">तत्र स्थाणुर्महातेजा महापारिषदं प्रभुः॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b2657"><div class="verse-block verse-justify"><div class="line">तथा देवा ददुस्तस्मै सेनां नैर्ऋतसङ्कुलाम्।</div><div class="line" style="position:relative;">देवशत्रुक्षयकरीमजय्यां विश्वरूपिणीम्॥२९॥</div></div></div>
