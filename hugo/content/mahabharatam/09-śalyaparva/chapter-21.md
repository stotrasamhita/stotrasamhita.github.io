<div class="verse-block-wrapper" id="09-śalyaparva-b1186"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1187"><div class="verse-block verse-justify"><div class="line">तस्मिंस्तु निहते शूरे साल्वे समितिशोभने।</div><div class="line" style="position:relative;">तवाभज्यद्बलं वेगाद्वातेनेव महाद्रुमः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1188"><div class="verse-block verse-justify"><div class="line">तत्प्रभग्नं बलं दृष्ट्वा कृतवर्मा महारथः।</div><div class="line" style="position:relative;">दधार समरे शूरः शत्रुसैन्यं महाबलः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1189"><div class="verse-block verse-justify"><div class="line">सन्निवृत्तास्तु ते शूरा दृष्ट्वा सात्वतमाहवे।</div><div class="line" style="position:relative;">शैलोपमं स्थिरं राजन्कीर्यमाणं शरैर्युधि॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1190"><div class="verse-block verse-justify"><div class="line">ततः प्रववृते युद्धं कुरूणां पाण्डवैः सह।</div><div class="line" style="position:relative;">निवृत्तानां महाराज मृत्युं कृत्वा निवर्तनम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1191"><div class="verse-block verse-justify"><div class="line">तत्राश्चर्यमभूद्युद्वं सात्वतस्य परैः सह।</div><div class="line" style="position:relative;">यदेको वारयामास पाण्डुसेनां दुरासदाम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1192"><div class="verse-block verse-justify"><div class="line">तेषामन्योन्यसुहृदां कृते कर्मणि दुष्करे।</div><div class="line" style="position:relative;">सिंहनादः प्रहृष्टानां दिविस्पृक्सुमहानभूत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1193"><div class="verse-block verse-justify"><div class="line">तेन शब्देन वित्रस्तान्पाञ्चालान्भरतर्षभ।</div><div class="line" style="position:relative;">शिनेर्नप्ता महाबाहुरन्वपद्यत सात्यकिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1194"><div class="verse-block verse-justify"><div class="line">स समासाद्य राजानं क्षेमधूर्तिं महाबलम्।</div><div class="line" style="position:relative;">सप्तभिर्निशितैर्बाणैरनयद्यमसादनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1195"><div class="verse-block verse-justify"><div class="line">तमायान्तं महाबाहुं प्रवपन्तं शिताञ्शरान्।</div><div class="line" style="position:relative;">जवेनाभ्यपतद्वीमान्हार्दिक्यः शिनिपुङ्गवम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1196"><div class="verse-block verse-justify"><div class="line">तौ सिंहाविव नर्दन्तौ धन्विनौ रथिनां वरौ।</div><div class="line" style="position:relative;">अन्योन्यमभिधावन्तौ शस्त्रप्रवरधारिणौ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1197"><div class="verse-block verse-justify"><div class="line">पाण्डवाः सहपाञ्चाला योधाश्चान्ये नृपोत्तमाः।</div><div class="line" style="position:relative;">प्रेक्षकाः समपद्यन्त तयोः पुरुषसिंहयोः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1198"><div class="verse-block verse-justify"><div class="line">नाराचैर्वत्सदन्तैश्च वृष्ण्यन्धकमहारथौ।</div><div class="line" style="position:relative;">अभिजघ्नतुरन्योन्यं प्रहृष्टाविव कुञ्चरौ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1199"><div class="verse-block verse-justify"><div class="line">चरन्तौ विविधान्मार्गान्हार्दिक्यशिनिपुङ्गवौ।</div><div class="line" style="position:relative;">मुहुरन्तर्दधाते तौ बाणवृष्ट्या परस्परम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1200"><div class="verse-block verse-justify"><div class="line">चापवेगबलोद्वूतान्मार्गणान्वृष्णिसिंहयोः।</div><div class="line" style="position:relative;">आकाशे समपश्याम पतङ्गानिव शीघ्रगान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1201"><div class="verse-block verse-justify"><div class="line">तमेकं सत्यकर्माणमासाद्य हृदिकात्मजः।</div><div class="line" style="position:relative;">अविध्यन्निशितैर्बाणैश्चतुर्भिश्चतुरो हयान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1202"><div class="verse-block verse-justify"><div class="line">स दीर्घबाहुः सङ्क्रुद्धस्तोत्रार्दित इव द्विपः।</div><div class="line" style="position:relative;">अष्टभिः कृतवर्माणमविध्यत्परमेषुभिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1203"><div class="verse-block verse-justify"><div class="line">ततः पूर्णायतोत्सृष्टैः कृतवर्मा शिलाशितैः।</div><div class="line" style="position:relative;">सात्यकिं त्रिभिराहत्य धनुरेकेन चिच्छिदे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1204"><div class="verse-block verse-justify"><div class="line">निकृत्तं तद्धनुःश्रेष्ठमपास्य शिनिपुङ्गवः।</div><div class="line" style="position:relative;">अन्यदादत्त वेगेन शैनेयः सशरं धनुः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1205"><div class="verse-block verse-justify"><div class="line">तदादाय धनुःश्रेष्ठं वरिष्ठः सर्वधन्विनाम्।</div><div class="line" style="position:relative;">आरोप्य च धनुः शीघ्रं महावीर्यो महाबलः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1206"><div class="verse-block verse-justify"><div class="line">अमृष्यमाणो धनुषश्छेदनं कृतवर्मणा।</div><div class="line" style="position:relative;">कुपितोऽतिरथः शीघ्रं कृतवर्माणमभ्ययात्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1207"><div class="verse-block verse-justify"><div class="line">ततः सुनिशितैर्बाणैर्दशभिः शिनिपुङ्गवः।</div><div class="line" style="position:relative;">जघान सूतं चाश्वांश्च ध्वजं च कृतवर्मणः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1208"><div class="verse-block verse-justify"><div class="line">ततो राजन्महेष्वासः कृतवर्मा महारथः।</div><div class="line" style="position:relative;">हताश्वसूतं सम्प्रेक्ष्य रथं हेमपरिष्कृतम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1209"><div class="verse-block verse-justify"><div class="line">रोषेण महताऽऽविष्टः शूलमुद्यम्य मारिष।</div><div class="line" style="position:relative;">चिक्षेप भुजवेगेन जिघांसुः शिनि पुङ्गवम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1210"><div class="verse-block verse-justify"><div class="line">तच्छूलं सात्वतो ह्याजौ निर्भिद्य निशितैः शरैः।</div><div class="line">चूर्णितं पातयामास मोहयन्निव माधवम्।</div><div class="line" style="position:relative;">ततोऽपरेण भल्लेन हृद्येनं समताडयत्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1211"><div class="verse-block verse-justify"><div class="line">सुयुद्वे युयुधानेन हताश्वो हतसारथिः।</div><div class="line" style="position:relative;">कृतवर्मा कृतास्त्रेण धरमीमन्वपद्यत॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1212"><div class="verse-block verse-justify"><div class="line">तस्मिन्सात्यकिना वीरे द्वैरथे विरथीकृते।</div><div class="line" style="position:relative;">समपद्यत सर्वेषां सैन्यानां सुमहद्भयम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1213"><div class="verse-block verse-justify"><div class="line">पुत्राणां तव चात्यर्थं विषादः समजायत।</div><div class="line" style="position:relative;">हतसूते हताश्वे तु विरथे कृतवर्मणि॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1214"><div class="verse-block verse-justify"><div class="line">हताश्वं च समालक्ष्य हतसूतमरिन्दम।</div><div class="line" style="position:relative;">अभ्यधावत्कृपो राजञ्जिघांसुः शिनिपुङ्गवम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1215"><div class="verse-block verse-justify"><div class="line">तमारोप्य रथोपस्थे मिषतां सर्वधन्विनाम्।</div><div class="line" style="position:relative;">अपोवाह महाबाहुं तूर्णमायोधनादपि॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1216"><div class="verse-block verse-justify"><div class="line">शैनेयेऽधिष्ठिते राजन्विरथे कृतवर्मणि।</div><div class="line" style="position:relative;">दुर्योधनबलं सर्वं पुनरासीत्पराङ्मुखम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1217"><div class="verse-block verse-justify"><div class="line">तत्परे नान्वबुध्यन्त सैन्येन रजसा वृताः।</div><div class="line" style="position:relative;">तावकाः प्रद्रुता राजन्दुर्योधनमृते नृपम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1218"><div class="verse-block verse-justify"><div class="line">दुर्योधनस्तु सम्प्रेक्ष्य भग्नं स्वबलमन्तिकात्।</div><div class="line" style="position:relative;">जवेनाभ्यपतत्तूर्णं सर्वांश्चैको न्यवारयत्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1219"><div class="verse-block verse-justify"><div class="line">पाण्डूंश्च सर्वान्सङ्क्रुद्धो धृष्टद्युम्नं च पार्षतम्।</div><div class="line" style="position:relative;">शिखण्डिनं द्रौपदेयान्पाञ्चालानां च ये गणाः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1220"><div class="verse-block verse-justify"><div class="line">केकयान्सोमकांश्चैव सृञ्जयांश्चैव मारिष।</div><div class="line" style="position:relative;">असम्भ्रमं दुराधर्षः शितैर्बाणैरवाकिरत्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1221"><div class="verse-block verse-justify"><div class="line">अतिष्ठदाहवे यत्तः पुत्रस्तव महाबलः।</div><div class="line">यथा यज्ञे महानग्निर्मन्त्रपूतः प्रकाशते।</div><div class="line" style="position:relative;">तथा दुर्योधनो राजा सङ्ग्रामे सर्वतोऽभवत्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1222"><div class="verse-block verse-justify"><div class="line">तं परे नाभ्यवर्तन्त मर्त्या मृत्युमिवाहवे।</div><div class="line" style="position:relative;">अथान्यं रथमास्थाय हार्दिक्यः समपद्यत॥३७॥</div></div></div></div>
