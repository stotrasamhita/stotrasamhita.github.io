<div class="verse-block-wrapper" id="09-śalyaparva-b2524"><div class="verse-block verse-justify"><div class="line">स शप्ता तेन क्रुद्धेन विश्वामित्रेण धीमता।</div><div class="line" style="position:relative;">तस्मिंस्तीर्थवरे शुभ्रं शोणितं समुपावहत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2525"><div class="verse-block verse-justify"><div class="line">अथाजग्मुस्ततो राजन्राक्षसास्तत्र भारत।</div><div class="line" style="position:relative;">तत्र ते शोणितं सर्वे पिबन्तः सुखमासते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2526"><div class="verse-block verse-justify"><div class="line">तृप्ताश्च सुभृशं तेन सुखिता विगतज्वराः।</div><div class="line" style="position:relative;">नृत्यन्तश्च हसन्तश्च यथा स्वर्गजितस्तथा॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2527"><div class="verse-block verse-justify"><div class="line">कस्यचित्त्वथ कालस्य ऋषयः सुतपोधनाः।</div><div class="line" style="position:relative;">तीर्थयात्रां समाजग्मुः सरस्वत्यां महीपते॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2528"><div class="verse-block verse-justify"><div class="line">तेषु सर्वेषु तीर्थेषु स्वाप्लुत्य मुनिपुङ्गवाः।</div><div class="line" style="position:relative;">प्राप्य प्रीतिं परां चापि तपोलुप्धा विशारदाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2529"><div class="verse-block verse-justify"><div class="line">प्रययुर्हि ततो राजन्येन तीर्थमसृग्वहम्।</div><div class="line" style="position:relative;">अथागम्य महाभागास्तत्तीर्थं दारुणं तदा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2530"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा तोयं सरस्वत्याः शोणितेन परिप्लुतम्।</div><div class="line" style="position:relative;">पीयमानं च रक्षोभिर्बहुभिर्नृपसत्तम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2531"><div class="verse-block verse-justify"><div class="line">तान्दृष्ट्वा राक्षसान्राजन्मुनयः संशितव्रताः।</div><div class="line" style="position:relative;">परित्राणे सरस्वत्याः परं यत्नं प्रचक्रिरे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2532"><div class="verse-block verse-justify"><div class="line">ते तु सर्वे महाभागाः समागम्य महाव्रताः।</div><div class="line" style="position:relative;">आहूय सरितां श्रेष्ठामिदं वचनमब्रुवन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2533"><div class="verse-block verse-justify"><div class="line">कारणं ब्रूहि कल्याणि किमर्थं ते हृदो ह्ययम्।</div><div class="line" style="position:relative;">एवमग्राह्यतां यातः श्रुत्वाऽध्यास्यामहे वयम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2534"><div class="verse-block verse-justify"><div class="line">ततः सा सर्वमाचष्ट यथावृत्तं प्रवेपती।</div><div class="line" style="position:relative;">दुःखितामथ तां दृष्ट्वा ऊचुस्ते वै तपोधनाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2535"><div class="verse-block verse-justify"><div class="line">कारणं श्रुतमस्माभिः शापश्चैव श्रुतोऽनघे।</div><div class="line" style="position:relative;">करिष्यन्ति तु यत्प्राप्तं सर्व एव तपोधनाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2536"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा सरिच्छ्रेष्ठामूचुस्तेऽथ परस्परम्।</div><div class="line" style="position:relative;">विमोचयामहे सर्वे शापादेतां सरस्वतीम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2537"><div class="verse-block verse-justify"><div class="line">ते सर्वे ब्राह्मणा राजंस्तपोभिर्नियमैस्तथा।</div><div class="line" style="position:relative;">उपवासैश्च विविधैर्यमैः कष्टव्रतैस्तथा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2538"><div class="verse-block verse-justify"><div class="line">आराध्य पशुभर्तारं महादेवं जगत्पतिम्।</div><div class="line" style="position:relative;">मोक्षयामासुस्तां देवीं सरिच्छ्रेष्ठां सरस्वतीम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2539"><div class="verse-block verse-justify"><div class="line">तेषां तु सा प्रभावेण प्रकृतिस्था सरस्वती।</div><div class="line">प्रसन्नसलिला जज्ञे यथा पूर्वं तथैव हि।</div><div class="line" style="position:relative;">निर्मुक्ता च सरिच्छ्रेष्ठा विबभौ सा यथा पुरा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2540"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा तोयं सरस्वत्या मुनिभिस्तैस्तथा कृतम्।</div><div class="line" style="position:relative;">तानेव शरणं जग्मू राक्षसाः क्षुधितास्तथा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2541"><div class="verse-block verse-justify"><div class="line">कृत्वाञ्जलिं ततो राजन्राक्षसाः क्षुधयाऽर्दिताः।</div><div class="line" style="position:relative;">ऊचुस्तान्वै मुनीन्सर्वान्कृपायुक्तान्पुनःपुनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2542"><div class="verse-block verse-justify"><div class="line">वयं च क्षुधिताश्चैव धर्माद्धीनाश्च शाश्वतात्।</div><div class="line" style="position:relative;">न च नः कामकारोऽयं यद्वयं पापकारिणा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2543"><div class="verse-block verse-justify"><div class="line">युष्माकं चाप्रसादेन दुष्कृतेन च कर्मणा।</div><div class="line">पक्षोऽयं वर्धतेऽस्माकं यतः स्मो ब्रह्मराक्षसाः।</div><div class="line" style="position:relative;">योषितां चैव पापेन योनिदोषकृतेन च॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2544"><div class="verse-block verse-justify"><div class="line">एवं हि वैश्यशूद्राणां क्षत्रियाणां तथैव च।</div><div class="line" style="position:relative;">ये ब्राह्मणान्प्रद्विषन्ति ते भवन्तीह राक्षसाः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2545"><div class="verse-block verse-justify"><div class="line">`शक्तिमन्तोऽपि ये केचिदाश्रितानां च रक्षणम्।</div><div class="line" style="position:relative;">न कुर्वन्ति मनुष्यास्ते सम्भवन्तीह राक्षसाः'॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2546"><div class="verse-block verse-justify"><div class="line">आचार्यमृत्विजं चैव गुरुं वृद्वजनं तथा।</div><div class="line" style="position:relative;">प्रणिनो येऽवमन्यन्ते ते भवन्तीह राक्षसाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2547"><div class="verse-block verse-justify"><div class="line">तत्कुरुध्वमिहास्माकं तारणं द्विजसत्तमाः।</div><div class="line" style="position:relative;">शक्ता भवन्तः सर्वेषां लोकानामपि तारणे॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2548"><div class="verse-block verse-justify"><div class="line">तेषां तु वचनं श्रुत्वा तुष्टुवुस्तां महानदीम्।</div><div class="line">मोक्षार्थं रक्षसां तेषामूचुः प्रयतमानसाः ॥ऋषय ऊचुः।</div><div class="line" style="position:relative;">॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2549"><div class="verse-block verse-justify"><div class="line">क्षुतं कीटावपन्नं च यच्चोच्छिष्टाचितं भवेत्।</div><div class="line" style="position:relative;">सकेशमवधूतं च रुदितोपहतं च यत्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2550"><div class="verse-block verse-justify"><div class="line">श्वभिः संसृष्टमन्नं च भागोऽसौ रक्षसामिह।</div><div class="line">तस्माज्ज्ञात्वा सदा विद्वानेतान्यत्नाद्विवर्जयेत्।</div><div class="line" style="position:relative;">राक्षसान्नमसौ भुङ्क्ते यो भुङ्क्ते ह्यन्नमीदृशम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2551"><div class="verse-block verse-justify"><div class="line">शोधयित्वा ततस्तीर्थमृषयस्ते तपोधनाः।</div><div class="line" style="position:relative;">मोक्षार्थं राक्षसानां च नदीं तां प्रत्यचोदयन्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2552"><div class="verse-block verse-justify"><div class="line">महर्षीणां मतं ज्ञात्वा ततः सा सरितां वरा।</div><div class="line" style="position:relative;">अरुणामानयामास स्वां तनुं पुरुषर्षभ॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2553"><div class="verse-block verse-justify"><div class="line">तस्यान्तेराक्षसाः स्नात्वा तनूस्त्यक्त्वा दिवं गताः।</div><div class="line" style="position:relative;">अरुणायां महाराज ब्रह्मवध्यापहा हि सा॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2554"><div class="verse-block verse-justify"><div class="line">एतमर्थमभिज्ञाय देवराजः शतक्रतुः।</div><div class="line">तस्मिंस्तीर्थे वरे स्नात्वा विमुक्तः पाप्मना किल ॥जनमेजय उवाच।</div><div class="line" style="position:relative;">॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2555"><div class="verse-block verse-justify"><div class="line">किमर्थं भगवाञ्शक्रो ब्रह्मवध्यामवाप्तवान्।</div><div class="line">कथमस्मिंश्च तीर्थे वै आप्लुत्याकल्मषोऽभवत् ॥वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2556"><div class="verse-block verse-justify"><div class="line">शृणुष्वैतदुपाख्यानं यथावृत्तं जनेश्वर।</div><div class="line" style="position:relative;">यथा बिभेद समयं नमुचेर्वासवः पुरा॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2557"><div class="verse-block verse-justify"><div class="line">जमुचिर्वासवाद्भीतः सूर्यरश्मिं समाविशत्।</div><div class="line" style="position:relative;">तेनेन्द्रः सख्यमकरोत्समयं चेदमब्रवीत्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2558"><div class="verse-block verse-justify"><div class="line">न चार्द्रेण न शुष्केण न रात्रौ नापि चाहनि।</div><div class="line" style="position:relative;">वधिष्याम्यसुरश्रेष्ठ सखे सत्येन ते शपे॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2559"><div class="verse-block verse-justify"><div class="line">एवं स कृत्वा समयं दृष्ट्वा नीहास्मीश्वरः।</div><div class="line" style="position:relative;">चिच्छेदास्य शिरो राजन्नपां फेनेन वासवः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b2560"><div class="verse-block verse-justify"><div class="line">तच्छिरो नमुचेश्छिन्नं पृष्ठतः शक्रमन्वियात्।</div><div class="line" style="position:relative;">भोभो मित्रह पापेति ब्रुवाणं शक्रमन्तिकात्॥३८॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b2562"><div class="verse-block verse-justify"><div class="line">तमब्रवील्लोकगुरुररुणायां यथाविधि।</div><div class="line" style="position:relative;">इष्ट्वोपस्पृश देवेन्द्र तीर्थे पापभयापहे॥४०॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b2567"><div class="verse-block verse-justify"><div class="line">स मुक्तः पाप्मना तेन ब्रह्मवध्याकृतेन च।</div><div class="line" style="position:relative;">जगाम संहृष्टमनास्त्रिदिवं त्रिदशेश्वरः॥४५॥</div></div></div>
