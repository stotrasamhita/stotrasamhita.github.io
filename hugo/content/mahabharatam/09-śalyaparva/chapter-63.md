<div class="verse-block-wrapper" id="09-śalyaparva-b3549"><div class="verse-block verse-justify"><div class="line">चिकीडुश्च तथैवान्ये जहसुश्च तवाहिताः।</div><div class="line" style="position:relative;">अब्रुवंश्चासकृद्वीरा भीमसेनमिदं वचः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3550"><div class="verse-block verse-justify"><div class="line">दुष्करं भवता कर्म रणेऽद्य सुमहत्कृतम्।</div><div class="line" style="position:relative;">कौरवेयं रणे हत्वा गदयाऽतिकृतश्रमम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3551"><div class="verse-block verse-justify"><div class="line">इन्द्रेणेव हि वृत्रस्य वधं परमसंयुगे।</div><div class="line" style="position:relative;">त्वया कृतममन्यन्त शत्रोर्वधमिमं जनाः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3552"><div class="verse-block verse-justify"><div class="line">चरन्तं विविधान्मार्गान्मण्डलानि च सर्वशः।</div><div class="line" style="position:relative;">दुर्योधनमिमं शूरं कोऽन्यो हन्याद्वृकोदरात्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3553"><div class="verse-block verse-justify"><div class="line">वैरस्य च गतः पारं त्वमिहान्यैः सुदुर्गमम्।</div><div class="line" style="position:relative;">अशक्यमेतदन्येन सम्पादयितुमीदृशम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3554"><div class="verse-block verse-justify"><div class="line">कुञ्जरेणेव मत्तेन वीरसङ्ग्राममूर्धनि।</div><div class="line" style="position:relative;">दुर्योधपशिरो दिष्ट्या पादेन मृदितं त्वया॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3555"><div class="verse-block verse-justify"><div class="line">सिंहेन महिषस्येव कृत्वा सङ्गरमुत्तमम्।</div><div class="line" style="position:relative;">दुःशासनस्य रुधिरं दिष्ट्या पीतं त्वयाऽनघ॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3556"><div class="verse-block verse-justify"><div class="line">ये विप्रकुर्वन्राजानं धर्मात्मानं युधिष्ठिरम्।</div><div class="line" style="position:relative;">मूर्ध्नि तेषां कृतः पादो दिष्ट्या ते स्वेन कर्मणा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3557"><div class="verse-block verse-justify"><div class="line">अमित्राणामधिष्ठानाद्वधाद्दुर्योधनस्य च।</div><div class="line" style="position:relative;">भीम दिष्ट्या पृथिव्यां ते प्रथितं सुमहद्यशः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3558"><div class="verse-block verse-justify"><div class="line">एवं नूनं हते वृत्रे शक्रं नन्दन्ति बन्दिनः।</div><div class="line" style="position:relative;">तथा त्वां निहतामित्रं वयं नन्दाम भारत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3559"><div class="verse-block verse-justify"><div class="line">दुर्योधनवधे यानि रोमाणि हृषितानि नः।</div><div class="line">अद्यापि न विकृष्यन्ते तानि तद्विद्धि भारत।</div><div class="line" style="position:relative;">इत्यब्रुन्भीमसेनं वातिकास्तत्र सङ्गताः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3560"><div class="verse-block verse-justify"><div class="line">तान्हृष्टान्पुरुषव्याघ्रान्पाञ्चालान्पाण्डवैः सह।</div><div class="line" style="position:relative;">ब्रुवतोऽसदृशं तत्र प्रोवाच मधुसूदनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3561"><div class="verse-block verse-justify"><div class="line">न न्याय्यं निहतं शत्रुं भूयो हन्तुं नराधिपाः।</div><div class="line" style="position:relative;">असकृद्वाग्भिरुग्राभिर्निहतो ह्येष मन्दधीः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3562"><div class="verse-block verse-justify"><div class="line">तदैवैष हतः पापो यदैव निरपत्रपः।</div><div class="line" style="position:relative;">लुब्धः पापसहायश्च सुहृदां शासनातिगः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3563"><div class="verse-block verse-justify"><div class="line">बहुशो विदुरद्रोणकृपगाङ्गेयसृञ्जयैः।</div><div class="line" style="position:relative;">पाण्डुभ्यः प्रार्थ्यमानोऽपि पित्र्यमंशं न दत्तवान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3564"><div class="verse-block verse-justify"><div class="line">नैव तेभ्योऽद्य मित्रं वा शत्रुर्वा पुरुषाधमः।</div><div class="line" style="position:relative;">किमनेन वितुन्नेन वाग्भिः काष्ठसधर्मणा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3565"><div class="verse-block verse-justify"><div class="line">रथेष्वारोहत क्षिप्रं गच्छामो वसुधाधिपाः।</div><div class="line" style="position:relative;">दिष्ट्या हतोऽयं पापात्मा सामात्यज्ञातिबान्धवः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3566"><div class="verse-block verse-justify"><div class="line">इति श्रुत्वा त्वधिक्षेपं कृष्णाद्दुर्योधनो नृपः।</div><div class="line" style="position:relative;">अमर्षवशमापन्न उदतिष्ठद्विशाम्पते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3567"><div class="verse-block verse-justify"><div class="line">स्फिग्देशेनोपविष्टः स दोर्भ्यां विष्टभ्य मेदिनीम्।</div><div class="line" style="position:relative;">दृष्टिं भ्रूसङ्कटां कृत्वा वासुदेवे न्यपातयत्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3568"><div class="verse-block verse-justify"><div class="line">अर्धोन्नतशरीरस्य रूपमासीन्नृपस्य तु।</div><div class="line" style="position:relative;">क्रुद्धस्याशीविषस्येव च्छिन्नपुच्छस्य भोगिनः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3569"><div class="verse-block verse-justify"><div class="line">प्राणान्तकरिणीं घोरां वेदनामप्यचिन्तयन्।</div><div class="line" style="position:relative;">दुर्योधनो वासुदेवं वाग्भिरुग्राभिरार्दयत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3570"><div class="verse-block verse-justify"><div class="line">कंसदासस्य दायाद न ते लज्जाऽस्त्यनेन वै।</div><div class="line">अधर्मेण गदायुद्धे यदर्हं विनिपातितः।</div><div class="line" style="position:relative;">ऊरूभिन्धीति भीमस्य स्मृतिं मिथ्याप्रयच्छता॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3571"><div class="verse-block verse-justify"><div class="line">किं न विज्ञातमेतन्मे यदर्जुनमवोचथाः।</div><div class="line">घातयित्वा महीपालानृजुयुद्धान्सहस्रशः।</div><div class="line" style="position:relative;">जिह्मैरुपायैर्बहुभिर्न ते लज्जा न ते घृणा॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3572"><div class="verse-block verse-justify"><div class="line">अहन्यहनि शूराणां कुर्वाणः कदनं महत्।</div><div class="line" style="position:relative;">शिखण्डिनं पुरस्कृत्य घातितस्ते पितामहः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3573"><div class="verse-block verse-justify"><div class="line">अश्वत्थाम्नः सनामानं हत्वा नागं सुदुर्मते।</div><div class="line" style="position:relative;">आचार्यो न्यासितः शस्त्रं किं तन्न विदितं मया॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3574"><div class="verse-block verse-justify"><div class="line">स चानेन नृशंसेन धृष्टद्युम्नेन वीर्यवान्।</div><div class="line" style="position:relative;">पात्यमानस्त्वया दृष्टो न चैनं त्वमवारयः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3575"><div class="verse-block verse-justify"><div class="line">वधार्थं पाण्डुपुत्रस्य याचितां शक्तिमेव च।</div><div class="line" style="position:relative;">घटोत्कचे व्यंसयतः कस्त्वत्तः पापकृत्तमः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3576"><div class="verse-block verse-justify"><div class="line">छिन्नहस्तः प्रायगतस्तथा भूरिश्रवा बली।</div><div class="line" style="position:relative;">त्वयाऽभिसृष्टेन हतः शैनेयेन दुरात्मना॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3577"><div class="verse-block verse-justify"><div class="line">कुर्वाणश्चोत्तमं कर्म कर्णः पार्थजिगीषया।</div><div class="line" style="position:relative;">व्यंसनेनाश्वसेनस्य पन्नगेन्द्रस्य वै पुनः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3578"><div class="verse-block verse-justify"><div class="line">पुनश्च पतिते चक्रे व्यसनार्तः पराजितः।</div><div class="line" style="position:relative;">पापितः समरे कर्णश्चक्रव्याग्रोऽग्रणीर्नृणाम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3579"><div class="verse-block verse-justify"><div class="line">यदि मां चापि कर्णं च भीष्मद्रोणौ च संयुगे।</div><div class="line" style="position:relative;">ऋजुना प्रतियुध्येथा न ते स्याद्विजयो ध्रुवम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3580"><div class="verse-block verse-justify"><div class="line">त्वया पुनरनार्येण जिह्ममार्गेण पार्थिवाः।</div><div class="line" style="position:relative;">स्वधर्ममनुतिष्ठन्तो वयं चान्ये च घातिताः॥३७॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3582"><div class="verse-block verse-justify"><div class="line">हतस्त्वमसि गान्धारे सभ्रातृसुतबान्धवः।</div><div class="line" style="position:relative;">सगणः ससुहृच्चैव पापं मार्गमनुष्ठितः॥३९॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3584"><div class="verse-block verse-justify"><div class="line">याच्यमानं मया मूढ पित्र्यमंशं न दित्ससि।</div><div class="line" style="position:relative;">पाण्डवेभ्यः स्वराज्यं च लोभाच्छकुनिनिश्चयात्॥४१॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3587"><div class="verse-block verse-justify"><div class="line">अनक्षज्ञं च धर्मज्ञं सौबलेनाक्षवेदिना।</div><div class="line" style="position:relative;">निकृत्या यत्पराजैषीस्तस्मादसि हतो रणे॥४४॥</div></div></div>
