<div class="verse-block-wrapper" id="09-śalyaparva-b1669"><div class="verse-block verse-justify"><div class="line">स़ञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1670"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धा महाराज सौबलस्य पदानुगाः।</div><div class="line" style="position:relative;">त्यक्त्वा जीवितमाक्रन्दे पाण्डवान्पर्यवारयन्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1671"><div class="verse-block verse-justify"><div class="line">तानर्जुनः प्रत्यगृह्णात्सहदेवजये धृतः।</div><div class="line" style="position:relative;">भीमसेनश्च तेजस्वी क्रुद्धाशीविषदर्शनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1672"><div class="verse-block verse-justify"><div class="line">शक्त्यृ-ष्टिप्रासहस्तानां सहदेवं जिघांसताम्।</div><div class="line" style="position:relative;">सङ्कल्पमकरोन्मोघं गाण्डीवेन धनञ्जयः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1673"><div class="verse-block verse-justify"><div class="line">सङ्गृहीतायुधान्बाहून्योधानामभिधावताम्।</div><div class="line" style="position:relative;">भल्लैश्चिच्छेद बीभत्सुः शिरांस्यपि हयानपि॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1674"><div class="verse-block verse-justify"><div class="line">ते हयाः प्रत्यपद्यन्त वसुधां विगतासवः।</div><div class="line" style="position:relative;">चरता लोकवीरेण प्रहताः सव्यसाचिना॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1675"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राजा दृष्ट्वा स्वबलसङ्क्षयम्।</div><div class="line" style="position:relative;">हतशेषान्समानीय क्रुद्धो रथगणान्बहून्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1676"><div class="verse-block verse-justify"><div class="line">कुञ्जरांश्च हयांश्चैव पादातांश्च समन्ततः।</div><div class="line" style="position:relative;">उवाच दुःखितान्सर्वान्धार्तराष्ट्र इदं वचः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1677"><div class="verse-block verse-justify"><div class="line">समासाद्य रणे सर्वान्पाण्डवान्ससुहृद्गणान्।</div><div class="line" style="position:relative;">पाञ्चाल्यं चापि सबलं हत्वा शीघ्रं न्यवर्तत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1678"><div class="verse-block verse-justify"><div class="line">तस्य ते शिरसा गृह्य वचनं युद्धदुर्मदाः।</div><div class="line" style="position:relative;">अभ्युद्ययू रणे पार्थांस्तव पुत्रस्य शासनात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1679"><div class="verse-block verse-justify"><div class="line">तानभ्यापततः शीघ्रं हतशेषान्महारणे।</div><div class="line" style="position:relative;">शरैराशीविषाकारैः पाण्डवाः समवाकिरन्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1680"><div class="verse-block verse-justify"><div class="line">तत्सैन्यं भरतश्रेष्ठ मुहूर्तेन महात्मभिः।</div><div class="line" style="position:relative;">अवध्यत रणं प्राप्य त्रातारं नाभ्यविन्दत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1681"><div class="verse-block verse-justify"><div class="line">पलायमानं तु भयान्नावतिष्ठति दंशितम्।</div><div class="line" style="position:relative;">अश्वैर्विपरिधावद्भिः सैन्येन रजसा वृते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1682"><div class="verse-block verse-justify"><div class="line">न प्राज्ञायन्त समरे दिशः सप्रदिशस्तथा।</div><div class="line" style="position:relative;">ततस्तु पाण्डवानीकान्निःसृत्य बहवो जनाः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1683"><div class="verse-block verse-justify"><div class="line">अभ्यघ्नंस्तावकान्युद्धे मुहूर्तादिव भारत।</div><div class="line" style="position:relative;">ततो निःशेषमभवत्तत्सन्यं तव भारत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1684"><div class="verse-block verse-justify"><div class="line">अक्षौहिण्यः समेतास्तु तव पुत्रस्य भारत।</div><div class="line" style="position:relative;">एकादश हता युद्धे ताः प्रभो पाण्डुसृञ्जयैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1685"><div class="verse-block verse-justify"><div class="line">तेषु राजसहस्रेषु तावकेषु महात्मसु।</div><div class="line" style="position:relative;">एको दुर्योधनो राजन्नदृश्यत भृशं क्षतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1686"><div class="verse-block verse-justify"><div class="line">ततो वीक्ष्य दिशः सर्वा दृष्ट्वा शून्यां च मेदिनीम्।</div><div class="line" style="position:relative;">विहीनः सर्वयोधैश्च पाण्डवान्वीक्ष्यं संयुगे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1687"><div class="verse-block verse-justify"><div class="line">मुदितान्सर्वतः सिद्धान्नर्दमानान्समन्ततः।</div><div class="line" style="position:relative;">बाणशब्दरवांश्चैव श्रुत्वा तेषां महात्मनाम्॥१९॥</div></div></div>
