<div class="verse-block-wrapper" id="09-śalyaparva-b1440"><div class="verse-block verse-justify"><div class="line">तत्सैन्यं भरतश्रेष्ठ वध्यमानं किरीटिना।</div><div class="line" style="position:relative;">सम्प्रदुद्राव सङ्ग्रामात्तव पुत्रस्य पश्यतः॥४॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b1443"><div class="verse-block verse-justify"><div class="line">अन्येषां सायकाः क्षीणास्तथाऽन्ये बाणपीडिताः।</div><div class="line" style="position:relative;">अक्षता युगपत्केचित्प्राद्रवन्भयपीडिताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1444"><div class="verse-block verse-justify"><div class="line">केचित्पुत्रानुपादाय हतभूयिष्ठबान्धवाः।</div><div class="line" style="position:relative;">विचुक्रुशुः पितॄंस्त्वन्ये सहायानपरे पुनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1445"><div class="verse-block verse-justify"><div class="line">बान्धवांश्च नरव्याघ्र भ्रातॄन्सम्बन्धिनस्तथा।</div><div class="line" style="position:relative;">दुद्रुवुः केचिदुत्सृज्य तत्रतत्र विशाम्पते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1446"><div class="verse-block verse-justify"><div class="line">बहवोऽत्र भृशं विद्धा मुह्यमाना महारथाः।</div><div class="line" style="position:relative;">निःश्वसन्ति स्म दृश्यन्ते पार्थबाणहता नराः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1447"><div class="verse-block verse-justify"><div class="line">तानन्ये रथमारोप्य ह्याश्वास्य च मुहूर्तकम्।</div><div class="line" style="position:relative;">विश्रान्ताश्च वितृष्णाश्च पुनर्युद्धाय जग्मिरे॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1448"><div class="verse-block verse-justify"><div class="line">तानपास्य गताः केचित्पुनरेव युयुत्सवः।</div><div class="line" style="position:relative;">कुर्वन्तस्तव पुत्रस्य शासनं युद्धदुर्मदाः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1449"><div class="verse-block verse-justify"><div class="line">पानीयमपरे पीत्वा पर्याश्वास्य च वाहनम्।</div><div class="line" style="position:relative;">वर्माणि च समारोप्य केचिद्भरतसत्तम॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1450"><div class="verse-block verse-justify"><div class="line">समाश्वास्यापरे भ्रातॄन्निक्षिप्य शिबिरेऽपि च।</div><div class="line" style="position:relative;">पुत्रानन्ये पितॄनन्ये पुनर्युद्धमरोचयन्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1451"><div class="verse-block verse-justify"><div class="line">सज्जयित्वा रथान्केचिद्यथामुख्यं विशाम्पते।</div><div class="line" style="position:relative;">आवृत्य पाण्डवानीकं पुनर्युद्धमरोचयन्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1452"><div class="verse-block verse-justify"><div class="line">ते शूराः किङ्किणीजालैः समाच्छन्ना बभासिरे।</div><div class="line" style="position:relative;">त्रैलोक्यविजये युक्ता यथा दैतेयदानवाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1453"><div class="verse-block verse-justify"><div class="line">आगम्य सहसा केचिद्रथैः स्वर्णविभूषितैः।</div><div class="line" style="position:relative;">पाण्डवानामनीकेषु धृष्टद्युम्नमयोधयन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1454"><div class="verse-block verse-justify"><div class="line">धृष्टद्युम्नोऽपि पाञ्चाल्यः शिखण्डी च महारथः।</div><div class="line" style="position:relative;">नाकुलिस्तु शतानीको रथानीकमयोधयन्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1455"><div class="verse-block verse-justify"><div class="line">पाञ्चाल्यस्तु ततः क्रुद्धः सैन्येन महता वृतः।</div><div class="line" style="position:relative;">अभ्यद्रवत्सुसङ्क्रुद्धस्तावकान्हन्तुमुद्यतः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1456"><div class="verse-block verse-justify"><div class="line">ततस्त्वापततस्तस्य तव पुत्रो जनाधिप।</div><div class="line" style="position:relative;">बाणसङ्घाननेकान्वै प्रेषयामास भारत॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1457"><div class="verse-block verse-justify"><div class="line">धृष्टद्युम्नस्ततो राजंस्तव पुत्रेण धन्विना।</div><div class="line" style="position:relative;">नाराचैरर्धनाराचैर्बहुभिः क्षिप्रकारिभिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1458"><div class="verse-block verse-justify"><div class="line">वत्सदन्तैश्च बाणैश्च कर्मारपरिमार्जितैः।</div><div class="line" style="position:relative;">अश्वांश्च चतुरो हत्वा बाह्वोरुरसि चार्पितः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1459"><div class="verse-block verse-justify"><div class="line">सोऽतिविद्धो महेष्वासस्तोत्रार्दित इव द्विपः।</div><div class="line">तस्याश्वांश्चतुरो बाणैः प्रेषयामास मृत्यवे।</div><div class="line" style="position:relative;">सारथेश्चास्य भल्लेन शिरः कायादपाहरत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1460"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राजा पृष्ठमारुह्य वाजिनः।</div><div class="line" style="position:relative;">अपाक्रामद्वतरथो नातिदूरमरिन्दमः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1461"><div class="verse-block verse-justify"><div class="line">दृष्टा तु हतविक्रान्तं स्वमनीकं महाबलः।</div><div class="line" style="position:relative;">तव पुत्रो महाराज प्रययौ यत्र सौबलः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1462"><div class="verse-block verse-justify"><div class="line">ततो रथेषु भग्नेषु त्रिसाहस्रा महाद्विपाः।</div><div class="line" style="position:relative;">पाण्डवान्रथिनः सर्वान्समन्तात्पर्यवारयन्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1463"><div class="verse-block verse-justify"><div class="line">ते वृताः समरे पञ्च गजानीकेन भारत।</div><div class="line" style="position:relative;">अशोभन्त महाराज ग्रहास्तारागणैरिव॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1464"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनो महाराज लब्धलक्षौ महाभुजः।</div><div class="line" style="position:relative;">विनिर्ययौ रथेनैव श्वेताश्वः कृष्णसारथिः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1465"><div class="verse-block verse-justify"><div class="line">तैः समन्तात्परिवृतः कुञ्जरैः पर्वतोपमैः।</div><div class="line" style="position:relative;">नाराचैर्विमलैस्तीक्ष्णैर्गजानीकमयोधयत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1466"><div class="verse-block verse-justify"><div class="line">तत्रैकबाणनिहतानपश्याम महागजान्।</div><div class="line" style="position:relative;">पतितान्पात्यमानांश्च निर्भिन्नान्सव्यसाचिना॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1467"><div class="verse-block verse-justify"><div class="line">भीमसेनस्तु तान्दृष्ट्वा नागान्मत्तगजोपमः।</div><div class="line">करेणादाय महतीं गदामभ्यपतद्बली।</div><div class="line" style="position:relative;">अथाप्लुत्य रथात्तूर्णं दण्डपाणिरिवान्तकः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1468"><div class="verse-block verse-justify"><div class="line">तमुद्यतगदं दृष्ट्वा पाण्डवानां महारथम्।</div><div class="line" style="position:relative;">वित्रेसुस्तावकाः सैन्याः शकृन्मूत्रं च सुस्रुवुः॥३१॥</div></div></div>
<p class="prose">आविग्नं च बलं सर्वं गदाहस्ते वृकोदरे</p>
<div class="verse-block-wrapper" id="09-śalyaparva-b1470"><div class="verse-block verse-justify"><div class="line">गदया भीमसेनेन भिन्नकुम्भान्निपातितान्।</div><div class="line" style="position:relative;">धावमानानपश्याम कुञ्जरान्पर्वतोपमान्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1471"><div class="verse-block verse-justify"><div class="line">प्राद्रवन्कुञ्जरास्ते तु भीमसेनगदाहताः।</div><div class="line" style="position:relative;">पेतुरार्तस्वरं कृत्वा छिन्नपक्षा इवाद्रयः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1472"><div class="verse-block verse-justify"><div class="line">प्रभिन्नकुम्भांस्तु बहून्द्रवमाणानितस्ततः।</div><div class="line" style="position:relative;">पतमानांश्च सम्प्रेक्ष्य वित्रेसुस्तव सैनिकाः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1473"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरोऽपि सङ्क्रुद्धो माद्रीपुत्रौ च पाण्डवौ।</div><div class="line" style="position:relative;">गार्ध्रपत्रैः शितैर्बाणैर्जघ्नुर्वै गजयोधिनः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1474"><div class="verse-block verse-justify"><div class="line">धृष्टद्युम्नस्तु समरे पारजित्य नराधिपम्।</div><div class="line" style="position:relative;">अपक्रान्ते तव सुते हयपृष्ठे समाश्रिते॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1475"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा च पाण्डवान्सर्वान्कुञ्चरैः परिवारितान्।</div><div class="line" style="position:relative;">धृष्टद्युम्नो महाराज सहसा समुपाद्रवत्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1476"><div class="verse-block verse-justify"><div class="line">पुत्रः पाञ्चालराजस्य जिघांसुः कुञ्जरान्ययौ।</div><div class="line" style="position:relative;">अदृष्ट्वा तु रथानीके दुर्योधनमरिन्दमम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1477"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा कृपश्चैव कृतवर्मा च सात्वतः।</div><div class="line" style="position:relative;">अपृच्छन्क्षत्रियांस्तत्र क्व नु दुर्योधनो गतः॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1478"><div class="verse-block verse-justify"><div class="line">तेऽपश्यमाना राजानं वर्तमाने जनक्षये।</div><div class="line">मन्वाना निहतं तत्र तव पुत्रं महारथाः।</div><div class="line" style="position:relative;">विवर्णवदना भूत्वा पर्यपृच्छन्त ते सुतम्॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1479"><div class="verse-block verse-justify"><div class="line">आहुः केचिद्वते सूते प्रयातो यत्र सौबलः।</div><div class="line" style="position:relative;">हित्वा पाञ्चालराजस्य तदनीकं दुरुत्सहम्॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1480"><div class="verse-block verse-justify"><div class="line">अपरे त्वब्रुवंस्तत्र क्षत्रिया भृशविक्षताः।</div><div class="line">दुर्योधनेन किं कार्यं द्रक्ष्यध्वं यदि जीवति।</div><div class="line" style="position:relative;">युध्यध्वं सहिताः सर्वे किं वो राजा करिष्यति॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1481"><div class="verse-block verse-justify"><div class="line">ते क्षत्रियाः क्षतैर्गात्रैर्हतभूयिष्ठवाहनाः।</div><div class="line" style="position:relative;">शरैः सम्पीड्यमानास्तु नातिव्यक्तमथाब्रुवन्॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1482"><div class="verse-block verse-justify"><div class="line">इदं सर्वं बलं हन्मो येन स्म परिवारिताः।</div><div class="line" style="position:relative;">एते सर्वे गजान्हत्वा उपयान्ति स्म पाण्डवाः॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1483"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु वचनं तेषामश्वत्थामा महाबलः।</div><div class="line" style="position:relative;">भित्त्वा पाञ्चालराजस्य तदनीकं दुरुत्सहम्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1484"><div class="verse-block verse-justify"><div class="line">कृपश्च कृतवर्मा च प्रययौ यत्र सौबलः।</div><div class="line" style="position:relative;">रथानीकं परित्यज्य शूराः सुदृढधन्विनः॥४६॥</div></div></div>
