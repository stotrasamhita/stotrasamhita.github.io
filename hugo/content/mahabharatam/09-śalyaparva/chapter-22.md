<div class="verse-block-wrapper" id="09-śalyaparva-b1229"><div class="verse-block verse-justify"><div class="line">यथा सैन्येन रजसा समुद्भूतेन वाहिनी।</div><div class="line" style="position:relative;">प्रत्यदृश्यत सञ्छन्ना तथा बाणैर्महात्मनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1230"><div class="verse-block verse-justify"><div class="line">बाणभूतामपश्याम पृथिवीं पृथिवीपते।</div><div class="line" style="position:relative;">दुर्योधनेन प्रकृतां क्षिप्रहस्तेन धन्विना॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1231"><div class="verse-block verse-justify"><div class="line">तेषु योधसहस्रेषु तावकेषु परेषु च।</div><div class="line" style="position:relative;">नास्ति दुर्योधनसमः पुमानिति मतिर्मम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1232"><div class="verse-block verse-justify"><div class="line">तत्राद्भुतमपश्याम तव पुत्रस्य विक्रमम्।</div><div class="line" style="position:relative;">यदेकं सहिताः पार्था नाभ्यवर्तन्त भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1233"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरं शतेनाजौ विव्याध भरतर्षभ।</div><div class="line" style="position:relative;">भीमसेनं च सप्तत्या सहदेवं च पञ्चभिः॥१०॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b1235"><div class="verse-block verse-justify"><div class="line">धनुश्चिच्छेद भल्लेन सहदेवस्य मारिष।</div><div class="line" style="position:relative;">तदपास्य धनुश्छिन्नं माद्रीपुत्रः प्रतापवान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1236"><div class="verse-block verse-justify"><div class="line">अभ्यद्रवत राजानं प्रगृह्यान्यन्महद्धनुः।</div><div class="line" style="position:relative;">ततो दुर्योधनं सङ्ख्ये विव्याध दशभिः शरैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1237"><div class="verse-block verse-justify"><div class="line">नकुलस्तु ततो वीरो राजानं नवभिः शरैः।</div><div class="line" style="position:relative;">घोररूपैर्महेष्वासो विव्याध च ननाद च॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1238"><div class="verse-block verse-justify"><div class="line">सात्यकिश्चैव राजानं शरेणानतपर्वणा।</div><div class="line">द्रौपदेयास्त्रिसप्तत्या धर्मराजश्च पञ्चभिः।</div><div class="line" style="position:relative;">अशीत्या भीमसेनश्च शरै राजानमार्पयन्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1239"><div class="verse-block verse-justify"><div class="line">समन्तात्कीर्यमाणस्तु बाणसङ्घैर्महात्मभिः।</div><div class="line" style="position:relative;">न चचाल महाराज सर्वसैन्यस्य पश्यतः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1240"><div class="verse-block verse-justify"><div class="line">लाघवात्सौष्ठवाच्चापि वीर्याच्चापि महात्मनः।</div><div class="line" style="position:relative;">अति सर्वाणि भूतानि ददृशुः सर्वपार्थिवाः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1241"><div class="verse-block verse-justify"><div class="line">धार्तराष्ट्रा हि राजेन्द्र योधास्तु स्वल्पमन्तरम्।</div><div class="line" style="position:relative;">अपश्यमाना राजानं पर्यवर्तन्त दंशिताः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1242"><div class="verse-block verse-justify"><div class="line">तेषामापततां घोरस्तुमुलः समपद्यत।</div><div class="line" style="position:relative;">क्षुब्धस्य हि समुद्रस्य प्रावृट््काले यथा स्वनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1243"><div class="verse-block verse-justify"><div class="line">समासाद्य रणे ते तु राजानमपराजितम्।</div><div class="line" style="position:relative;">प्रत्युद्ययुर्महेष्वासाः पाण्डवानाततायिनः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1244"><div class="verse-block verse-justify"><div class="line">भीमसेनं रणे क्रुद्धो द्रोणपुत्रो न्यवारयत्।</div><div class="line">तयोर्बाणैर्महाराज प्रमुक्तैः सर्वतोदिशम्।</div><div class="line" style="position:relative;">नाज्ञायन्त रणे वीरा न दिशः प्रदिशस्तथा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1245"><div class="verse-block verse-justify"><div class="line">तावुभौ क्रूरकर्माणावुभौ भारतदुःसहौ।</div><div class="line" style="position:relative;">घोररूपमयुध्येतां कृतप्रतिकृतैषिणौ॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1246"><div class="verse-block verse-justify"><div class="line">त्रासयन्तौ दिशः सर्वा ज्याक्षेपकठिनत्वचौ।</div><div class="line" style="position:relative;">शकुनिस्तु रणे वीरो युधिष्ठिरमपीडयत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1247"><div class="verse-block verse-justify"><div class="line">तस्याश्वांश्चतुरो हत्वा सुबलस्य सुतो विभो।</div><div class="line" style="position:relative;">नादं चकार बलवत्सर्वसैन्यानि कम्पयन्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1248"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नन्तरे वीरं राजानमपराजितम्।</div><div class="line" style="position:relative;">अपोवाह रथेनाजौ सहदेवः प्रतापवान्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1249"><div class="verse-block verse-justify"><div class="line">अथान्यं रथमास्थाय धर्मपुत्रो युधिष्ठिरः।</div><div class="line">शकुनिं नवभिर्विद्ध्वा पुनर्विव्याध पञ्चभिः।</div><div class="line" style="position:relative;">ननाद च महानादं प्रवरः सर्वधन्विनाम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1250"><div class="verse-block verse-justify"><div class="line">तद्युद्धमभवच्चित्रं घोररूपं च मारिष।</div><div class="line" style="position:relative;">प्रेक्षतां प्रीतिजननं सिद्धचारणसेवितम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1251"><div class="verse-block verse-justify"><div class="line">उलूकस्तु महेष्वासं नकुलं युद्धदुर्मदम्।</div><div class="line" style="position:relative;">अभ्यवर्षदमेयात्मा शरवर्षैः समन्ततः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1252"><div class="verse-block verse-justify"><div class="line">तथैव नकुलः शूरः सौबलस्य सुतं रणे।</div><div class="line" style="position:relative;">शरवर्षेण महता समन्तात्पर्यवारयत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1253"><div class="verse-block verse-justify"><div class="line">तौ तत्र समरे वीरौ कुलपुत्रौ महारथौ।</div><div class="line" style="position:relative;">योधयन्तावपश्येतां कृतप्रतिकृतैषिणौ॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1254"><div class="verse-block verse-justify"><div class="line">तथैव कृतवर्माणं शैनेयः शत्रुतापनः।</div><div class="line" style="position:relative;">योधयञ्शुशुभे राजन्बलिं शक्र इवाहवे॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1255"><div class="verse-block verse-justify"><div class="line">दुर्योधनो धनुश्छित्त्वा धृष्टद्युम्नस्य संयुगे।</div><div class="line" style="position:relative;">अथैनं छिन्नधन्वानं विव्याध निशितैः शरैः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1256"><div class="verse-block verse-justify"><div class="line">धृष्टद्युम्नोऽपि समरे प्रगृह्य परमायुधम्।</div><div class="line" style="position:relative;">राजानं योधयामास पश्यतां सर्वधन्विनाम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1257"><div class="verse-block verse-justify"><div class="line">तयोर्युद्धं महच्चासीत्सङ्ग्रामे भरतर्षभ।</div><div class="line" style="position:relative;">प्रभिन्नयोर्यथा सक्तं मत्तयोर्वनहस्तिनोः॥३४॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b1262"><div class="verse-block verse-justify"><div class="line">नराश्चैव नरैः सार्धं दन्तिनो दन्तिभिस्तथा।</div><div class="line" style="position:relative;">हया हयैः समासक्ता रथिनो रथिभिः सह॥३९॥</div></div></div>
