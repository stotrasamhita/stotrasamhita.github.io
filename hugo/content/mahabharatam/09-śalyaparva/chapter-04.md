<div class="verse-block-wrapper" id="09-śalyaparva-b0260"><div class="verse-block verse-justify"><div class="line">शृणु राजन्नवहितो यथावृत्तो महान्क्षयः।</div><div class="line" style="position:relative;">कुरूणां पाण्डवानां च समासाद्य परस्परम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0261"><div class="verse-block verse-justify"><div class="line">निहते सूतपुत्रे च फल्गुनेन महात्मना।</div><div class="line" style="position:relative;">विद्रुतेषु च सैन्येषु समानीतेषु चासकृत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0262"><div class="verse-block verse-justify"><div class="line">विमुखे तव पुत्रे च शोकोपहतचेतसि।</div><div class="line" style="position:relative;">भृशोद्विग्नेषु सैन्येषु दृष्ट्वा पार्थस्य विक्रमम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0263"><div class="verse-block verse-justify"><div class="line">ध्यायमानेषु योधेषु दुऋखं प्राप्तेषु भारत।</div><div class="line" style="position:relative;">बलानां मत्यमानानां श्रुत्वा निनदमुत्तमम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0264"><div class="verse-block verse-justify"><div class="line">अभिज्ञानं नरेन्द्राणां विकृतं प्रेक्ष्य संयुगे।</div><div class="line">पतितानवनीपालान्धवजांश्चैव महात्मनाम्।</div><div class="line" style="position:relative;">रणे विनिहतान्नागान्दृष्ट्वा पत्तींश्च भारत॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0265"><div class="verse-block verse-justify"><div class="line">आयोधनं महाघोरं रुद्रस्याक्रीडसन्निभम्।</div><div class="line" style="position:relative;">अप्रख्यातिं गतानां तु राज्ञां शतसहस्रशः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0266"><div class="verse-block verse-justify"><div class="line">कृपाविष्टः कृपो दृष्ट्वा वयः शीलसमन्वितः।</div><div class="line">अब्रवीत्तत्र तेजस्वी सोऽभिसृत्य जनाधिपम्।</div><div class="line" style="position:relative;">दुर्योधनमनुक्रोशाद्वाक्यं वाक्यविशारदः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0267"><div class="verse-block verse-justify"><div class="line">दुर्योधन निबोधेयं यत्त्वां वक्ष्यामि कौरव।</div><div class="line" style="position:relative;">श्रुत्वा कुरु महाराज यदि ते रोचतेऽनघ॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0268"><div class="verse-block verse-justify"><div class="line">न युद्धधर्माच्छ्रेयान्वै पन्था राजेन्द्र विद्यते।</div><div class="line" style="position:relative;">यं समाश्रित्य युध्यन्ते क्षत्रियाः क्षत्रियर्षभ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0269"><div class="verse-block verse-justify"><div class="line">पुत्रो भ्राता पिता चैव स्वस्रीयो मातुलस्तथा।</div><div class="line" style="position:relative;">सम्बन्धिबान्धवाश्चैव योद्वव्याः क्षत्रजीविन॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0270"><div class="verse-block verse-justify"><div class="line">वधे चैव परो धर्मस्तथाऽधर्मः पलायने।</div><div class="line">ते स्म घोरां समापन्ना जीविकां जीवितार्थिनः।</div><div class="line" style="position:relative;">तदत्र प्रतिवक्ष्यामि किञ्चिदेव हितं वचः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0271"><div class="verse-block verse-justify"><div class="line">हते भीष्मे च द्रोणे च कर्णे चैव महारथे।</div><div class="line">जयद्रथे च निहते तव भ्रातृषु चानघ।</div><div class="line" style="position:relative;">लक्ष्मणे तव पुत्रे च किं शेषं पर्युपास्महे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0272"><div class="verse-block verse-justify"><div class="line">येषु भारं समासज्य राज्ये मतिमकुर्महि।</div><div class="line" style="position:relative;">ते सन्त्यज्य तनूर्याताः शूरा ब्रह्मविदां गतिम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0273"><div class="verse-block verse-justify"><div class="line">वयं त्विह विनाभूता गुणवद्भिर्महारथैः।</div><div class="line" style="position:relative;">कृपणं वर्तयिष्यामः पातयित्वा नृपान्बहून्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0274"><div class="verse-block verse-justify"><div class="line">सर्वैरथ च जीवद्भिर्बीभत्सुरपराजितः।</div><div class="line" style="position:relative;">कृष्णनेत्रो महाबाहुर्देवैरपि दुरासदः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0275"><div class="verse-block verse-justify"><div class="line">इन्द्रकार्मुकवज्राभमिन्द्रकेतुमिवोच्छ्रितम्।</div><div class="line" style="position:relative;">वानरं केतुमासाद्य सञ्चचाल महाचमूः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0276"><div class="verse-block verse-justify"><div class="line">सिंहनादेन भीमस्य पाञ्चजन्यस्वनेन च।</div><div class="line" style="position:relative;">गाण्डीवस्य च निर्घोषात्सम्मुह्यन्ते मनांसि नः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0277"><div class="verse-block verse-justify"><div class="line">स्फुरन्तीव महाविद्युन्मुष्णन्ती नयनप्रभाम्।</div><div class="line" style="position:relative;">अलातमिव चाविद्वं गाण्डवीं समदृश्यत॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0278"><div class="verse-block verse-justify"><div class="line">जाम्बूनदविचित्रं च धूयमानं महद्धनुः।</div><div class="line" style="position:relative;">दृश्यते दिक्षु स्रवासु विद्युदभ्रघनेष्विव॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0279"><div class="verse-block verse-justify"><div class="line">उह्यमानश्च कृष्णेन वायुनेव बलाहकः।</div><div class="line">तावकं तद्बलं राजन्नर्जुनोऽस्त्रविशारदः।</div><div class="line" style="position:relative;">गहनं शिशिरापाये ददाहाग्निरिवोल्बणः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0280"><div class="verse-block verse-justify"><div class="line">गाहमानमनीकानि महेन्द्रसदृशप्रभम्।</div><div class="line">विक्षोभयन्तं सेनां वै त्रासयन्तं च पार्थिवान्।</div><div class="line" style="position:relative;">धनञ्जयमपश्याम नलिनीमिव कुञ्जरम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0281"><div class="verse-block verse-justify"><div class="line">त्रासयन्तं तथा योधान्धनुर्घोषेण पांण्डवम्।</div><div class="line" style="position:relative;">भूय एनमपश्याम सिंहं मृगगणानिव॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0282"><div class="verse-block verse-justify"><div class="line">सर्वलोकमहेष्वासो वृषभौ सर्वधन्विनाम्।</div><div class="line" style="position:relative;">आमुक्तकवचौ कृष्णौ लोकमध्ये विरेजतुः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0283"><div class="verse-block verse-justify"><div class="line">अद्य सप्तदशाहानि वर्तमानस्य भारत।</div><div class="line" style="position:relative;">सङ्ग्रामस्यातिघोरस्य युध्यतां चाभितो युधि॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0284"><div class="verse-block verse-justify"><div class="line">वायुनेव विधूतानि एव सैन्यानि गच्छता।</div><div class="line" style="position:relative;">शरदम्भोदजालानि विशीर्यन्ते समन्ततः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0285"><div class="verse-block verse-justify"><div class="line">तां नावमिव पर्यस्तां मज्जमानां महार्णवे।</div><div class="line" style="position:relative;">तव सेनां महाराज सव्यसाची व्यकम्पयत्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0286"><div class="verse-block verse-justify"><div class="line">क्वनु ते सूतपुत्रोऽभूत्क्वनु द्रोणः सहात्मजः।</div><div class="line" style="position:relative;">अहं क्व च क्व चात्मा ते हार्दिक्यश्च तथा क्वनु॥२८॥</div></div></div>
<p class="prose">दुःशासनश्च ते भ्राता भ्रातृभिः सहितः क्वनु</p>
<div class="verse-block-wrapper" id="09-śalyaparva-b0288"><div class="verse-block verse-justify"><div class="line">वाणगोचरसम्प्राप्तं युध्यमानं जयद्रथम्।</div><div class="line">सम्बन्धिनस्ते भ्रातॄंश्च साहयान्मातुलांस्तथा।</div><div class="line" style="position:relative;">सर्वान्विक्रम्य मिपतो लोकमाक्रम्य मूर्धनि॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0289"><div class="verse-block verse-justify"><div class="line">जयद्रथो हतो राजन्किन्नु शेषमुपास्महे।</div><div class="line" style="position:relative;">को वेह स पुमानास्ते यो विजेष्यति पाण्डवम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0290"><div class="verse-block verse-justify"><div class="line">तस्य चास्त्राणि दिव्यानि विदितानि महात्मनः।</div><div class="line" style="position:relative;">गाण्डीवस्य च निर्घोषो धैर्याणि हरते हि नः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0291"><div class="verse-block verse-justify"><div class="line">नष्टचन्द्रा यथा रात्रिः सेनेयं हतनायका।</div><div class="line" style="position:relative;">नागभग्नद्रुमा शुष्का नदीव प्रतिभाति मे॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0292"><div class="verse-block verse-justify"><div class="line">ध्वजिन्यां हतनेत्रायां यथेष्टं श्वेतवाहनः।</div><div class="line" style="position:relative;">चरिष्यति महाराजः कक्षेष्वग्निरिव ज्वलन्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0293"><div class="verse-block verse-justify"><div class="line">सात्यकेश्चैव यो वेगो भीमसेनस्य चोभयोः।</div><div class="line" style="position:relative;">दारयेत गिरीन्सर्वाञ्शोषयेच्चैव सागरान्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0294"><div class="verse-block verse-justify"><div class="line">उवाच वाक्यं यद्भीमः सभामध्ये विशाम्पते।</div><div class="line" style="position:relative;">कृतं तत्सफलं सर्वं भूयश्चैव करिष्यति॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0295"><div class="verse-block verse-justify"><div class="line">प्रमुखस्थे तदा कर्णे बलं पाण्डवरक्षितम्।</div><div class="line" style="position:relative;">दुरासदं तदा गुप्तं व्यूढं गाण्डीवधन्वना॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0296"><div class="verse-block verse-justify"><div class="line">युष्माभिस्तानि चीर्णानि यान्यसाधूनि साधुषु।</div><div class="line" style="position:relative;">अकारणकृतान्येव तेषां वः फलमागतम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0297"><div class="verse-block verse-justify"><div class="line">आत्मनोऽर्थे त्वया लोके यत्नतः सर्व आहृतः।</div><div class="line" style="position:relative;">स ते संशयितस्तात आत्मा च भरतर्षभ॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0298"><div class="verse-block verse-justify"><div class="line">रक्ष दुर्योधनात्मानमात्मा सर्वस्य भाजनम्।</div><div class="line" style="position:relative;">भिन्ने हि भाजने तात दिशो गच्छति तद्गतम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0299"><div class="verse-block verse-justify"><div class="line">हीयमानेन वै सन्धिः पर्येष्टव्यः समेन वा।</div><div class="line" style="position:relative;">विग्रहो वर्धमानेन नीतिरेषा बृहस्पतेः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0300"><div class="verse-block verse-justify"><div class="line">ते वयं पाण्डुपुत्रेभ्यो हीनाः स्म बलशक्तितः।</div><div class="line" style="position:relative;">अत्र ते पाण्डवैः सार्धं सन्धिं मन्ये क्षमं प्रभो॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0301"><div class="verse-block verse-justify"><div class="line">न जानीते हि यः श्रेयः श्रेयसश्चावमन्यते।</div><div class="line" style="position:relative;">स क्षिप्रं भ्रश्यते राज्यान्न च श्रेयोऽनुविन्दति॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0302"><div class="verse-block verse-justify"><div class="line">अणिपत्य हि राजानं राज्यं यदि लभेमहि।</div><div class="line" style="position:relative;">श्रेयः स्यान्न तु मौढ्येन राजन्गन्तुं पराभवम्॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0303"><div class="verse-block verse-justify"><div class="line">वैचित्रवीर्यवचनात्कृपाशीलो युधिष्ठिरः।</div><div class="line" style="position:relative;">विनियुञ्जीत राज्ये त्वां गोविन्दवचनेन च॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0304"><div class="verse-block verse-justify"><div class="line">`अजातशत्रुः कौरव्यो गुरुशुश्रूषणे रतः।</div><div class="line" style="position:relative;">धृतराष्ट्रस्य वचनं नावमंस्यति धार्मिकः॥४५॥</div></div></div>
<p class="prose">कुर्वन्ति भ्रातरश्चास्य वचनं नात्र संशयः</p>
<div class="verse-block-wrapper" id="09-śalyaparva-b0306"><div class="verse-block verse-justify"><div class="line">यद्ब्रूयाद्धि हृषीकेशो राजानमपराजितम्।</div><div class="line" style="position:relative;">अर्जुनो भीमसेनश्च सर्वे कुर्युरसंशयम्॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0307"><div class="verse-block verse-justify"><div class="line">नातिक्रमिष्यते कृष्णो वचनं पाण्डवस्य तु।</div><div class="line" style="position:relative;">धृतराष्ट्रस्य मन्येऽहं नापि कृष्णस्य पाण्डवः॥४७॥</div></div></div>
