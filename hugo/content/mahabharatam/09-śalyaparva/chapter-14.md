<div class="verse-block-wrapper" id="09-śalyaparva-b0791"><div class="verse-block verse-justify"><div class="line">द्रौणिं विव्याध समरे त्रिभिरेव शिलीमुखैः।</div><div class="line">तथेतरान्महेष्वासान्द्वाभ्यां द्वाभ्यां धनञ्जयः।</div><div class="line" style="position:relative;">भूयश्चैव महाराज शरवर्षैरवाकिरत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0792"><div class="verse-block verse-justify"><div class="line">शरकण्टकितास्ते तु तावका भरतर्षभ।</div><div class="line" style="position:relative;">न जहुः पार्थमासाद्य ताड्यमानाः शितैः शरैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0793"><div class="verse-block verse-justify"><div class="line">अर्जुनं रथवंशेन द्रोणपुत्रपुरोगमाः।</div><div class="line" style="position:relative;">परिवार्य मुदा युक्ता योधयन्तश्चकाशिरे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0794"><div class="verse-block verse-justify"><div class="line">तैस्तु क्षिप्ताः शरा राजन्कार्तस्वरविभूषिताः।</div><div class="line" style="position:relative;">अर्जुनस्य रथोपस्थं पूरयामासुरञ्जसा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0795"><div class="verse-block verse-justify"><div class="line">तथा कृष्णौ महेष्वासौ वृषभौ सर्वधन्विनाम्।</div><div class="line" style="position:relative;">विव्यधुश्च शरैर्घोरैः प्रहृष्टा युद्धदुर्मदाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0796"><div class="verse-block verse-justify"><div class="line">कूबरं रथचक्राणि ईषा योक्त्राणि वा विभो।</div><div class="line" style="position:relative;">युगं चैवानुकर्षं च शरभूतमभूत्तदा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0797"><div class="verse-block verse-justify"><div class="line">नैतादृशं दृष्टपूर्वं राजन्नेव च नः श्रुतम्।</div><div class="line" style="position:relative;">यादृशं तत्र पार्थस्य तावकाः सम्प्रचक्रिरे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0798"><div class="verse-block verse-justify"><div class="line">स रथः सर्वतो भाति चित्रपुङ्खैः शितैः शरैः।</div><div class="line" style="position:relative;">उल्काशतैः सम्प्रदीप्तं विमानमिव भूतले॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0799"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनो महाराज शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">अवाकिरत्तां पृतनां मेघो धृष्ट्येव पर्वतम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0800"><div class="verse-block verse-justify"><div class="line">ते वध्यमानाः समरे पार्थनामाङ्कितैः शरैः।</div><div class="line" style="position:relative;">पार्थभूतममन्यन्त प्रेक्षमाणास्तथाविधम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0801"><div class="verse-block verse-justify"><div class="line">कोपोद्धूतशरज्वालो धनुःशब्दानिलो महान्।</div><div class="line" style="position:relative;">सैन्येन्धनं ददाहाशु तावकं पार्थपावकः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0802"><div class="verse-block verse-justify"><div class="line">चक्राणां पततां चापि युगानां च धरातले।</div><div class="line" style="position:relative;">तूणीराणां पताकानां ध्वजानां च रथैः सह॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0803"><div class="verse-block verse-justify"><div class="line">ईषाणामनुकर्षाणां त्रिवेणूनां च भारत।</div><div class="line" style="position:relative;">अक्षाणामथ योक्त्राणां प्रतोदानां च राशयः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0804"><div class="verse-block verse-justify"><div class="line">शिरसां पततां चापि कुण्डलोष्णीषधारिणाम्।</div><div class="line" style="position:relative;">भुजानां च महाभाग स्कन्धानां च समन्ततः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0805"><div class="verse-block verse-justify"><div class="line">छत्राणां व्यजनैः सार्धं मकुटानां च राशयः।</div><div class="line" style="position:relative;">समदृश्यन्त पार्थस्य रथमार्गेषु भारत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0806"><div class="verse-block verse-justify"><div class="line">अगम्यरूपा पृथिवी मांसशोणितकर्दमा।</div><div class="line">भीरूणां त्रासजननी शूराणां हर्षवर्धिनी।</div><div class="line" style="position:relative;">बभूव भरतश्रेष्ठ रुद्रस्याक्रीडनं यथा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0807"><div class="verse-block verse-justify"><div class="line">हत्वा तु समरे पार्थः सहस्रे द्वे परन्तपः।</div><div class="line" style="position:relative;">रथानां सवरूथानं विधूमोऽग्निरिव ज्वलन्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0808"><div class="verse-block verse-justify"><div class="line">यथा हि भगवानग्निर्जगद्दग्ध्वा चराचरम्।</div><div class="line" style="position:relative;">विधूमो दृश्यते राजंस्तथा पार्थो धऩञ्जयः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0809"><div class="verse-block verse-justify"><div class="line">द्रौणिस्तु समरे दृष्ट्वा पाण्डवस्य पराक्रमम्।</div><div class="line" style="position:relative;">रथेनातिपताकेन पाण्डवं प्रत्यवारयत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0810"><div class="verse-block verse-justify"><div class="line">तावुभौ पुरुषव्याघ्रौ श्वेताश्वौ धन्विनां वरौ।</div><div class="line" style="position:relative;">समीयतुस्तदाऽन्योन्यं परस्परवधैषिणौ॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0811"><div class="verse-block verse-justify"><div class="line">तयोरासीन्महाराज बाणवर्षं सुदारुणम्।</div><div class="line" style="position:relative;">जीमूतयोर्यथा वृष्टिस्तपान्ते भरतर्षभ॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0812"><div class="verse-block verse-justify"><div class="line">अन्योन्यस्पर्धिनौ तौ तु शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">ततक्षतुस्तदाऽन्योन्यं शृङ्गाभ्यां वृषभाविव॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0813"><div class="verse-block verse-justify"><div class="line">तयोर्युद्धं महाराज चिरं सममिवाभवत्।</div><div class="line" style="position:relative;">शस्त्राणां सङ्गमश्चैव घोरस्तत्राभवत्पुनः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0814"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनं द्वादशभी रुक्मपुङ्खैः सुतेजनैः।</div><div class="line" style="position:relative;">वासुदेवं च दशभिर्द्रौणिर्विव्याध भारत॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0815"><div class="verse-block verse-justify"><div class="line">ततः प्रहस्य बीभत्सुर्व्याक्षिपद्गाण्डिवं धनुः।</div><div class="line" style="position:relative;">मानयित्वा मुहूर्तं तु गुरुपुत्रं महाहवे॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0816"><div class="verse-block verse-justify"><div class="line">व्यश्वसूतरथं चक्रे सव्यसाची परन्तपः।</div><div class="line" style="position:relative;">मृदुपूर्वं ततश्चैनं पुनःपुनरताडयत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0817"><div class="verse-block verse-justify"><div class="line">हताश्वे तु रथे तिष्ठन्द्रोणपुत्रस्त्वयस्मयम्।</div><div class="line" style="position:relative;">मुसलं पाण्डुपुत्राय चिक्षेप परिघोपमम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0818"><div class="verse-block verse-justify"><div class="line">तमापतन्तं सहसा हेमपट्टविभूषितम्।</div><div class="line" style="position:relative;">चिच्छेद सप्तधा वीरः पार्थः शत्रुनिबर्हणः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0819"><div class="verse-block verse-justify"><div class="line">स च्छिन्नं मुसलं दृष्ट्वा द्रौणिः परमकोपनः।</div><div class="line">आददे परिघं घोरं नगेन्द्रशिखरोपमम्।</div><div class="line" style="position:relative;">चिक्षेप चैव पार्थाय द्रौणिर्युद्धविशारदः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0820"><div class="verse-block verse-justify"><div class="line">तमन्तकमिव क्रुद्दं परिघं प्रेक्ष्य पाण्डवः।</div><div class="line" style="position:relative;">अर्जुनस्त्वरितो जघ्ने पञ्चभिः सायकोत्तमैः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0821"><div class="verse-block verse-justify"><div class="line">स च्छिन्नः पतितो भूमौ पार्थबाणैर्महाहवे।</div><div class="line" style="position:relative;">दारयन्पृथिवीन्द्राणां मनः सब्देन भारत॥३३॥</div></div></div>
<p class="prose">ततोऽपरैस्त्रिभिर्भल्लैर्द्रौणिं विव्याध पाण़्डवः</p>
<div class="verse-block-wrapper" id="09-śalyaparva-b0823"><div class="verse-block verse-justify"><div class="line">सोऽतिविद्धो बलवता पार्थेन सुमहात्मना।</div><div class="line" style="position:relative;">नाकम्पत तदा द्रौणिः पौरुषेषु व्यवस्थितः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0824"><div class="verse-block verse-justify"><div class="line">सुरथं च ततो राजन्भारद्वाजो महारथम्।</div><div class="line" style="position:relative;">अवाकिरच्छरव्रातैः सर्वक्षत्रस्य पश्यतः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0825"><div class="verse-block verse-justify"><div class="line">ततस्तु सुरथोऽप्याजौ पाञ्चालानां महारथः।</div><div class="line" style="position:relative;">रथेन मेघघोषेण द्रौणिमेवाभ्यधावत॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0826"><div class="verse-block verse-justify"><div class="line">विकर्षन्वै धनुःश्रेष्ठं सर्वभारसहं दृढम्।</div><div class="line" style="position:relative;">ज्वलनाशीविषनिभैः शरैश्चैनमवाकिरत्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0827"><div class="verse-block verse-justify"><div class="line">सुरथं तं ततः क्रुद्धमापतन्तं महारथम्।</div><div class="line" style="position:relative;">चुकोप समरे द्रौणिर्दण्डाहत इवोरगः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0828"><div class="verse-block verse-justify"><div class="line">त्रिशिखां भ्रुकुटीं कृत्वा सृक्विणी परिसंलिहन्।</div><div class="line">उद्वीक्ष्य सुरथं रोषाद्धनुर्ज्यामवमृज्य च।</div><div class="line" style="position:relative;">मुमोच तीक्ष्णं नाराचं यमदण्डोपमद्युतिम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0829"><div class="verse-block verse-justify"><div class="line">स तस्य हृदयं भित्त्वा प्रविवेशातिवेगितः।</div><div class="line" style="position:relative;">शक्ताशनिरिवोत्सृष्टो विदार्य धरणीतलम्॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0830"><div class="verse-block verse-justify"><div class="line">ततः स पतितो भूमौ नाराचेन समाहतः।</div><div class="line" style="position:relative;">वज्रेण च यथा शृङ्गं पर्वतस्येव दीर्यतः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0831"><div class="verse-block verse-justify"><div class="line">तस्मिन्विनिहते वीरे द्रोणपुत्रः प्रतापवान्।</div><div class="line" style="position:relative;">आरुरोह रथं तूर्णं तमेव रथिनां वरः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0832"><div class="verse-block verse-justify"><div class="line">ततः सज्जो महाराज द्रौणिराहवदुर्मदः।</div><div class="line" style="position:relative;">अर्जुनं योधयामास संशप्तकवृतो रणे॥४३॥</div></div></div>
