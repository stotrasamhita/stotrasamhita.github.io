<div class="verse-block-wrapper" id="09-śalyaparva-b1543"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1544"><div class="verse-block verse-justify"><div class="line">दुर्योधनो महाराज सुदर्शश्चापि ते सुतः।</div><div class="line" style="position:relative;">हतशेषौ तदा सङ्ख्ये वाजिमध्ये व्यवस्थितौ॥२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1545"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनं दृष्ट्वा वाजिमध्ये व्यवस्थितम्।</div><div class="line" style="position:relative;">उवाच देवकीपुत्रः कुन्तीपुत्रं धनञ्जयम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1546"><div class="verse-block verse-justify"><div class="line">शत्रवो हतभूयिष्ठा ज्ञातयः परिपालिताः।</div><div class="line" style="position:relative;">गृहीत्वा सञ्जयं चासौ निवृत्तः शिनिपुङ्गवः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1547"><div class="verse-block verse-justify"><div class="line">परिश्रान्तश्च नकुलः सहदेवश्च भारत।</div><div class="line" style="position:relative;">योधयित्वा रणे पापान्धार्तराष्ट्रान्सहानुगान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1548"><div class="verse-block verse-justify"><div class="line">दुर्योधनमतिक्रम्य त्रय एते व्यवस्थिन्ताः।</div><div class="line" style="position:relative;">कृपश्च कृतवर्मा च द्रौणिश्चैव महारथः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1549"><div class="verse-block verse-justify"><div class="line">असौ तिष्ठति पाञ्चाल्यः श्रिया परमया युतः।</div><div class="line" style="position:relative;">दुर्योधनबलं हत्वा सह सर्वैः प्रभद्रकैः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1550"><div class="verse-block verse-justify"><div class="line">असौ दुर्योधनः पार्थ वाजिमध्ये व्यवस्थितः।</div><div class="line" style="position:relative;">छत्रेण ध्रियामणेन प्रेक्षमाणो मुहुर्मुहुः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1551"><div class="verse-block verse-justify"><div class="line">प्रतिव्यूह्य बलं सर्वं रणमध्ये व्यवस्थितः।</div><div class="line" style="position:relative;">एनं हत्वा शितैर्बाणैः कृतकृत्यो भविष्यसि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1552"><div class="verse-block verse-justify"><div class="line">गजानीकं हतं दृष्ट्वा त्वां च प्राप्तमरिन्दम।</div><div class="line" style="position:relative;">यावन्न विद्रवन्त्येते तावज्जहि सुयोधन्सम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1553"><div class="verse-block verse-justify"><div class="line">यातु कश्चित्तुं पाञ्चाल्यं क्षिप्रमागम्यतामिति।</div><div class="line" style="position:relative;">परिश्रान्तबलस्तात नैष मुच्येत किल्बिषी॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1554"><div class="verse-block verse-justify"><div class="line">हत्वा तव बलं सर्वं सङ्ग्रमे धृतराष्ट्रजः।</div><div class="line" style="position:relative;">जितान्पाण्डुसुतान्प्रत्वा रूपं धारयते महत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1555"><div class="verse-block verse-justify"><div class="line">निहतं स्वबलं दृष्ट्वा पीडितं चापि पाण्डवैः।</div><div class="line" style="position:relative;">ध्रुवमेष्यति सङ्ग्रमे वधायैवात्मनो नृपः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1556"><div class="verse-block verse-justify"><div class="line">एवमुक्तः फल्गुनस्तु कृष्णं वचनमब्रवीत्।</div><div class="line">धृतराष्ट्रसुताः सर्वे हता भीमेन माधव।</div><div class="line" style="position:relative;">यावेतावास्थितौ कृष्ण तावद्य नभविष्यतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1557"><div class="verse-block verse-justify"><div class="line">हतो भीष्मो हतो द्रोणः कर्णो वैकर्तनो हतः।</div><div class="line" style="position:relative;">मद्रराजो हतः शल्यो हतः कृष्ण जयद्रथः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1558"><div class="verse-block verse-justify"><div class="line">हयाः पञ्चशताः शिष्टाः शकुनेः सौबलस्य च।</div><div class="line">रथानां तु शते शिष्टे द्वे एव तु जनार्दन।</div><div class="line" style="position:relative;">दन्तिनां च शतं साग्रं त्रिसाहस्राः पदातयः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1559"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा कृपश्चैव त्रिगर्ताधिपतिस्तथा।</div><div class="line" style="position:relative;">उलूकः शकुनिश्चैव कृतवर्मा च सात्वतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1560"><div class="verse-block verse-justify"><div class="line">एतद्बलमभूच्छेषं धार्तराष्ट्रस्य माधव।</div><div class="line" style="position:relative;">मोक्षो न नूनं कालात्तु विद्यते भुवि कस्यचित्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1561"><div class="verse-block verse-justify"><div class="line">तथा विनिहते सैन्ये पश्य दुर्योधनं स्थितम्।</div><div class="line">अद्याह्ना हि महाराजो हतामित्रो भविष्यति।</div><div class="line" style="position:relative;">न हि मे मोक्ष्यते कश्चित्परेषामिह चिन्तये॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1562"><div class="verse-block verse-justify"><div class="line">ये त्वद्य समरं कृष्ण न हास्यन्ति मदोत्कटाः।</div><div class="line" style="position:relative;">तान्वै सर्वान्हनिष्यामि यद्यपि स्युरमानुषाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1563"><div class="verse-block verse-justify"><div class="line">अद्य युद्धे समुत्पन्नं दीर्घं राज्ञः प्रजागरम्।</div><div class="line" style="position:relative;">अपनेष्यामि गान्धारिं घातयित्वा शितैः शरैः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1564"><div class="verse-block verse-justify"><div class="line">निकृत्या वै दुराचारो यानि रत्नानि सौबलः।</div><div class="line" style="position:relative;">सभायामहरद्द्यूते पुनस्तान्याहराम्यहम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1565"><div class="verse-block verse-justify"><div class="line">अद्य वेत्स्यन्ति मच्छक्तिं सर्वा नागपुरे स्त्रियः।</div><div class="line" style="position:relative;">श्रुत्वा पतींश्च पुत्रांश्च पाण्डवैर्निहतान्युधि॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1566"><div class="verse-block verse-justify"><div class="line">समाप्तमद्य वै कर्म सर्वं कृष्ण भविष्यति।</div><div class="line" style="position:relative;">अद्य दुर्योधनो दीप्तां श्रियं प्राणांश्च मोक्ष्यति॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1567"><div class="verse-block verse-justify"><div class="line">नापयाति भयात्कृष्ण सङ्ग्रामाद्यदि चेन्मम।</div><div class="line" style="position:relative;">निहतं विद्धि वार्ष्णेय धार्तराष्ट्रं सुबालिशम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1568"><div class="verse-block verse-justify"><div class="line">मम ह्येतदपर्याप्तं वाजिबृन्दमरिन्दम।</div><div class="line">सोढुं ज्यातलनिर्घोषं याहि यावन्निहन्म्यहम् ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1569"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु दाशार्हः पाण्डवेन यशस्विना।</div><div class="line" style="position:relative;">अचोदयद्धायन्राजन्दुर्योधनबलं प्रति॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1570"><div class="verse-block verse-justify"><div class="line">तदनीकमभिप्रेक्ष्य त्रयः सज्जा महारथाः।</div><div class="line">भीमसेनोऽकर्जुनश्चैव सहदेवश्च मारिष।</div><div class="line" style="position:relative;">प्रययुः सिंहनादेन दुर्योधनजिघांसया॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1571"><div class="verse-block verse-justify"><div class="line">तान्प्रेक्ष्य सहितान्सर्वाञ्जवेनोद्यतकार्मुकान्।</div><div class="line" style="position:relative;">सौबलोऽभ्यद्रवद्युद्धे पाण्डवानाततायिनः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1572"><div class="verse-block verse-justify"><div class="line">सुदर्शनस्त्व मतो भीमसेनं समभ्ययात्।</div><div class="line" style="position:relative;">सुशर्मा शकुनिश्चैव युयुधाते किरीटिना॥३०॥</div></div></div>
<p class="prose">सहदेवं तव सुतो हयपृष्ठगतोऽभ्ययात्</p>
<div class="verse-block-wrapper" id="09-śalyaparva-b1574"><div class="verse-block verse-justify"><div class="line">ततो हि यत्नतः क्षिप्रं तवं पुत्रो जनाधिप।</div><div class="line" style="position:relative;">प्रासेन सहदेवस्य शिरसि प्राहरद्भृशम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1575"><div class="verse-block verse-justify"><div class="line">सोपाविशद्रथोपस्थे तव सुत्रेण ताडितः।</div><div class="line" style="position:relative;">रुधिराप्लुतसर्वाङ्ग आशीविष इव श्वसन्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1576"><div class="verse-block verse-justify"><div class="line">प्रतिलभ्य ततः सञ्ज्ञां सहदेवो विशाम्पते।</div><div class="line" style="position:relative;">दुर्योधनं शरैस्तीक्ष्णैः सङ्क्रुद्धः समवाकिरत्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1577"><div class="verse-block verse-justify"><div class="line">पार्थोऽपि युधि विक्रम्य कुन्तीपुत्रो धनञ्जयः।</div><div class="line" style="position:relative;">शूराणामश्वपृष्ठेभ्यः शिरांसि निचकर्त ह॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1578"><div class="verse-block verse-justify"><div class="line">तदनीकं तदा पार्थो व्यधमद्बहुभिः शरैः।</div><div class="line" style="position:relative;">पातयित्वा हयान्सर्वांस्त्रिगर्तानां रथान्ययौ॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1579"><div class="verse-block verse-justify"><div class="line">ततस्ते सहिता भूत्वा त्रिगर्तानां महारथाः।</div><div class="line" style="position:relative;">अर्जुनं वासुदेवं च शरवर्षैरवाकिरन्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1580"><div class="verse-block verse-justify"><div class="line">सत्यकर्माणमाक्षिप्य क्षुरप्रेण महायशाः।</div><div class="line" style="position:relative;">ततोऽस्य स्यन्दनस्येषां चिच्छिदे पाण्डुनन्दनः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1581"><div class="verse-block verse-justify"><div class="line">शिलाशितेन च विभो क्षुरप्रेण महायशाः।</div><div class="line" style="position:relative;">शिरश्चिच्छेद सहसा तप्तकुण्डलभूषणम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1582"><div class="verse-block verse-justify"><div class="line">सत्येषुमथ चादत्त योधानां मिषतां ततः।</div><div class="line" style="position:relative;">यथा सिंहो वने राजन्मृगं परि बुभुक्षितः॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1583"><div class="verse-block verse-justify"><div class="line">तं निहत्य ततः पार्थः सुशर्माणं त्रिभिः शरैः।</div><div class="line" style="position:relative;">विद्ध्वा तानहनत्सर्वान्रथान्रुक्मविभूओषितान्॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1584"><div class="verse-block verse-justify"><div class="line">ततः प्रायात्त्वरन्पार्थो दीर्घकालं सुसंवृतम्।</div><div class="line" style="position:relative;">मुञ्जन्क्रोधविषं तीक्ष्णं प्रस्थलाधिपतिं प्रति॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1585"><div class="verse-block verse-justify"><div class="line">तमर्जुनः पृषत्कानां शतेन भरतर्षभ।</div><div class="line" style="position:relative;">पूरयित्वा ततो वाहान्प्राहरत्तस्य धन्विनः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1586"><div class="verse-block verse-justify"><div class="line">ततः शरं समादाय यमदण्डोपमं तदा।</div><div class="line" style="position:relative;">सुशर्माणं समुद्दिश्य चिक्षेपाशु हसन्निव॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1587"><div class="verse-block verse-justify"><div class="line">स शरः प्रेषितस्तेन क्रोधदीप्तेन धन्विना।</div><div class="line" style="position:relative;">सुशर्माणं समासाद्य बिभेद हृदयं रणे॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1588"><div class="verse-block verse-justify"><div class="line">स गतासुर्महाराज पपात धरणीतले।</div><div class="line" style="position:relative;">नन्दयन्पाण्डवान्सर्वान्व्यथयंश्चापि तावकान्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1589"><div class="verse-block verse-justify"><div class="line">सुशर्माणं रणे हत्वा पुत्रानस्य महारथान्।</div><div class="line" style="position:relative;">सप्त चाष्टौ च त्रिंशच्च सायकैरनयत्क्षयम्॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b1590"><div class="verse-block verse-justify"><div class="line">ततोऽस्य निशितैर्बाणैः सर्वान्हत्वा पदानुगान्।</div><div class="line" style="position:relative;">अभ्यगाद्भारतीं सेनां हतशेषां महारथः॥४७॥</div></div></div>
