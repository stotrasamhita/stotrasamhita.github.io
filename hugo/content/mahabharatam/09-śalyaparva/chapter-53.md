<div class="verse-block-wrapper" id="09-śalyaparva-b3060"><div class="verse-block verse-justify"><div class="line">तस्यातितपसः शक्रो बिभेति सततं विभो।</div><div class="line" style="position:relative;">न स लोभयितुं शक्यः फलैर्बहुविधैरपि॥७॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3064"><div class="verse-block verse-justify"><div class="line">कुक्षौ चाप्यदधद्वृष्टा तद्रेतः पुरुषर्षभ।</div><div class="line" style="position:relative;">सा दधार च तं गर्भं पुत्रहेतोर्महात्मनः ॥थ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3065"><div class="verse-block verse-justify"><div class="line">सुषुवे चापि समये पुत्रं सारस्वतं वरम्।</div><div class="line" style="position:relative;">जगाम पुत्रमादाय तमृषिं प्रति च प्रभो॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3067"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा तेऽप्सरसं रेतो यत्स्कन्नं प्रागलम्बुसाम्।</div><div class="line" style="position:relative;">तत्कुक्षिणाऽहं ब्रह्मर्षे त्वद्भक्त्या धृतवत्यहम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3068"><div class="verse-block verse-justify"><div class="line">न विनाशमिदं गच्छेत्त्वत्तेज इति निश्चयात्।</div><div class="line" style="position:relative;">प्रतिगृह्णीष्व पुत्रं स्वं मया दत्तमनिन्दितम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3069"><div class="verse-block verse-justify"><div class="line">इत्युक्तः प्रतिजग्राह प्रीतिं चावाप पुष्कलाम्।</div><div class="line" style="position:relative;">पितृवच्चोपजिघ्रत्तं मूर्ध्नि प्रेम्णा द्विजोत्तमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3070"><div class="verse-block verse-justify"><div class="line">परिष्वज्य चिरं कालं तदा भरतसत्तम।</div><div class="line" style="position:relative;">सरस्वत्यै वरं प्रादात्प्रीयमाणो महामुनिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3071"><div class="verse-block verse-justify"><div class="line">विश्वेदेवाः सपितरो गन्धर्वाप्सरसां गणाः।</div><div class="line" style="position:relative;">तृप्तिं यास्यन्ति सुभगे तर्प्यमाणास्तवाम्भसा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3072"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा स तु तुष्टाव वचोभिर्वै महानदीम्।</div><div class="line">प्रीतः परमहृष्टात्मा यथावच्छृणु पार्थिव ॥दधीचिरुवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3073"><div class="verse-block verse-justify"><div class="line">प्रस्रुताऽसि महाभागे सरसो ब्रह्मणः पुरा।</div><div class="line" style="position:relative;">जानन्ति त्वां सरिच्छ्रेष्ठे मुनयः संशितव्रताः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3074"><div class="verse-block verse-justify"><div class="line">मम प्रियकरी चापि सततं प्रियदर्शने।</div><div class="line" style="position:relative;">तस्मात्सारस्वतं पुत्रमदधा वरवर्णिनि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3075"><div class="verse-block verse-justify"><div class="line">तवैव नाम्ना प्रथितः पुत्रस्ते लोकभावनः।</div><div class="line" style="position:relative;">सारस्वत इति ख्यातो भविष्यति महातपाः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3076"><div class="verse-block verse-justify"><div class="line">एष द्वादशवार्षिक्यामनावृष्ट्यां द्विजर्षभान्।</div><div class="line" style="position:relative;">सारस्वतो महाभागे वेदानध्यापयिष्यति॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3077"><div class="verse-block verse-justify"><div class="line">पुण्याभ्यश्च सरिद्भ्यस्त्वं सदा पुण्यतमा शुभे।</div><div class="line" style="position:relative;">भविष्यसि महाभागे मत्प्रसादात्सरस्वति॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b3081"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीत्सुराञ्शक्रो न मे शक्याः सुरारयः।</div><div class="line" style="position:relative;">ऋतेऽस्थिभिर्दधीचस्य निहन्तुं त्रिदशद्विषः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3082"><div class="verse-block verse-justify"><div class="line">तस्माद्यत्नादृषिश्रेष्ठो याच्यतां कार्यसिद्धये।</div><div class="line" style="position:relative;">दधीचास्थीनि देहीति तैर्वधिष्यामहे रिपून्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3083"><div class="verse-block verse-justify"><div class="line">स च तैर्याचितोऽस्थीनि यत्नादृषिवरस्तदा।</div><div class="line" style="position:relative;">साहाय्यं नः कुरुष्वेति चकारैवाविचारयन्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3084"><div class="verse-block verse-justify"><div class="line">स लोकानक्षयान्प्राप्तो देवप्रियकरस्तदा।</div><div class="line" style="position:relative;">तस्यास्थिभिरथो शक्रः सम्प्रहृष्टमनास्तदा॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3085"><div class="verse-block verse-justify"><div class="line">कारयामास दिव्यानि नानाप्रहरणान्युत।</div><div class="line" style="position:relative;">गदा वज्राणि चक्राणि गुरून्दण्डांश्च पुष्कलान्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b3086"><div class="verse-block verse-justify"><div class="line">स हि तीव्रेण तपसा सम्भृतः परमर्षिणा।</div><div class="line" style="position:relative;">प्रजापतिसुतेनाथ भृगुणा लोकभावनः॥३३॥</div></div></div>
