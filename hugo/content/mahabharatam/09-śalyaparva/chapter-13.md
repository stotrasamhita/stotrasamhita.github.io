<div class="verse-block-wrapper" id="09-śalyaparva-b0742"><div class="verse-block verse-justify"><div class="line">सात्यकिश्च शतेनैनं धर्मपुत्रपरीप्सया।</div><div class="line" style="position:relative;">मद्रेश्वरमवाकीर्य सिंहनादमथानदत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0743"><div class="verse-block verse-justify"><div class="line">नकुलः पञ्चभिश्चैनं सहदेवश्च प़ञ्चभिः।</div><div class="line" style="position:relative;">विद्ध्वा तं तु पुनस्तूर्णं ततो विव्याध सप्तभिः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0744"><div class="verse-block verse-justify"><div class="line">स तु शूरो रणे यत्तः पीडितस्तैर्महारथैः।</div><div class="line" style="position:relative;">विकृष्य कार्मुकं घोरं भारघ्नं वेगवत्तरम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0745"><div class="verse-block verse-justify"><div class="line">सात्यकिं पञ्चविंशत्या शल्यो विव्याध मारिष।</div><div class="line" style="position:relative;">भीमसेनं तु सप्तत्या नकुलं सप्तभिस्तथा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0746"><div class="verse-block verse-justify"><div class="line">ततः स विशिखं चापं सहदेवस्य धन्विनः।</div><div class="line" style="position:relative;">छित्त्वा भल्लेन समरे विव्याधैनं त्रिसप्तभिः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0747"><div class="verse-block verse-justify"><div class="line">सहदेवस्तु समरे मातुलं भूरिवर्चसम्।</div><div class="line">सज्यमन्यद्धनुः कृत्वा पञ्चभिः समताडयत्।</div><div class="line" style="position:relative;">शरैराशीविषाकारैर्ज्वलज्ज्वलनसन्निभैः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0748"><div class="verse-block verse-justify"><div class="line">सारथिं चास्य समरे शरेणानतपर्वणा।</div><div class="line" style="position:relative;">विव्याध भृशसङ्क्रुद्धस्तं वै भूयस्त्रिभिः शरैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0749"><div class="verse-block verse-justify"><div class="line">भीमसेनस्तु सप्तत्या सात्यकिर्नवभिः शरैः।</div><div class="line" style="position:relative;">धर्मराजस्तथा षष्ट्या गात्रे शल्यं समार्पयत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0750"><div class="verse-block verse-justify"><div class="line">ततः शल्यो महाराज निर्विद्धस्तैर्महारथैः।</div><div class="line" style="position:relative;">सुस्राव रुधिरं गात्रैर्गैरिकं पर्वतो यथा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0751"><div class="verse-block verse-justify"><div class="line">तांश्च सर्वान्महेष्वासान्पञ्चभिः पञ्चभिः शरैः।</div><div class="line" style="position:relative;">विव्याध तरसा राजंस्तदद्भुतमिवाभवत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0752"><div class="verse-block verse-justify"><div class="line">ततोऽपरेण भल्लेन धर्मपुत्रस्य मारिष।</div><div class="line" style="position:relative;">धनुश्चिच्छेद समरे सज्जयं स सुमहारथः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0753"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय धर्मपुत्रो युधिष्ठिरः।</div><div class="line" style="position:relative;">साश्वसूतध्वजरथं शल्यं प्राच्छादयच्छरैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0754"><div class="verse-block verse-justify"><div class="line">स च्छाद्यमानः समरे धर्मपुत्रस्य सायकैः।</div><div class="line" style="position:relative;">युधिष्ठिरमथाविध्यद्दशभिर्निशितैः शरैः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0755"><div class="verse-block verse-justify"><div class="line">सात्यकिस्तु ततः क्रुद्धो धर्मपुत्रे शरार्दिते।</div><div class="line" style="position:relative;">मद्राणामधिपं शूरं शरैर्विव्याध पञ्चभिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0756"><div class="verse-block verse-justify"><div class="line">स सात्यकेः प्रचिच्छेद क्षुरप्रेण महद्धनुः।</div><div class="line" style="position:relative;">भीमसेनमुखांस्तांश्च त्रिभिस्त्रिभिरताडयत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0757"><div class="verse-block verse-justify"><div class="line">तस्य क्रुद्धो महाराज सात्यकिः सत्यविक्रमः।</div><div class="line" style="position:relative;">तोमरं प्रेषयामास स्वर्णदण्डं महारणे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0758"><div class="verse-block verse-justify"><div class="line">भीमसेनोऽथ नाराचं ज्वलन्तमिव पन्नगम्।</div><div class="line">नकुलः समरे शक्तिं सहदेवो गदां शुभाम्।</div><div class="line" style="position:relative;">धर्मराजः शतघ्नीं च जिघांसुः शल्यमाहवे॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0759"><div class="verse-block verse-justify"><div class="line">तानापतत एवाशु पञ्चानां वै भुजच्युतान्।</div><div class="line" style="position:relative;">वारयामास समरे शस्त्रसङ्घैः स मद्रराट्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0760"><div class="verse-block verse-justify"><div class="line">सात्यकिप्रहितं शल्यो भल्लैश्चिच्छेद तोमरम्।</div><div class="line" style="position:relative;">प्रहितं भीमसेनेन शरं कनकभूषणम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0761"><div class="verse-block verse-justify"><div class="line">द्विधा चिच्छेद समरे कृतहस्तः प्रतापवान्।</div><div class="line" style="position:relative;">नकुलप्रेषितां शक्तिं हेमदण्डां भयावहाम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0762"><div class="verse-block verse-justify"><div class="line">गदां च सहदेवेन शरौघैः समवारयत्।</div><div class="line" style="position:relative;">शराभ्यां च शतघ्नीं तां राज्ञश्चिच्छेद भारत॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0763"><div class="verse-block verse-justify"><div class="line">पश्यतां पाण्डुपुत्राणां सिंहनादं ननाद च।</div><div class="line" style="position:relative;">नामृष्यत्तत्र शैनेयः शत्रोर्विजयमाहवे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0764"><div class="verse-block verse-justify"><div class="line">अथान्यद्धनुरादाय सात्यकिः क्रोधमूर्च्छितः।</div><div class="line" style="position:relative;">द्वाभ्यां मद्रेश्वरं विद्वा सारथिं च त्रिभिः शरैः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0765"><div class="verse-block verse-justify"><div class="line">ततः शल्यो रणे राजन्सर्वांस्तान्दशभिः शरैः।</div><div class="line" style="position:relative;">विव्याध भृशसङ्क्रुद्धस्तोत्रैरिव महाद्विपान्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0766"><div class="verse-block verse-justify"><div class="line">ते वार्यमाणाः समरे मद्रराज्ञा महारथाः।</div><div class="line" style="position:relative;">न शेकुः सम्मुखे स्थातुं तस्य शत्रुनिषूदनाः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0767"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनो राजा दृष्ट्वा शल्यस्य विक्रमम्।</div><div class="line">निहतान्पाण्डवान्मेने पाञ्चालानथ सृञ्जयान्।</div><div class="line" style="position:relative;">॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0768"><div class="verse-block verse-justify"><div class="line">`तथाविधं महाराज मद्रराजस्य विक्रमम्।</div><div class="line" style="position:relative;">असह्यं मानवैर्युद्धे तद्बभूव नरर्षभ ॥'॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0769"><div class="verse-block verse-justify"><div class="line">ततो राजन्महाबाहुर्भीमसेनः प्रतापवान्।</div><div class="line" style="position:relative;">सन्त्यज्य मनसा प्राणान्मद्राधिपमयोधयत्॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="09-śalyaparva-b0771"><div class="verse-block verse-justify"><div class="line">स चतुर्भिर्महेष्वासैः पाण्डवानां महारथः।</div><div class="line" style="position:relative;">वृतस्तान्योधयामास मद्रराजः प्रतापवान्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0772"><div class="verse-block verse-justify"><div class="line">तस्य धर्मसुतो राजन्क्षुरप्रेण महाहवे।</div><div class="line" style="position:relative;">चक्ररक्षं जघानाशु मद्रराजस्य पार्थिवः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0773"><div class="verse-block verse-justify"><div class="line">तस्मिंस्तु निहते शूरे चक्ररक्षे महारथे।</div><div class="line" style="position:relative;">मद्रराजोऽपि बलवान्सैनिकानावृणोच्छरैः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="09-śalyaparva-b0774"><div class="verse-block verse-justify"><div class="line">समावृतांस्ततस्तांस्तु राजन्वीक्ष्य स्वसैनिकान्।</div><div class="line" style="position:relative;">चिन्तयामास समरे धर्मपुत्रो युधिष्ठिरः॥३७॥</div></div></div>
