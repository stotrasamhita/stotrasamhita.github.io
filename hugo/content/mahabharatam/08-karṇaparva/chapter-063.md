<div class="verse-block-wrapper" id="08-karṇaparva-b3258"><div class="verse-block verse-justify"><div class="line">धनूंषि बाणान्परिघानसिपट्टसतोमरान्।</div><div class="line" style="position:relative;">मुसलानि भुशुण्डीश्च शक्तिरिष्टिपरश्वथान्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3259"><div class="verse-block verse-justify"><div class="line">गदाः प्रासाञ्शितान्कुन्तान्भिंडिपालान्महाङ्कुशान्।</div><div class="line" style="position:relative;">प्रगृह्य क्षिप्रमापेतुः परस्परजिघांसया॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3260"><div class="verse-block verse-justify"><div class="line">बाणज्यातलशब्देन द्यां दिशः प्रदिशो वियत्।</div><div class="line" style="position:relative;">पृथिवीं नेमिघोषेण नादयन्तोऽभ्ययुः परान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3261"><div class="verse-block verse-justify"><div class="line">तेन शब्देन महता संहृष्टाश्चक्रुराहवम्।</div><div class="line" style="position:relative;">वीरा वीरैर्महाप्तोरकलहान्तं तितीर्षवः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3262"><div class="verse-block verse-justify"><div class="line">ज्यातलत्रधनुःशब्दाञ्शस्त्राणां च निपात्यताम्।</div><div class="line" style="position:relative;">ताडितानां च पतता निनादांश्च पदातिनाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3263"><div class="verse-block verse-justify"><div class="line">बाणशब्दांश्च विविधाञ्शूराणां चाभिगर्जताम्।</div><div class="line" style="position:relative;">श्रुत्वा गजा भृशं त्रेसुः पेतुर्मम्लुश्च सैनिकाः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3264"><div class="verse-block verse-justify"><div class="line">तेषां निनदतां चैव शस्त्रवर्षं च मुञ्चताम्।</div><div class="line" style="position:relative;">बहूनाधिरथिर्वीरः प्रममाथेषुभिः परान्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3265"><div class="verse-block verse-justify"><div class="line">पञ्च पाञ्चालवीराणां रथान्दश च पञ्च च।</div><div class="line" style="position:relative;">साश्वसूतध्वजान्कर्णः शरैर्निन्ये यमक्षयम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3266"><div class="verse-block verse-justify"><div class="line">योधमुख्या महावीर्याः पाण्डूनां कर्णमाहवे।</div><div class="line" style="position:relative;">शिक्षितास्तमभिद्रुत्य परिवव्रुः समन्ततः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3267"><div class="verse-block verse-justify"><div class="line">ततः कर्णो द्विषत्सेनां शरवर्षैर्विलोडयन्।</div><div class="line" style="position:relative;">विजगाहाण्डजाकीर्णां पद्मिनीमिव यूथपः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3268"><div class="verse-block verse-justify"><div class="line">द्विषत्सैन्यमवस्कन्द्य राधेयो धनुरुत्तमम्।</div><div class="line" style="position:relative;">विधुन्वानः शितैर्बाणैः शिरांस्युन्मथ्य पातयत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3269"><div class="verse-block verse-justify"><div class="line">`हस्तिनः समहामात्रान्साश्वारोहान्हयानपि।</div><div class="line" style="position:relative;">रथिनोऽप्येकबाणेन भ्रमतश्चावपातयत्'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3270"><div class="verse-block verse-justify"><div class="line">वर्म वा चर्म वाऽश्रित्य निपातमपि देहिनः।</div><div class="line" style="position:relative;">विषेहुर्नास्य संस्पर्शं द्वितीयस्य पतत्त्रिणः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3271"><div class="verse-block verse-justify"><div class="line">वर्मयुक्तेषु देहेषु धनुषोऽस्यञ्शिताञ्शरान्।</div><div class="line" style="position:relative;">मौर्व्या तलत्रे न्यहनत्कशया वाजिनो यथा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3272"><div class="verse-block verse-justify"><div class="line">पाण्डुसृञ्जयपाञ्चालाञ्शरगोचरमागतान्।</div><div class="line" style="position:relative;">ममर्द तरसा कर्णः सिंहो मृगगणानिव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3273"><div class="verse-block verse-justify"><div class="line">ततः पाञ्चालपुत्रश्च द्रौपदेयाश्च मारिष।</div><div class="line" style="position:relative;">यमौ च युयुधानश्च त्वरिताः कर्णमभ्ययुः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3274"><div class="verse-block verse-justify"><div class="line">तेषु व्यायच्छमानेषु कुरुपाञ्चालपाण्डुषु।</div><div class="line" style="position:relative;">प्रियानसून्रणे त्यक्त्वा योधा जघ्नुः परस्परम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3275"><div class="verse-block verse-justify"><div class="line">सुसन्नद्धाः कवचिनः सशिरस्त्राणभूषणाः।</div><div class="line" style="position:relative;">गदाभिर्मुसलैश्चान्ये परिघैश्च महाबालाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3276"><div class="verse-block verse-justify"><div class="line">समभ्यधावन्त भृशं कालदण्डैरिवोद्यतैः।</div><div class="line" style="position:relative;">नर्दन्तश्चाह्वयन्तश्च प्रवल्गन्तश्च मारिष॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3277"><div class="verse-block verse-justify"><div class="line">ततो निजघ्नुरन्योन्यं पेतुश्चान्योन्यताडिताः।</div><div class="line" style="position:relative;">वमन्तो रुधिरं गात्रैर्विमस्तिष्केक्षमायुधाः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3278"><div class="verse-block verse-justify"><div class="line">दन्तपूर्णैः सरुधिरैर्वक्रैर्दाडिमसन्निभैः।</div><div class="line" style="position:relative;">जीवन्त इव चाप्येके तस्थुः शस्त्रौघबृंहिताः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3279"><div class="verse-block verse-justify"><div class="line">परश्वथैश्चाप्यपरे पट्टसैरसिभिस्तथा।</div><div class="line" style="position:relative;">शक्तिभिर्भिण्डिपालैश्च नखरप्रासतोमरैः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3280"><div class="verse-block verse-justify"><div class="line">ततक्षुश्चिच्छिदुश्चान्ये बिभिदुश्चिक्षिपुस्तथा।</div><div class="line" style="position:relative;">सञ्चकर्तुश्च जघ्नुश्च राजन्योधा महारणे॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3281"><div class="verse-block verse-justify"><div class="line">पेतुरन्योन्यनिहता व्यसवो रुधिरोक्षिताः।</div><div class="line" style="position:relative;">क्षरन्तः स्वरसं रक्तं प्रकृत्ताश्चन्दना इव॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3282"><div class="verse-block verse-justify"><div class="line">रथै रथा विनिहता हस्तिभिश्चापि हस्तिनः।</div><div class="line" style="position:relative;">नरैर्नरा हताः पेतुरश्वाश्चाश्वैः सहस्रशः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3283"><div class="verse-block verse-justify"><div class="line">ध्वजाः शिरांसि च्छिन्नाः पेतुर्महीतले ॥`वध्यतां दारुणाः शब्दाः पततां स्तनतामपि।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
