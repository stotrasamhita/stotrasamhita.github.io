<div class="verse-block-wrapper" id="08-karṇaparva-b2523"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2524"><div class="verse-block verse-justify"><div class="line">सुदुष्करमिदं कर्म कृतं भीमेन सञ्जय।</div><div class="line" style="position:relative;">येन कर्णो महाबाहू रथोपस्थे निपातितः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2525"><div class="verse-block verse-justify"><div class="line">कर्णो ह्येको रणे हन्ता पाण्डवान्सृञ्जयैः सह।</div><div class="line" style="position:relative;">इति दुर्योधनः सूत प्राब्रवीन्मां मुहुर्मुहुः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2526"><div class="verse-block verse-justify"><div class="line">पराजितं तु राधेयं दृष्ट्वा भीमेन संयुगे।</div><div class="line">ततः परं किमकरोत्पुत्रो दुर्योधनो मम ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2527"><div class="verse-block verse-justify"><div class="line">विमुखं प्रेक्ष्य राधेयं सूतपुत्रं महाहवे।</div><div class="line" style="position:relative;">पुत्रस्तव महाराज सोदर्यानिदमब्रवीत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2528"><div class="verse-block verse-justify"><div class="line">शीघ्रं गच्छत भद्रं वो राधेयं परिरक्षत।</div><div class="line" style="position:relative;">भीमसेनभयेऽगाधे म़ज्जन्तं व्यसनार्णवे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2529"><div class="verse-block verse-justify"><div class="line">ते तु राज्ञा समादिष्टा भीमसेनं जिघांसवः।</div><div class="line" style="position:relative;">अभ्यवर्तन्त सङ्क्रुद्धाः पतङ्गाः पावकं यथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2530"><div class="verse-block verse-justify"><div class="line">श्रुतायुर्दुर्धरः क्राथो विवित्सुर्विकटः समः।</div><div class="line" style="position:relative;">निषङ्गी कवची पाशी तथा नन्दोपनन्दकौ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2531"><div class="verse-block verse-justify"><div class="line">दुष्प्रधर्षः सुबाहुश्च वातवेगसुवर्चसौ।</div><div class="line" style="position:relative;">धनुर्ग्राहो दुर्मदश्च जलसन्धः शलः सहः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2532"><div class="verse-block verse-justify"><div class="line">एते रथैः परिवृता वीर्यवन्तो महाबलाः।</div><div class="line" style="position:relative;">भीमसेनं समासाद्य समन्तात्पर्यवारयन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2533"><div class="verse-block verse-justify"><div class="line">ते व्यमुञ्चञ्छरव्रातान्नानालिङ्गान्समन्ततः।</div><div class="line" style="position:relative;">स तैरभ्यर्द्यमानस्तु भीमसेनो महाबलः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2534"><div class="verse-block verse-justify"><div class="line">तेषामापततां क्षिप्रं सुतानां ते जनाधिप।</div><div class="line" style="position:relative;">रथैः पञ्चशतैः सार्धं पञ्चाशदहनद्रथान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2535"><div class="verse-block verse-justify"><div class="line">विवित्सोस्तु ततः क्रुद्धो भल्लेनापाहरच्छिरः।</div><div class="line" style="position:relative;">भीमसेनो महाराज तत्पपात हतं भुवि॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2536"><div class="verse-block verse-justify"><div class="line">सकुण्डलशिरस्त्राणं पूर्णचन्द्रोपमं तथा।</div><div class="line" style="position:relative;">`अशोभत महाराज पूर्णचन्द्र इवाम्बरे'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2537"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा निहतं शूरं भ्रातरः सर्वतस्तदा।</div><div class="line" style="position:relative;">अभ्यद्रवन्त समरे भीमं भीमपराक्रमम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2538"><div class="verse-block verse-justify"><div class="line">ततोऽपराभ्यां भल्लाभ्यां पुत्रयोस्ते महाहवे।</div><div class="line" style="position:relative;">जहार समरे प्राणान्भीमो भीमपराक्रमः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2539"><div class="verse-block verse-justify"><div class="line">तौ धरामन्वपद्येतां वातरुग्णाविव द्रुमौ।</div><div class="line" style="position:relative;">विकटश्च सहश्चोमौ देवपुत्रोपमौ नृप॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2540"><div class="verse-block verse-justify"><div class="line">ततस्तु त्वरितो भीमः क्राथं निन्ये यमक्षयम्।</div><div class="line" style="position:relative;">नाराचेन सुतीक्ष्णेन स हतो न्यपतद्भुवि॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2541"><div class="verse-block verse-justify"><div class="line">हाहाकारस्ततस्तीव्रः सम्बभूव जनेश्वर।</div><div class="line" style="position:relative;">वध्यमानेषु वीरेषु तव पुत्रेषु धन्विषु॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2542"><div class="verse-block verse-justify"><div class="line">तेषां सुलुलिते सैन्ये पुनर्भीमो महाबलः।</div><div class="line" style="position:relative;">नन्दोपनन्दौ समरे प्रैषयद्यमसादनम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2543"><div class="verse-block verse-justify"><div class="line">ततस्ते प्राद्रवन्भीताः पुत्रास्ते विह्वलीकृताः।</div><div class="line" style="position:relative;">भीमसेनं रणे दृष्ट्वा कालान्तकयमोपमम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2544"><div class="verse-block verse-justify"><div class="line">पुत्रांस्ते निहतान्दृष्ट्वा सूतपुत्रः सुदुर्मनाः।</div><div class="line" style="position:relative;">हंसवर्णान्हयान्भूयः प्रैषयद्यत्र पाण्डवः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2545"><div class="verse-block verse-justify"><div class="line">ते प्रेषिता महाराज मद्रराजेन वाजिनः।</div><div class="line" style="position:relative;">भीमसेनरथं प्राप्य समसज्जन्त वेगिताः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2546"><div class="verse-block verse-justify"><div class="line">स सन्निपातस्तुमुलो घोररूपो विशाम्पते।</div><div class="line" style="position:relative;">आसीद्रौद्रो महाराज कर्णपाण्डवयोर्मृधे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2547"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा मम महाराज तौ समेतौ महारथौ।</div><div class="line" style="position:relative;">आसीद्बुद्विः कथं युद्वमेतदद्य भविष्यति॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2548"><div class="verse-block verse-justify"><div class="line">ततो भीमो रणश्लाघी छादयामास पत्रिभिः।</div><div class="line" style="position:relative;">कर्णं रणे महाराज पुत्राणां तव पश्यताम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2549"><div class="verse-block verse-justify"><div class="line">ततः कर्णो भृशं क्रुद्धो भीमं नवभिरायसैः।</div><div class="line" style="position:relative;">विव्याध परमास्त्रज्ञो भल्लैः सन्नतपर्वभिः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2550"><div class="verse-block verse-justify"><div class="line">तान्निहत्य महाबाहुर्भीमो भीमपराक्रमः।</div><div class="line" style="position:relative;">आकर्णपूर्णैर्विशिखैः कर्णं विव्याध सप्तभिः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2551"><div class="verse-block verse-justify"><div class="line">ततः कर्णो महाराज आशीविष इव श्वसन्।</div><div class="line" style="position:relative;">शरवर्षेण महता छादयामास पाण्डवम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2552"><div class="verse-block verse-justify"><div class="line">भीमोऽपि ततं शरव्रातैश्छादयित्वा महारथम्।</div><div class="line" style="position:relative;">पश्यतां कौरवेयाणां विननर्द महाबलः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2553"><div class="verse-block verse-justify"><div class="line">ततः कर्णो भृशं क्रुद्धो दृढमादाय कार्मुकम्।</div><div class="line">भीमं विव्याध दशभिः कङ्कपत्रैः शिलाशितैः।</div><div class="line" style="position:relative;">कार्मुकं चास्य चिच्छेद भल्लेन निशितेन च॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2554"><div class="verse-block verse-justify"><div class="line">ततो भीमो महाबाहुर्हेमपट्टविभूषितम्।</div><div class="line">परिघं घोरमादाय मृतत्युदण्डमिवापरम्।</div><div class="line" style="position:relative;">कर्णस्य निधनाकाङ्क्षी चिक्षेपातिबलो नदन्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2555"><div class="verse-block verse-justify"><div class="line">तमापतन्तं परिघं वज्राशनिसमस्वनम्।</div><div class="line" style="position:relative;">चिच्छेद बहुधा कर्णः शरैराशीविषोपमैः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2556"><div class="verse-block verse-justify"><div class="line">ततः कार्मुकमादाय भीमो दृढतरं तदा।</div><div class="line" style="position:relative;">छादयामास विशिखैः कर्णं परबलार्दनम्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2557"><div class="verse-block verse-justify"><div class="line">ततो युद्धमभूद्धोरं कर्णपाण्डवयोर्मृधे।</div><div class="line" style="position:relative;">बलीन्द्रयोरिव मुहुः परस्परवधैषिणोः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2558"><div class="verse-block verse-justify"><div class="line">ततः कर्णो महाराज भीमसेनं त्रिभिः शरैः।</div><div class="line" style="position:relative;">आकर्णपूर्णैर्विव्याध दृढमानम्य कार्मुकम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2559"><div class="verse-block verse-justify"><div class="line">सोऽतिविद्धो महेष्वासः कर्णेन बलिनां वरः।</div><div class="line" style="position:relative;">घोरमादत्त विशिखं कर्णकायावदारणम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2560"><div class="verse-block verse-justify"><div class="line">तस्य भित्त्वा तनुत्राणं भित्त्वा कायं च सायकः।</div><div class="line" style="position:relative;">प्राविशद्वरणीं राजन्वल्मीकमिव पन्नगः॥३८॥</div></div></div>
<p class="prose">स तेनातिप्रहारेण व्यथितो विह्वलन्निव ॥सञ्चचाल रथे कर्णः क्षितिकम्पे यथाऽचलः</p>
<div class="verse-block-wrapper" id="08-karṇaparva-b2562"><div class="verse-block verse-justify"><div class="line">ततः कर्णो महाराज रोषामर्षसमन्वितः।</div><div class="line" style="position:relative;">भीमं तं पञ्चविंशत्या नाराचानां समार्पयत्॥३९॥</div></div></div>
<p class="prose">चिच्छेद कार्मुकं तूर्णं पाण्डवस्याशु पत्रिणा</p>
<div class="verse-block-wrapper" id="08-karṇaparva-b2564"><div class="verse-block verse-justify"><div class="line">तततो मुहूर्ताद्राजेन्द्र नातिकृच्छ्राद्वसन्निव।</div><div class="line" style="position:relative;">विरथं भीमकर्माणं भीमं कर्णश्चकार ह॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2565"><div class="verse-block verse-justify"><div class="line">विरथो भरतश्रेष्ठ प्रहसन्ननिलोपमः।</div><div class="line" style="position:relative;">गदां गृह्य महाबाहुरपतत्स्यन्दनोत्तमात्॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2566"><div class="verse-block verse-justify"><div class="line">गदया च महाराज कर्णस्य रथकूबरम्।</div><div class="line" style="position:relative;">पोथयामास सङ्क्रुद्धः समरे शत्रुतापनः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2567"><div class="verse-block verse-justify"><div class="line">स क्रोधवशमापन्नः पाण्डुपुत्रः प्रतापवान्।</div><div class="line" style="position:relative;">विद्राव्य गदया वीरस्तव पुत्रान्महाहवे॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2568"><div class="verse-block verse-justify"><div class="line">नागान्सप्तशतान्राजन्नीषादन्तान्प्रहारिणः।</div><div class="line" style="position:relative;">व्यधमत्सहसा भीमः क्रुद्धरूपः परन्तपः॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2569"><div class="verse-block verse-justify"><div class="line">दन्तवेष्टेषु नेत्रेषु कुम्भेषु च कटेषु च।</div><div class="line" style="position:relative;">मर्मस्वपि च मर्मज्ञस्तान्नागानवधीद्बली॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2570"><div class="verse-block verse-justify"><div class="line">`अर्दिता भीमसेनेन विनदन्तो भृशातुराः'।</div><div class="line">ततस्ते प्राद्रवन्भीताः प्रहताश्च पुनःपुनः।</div><div class="line" style="position:relative;">महामात्रास्तमावव्रुर्मेघा इव दिवाकरम्॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2571"><div class="verse-block verse-justify"><div class="line">तान्स सप्तशतान्नागान्सारोहायुधकेतनान्।</div><div class="line" style="position:relative;">भूमिष्ठो गदया जघ्ने वज्रेणेन्द्र इवाचलान्॥४७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2572"><div class="verse-block verse-justify"><div class="line">ततः सुबलपुत्रस्य नागानतिबलान्पुनः।</div><div class="line" style="position:relative;">पोथयामास कौन्तेयो द्विपञ्चाशदरिन्दमः॥४८॥</div></div></div>
