<div class="verse-block-wrapper" id="08-karṇaparva-b5393"><div class="verse-block verse-justify"><div class="line">शृणु राजन्नवहितो यथा वृत्तो महाक्षयः।</div><div class="line" style="position:relative;">घोरो मनुष्यदेहानामाजौ च गजवाजिनाम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b5396"><div class="verse-block verse-justify"><div class="line">वणिजो नावि भिन्नायामगाधे विप्लुवे यथा।</div><div class="line" style="position:relative;">अपारे पारमिच्छन्तो हते द्वीपे किरीटिना॥६॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b5398"><div class="verse-block verse-justify"><div class="line">भग्नशृङ्गा वृषा यद्वद्भग्नदंष्ट्रा इवोरगाः।</div><div class="line" style="position:relative;">प्रत्यपायाम सायाह्ने निर्जिताः सव्यसाचिना॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5399"><div class="verse-block verse-justify"><div class="line">हतप्रवीरा विध्वस्ता निकृत्ता निशितैः शरैः।</div><div class="line" style="position:relative;">सूतपुत्रे हते राजन्पुत्रास्ते दुद्रुवुर्भयात्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5400"><div class="verse-block verse-justify"><div class="line">विस्रस्तयन्त्रकवचाः कान्दिग्भूता विचेतसः।</div><div class="line" style="position:relative;">अन्योन्यमवमृद्गन्तो वीक्षमाणा भयार्दिताः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5401"><div class="verse-block verse-justify"><div class="line">मामेव नूनं बीभत्सुर्मामेव च वृकोदरः।</div><div class="line" style="position:relative;">अभियातीति मन्वानाः पेतुर्मम्लुश्च सम्भ्रमात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5402"><div class="verse-block verse-justify"><div class="line">हयानन्ये गजानन्ये रथानन्ये महारथाः।</div><div class="line" style="position:relative;">आरुह्य जवसम्पन्नाः पदातीन्प्रजहुर्भयात्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5403"><div class="verse-block verse-justify"><div class="line">कुञ्जरैः स्यन्दनाः क्षुण्णाः सादिनश्च महारथैः।</div><div class="line" style="position:relative;">पदातिसङ्घाश्चाश्वौधैः पलायद्भिर्भयार्दितैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5404"><div class="verse-block verse-justify"><div class="line">व्यालतस्करसङ्कीर्णे सार्थहीना यथा वने।</div><div class="line" style="position:relative;">सूतपुत्रे हते राजंस्तव योधास्तथाऽभवन्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5405"><div class="verse-block verse-justify"><div class="line">हतारोहा यथा नागाश्छिन्नहस्ता यथा नराः।</div><div class="line" style="position:relative;">सर्वे पार्थमयं लोकं सम्पश्यन्तो भयार्दिताः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5406"><div class="verse-block verse-justify"><div class="line">सम्प्रेक्ष्य द्रवतः सर्वान्भीमसेनभयार्दितान्।</div><div class="line" style="position:relative;">दुर्योधनोऽथ स्वं सूतं हाहाकृत्वेदमब्रवीत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5407"><div class="verse-block verse-justify"><div class="line">नातिक्रमेच्च मां पार्थो धनुष्पाणिमवस्थितम्।</div><div class="line" style="position:relative;">जघने सर्वसैन्यानां शनैरश्वान्प्रचोदय॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5408"><div class="verse-block verse-justify"><div class="line">युध्यमानं हि कौन्तेयं हनिष्यामि न संशयः।</div><div class="line" style="position:relative;">नोत्सहेन्मामतिक्रान्तुं वेलामिव महोदधिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5409"><div class="verse-block verse-justify"><div class="line">अद्यार्जुनं सगोविन्दं मानिनं च वृकोदरम्।</div><div class="line" style="position:relative;">अन्याञ्शिष्टांस्तथा शत्रून्कर्णस्यानृण्यमाप्नुयाम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5410"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा कुरुराजस्य शूरार्यसदृशं वचः।</div><div class="line" style="position:relative;">सूतो हेमपरिच्छन्नाञ्शनैरश्वानचोदयत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5411"><div class="verse-block verse-justify"><div class="line">रथाश्वनागहीनास्तु पादातास्तव मारिष।</div><div class="line" style="position:relative;">पञ्चविंशतिसाहस्रा युद्वायैव व्यवस्थिताः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5412"><div class="verse-block verse-justify"><div class="line">तान्भीमसेनः सङ्क्रुद्धो धृष्ट्यद्युम्नश्च पार्षतः।</div><div class="line" style="position:relative;">बलेन चतुरङ्गेण संवृत्याजघ्नतुः शरैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5413"><div class="verse-block verse-justify"><div class="line">प्रत्ययुध्यन्त समरे भीमसेनं सपार्षतम्।</div><div class="line" style="position:relative;">पार्थपार्षतयोश्चान्ये जगृहुस्तत्र नामनी॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5414"><div class="verse-block verse-justify"><div class="line">अक्रुध्यत रणे भीमस्तैस्तदा पर्यवस्थितैः।</div><div class="line" style="position:relative;">सोऽवतीर्य रथात्तूर्णं गदापाणिरयुध्यत॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5415"><div class="verse-block verse-justify"><div class="line">न तान्रथस्थो भूमिष्ठान्धर्मापेक्षी वृकोदरः।</div><div class="line" style="position:relative;">योधयामास कौन्तेयो भुजवीर्यव्यपाश्रयः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5416"><div class="verse-block verse-justify"><div class="line">जातरूपपरिच्छन्नां प्रगृह्य महतीं गदाम्।</div><div class="line" style="position:relative;">अवधीत्तावकान्सर्वान्दण्डपाणिरिवान्तकः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5417"><div class="verse-block verse-justify"><div class="line">पदातिनोऽपि सन्त्यक्त्वा प्रियं जीवितमात्मनः।</div><div class="line" style="position:relative;">भीममभ्यद्रवन्सङ्ख्ये पतङ्गा ज्वलनं यथा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5418"><div class="verse-block verse-justify"><div class="line">आसाद्य भीमसेनं तु संरब्धा युद्धदुर्मदाः।</div><div class="line" style="position:relative;">विनेशुः सहसा दृष्ट्वा भूतग्रामा इवान्तकम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5419"><div class="verse-block verse-justify"><div class="line">श्येनवद्विचरन्भीमो गदाहस्तो महाबलः।</div><div class="line" style="position:relative;">पञ्चविंशतिसाहस्रांस्तावकानवपोथयत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5420"><div class="verse-block verse-justify"><div class="line">हत्वा तत्पुरुषानीकं भीमः सत्यपराक्रमः।</div><div class="line" style="position:relative;">धृष्टद्युम्नं पुरस्कृत्य तस्थौ तत्र महाबलः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5421"><div class="verse-block verse-justify"><div class="line">धनञ्जयो रथानीकमभ्यवर्तत वीर्यवान्।</div><div class="line">माद्रीपुत्रौ तु शकुनिं सात्यकिश्च महारथः।</div><div class="line" style="position:relative;">जवेनाभ्यपतन्हृष्टा घ्नन्तो दौर्योधनं बलम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5422"><div class="verse-block verse-justify"><div class="line">तस्याश्वसादीन्सुबहूंस्ते निहत्य शितैः शरैः।</div><div class="line" style="position:relative;">समभ्यधावंस्त्वरितास्तत्र युद्धमभून्महत्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5423"><div class="verse-block verse-justify"><div class="line">धनञ्जयोऽपि चाभ्येत्य रथानीकं तव प्रभो।</div><div class="line" style="position:relative;">विश्रुतं त्रिषु लोकेषु गाण्डीवं विक्षिपन्धनुः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5424"><div class="verse-block verse-justify"><div class="line">कृष्णसारथिमायान्तं दृष्ट्वा श्वेतहयं रथम्।</div><div class="line" style="position:relative;">अर्जुनं चापि योद्वारं त्वदीयाः प्राद्रवन्भयात्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5425"><div class="verse-block verse-justify"><div class="line">विप्रहीणरथाश्चैव शरैश्च परिकर्शिताः।</div><div class="line" style="position:relative;">पञ्चविंशतिसाहस्राः कालमार्छन्पदातयः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5426"><div class="verse-block verse-justify"><div class="line">हत्वा तान्पुरुषव्याघ्रः पाञ्चालानां महारथः।</div><div class="line" style="position:relative;">पुत्राः पाञ्चालराजस्य धृष्टद्युम्नो महामनाः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5427"><div class="verse-block verse-justify"><div class="line">भीमसेनं पुरस्कृत्य नचिरात्प्रत्यदृश्यत।</div><div class="line" style="position:relative;">महाधनुर्धरः श्रीमानमित्रगणतापनः॥३७॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b5429"><div class="verse-block verse-justify"><div class="line">गान्धारराजं शीघ्रास्त्रमनुसृत्य यशस्विनौ।</div><div class="line" style="position:relative;">नचिरात्प्रत्यदृश्येतां माद्रीपुत्रौ ससात्यकी॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5430"><div class="verse-block verse-justify"><div class="line">वेकितानः शिखण्डी च द्रौपदेयाश्च मारिष।</div><div class="line" style="position:relative;">हत्वा त्वदीयं सुमहत्सैन्यं शङ्खांस्तथाऽधमन्॥४०॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b5433"><div class="verse-block verse-justify"><div class="line">धनञ्जयो रथानीकमभ्यवर्तत वीर्यवान्।</div><div class="line" style="position:relative;">विश्रुतं त्रिषु लोकेषु व्याक्षिपद्गाण्डिवं धनुः॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5434"><div class="verse-block verse-justify"><div class="line">तत एनाञ्शरव्रातैः सहसा समवाकिरत्।</div><div class="line" style="position:relative;">तमसा संवृतेनाथ न स्म किञ्चिद्वदृश्यत॥४४॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b5441"><div class="verse-block verse-justify"><div class="line">तत्राद्भुतमपश्याम तव पुत्रस्य पौरुषम्।</div><div class="line">यदेकः सहितान्सर्वान्रणेऽयुध्यत पाण्डवान्।</div><div class="line" style="position:relative;">ततोऽपश्यन्महात्मा स स्वसैन्यं भृशदुःखितम्॥५०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5442"><div class="verse-block verse-justify"><div class="line">ततोऽवस्थाप्य राजेन्द्र कृतबुद्धिस्तवात्मजः।</div><div class="line" style="position:relative;">हर्षयन्निव तान्योधानिदं वचनमब्रवीत्॥५१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5443"><div class="verse-block verse-justify"><div class="line">न तं देशं प्रपश्यामि यत्र याता भयार्दिताः।</div><div class="line" style="position:relative;">गतानां यत्र वै मोक्षः पाण्डवात्किं गतेन वः॥५२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5444"><div class="verse-block verse-justify"><div class="line">अल्पं च बलमेतेषां कृष्णौ च भृशविक्षतौ।</div><div class="line" style="position:relative;">अद्य सर्वान्हनिष्यामिध्रुवो हि विजयो भवेत्॥५३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5445"><div class="verse-block verse-justify"><div class="line">विप्रयातांस्तु वो भिन्नान्पाण्डवाः कृतकिल्विषान्।</div><div class="line" style="position:relative;">अनुसृत्य वधिष्यन्ति श्रेयान्नः समरे वधः॥५४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5446"><div class="verse-block verse-justify"><div class="line">सुखं साङ्ग्रामिको मृत्युः क्षत्रधर्मेण युध्यताम्।</div><div class="line" style="position:relative;">मृतो दुःखं न जानीते प्रेत्य चानन्त्यमश्नुते॥५५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5447"><div class="verse-block verse-justify"><div class="line">शृणुध्वं क्षत्रियाः सर्वे यावन्तः स्थ समागताः।</div><div class="line" style="position:relative;">यदा शूरं च भीरुं च मारयत्यन्तको मयः॥५६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5448"><div class="verse-block verse-justify"><div class="line">को न मूढो न युध्येत मादृशः क्षत्रियव्रतः।</div><div class="line" style="position:relative;">द्विषतो भीमसेनस्य क्रद्धस्य वशमेष्यथा॥५७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5449"><div class="verse-block verse-justify"><div class="line">पितामहैराचरितं न धर्मं हातुमर्हथ।</div><div class="line" style="position:relative;">न ह्यधर्मोऽस्ति पापीयान्क्षत्रियस्य पलायनात्॥५८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5450"><div class="verse-block verse-justify"><div class="line">न युद्धधर्माच्छ्रेयो हि पन्थाः स्वर्गस्य कौरवाः।</div><div class="line">अचिरेण हता लोकान्सद्यो योधाः समश्नुत ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥५९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5451"><div class="verse-block verse-justify"><div class="line">एवं ब्रुवति पुत्रे ते सैनिका भृशविक्षताः।</div><div class="line">अनवेक्ष्यैव तद्वाक्यं प्राद्रवन्सर्वतो दिशः ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥६०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b5452"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा तु सैन्यं परिवृत्यमानंपुत्रेण ते मद्रपतिस्तदानीम्।</div><div class="line" style="position:relative;">सन्त्रस्तरूपः परिमूढचेतादुर्योधनं वाक्यमिदं बभाषे]॥६१॥</div></div></div></div>
