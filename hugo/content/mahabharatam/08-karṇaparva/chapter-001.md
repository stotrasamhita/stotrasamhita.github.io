<div class="verse-block-wrapper" id="08-karṇaparva-b0009"><div class="verse-block verse-justify"><div class="line">गोष्ठमध्येव ऋषभो गोव्रजैः परिवारितः।</div><div class="line" style="position:relative;">व्यालेन निहतो यद्वत्तथाऽसौ निहतः परैः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0010"><div class="verse-block verse-justify"><div class="line">निराशान्पाण्डवान्कृत्वा दृढं राजन्ससात्यकान्।</div><div class="line" style="position:relative;">पाञ्चालानां रथांश्चैव विनिहत्य सहस्रशः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0011"><div class="verse-block verse-justify"><div class="line">जित्वा शूरान्महेष्वासान्विद्राव्य च दिशोदश।</div><div class="line" style="position:relative;">हतो वैकर्तनः कर्णः पाण्डवेन किरीटिना॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0012"><div class="verse-block verse-justify"><div class="line">वैरस्य गत आनृण्यं दुर्गमस्य दुरात्मभिः।</div><div class="line" style="position:relative;">हत्वा कर्णं महाराज विशल्यः पाण्डवोऽभवत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0013"><div class="verse-block verse-justify"><div class="line">शोषणं सागराणां वा पतनं वा विवस्वतः।</div><div class="line" style="position:relative;">विशीर्णत्वं यथा मेरोस्तथा कर्णस्य पातनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0014"><div class="verse-block verse-justify"><div class="line">योधाश्च बहवो राजन्हतास्तत्र जयैषिणः।</div><div class="line" style="position:relative;">राजानो राजपुत्राश्च शूराः परिघबाहवः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0015"><div class="verse-block verse-justify"><div class="line">रथौघाश्च नरौघाश्च हता राजन्सहस्रशः।</div><div class="line" style="position:relative;">वारणा निहतास्तत्र वाजिनश्च महाहवे॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0016"><div class="verse-block verse-justify"><div class="line">क्षत्रियाश्च महाराज सेनयोरुभयोर्हताः।</div><div class="line" style="position:relative;">परस्परं वै वीक्ष्यात्र परस्परकृतागसः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0017"><div class="verse-block verse-justify"><div class="line">किञ्चिच्छेषान्परान्कृत्वा तीर्त्वा पाण्डववाहिनीम्।</div><div class="line" style="position:relative;">पार्थवेलां समासाद्य हतो वैकर्तनो वृषा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0018"><div class="verse-block verse-justify"><div class="line">जयाशा धार्तराष्ट्राणां वैरस्य च मुखं नृप।</div><div class="line" style="position:relative;">तीर्णं तत्पाण्डवै राजन्यत्पुरा नावबुध्यसे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0019"><div class="verse-block verse-justify"><div class="line">प्रोच्यमानं महाराज बन्धुभिर्हितबुद्धिभिः।</div><div class="line" style="position:relative;">तदिदं समनुप्राप्तं व्यसनं त्वां महाभयम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0020"><div class="verse-block verse-justify"><div class="line">पुत्राणां राज्यकामेन त्वया राजन्हितैषिणा।</div><div class="line" style="position:relative;">चरितान्यहितान्येव तेषां ते फलमागतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0021"><div class="verse-block verse-justify"><div class="line">हतो दुःशासनो राजन्यथोक्तं पाण्डवेन तु।</div><div class="line" style="position:relative;">प्रतिज्ञा भीमसेनेन निस्तीर्णा सा चमूमुखे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0022"><div class="verse-block verse-justify"><div class="line">पीतं च क्षतजं तस्य धार्तराष्ट्रस्य संयुगे।</div><div class="line">पाण्डवेन महाराज कर्म कृत्वा सुदुष्करम्*' ॥वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0023"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा महाराज धृतराष्ट्रोऽम्बिकासुतः।</div><div class="line">शोकस्यान्तमपश्यन्वै हतं मत्वा सुयोधनम्।</div><div class="line" style="position:relative;">विह्वलः पतितो भूमौ नष्टचेता इव द्विपः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0024"><div class="verse-block verse-justify"><div class="line">तस्मिन्निपतिते भूमौ विह्वले राजसत्तमे।</div><div class="line" style="position:relative;">आर्तनादो महानासीत्स्त्रीणां भरतसत्तम॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0026"><div class="verse-block verse-justify"><div class="line">शोकार्णवे महाघोरे निमग्ना भरतस्त्रियः।</div><div class="line" style="position:relative;">रुरुदुर्दुःखशोकार्ता भृशमुद्विग्नचेतसः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0027"><div class="verse-block verse-justify"><div class="line">राजानं च समासाद्य गान्धारी भरतर्षभ।</div><div class="line" style="position:relative;">निःसंज्ञा पतिता भूमौ सर्वाण्यन्तः पुराणि च॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0028"><div class="verse-block verse-justify"><div class="line">ततस्ताः स़ञ्जयो राजन्समाश्वासयदातुराः।</div><div class="line" style="position:relative;">मुह्यामानाः सुबहुशो मुञ्चन्तीर्वारि नेत्रजम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0029"><div class="verse-block verse-justify"><div class="line">समाश्वस्ताः स्त्रियस्तास्तु वेपमाना मुहुर्मुहुः।</div><div class="line" style="position:relative;">कदल्य इव वातेन धूयमानाः समन्ततः॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0031"><div class="verse-block verse-justify"><div class="line">स लब्ध्वा सनकैः संज्ञां ताश्च दृष्ट्वा स्त्रियो नृपः।</div><div class="line" style="position:relative;">उन्मत्त इव राजेन्द्र स्थितस्तूष्णीं विशाम्पते॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0032"><div class="verse-block verse-justify"><div class="line">ततो ध्यात्वा चिरं कालं निःश्वस्य च पुनःपुनः।</div><div class="line" style="position:relative;">स्वान्पुत्रान्गर्हयामास बहुमेने च पाण्डवान्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0033"><div class="verse-block verse-justify"><div class="line">गर्हयंश्चात्मनो बुद्धिं शकुनेः सौबलस्य च।</div><div class="line" style="position:relative;">ध्यात्वा तु सुचिरं कालं वेपमानो मुहुर्मुहुः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0034"><div class="verse-block verse-justify"><div class="line">संस्तभ्य च मनो भूयो राजा धैर्यसमन्वितः।</div><div class="line" style="position:relative;">पुनर्गावल्गणिं सूतं पर्यपृच्छत सञ्जयम्॥३२॥</div></div></div>
