<div class="verse-block-wrapper" id="08-karṇaparva-b3590"><div class="verse-block verse-justify"><div class="line">द्रौणिस्तु रथवंशेन महता परिवारितः।</div><div class="line" style="position:relative;">अपतत्सहसा राजन्यत्र पार्थो व्यवस्थितः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3591"><div class="verse-block verse-justify"><div class="line">तमापतन्तं सहसा शूरः शौरिसहायवान्।</div><div class="line" style="position:relative;">दधार समरे पार्थो वेलेव मकरालयम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3592"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धो महाराज द्रोणपुत्रः प्रतापवान्।</div><div class="line" style="position:relative;">अर्जुनं वासुदेवं च च्छादयामास सायकैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3593"><div class="verse-block verse-justify"><div class="line">अवच्छन्नौ ततः कृष्णौ दृष्ट्वा तत्र महारथाः।</div><div class="line" style="position:relative;">विस्मयं परमं गत्वा प्रैक्षन्त कुरवस्तदा॥५॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3595"><div class="verse-block verse-justify"><div class="line">यद्यद्धि व्याक्षिपद्युद्धे फल्गुनोऽस्त्रं जिघांसया।</div><div class="line" style="position:relative;">तत्तदस्त्रैर्महेष्वासो द्रोणपुत्रो जघान ह॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3596"><div class="verse-block verse-justify"><div class="line">अस्त्रयुद्धे महाराज वर्तमाने महाभये।</div><div class="line" style="position:relative;">अपश्याम रणे द्रौणिं व्यात्ताननमिवान्तकम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3597"><div class="verse-block verse-justify"><div class="line">स दिशः प्रदिशश्चैव च्छादयित्वा ह्यजिह्मगैः।</div><div class="line" style="position:relative;">वासुदेवं त्रिभिर्बाणैरविध्यद्दक्षिणे भुजे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3598"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनो रथान्सर्वान्हत्वा तस्य पदानुगान्।</div><div class="line">चकार समरे भूमिं शोणितौघपरिप्लुताम्।</div><div class="line" style="position:relative;">[सर्बलोकवहां रौद्रां परलोकवहां नदीम्]॥१०॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3601"><div class="verse-block verse-justify"><div class="line">तयोस्तु व्याकुले युद्धे द्रौणेः पार्थस्य दारुणे।</div><div class="line" style="position:relative;">अमर्यादं योधयन्तः पर्यधावन्त पृष्ठतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3602"><div class="verse-block verse-justify"><div class="line">रथैर्हताश्वसूतैश्च हतारोहैश्च वाजिभिः।</div><div class="line" style="position:relative;">द्विरदैश्च हतारोहैर्महामात्रैर्हतद्विपैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3603"><div class="verse-block verse-justify"><div class="line">पार्थेन समरे राजन्कृतो घोरो जनक्षयः।</div><div class="line" style="position:relative;">निहता रथिनः पेतुः पार्थचापच्युतैः शरैः ॥]॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3604"><div class="verse-block verse-justify"><div class="line">हयाश्च पर्यधावन्त मुक्तयोक्त्रास्ततस्ततः।</div><div class="line" style="position:relative;">तद्दृष्ट्वा कर्म पार्थस्य द्रौणिराहवशोभिनः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3605"><div class="verse-block verse-justify"><div class="line">अर्जुनं जयतां श्रेष्ठं त्वरितोऽभ्येत्य वीर्यवान्।</div><div class="line" style="position:relative;">विधुन्वानो महच्चापं कार्तस्वरविभूषितम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3606"><div class="verse-block verse-justify"><div class="line">अवाकिरत्ततो द्रौमिः समन्तान्निशितैः शरैः।</div><div class="line" style="position:relative;">भूयोऽर्जुनं महाराज द्रौणिरायम्य पत्रिणा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3607"><div class="verse-block verse-justify"><div class="line">वक्षोदेशे भृशं पार्थं ताडयामास निर्दयम्।</div><div class="line" style="position:relative;">सोऽतिविद्धो रणे तेन द्रोणपुत्रेण भारत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3608"><div class="verse-block verse-justify"><div class="line">गाण्डीवधन्वा प्रसभं शरवर्षैरुदारधीः।</div><div class="line" style="position:relative;">सञ्छाद्य समरे द्रौणिं चिच्छेदास्य च कार्मुकम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3609"><div class="verse-block verse-justify"><div class="line">स च्छिन्नधन्वा परिधं वज्रस्पर्शसमं युधि।</div><div class="line" style="position:relative;">आदाय चिक्षेप तदा द्रोणपुत्रः किरीटिने॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3610"><div class="verse-block verse-justify"><div class="line">तमापन्ततं परिघं जाम्बूनदपरिष्कृतम्।</div><div class="line" style="position:relative;">चिच्छेद सहसा राजन्प्रहसन्निव पाण्डवः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3611"><div class="verse-block verse-justify"><div class="line">स पपात तदा भूमौ निकृत्तः पार्थसायकैः।</div><div class="line" style="position:relative;">विकीर्णः पर्वतो राजन्यथा वज्रेण ताडितः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3612"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धो महाराज द्रोणपुत्रो महारथः।</div><div class="line" style="position:relative;">ऐन्द्रेण चास्त्रवेगेन बीभत्सुं समवाकिरत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3613"><div class="verse-block verse-justify"><div class="line">तस्येन्द्रजालावततं समीक्ष्यपार्थो राजन्गाण्डिवमाददे सः।</div><div class="line" style="position:relative;">ऐन्द्रं जालं प्रत्यहरत्तरस्वीवरास्त्रमादाय महेन्द्रसृष्टम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3614"><div class="verse-block verse-justify"><div class="line">विदार्य तज्जालमथेन्द्रमुक्तंपार्थस्ततो द्रौणिरथं क्षणेन।</div><div class="line" style="position:relative;">प्रच्छादयामास ततोऽभ्युपेत्यद्रौणिस्तदा पार्थशराभिभूतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3615"><div class="verse-block verse-justify"><div class="line">विगाह्य तां पाण्डवबाणवृष्टिंशरैः परं नाम ततः प्रकाश्य।</div><div class="line" style="position:relative;">शतेन कृष्णं सहसाभ्यविद्ध्या--त्त्रिभिः शतैरर्जुनं क्षुद्रकाणाम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3616"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनः सायकानां शतेनगुरोः सुतं मर्मसु निर्बिभेद।</div><div class="line" style="position:relative;">अश्वांश्च सूतं च तथा धनुर्ज्या--मवाकिरत्पश्यतां तावकानाम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3617"><div class="verse-block verse-justify"><div class="line">स विद्ध्वा मर्मसु द्रौणिं पाण्डवः परवीरहा।</div><div class="line" style="position:relative;">सारथिं चास्य भल्लेन रथनीडादपातयत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3618"><div class="verse-block verse-justify"><div class="line">स सङ्गृह्य स्वयं वाहान्कृष्णौ प्राच्छदयच्छरैः।</div><div class="line" style="position:relative;">तत्राद्भुतमपरश्याम द्रौणेराशु पराक्रमम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3619"><div class="verse-block verse-justify"><div class="line">संयच्छंस्तुरगान्यच्च फल्गुनं चाप्ययोधयत्।</div><div class="line" style="position:relative;">तदस्य समरे राजन्सर्वभूतान्यपूजयन्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3620"><div class="verse-block verse-justify"><div class="line">`तेनास्य समरे योधास्तुष्टुवुः सर्व एव ते।</div><div class="line" style="position:relative;">यदा निजगृहे पार्थो द्रोणपुत्रेण धन्विना॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3621"><div class="verse-block verse-justify"><div class="line">तदा प्रहस्य बीभस्तुर्द्रोणपुत्रस्य संयुगे।</div><div class="line" style="position:relative;">क्षिप्रं रश्मीनथाश्वानां क्षुरप्रैश्चिच्छिदे जयः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3622"><div class="verse-block verse-justify"><div class="line">प्राद्रवंस्तुरगास्तस्य शरवेगप्रपीडिताः।</div><div class="line" style="position:relative;">ततोऽभून्निनदो घोरस्तव सैन्यस्य भारत॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3623"><div class="verse-block verse-justify"><div class="line">पाण्डवास्तु जयं लब्ध्वा तव सैन्यं समाद्रवन्।</div><div class="line" style="position:relative;">समन्तान्निशितान्बाणान्विमुञ्चन्तो जयैषिणः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3624"><div class="verse-block verse-justify"><div class="line">तावकी चतुरङ्गैषा पाण्डवैस्तु महाचमूः।</div><div class="line" style="position:relative;">पुनःपुनरथो वीरैः प्रभग्ना जितकाशिभिः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3625"><div class="verse-block verse-justify"><div class="line">पश्यतां ते महाराज पुत्राणां चित्रयोधिनाम्।</div><div class="line" style="position:relative;">शकुनेः सौबलेयस्य कर्णस्य च विशाम्पते॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3626"><div class="verse-block verse-justify"><div class="line">वार्यमाणा महासेना पुत्रैस्तव जनेश्वर।</div><div class="line" style="position:relative;">न चातिष्ठत सङ्ग्रामे पीड्यमाना समन्ततः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3627"><div class="verse-block verse-justify"><div class="line">ततो योधैर्महाराज पलायद्भिः समन्ततः।</div><div class="line" style="position:relative;">अभवद्य्वाकुलं भीतं पुत्राणां ते महद्बलम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3628"><div class="verse-block verse-justify"><div class="line">तिष्ठतिष्ठेति सततं सूतपुत्रस्य जल्पतः।</div><div class="line" style="position:relative;">नावतिष्ठति सा सेना वध्यमाना महात्मभिः॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3629"><div class="verse-block verse-justify"><div class="line">अथोत्क्रुष्टं महाराज पाण्डवैर्जितकाशिभिः।</div><div class="line" style="position:relative;">धार्तराष्ट्रबलं दृष्ट्वा विद्रुतं वै समन्ततः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3630"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनः कर्णमब्रवीत्प्रणयादिव।</div><div class="line" style="position:relative;">पश्य कर्ण महासेनां पाण्डवैरर्दितां भृशम्॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3631"><div class="verse-block verse-justify"><div class="line">त्वयि तिष्ठति सन्त्रासात्समन्तात्प्रपलायते।</div><div class="line" style="position:relative;">एतज्ज्ञत्वा महाबाहो कुरु प्राप्तमरिन्दम॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3632"><div class="verse-block verse-justify"><div class="line">सहस्राणि च योधानां त्वामेव पुरुषोत्तम।</div><div class="line">क्रोशन्ते समरे वीर द्राव्यमाणानि पाण्डवैः ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3633"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा तु राधेयो दुर्योधनवचो महत्।</div><div class="line" style="position:relative;">मद्रराजमिदं वाक्यमब्रवीत्प्रहसन्निव॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3634"><div class="verse-block verse-justify"><div class="line">पश्य मे भुजयोर्वीर्यमस्त्राणां च जनेश्वर।</div><div class="line" style="position:relative;">अद्य हन्मि रणे सर्वान्पाञ्चालान्पाण्डुभिः सह॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3635"><div class="verse-block verse-justify"><div class="line">वाहयाश्वान्नरव्याघ्र मद्राधिप न संशयः।</div><div class="line" style="position:relative;">एवमुक्त्वा महाराज सूतपुत्रः प्रतापवान्॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3636"><div class="verse-block verse-justify"><div class="line">प्रगृह्य विजयं वीरो धनुःश्रेष्ठं पुरातनम्।</div><div class="line" style="position:relative;">सज्यं कृत्वा महाराज सम्मृज्य च पुनः पुनः॥४७॥</div></div></div>
