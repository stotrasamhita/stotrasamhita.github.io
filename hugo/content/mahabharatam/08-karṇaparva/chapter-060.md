<div class="verse-block-wrapper" id="08-karṇaparva-b3127"><div class="verse-block verse-justify"><div class="line">स शक्तिप्रासतूणीरानश्वारोहान्हयानपि।</div><div class="line" style="position:relative;">पुलिन्दखसबाह्लीकनिषादान्ध्रककुन्तलान्॥११॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3130"><div class="verse-block verse-justify"><div class="line">आभाष्य चैनं मधुरमभीतं तमभीतवत्।</div><div class="line" style="position:relative;">प्राह प्रहरतां श्रेष्ठः स्मितपूर्वं समाह्वयत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3131"><div class="verse-block verse-justify"><div class="line">राजन्कमलपत्राक्ष प्रधानायुधवाहन।</div><div class="line" style="position:relative;">वज्रसंहननप्रख्य प्रख्यातबलपौरुष॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3132"><div class="verse-block verse-justify"><div class="line">मुष्टिक्लिष्टाङ्गुलिभ्यां च व्यायताभ्यां महद्धनुः।</div><div class="line" style="position:relative;">दोर्भ्यां विस्फारयन्भासि महाजलदवद्भृशम्॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3135"><div class="verse-block verse-justify"><div class="line">महता रथघोषेण दिवं भूमिं च नादयन्।</div><div class="line" style="position:relative;">वर्षान्ते सस्यां सूर्यो भाभिरादीपयन्निव॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3136"><div class="verse-block verse-justify"><div class="line">संस्पृशानः शरैः पूर्णौ तूणी चाशीविषोपमैः।</div><div class="line" style="position:relative;">मयैवैकेन युध्यस्व त्र्यम्बकेनान्धको यथा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3137"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तथेत्युक्त्वा प्रमथ्यैनं स पार्थिवः।</div><div class="line" style="position:relative;">कर्णिना द्रोणतनयं विव्याध मलयध्वजः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3138"><div class="verse-block verse-justify"><div class="line">मर्मभेदिभिरत्युग्रैर्बाणैरग्निशिखोपमैः।</div><div class="line" style="position:relative;">मर्मस्वभ्यहनद्द्रौणिः पाण्ड्यमाचार्यनन्दनः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3139"><div class="verse-block verse-justify"><div class="line">ततोऽपरान्नवांस्तूर्णं नाराचान्कङ्कवाससः।</div><div class="line" style="position:relative;">गत्या दशम्या संयुक्तानश्वत्थामाऽप्यवासृजत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3140"><div class="verse-block verse-justify"><div class="line">तानच्छिनत्तदा पाण्ड्यश्चतुर्भिरपरैः शरैः।</div><div class="line" style="position:relative;">चतुरोऽभ्याहनच्चाश्वानाशु ते व्यसवोऽभवन्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3141"><div class="verse-block verse-justify"><div class="line">अथ द्रोणसुतस्येषूंस्ताञ्छित्त्वा निशितैः शरैः।</div><div class="line" style="position:relative;">धनुर्ज्यां विततां पाण्ड्यश्चिच्छेदादित्यतेजसः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3142"><div class="verse-block verse-justify"><div class="line">दिव्यं धनुरथाधिज्यं कृत्वा द्रौणिरमित्रहा।</div><div class="line" style="position:relative;">प्रेक्ष्य चाशु रथे युक्तान्नरैरन्यान्हयोत्तमान्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3143"><div class="verse-block verse-justify"><div class="line">ततः शरसहस्राणि प्रेषयामास वै द्विजः।</div><div class="line" style="position:relative;">इषुसम्बाधमाकाशमकरोद्दिश एव च॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3144"><div class="verse-block verse-justify"><div class="line">ततस्तानस्यतः सर्वान्द्रौणेर्बाणान्महात्मनः।</div><div class="line" style="position:relative;">जानानोप्यक्षयान्पाण्ड्यो शातयत्पुरुषर्षभः॥२८॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3146"><div class="verse-block verse-justify"><div class="line">अथ तल्लाघवं दृष्ट्वा मण्डलीकृतकार्मुकः।</div><div class="line" style="position:relative;">प्रास्यद्द्रोणसुतो बाणान्वृष्टिं पूषानुजो यथा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3147"><div class="verse-block verse-justify"><div class="line">अष्टावष्टगवान्यूहुः शकटानि यदायुधम्।</div><div class="line" style="position:relative;">अह्नस्तदष्टभागेन द्रौणिश्चिक्षेप मारिष॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3148"><div class="verse-block verse-justify"><div class="line">तमन्तकमिव क्रुद्धमन्तकालान्तकोपमम्।</div><div class="line" style="position:relative;">ये ये ददृशिरे रूपं विसंज्ञाः प्रायशोऽभवन्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3149"><div class="verse-block verse-justify"><div class="line">आचार्यपुत्रस्तां सेनां बाणवृष्ट्या व्यवीवृषत्।</div><div class="line" style="position:relative;">पर्जन्य इव घर्मान्ते वृष्ट्या साद्रिद्रुमां महिम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3150"><div class="verse-block verse-justify"><div class="line">द्रौणिपर्जन्यमुक्तां तां बाणवृष्टिं सुदुःसहाम्।</div><div class="line" style="position:relative;">वायव्यास्त्रेण स क्षिप्रं विद्ध्वा पाण्ड्यातिलोऽनदत्॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3151"><div class="verse-block verse-justify"><div class="line">तस्य नानदतः केतुं चन्दनागुरुरूषितम्।</div><div class="line" style="position:relative;">मलयप्रतिमं द्रौणिश्छित्त्वाश्वांश्चतुरोऽहनत्॥३५॥</div></div></div>
