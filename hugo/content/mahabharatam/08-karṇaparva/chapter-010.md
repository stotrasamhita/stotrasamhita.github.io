<div class="verse-block-wrapper" id="08-karṇaparva-b0441"><div class="verse-block verse-justify"><div class="line">दैवमेव परं मन्ये धिक्पौरुषमनर्थकम्।</div><div class="line" style="position:relative;">यत्र सालप्रतीकाशः कर्णोऽहन्यत संयुगे॥४॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0443"><div class="verse-block verse-justify"><div class="line">मोहियित्वा रणे पार्थान्वज्नहस्त इवासुरान्।</div><div class="line" style="position:relative;">स कथं निहतः शेते वायुरुग्ण इव द्रुमः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0444"><div class="verse-block verse-justify"><div class="line">शोकस्यान्तं न पश्यामि पारं जलनिधेरिव।</div><div class="line" style="position:relative;">चिन्ता मे वर्धतेऽतीव मुमूर्षा चापि जायते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0445"><div class="verse-block verse-justify"><div class="line">कर्णस्य निधनं श्रुत्वा विजयं फल्गुनस्य च।</div><div class="line" style="position:relative;">अश्रद्धेयमहं मन्ये वधं कर्णस्य सञ्जय॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0446"><div class="verse-block verse-justify"><div class="line">वज्रसारमिदं नूनं हृदयं दुर्भिदं मम।</div><div class="line" style="position:relative;">यच्छ्रुत्वा पुरुषव्याघ्रं हतं कर्णं न दीर्यते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0447"><div class="verse-block verse-justify"><div class="line">आर्युर्नूनं सुदीर्घं मे विहितं दैवतैः पुरा।</div><div class="line" style="position:relative;">यत्र कर्णं हतं श्रुत्वा जीवामीह सुदुःखितः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0448"><div class="verse-block verse-justify"><div class="line">धिग्जीवितमिदं चैव सुहृद्वीनस्य सञ्जय।</div><div class="line">अद्य चाहं दशामेतां गतः सञ्जय गर्हिताम्।</div><div class="line" style="position:relative;">कृपणं वर्तयिष्यामि शोच्यः सर्वस्य मन्दधीः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0449"><div class="verse-block verse-justify"><div class="line">अहमेव पुरा भूत्वा सर्वलोकस्य सत्कृतः।</div><div class="line" style="position:relative;">परिभूतः कथं सूत परैः शक्ष्यामि जीवितुम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0450"><div class="verse-block verse-justify"><div class="line">दुःखाद्दुःखतरं भन्ये प्राप्तवानस्मि सञ्जय।</div><div class="line" style="position:relative;">भीष्मद्रोणवधेनैव कर्णस्य च महात्मनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0451"><div class="verse-block verse-justify"><div class="line">नावशेषं प्रपश्यामि सूतपुत्रे हते युधि।</div><div class="line" style="position:relative;">सहि पारं महानासीत्पुत्राणां मम सञ्जय॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0452"><div class="verse-block verse-justify"><div class="line">युद्धे हि निहतः शूरो विसृजन्सायकान्बहून्।</div><div class="line" style="position:relative;">को हि मे जीवितेनार्थस्तमृते पुरुषर्षभम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0453"><div class="verse-block verse-justify"><div class="line">रथादाधिरथिर्नूनं न्यपतत्सायकार्दितः।</div><div class="line" style="position:relative;">पर्वतस्येव शिखरं व ज्रपाताद्विदारितम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0454"><div class="verse-block verse-justify"><div class="line">स शेते पृथिवीं नूनं शोभयन्रुधिरोक्षितः।</div><div class="line" style="position:relative;">मातङ्ग इव मत्तेन मताङ्गेन निपातितः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0455"><div class="verse-block verse-justify"><div class="line">यो बलं धार्तराष्ट्राणां पाण्डवानां यतो भयम्।</div><div class="line" style="position:relative;">सोऽर्जुनेन हतः कर्णः प्रतिमानं धनुष्मताम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0456"><div class="verse-block verse-justify"><div class="line">स हि वीरो महेष्वासो मित्राणामभयङ्करः।</div><div class="line" style="position:relative;">शेते विनिहतो वीरः शक्रेणेव पुरा बलः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0457"><div class="verse-block verse-justify"><div class="line">पङ्गोरिवाध्वगमनं दरिद्रस्येव कामितम्।</div><div class="line" style="position:relative;">दुर्योधनस्य लोभश्च समान्येतानि सञ्जय॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0458"><div class="verse-block verse-justify"><div class="line">अन्यथा चिन्तितं कार्यमन्यथा तत्तु जायते।</div><div class="line" style="position:relative;">अहो नु बलवद्दैवं कालश्च दुरतिक्रमः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0459"><div class="verse-block verse-justify"><div class="line">पलायमानः कृपणो दीनात्मा दीनपौरुषः।</div><div class="line" style="position:relative;">कच्चिद्विनिहतः सूत पुत्रो दुःशासनो मम॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0460"><div class="verse-block verse-justify"><div class="line">कच्चिन्न दीनाचरितं कृतवांस्तात संयुगे।</div><div class="line" style="position:relative;">कच्चिन्न निहतः शूरा यथाऽन्ये क्षत्रियर्षभाः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0461"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरस्य वचनं मा युद्धमिति सर्वदा।</div><div class="line" style="position:relative;">दुर्योधनो नाभ्यगृह्णान्मूढः पथ्यमिवौषधम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0462"><div class="verse-block verse-justify"><div class="line">शतल्पे शयानेन भीष्मेण सुमहात्मना।</div><div class="line" style="position:relative;">पानीयं याचितः पार्थः सोविध्यन्मेदिनीतलम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0463"><div class="verse-block verse-justify"><div class="line">जलस्य धारां जनितां दृष्ट्वा पाण्डुसुतेन च।</div><div class="line" style="position:relative;">अब्रवीत्स महाबाहुस्तात संशाम्य पाण्डवैः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0464"><div class="verse-block verse-justify"><div class="line">प्रशमाद्वि भवेच्छान्तिर्मदन्तं युद्धमस्तु वः।</div><div class="line" style="position:relative;">भ्रातृभावेन पृथिवीं भुङ्क्ष्व पाण्डुसुतैः सह॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0465"><div class="verse-block verse-justify"><div class="line">अकुर्वन्वचनं तस्य नूनं शोचति पुत्रकः।</div><div class="line" style="position:relative;">तदिदं समनुप्राप्त वचनं दीर्घदर्शिनः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0466"><div class="verse-block verse-justify"><div class="line">अहं तु निहतामात्यो हतपुत्रश्च सञ्जय।</div><div class="line" style="position:relative;">द्यूततः कृच्छ्रमापन्नो लूनपक्ष इव द्विजः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0467"><div class="verse-block verse-justify"><div class="line">यथा हि शकुनिं गृह्य छित्त्वा पक्षौ च सञ्जय।</div><div class="line" style="position:relative;">विसर्जयन्ति संहृष्टाः क्रीडमानाः कुमारकाः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0468"><div class="verse-block verse-justify"><div class="line">लूनपक्षतया तस्य गमनं नोपपद्यते।</div><div class="line" style="position:relative;">तथाहमपि सम्प्राप्तो लूनपक्ष इव द्विजः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0469"><div class="verse-block verse-justify"><div class="line">क्षीणः सर्वार्थहीनश्च निर्ज्ञातिर्बन्धुवर्जितः।</div><div class="line">कां दिशं प्रतिपत्स्यामि दीनः शत्रुवशं गतः ॥वैशम्पायन उवाच।</div><div class="line" style="position:relative;">॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0470"><div class="verse-block verse-justify"><div class="line">इत्येवं धृतराष्ट्रोऽथ विलप्य बहुदुःखितः।</div><div class="line">प्रोवाच सञ्जयं भूयः सोकव्याकुलमानसः ॥धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0471"><div class="verse-block verse-justify"><div class="line">योऽजयत्सर्वकाम्भोजानम्बष्ठान्कैकयैः सह।</div><div class="line" style="position:relative;">गान्धारांश्च विदेहांश्च जित्वा कार्यार्थमाहवे॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0472"><div class="verse-block verse-justify"><div class="line">दुर्योधनस्य वृद्ध्यर्थं योऽजयत्पृथिवीं प्रभुः।</div><div class="line" style="position:relative;">स जितः पाण्डवैः शूरैः समरे वाहुशालिभिः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0473"><div class="verse-block verse-justify"><div class="line">तस्मिन्हते महेष्वासे कर्णे युधि किरीटिना।</div><div class="line" style="position:relative;">के वीराः पर्यतिष्ठन्त तन्ममाचक्ष्व सञ्जय॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0474"><div class="verse-block verse-justify"><div class="line">कच्चिन्नैकः परित्यक्तः पाण्डवैर्निहतो रणे।</div><div class="line" style="position:relative;">उक्तं त्वया पुरा तात यथा वीरो निपातितः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0475"><div class="verse-block verse-justify"><div class="line">भीष्ममप्रतियुध्यन्तं शिखम्डी सायकोत्तमैः।</div><div class="line" style="position:relative;">पातयामास समरे सर्वशस्त्रभृतां वरम्॥३८॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0477"><div class="verse-block verse-justify"><div class="line">युक्तयोगो महेष्वासः शरैर्बहुभिराचितः।</div><div class="line" style="position:relative;">निहतः खङ्गमुद्यम्य धृष्टद्युम्नेन सञ्जय॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0478"><div class="verse-block verse-justify"><div class="line">अन्तरेण हतावेतौ छलेन च विशेषतः।</div><div class="line" style="position:relative;">अश्रौषमहमेतद्वै भीष्मद्रोणौ निपातितौ॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0479"><div class="verse-block verse-justify"><div class="line">भीष्मद्रोणौ हि समरे न हन्याद्वज्रभृत्स्वयम्।</div><div class="line" style="position:relative;">न्यायेन युध्यन्समरे तद्वै सत्यं ब्रवीमि ते॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0480"><div class="verse-block verse-justify"><div class="line">कर्णं त्वस्यन्तमस्त्राणि दिव्यानि च बहूनि च।</div><div class="line" style="position:relative;">कथमिन्द्रोपमं वीरं मृत्युर्युद्वे समस्पृशत्॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0481"><div class="verse-block verse-justify"><div class="line">यस्य विद्युत्प्रभां शक्तिं दिव्यां कनकभूषणाम्।</div><div class="line" style="position:relative;">प्रायच्छद्द्विषतां हन्त्रीं कुण्डलाभ्यां पुरन्दरः॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0482"><div class="verse-block verse-justify"><div class="line">यस्य सर्पमुखो दिव्यः शरः काञ्चनभूषणः।</div><div class="line" style="position:relative;">अशेत निशितः पत्री समरेष्वरिसूदनः॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0483"><div class="verse-block verse-justify"><div class="line">भीष्मद्रोणमुखान्वीरान्योऽवमत्य महारथान्।</div><div class="line" style="position:relative;">जामदग्न्यान्महाघोरं ब्राह्ममस्त्रमशिक्षत॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0484"><div class="verse-block verse-justify"><div class="line">यश्च द्रोणमुखान्दृष्ट्वा विमुखानर्दिताञ्शरैः।</div><div class="line" style="position:relative;">सौभद्रस्य महाबाहुर्व्यधमत्कार्मुकं शितैः॥४७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0485"><div class="verse-block verse-justify"><div class="line">यश्च नागायुतप्राणां वज्ररंहसमच्युतम्।</div><div class="line" style="position:relative;">विरथं सहसा कृत्वा भीमसेनमपाहसत्॥४८॥</div></div></div>
