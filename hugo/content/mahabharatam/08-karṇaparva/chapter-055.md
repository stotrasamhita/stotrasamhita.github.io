<div class="verse-block-wrapper" id="08-karṇaparva-b2801"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2802"><div class="verse-block verse-justify"><div class="line">भीमसेनं सपाञ्चालं चेदिकेकयसंवृतम्।</div><div class="line" style="position:relative;">सेनापतिः स्वयं क्रुद्धो वारयामास सायकैः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2803"><div class="verse-block verse-justify"><div class="line">ततस्तु चेदिकारूशान्सृढ्जयांश्च महारथान्।</div><div class="line" style="position:relative;">कर्णो जघान समरे भीमसेनस्य पश्यतः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2804"><div class="verse-block verse-justify"><div class="line">भीमसेनस्ततः कर्णे विहाय रथसत्तमम्।</div><div class="line" style="position:relative;">प्रययौ कौरवं सैन्यं कक्षमग्निरिव ज्वलन्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2805"><div class="verse-block verse-justify"><div class="line">सूतपुत्रोऽपि समरे पाञ्चालान्केकयांस्तथा।</div><div class="line" style="position:relative;">सृञ्जयांश्च महेष्वासान्निजघानं सहस्रशः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2806"><div class="verse-block verse-justify"><div class="line">संशप्तकेषु पार्थश्च कौरवेषु वृकोदरः।</div><div class="line" style="position:relative;">पाञ्चालेषु तथा कर्णः क्षयं चक्रुर्महारथाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2807"><div class="verse-block verse-justify"><div class="line">ते क्षत्रिया दह्यामानास्त्रिभिस्तैः पावकोपमैः।</div><div class="line" style="position:relative;">जग्मुर्विनाशं समरे राजन्दुर्मन्त्रिते तव॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2808"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनः क्रुद्दो नकुलं नवभिः शरैः।</div><div class="line" style="position:relative;">विव्याध भरतश्रेष्ठ चतुरश्चास्य वाजिनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2809"><div class="verse-block verse-justify"><div class="line">ततः परममेयात्मा तव पुत्रो जनाधिप।</div><div class="line" style="position:relative;">क्षुरेण सहदेवस्य ध्वजं चिच्छेद काञ्चनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2810"><div class="verse-block verse-justify"><div class="line">नकुलस्तु ततः क्रुद्धस्तव पुत्रं च सप्तभिः।</div><div class="line" style="position:relative;">जघान समरे राजन्सहदेवश्च पञ्चभिः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2811"><div class="verse-block verse-justify"><div class="line">तावुभौ भरतश्रेष्ठौ ज्येष्ठौ सर्वधनुष्मताम्।</div><div class="line" style="position:relative;">विव्याधोरसि सङ्क्रुद्धः पञ्चभिः पञ्चभिः शरैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2812"><div class="verse-block verse-justify"><div class="line">ततोऽपराभ्यां भल्लाब्यां धनुषी समकृन्तत।</div><div class="line" style="position:relative;">यमयोः प्रसभं वीरो विव्याधाशु त्रिभिस्त्रिभिः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2813"><div class="verse-block verse-justify"><div class="line">तावन्ये धनुषी श्रेष्ठे शक्रचापनिभे शुभे।</div><div class="line" style="position:relative;">प्रगृह्य रेजतुः शूरौ देवपुत्रसमौ युधि॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2814"><div class="verse-block verse-justify"><div class="line">ततस्तौ रभसौ युद्धे भ्रातरौ भ्रातरं युधि।</div><div class="line" style="position:relative;">शरैर्ववर्षतुर्घोरैर्महामेघौ यथाऽचलम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2815"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धो महाराज तव पुत्रो महारथः।</div><div class="line" style="position:relative;">पाण़्डुपुत्रौ महेष्वासौ वारयामास पत्रिभिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2816"><div class="verse-block verse-justify"><div class="line">धनुर्मण्डलमेवास्य ददृशे युधि भारत।</div><div class="line" style="position:relative;">सायकाश्चाप्यदृश्यन्त निश्चरन्तः समन्ततः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2817"><div class="verse-block verse-justify"><div class="line">`तस्य सायकसञ्छन्नौ माद्रेयौ न विरेजतुः।</div><div class="line" style="position:relative;">मेघच्छन्नौ यथा व्योम्नि चन्द्रसूर्यौ गतप्रभौ॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2818"><div class="verse-block verse-justify"><div class="line">ते तु बाणा महाराज स्वर्णपुङ्खाः शिलाशिताः'।</div><div class="line" style="position:relative;">आच्छादयन्दिशः सर्वाः सूर्यस्येव मरीचयः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2819"><div class="verse-block verse-justify"><div class="line">वाणभूते ततस्तस्मिन्सञ्छन्ने च नभस्तले।</div><div class="line" style="position:relative;">राज्ञस्तु ददृशे रूपं कालान्तकयमोपमम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2820"><div class="verse-block verse-justify"><div class="line">पराक्रमं तु तं दृष्ट्वा तव सूनोर्महारथाः।</div><div class="line" style="position:relative;">मृत्योरुपान्तिकं प्राप्तौ माद्रीपुत्रौ स्म मेनिरे॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2821"><div class="verse-block verse-justify"><div class="line">ततः सेनापती राजन्पाण्डवानां महारथः।</div><div class="line" style="position:relative;">पार्षतः प्रययौ तत्र यत्र राजा सुयोधनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2822"><div class="verse-block verse-justify"><div class="line">माद्रीपुत्रौ ततः शूरौ व्यतिक्रम्य महारथौ।</div><div class="line" style="position:relative;">धृष्टद्युम्नस्तव सुतं पीडयामास सायकैः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2823"><div class="verse-block verse-justify"><div class="line">तमविध्यदमेयात्मा तव पुत्रो ह्यमर्पणः।</div><div class="line" style="position:relative;">पाञ्चाल्यं पञ्चविंशत्या प्रहस्य पुरुषर्षभः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2824"><div class="verse-block verse-justify"><div class="line">ततः पुनरमेयात्मा तव पुत्रो ह्यमर्षणः।</div><div class="line" style="position:relative;">विद्ध्वा ननाद पाञ्चाल्यं षष्ट्या पञ्चभिरेव च॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2825"><div class="verse-block verse-justify"><div class="line">तथास्य सशरं चापं मुष्टिदेशे विशाम्पते।</div><div class="line" style="position:relative;">क्षुरप्रेण सुतीक्ष्णेन राजा चिच्छेद संयुगे॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2826"><div class="verse-block verse-justify"><div class="line">तदपास्य धनुश्छिन्नं पाञ्चाल्यः शत्रुकर्शनः।</div><div class="line" style="position:relative;">अन्यदादत्त वेगेन धनुर्भारसहं नवम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2827"><div class="verse-block verse-justify"><div class="line">क्रोधाद्रुधिररक्ताक्षः संरम्भात्प्रज्वलन्निव।</div><div class="line" style="position:relative;">अशोभत महेष्वासो धृष्टद्युम्नः कृतव्रणः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2828"><div class="verse-block verse-justify"><div class="line">स पञ्चदश नाराचाञ्श्वसतः पन्नगानिव।</div><div class="line" style="position:relative;">जिघांसुर्भरतश्रेष्ठं धृष्टद्युम्नो व्यपासृजत्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2829"><div class="verse-block verse-justify"><div class="line">ते वर्म हेमविकृतं भित्त्वा राज्ञः शिलीमुखाः।</div><div class="line" style="position:relative;">विविशुर्वसुधां वेगात्कङ्कबर्हिणवाससः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2830"><div class="verse-block verse-justify"><div class="line">सोऽतिविद्धो महाराज पुत्रस्तेऽतिव्यराजत।</div><div class="line" style="position:relative;">वसन्तकाले सुमहान्प्रफुल्ल इव किंशुकः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2831"><div class="verse-block verse-justify"><div class="line">स च्छिन्नवर्मा नाराचप्रहारैर्झर्झरीकृतः।</div><div class="line" style="position:relative;">धृष्टद्युम्नस्य भल्लेन क्रुद्धश्चिच्छेद कार्मुकम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2832"><div class="verse-block verse-justify"><div class="line">अथैनं छिन्नधन्वानं त्वरमाणो महीपतिः।</div><div class="line" style="position:relative;">सायकैर्दशभी राजन्भ्रुवोर्मध्ये समार्पयत्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2833"><div class="verse-block verse-justify"><div class="line">तस्य तेऽशोभयन्वक्त्रं कर्मारपरिमार्जिताः।</div><div class="line" style="position:relative;">प्रफुल्लं पङ्कजं यद्वद्धमरा मधुलिप्सवः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2834"><div class="verse-block verse-justify"><div class="line">तदपास्य धनुश्छिन्नं दृष्टद्युम्नो महामनाः।</div><div class="line" style="position:relative;">अन्यदादत्त वेगेन धनुर्भल्लांश्च षोडश॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2835"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनस्याश्वान्हत्वा सूतं च पञ्चभिः।</div><div class="line" style="position:relative;">धनुश्चिच्छेद भल्लेन जातरूपपरिष्कृतम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2836"><div class="verse-block verse-justify"><div class="line">रथं सोपस्करं छत्रं शक्तिं खङ्गं गदां ध्वजम्।</div><div class="line" style="position:relative;">भल्लैश्चिच्छेद दशभिः पुत्रस्य तव पार्षतः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2837"><div class="verse-block verse-justify"><div class="line">तपनीयाङ्गदं छत्रं नागं मणिमयं शुभम्।</div><div class="line" style="position:relative;">ध्वजं कुरुपतेश्छिन्नं ददृशुः सर्वपार्थिवाः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2838"><div class="verse-block verse-justify"><div class="line">दुर्योधनं तु विरथं छिन्नवर्मायुधध्वजम्।</div><div class="line" style="position:relative;">भ्रातरं समुदैक्षन्त सोदरा भरतर्षभ॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2839"><div class="verse-block verse-justify"><div class="line">तमारोप्य रथे राजन्कुण्डधारो महारथम्।</div><div class="line" style="position:relative;">अपाहरदसम्भ्रान्तो धृष्टद्युम्नस्य पश्यतः॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2840"><div class="verse-block verse-justify"><div class="line">कर्णस्तु सात्यकिं हित्वा राजगृघ्नुरमर्षणः।</div><div class="line" style="position:relative;">द्रोमहन्तारमुग्रेषुं ससाराभिमुखो रणे॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2841"><div class="verse-block verse-justify"><div class="line">तं पृष्ठतोऽन्वयात्तूर्णं शैनेयोऽभिहतः शरैः।</div><div class="line" style="position:relative;">वारणं जघनोपान्ते विषाणाभ्यामिव द्विपः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2842"><div class="verse-block verse-justify"><div class="line">स भारतपुरोगानां राज्ञां च सुमहात्मनाम्।</div><div class="line" style="position:relative;">कर्णपार्षतयोर्युद्वे सङ्क्रद्धानां महारणे॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2843"><div class="verse-block verse-justify"><div class="line">न पाण्डवानां नास्माकं कश्चिदासीत्पराङ्मुखः।</div><div class="line" style="position:relative;">प्रत्यदृश्यत यत्कर्णः पाञ्चालांस्त्वरितो ययौ॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2844"><div class="verse-block verse-justify"><div class="line">तस्मिन्क्षणे नरश्रेष्ठ गजवाजिरथक्षयः।</div><div class="line" style="position:relative;">प्रादुरासीदुभयतो राजन्मध्यगतेऽहनि॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2845"><div class="verse-block verse-justify"><div class="line">पाञ्चालास्तु महाराज त्वरिता विजिगीषवः।</div><div class="line" style="position:relative;">सर्वतोऽभ्यद्रवन्कर्णं पतत्त्रिण इव द्रुमम्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2846"><div class="verse-block verse-justify"><div class="line">तेषामाधिरथिः क्रुद्धः प्रधानान्वै तरस्विनः।</div><div class="line" style="position:relative;">विचिन्वन्निव बाणौघैः समासादयदग्रतः॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2847"><div class="verse-block verse-justify"><div class="line">व्याघ्रकेतुः सुशर्मा च शुक्रश्चित्रायुधः क्रतुः।</div><div class="line">दुर्जयो रोचमानश्च सिंहसेनस्तथाऽष्टमः।</div><div class="line" style="position:relative;">महता रथवंशेन परिवव्रुर्नरोत्तमम्॥४७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2848"><div class="verse-block verse-justify"><div class="line">सृजन्तः सायकांस्तूर्णं कर्णमाहवशोभिनम्।</div><div class="line">यतमानांस्तु ताञ्शूरान्मनुजेन्द्रः शितैः शरैः।</div><div class="line" style="position:relative;">अष्टाभिरष्टौ राधेयः पाञ्चालान्न्यहनद्रणे॥४८॥</div></div></div>
