<div class="verse-block-wrapper" id="08-karṇaparva-b0870"><div class="verse-block verse-justify"><div class="line">नकुलं रभसं युद्धे द्रावयन्तं वरूथिनीम्।</div><div class="line" style="position:relative;">कर्णो वैकर्तनो राजन्वारयामास वै रुषा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0871"><div class="verse-block verse-justify"><div class="line">नकुलस्तु ततः कर्णं प्रहसन्निदमब्रवीत् ॥चिरस्य बत दृष्टोऽहं दैवतैः सौम्यचक्षुषा।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0872"><div class="verse-block verse-justify"><div class="line">यस्य मे त्वं रणे पाप चक्षुर्विषयमागतः ॥त्वं हि मूलमनर्थानां वैरस्य कलहस्य च।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0873"><div class="verse-block verse-justify"><div class="line">त्वद्दोषात्कुरवः क्षीणाः समासाद्य परस्परम्।</div><div class="line" style="position:relative;">त्वामद्य समरे हत्वा कृतकृत्योऽस्मि विज्वरः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0874"><div class="verse-block verse-justify"><div class="line">एवमुक्तः प्रत्युवाच नकुलं सूतनन्दनः।</div><div class="line" style="position:relative;">सदृशं राजपुत्रस्य धन्विनश्च विशेषतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0875"><div class="verse-block verse-justify"><div class="line">प्रहरस्व च मे वीर पश्यामस्तव पौरुषम्।</div><div class="line" style="position:relative;">कर्म कृत्वा रमे शूर ततः कत्थितुमर्हसि॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0876"><div class="verse-block verse-justify"><div class="line">अनुक्त्वा समरे तात शूरा युध्यन्ति शक्तितः।</div><div class="line" style="position:relative;">प्रयुध्यस्व मया शक्त्या हनिष्ये दर्पमेव ते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0877"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा प्राहरत्तूर्णं पाण्डुपुत्राय सूतजः।</div><div class="line" style="position:relative;">विव्याध चैनं समरे त्रिसप्तत्या शिलीमुखैः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0878"><div class="verse-block verse-justify"><div class="line">नकुलस्तु ततो विद्वः सूतपुत्रेण भारत।</div><div class="line" style="position:relative;">अशीत्याऽऽशीविषप्रख्यैः सूतपुत्रमविध्यत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0879"><div class="verse-block verse-justify"><div class="line">तस्य कर्णो धनुश्छित्त्वा स्वर्णपुङ्खैः शिलाशितैः।</div><div class="line" style="position:relative;">त्रिंशता परमेष्वासः शरैः पाण्डवमर्दयत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0880"><div class="verse-block verse-justify"><div class="line">ते तस्य कवचं भित्त्वा पपुः शोणितमाहवे।</div><div class="line" style="position:relative;">आशीविषा यथा नागा भित्त्वा गां सलिलं पपुः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0881"><div class="verse-block verse-justify"><div class="line">अथान्यद्वनुरादाय हेमपृष्ठं दुरासदम्।</div><div class="line" style="position:relative;">कर्णं विव्याध सप्तत्या सारथिञ्च त्रिभिः शरैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0882"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धो महाराज नकुलः परवीहा।</div><div class="line" style="position:relative;">क्षुरप्रेण सुतीक्ष्णेन कर्णस्य धनुराच्छिनत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0883"><div class="verse-block verse-justify"><div class="line">अथैनं छिन्नधन्वानं सायकानां शतैस्त्रिभिः।</div><div class="line" style="position:relative;">आजघ्ने प्रहसन्वीरः सर्वलोकमहारथम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0884"><div class="verse-block verse-justify"><div class="line">कर्णमभ्यर्दितं दृष्ट्वा पाण्डुपुत्रेण मारिष।</div><div class="line" style="position:relative;">विस्मयं परमं जग्मुर्ऋषयः सह दैवतैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0885"><div class="verse-block verse-justify"><div class="line">अथान्यद्वनुरादाय कर्णो वैकर्तनस्तदा।</div><div class="line" style="position:relative;">नकुलं पञ्छभिर्बाणैर्जत्रुदेशे समार्पयत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0886"><div class="verse-block verse-justify"><div class="line">उरस्थैरथ तैर्बाणैर्माद्रीपुत्रो व्यरोचत।</div><div class="line" style="position:relative;">स्वरश्मिभिरिवादित्यो भुवने विसृजन्प्रभाम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0887"><div class="verse-block verse-justify"><div class="line">नकुलस्तु ततः कर्णं विद्व्वा सप्तभिराशुगैः।</div><div class="line" style="position:relative;">अथास्य धनुषः कोटिं पुनश्चिच्छेद मारिष॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0888"><div class="verse-block verse-justify"><div class="line">सोऽन्यत्कार्मुकमादाय समरे वेगवत्तरम्।</div><div class="line" style="position:relative;">नकुलस्य ततो बैणैः समन्ताच्छादयद्दिशः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0889"><div class="verse-block verse-justify"><div class="line">सञ्छाद्यमानः सहसा कर्णचाप्युतैः सरैः।</div><div class="line" style="position:relative;">चिच्छेद स शरांस्तूर्णं शरैरेव महारथः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0890"><div class="verse-block verse-justify"><div class="line">ततो बाणमयं जालं विततं व्योम्न्यदृश्यत।</div><div class="line" style="position:relative;">खद्योतानामिव व्रातैः सम्पतद्भिर्यथा नभः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0891"><div class="verse-block verse-justify"><div class="line">तैर्विमुक्तैः शरशतैश्छादितं गगनं तदा।</div><div class="line" style="position:relative;">शलभानां यथा व्रातैस्तद्वदासीद्विशाम्पते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0892"><div class="verse-block verse-justify"><div class="line">ते शरा हेमविकृताः सम्पन्तौ मुहुर्मुहुः।</div><div class="line" style="position:relative;">श्रेणीकृता व्यकाशान्त क्रौञ्चाः श्रेणीकृता इव॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0893"><div class="verse-block verse-justify"><div class="line">बाणजालवृते व्योम्नि च्छादिते च दिवाकरे।</div><div class="line" style="position:relative;">न स्म सम्पतते भूतं किञ्चिदप्यन्तरिक्षगम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0894"><div class="verse-block verse-justify"><div class="line">निरुद्वे तत्र मार्गे च शरसङ्घैः समन्ततः।</div><div class="line" style="position:relative;">व्यरोचेतां महात्मानौ कालसूर्याविवोदितौ॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0895"><div class="verse-block verse-justify"><div class="line">कर्णचापच्युतैर्बाणैर्वध्यमानास्तु सोमकाः।</div><div class="line" style="position:relative;">अवालीयन्त राजेन्द्र वेदनार्ता भूशार्दिताः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0896"><div class="verse-block verse-justify"><div class="line">नकुलस्य तथा बाणैर्हन्यमाना चमूस्तव।</div><div class="line" style="position:relative;">व्यशीर्यत दिशो राजन्वातनुन्ना इवाम्बुदाः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0897"><div class="verse-block verse-justify"><div class="line">ते सेने हन्यमाने तु ताभ्यां दिव्यैर्महाशरैः।</div><div class="line" style="position:relative;">शरपातमपाक्रम्य तस्थतुः प्रेक्षिके तदा॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0898"><div class="verse-block verse-justify"><div class="line">प्रोत्सारितजने तस्मिन्कर्णपाण्डवयोः शरैः।</div><div class="line" style="position:relative;">अविध्येतां महात्मानावन्योन्यं शरवृष्टिभिः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0899"><div class="verse-block verse-justify"><div class="line">विदर्शयन्तौ दिव्यानि शस्त्राणि रणमूर्धनि।</div><div class="line" style="position:relative;">छादयन्तौ च सहसा परस्परवधैषिणौ॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0900"><div class="verse-block verse-justify"><div class="line">नकुलेन शरा मुक्ता कङ्कबर्हिणवाससः।</div><div class="line" style="position:relative;">सूतपुत्रमवच्छाद्य व्यतिष्ठन्त यथाऽम्बरे॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0901"><div class="verse-block verse-justify"><div class="line">तथैव सूतपुत्रेण प्रेषिताः परमाहवे।</div><div class="line" style="position:relative;">पाण्डुपुत्रमवच्छाद्य व्यतिष्ठन्ताम्बरे शराः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0902"><div class="verse-block verse-justify"><div class="line">शरवेश्मप्रविष्टौ तौ ददृशाते न कैश्चन।</div><div class="line" style="position:relative;">सूर्याचन्द्रमसौ राजञ्छाद्यमानौ घनैरिव॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0903"><div class="verse-block verse-justify"><div class="line">ततः क्रुद्धो रणे कर्णः कृत्वा घोरतरं वपुः।</div><div class="line" style="position:relative;">पाण्डवं छादयामास समन्ताच्छरवृष्टिभिः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0904"><div class="verse-block verse-justify"><div class="line">सोऽतिच्छन्नो महाराज सूतपुत्रेण पाण्डवः।</div><div class="line" style="position:relative;">न चकार व्यथां राजन्भास्करो जलदैर्यथा॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0905"><div class="verse-block verse-justify"><div class="line">ततः प्रहस्याधिरथिः शरजालानि मारिष।</div><div class="line" style="position:relative;">प्रेषयामास समरे शतशोऽथ सहस्रशः॥३७॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0910"><div class="verse-block verse-justify"><div class="line">शतचन्द्रं च तच्चर्म सर्वापकरणानि च।</div><div class="line" style="position:relative;">`सुवर्णविकृतं तच्च धनुः शरमाहवे'॥४१॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0913"><div class="verse-block verse-justify"><div class="line">व्यायुधं चैनमालक्ष्य शरैः सन्नतपर्वभिः।</div><div class="line" style="position:relative;">आर्पयद्बहुभिः कर्णो न चैनं समपीडयत्॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0914"><div class="verse-block verse-justify"><div class="line">स हन्यमानः समरे कृतास्त्रेण बलीयसा।</div><div class="line" style="position:relative;">प्राद्रवत्सहसा राजन्नकुलो व्याकुलेन्द्रियः॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b0915"><div class="verse-block verse-justify"><div class="line">तमभिद्रुत्य राधेयः प्रहसन्वै पुनःपुनः।</div><div class="line" style="position:relative;">सज्यमस्य धनुः कण्ठे व्यवासृजत भारत॥४६॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b0917"><div class="verse-block verse-justify"><div class="line">यथैव चासितो मेघः शक्रचापेन शोभितः।</div><div class="line" style="position:relative;">`अशोभत महाराज पाण्डुपुत्रस्तथा रणे'॥४८॥</div></div></div>
