<div class="verse-block-wrapper" id="08-karṇaparva-b3555"><div class="verse-block verse-justify"><div class="line">स समासाद्य राजानं दुर्योधनहितेप्सया।</div><div class="line" style="position:relative;">सूतपुत्रस्त्रिभिस्तीक्ष्णैर्विव्याध परमेषुभिः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3556"><div class="verse-block verse-justify"><div class="line">तथैव राजा राधेयं प्रत्यविध्यत्स्तनान्तरे।</div><div class="line" style="position:relative;">शरैस्त्रिभिश्च यन्तारं चतुर्भिश्चतुरो हयान्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3557"><div class="verse-block verse-justify"><div class="line">चक्ररक्षौ नृपसुतौ माद्रीपुत्रौ परन्तपौ।</div><div class="line" style="position:relative;">तावप्यधावतां कर्णं राजानं मा वधीदिति॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3558"><div class="verse-block verse-justify"><div class="line">तौ पृथक्शरवर्षाभ्यां राधेयं समवर्षताम्।</div><div class="line" style="position:relative;">नकुलः सहदेवश्च परमं यत्नमास्थितौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3559"><div class="verse-block verse-justify"><div class="line">तथैव तौ प्रत्यविध्यत्सूतपुत्रः प्रतापवान्।</div><div class="line" style="position:relative;">भल्लाभ्यां शितधाराभ्यां महात्मानावरिन्दमौ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3560"><div class="verse-block verse-justify"><div class="line">दन्तवर्णांस्तु राधेयो निजघान मनोजवान्।</div><div class="line" style="position:relative;">युधिष्ठिरस्य सङ्ग्रामे कालवालान्हयोत्तमान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3561"><div class="verse-block verse-justify"><div class="line">ततोऽपरेण भल्लेन शिरस्त्राणमपातयत्।</div><div class="line" style="position:relative;">कौन्तेयस्य महेष्वासः प्रहसन्निव सूतजः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3562"><div class="verse-block verse-justify"><div class="line">तथैव नकुलस्यापि हयान्हत्वा महारथः।</div><div class="line">ततोऽपरेण भल्लेन भृशं तीक्ष्णेन भारत।</div><div class="line" style="position:relative;">धनुश्चिच्छेद वीरस्य माद्रीपुत्रस्य धीमतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3563"><div class="verse-block verse-justify"><div class="line">तौ हताश्वौ हतरथौ पाण्डवौ भृशविक्षतौ।</div><div class="line" style="position:relative;">भ्रातरावारुरुहतुः सहदेवरथं तदा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3564"><div class="verse-block verse-justify"><div class="line">तौ दृष्ट्वा मातुलस्तत्र विरथौ परवीरहा।</div><div class="line" style="position:relative;">अभ्यभाषत राधेयं मद्रराजोऽनुकम्पया॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3567"><div class="verse-block verse-justify"><div class="line">तथापि कर्णः संरब्धो युधिष्ठिरमपीडयत्।</div><div class="line" style="position:relative;">शरैस्तीक्ष्णैर्महावीर्यैर्माद्रीपुत्रौ च पाण्डवौ॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b3568"><div class="verse-block verse-justify"><div class="line">ततः शल्यः प्रहस्येदं कर्णं पुनरुवाच ह।</div><div class="line" style="position:relative;">रथस्थमतिसंरब्धं युधिष्ठिरवधे स्थितम्॥२०॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b3571"><div class="verse-block verse-justify"><div class="line">शङ्खयोर्ध्मातयोः शब्दः सुमहानेष कृष्णयोः।</div><div class="line" style="position:relative;">श्रूयते चापघोषश्च प्रावृषीवाम्बुदस्य ह॥२३॥</div></div></div>
