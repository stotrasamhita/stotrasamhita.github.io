<div class="verse-block-wrapper" id="08-karṇaparva-b1475"><div class="verse-block verse-justify"><div class="line">तेजसोऽर्धं सुरा दत्त्वा शङ्कराय महात्मने।</div><div class="line" style="position:relative;">पशुत्वमपि चोपेत्य विश्वकर्माणमव्ययम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1476"><div class="verse-block verse-justify"><div class="line">ऊचुः सर्वे समाभाष्य रथः सङ्कल्प्यतामिति।</div><div class="line" style="position:relative;">विश्वकर्माऽपि सञ्चिन्त्य रथं दिव्यमकल्पयत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1477"><div class="verse-block verse-justify"><div class="line">समेतां पृथिवीं देवी विशालां पुरमालिनीम्।</div><div class="line" style="position:relative;">सपर्वतवनद्वीपां चक्रे भूतधरां रथम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1478"><div class="verse-block verse-justify"><div class="line">ईषां नक्षत्रवंशं च छत्रं मेरुमहागिरिम्।</div><div class="line" style="position:relative;">अनेकद्रुसञ्छन्नं रत्नाकरमनुत्तमम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1479"><div class="verse-block verse-justify"><div class="line">हिमवन्तं च विन्ध्यं च नानाद्रुमलताकुलम्।</div><div class="line" style="position:relative;">अवस्करं प्रतिष्ठानं कल्पयामास वै तदा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1480"><div class="verse-block verse-justify"><div class="line">अस्तंगिरिमधिष्ठानं नानाद्विजगणायुतम्।</div><div class="line" style="position:relative;">चकार भगवांस्त्वष्टा उदयं रथकूबरम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1481"><div class="verse-block verse-justify"><div class="line">मीननक्रझषावासं दानवालयमुत्तमम्।</div><div class="line" style="position:relative;">समुद्रमक्षं विदधे पत्तनाकरशोभितम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1482"><div class="verse-block verse-justify"><div class="line">चक्रं चक्रे चन्द्रमसं तारकागणमण्डितम्।</div><div class="line" style="position:relative;">दिवाकरं चाप्यपरं चक्रं चक्रेंऽशुमालिनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1483"><div class="verse-block verse-justify"><div class="line">गङ्गां सरस्वतीं तूणीं चक्रे विश्वकृदव्ययः।</div><div class="line" style="position:relative;">अलङ्कारा रथस्यासन्नापगाः सरितस्तथा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1484"><div class="verse-block verse-justify"><div class="line">त्रीनग्नीन्मन्त्रवच्चक्रे रथस्याथ त्रिवेणुकम्।</div><div class="line" style="position:relative;">अनुकर्षान्रथे दीप्तान्वरूथांश्चापि तारकाः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1485"><div class="verse-block verse-justify"><div class="line">ओषधीर्वीरुधश्चैव घण्टाजालं च भानुमत्।</div><div class="line" style="position:relative;">अलञ्चकार च रथं मासपक्षर्तुभिर्विभुः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1486"><div class="verse-block verse-justify"><div class="line">अहोरात्रैः कलाभिश्च काष्ठाभिरयनैस्तथा।</div><div class="line" style="position:relative;">द्यां युगं युगपर्वाणि संवर्तकबलाहरकान्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1487"><div class="verse-block verse-justify"><div class="line">शम्यां धृतिं च मेधां च स्थितिं सन्नतिमेव च।</div><div class="line" style="position:relative;">ऋघ्वेदं सामवेदं च धुर्यावश्वावकल्पयत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1488"><div class="verse-block verse-justify"><div class="line">पृष्ठाश्वौ तु यजुर्वेदः कल्पितोऽथर्वणस्तथा।</div><div class="line" style="position:relative;">अश्वानां चाप्यलङ्कारं विदधे पदसञ्चयम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1489"><div class="verse-block verse-justify"><div class="line">सिनीवालीमनुमतिं कुहूं राकां च सुप्रभाम्।</div><div class="line">योक्ताणि चक्रे चाश्वानां कूश्माण्डांश्चापि पन्नगान्।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1490"><div class="verse-block verse-justify"><div class="line">तालपृष्ठोऽथ नहुषः कार्कोटकधनञ्जयौ।</div><div class="line" style="position:relative;">इतरे चाभवन्नागा हयानां वाहबन्धनम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1491"><div class="verse-block verse-justify"><div class="line">अभीशवः षडङ्गानि कल्पितानि महीपते।</div><div class="line" style="position:relative;">ओङ्कारः कल्पितस्तस्य प्रतोदो विश्वकर्मणा॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1492"><div class="verse-block verse-justify"><div class="line">यज्ञाः सर्वे पृथक्लृप्ता रथाङ्गानि च भागशः।</div><div class="line" style="position:relative;">अधिष्ठानं मनश्चासीत्परिरथ्या सरस्वती॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1493"><div class="verse-block verse-justify"><div class="line">नानावर्णानि शस्त्राणि पताकाः पवनेरिताः।</div><div class="line" style="position:relative;">विद्युदिन्द्रधनुर्युक्तं रथं दीप्त्या व्यदीपयत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1494"><div class="verse-block verse-justify"><div class="line">वर्म योद्धुं च विहितं नमो ग्रहगणाकुलम्।</div><div class="line" style="position:relative;">अभेद्यं भानुमच्चित्रं कालचक्रपरिक्षतम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1495"><div class="verse-block verse-justify"><div class="line">एवमस्मिन्महाराज कल्पिते रथसत्तमे।</div><div class="line" style="position:relative;">त्वष्ट्रा मनुजशार्दूल द्विषतां भयवर्धने॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1496"><div class="verse-block verse-justify"><div class="line">स्वान्यायुधानि दिव्यानि न्यदधाच्छङ्करो रथे।</div><div class="line" style="position:relative;">ध्वजयष्टिं वियत्कृत्वा स्थापयामास गोवृषम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1497"><div class="verse-block verse-justify"><div class="line">ब्रह्मदण़्डः कालदम्डो रुद्रदण्डश्च ते ज्वराः।</div><div class="line" style="position:relative;">परिष्कारा र्थस्यासन्समन्ताद्दिशमुद्धताः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1498"><div class="verse-block verse-justify"><div class="line">विचित्रमृतुभिः षड्भिः कृत्वा संवत्सरं धनुः।</div><div class="line" style="position:relative;">छायामेवात्मनश्चक्रे धनुर्ज्यामक्षयां ध्रुवाम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1499"><div class="verse-block verse-justify"><div class="line">कालो हि भगवान्रुद्रस्तच्च संवत्सरं धनुः।</div><div class="line" style="position:relative;">तस्माद्रौद्री कालरात्री ज्या कृता धनुषो जरा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1500"><div class="verse-block verse-justify"><div class="line">ततो रथे रथाश्वांस्तानृषयः समयोजयन्।</div><div class="line" style="position:relative;">एकैकशः सुसंहृष्टानादाय सुधृतव्रताः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1501"><div class="verse-block verse-justify"><div class="line">दक्षिणस्यां धुरि कृत ऋग्वेदो मन्त्रपारगैः।</div><div class="line" style="position:relative;">सव्यतः सामवेदश्च युक्तो राजन्महर्षिभिः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1502"><div class="verse-block verse-justify"><div class="line">पार्ष्ठिदक्षिणतो युक्तो यजुर्वेदः सुरद्विजैः।</div><div class="line" style="position:relative;">इतरस्यां तथा पार्ष्ठ्यां युक्तो राजन्नथर्वणः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1503"><div class="verse-block verse-justify"><div class="line">एवं ते वाजिनो युक्ता यज्ञविद्भिस्तथा रथे।</div><div class="line" style="position:relative;">अशोभन्त तथा युक्ता यथैवाध्वरमध्यगाः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1504"><div class="verse-block verse-justify"><div class="line">कल्पयित्वा रथं दिव्यं ततो बाणमकल्पयत्।</div><div class="line" style="position:relative;">चिन्तयित्वा हरिं विष्णुमव्ययं यज्ञवाहनम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1505"><div class="verse-block verse-justify"><div class="line">शरं सङ्कल्पयाञ्चक्रे विश्वकर्मा महामनाः।</div><div class="line" style="position:relative;">तस्य वाजांश्च पुङ्खं च कल्पयामास वै तदा॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1506"><div class="verse-block verse-justify"><div class="line">पुण्यगन्धवहं राजञ्श्वसनं राजसत्तम।</div><div class="line" style="position:relative;">अग्नीषोमौ शरमुखे कल्पयामास वै तदा॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1507"><div class="verse-block verse-justify"><div class="line">अग्नीषोमात्मकं कृत्स्नमुच्यते वैष्णवं जगत्।</div><div class="line" style="position:relative;">विष्णुरात्मा भगवतो भवस्यामिततेजसः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1508"><div class="verse-block verse-justify"><div class="line">तस्माद्धनुर्ज्यासंस्पर्शं स विषेहे शरस्य वै।</div><div class="line" style="position:relative;">तस्मिञ्शरे तीक्ष्णमन्युममुञ्चद्दुःसहं प्रभुः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1509"><div class="verse-block verse-justify"><div class="line">भृग्वङ्गिरोमन्युभवः क्रोधाग्निरतिदुःसहः।</div><div class="line" style="position:relative;">स नीललोहितो धूम्रः कृत्तिवासा भयानकः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1510"><div class="verse-block verse-justify"><div class="line">आदित्यायुतसङ्काशस्तेजोज्वालावृतो भवः।</div><div class="line" style="position:relative;">दुश्चर्यच्यावको जेता हन्ता ब्रह्मद्विषां वरः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1511"><div class="verse-block verse-justify"><div class="line">तस्याङ्गानि समाश्रित्य स्थितं विश्वमिदं जगत्।</div><div class="line" style="position:relative;">जङ्गमाजङ्गमं राजञ्छुशुभेऽद्भुतदर्शनम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1512"><div class="verse-block verse-justify"><div class="line">दृष्टा तु तं रथं दिव्यं कवची स शरासनी।</div><div class="line" style="position:relative;">आददे स शरं दिव्यं सोमविष्ण्वग्निवायुजम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1513"><div class="verse-block verse-justify"><div class="line">तमादाय महादेवस्त्रासयन्दैत्यदानवान्।</div><div class="line" style="position:relative;">आरुरोह तदा यत्तः कम्पयन्निव रोदसी॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1514"><div class="verse-block verse-justify"><div class="line">महर्षिभिः स्तूयमानो वन्द्यमानश्च वन्दिभिः।</div><div class="line" style="position:relative;">उपनृत्तश्चाप्सरसां गणैर्नृत्तविशारदैः॥४१॥</div></div></div>
