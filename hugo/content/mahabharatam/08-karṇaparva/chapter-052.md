<div class="verse-block-wrapper" id="08-karṇaparva-b2653"><div class="verse-block verse-justify"><div class="line">संशप्तकानामकरोत्कदनं यत्र पाण्डवः।</div><div class="line" style="position:relative;">कोसलानां तथा राजन्नारायणबलस्य च॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2654"><div class="verse-block verse-justify"><div class="line">संशप्तकास्तु समरे शरवृष्टिः समन्ततः।</div><div class="line" style="position:relative;">अपातयन्पार्थमूर्ध्नि जयगृद्धाः प्रमन्यवः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2655"><div class="verse-block verse-justify"><div class="line">तां शस्त्रवृष्टिमायान्तीमुरसा धारयन्प्रभुः।</div><div class="line" style="position:relative;">व्यगाहत परान्पार्थो विनिघ्नन्रथिनां वरान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2656"><div class="verse-block verse-justify"><div class="line">विक्षोभ्य तु रथानीकं कङ्कपत्रैः शिलाशितैः।</div><div class="line" style="position:relative;">आससाद ततः पार्थः सुशर्माणं महारथम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2657"><div class="verse-block verse-justify"><div class="line">स तस्य शरवर्षाणि ववर्ष रथिनां वरः।</div><div class="line" style="position:relative;">तथा संशप्तकाश्चैव पार्थस्य समरे स्थिताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2658"><div class="verse-block verse-justify"><div class="line">सुशर्मा तु ततः पार्थं विद्ध्वा दशभिराशुगैः।</div><div class="line" style="position:relative;">जनार्दनं त्रिभिर्बाणैरहनद्दक्षिणे भुजे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2659"><div class="verse-block verse-justify"><div class="line">ततोऽपरेण भल्लेन केतुं पार्थस्य मारिष।</div><div class="line" style="position:relative;">विव्याध समरे राजन्सुशर्मा क्रोधमूर्च्छितः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2660"><div class="verse-block verse-justify"><div class="line">स वानरवरो राजन्विश्वकर्मकृतो महान्।</div><div class="line" style="position:relative;">ननाद सुमहानादं नृत्यन्निव विभीषयन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2661"><div class="verse-block verse-justify"><div class="line">कपेस्तु निनदं श्रुत्वा सन्त्रस्ता तव वाहिनी।</div><div class="line" style="position:relative;">भयं विपुलमासाद्य निश्चेष्टा समपद्यत॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2662"><div class="verse-block verse-justify"><div class="line">ततः सा शुशुभे सेना निश्चेष्टाऽवस्थिता नृप।</div><div class="line" style="position:relative;">नानापुष्पसमाकीर्णं यथा चित्रीकृतं वनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2663"><div class="verse-block verse-justify"><div class="line">प्रतिलभ्य ततः संज्ञां योधास्ते कुरुसत्तम।</div><div class="line" style="position:relative;">अर्जुनं सिषिचुर्बाणैः पर्वतं जलदा इव॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2664"><div class="verse-block verse-justify"><div class="line">सञ्छाद्य समरे पार्थं परिवव्रुः समन्ततः।</div><div class="line">ते हयान्रथचक्रे तु रथेषां चापि मारिष।</div><div class="line" style="position:relative;">ग्रहीतुं प्रचक्रमुश्चैव क्रोधाविष्टाः समन्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2665"><div class="verse-block verse-justify"><div class="line">निगृह्य तु रथं तस्य योधास्ते तु सहस्रशः।</div><div class="line" style="position:relative;">रथबन्धं प्रचक्रुर्हि पाण्डवस्यामितौजसः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2666"><div class="verse-block verse-justify"><div class="line">रथमारुरुहुः केचित्कृष्णपार्थौ जिघृक्षवः।</div><div class="line" style="position:relative;">संशप्तकानां योधास्ते सिंहनादांश्च नेदिरे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2667"><div class="verse-block verse-justify"><div class="line">अपरे जगृहुश्चैव केशवस्य महाभुजौ।</div><div class="line" style="position:relative;">पार्थं चैके महाराज रथस्थं जगृहुर्मुदा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2668"><div class="verse-block verse-justify"><div class="line">अच्युतः स महाबाहुर्विधुन्वन्रणमूर्धनि।</div><div class="line" style="position:relative;">पातयामास तान्सर्वान्दुष्टहस्तीव हस्तिपान्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2669"><div class="verse-block verse-justify"><div class="line">स रथस्तैर्गृहीतस्तु पाण्डवस्य महात्मनः।</div><div class="line" style="position:relative;">स्पन्दितुं नाशकद्राजंस्तदद्भुतमिवाभवत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2670"><div class="verse-block verse-justify"><div class="line">ततः पार्थो महाबाहुः संवृतः स्तैर्महारथैः।</div><div class="line" style="position:relative;">निगृहीतं रथं दृष्ट्वा तांश्चाप्याद्रवतो बहून्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2671"><div class="verse-block verse-justify"><div class="line">रथारूढांश्च सुबहून्पदाऽऽक्षिप्य न्यपातयत्।</div><div class="line" style="position:relative;">अपातयदसम्भ्रान्तः शरैरासन्नयोधिभिः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2672"><div class="verse-block verse-justify"><div class="line">तांस्तापयित्वा समरे पार्थः परपुरञ्जयः।</div><div class="line" style="position:relative;">स्मयन्निव महाबाहुः केशवं वाक्यमब्रवीत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2673"><div class="verse-block verse-justify"><div class="line">पश्य कृष्ण महाबाहो संशप्तकगणान्बहून्।</div><div class="line" style="position:relative;">कुर्वतोऽसुकरं कर्म मुमूर्षून्कालचोदितान्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2674"><div class="verse-block verse-justify"><div class="line">रथबन्धमिमं प्राप्य पृथिव्यां नास्ति कश्चन।</div><div class="line" style="position:relative;">यः सहेत पुमाँल्लोके मदन्यः क्षत्रियर्षभः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2675"><div class="verse-block verse-justify"><div class="line">पश्यैततानद्य समरे मत्प्रयुक्तैः सुतेजनैः।</div><div class="line" style="position:relative;">पात्यमानान्रणे कृष्ण शरैराशीविषोपमैः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2676"><div class="verse-block verse-justify"><div class="line">इत्येवमुक्त्वा बीभत्सुः शङ्खप्रवरमुत्तमम्।</div><div class="line" style="position:relative;">व्यनादयदमेयात्मा देवदत्तं महामृधे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2677"><div class="verse-block verse-justify"><div class="line">देवदत्तस्वनं श्रुत्वा केशवोऽपि महायशाः।</div><div class="line" style="position:relative;">पाञ्चजन्यस्वनं चक्रे पूरयन्निव रोदसी॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2678"><div class="verse-block verse-justify"><div class="line">तयोः शङ्खस्वनं श्रुत्वा संशप्तकवरूथिनी।</div><div class="line" style="position:relative;">सञ्चचाल महाराज वित्रस्ता चाद्रवद्भृशम्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2679"><div class="verse-block verse-justify"><div class="line">नागमस्त्रं ततः पार्थः प्रादुश्चक्रे हसन्निव।</div><div class="line" style="position:relative;">पादबन्धं स तेषां वै चक्रे तेन महास्त्रवित्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2680"><div class="verse-block verse-justify"><div class="line">यानुद्दिश्य रणे पार्थः पादबन्धं चकार ह।</div><div class="line" style="position:relative;">ते बद्धाः पादबन्धेन योधाः संशप्तकास्तदा॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2681"><div class="verse-block verse-justify"><div class="line">निर्विचेष्टास्तदाऽभूवन्पाण्डवस्यास्त्रतेजसा।</div><div class="line">निर्विचेष्टांस्ततो योधानवधीत्पाण्डुनन्दनः।</div><div class="line" style="position:relative;">यथेन्द्रः समरे दैत्यांस्तारकस्य वधे पुरा॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2682"><div class="verse-block verse-justify"><div class="line">ते वध्यमानाः समरे मुमुचुस्तं रथोत्तमम्।</div><div class="line" style="position:relative;">आयुधानि च सर्वाणि विस्रष्टुमुपचक्रमुः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2683"><div class="verse-block verse-justify"><div class="line">ततः सुशर्मा राजेन्द्र गृहीतां वीक्ष्य वाहिनीम्।</div><div class="line" style="position:relative;">सौपर्णमस्त्रं त्वरितः प्रादुश्चक्रे महारथः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2684"><div class="verse-block verse-justify"><div class="line">ततः सुपर्णाः सम्पेतुर्भक्षयन्तो भुजङ्गमान्।</div><div class="line" style="position:relative;">ततो विदुद्रुवुर्नागास्तान्दृष्ट्वा खेचरान्नृप॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2685"><div class="verse-block verse-justify"><div class="line">तद्विमुक्तं बलं रेजे पादबन्धाद्विशाम्पते।</div><div class="line" style="position:relative;">मेघबन्धाद्यथा मुक्तो भास्करस्तापयन्प्रजाः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2686"><div class="verse-block verse-justify"><div class="line">विप्रमुक्ताः स्वका योधाः फल्गुनस्य रथं प्रति।</div><div class="line" style="position:relative;">ससृजुः शरसङ्घांश्च शस्त्रसङ्घांश्च मारिष॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2687"><div class="verse-block verse-justify"><div class="line">तां महास्त्रमयीं वृष्टिं शरैः सञ्छिद्य भारत।</div><div class="line" style="position:relative;">अवधीत्सततो योधान्वासविः परवीरहा॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2688"><div class="verse-block verse-justify"><div class="line">सुशर्मा तु ततो राजन्बाणेन नतपर्वणा।</div><div class="line" style="position:relative;">अर्जुनं हृदये विद्ध्वा विव्याधान्यैस्त्रिसप्तभिः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2689"><div class="verse-block verse-justify"><div class="line">स गाढविद्धो व्यथितो रथोपस्थ उपाविशतत्।</div><div class="line" style="position:relative;">तत उच्चुक्रुशुः सर्वे हतः पार्थ इति ब्रुवन्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2690"><div class="verse-block verse-justify"><div class="line">ततः शङ्खनिनादाश्च भेरीशब्दाश्च पुष्कलाः।</div><div class="line" style="position:relative;">नानावादित्रनिनदाः सिंहनादाश्च जज्ञिरे॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2691"><div class="verse-block verse-justify"><div class="line">प्रतिलभ्य ततः संज्ञां श्वेताश्वः कृष्णसारथिः।</div><div class="line">सोऽतिविद्धो महेष्वासः शरैराशीविपोपमैः।</div><div class="line" style="position:relative;">सुशर्माणं महाराज क्रोधाविष्टो महारथः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2692"><div class="verse-block verse-justify"><div class="line">ततः शरशतैः पार्थः सञ्छाद्यैनं क्षणाद्रणे।</div><div class="line" style="position:relative;">दिशस्तु वारयामास बाणैस्तत्र महास्त्रवित्॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2693"><div class="verse-block verse-justify"><div class="line">विमुखीकृत्य समरे सुशर्माणं धनञ्जयः।</div><div class="line" style="position:relative;">ऐन्द्रमस्त्रममेयात्मा प्रादुश्चक्रे हसन्निव॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2694"><div class="verse-block verse-justify"><div class="line">ततो बाणसहस्राणि तदुत्सृष्टानि मारिष।</div><div class="line" style="position:relative;">सर्वदिक्षु व्यदृश्यन्त सूदयन्ति रथद्विपान्॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2695"><div class="verse-block verse-justify"><div class="line">हयान्पत्तींश्च समरे शस्त्रैः शतसहस्रशः।</div><div class="line" style="position:relative;">रराज समरे राजञ्शक्रो निघ्नन्निवासुरान्॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2696"><div class="verse-block verse-justify"><div class="line">वध्यमाने ततः सैन्ये भयं सुमहदाविशत्।</div><div class="line" style="position:relative;">संशप्तकगणानां च गोपालानां च भारत॥४६॥</div></div></div>
