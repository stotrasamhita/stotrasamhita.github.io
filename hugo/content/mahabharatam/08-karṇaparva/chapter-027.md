<div class="verse-block-wrapper" id="08-karṇaparva-b1297"><div class="verse-block verse-justify"><div class="line">मद्राजं च समरे प्रणम्य च महारथम्।</div><div class="line" style="position:relative;">विनयेनोपसङ्गम्य प्रणयाद्वाक्यमब्रवीत्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1298"><div class="verse-block verse-justify"><div class="line">सत्यव्रत महाभाग द्विषतां तापवर्धन।</div><div class="line" style="position:relative;">मद्रेश्वर रणेशूर परसैन्यभयङ्कर॥४॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b1300"><div class="verse-block verse-justify"><div class="line">तत्त्वामप्रतिवीर्याद्य शत्रुपक्षक्षयावह।</div><div class="line" style="position:relative;">मद्रेश्वर प्रयाचेऽहं शिरसा विनयेन च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1301"><div class="verse-block verse-justify"><div class="line">तस्मात्पार्थविनाशार्थं हितार्थं मम चैव हि।</div><div class="line" style="position:relative;">त्वं पाहि सर्वतः कर्णं भवं ब्रह्मेव सुव्रत॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1302"><div class="verse-block verse-justify"><div class="line">सारथ्यं रथिनां श्रेष्ठ प्रणयात्कर्तुमर्हसि।</div><div class="line" style="position:relative;">त्वयि यन्तरि राधेयो विद्विषो मे विजेष्यते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1303"><div class="verse-block verse-justify"><div class="line">अभीषूणां हि कर्णस्य ग्रहीतान्यो न विद्यते।</div><div class="line" style="position:relative;">ऋते हि त्वां महाभाग वासुदेवसमं युधि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1304"><div class="verse-block verse-justify"><div class="line">स पाहि सर्वथा कर्णं यथा ब्रह्मा महेश्वरम्।</div><div class="line">यथा च सर्वथाऽऽपत्सु वार्ष्णेयः पाति पाण्डवम्।</div><div class="line" style="position:relative;">तथा मद्रेश्वराद्य त्वं राधेयं प्रतिपालय॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1305"><div class="verse-block verse-justify"><div class="line">भीष्मो द्रोणः कृपः कर्णो भवान्भोजश्च वीर्यवान्।</div><div class="line" style="position:relative;">शकुनिः सौबलो द्रौणिरहमेव च नो बलम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1306"><div class="verse-block verse-justify"><div class="line">एवमेष कृतो भागो नवधा पृथिवीपते।</div><div class="line" style="position:relative;">न च भागोऽत्र भीष्मस्य द्रोणस्य च महात्मनः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1307"><div class="verse-block verse-justify"><div class="line">ताभ्यामतीत्य तौ भागौ निहता मम शत्रवः।</div><div class="line" style="position:relative;">वृद्वौ हि तौ महेष्वासौ छलेन निहतौ युधि॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1308"><div class="verse-block verse-justify"><div class="line">कृत्वा चासुकरं कर्म गतौ स्वर्गमितोऽनघ।</div><div class="line" style="position:relative;">तथान्ये पुरुषव्याघ्राः परैर्विनिहता युधि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1309"><div class="verse-block verse-justify"><div class="line">अस्मदीयाश्च बहवः स्वर्गाय गमिताः परैः।</div><div class="line">त्यक्त्वा प्राणानय्थाशक्ति चेष्टां कृत्वा च पुष्कलाम्।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1310"><div class="verse-block verse-justify"><div class="line">तदिदं हतभूयिष्ठं बलं मम नराधिप।</div><div class="line" style="position:relative;">पूर्वमप्यल्पकैः पार्थैर्हतं किमुत साम्प्रतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1311"><div class="verse-block verse-justify"><div class="line">बलवन्तो महात्मानः कौन्तेयाः सत्यविक्रमाः।</div><div class="line" style="position:relative;">बलं शेषं न हन्युर्मे यथा तत्कुरु पार्थिव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1312"><div class="verse-block verse-justify"><div class="line">हतवीरमिदं सैन्यं पाण्डवैः समरे विभो।</div><div class="line" style="position:relative;">कर्णो ह्येको महाबाहुरस्मत्प्रियहिते रतः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1313"><div class="verse-block verse-justify"><div class="line">भवांश्च पुरुषव्याघ्र सर्वलोकमहारथः।</div><div class="line" style="position:relative;">शल्य कर्णोऽर्जुनेनाद्य योद्वुमिच्छति संयुगे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1314"><div class="verse-block verse-justify"><div class="line">तस्मिञ्जयाशा विपुला मद्रराज नराधिप।</div><div class="line" style="position:relative;">तस्याभीषुग्रहवरो नान्योऽस्ति भुवि कश्चन॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1315"><div class="verse-block verse-justify"><div class="line">पार्थस्य समरे कृष्णो यथाऽभीषुग्रहो वरः।</div><div class="line" style="position:relative;">तथा त्वमपि कर्णस्य रथेऽभीषुग्रहो भव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1316"><div class="verse-block verse-justify"><div class="line">तेन युक्तो रणे पार्थो रक्ष्यमाणश्च पार्थिव।</div><div class="line" style="position:relative;">यानि कर्माणि कुरुते प्रत्यक्षाणि तथैव तत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1317"><div class="verse-block verse-justify"><div class="line">पूर्वं नः समरे ह्येवमधीदर्जुनो रिपून्।</div><div class="line" style="position:relative;">इदानीं विक्रमो ह्यस्य कृष्णेन सहितस्य च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1318"><div class="verse-block verse-justify"><div class="line">कृष्णेन सहितः पार्थो धार्तराष्ट्रीं महाचमूम्।</div><div class="line" style="position:relative;">अहन्यहनि मद्रेश द्रावयन्दृश्यते युधि॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1319"><div class="verse-block verse-justify"><div class="line">भागोऽवशिष्टः कर्णस्य तव चैव महाद्युते।</div><div class="line" style="position:relative;">तं भागं सह कर्णेन युगपन्नाशयाद्य हि॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1320"><div class="verse-block verse-justify"><div class="line">अरुणेन यथा सार्धं तमः सूर्यो व्यपोहति।</div><div class="line" style="position:relative;">तथा कर्णेन सहितो जहि पार्थं महाहवे॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1321"><div class="verse-block verse-justify"><div class="line">उद्यन्तौ च यथा सूर्यौ बालसूर्यसमप्रभौ।</div><div class="line" style="position:relative;">कर्णशल्यौ रणे दृष्ट्वा विद्रवन्तु महारथाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1322"><div class="verse-block verse-justify"><div class="line">सूर्यारुषौ यथा दृष्ट्वा तमो नश्यति मारिष।</div><div class="line" style="position:relative;">तथा नश्यन्तु कौन्तेयाः सपाञ्चालाः ससृञ्जयाः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1323"><div class="verse-block verse-justify"><div class="line">रथिनां प्रवरः कर्णो यन्तॄणां प्रवरो भवान्।</div><div class="line" style="position:relative;">संयोगो युवयोर्लोके नाभून्न च भविष्यति॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1324"><div class="verse-block verse-justify"><div class="line">यथा सर्वास्ववस्थासु वार्ष्मेयः पाति पाण्डवम्।</div><div class="line">तथा भवान्परित्रातु कर्णं वैकर्तनं रणे।</div><div class="line" style="position:relative;">`सारथ्यं क्रियतां तस्य युध्यमानस्य संयुगे॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1325"><div class="verse-block verse-justify"><div class="line">यथा च समरे कृष्णो रक्षते सर्वतोऽर्जुनम्।</div><div class="line" style="position:relative;">तथा त्वमपि राधेयं रक्षस्व च महारणे'॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1326"><div class="verse-block"><div class="line">त्वया सारथिना ह्येष अप्रधृष्यो भविष्यति</div><div class="line pada-even">देवतानामपि रणे सशक्राणां महीपते।</div><div class="line">किं पुनः पाण्डवेयानां मा विशङ्कीर्वचो मम ॥सञ्जय उवाच</div><div class="line pada-even" style="position:relative;">॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1327"><div class="verse-block verse-justify"><div class="line">दुर्योधनवचः श्रुत्वा शल्यः क्रोधसमन्वितः।</div><div class="line" style="position:relative;">`धार्तराष्ट्रमथोवाच सृक्विणी परिलेलिहन्'॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1328"><div class="verse-block"><div class="line">त्रिशिखां भ्रुकुटिं कृत्वा धुन्वन्हस्तौ पुनःपुनः</div><div class="line pada-even">क्रोधरक्ते महानेत्रे परिवृत्य महाभुजः।</div><div class="line">कुलैश्वर्यश्रुतबलैर्दृप्तः शल्योऽब्रवीदिदम् ॥शल्य उवाच</div><div class="line pada-even" style="position:relative;">॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1329"><div class="verse-block verse-justify"><div class="line">न मामर्हसि राजेद्र नियोक्तुं कर्मणीदृशे।</div><div class="line" style="position:relative;">न हि पापीयसः श्रेयान्प्रेष्यत्वं कर्तुमर्हति॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1330"><div class="verse-block verse-justify"><div class="line">यो ह्यभ्युपगतं मित्रं गरीयांसं विशेषतः।</div><div class="line" style="position:relative;">वशे पापीयसो धत्ते तत्कार्यमधरोत्तरम्॥३६॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b1332"><div class="verse-block verse-justify"><div class="line">तेभ्यो वर्णविशेषाश्च प्रतिलोमानुलोमजाः।</div><div class="line" style="position:relative;">अथान्योन्यस्य संयोगाच्चातुर्वर्ण्यस्य भारत॥३८॥</div></div></div>

<div class="verse-block-wrapper" id="08-karṇaparva-b1338"><div class="verse-block verse-justify"><div class="line">सोऽमहेतादृशो भूत्वा नेहारिबलसूदनः।</div><div class="line" style="position:relative;">सूतपुत्रस्य सङ्ग्रामे सारथ्यं कर्तुमुत्सहे॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1339"><div class="verse-block verse-justify"><div class="line">अवमन्यसे मां गान्धारे ध्रुवं चाप्यतिशङ्कसे।</div><div class="line" style="position:relative;">यस्माद्ब्रवीषि विस्रब्धः सारथ्यं क्रियतामिति॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1340"><div class="verse-block verse-justify"><div class="line">अस्मत्तोऽभ्यधिकं कर्णं मन्वानस्तं प्रशंससि।</div><div class="line" style="position:relative;">न चाहं युधि राधेयं गणये तुल्यमात्मनः॥४५॥</div></div></div>
