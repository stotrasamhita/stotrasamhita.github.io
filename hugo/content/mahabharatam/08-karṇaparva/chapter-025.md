<div class="verse-block-wrapper" id="08-karṇaparva-b1171"><div class="verse-block verse-justify"><div class="line">ततः कर्णं पुरस्कृत्य त्वदीया युद्वदुर्मदाः।</div><div class="line" style="position:relative;">पुनरावृत्य सङ्ग्रामं चक्रुर्देवासुरोपमम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1172"><div class="verse-block verse-justify"><div class="line">द्विरदनररथाश्वशङ्खशब्दैःपरिहृषिता विविधैश्च शस्त्रपातैः।</div><div class="line" style="position:relative;">द्विरदरथपदातिसादिसङ्घाःपरिकुपिताभिमुखाः प्रजघ्निरे ते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1173"><div class="verse-block verse-justify"><div class="line">शितपरश्वथसासिपट्टसै--रिषुभिरनेकविधैश्च सूदिताः।</div><div class="line" style="position:relative;">द्विरदरथहया महाहवेवरपुरुषैः पुरुषाश्च वाहनैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1174"><div class="verse-block verse-justify"><div class="line">कमलदिनकरेन्दुसन्निभैःसितदशनैः सुमुखाक्षिनासिकैः।</div><div class="line" style="position:relative;">रुचिरमकुटकुण्डलैर्महीपुरुषशिरोभिरुपस्तृता बभौ॥५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1175"><div class="verse-block verse-justify"><div class="line">परिघमुसलशक्तितौमरै--र्नखरभुशुण्डिगदाशतैर्हताः।</div><div class="line" style="position:relative;">द्विरदनरहयाः सहस्रशोरुधिरनदीप्रवहास्तदाऽभवन्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1176"><div class="verse-block verse-justify"><div class="line">प्रहतरथनराश्वकुञ्जरंप्रतिभयदर्शनमुल्बणव्रणम्।</div><div class="line" style="position:relative;">तदहितहतमाबभौ बलंपितृपतिराष्ट्रमिव प्रजाक्षये॥७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1177"><div class="verse-block verse-justify"><div class="line">अथ तव नरदेव सैनिका--स्तव च सुताः सुरसूनुसन्निभाः।</div><div class="line" style="position:relative;">अमितबलपुरः सरा रणेकुरुवृषभाः शिनिपुत्रमभ्ययुः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1178"><div class="verse-block verse-justify"><div class="line">तदतिरुधिरभीममाबभौपुरुषवराश्वरथद्विपाकुलम्।</div><div class="line" style="position:relative;">लवणजलसमुद्वतस्वनंबलमसुरामरसैन्यसप्रभम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1179"><div class="verse-block verse-justify"><div class="line">सुरपतिसमविक्रमस्तत--स्त्रिदशवरावरजोपमं युधि।</div><div class="line" style="position:relative;">दिनकरकिरणप्रभैः पृषत्कैरवितनयोऽभ्यहनच्छिनिप्रवीरम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1180"><div class="verse-block verse-justify"><div class="line">तमपि सरथवाजिसारथिंशिनिवृषभो विविधैः शरैस्त्वरन्।</div><div class="line" style="position:relative;">भुजगविषसमप्रभै रणेपुरुषवरं समवास्तृणोत्तदा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1181"><div class="verse-block verse-justify"><div class="line">शिनिवृषभशरैर्निपीडितंतव सुहृदो वसुषेणमभ्ययुः।</div><div class="line" style="position:relative;">त्वरितमतिरथा रथर्षभंद्विरदरथाश्वपदातिभिः सह॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1182"><div class="verse-block verse-justify"><div class="line">तदुदधिनिभमाद्रवद्बलंत्वरिततरैः समभिद्रुतं परैः।</div><div class="line" style="position:relative;">द्रुपदसुतमुखैस्तदाऽभवत्पुरुषरथाश्वगजक्षयो महान्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1183"><div class="verse-block verse-justify"><div class="line">अथ पुरुषवरौ कृताह्निकौभवमभिपूज्य यथाविधि प्रभुम्।</div><div class="line" style="position:relative;">अरिवधकृतनिश्चयौ द्रुतंतव बलमर्जुनकेशवौ सृतौ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1184"><div class="verse-block verse-justify"><div class="line">जलदनिनदनिःस्वनं रथंपवनविधूतपताककेतनम्।</div><div class="line" style="position:relative;">सितहयमुपयान्तमन्तिकंकृतमनसो ददृशुस्तदाऽरयः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1185"><div class="verse-block verse-justify"><div class="line">अथ विष्फार्य गाण्डीवं रथे नृत्यन्निवार्जुनः।</div><div class="line" style="position:relative;">शरसम्बाधमकरोत्खं दिशः प्रदिशस्तथा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1186"><div class="verse-block verse-justify"><div class="line">रथान्विमानप्रतिमान्मज्जयन्सायुधध्वजान्।</div><div class="line" style="position:relative;">ससारथींस्तदा बाणैरभ्राणीवानिलोऽवधीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1187"><div class="verse-block verse-justify"><div class="line">गजान्गजप्रयन्तॄंश्च वैजयन्त्यायुधध्वजान्।</div><div class="line" style="position:relative;">सादिनोऽश्वांश्च पत्तींश्च शरैर्निन्ये यमक्षयम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1188"><div class="verse-block verse-justify"><div class="line">तमन्तकमिव क्रुद्धमनिवार्यं महारथम्।</div><div class="line" style="position:relative;">दुर्योधनोऽभ्ययादेको निघ्नन्बाणैरजिह्मगैः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1189"><div class="verse-block verse-justify"><div class="line">तस्यार्जुनो धनुः सूतमश्वान्केतुं च सायकैः।</div><div class="line" style="position:relative;">हत्वा सप्तभिरेकेन च्छत्रं चिच्छेद पत्रिणा॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1190"><div class="verse-block verse-justify"><div class="line">नवमं च समाधाय व्यसृजत्प्राणघातिनम्।</div><div class="line" style="position:relative;">दुर्योधनायेषुवरं तं द्रौणिः सप्तधाऽच्छिनत्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1191"><div class="verse-block verse-justify"><div class="line">ततो द्रौणेर्धनुश्छित्वा हत्वा चाश्वरथाञ्शरैः।</div><div class="line" style="position:relative;">कृपस्यापि तदत्युग्रं धनुश्चिच्छेद पाण्डवः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1192"><div class="verse-block verse-justify"><div class="line">हार्दिक्यस्य धनुश्छित्त्वा ध्वजं चाश्वांस्तदाऽवधीत्।</div><div class="line" style="position:relative;">दुःशासनस्येष्वसनं छित्त्वा राधेयमभ्ययात्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1193"><div class="verse-block verse-justify"><div class="line">अथ सात्यकिमुत्सृज्य त्वरन्कर्णोऽर्जुनं त्रिभिः।</div><div class="line" style="position:relative;">विद्व्वा विव्याध विंशत्या कृष्णं पार्थं पुनःपुनः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1194"><div class="verse-block verse-justify"><div class="line">न ग्लानिरासीत्कर्णस्य क्षिपतः सायकान्बहून्।</div><div class="line" style="position:relative;">रणे विनिघ्नतः शत्रून्क्रुद्धस्येव शतक्रतोः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1195"><div class="verse-block verse-justify"><div class="line">अथ सात्यकिरागत्य कर्णं विद्द्वा शितैः शरैः।</div><div class="line" style="position:relative;">नवत्या नवभिश्चोग्रैः शतेन पुनरार्पयत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1196"><div class="verse-block verse-justify"><div class="line">ततः प्रवीराः पार्थानां सर्वे कर्णमपीडयन्।</div><div class="line" style="position:relative;">युधामन्युः शिखण्डी च द्रौपदेयाः प्रभद्रकाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1197"><div class="verse-block verse-justify"><div class="line">उत्तमौजा युयुत्सुश्च यमौ पार्षत एव च।</div><div class="line" style="position:relative;">चेदिकारूशमात्स्यानां केकयानां च यद्बलं॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1198"><div class="verse-block verse-justify"><div class="line">चेकितानश्च बलवान्धर्मराजश्च सुव्रतः।</div><div class="line" style="position:relative;">एते रथाश्वद्विरदैः पत्तिभिश्चोग्रविक्रमैः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1199"><div class="verse-block verse-justify"><div class="line">परिवार्य रणे कर्णं नानाशस्त्रैरवाकिरन्।</div><div class="line" style="position:relative;">भाषन्तो वाग्भिरुग्राभिः सर्वे कर्णवधे धृताः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1200"><div class="verse-block verse-justify"><div class="line">तां शस्त्रवृष्टिं बहुधा कर्णश्छित्त्वा शितैः शरैः।</div><div class="line" style="position:relative;">अपोवाहास्त्रवीर्येण द्रुमं भङ्क्त्वेव मारुतः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1201"><div class="verse-block verse-justify"><div class="line">रथिनः समहामात्रान्गजानश्वान्ससादिनः।</div><div class="line" style="position:relative;">पत्तिव्रातांश्च सङ्क्रुद्वो निघ्नन्कर्णो व्यदृश्यत॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1202"><div class="verse-block verse-justify"><div class="line">तद्वध्यमानं पाण्डूनां बलं कर्णास्त्रतेजसा।</div><div class="line" style="position:relative;">विशस्त्रपत्रदेहासु प्राय आसीत्पराङ्मुखम्॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1203"><div class="verse-block verse-justify"><div class="line">अथ कर्णास्त्रमस्त्रेण प्रतिहत्यार्जुनः स्मयन्।</div><div class="line" style="position:relative;">दिशं खं चैव भूमिं च प्रावृणोच्छरवृष्टिभिः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1204"><div class="verse-block verse-justify"><div class="line">मुसलानीव सम्पेतुः परिघा इव चेषवः।</div><div class="line" style="position:relative;">शतघ्न्य इव चाप्यन्ये वज्राण्युग्राणि चापरे॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1205"><div class="verse-block verse-justify"><div class="line">तैर्वध्यमानं तत्सैन्यं सपत्त्यश्वरथद्विपम्।</div><div class="line" style="position:relative;">निमीलिताक्षमत्यर्थं बभ्राम च ननाद च॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1206"><div class="verse-block verse-justify"><div class="line">निष्कैवल्यं तदा युद्वं प्रापुरश्वनरद्विपाः।</div><div class="line" style="position:relative;">हन्यमानाः शरैरार्तास्तदा भीताः प्रदुद्रुवुः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1207"><div class="verse-block verse-justify"><div class="line">त्वदीयानां तदा युद्वे संसक्तानां जयैषिणाम्।</div><div class="line" style="position:relative;">गिरिमस्तं समासाद्य प्रत्यपद्यत भानुमान्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1208"><div class="verse-block verse-justify"><div class="line">तमसा च महाराज रजसा च विशेषतः।</div><div class="line" style="position:relative;">न किञ्चित्प्रत्यपश्याम शुभं वा यदि वाऽशुभम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1209"><div class="verse-block verse-justify"><div class="line">ते त्रस्यन्तो महेष्वासा रात्रियुद्वस्य भारत।</div><div class="line" style="position:relative;">अपयानं ततश्चक्रुः सहिताः सर्वयोधिभिः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1210"><div class="verse-block verse-justify"><div class="line">कौरवेष्वपयातेषु तदा राजन्दिनक्षये।</div><div class="line" style="position:relative;">जयं सुमनसः प्राप्य पार्थाः स्वशिबिरं ययुः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1211"><div class="verse-block verse-justify"><div class="line">वादित्रशब्दैर्विविधैः सिंहनादैः सगर्जितैः।</div><div class="line" style="position:relative;">परानुवहन्तश्च स्तुवन्तश्चाच्युतार्जुनौ॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1212"><div class="verse-block verse-justify"><div class="line">कृतेऽवहारे तैर्वीरैः सैनिकाः सर्व एव ते।</div><div class="line" style="position:relative;">आशीर्वाचः पाण्डवेषु प्रायुञ्जन्त नरेश्वराः॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1213"><div class="verse-block verse-justify"><div class="line">ततः कृतेऽवहारे च प्रहृष्टास्तत्र पाण्डवाः।</div><div class="line" style="position:relative;">निशायां शिबिरं गत्वा न्यवसन्त नरेश्वराः॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b1214"><div class="verse-block verse-justify"><div class="line">ततो रक्षः पिशाचाश्च श्वापदाश्चैव सङ्घशः।</div><div class="line" style="position:relative;">जग्मुरायोधनं घोरं रुद्रस्याक्रीडसन्निभम्॥४५॥</div></div></div></div>
