<div class="verse-block-wrapper" id="08-karṇaparva-b2195"><div class="verse-block verse-justify"><div class="line">चिरकालेप्सितमिदमद्येदानीं भविष्यति।</div><div class="line" style="position:relative;">अद्य द्रक्ष्याम सङ्ग्रामं घोरं देवासुरोपमम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2196"><div class="verse-block verse-justify"><div class="line">अद्येदानीं महद्युद्धं भविष्यति भयानकम्।</div><div class="line" style="position:relative;">अद्येदानीं जयो नित्यमेकस्यैकस्य वा रणे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2197"><div class="verse-block verse-justify"><div class="line">अर्जुनं किल राधेयो वधिष्यति महारणे।</div><div class="line" style="position:relative;">अथवा कं नरं लोके न स्पृशन्ति मनोरथाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2198"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा विविधा वाचः कुरवः कुरुनन्दन।</div><div class="line" style="position:relative;">आजघ्नुः पटहांश्चैव तूर्यांश्चैव सहस्रशः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2199"><div class="verse-block verse-justify"><div class="line">भेरीनादांश्च विविधान्सिंहनादांश्च पुष्कलान््।</div><div class="line" style="position:relative;">मुरजानां महाशब्दानानकानां महारवान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2200"><div class="verse-block verse-justify"><div class="line">नृत्यमानाश्च बहुशस्तर्जमानाश्च मारिष।</div><div class="line" style="position:relative;">अन्योन्यमभ्ययुर्युद्धे युद्धरङ्गगता नराः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2201"><div class="verse-block verse-justify"><div class="line">तेषां पदाता नागानां पादरक्षाः समन्ततः।</div><div class="line" style="position:relative;">पट्टसासिधराः शूराश्चापबाणमुसुण्ठिनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2202"><div class="verse-block verse-justify"><div class="line">भिण्डिपालधराश्चैव शूरलहस्ताः सुचक्रिणः।</div><div class="line">तेषां समागमो घोरो देवासुररणोपमः ॥धृतराष्ट्रं उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2203"><div class="verse-block verse-justify"><div class="line">कथं सञ्जय राधेयः प्रत्यव्यूहत पाण्डवान्।</div><div class="line">धृष्टद्युम्नमुखान्सङ्ख्ये भीमसेनाभिरक्षितान्।</div><div class="line" style="position:relative;">सर्वानेव महेष्वासानजय्यानमरैरपि॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2204"><div class="verse-block verse-justify"><div class="line">के च प्रपक्षौ पक्षौ वा मम सैन्यस्य सञ्जय।</div><div class="line" style="position:relative;">प्रविभज्य यथान्यायमवतस्थुः सुदंशिताः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2205"><div class="verse-block verse-justify"><div class="line">कथं पाण्डुसुताश्चापि प्रत्यव्यूहन्त मामकान्।</div><div class="line" style="position:relative;">कथं चैतन्महद्युद्धं प्रावर्तत सुदारुणम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2206"><div class="verse-block verse-justify"><div class="line">क्व च बीभत्सुरभवद्यत्कर्णोऽयाद्युधिष्ठिरम्।</div><div class="line" style="position:relative;">को ह्यर्जुनस्य सान्निध्ये शक्तो जेतुं युधिष्ठिरम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2207"><div class="verse-block verse-justify"><div class="line">सर्वभूतानि यो ह्येकः खाण्डवे जितबान्पुरा।</div><div class="line">कस्तमन्यस्तु राधेयात्प्रतियुध्येज्जिजीविषुः ॥सञ्जय उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2208"><div class="verse-block verse-justify"><div class="line">शृणु व्यूहस्य रचनामर्जुनश्च यथा गतः।</div><div class="line" style="position:relative;">परिवार्य नृपान्सर्वान्सङ्ग्रामश्चाभवद्यथा॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2209"><div class="verse-block verse-justify"><div class="line">कृपः शारद्वतो राजन्मागधाश्च तरस्विनः।</div><div class="line" style="position:relative;">सात्वतः कृतवर्मा च दक्षिणं पक्षमाश्रिताः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2210"><div class="verse-block verse-justify"><div class="line">तेषां प्रपक्षे शकुनिरुलूकश्च महारथः।</div><div class="line" style="position:relative;">सादिभिर्विमलप्रासैस्तदानीमभ्यरक्षताम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2211"><div class="verse-block verse-justify"><div class="line">गान्धारैश्चाप्यसम्भ्रान्तैः पार्वतीयैश्च दुर्जयैः।</div><div class="line" style="position:relative;">शलभानामिव व्रातैः पिशाचैरिव दुर्दृशैः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2212"><div class="verse-block verse-justify"><div class="line">चतुर्विंशत्सहस्राणि रथानामनिवर्तिनाम्।</div><div class="line" style="position:relative;">संशप्तका युद्वशौण्डा वामपाश्वमपालयन्॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2213"><div class="verse-block verse-justify"><div class="line">समन्वितास्तव सुतैः कृष्णार्जुनजिघांसवः।</div><div class="line" style="position:relative;">तेषां प्रपक्षाः काम्भोजाः शकाश्च यवनैः सह॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2214"><div class="verse-block verse-justify"><div class="line">निदेशात्सूतपुत्रस्य सरथाश्वेभपत्तयः।</div><div class="line" style="position:relative;">आह्वयन्तोऽर्जुनं तस्थुः केशवं च महाबलम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2215"><div class="verse-block verse-justify"><div class="line">ततः सेनामुखे कर्णोऽप्यवातिष्ठत दंशितः।</div><div class="line" style="position:relative;">चित्रवर्माङ्गदः स्रग्वी पालयन्वाहिनीमुखम्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2216"><div class="verse-block verse-justify"><div class="line">रक्षमाणैः सुसंरब्धैः पुत्रैः शस्त्रभृतां वरः।</div><div class="line" style="position:relative;">वाहिनीं प्रमुखे वीरः सम्प्रकर्षन्नशोभत॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2217"><div class="verse-block verse-justify"><div class="line">अशोकस्तु महाबाहुः सूर्यवैश्वानरप्रभः।</div><div class="line">महाद्विपस्कन्धगतः पिङ्गाक्षः प्रियदर्शनः।</div><div class="line" style="position:relative;">दुःशासनो वृतः सैन्यैः स्थितो व्यूहस्य पृष्ठतः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2218"><div class="verse-block verse-justify"><div class="line">तन्मध्ये स महाबाहुः स्वयं दुर्योधनो नृपः।</div><div class="line" style="position:relative;">चित्राश्वैश्चित्रसन्नाहैः सोदर्यैरभिरक्षितः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2219"><div class="verse-block verse-justify"><div class="line">रक्ष्यमाणो महावीर्यैः सहितैर्मद्रकेकयैः।</div><div class="line" style="position:relative;">अशोभत महाराज देवैरिव शतक्रतुः॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="08-karṇaparva-b2220"><div class="verse-block verse-justify"><div class="line">अश्वत्थामा कुरूणां च ये प्रवीरा महारथाः।</div><div class="line">नित्यमत्ताश्च मातङ्गाः शूरैर्म्लेच्छैः समन्विताः।</div><div class="line" style="position:relative;">अन्वयुस्तद्रथानीकं क्षरन्त इव तोयदाः॥३६॥</div></div></div>
