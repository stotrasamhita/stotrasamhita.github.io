<div class="verse-block-wrapper" id="03-araṇyaparva-b9070"><div class="verse-block verse-justify"><div class="line">प्रजासु तासु सर्वासु रूपेणाप्रतिमाऽभवत्।</div><div class="line" style="position:relative;">देवी भानुमती नाम प्रथमाऽङ्गिरसः सुता॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9071"><div class="verse-block verse-justify"><div class="line">भूतानामेव सर्वेषां यस्यां रागस्तदाऽभवत्।</div><div class="line" style="position:relative;">रागाद्रागेति यामाहुर्द्वितीयाऽङ्गिरसः सुता॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9072"><div class="verse-block verse-justify"><div class="line">यां कपर्दिसुतामाहुर्दृश्यादृश्येति देहिनः।</div><div class="line" style="position:relative;">तनुत्वात्सा सिनीवाली तुतीयाऽङ्गिरसः सुता॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9073"><div class="verse-block verse-justify"><div class="line">यां तु दृष्ट्वा भगवतीं जनः कुहकुहायते।</div><div class="line" style="position:relative;">एकानकेति यामाहुश्चतुर्थ्यङ्गिरसः सुता॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9074"><div class="verse-block verse-justify"><div class="line">पञ्चम्यर्चिष्मती नाम्ना हविर्भिश्च हविष्मती।</div><div class="line" style="position:relative;">षष्ठीम्गिरसः कन्यां पुण्यामाहुर्महिष्मतीम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9075"><div class="verse-block verse-justify"><div class="line">महामखेष्वाङ्गिरसी दीप्तिमत्सु महीयती।</div><div class="line" style="position:relative;">महामतीति विख्याता सप्तमी कथ्यते सुता॥९॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b9077"><div class="verse-block verse-justify"><div class="line">आहुतिष्वेव यस्याग्नेर्हविराज्यं विधीयते।</div><div class="line" style="position:relative;">सोग्निर्बृहस्पतेः पुत्र शंयुर्नाम महाप्रभः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9078"><div class="verse-block verse-justify"><div class="line">चातुर्मास्येषु यस्येष्टमश्वमेधाग्रभागभूत्।</div><div class="line" style="position:relative;">दीप्तिज्वालैरनेकाग्रैरग्निष्टोमोऽथ वीर्यवान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9079"><div class="verse-block verse-justify"><div class="line">शंयोरप्रतिमा भार्या सत्या सत्याऽथ धर्मजा।</div><div class="line" style="position:relative;">अग्नितस्य सुतो दीप्तस्तिस्रः कन्याश्च सुव्रताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9080"><div class="verse-block verse-justify"><div class="line">प्रथमेनाज्यभागेन पूज्यते योऽग्निरध्वरे।</div><div class="line" style="position:relative;">अग्निस्तस्य भरद्वाजः प्रथमः पुत्र उच्यते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9081"><div class="verse-block verse-justify"><div class="line">पौर्णमासेषु सर्वेषु हविराज्यं ध्रुवोद्यतम्।</div><div class="line" style="position:relative;">भरतो नामतः सोग्निर्द्वितीयः शंयुतः सुतः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9082"><div class="verse-block verse-justify"><div class="line">तिस्रः कन्या भवन्त्यन्या यासां स भरत पतिः।</div><div class="line" style="position:relative;">भारतस्तु सुतस्तस्य भारत्येका च पुत्रिका॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9083"><div class="verse-block verse-justify"><div class="line">भारतो भरतस्याग्नेः पावकस्तु प्रजायते।</div><div class="line" style="position:relative;">महानत्यर्थमहितस्तथा भरतसत्तम॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b9085"><div class="verse-block verse-justify"><div class="line">हविषा यो द्वितीयेन सोमेन सह युज्यते।</div><div class="line" style="position:relative;">रथप्रभू रथध्वानः कुम्भरेताः स उच्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9086"><div class="verse-block verse-justify"><div class="line">सरय्वां जनयत्सिद्धिं भानुं भाभिः समावृणोत्।</div><div class="line" style="position:relative;">आग्नेयमनयन्मिथ्यं मिथ्यो नामैव कथ्यते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9087"><div class="verse-block verse-justify"><div class="line">यस्तु न च्यवते नित्यं शयसा वर्चसा श्रिया।</div><div class="line" style="position:relative;">अग्निर्निश्च्यवनो नाम पृथिवीं स्तौति केवलम्॥२१॥</div></div></div>
