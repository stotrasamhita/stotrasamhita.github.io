<div class="verse-block-wrapper" id="03-araṇyaparva-b4669"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वान्तरिक्षाच्च स राजा राजसत्तमः।</div><div class="line" style="position:relative;">यथोक्तं तच्चराकाथ श्रद्दधद्भरतर्षभ॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4670"><div class="verse-block verse-justify"><div class="line">[एकैकशस्ततः कृत्वा बीजं बीजं नराधिपः।</div><div class="line" style="position:relative;">घृतपूर्णेषु कुम्भेषु तान्भागान्विदधे ततः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4671"><div class="verse-block verse-justify"><div class="line">धात्रीश्चैकैकशः प्रादात्पुत्ररक्षणतत्परः।</div><div class="line" style="position:relative;">ततः कालेन महता समुत्तस्थुर्महाबलाः ॥]॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4672"><div class="verse-block verse-justify"><div class="line">षष्टिः पुत्रसहस्रणि तस्याप्रतिमतेजसः।</div><div class="line" style="position:relative;">रुद्रप्रसादाद्राजर्षेः समजायन्त पार्थिव॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4673"><div class="verse-block verse-justify"><div class="line">ते घोराः क्रूरकर्माण आकाशपरिसर्पिणः।</div><div class="line" style="position:relative;">बहुत्वाच्चावजानन्तः सर्वाल्लोँकान्सहामरान्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4674"><div class="verse-block verse-justify"><div class="line">त्रिदशांश्चाप्यबाधन्त तथा गन्धर्वराक्षसान्।</div><div class="line" style="position:relative;">सर्वाणि चैव भूतानि शूरा समरशालिनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4675"><div class="verse-block verse-justify"><div class="line">वध्यमानास्तदा लोकाः सागरैर्मन्दबुद्धिभिः।</div><div class="line" style="position:relative;">ब्रह्माणं शरणं जग्मुः सहिताः सर्वदैवतैः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4676"><div class="verse-block verse-justify"><div class="line">तानुवाच महाभागः सर्वलोकपितामहः।</div><div class="line" style="position:relative;">गच्छध्वं त्रिदशाः सर्वे लोकैः सार्धं यथागतम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4677"><div class="verse-block verse-justify"><div class="line">नातिदीर्घेण कालेन सागराणां क्षयो महान्।</div><div class="line" style="position:relative;">विष्यति महाघोरः स्वकृतैः कर्मभिः सुराः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4678"><div class="verse-block verse-justify"><div class="line">एवमुक्तास्तु ते देवा लोकाश् मनुजेश्वर।</div><div class="line" style="position:relative;">पितामहमनुज्ञाप्य विप्रजग्मुर्यथागतम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4679"><div class="verse-block verse-justify"><div class="line">ततः काले बहुतिथे व्यतीते भरतर्षभ।</div><div class="line" style="position:relative;">दीक्षितः सगरो राजा हयमेधेन वीर्यवान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4680"><div class="verse-block verse-justify"><div class="line">तस्याश्वो व्यचरद्भूमिं पुत्रैः सुपरिरक्षितः।</div><div class="line" style="position:relative;">`सर्वैरेव महोत्साहैः स्वच्छन्दप्रचरो नृप'॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4681"><div class="verse-block verse-justify"><div class="line">समुद्रं स समासाद्य निस्तोयं भीमदर्शनम्।</div><div class="line" style="position:relative;">रक्ष्यमाणः प्रयत्नेन तत्रैवान्तरधीयत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4682"><div class="verse-block verse-justify"><div class="line">ततस्ते सागरास्तात हृतं मत्वा हयोत्तमम्।</div><div class="line">आगम्य पितुराचख्युरदृश्यं तुरगं हृतम्।</div><div class="line" style="position:relative;">तेनोक्ता दिक्षु सर्वासु पुत्रा मार्गत वाजिनम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4683"><div class="verse-block verse-justify"><div class="line">ततस्ते पितुराज्ञाय दिक्षु सर्वासु तं हयम्।</div><div class="line" style="position:relative;">अमार्गन्त महाराज सर्वंच पृथिवीतलम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4684"><div class="verse-block verse-justify"><div class="line">ततस्ते सागराः सर्वे समुपेत्य परस्परम्।</div><div class="line" style="position:relative;">नाध्यगच्छन्त तुरगमश्वहर्तारमेव च॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b4686"><div class="verse-block verse-justify"><div class="line">ससमुद्रवनद्वीपा सनदीनदकन्दरा।</div><div class="line" style="position:relative;">सपर्वतवनोद्देशा निखिलेन मही नृप॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4687"><div class="verse-block verse-justify"><div class="line">अस्माभिर्विचिता राजञ्शासनात्तव पार्थिव।</div><div class="line" style="position:relative;">न चाश्वमधिगच्छामो नाश्वहर्तारमेव च॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4688"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु वचनं तेषां स राजा क्रोधमूर्चितः।</div><div class="line" style="position:relative;">उवाच वचनं सर्वांस्तदा दैववशान्नृप॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4689"><div class="verse-block verse-justify"><div class="line">अनागमाय गच्छध्वं भूयो मार्गत वाजिनम्।</div><div class="line" style="position:relative;">याज्ञीयं तं विनाह्यश्वं नागन्तव्यं हि पुत्रकाः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4690"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्य तु संदेशं पितुस्ते सगरात्मजाः।</div><div class="line" style="position:relative;">भूय एव महीं कृत्स्नां विचेतुमुपचक्रमुः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4691"><div class="verse-block verse-justify"><div class="line">अथापश्यन्त ते वीराः पृथिवीमवदारिताम्।</div><div class="line" style="position:relative;">`समुद्रे पृथिवीपाल पदं मार्गं च वाजिनः'॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4692"><div class="verse-block verse-justify"><div class="line">समासाद्य बिलं तच्चाप्यखनन्सगरात्मजाः।</div><div class="line" style="position:relative;">कुद्दालैर्मुसलैश्चैव समुद्रं यत्नमास्थिताः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4693"><div class="verse-block verse-justify"><div class="line">स खन्यमानः सहितैः सागरैर्वरुणालयः।</div><div class="line" style="position:relative;">अगच्छत्परमामार्तिं दीर्यमाणः समन्ततः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4694"><div class="verse-block verse-justify"><div class="line">असुरोरगरक्षांसि सत्वानि विविधानि च।</div><div class="line" style="position:relative;">आर्तनादमकुर्वन्त वध्यमानानि सागरैः॥२६॥</div></div></div>
