<div class="verse-block-wrapper" id="03-araṇyaparva-b4122"><div class="verse-block verse-justify"><div class="line">धौम्य उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4123"><div class="verse-block verse-justify"><div class="line">अवन्तीषु प्रतीच्यां वै कीर्तयिष्यामि ते दिशि।</div><div class="line" style="position:relative;">यानि तत्रपवित्राणि पुण्यान्यायतनानि च॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4124"><div class="verse-block verse-justify"><div class="line">प्रियङ्ग्वाम्रवणोपेता वानीरफलमालिनी।</div><div class="line" style="position:relative;">प्रत्यक्स्रोता नदी पुण्या नर्मदा तत्र भारत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4125"><div class="verse-block verse-justify"><div class="line">त्रैलोक्ये यानि तीर्थानि पुण्यान्यायतनानि च।</div><div class="line" style="position:relative;">सरिद्वनानि शैलेन्द्रा देवाश्च सपितामहाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4126"><div class="verse-block verse-justify"><div class="line">नर्मदायां कुरुश्रेष्ठ सहसिद्धर्षिचारणैः।</div><div class="line" style="position:relative;">स्नातुमायान्ति पुण्यौधैः सदा वारिषु भारत॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4127"><div class="verse-block verse-justify"><div class="line">निरेतः श्रूयते पुण्यो यत्र विश्रवसो मुनेः।</div><div class="line" style="position:relative;">जज्ञे धनपतिर्यत्र कुबेरो नरवाहनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4128"><div class="verse-block verse-justify"><div class="line">वैढूर्यशिखरो नाम पुण्यो गिरिवरः शिवः।</div><div class="line" style="position:relative;">नित्यपुष्पफलास्तत्र पादपा हरितच्छदाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4129"><div class="verse-block verse-justify"><div class="line">तस्य शैलस्य शिखरे सरः पुण्यं महीपते।</div><div class="line" style="position:relative;">फुल्लपद्मं महाराज देवगन्धर्वसेवितम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4130"><div class="verse-block verse-justify"><div class="line">बह्वाश्चर्यं महाराज दृश्यते तत्र पर्वते।</div><div class="line" style="position:relative;">पुण्ये स्वर्गोपमे चैव देवर्षिगणसेविते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4131"><div class="verse-block verse-justify"><div class="line">ह्रदिनी पुण्यतीर्था च राजर्षेस्तत्र वै सरित्।</div><div class="line" style="position:relative;">विश्चामित्रेण तपसा निर्मिता सर्वपावनी॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4132"><div class="verse-block verse-justify"><div class="line">यस्यास्तीरे सतां मध्ये ययातिर्नहुषात्मजः।</div><div class="line" style="position:relative;">पपात स पुनर्लोकाँल्लेभे धर्मान्सनातनान्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4133"><div class="verse-block verse-justify"><div class="line">तत्रपुण्यो ह्रदः ख्यातो मैनाकश्चैव पर्वतः।</div><div class="line" style="position:relative;">बहुमूलफलोपेतस्त्वमितो नाम पर्वतः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4134"><div class="verse-block verse-justify"><div class="line">आश्रमः कक्षसेनस्य पुण्यस्तत्रयुधिष्ठिरः।</div><div class="line">च्यवनस्याश्रमश्चैव विख्यातस्तत्रपाण्डव।</div><div class="line" style="position:relative;">तत्राल्पेनैव सिद्ध्यन्ति मानवास्तपसा विभो॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4135"><div class="verse-block verse-justify"><div class="line">जम्बूमार्गो महाराज ऋषीणां भावितात्मनाम्।</div><div class="line" style="position:relative;">आश्रम शाम्यतां श्रेष्ठ मृगद्विजनिषेवितः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4136"><div class="verse-block verse-justify"><div class="line">ततः पुण्यतमा राजन्सततं तापसैर्युता।</div><div class="line">केतुमाला च मेध्या च गङ्गाद्वारं च भूमिप।</div><div class="line" style="position:relative;">ख्यातं च सैन्धवारण्यं पुण्यं द्विजनिषेवितम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4137"><div class="verse-block verse-justify"><div class="line">पितामहसरः पुण्यं पुष्करं नाम नामतः।</div><div class="line" style="position:relative;">वैखानसानां सिद्धानामृषीणामाश्रमः प्रियः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4138"><div class="verse-block verse-justify"><div class="line">अप्यत्र संश्रयार्थाय प्रजापतिरथो जगौ।</div><div class="line" style="position:relative;">पुष्करेषु कुरुश्रेष्ठ गाथां सुकृतिनांवर॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4139"><div class="verse-block verse-justify"><div class="line">मनसाऽप्यभिकामस्य पुष्कराणि मनखिनः।</div><div class="line" style="position:relative;">विप्रणश्यन्ति पापानि नाकपृष्ठे च मोदते॥१८॥</div></div></div></div>
