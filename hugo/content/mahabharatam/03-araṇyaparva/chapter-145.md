<div class="verse-block-wrapper" id="03-araṇyaparva-b5862"><div class="verse-block verse-justify"><div class="line">परिगृह्य द्विजश्रेष्ठाञ्ज्येष्ठाः सर्वधनुष्मताम्।</div><div class="line" style="position:relative;">पञ्चालीसहिता राजन्प्रययुर्गन्धमादनम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5863"><div class="verse-block verse-justify"><div class="line">सरांसि सरितश्चैव पर्वतांश्च वनानि च।</div><div class="line" style="position:relative;">वृक्षांश्च बहुलच्छायान्ददृशुर्गिरिमूर्धनि॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5864"><div class="verse-block verse-justify"><div class="line">नित्यपुष्पफलान्देशान्देवर्षिगणसेवितान्।</div><div class="line" style="position:relative;">आत्मन्यात्मानमाधाय वीरा मूलफलाशिनः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5865"><div class="verse-block verse-justify"><div class="line">चेरुरुच्चावचाकारान्देशान्विषमसंकटान्।</div><div class="line" style="position:relative;">पश्यन्तो मृगजातानि बहूनि विविधानि च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5866"><div class="verse-block verse-justify"><div class="line">ऋषिसिद्धामरयुतं गन्धर्वाप्सरसां प्रियम्।</div><div class="line" style="position:relative;">विविशुस्ते महात्मानः किन्नराचरितं गिरिम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5867"><div class="verse-block verse-justify"><div class="line">प्रविशत्स्वथ वीरेषु पर्वतं गन्धमादनम्।</div><div class="line" style="position:relative;">चणअडवातं महद्वर्षं प्रादुरासीद्विशांपते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5868"><div class="verse-block verse-justify"><div class="line">ततो रेणुः समुद्भूतः सपत्रबहुलो महान्।</div><div class="line" style="position:relative;">पृथिवीं चान्तरिक्षं च द्यां चैव सहसाऽवृणोत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5869"><div class="verse-block verse-justify"><div class="line">न स्म प्रज्ञायते किंचिदावृते व्योम्नि रेणुना।</div><div class="line" style="position:relative;">न चापि शेकुस्तत्कर्तुमन्योन्यस्याभिभाषणम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5870"><div class="verse-block verse-justify"><div class="line">न चापश्यंस्ततोऽन्योन्यं तमसाऽऽवृतचक्षुषः।</div><div class="line" style="position:relative;">आकृष्यमाणा वातेन साश्मचूर्णेन भारत॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5871"><div class="verse-block verse-justify"><div class="line">द्रुमाणां वातरुग्णआनां पततां भूतलेऽनिशम्।</div><div class="line" style="position:relative;">अन्येषां च महीजानां शब्दः समभवन्महान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5872"><div class="verse-block verse-justify"><div class="line">द्यौः स्वित्पतति किं भूमिर्दीर्यते पर्वतोनु किम्।</div><div class="line" style="position:relative;">इति ते मेनिरे सर्वेपवनेन विमोहिताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5873"><div class="verse-block verse-justify"><div class="line">ते पथाऽनन्तरांन्वृक्षान्वल्मीकान्विषमाणि च।</div><div class="line" style="position:relative;">पाणिभिः परिमार्गन्तो भीता वायोर्निलिल्यिरे॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5874"><div class="verse-block verse-justify"><div class="line">ततः कार्मुकमादाय भीमसेनो महाबलः।</div><div class="line" style="position:relative;">कृष्णामादाय संगम्य तस्थावाश्रित्य पादपम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5875"><div class="verse-block verse-justify"><div class="line">धर्मराजश्च धौम्यश् निलिल्याते महावने।</div><div class="line" style="position:relative;">अग्निहोत्राण्युपादाय सहदेवस्तु पर्वते॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5876"><div class="verse-block verse-justify"><div class="line">नकुलो ब्राह्मणाश्चान्ये लोमशश्च महातपाः।</div><div class="line" style="position:relative;">वृक्षानासाद्य संत्रस्तास्तत्रतत्र निलिल्यिरे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5877"><div class="verse-block verse-justify"><div class="line">मन्दीभूते तु पवने तस्मिन्रजसि शाम्यति।</div><div class="line" style="position:relative;">महद्भिर्जलधारौर्घर्वर्षमभ्याजगाम ह॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5878"><div class="verse-block verse-justify"><div class="line">भृशं चटचटाशब्दो वज्राणां क्षिप्यतामिव।</div><div class="line" style="position:relative;">ततस्ताश्चञ्चलाभासश्चेरुरभ्रेषु विद्युतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5879"><div class="verse-block verse-justify"><div class="line">ततोऽश्मसहिता धाराः संवृण्वन्त्यः समन्ततः।</div><div class="line" style="position:relative;">प्रपेतुरनिशं तत्र शीघ्रवातसमीरिताः॥२०॥</div></div></div>
