<div class="verse-block-wrapper" id="03-araṇyaparva-b5167"><div class="verse-block verse-justify"><div class="line">लोमश उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5168"><div class="verse-block verse-justify"><div class="line">कस्य चित्त्वथ कालस्य त्रिदशावश्विनौ नृप।</div><div class="line" style="position:relative;">कृताभिषेकां विवृतां सुकन्यां तामपश्यताम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5169"><div class="verse-block verse-justify"><div class="line">तां दृष्ट्वा दर्शनीयाङ्गीं देवराजसुतामिव।</div><div class="line" style="position:relative;">ऊचतुः समभिद्रुत्य नासत्यावश्विनाविदम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5170"><div class="verse-block verse-justify"><div class="line">कस्य त्वमसि वामोरु वनेऽस्मिन्किं करोषि च।</div><div class="line" style="position:relative;">इच्छाव भद्रे ज्ञातुं त्वां तत्वमाख्याहि शोभने॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5171"><div class="verse-block verse-justify"><div class="line">ततः सुकन्या संवीता तावुवाच सुरोत्तमौ।</div><div class="line" style="position:relative;">शर्यातितनयां वित्तं भार्यां मां च्यवनस्य च॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5172"><div class="verse-block verse-justify"><div class="line">`नाम्ना चाहं सुकन्येति नृलोकेऽस्मिन्प्रतिष्ठिता।</div><div class="line" style="position:relative;">साऽहंसर्वात्मना नित्यं भर्तारमनुवर्तिनी'॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5173"><div class="verse-block verse-justify"><div class="line">अथाश्विनौ प्रहस्यैतामब्रूतां पुनरेव तु।</div><div class="line" style="position:relative;">कथं त्वमसि कल्याणि पित्रा दत्ताऽऽगता वने॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5174"><div class="verse-block verse-justify"><div class="line">भ्राजसेऽस्मिन्वने भीरु विद्युत्सौदामनी यथा।</div><div class="line" style="position:relative;">न देवेष्वपि तुल्यां हि त्वया पश्याव भामिनि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5175"><div class="verse-block verse-justify"><div class="line">अनाभरणसंपन्ना परमाम्बरवर्जिता।</div><div class="line" style="position:relative;">शोभयस्यधिकं भद्रे वनमप्यनलंकृता॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5176"><div class="verse-block verse-justify"><div class="line">सर्वाभरणसंपनना परमाम्वरधारिणी।</div><div class="line" style="position:relative;">शोभेथास्त्वनवद्याङ्गि न त्वेवं मलपङ्किनी॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5177"><div class="verse-block verse-justify"><div class="line">कस्मादेवंविधा भूत्वा जराजर्जरितं पतिम्।</div><div class="line" style="position:relative;">त्वमुपास्से ह कल्याणि कामभोगबहिष्कृतम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5178"><div class="verse-block verse-justify"><div class="line">असमर्थं परित्राणे पोषणे तु शुचिस्मिते।</div><div class="line" style="position:relative;">सा त्वं च्यवनमुत्सृज्यवरयस्वैकमावयोः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5179"><div class="verse-block verse-justify"><div class="line">पत्यर्थं देवगर्भाभे मा वृथा यौवनं कृथाः।</div><div class="line" style="position:relative;">एवमुक्ता सुकन्याऽपि सुरौ ताविदमब्रवीत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5180"><div class="verse-block verse-justify"><div class="line">रताऽहं च्यवने पत्यौ मैवं मां पर्यशङ्कतम्।</div><div class="line" style="position:relative;">तावब्रूतां पुनस्त्वेनामावां देवभिषग्वरौ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5181"><div class="verse-block verse-justify"><div class="line">युवानं रूपसंपन्नं करिष्यावः पतिं तव।</div><div class="line">ततस्तस्यावयोश्चैव वृणीष्वान्यतमं पतिम्।</div><div class="line" style="position:relative;">एतेन समयेनैनमामन्त्रय पतिं शुभे॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5182"><div class="verse-block verse-justify"><div class="line">सा तयोर्वचनाद्राजन्नुपसंगम्य भार्गवम्।</div><div class="line" style="position:relative;">उवाच वाक्यं यत्ताभ्यामुक्तं भृगुसुतं प्रति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5183"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा च्यवनो भार्यामुवाच क्रियतामिति।</div><div class="line" style="position:relative;">`सा भर्त्रा समनुज्ञाता क्रियतामित्यथाब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5184"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तदश्विनौ वाक्यं तस्यास्तत्क्रियतामिति'।</div><div class="line" style="position:relative;">ऊचतू राजपुत्रीं तां पतिस्तव विशत्वपः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5185"><div class="verse-block verse-justify"><div class="line">ततोऽम्भश्च्यवनः शीघ्रं रूपार्थी प्रविवेश ह।</div><div class="line" style="position:relative;">अश्विनावपि तद्राजन्सरः प्राविशतां तदा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5186"><div class="verse-block verse-justify"><div class="line">ततो मुहूर्तादुत्तीर्णाः सर्वे ते सरसस्त्रयः।</div><div class="line">दिव्यरूपधराः सर्वे युवानो मृष्टकुण्डलाः।</div><div class="line" style="position:relative;">तुल्यवेषधराश्चैव मनसः प्रीतिवर्धनाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5187"><div class="verse-block verse-justify"><div class="line">तेऽब्रुवन्सहिताः सर्वे वृणीष्वान्यतमं शुभे।</div><div class="line" style="position:relative;">अस्माकमीप्सितं भद्रे पतित्वे वरवर्णिनि॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5188"><div class="verse-block verse-justify"><div class="line">`त्वमश्विनोरन्यतरं च्यवनं वा यशस्विनि'।</div><div class="line" style="position:relative;">यत्र वाऽप्यभिकामाऽसि तं वृणीष्व सुशोभने॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5189"><div class="verse-block verse-justify"><div class="line">सा समीक्ष्य तु तान्सर्वांस्तुल्यरूपधरान्स्थितान्।</div><div class="line" style="position:relative;">निश्चित्य मनसा बुद्ध्या देवी वव्रे स्वकं पतिम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5190"><div class="verse-block verse-justify"><div class="line">लब्ध्वा तु च्यवनो भार्यांवयोरूपं च वाञ्छितम्।</div><div class="line" style="position:relative;">हृष्टोऽब्रवीन्महातेजास्तौ नासत्याविदं वचः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5191"><div class="verse-block verse-justify"><div class="line">यथाऽहं रूपसंपन्नो वयसा च समन्वितः।</div><div class="line" style="position:relative;">कृतो भवद्भ्यां वृद्धः सन्भार्यां च प्राप्तवानिमाम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5192"><div class="verse-block verse-justify"><div class="line">तस्माद्युवां करिष्यामि प्रीत्याऽहं सोमपीथिनौ।</div><div class="line" style="position:relative;">मिषतो देवराजस्य सत्यमेतद्ब्रवीमि वाम्॥२६॥</div></div></div>
