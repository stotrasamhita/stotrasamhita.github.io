<div class="verse-block-wrapper" id="03-araṇyaparva-b7027"><div class="verse-block verse-justify"><div class="line">ततोऽहं स्तूयमानस्तु तत्रतत्र सुरपिभिः।</div><div class="line" style="position:relative;">अपश्यमुदधिं भीममपांपतिमथाव्ययम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7028"><div class="verse-block verse-justify"><div class="line">फेनवत्यः प्रकीर्णाश्च संहताश्च समुच्छिताः।</div><div class="line" style="position:relative;">ऊर्मयश्चात्र दृश्यन्ते चलन्तइव पर्वताः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7029"><div class="verse-block verse-justify"><div class="line">नावः सहस्रशस्तत्र रत्नपूर्णाः समन्ततः।</div><div class="line" style="position:relative;">`नभसीव विमानानि विचरन्त्यो विरेजिरे'॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7030"><div class="verse-block verse-justify"><div class="line">तिमिङ्गिलाः कच्छपाश् तथा तिमितिभिङ्गिलाः।</div><div class="line" style="position:relative;">मकराश्चात्र दृश्यन्ते जले मप्रा इवाद्रयः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7031"><div class="verse-block verse-justify"><div class="line">शङ्खानां च महबाणि मग्रान्यप्सु ममन्ततः।</div><div class="line" style="position:relative;">दृश्यन्ते स्म यथा रात्रौ तारास्तन्वभ्रमंवृताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7032"><div class="verse-block verse-justify"><div class="line">तथा सहस्रशस्तत्ररत्नसङ्घाः प्लवन्त्युत।</div><div class="line" style="position:relative;">वायुश् घृर्णते भीमस्तदद्भुतमिवाभवत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7033"><div class="verse-block verse-justify"><div class="line">तमतीत्य महावेगं सर्वाम्भोनिधिमुत्तमम्।</div><div class="line" style="position:relative;">अपश्यं दानवाकीर्णं तद्दैत्यपुरमन्तिकात्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7034"><div class="verse-block verse-justify"><div class="line">तत्रैव मातलिस्तृणं निपात्य पृथिवीतले।</div><div class="line" style="position:relative;">दानवान्रथघोपेण तन्पुरं समुपाद्रवत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7035"><div class="verse-block verse-justify"><div class="line">रथघोषं तु तं श्रुत्वा स्तनयित्नोग्विम्बरे।</div><div class="line" style="position:relative;">मन्वाना देवराजं मामाविग्ना दानवाऽभवन्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7036"><div class="verse-block verse-justify"><div class="line">सर्वे संभ्रान्तमनसः शग्चापधराः स्थिताः।</div><div class="line" style="position:relative;">तथाऽसिशूलपरशुगदामुसलपाणयः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7037"><div class="verse-block verse-justify"><div class="line">ततो द्वाराणि पिदधुर्दानवास्त्रस्तचेतसः।</div><div class="line" style="position:relative;">संविधाय पुरे रक्षां न स्म कश्चन दृश्यते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7038"><div class="verse-block verse-justify"><div class="line">ततः शङ्खमुपादाय देवदत्तं महास्वनम्।</div><div class="line" style="position:relative;">परमां मुदमाश्रित्य प्राधमं तं शनैरहम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7039"><div class="verse-block verse-justify"><div class="line">स तु शब्दो दिवं स्तब्ध्वा प्रतिशब्दमजीजनत्।</div><div class="line" style="position:relative;">वित्रेसुश्च निलिल्युश्च भूतानि सुमहान्त्यपि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7040"><div class="verse-block verse-justify"><div class="line">ततो निवातकवचाः सर्व एव समन्ततः।</div><div class="line" style="position:relative;">दंशिता विविधैस्त्राणैर्विचित्रायुपाणयः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7041"><div class="verse-block verse-justify"><div class="line">आयसैश्र महाशूलैर्गदाभिर्मुसलैरपि।</div><div class="line" style="position:relative;">पट्टसैः करवालैश्च रथचक्रैश्च भारत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7042"><div class="verse-block verse-justify"><div class="line">शतघ्नीभिर्भुशुण्डीभिः खङ्गैश्चित्रैः स्वलंकृतैः।</div><div class="line" style="position:relative;">प्रगृहीतैर्दितेः पुत्राः प्रादुरासन्सहस्रशः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7043"><div class="verse-block verse-justify"><div class="line">ततो विचार्य बहुशो रथमार्गेषु तान्हयान्।</div><div class="line" style="position:relative;">प्राचोदयत्समे देशे मातलिर्भरतर्षभ॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7044"><div class="verse-block verse-justify"><div class="line">तेन तेषां प्रणुन्नानासाशुत्वाच्छीघ्रगामिनाम्।</div><div class="line" style="position:relative;">नान्वपश्यत्तदा कश्रित्तन्मेऽद्भुतमिवाभवत्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7045"><div class="verse-block verse-justify"><div class="line">ततस्ते दानवास्तत्रयोधवातान्यनेकशः।</div><div class="line" style="position:relative;">विकृतस्वररूपाणि भृशं सर्वाण्यचोदयन्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7046"><div class="verse-block verse-justify"><div class="line">तेन शब्देन सहसा समुद्रे पर्वतोपमाः।</div><div class="line" style="position:relative;">आप्लवन्त गतैः सत्वैर्मत्स्याः शतसहस्रशः॥२१॥</div></div></div></div>
