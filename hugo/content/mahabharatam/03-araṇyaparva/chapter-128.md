<div class="verse-block-wrapper" id="03-araṇyaparva-b5300"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5301"><div class="verse-block verse-justify"><div class="line">कथंवीर्यः स राजाऽभूत्सोमको ददतांवरः।</div><div class="line">कर्माण्यस्य प्रभावं च श्रोतुमिच्छामि तत्त्वतः ॥लोमश उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5302"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरासीन्नृपतिः सोमको नाम धार्मिकः।</div><div class="line" style="position:relative;">तस्य भार्याशतं राजन्सदृशीनामभूत्तदा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5303"><div class="verse-block verse-justify"><div class="line">स वै यत्नेन महता तासु पुत्रं महीपतिः।</div><div class="line" style="position:relative;">कंचिन्नासादयामास कालेन महता ह्यपि॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5304"><div class="verse-block verse-justify"><div class="line">कदाचित्तस् वृद्धस्य यतमानस्य धीमतः।</div><div class="line" style="position:relative;">जन्तुर्नाम सुतस्तस्य ज्येष्ठायां समजायत॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5305"><div class="verse-block verse-justify"><div class="line">तं जातं मातरः सर्वाः परिवार्य समासते।</div><div class="line" style="position:relative;">सततं पृष्ठतः कृत्वा कामभोगान्विशांपते॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5306"><div class="verse-block verse-justify"><div class="line">ततः पिपीलिका जन्तुं कदाचिददशत्स्फिचि।</div><div class="line" style="position:relative;">स दष्टो ह्यरुदद्राजंस्तेन दुःखेन बालकः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5307"><div class="verse-block verse-justify"><div class="line">ततस्ता मातरः सर्वाः प्राक्रोशन्भृशदुःखिताः।</div><div class="line" style="position:relative;">प्रवार्य जनतुं सहसा स शब्दस्तुमुलोऽभवत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5308"><div class="verse-block verse-justify"><div class="line">तमार्तनादं सहसा शुश्राव स महीपतिः।</div><div class="line" style="position:relative;">अमात्यपर्षदो मध्ये उपविष्टः सहर्त्विजा॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5309"><div class="verse-block verse-justify"><div class="line">ततः प्रस्थापयामास किमेतदिति पार्थिवः।</div><div class="line" style="position:relative;">तस्मै क्षत्ता यथावृत्तमाचचक्षे सुतं प्रति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5310"><div class="verse-block verse-justify"><div class="line">त्वरमाणः स चोत्थाय सोमकः सह मन्त्रिभिः।</div><div class="line" style="position:relative;">प्रविश्यान्तःपुरं पुत्रमाश्वासयदरिंदमः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5311"><div class="verse-block verse-justify"><div class="line">सान्त्वयित्वा तु तं पुत्रं निष्क्रम्यान्तःपुरान्नृपः।</div><div class="line">ऋत्विजा सहितो राजन्सहामात्य उपाविशत् ॥सोमक उवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5312"><div class="verse-block verse-justify"><div class="line">धिगस्त्विहैकपुत्रत्वमपुत्रत्वं वरं भवेत्।</div><div class="line" style="position:relative;">नित्यातुरत्वाद्भूतानां शोक एवैकपुत्रता॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5313"><div class="verse-block verse-justify"><div class="line">इदं भार्याशतं ब्रह्मन्परीक्ष्यसदृशं प्रभो।</div><div class="line" style="position:relative;">पुत्रार्थिना मया वोढं न तासां विद्यते प्रजा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5314"><div class="verse-block verse-justify"><div class="line">एकः कथंचिदुत्पन्नः पुत्रो जन्तुरयं मम।</div><div class="line" style="position:relative;">यतमानासु सर्वासु किंनु दुःखमतः परम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5315"><div class="verse-block verse-justify"><div class="line">वयश्च समतीतं मे सभार्यस्यं द्विजोत्तम।</div><div class="line" style="position:relative;">आसां प्राणाः समायत्ता मम चात्रैकपुत्रके॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5316"><div class="verse-block verse-justify"><div class="line">स्यात्तु कर्म तथा युक्तं येन पुत्रशतं भवेत्।</div><div class="line">महता लघुना वाऽपि कर्मणा दुष्करेण वा ॥ऋत्विगुवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5317"><div class="verse-block verse-justify"><div class="line">अस्ति चैतादृशं कर्म येन पुत्रशतं भवेत्।</div><div class="line">यदि शक्नोषि तत्कर्तुमथ वक्ष्यामि सोमक ॥सोमक उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5318"><div class="verse-block verse-justify"><div class="line">कार्यं वा यदि वाऽकार्यं येन पुत्रशतं भवेत्।</div><div class="line">कृतमेवेति तद्विद्धि भगवान्प्रब्रवीतु मे ॥ऋत्विगुवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5319"><div class="verse-block verse-justify"><div class="line">यजस्व जन्तुना राजंस्त्वं मया वितते क्रतौ।</div><div class="line" style="position:relative;">ततः पुत्रशतं श्रीमद्भविष्यत्यचिरेण ते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5320"><div class="verse-block verse-justify"><div class="line">वपार्या हूयमानायां धूममाघ्राय मातरः।</div><div class="line" style="position:relative;">ततस्ताः सुमहावीर्याञ्जनयिष्यन्ति ते सुतान्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5321"><div class="verse-block verse-justify"><div class="line">तस्यामेव तु ते जन्तुर्भविता पुनरात्मजः।</div><div class="line" style="position:relative;">उत्तरे चास्य सौवर्णं लक्ष्म पार्श्वे भविष्यति॥२२॥</div></div></div></div>
