<div class="verse-block-wrapper" id="03-araṇyaparva-b12086"><div class="verse-block verse-justify"><div class="line">यच्चापि ते भयं तीव्रं न च कीर्तयसे क्वचित्।</div><div class="line" style="position:relative;">तच्चाप्यपहरिष्यामि धनंजय इतो गते॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12087"><div class="verse-block verse-justify"><div class="line">किंनु रतज्जपतांश्रेष्ठ कर्णं प्रति महद्भयम्।</div><div class="line">आसीन्न च स धर्मात्मा कथयामास कस्यचित् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12088"><div class="verse-block verse-justify"><div class="line">अहं ते राजशार्दूल कथयामि कथामिमाम्।</div><div class="line" style="position:relative;">पृच्छतो भरतश्रेष्ठ शुश्रूषस्व गिरं मम॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12089"><div class="verse-block verse-justify"><div class="line">द्वादशे समतिक्रान्ते वर्षे प्राप्ते त्रयोदशे।</div><div class="line" style="position:relative;">पाण्डूनां हितकृच्छक्रः कर्णं भिक्षितुमुद्यतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12090"><div class="verse-block verse-justify"><div class="line">अभिप्रायमथो ज्ञात्वा महेन्द्रस्य विभावसुः।</div><div class="line" style="position:relative;">कुण्डलार्थे महाराज सूर्यः कर्णमुपागतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12091"><div class="verse-block verse-justify"><div class="line">महार्हे शयने वीरं स्पर्द्ध्यास्तरणसंवृते।</div><div class="line" style="position:relative;">शयानमतिविश्वस्तं ब्रह्मण्यं सत्यवादिनम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12092"><div class="verse-block verse-justify"><div class="line">स्वप्नान्ते निशि राजेन्द्र दर्शयामास रश्मिवान्।</div><div class="line" style="position:relative;">कृपया परयाऽऽविष्टः पुत्रस्नेहाच्च भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12093"><div class="verse-block verse-justify"><div class="line">ब्राह्मणो वेदविद्भूत्वा सूर्यो योगर्द्धिरूपवान्।</div><div class="line" style="position:relative;">हितार्थमब्रवीत्कर्णं सान्त्वपूर्वमिदं वचः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12094"><div class="verse-block verse-justify"><div class="line">कर्ण मद्वचनं तात शृणु सत्यभृतांवर।</div><div class="line" style="position:relative;">ब्रुवतोऽद्य महाबाहो सौहृदात्परमं हितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12095"><div class="verse-block verse-justify"><div class="line">उपायास्यति शक्रस्त्वां पाण्डवानां हितेप्सया।</div><div class="line" style="position:relative;">ब्राह्मणच्छद्मना कर्ण कुण्डलोपजिहीर्षया॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12096"><div class="verse-block verse-justify"><div class="line">विदितं तेन शीलं ते सर्वस्य जगतस्तथा।</div><div class="line" style="position:relative;">यथा त्वं भिक्षितः सद्भिर्ददास्येव न याचसे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12097"><div class="verse-block verse-justify"><div class="line">त्वं हि तात ददास्येव ब्राह्मणेभ्यः प्रयाचितम्।</div><div class="line" style="position:relative;">वित्तं यच्चान्यदप्याहुर्न प्रत्याख्यासि कस्यचित्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12098"><div class="verse-block verse-justify"><div class="line">त्वां तु चैवंविधं ज्ञात्वा स्वयं वै पाकशासनः।</div><div class="line" style="position:relative;">आगन्ता रकुण्डलार्थाय कवचं चैव भिक्षितुम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12099"><div class="verse-block verse-justify"><div class="line">तस्मै प्रयाचमानाय न देये कुण्डले त्वया।</div><div class="line" style="position:relative;">अनुनेयः परं शक्त्या श्रेय एतद्धि ते परम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12100"><div class="verse-block verse-justify"><div class="line">कुण्डलार्थेऽब्रुवंस्तात कारणैर्बहुभिस्त्वया।</div><div class="line" style="position:relative;">अन्यैर्बहुविधैर्वित्तैः सन्निवार्यः पुनःपुनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12101"><div class="verse-block verse-justify"><div class="line">रत्नैः स्त्रीभिस्तथा गोभिर्धनैर्बहुविधैरपि।</div><div class="line" style="position:relative;">निदर्शनैश्च बहुभिः कुण्डलेप्सुः पुरंदरः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12102"><div class="verse-block verse-justify"><div class="line">यदि दास्यसि कर्ण त्वं सहजे कुण्डले शुभे।</div><div class="line" style="position:relative;">आयुषः प्रक्षयं गत्वा मृत्योर्वशमुपैष्यसि॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12103"><div class="verse-block verse-justify"><div class="line">कवचेन समायुक्तः कुण्डलाभ्यां च मानद।</div><div class="line" style="position:relative;">अवध्यस्त्वं रणेऽरीणामिति विद्धि वचो मम॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12104"><div class="verse-block verse-justify"><div class="line">अमृतादुत्थितं ह्येतदुभयं रत्नसंमितम्।</div><div class="line">तस्माद्रक्ष्यं त्वया कर्ण जीवितं चेत्प्रियं तव ॥कर्ण उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12105"><div class="verse-block verse-justify"><div class="line">को मामेवं भवान्प्राह दर्शयन्सौहृदं परम्।</div><div class="line">कामया भगवन्ब्रूहि को भवान्द्विजवेषधृक् ॥ब्रह्मण उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12106"><div class="verse-block verse-justify"><div class="line">अहं तात सहस्रांशुः सौहृदात्त्वां निदर्शये।</div><div class="line">कुरुष्वैतद्वयो मे त्वमेतच्छ्रेयः परं हि ते ॥कर्ण उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12107"><div class="verse-block verse-justify"><div class="line">श्रेय एव ममात्यन्तं यस्य मे गोपतिः प्रभुः।</div><div class="line" style="position:relative;">प्रवक्ताऽद्य हितान्वेषी शृणु चेदं वचो मम॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12108"><div class="verse-block verse-justify"><div class="line">प्रसादये त्वां वरदं प्रणयाच्च ब्रवीम्यहम्।</div><div class="line" style="position:relative;">न निवार्यो व्रतादस्मादहं यद्यस्मि ते प्रियः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12109"><div class="verse-block verse-justify"><div class="line">व्रतं वै मम लोकोऽयं वेत्ति कृत्स्नं विभावसो।</div><div class="line" style="position:relative;">यथाऽहं द्विजमुख्येभ्यो दद्यां प्राणानपिध्रुवम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12110"><div class="verse-block verse-justify"><div class="line">यद्यागच्छति मां शक्रो ब्राह्मणच्छद्मना वृतः।</div><div class="line" style="position:relative;">हितार्थं पाण्डुपुत्राणां खेचरोत्तम भिक्षितुम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12111"><div class="verse-block verse-justify"><div class="line">दास्यामि विबुधश्रेष्ठ कुण्डले वर्म चोत्तमम्।</div><div class="line" style="position:relative;">न मे कीर्तिः प्रणश्येत त्रिषु लोकेषु विश्रुता॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12112"><div class="verse-block verse-justify"><div class="line">मद्विधस्यायशस्यं हि न युक्तं प्राणरक्षणम्।</div><div class="line" style="position:relative;">युक्तं हि यशसा युक्तं मरणं लोकसंमतम्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12113"><div class="verse-block verse-justify"><div class="line">सोहमिन्द्राय दास्यामि कुण्डले सह वर्मणा।</div><div class="line" style="position:relative;">यदि मां बलवृत्रघ्नो भिक्षार्थमुपयास्यति॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12114"><div class="verse-block verse-justify"><div class="line">हितार्थं पाण्डुपुत्राणां कुण्डले मे प्रयाचितुम्।</div><div class="line" style="position:relative;">तन्मे कीर्तिकरं लोके तस्याकीर्तिर्भविष्यति॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12115"><div class="verse-block verse-justify"><div class="line">वृणोमि कीर्तिं लोके हि जीवितेनापि भानुमन्।</div><div class="line" style="position:relative;">कीर्तिमानश्नुते स्वर्गं हीनकीर्तिस्तु नश्यति॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12116"><div class="verse-block verse-justify"><div class="line">कीर्तिर्हि पुरुषं लोके संजीवयति मातृवत्।</div><div class="line" style="position:relative;">अकीर्तिर्जीवितं हन्ति जीवतोपि शरीरिणः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12117"><div class="verse-block verse-justify"><div class="line">अयंपुराणः श्लोको हि स्वयं गीतो विभावसो।</div><div class="line" style="position:relative;">धात्रा लोकेश्वर यथा कीर्तिरायुर्नरस्य ह॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12118"><div class="verse-block verse-justify"><div class="line">पुरुषस्य परे लोके कीर्तिरेव परायणम्।</div><div class="line" style="position:relative;">इह लोके विशुद्धा च कीर्तिरायुर्विवर्धनी॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12119"><div class="verse-block verse-justify"><div class="line">सोहं शरीरजे दत्त्वा कीर्तिं प्राप्स्यामि शाश्वतीम्।</div><div class="line" style="position:relative;">दत्त्वा च विधिवद्दानं ब्राह्मणेभ्यो यथाविधि॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12120"><div class="verse-block verse-justify"><div class="line">हुत्वा शरीरं रसंग्रामे कृत्वा कर्म सुदुष्करम्।</div><div class="line" style="position:relative;">विजित्य च परानाजौ यशः प्राप्स्यामि केवलम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12121"><div class="verse-block verse-justify"><div class="line">भीतानामभयं दत्त्वा संग्रामे जीवितार्थिनाम्।</div><div class="line" style="position:relative;">वृद्धान्वालान्द्विजातींश्च मोक्षयित्वा महाभयात्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12122"><div class="verse-block verse-justify"><div class="line">प्राप्स्यामि परमं लोके यशः स्वर्ग्यमनुत्तमम्।</div><div class="line" style="position:relative;">जीवितेनापि मे रक्ष्या कीर्तिस्तद्विद्वि मे व्रतम्॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12123"><div class="verse-block verse-justify"><div class="line">सो हं दत्त्वा मघवते भिक्षामेतामनुत्तमाम्।</div><div class="line" style="position:relative;">ब्राह्मणच्छद्मिने देव लोके गन्ता परां गतिम्॥४०॥</div></div></div></div>
