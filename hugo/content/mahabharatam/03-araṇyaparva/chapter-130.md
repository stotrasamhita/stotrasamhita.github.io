<div class="verse-block-wrapper" id="03-araṇyaparva-b5347"><div class="verse-block verse-justify"><div class="line">`वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5348"><div class="verse-block verse-justify"><div class="line">सोमकस्याश्रमे पुण्ये धर्मराजो युधिष्ठिरः।</div><div class="line">षड्रात्रमुष्य नियतो भ्रात्रादिभिररिंदमः।</div><div class="line" style="position:relative;">तस्मान्निर्गम्य सहसा दिशं प्रायात्तथोत्तराम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5349"><div class="verse-block verse-justify"><div class="line">बहुदूरं ततो गत्वा वनं तत्र मनोहरम्।</div><div class="line" style="position:relative;">बहुपष्पफलाकीर्णं महानद्युपशोभितम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b5351"><div class="verse-block verse-justify"><div class="line">बहुपुष्पफलोपेतं मुनिसङ्घैर्निषेवितम्।</div><div class="line" style="position:relative;">स्रवन्त्या च समायुक्तं महत्या पुण्यतोयया॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5352"><div class="verse-block"><div class="line">ऋषीणामाश्रमाः पुण्या दृश्यन्ते विविधा मुने</div><div class="line pada-even">कस्यायमाश्रमः पुण्यः कस्येमे मुनयोऽमलाः।</div><div class="line">एतदिच्छाम्यहं श्रोतुं वद त्वं वदतांवर ॥लोमश उवाच</div><div class="line pada-even" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5353"><div class="verse-block verse-justify"><div class="line">शृणु राजेनद््र भद्रं ते वनस्यास्य पुरातनम्।</div><div class="line" style="position:relative;">वृत्तान्तं निखिलेनाद्य प्रोच्यमानं मयाऽनघ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5354"><div class="verse-block verse-justify"><div class="line">मृकण्डुपुत्रो मेधावी मार्कण्डेयो महामुनिः।</div><div class="line" style="position:relative;">बाल एव महाबुद्धिः सर्वविद्याविशारदः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5355"><div class="verse-block verse-justify"><div class="line">मातापित्रोः प्रियं कुर्वंस्तपोर्थं वनमाविशत्।</div><div class="line" style="position:relative;">अत्राश्रमपदं कृत्वा तपस्तेपे सुदारुणम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5356"><div class="verse-block verse-justify"><div class="line">ग्रीष्मे पञ्चतपा भूत्वा वर्षास्वाकाशसंश्रयः।</div><div class="line" style="position:relative;">जलस्थः शिशिरे योगी बहुकालमवर्तत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5357"><div class="verse-block verse-justify"><div class="line">ऊर्ध्वबाहुर्निरालम्बः पादाङ्गुष्ठाग्रविष्ठितः।</div><div class="line">जितेन्द्रियो जितप्राणश्चिन्तयन्दृढमव्ययम्।</div><div class="line" style="position:relative;">अनाहारो जितक्रोधश्चिरमेवमवर्तत॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5358"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नन्तरे राजन्ननावृष्टिः सुदारुणा।</div><div class="line" style="position:relative;">संभूता सर्वसंहर्त्री तया दग्धं चराचरम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5359"><div class="verse-block verse-justify"><div class="line">अनावृष्ट्यां प्रवृत्तायां सर्वे च निधनं गताः।</div><div class="line" style="position:relative;">केचिदन्ये महात्मानो मुनयो द्विजपुङ्गवाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5360"><div class="verse-block verse-justify"><div class="line">ब्राह्मणाः क्षत्रिया वैश्याः शूद्रा सर्वाश्च योषितः।</div><div class="line">पशुपक्षिमृगाः सर्वे क्षुत्पिपासासमाकुलाः।</div><div class="line" style="position:relative;">कृशाः शुष्कोष्ठकण्ठाश्च श्रान्ता भ्रान्ता विचेतसः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5361"><div class="verse-block verse-justify"><div class="line">केनचित्पुण्यशेषेण मार्कण्डेयस्य धीमतः।</div><div class="line" style="position:relative;">आश्रमं शनकैः प्राप्ता मूर्च्छिताः सहसाऽपतन्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5362"><div class="verse-block verse-justify"><div class="line">समाधिविरतो योगी मार्कण्डेयो महातपाः।</div><div class="line" style="position:relative;">तद्वनं निबिडं दृष्ट्वा जनैः क्षुत्तृट्समाकलैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5363"><div class="verse-block verse-justify"><div class="line">दयार्द्रहृदयो योगी शिवं ध्यात्वा हृदम्बुजे।</div><div class="line">तेषां संरक्षणार्थाय वेदगन्धिस्वरेण सः।</div><div class="line" style="position:relative;">आजुहाव तदा गङ्गां तपोयोगेन भारत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5364"><div class="verse-block verse-justify"><div class="line">गङ्गा समागता शीघ्रं तेनाहूताऽतिपावना।</div><div class="line" style="position:relative;">नाम्ना वेदनदीत्येव प्रख्याता लोकपावना॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5365"><div class="verse-block verse-justify"><div class="line">पर्जन्यश्च समाहूतः सुखं वर्षति भारत।</div><div class="line">सस्यानि च समृद्धानि फलमूलान्यनेकशः।</div><div class="line" style="position:relative;">संभूतान्यत्रराजेनद्र सर्वे ते रक्षिता जनाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5366"><div class="verse-block verse-justify"><div class="line">मार्कण्डेयं प्रशंसन्तो जनाः सर्वे द्विजात्तयः।</div><div class="line" style="position:relative;">चिरं सुखमवर्तन्त तेन संरक्षिता नृप॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5367"><div class="verse-block verse-justify"><div class="line">एवं विधाय रक्षां स सर्वेषां पुण्यकर्मणाम्।</div><div class="line" style="position:relative;">पुनश्चचार च तपः परमेश्वरतुष्टये॥२१॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b5370"><div class="verse-block verse-justify"><div class="line">गन्धर्वयक्षप्रवरैः सकिन्नरपतत्रिभिः।</div><div class="line">स्तूयमानं महादेवमव्ययं निष्कलं शिवम्।</div><div class="line" style="position:relative;">प्रणनाम मुनिर्भक्त्या साष्टाङ्गं च पुनः पुनः॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b5372"><div class="verse-block verse-justify"><div class="line">तमुवाच महादेवो मार्कण्डेयं महामुनिम्।</div><div class="line" style="position:relative;">वरं वरय भद्रं ते वरदोस्मि मुने तव॥२५॥</div></div></div>
