<div class="verse-block-wrapper" id="03-araṇyaparva-b7234"><div class="verse-block verse-justify"><div class="line">अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7235"><div class="verse-block verse-justify"><div class="line">ततो मामिविश्वस्तं संरूढशरविक्षतम्।</div><div class="line" style="position:relative;">देवराजोऽनुगृह्येदं काले वचनमब्रवीत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7236"><div class="verse-block verse-justify"><div class="line">दिव्यान्यस्त्राणि सर्वाणि त्वयि तिष्ठन्ति भारत।</div><div class="line" style="position:relative;">न त्वाऽभिभवितुं शक्तो मानुषो भुवि कश्चन॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7237"><div class="verse-block verse-justify"><div class="line">भीष्मो द्रोणः कृपः कर्णः शकुनिः सह राजभिः।</div><div class="line" style="position:relative;">संग्रामस्थस्य ते पुत्र कलां नार्हन्ति षोडशीम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7238"><div class="verse-block verse-justify"><div class="line">इदं च मे तनुत्राणं प्रायच्छन्मघवान्प्रभुः।</div><div class="line" style="position:relative;">अभेद्यं कवचं दिव्यं स्रजं चैव हिरण्मयीम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7239"><div class="verse-block verse-justify"><div class="line">देवदत्तं च मे शङ्खं देवः प्रादान्महारवम्।</div><div class="line" style="position:relative;">दिव्यं चेदं किरीटं मे स्वयमिन्द्रो युयोज ह॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7240"><div class="verse-block verse-justify"><div class="line">ततो दिव्यानि वस्त्राणि दिव्यान्याभरणानि च।</div><div class="line" style="position:relative;">प्रादाच्छक्रो ममैतानि रुचिराणि बृहन्ति च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7241"><div class="verse-block verse-justify"><div class="line">एवं संपूजितस्तत्रसुखमस्म्युषितो नृप।</div><div class="line" style="position:relative;">इन्द्रस् भवने पुण्ये गन्धर्वशिशुभिः सह॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7242"><div class="verse-block verse-justify"><div class="line">ततो मामब्रवीच्छक्रः प्रीतिमानमरैः सह।</div><div class="line" style="position:relative;">समयोऽर्जुन गन्तुं ते भ्रातरो हि स्मरन्ति ते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7243"><div class="verse-block verse-justify"><div class="line">एवमिन्द्रस्य भवने पञ्चवर्षाणि भारत।</div><div class="line" style="position:relative;">उषितानि मया राजन्स्मरता द्यूतजं कलिम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7244"><div class="verse-block verse-justify"><div class="line">ततो भवन्तमद्राक्षं भ्रातृभिः परिवारितम्।</div><div class="line">गन्धमादनमासाद्य पर्वतस्यास्य मूर्धनि ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7245"><div class="verse-block verse-justify"><div class="line">दिष्ट्या धनंजयास्त्राणि त्वया प्राप्तानि भारत।</div><div class="line" style="position:relative;">दिष्ट्या चाराधितो राजा देवानामीश्वरः प्रभुः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7246"><div class="verse-block verse-justify"><div class="line">दिष्ट्या च भगवन्स्थाणुर्देव्या सह परंतपः।</div><div class="line" style="position:relative;">साक्षाद्दृष्टः स्वयुद्धेन तोषितश् त्वयाऽनघ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7247"><div class="verse-block verse-justify"><div class="line">दिष्ट्या च लोकपालैस्त्वं समेतो भरतर्षभ।</div><div class="line" style="position:relative;">दिष्ट्या वर्धामहे पार्थ दिष्ट्याऽसि पुनरागतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7248"><div class="verse-block verse-justify"><div class="line">अद्य कृत्स्नामिमां देवीं विजितां पुरमालिनीम्।</div><div class="line">`पश्यामि भूमिं कौन्तेय त्वया मे प्रतिपादिताम्'।</div><div class="line" style="position:relative;">मन्ये च धृतराष्ट्रस्य पुत्रानपि वशीकृतान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7249"><div class="verse-block verse-justify"><div class="line">इच्छामि तानि चास्त्राणि द्रष्टुं दिव्यानि भारत।</div><div class="line">यैस्तथा वीर्यवन्तस्ते निवातकवचा हताः ॥अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7250"><div class="verse-block verse-justify"><div class="line">सः प्रभाते भवान्द्रष्टा दिव्यान्यस्त्राणि सर्वशः।</div><div class="line">निवातकवचा घोरा यैर्मया विनिपातिताः ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7251"><div class="verse-block verse-justify"><div class="line">एवमागमनं तत्र कथयित्वा धनंजयः।</div><div class="line" style="position:relative;">भ्रातृभिः सहितः सर्वै रजनीं तामुवास ह॥१८॥</div></div></div></div>
