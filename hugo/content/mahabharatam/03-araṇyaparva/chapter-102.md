<div class="verse-block-wrapper" id="03-araṇyaparva-b4564"><div class="verse-block verse-justify"><div class="line">देवा ऊचुः।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4565"><div class="verse-block verse-justify"><div class="line">तव प्रसादाद्वर्धन्ते प्रजाः सर्वाश्चतुर्विधाः।</div><div class="line" style="position:relative;">ता भाविता भावयनति हव्यकव्यैर्दिवौकसः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4566"><div class="verse-block verse-justify"><div class="line">लोका ह्येवं विवर्धन्ते ह्यन्योन्यं समुपाश्रिताः।</div><div class="line" style="position:relative;">त्वत्प्रसादान्निरुद्विग्नास्त्वयैव परिरक्षिताः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4567"><div class="verse-block verse-justify"><div class="line">इदं च समनुप्राप्तं लोकानां भयमुत्तमम्।</div><div class="line" style="position:relative;">न च जानीम केनेति रात्रौ वध्यन्ति ब्राह्मणाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4568"><div class="verse-block verse-justify"><div class="line">क्षीणेषु च ब्राह्मणेषु पृथिवी क्षयमेष्यति।</div><div class="line" style="position:relative;">तत पृथिव्यां क्षीणायां त्रिदिवं क्षयमेष्यति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4569"><div class="verse-block verse-justify"><div class="line">त्वत्प्रसादान्महाबाहो लोकाः सर्वे जगत्पते।</div><div class="line">विनाशं नाधिगच्छेयुस्त्वया वै परिरक्षिताः ॥विष्णुरुवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4570"><div class="verse-block verse-justify"><div class="line">विदितं मे सुराः सर्वं प्रजानां क्षयकारणम्।</div><div class="line" style="position:relative;">भवतां चापि वक्ष्यामि शृणुध्वं विगतज्वराः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4571"><div class="verse-block verse-justify"><div class="line">कालेया इति विख्याता गणाः परमदारुणाः।</div><div class="line" style="position:relative;">तैश्च वृत्रं समाश्रित्य जगत्सर्वं प्रमाथितम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4572"><div class="verse-block verse-justify"><div class="line">ते वृत्रं निहतं दृष्ट्वा सहस्राक्षेण धीमता।</div><div class="line" style="position:relative;">जीवितं परिरक्षन्तः प्रविष्टा वरुणालयम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4573"><div class="verse-block verse-justify"><div class="line">ते प्रविश्योदधिं घोरं नक्रग्राहसमाकुलम्।</div><div class="line" style="position:relative;">उत्सादार्थं लोकानां रात्रौ घ्नन्ति ऋषीनिह॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4574"><div class="verse-block verse-justify"><div class="line">न तु शक्याः क्षयं नेतुं समुद्राश्रयिणो हि ते।</div><div class="line" style="position:relative;">समुद्रस्य क्षये बुद्धिर्भवद्भिः संप्रधार्यताम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4575"><div class="verse-block verse-justify"><div class="line">`एतच्छ्रुत्वा वचो देवा विष्णुना समुदीरितम्।</div><div class="line" style="position:relative;">विष्णुमेव पुरस्कृत्यब्रह्माणं समुपस्थिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4576"><div class="verse-block verse-justify"><div class="line">ततस्ते प्रणता भूत्वा तमेवार्धं न्यवेदयन्।</div><div class="line" style="position:relative;">सर्वलोकविनाशार्थं कालेया कृतनिश्चयाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4577"><div class="verse-block verse-justify"><div class="line">एषां तद्वचनं श्रुत्वा पद्मयोनिः सनातनः।</div><div class="line" style="position:relative;">उवाच परमप्रीतस्त्रिदशानर्थवद्वचः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4578"><div class="verse-block verse-justify"><div class="line">विदितं मे सुराः सर्वे दानवानां विवेष्टितम्।</div><div class="line" style="position:relative;">मनुष्यादेश्च निधनं कालेयैः कालचोदितैः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4579"><div class="verse-block verse-justify"><div class="line">क्षयस्तेपामनुप्राप्तः कालेनोपहताश्च ये।</div><div class="line" style="position:relative;">उपायं संप्रवक्ष्यामि समुद्रस्य विशोषणे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4580"><div class="verse-block verse-justify"><div class="line">अगस्त्य इतिविग्व्यातो वारुणिः सुसमाहितः।</div><div class="line" style="position:relative;">तमुपागम्य सहिता इममर्थं प्रयाचत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4581"><div class="verse-block verse-justify"><div class="line">स हि शक्तो महातेजाः क्षणात्पातुं महोदधिम्'।</div><div class="line" style="position:relative;">अगस्त्येन विना को हि शक्तोऽन्योऽर्णवशोपणे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4582"><div class="verse-block verse-justify"><div class="line">अन्यथा हि न शक्यास्ते विना सागरशोपणम्।</div><div class="line" style="position:relative;">समुद्रे च क्षयं नीते कालेयान्निहनिष्यथ॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4583"><div class="verse-block verse-justify"><div class="line">एवं श्रुत्वा वचो देवा ब्रह्मणः परमेष्ठिनः।</div><div class="line" style="position:relative;">परमेष्ठिनमाज्ञाप्यअगस्त्यस्याश्रमं ययुः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4584"><div class="verse-block verse-justify"><div class="line">तत्रापश्यन्महात्मानं वारुणिं दीप्ततेजसम्।</div><div class="line" style="position:relative;">उपास्यमानमृपिभिर्देवैरिव पितामहम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4585"><div class="verse-block verse-justify"><div class="line">तेऽभिगम्य महात्मानं मैत्रावरुणिमच्युतम्।</div><div class="line">आश्रमश्थं तपोराशिं कर्मभिः स्वैस्तु तुष्टुवुः ॥देवा ऊचुः।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4586"><div class="verse-block verse-justify"><div class="line">नहुपेणाभितप्तानां त्वं लोकानां गतिः पुरा।</div><div class="line" style="position:relative;">भ्रंशितश्च सुरैश्वर्याल्लोकार्थं लोककण्टकः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4587"><div class="verse-block verse-justify"><div class="line">क्रोधात्प्रवृद्धस्तरणं भास्करस्य नभोगतः।</div><div class="line" style="position:relative;">वचस्तवानतिक्रामन्विन्ध्यः शैलो न वर्धते॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4588"><div class="verse-block verse-justify"><div class="line">तमसा चावृतेलोके मृत्युनाऽभ्यर्दिताः प्रजाः।</div><div class="line" style="position:relative;">त्वामेव नाथमासाद्य निर्वृतिं परमां गताः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4589"><div class="verse-block verse-justify"><div class="line">अस्माकं भयभीतानां नित्यशो भगवान्गतिः।</div><div class="line" style="position:relative;">ततस्त्वार्ताः प्रयाचामो वरं त्वां वरदो ह्यसि॥२६॥</div></div></div></div>
