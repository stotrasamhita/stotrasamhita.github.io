<div class="verse-block-wrapper" id="03-araṇyaparva-b0734"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा रौक्मिणेयो यादवान्यादवर्षभ।</div><div class="line" style="position:relative;">दंशितैर्हरिभिर्युक्तं रथमास्थाय काञ्चनम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0735"><div class="verse-block verse-justify"><div class="line">उच्छ्रित्य मकरं केतुं व्यात्ताननमलङ्कृतम्।</div><div class="line" style="position:relative;">उत्पतद्भिरिवाकाशं तैर्हयैरन्वयात्परान्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0736"><div class="verse-block verse-justify"><div class="line">विक्षिपन्नादयंश्चापि धनुः श्रेष्ठं महाबलैः।</div><div class="line" style="position:relative;">तूणखङ्गधरः शूरो बद्धगोधाङ्गुलित्रवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0737"><div class="verse-block verse-justify"><div class="line">सविद्युच्छुरितं चापं विहरन्वै तलात्तलम्।</div><div class="line" style="position:relative;">मोहयामास दैतेयान्सर्वान्सौभनिवासिनः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0738"><div class="verse-block verse-justify"><div class="line">तस्य विक्षिपतश्चापं संदधानस्य चासकृत्।</div><div class="line" style="position:relative;">नान्तरं ददृशे कश्चिन्नघ्नितः शात्रवान्रणे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0739"><div class="verse-block verse-justify"><div class="line">मुखस्य वर्णो न विकल्पतेऽस्यचेलुश्च गात्राणि न चापि तस्य।</div><div class="line" style="position:relative;">सिंहोन्नतं चाप्यभिगर्जतोऽस्यशुश्राव लोकोऽद्भुतवीर्यमग्र्यम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0740"><div class="verse-block verse-justify"><div class="line">जलेचरः काञ्चनयष्टिसंस्थोव्यात्ताननः शत्रुबलप्रमाथी।</div><div class="line" style="position:relative;">वित्रासयन्राजति वाहमुख्येसाल्वस्य सेनाप्रमुखे ध्वजाग्र्यः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0741"><div class="verse-block verse-justify"><div class="line">ततस्तूर्णं विनिष्पत्य प्रद्युम्नः शत्रुकर्शनः।</div><div class="line" style="position:relative;">साल्वमेवाभिदुद्राव विधित्सुः कलहं नृप॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0742"><div class="verse-block verse-justify"><div class="line">अभियानं तु वीरेण प्रद्युम्नेन महारणे।</div><div class="line" style="position:relative;">नामर्षयत संक्रुद्वः साल्वः कुरुकुलोद्वह॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0743"><div class="verse-block verse-justify"><div class="line">सरोषमहमत्तो वै कामगादवरुह्य च।</div><div class="line" style="position:relative;">प्रद्युम्नं योधयामास साल्वः परपुरंजयः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0744"><div class="verse-block verse-justify"><div class="line">तयोः सुतुमुलं युद्धं साल्ववृष्णिप्रवीरयोः।</div><div class="line" style="position:relative;">समेता ददृशुर्लोका बलिवासवयोरिव॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0745"><div class="verse-block verse-justify"><div class="line">तस्य मायामयो वीर रथो हेमपरिष्कृतः।</div><div class="line" style="position:relative;">सपताकः सध्वजश्च सानुकर्षः स तूणवान्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0746"><div class="verse-block verse-justify"><div class="line">स तं रथवरं श्रीमान्समारुह्य किल प्रभो।</div><div class="line" style="position:relative;">मुमोच बाणान्कौख्य प्रद्युम्नाय महाबलः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0747"><div class="verse-block verse-justify"><div class="line">ततो बाणमयं वर्षं व्यसृजत्तरसा रणे।</div><div class="line" style="position:relative;">प्रद्युम्नो भुजवेगेन साल्वं संमोहयन्निव॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0748"><div class="verse-block verse-justify"><div class="line">स तैरभिहतः सङ्ख्ये नामर्यत सौभराट्।</div><div class="line" style="position:relative;">शरान्दीप्ताग्निसंकाशान्मुमोच तनये मम॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0749"><div class="verse-block verse-justify"><div class="line">तमापतन्तं बाणौधं स चिच्छेद महाबलः।</div><div class="line" style="position:relative;">ततश्चान्याञ्शरान्दीप्तान्प्रचिक्षेप सुते मम॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0750"><div class="verse-block verse-justify"><div class="line">स साल्वबाणै राजेन्द्र विद्धो रुक्मिणिनन्दनः।</div><div class="line" style="position:relative;">मुमोच बाणं त्वरितो मर्मभेदिनमाहवे॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0751"><div class="verse-block verse-justify"><div class="line">तस्य वर्म विभिद्याशु स बाणो मत्सुतेरितः।</div><div class="line" style="position:relative;">विवेश हृदयं पत्री स पपात भृशहतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0752"><div class="verse-block verse-justify"><div class="line">तस्मिन्निपतिते वीरे साल्वराजे विचेतसि।</div><div class="line" style="position:relative;">संप्रद्रबन्दानवेन्द्रा दारयन्तो वसुंधराम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0753"><div class="verse-block verse-justify"><div class="line">हाहाकृतमभूत्सैन्यं साल्वस्य पृथिवीपते।</div><div class="line" style="position:relative;">नष्टसंज्ञे निपतिते तदा सौभपतौ नृपे॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0754"><div class="verse-block verse-justify"><div class="line">तत उत्थाय राजेन्द्र प्रतिलभ्य च चेतनाम्।</div><div class="line" style="position:relative;">मुमोच बाणान्सहसा प्रद्युम्नाय महाबलः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0755"><div class="verse-block verse-justify"><div class="line">तेन बाणेन महता प्रद्युम्नः समरे स्थितः।</div><div class="line" style="position:relative;">जत्रुदेशे भृशं विद्धो व्यथितो ददृशे तदा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0756"><div class="verse-block verse-justify"><div class="line">तं स विद्ध्वा महाराज साल्वो रुक्मिणिनन्दनम्।</div><div class="line" style="position:relative;">ननाद सिंहनादं वै नादेनापूरयन्महीम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0757"><div class="verse-block verse-justify"><div class="line">ततो मोहं समापन्ने तनये मम भारत।</div><div class="line" style="position:relative;">मुमोच बाणांस्त्वरितः पुनरन्यान्दुरात्मवान्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0758"><div class="verse-block verse-justify"><div class="line">स तैरभिहतो बाणैर्बहुभिस्तेन मोहितः।</div><div class="line" style="position:relative;">निश्चेष्टः कौरवश्रेष्ठ प्रद्युम्नोऽभूद्रणाजिरे॥२६॥</div></div></div></div>
