<div class="verse-block-wrapper" id="03-araṇyaparva-b12065"><div class="verse-block verse-justify"><div class="line">मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12066"><div class="verse-block verse-justify"><div class="line">तस्यां रात्र्यां व्यतीतायामुदिते सूर्यमण्डले।</div><div class="line" style="position:relative;">कृतपौर्वाह्णिकाः सर्वे समेयुस्ते तपोधनाः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12067"><div class="verse-block verse-justify"><div class="line">तदेव सर्वं सावित्र्या महाभाग्यं महर्षयः।</div><div class="line" style="position:relative;">द्युम्सेनाय नातृप्यन्कथयन्तः पुनः पुनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12068"><div class="verse-block verse-justify"><div class="line">ततः प्रकृतयः सर्वाः साल्वेभ्योऽभ्यागा नृपम्।</div><div class="line" style="position:relative;">आचख्युर्निहतं चैव स्वेनामात्येन तं द्विषम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12069"><div class="verse-block verse-justify"><div class="line">तं मन्त्रिणा हतं प्रोच्य ससहायं सबान्धवम्।</div><div class="line" style="position:relative;">न्यवेदयन्यथावृत्तं विद्रुतं च द्विषद्बलम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12070"><div class="verse-block verse-justify"><div class="line">ऐकमत्यं च सर्वस्य जनस्य स्वं नृपं प्रति।</div><div class="line" style="position:relative;">सचक्षुर्वाऽप्यचक्षुर्वा स नो राजा भवत्विति॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12071"><div class="verse-block verse-justify"><div class="line">अनेन निश्चयेनेह वयं प्रस्थापिता नृप।</div><div class="line" style="position:relative;">प्राप्तानीमानि यानानि चतुरङ्गं च ते बलम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12072"><div class="verse-block verse-justify"><div class="line">प्रयाहि राजन्भद्रं ते घुष्टस्ते नगरे जयः।</div><div class="line">अध्यास्स्व चिररात्राय पितृपैतामहं पदम् ॥मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12073"><div class="verse-block verse-justify"><div class="line">चक्षुष्मन्तं च तं दृष्ट्वा राजानं वपुषाऽन्वितम्।</div><div class="line" style="position:relative;">मूर्ध्ना निपतिताः सर्वेविस्मयोत्फुल्ललोचनाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12074"><div class="verse-block verse-justify"><div class="line">तोऽभिवाद्य तान्वृद्धान्द्विजानाश्रमवासिनः।</div><div class="line" style="position:relative;">तैश्चाभिपूजितः सर्वैः प्रययौ नगरं प्रति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12075"><div class="verse-block verse-justify"><div class="line">शैव्या च सह सावित्र्या स्वास्तीर्णेन सुवर्चसा।</div><div class="line" style="position:relative;">नरयुक्तेन यानेन प्रययौ सेनया वृता॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12076"><div class="verse-block verse-justify"><div class="line">ततोऽभिषिषिचुः प्रीत्या द्युमत्सेनं पुरोहिताः।</div><div class="line" style="position:relative;">पुत्रं चास्य महात्मानं यौवराज्येऽभ्यषेचयन्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12077"><div class="verse-block verse-justify"><div class="line">ततः कालेन महता सावित्र्याः कीर्तिवर्धनम्।</div><div class="line" style="position:relative;">तद्वै पुत्रशतं जज्ञे शूराणामनिवर्तिनाम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12078"><div class="verse-block verse-justify"><div class="line">भ्रातृणां सोदराणां च तथैवास्याभवच्छतम्।</div><div class="line" style="position:relative;">मद्राधिपस्याश्वपतेर्मालव्यां सुमहाबलम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12079"><div class="verse-block verse-justify"><div class="line">एवमात्मा पिता माता श्वश्रूः श्वशुर एव च।</div><div class="line" style="position:relative;">भर्तुः कुलं च सावित्र्या सर्वं कृच्छ्रात्समुद्धृतं॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12080"><div class="verse-block verse-justify"><div class="line">तथैवैषा हि कल्याणी द्रौपदी शीलसंमता।</div><div class="line">तारयिष्यति वः सर्वान्सावित्रीव कुलाङ्गना ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12081"><div class="verse-block verse-justify"><div class="line">एवं स पाण्डवस्तेन अनुनीतो महात्मना।</div><div class="line" style="position:relative;">विशोको विज्वरो राजन्काम्यके न्यवसत्तदा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12082"><div class="verse-block verse-justify"><div class="line">यश्चेदं शृणुयाद्भक्त्या सावित्र्याख्यानमुत्तमम्।</div><div class="line" style="position:relative;">स सुखी सर्वसिद्धार्थो न दुःखं प्राप्नुयान्नरः॥१८॥</div></div></div></div>
