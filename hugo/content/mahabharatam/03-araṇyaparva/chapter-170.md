<div class="verse-block-wrapper" id="03-araṇyaparva-b7000"><div class="verse-block verse-justify"><div class="line">न त्वमद्य युधा जेतुं शक्यः सुरगणैरपि।</div><div class="line" style="position:relative;">किं पुनर्मानुषे लोके मानुषैरकृतात्मभिः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7001"><div class="verse-block verse-justify"><div class="line">अप्रमेयोऽप्रधृष्यश्च युद्धेष्वप्रतिमस्तथा।</div><div class="line" style="position:relative;">`अजेयस्त्वं हि सङ्ग्रामे सर्वैरपिसुरासुरैः'॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7002"><div class="verse-block verse-justify"><div class="line">अथाब्रवीत्पुनर्देवः संप्रहृष्टतनूरुहः।</div><div class="line" style="position:relative;">अस्त्रयुद्धे समो वीर न ते कश्चिद्भविष्यति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7003"><div class="verse-block verse-justify"><div class="line">अप्रमत्तः सदा दक्षः सत्यवादी जितेन्द्रियः।</div><div class="line" style="position:relative;">ब्रह्मण्यश्चास्त्रविच्चासि शूरश्चासि कुरूद्वह॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7004"><div class="verse-block verse-justify"><div class="line">अस्त्राणि समवाप्तानि त्वया दश च पञ्च च।</div><div class="line" style="position:relative;">पञ्चभिर्विधिभिः पार्थ विद्यते न त्वया समः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7005"><div class="verse-block verse-justify"><div class="line">प्रयोगमुपसंहारमावृत्तिं च धनञ्जय।</div><div class="line" style="position:relative;">प्रायश्चित्तं च वेत्थ त्वं प्रतीघातं च सर्वशः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7006"><div class="verse-block verse-justify"><div class="line">तव गुर्वर्थकालोऽयं समुत्पन्नः परंतप।</div><div class="line" style="position:relative;">प्रतिजानीष्व तं कर्तुं ततो वेत्स्याम्यहं परम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7007"><div class="verse-block verse-justify"><div class="line">ततोऽहमब्रवं राजन्देवराजमिदं वचः।</div><div class="line" style="position:relative;">विषह्यं यन्मया कर्तुं कृतमेव निबोध तत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7008"><div class="verse-block verse-justify"><div class="line">ततो मामब्रवीद्राजन्प्रहसन्बलवृत्रहा।</div><div class="line" style="position:relative;">नाविपह्यं तवाद्यास्ति त्रिषु लोकेषु किंचन॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7009"><div class="verse-block verse-justify"><div class="line">निवातकवचा नाम दानवा मम शत्रवः।</div><div class="line" style="position:relative;">समुद्रकुक्षिमाश्रित्य दुर्गे प्रतिवसन्त्युत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7010"><div class="verse-block verse-justify"><div class="line">तिस्रः कोट्यः समाख्यातास्तुल्यरूपबलप्रभाः।</div><div class="line" style="position:relative;">तांस्तत्रजहि कौन्तेय गुर्वर्थस्ते भविष्यति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7011"><div class="verse-block verse-justify"><div class="line">ततो मातलिसंयुक्तं मयूरसमरोमभिः।</div><div class="line" style="position:relative;">हयैरुपेतं प्रादान्मे रथं दिव्यं महाप्रभम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7012"><div class="verse-block verse-justify"><div class="line">बबन्ध चैव मे मूर्ध्नि किरीटमिदमुत्तमम्।</div><div class="line" style="position:relative;">स्वरूपसदृशं चैव प्रादादङ्गविभूषणम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7013"><div class="verse-block verse-justify"><div class="line">अभेद्यं कवचं चेदं स्पर्सरूपवदुत्तमम्।</div><div class="line" style="position:relative;">अजरां ज्यामिमां चापि गाण्डीवे समयोजयत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7014"><div class="verse-block verse-justify"><div class="line">ततः प्रायामहं तेन स्यन्दनेन विराजता।</div><div class="line" style="position:relative;">येनाजयद्देवपतिर्बलिं वैरोचनिं पुरा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7015"><div class="verse-block verse-justify"><div class="line">ततो देवाः सर्व एव तेन घोषेण बोधिताः।</div><div class="line" style="position:relative;">मन्वाना देवराजं मां समाजग्मुर्विशांपते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7016"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा च मामपृच्छन्त किं करिष्यसि फल्गुन।</div><div class="line" style="position:relative;">तानब्रुवं यथाभूतमिदं कर्ताऽस्मि संयुगे॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7017"><div class="verse-block verse-justify"><div class="line">निवातकवचानां तु प्रस्थितं मां तधैपिणम्।</div><div class="line" style="position:relative;">निबोधत महाभागा शिवं चाशास्त मेऽनघाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7018"><div class="verse-block verse-justify"><div class="line">`ततो वाग्भिः प्रशस्ताभिस्त्रिदशाः पृथिवीपते'।</div><div class="line" style="position:relative;">तुष्टुवुर्मां प्रसन्नास्ते यथा देवं पुरंदरम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7019"><div class="verse-block verse-justify"><div class="line">रथेनानेन मघवा जितवाञ्शम्बरं युधि।</div><div class="line" style="position:relative;">नमुचिं बलवृत्रौ च प्रह्लादनरकावपि॥२२॥</div></div></div>
