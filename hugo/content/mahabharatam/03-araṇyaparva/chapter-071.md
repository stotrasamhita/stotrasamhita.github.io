<div class="verse-block-wrapper" id="03-araṇyaparva-b3069"><div class="verse-block verse-justify"><div class="line">स भीमवचनाद्राजा कुण्डिनं प्राविशत्पुरम्।</div><div class="line" style="position:relative;">नादयन्रथघोषेण सर्वाः स विदिशो दिशः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3070"><div class="verse-block verse-justify"><div class="line">ततस्तं रथनिर्घोषं नलाश्वास्तत्र शुश्रुवुः।</div><div class="line" style="position:relative;">श्रुत्वा तु समहृष्यन्त पुरेव नलसन्निधौ॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3071"><div class="verse-block verse-justify"><div class="line">दमयन्ती तु शुश्राव रथघोषं नलस्य तम्।</div><div class="line" style="position:relative;">यथा मेघस्य नदतो गम्भीरं जलदागमे॥५॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b3073"><div class="verse-block verse-justify"><div class="line">प्रासादस्थाश्च शिखिनः शालास्थाश्चैव वारणाः।</div><div class="line" style="position:relative;">हयाश्च शुश्रुवुस्तस्य रथघोषं महीपतेः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3074"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा रथनिर्घोषं वारणाः शिखिनस्तथा।</div><div class="line">प्रवेदुरुन्मुखा राजन्दृष्ट्वेव जलदोदयम् ॥दमयन्त्युवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3075"><div class="verse-block verse-justify"><div class="line">यथाऽसौ रथनिर्घोषः पूरयन्निव मेदिनीम्।</div><div class="line" style="position:relative;">ममाह्लादयते चेतो नल एव महीपतिः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3076"><div class="verse-block verse-justify"><div class="line">अद्य नन्द्राभवक्रं तं न पश्यामि नलं यदि।</div><div class="line" style="position:relative;">असंख्येयगुणं वीरं विनङ्क्ष्यामि न संशयः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3077"><div class="verse-block verse-justify"><div class="line">यदि वै तस्य वीरस्य हाह्वोर्नाद्याहमन्तरम्।</div><div class="line" style="position:relative;">प्रविशामि सुखस्पर्शं नभविष्याम्यसंशयम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3078"><div class="verse-block verse-justify"><div class="line">यदि मां मेघनिर्घोषो नोपगच्छति नैषधः।</div><div class="line" style="position:relative;">अद्य चामीकरप्रख्यं प्रवेक्ष्यामि हुताशनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3079"><div class="verse-block verse-justify"><div class="line">यदि मां सिंहविक्रान्तो मत्तवारणविक्रमः।</div><div class="line" style="position:relative;">नाभिगच्छतिराजेन्द्रो विनङ्क्ष्यामि न संशयः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3080"><div class="verse-block verse-justify"><div class="line">न स्मराम्यनृतं किंचिन्न स्मराम्यपकारताम्।</div><div class="line" style="position:relative;">न च पर्युषितं वाक्यं स्वैरेष्वपि महात्मनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3081"><div class="verse-block verse-justify"><div class="line">प्रभुः क्षमावान्वीरश्च दाता चाप्यधिको नृपैः।</div><div class="line" style="position:relative;">अहो नीचानुवर्ती च क्लीबवन्मम नैषध॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3082"><div class="verse-block verse-justify"><div class="line">गुणांस्तस्य स्मरन्त्या मे तत्पराया दिवानिशम्।</div><div class="line" style="position:relative;">हृदयं दीर्यत इदं शोकात्प्रियविनाकृतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3083"><div class="verse-block verse-justify"><div class="line">एवं विलपमाना सा नष्टसंज्ञेव भारत।</div><div class="line" style="position:relative;">आरुरोह महद्वेश्म पुण्यश्लोकदिदृक्षया॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3084"><div class="verse-block verse-justify"><div class="line">ततो मध्यमकक्षायां ददर्श रथमास्थितम्।</div><div class="line" style="position:relative;">ऋतुपर्णं महीपालं सहवार्ष्णेयबाहुकम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3085"><div class="verse-block verse-justify"><div class="line">ततोऽवतीर्य वार्ष्णएयो बाहुकश्च रथोत्तमात्।</div><div class="line" style="position:relative;">हयांस्तानवमुच्याथ स्थापयामासतू रथम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3086"><div class="verse-block verse-justify"><div class="line">सोऽवतीर्य रथोपस्थादृतुपर्णो नराधिपः।</div><div class="line" style="position:relative;">उपतस्थे महाराजं भीमं भीमपराक्रमम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3087"><div class="verse-block verse-justify"><div class="line">तं भीमः प्रतिजग्राह पूजया परया मुदा।</div><div class="line" style="position:relative;">स तेन पूजितो राज्ञा ऋतुपर्णो नराधिपः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3088"><div class="verse-block verse-justify"><div class="line">स तत्र कुण्डिने रम्ये वसमानो महीपतिः।</div><div class="line">न च किंचित्तदाऽपश्यत्प्रेक्षमाणो मुहुर्मुहुः।</div><div class="line" style="position:relative;">स तु राज्ञा समागम्य विदर्भपतिना तदा॥२२॥</div></div></div>
