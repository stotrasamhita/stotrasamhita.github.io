<div class="verse-block-wrapper" id="03-araṇyaparva-b1056"><div class="verse-block verse-justify"><div class="line">सुखोचितमदुःखार्हं दुरात्मा ससुहृद्गणः।</div><div class="line" style="position:relative;">ईदृशं दुःखमानीय मोदते पापपूरुषः॥७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b1059"><div class="verse-block verse-justify"><div class="line">इतरेषां तु सर्वेषां कुरूणां कुरुसत्तम।</div><div class="line" style="position:relative;">दुःखेनाभिपरीतानां नेत्रेभ्यः प्रापतज्जलम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1060"><div class="verse-block verse-justify"><div class="line">इदं च शयनं दृष्ट्वा यच्चासीत्ते पुरातनम्।</div><div class="line" style="position:relative;">शोचामित्वां महाराजदुःखानर्हंसुखोचितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1061"><div class="verse-block verse-justify"><div class="line">दान्तं यच्च सभामध्य आसनं रत्नभूषितम्।</div><div class="line" style="position:relative;">दृष्ट्वा कुशबृशीं चेमां शोको मां दारयत्ययम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1062"><div class="verse-block verse-justify"><div class="line">यदपश्यं सभायां त्वां राजभिः परिवारितम्।</div><div class="line" style="position:relative;">तच्च राजन्नपश्यन्त्याः का सान्तिर्हृदयस्यमे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1063"><div class="verse-block verse-justify"><div class="line">या त्वाऽह्रं चन्दनादिग्धमपश्यं सूर्यवर्चसम्।</div><div class="line" style="position:relative;">सा त्वां पङ्कमलादिग्धं दृष्ट्वा मुह्यामि भारत॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1064"><div class="verse-block verse-justify"><div class="line">या त्वाऽहं कौशिकैर्वस्त्रैः शुभ्रैराच्छादितं पुरा।</div><div class="line" style="position:relative;">दृष्टवत्यस्मिराजेनद््र साऽद्य पश्यामि चीरिणम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1065"><div class="verse-block verse-justify"><div class="line">यच्च तद्रुक्मपात्रीभिर्ब्राह्मणेभ्यः सहस्रशः।</div><div class="line" style="position:relative;">ह्रियते ते गृहादन्नं संस्कृतंसार्वकामिकम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1066"><div class="verse-block verse-justify"><div class="line">यतीनामगृहाणां ते तथैव गृहमेधिनाम्।</div><div class="line" style="position:relative;">दीयते भोजनं राजन्नतीव गुणवत्प्रभो॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b1068"><div class="verse-block verse-justify"><div class="line">यत्ते भ्रातॄन्महाराज युवानो मृष्टकुण्डलाः।</div><div class="line" style="position:relative;">अभोजयन्त मृष्टान्नैः सूदाः परमसंस्कृतैः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1069"><div class="verse-block verse-justify"><div class="line">सर्वांस्तानद्य पश्यामि वने वन्येन जीविनः।</div><div class="line" style="position:relative;">अदुःखार्हान्मनुष्येन्द्र नोपशाम्यति मे मनः॥२०॥</div></div></div>
