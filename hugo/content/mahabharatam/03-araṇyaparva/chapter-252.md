<div class="verse-block-wrapper" id="03-araṇyaparva-b10156"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10157"><div class="verse-block verse-justify"><div class="line">प्रायोपविष्टं राजानं दुर्योधनममर्षणम्।</div><div class="line" style="position:relative;">उवाच सान्त्वयन्राजञ्शकुनिः सौबलस्तदा॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10158"><div class="verse-block verse-justify"><div class="line">सम्यगुक्तं हि कर्णेन तच्छ्रुतं कौरव त्वया।</div><div class="line" style="position:relative;">मया हृतां श्रियं स्फीतां तां मोहादपहासि किम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10159"><div class="verse-block verse-justify"><div class="line">त्वमबुद्ध्या नृपवर प्राणानुत्स्रष्टुमिच्छसि।</div><div class="line" style="position:relative;">अद्य वाऽप्यवगन्छामि न वृद्धाः सेवितास्त्वया॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10160"><div class="verse-block verse-justify"><div class="line">यः समुत्पतितं हर्षं दैन्यं वा न नियच्छति।</div><div class="line" style="position:relative;">स नश्यति श्रियं प्राप्य पात्रमाममिवाम्भसि॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10161"><div class="verse-block verse-justify"><div class="line">अतिभीरुं मृदुं क्लीबं दीर्घसूत्रं प्रमादिनम्।</div><div class="line" style="position:relative;">व्यसनाद्विषयाक्रान्तं न भजन्ति नृपं श्रिताः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10162"><div class="verse-block verse-justify"><div class="line">सत्कृतस्य हि ते शोको विपरीते कथं भवेत्।</div><div class="line" style="position:relative;">मा कृतं शोभनं पार्थैः शोकमालम्ब्य नाशय॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10163"><div class="verse-block verse-justify"><div class="line">यत्र हर्षस्त्वया कार्यः सत्कर्तव्याश्च पाण्डवाः।</div><div class="line" style="position:relative;">तत्र शोचसि राजेन्द्रविपरीतमिदं तव॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10164"><div class="verse-block verse-justify"><div class="line">प्रसीदमा त्यजात्मानं तुष्टश्च सुकृतं स्मर।</div><div class="line">कप्रयच्छ राज्यं पार्थानां यशो धर्ममवाप्नुहि।</div><div class="line" style="position:relative;">क्रियामेतां समाज्ञाय कृतघ्नो न भविष्यसि॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10165"><div class="verse-block verse-justify"><div class="line">सौभ्रात्रं पाण्डवैः कृत्वा समवस्ताप्य चैव तान्।</div><div class="line">पित्र्यं राज्यं प्रयच्छैषां ततः सुखमवाप्स्यसि ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10166"><div class="verse-block verse-justify"><div class="line">शकुनेस्तु वचः श्रुत्वा दुःशासनमवेक्ष्य च।</div><div class="line" style="position:relative;">पादयोः पतितं वीरं विकृतं भ्रातृसौहृदात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10167"><div class="verse-block verse-justify"><div class="line">बाहुभ्यां साधुजाताभ्यां दुःशासनमरिंदमम्।</div><div class="line" style="position:relative;">उत्थाप्य संपरिष्वज्य प्रीत्याऽजिघ्रत मूर्धनि॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10168"><div class="verse-block verse-justify"><div class="line">कर्णसौबलयोश्चापि संश्रुत्य वचनान्यसौ।</div><div class="line">निर्वेदं परमं गत्वा राजा दुर्योधनस्तदा।</div><div class="line" style="position:relative;">व्रीडयाऽभिपरीतात्मा नैराश्यमगमत्परम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10169"><div class="verse-block verse-justify"><div class="line">सुहृदां चैव तच्छ्रुत्वा समन्युरिदमब्रवीत्।</div><div class="line" style="position:relative;">न धर्मधनसौख्येन नैश्वर्येण न चाज्ञया॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10170"><div class="verse-block verse-justify"><div class="line">नैव भोगैश्च मे कार्यं मा विहन्यत गच्छत।</div><div class="line" style="position:relative;">निश्चितेयं मम मतिः स्थिता प्रायोपवेशने॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10171"><div class="verse-block verse-justify"><div class="line">गच्छध्वं नगरं सर्वे पूज्याश्च गुरवो मम।</div><div class="line" style="position:relative;">त एवमुक्ताः प्रत्यूच् राजानमरिमर्दनम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10172"><div class="verse-block verse-justify"><div class="line">या गतिस्तव राजेन्द्र साऽस्माकमपि भारत।</div><div class="line">कथं वा संप्रवेक्ष्यामस्त्वद्विहीनाः पुरं वयम् ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10173"><div class="verse-block verse-justify"><div class="line">स सुहृद्भिरमात्यैश्च भातृभिः स्वजनेन च।</div><div class="line" style="position:relative;">बहुप्रकारमप्युक्तो निश्चयान्न व्यचाल्यत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10174"><div class="verse-block verse-justify"><div class="line">दर्भास्तरणमास्तीर्य निश्चयाद्धृतराष्ट्रजः।</div><div class="line" style="position:relative;">संस्पृश्यापः शुचिर्भूत्वा भूतले समुपस्थितः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10175"><div class="verse-block verse-justify"><div class="line">कुशचीराम्बरधरः परं नियममास्थितः।</div><div class="line" style="position:relative;">वाग्यतो राजशार्दूलः स स्वर्गगतिकाम्यया॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10176"><div class="verse-block verse-justify"><div class="line">मनसोपचितिं कृत्वा निरस् च बहिःक्रियाः।</div><div class="line" style="position:relative;">`तस्थौप्रायोपवेशेऽथमतिं कृत्वा सुनिश्चयाम्'॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10177"><div class="verse-block verse-justify"><div class="line">अथ तं निश्चयं तस्य बुद्ध्वा दैतेयदानवाः।</div><div class="line" style="position:relative;">पातालवासिनो रौद्राः पूर्वं देवैर्विनिर्जिताः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10178"><div class="verse-block verse-justify"><div class="line">ते स्वपक्षक्षयं तं तु ज्ञात्वा दुर्योधनस्य वै।</div><div class="line" style="position:relative;">आह्वानाय तदा चक्रुः कर्म वैतानसंभवम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10179"><div class="verse-block verse-justify"><div class="line">बृहस्पत्युशनोक्तैश्च मन्त्रैर्मन्त्रविशारदाः।</div><div class="line">अथर्ववेदप्रोक्तैश्च याश्चौपनिषदाः क्रियाः।</div><div class="line" style="position:relative;">मन्त्रजप्यसमायुक्तास्तास्तदा समवर्तयन्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10180"><div class="verse-block verse-justify"><div class="line">जुह्वत्यग्नौ हविः क्षीरं मन्त्रवत्सुसमाहिताः।</div><div class="line" style="position:relative;">ब्राह्मणा वेदवेदाङ्गपारगाः सुदृढव्रताः॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b10183"><div class="verse-block verse-justify"><div class="line">आहुर्दैत्याश्च तां तत्र सुप्रीतेनान्तरात्मना।</div><div class="line" style="position:relative;">प्रायोपविष्टं राजानं धार्तराष्ट्रमिहानय॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10184"><div class="verse-block verse-justify"><div class="line">तथेति च प्रतिश्रुत्य सा कृत्या प्रययौ तदा।</div><div class="line" style="position:relative;">निमेषादगमच्चापि यत्र राजा सुयोधनः॥२८॥</div></div></div>
