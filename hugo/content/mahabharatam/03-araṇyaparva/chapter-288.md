<div class="verse-block-wrapper" id="03-araṇyaparva-b11543"><div class="verse-block verse-justify"><div class="line">मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11544"><div class="verse-block verse-justify"><div class="line">ततो निर्याय स्वपुरात्कुम्भकर्णः सहानुगः।</div><div class="line" style="position:relative;">अपश्यत्कपिसैन्यं रतज्जितकाश्यग्रतः स्थितम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11545"><div class="verse-block verse-justify"><div class="line">स वीक्षमाणस्तत्सैन्यं रामदर्शनकाङ्क्षया।</div><div class="line" style="position:relative;">अपश्यच्चापि सौमित्रिं धनुष्पाणिं व्यवस्थितम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11546"><div class="verse-block verse-justify"><div class="line">तमभ्येत्याशु हरयः परिवब्रुः समन्ततः।</div><div class="line" style="position:relative;">`शैलवृक्षायुधा नादानमुञ्चन्भीषणास्ततः'॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11547"><div class="verse-block verse-justify"><div class="line">अभ्यघ्नंश्च महाकायैर्बहुभिर्जगतीरुहैः।</div><div class="line" style="position:relative;">करजैरतुदंश्चान्ये विहाय भयमुत्तमम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11548"><div class="verse-block verse-justify"><div class="line">बहुधा युध्यमानास्ते युद्धमार्गैः प्लवंगमाः।</div><div class="line" style="position:relative;">नानाप्रहरणैर्भीमै राक्षसेन्द्रमताडयन्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11549"><div class="verse-block verse-justify"><div class="line">स ताड्यमानः प्रहसन्भक्षयामास वानरान्।</div><div class="line" style="position:relative;">बलं चण्डबलाख्यं च वज्रबाहुं च वानरम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11550"><div class="verse-block verse-justify"><div class="line">तद्दृष्ट्वा व्यथनं कर्म कुम्भकर्णस्य रक्षसः।</div><div class="line" style="position:relative;">उदक्रोशन्परित्रस्तास्तारप्रभृतयस्तदा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11551"><div class="verse-block verse-justify"><div class="line">तानुच्चैः क्रोशतः सैन्याञ्श्रुत्वा स हरियूथपान्।</div><div class="line" style="position:relative;">अभिदुद्राव सुग्रीवः कुम्भकर्णमपेतभीः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11552"><div class="verse-block verse-justify"><div class="line">ततो निपत्य वेगेन कुम्भकर्णं महामना।</div><div class="line" style="position:relative;">सालेन जघ्निवान्मूर्ध्निं बलेन कपिकुञ्जरः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11553"><div class="verse-block verse-justify"><div class="line">स महात्मा महावेगः कुम्भकर्णस् मूर्धनि।</div><div class="line" style="position:relative;">बिभेद सालं सुग्रीवो न चैवाव्यथयत्कपिः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11554"><div class="verse-block verse-justify"><div class="line">ततो विनद्यसहसा सालस्पर्शविबोधितः।</div><div class="line" style="position:relative;">दोर्भ्यामादाय सुग्रीवं कुम्भकर्णोऽहरद्बलात्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11555"><div class="verse-block verse-justify"><div class="line">ह्रियमाणं तु सुग्रीवं कुम्भकर्णेन रक्षसा।</div><div class="line" style="position:relative;">अवेक्ष्याभ्यद्रवद्वीरः सौमित्रिर्मित्रनन्दनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11556"><div class="verse-block verse-justify"><div class="line">सोऽभिपत्य महर्वेगं रुक्मपुङ्खं महाशरम्।</div><div class="line" style="position:relative;">प्राहिणोत्कुम्भकर्णाय लक्ष्मणः परवीरहा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11557"><div class="verse-block verse-justify"><div class="line">स तस्य देहावरणं भित्त्वा देहं च सायकः।</div><div class="line" style="position:relative;">जगाम दारयन्भूमिं रुधिरेण समुक्षितः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11558"><div class="verse-block verse-justify"><div class="line">तथा स भिन्नहृदयः समुत्सृज्य कपीश्वरम्।</div><div class="line" style="position:relative;">`वेगेन महताऽऽविष्टस्तिष्ठतिष्ठेति चाब्रवीत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11559"><div class="verse-block verse-justify"><div class="line">कुम्भकर्णो महेष्वासः प्रगृहीतशिलायुधः।</div><div class="line" style="position:relative;">अभिदुद्राव सौमित्रिमुद्यम्य महतीं शिलाम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11560"><div class="verse-block verse-justify"><div class="line">तस्याभिपततस्तूर्णं क्षुराभ्यामुच्छितौ करौ।</div><div class="line" style="position:relative;">चिच्छेद निशिताग्राभ्यां स बभूव चतुर्भुजः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11561"><div class="verse-block verse-justify"><div class="line">तानप्यस् भुजान्सर्वान्प्रगृहीतशिलायुधान्।</div><div class="line" style="position:relative;">क्षुरैश्चिच्छेदलघ्वस्त्रं सौमित्रिः प्रतिदर्शयन्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11562"><div class="verse-block verse-justify"><div class="line">स बभूवातिकायश्च बहुपादशिरोभुजः।</div><div class="line" style="position:relative;">तं ब्रह्मास्त्रेण सौमित्रिर्ददाराद्रिचयोपमम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11563"><div class="verse-block verse-justify"><div class="line">स पपात महावीर्यो दिव्यास्त्राभिहतो रणे।</div><div class="line" style="position:relative;">महाशनिविनिर्दग्धः पादपोऽङ्कुरवानिव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11564"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा वृत्रसंकाशं कुम्भकर्णं तरस्विनम्।</div><div class="line" style="position:relative;">गतासुं पतितं भूमौ राक्षसाः प्राद्रवन्भयात्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11565"><div class="verse-block verse-justify"><div class="line">तथातान्द्रवतो योधान्दृष्ट्वा तौ दूषणानुजौ।</div><div class="line" style="position:relative;">अवस्थाप्याथ सौमित्रिं संक्रुद्धावभ्यधावताम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11566"><div class="verse-block verse-justify"><div class="line">तावाद्रवन्तौ संक्रुद्धौ वज्रवेगप्रमाथिनौ।</div><div class="line" style="position:relative;">अभिजग्राह सौमित्रिर्विनद्योभौ पतत्रिभिः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11567"><div class="verse-block verse-justify"><div class="line">ततः सुतुमुलं युद्धमभवद्रोमहर्षणम्।</div><div class="line" style="position:relative;">दूषणानुजयोः पार्थ लक्ष्मणस् च धीमतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11568"><div class="verse-block verse-justify"><div class="line">महता शरवर्षेण राक्षसौ सोऽभ्यवर्पत।</div><div class="line" style="position:relative;">तं चापिवीरौ संक्रुद्धावुभौ तौ समवर्षताम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11569"><div class="verse-block verse-justify"><div class="line">मुहूर्तमेवमभवद्वज्रवेगप्रमाथिनोः।</div><div class="line" style="position:relative;">सौमित्रेश्च महाबाहोः संप्रहारः सुदारुणः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11570"><div class="verse-block verse-justify"><div class="line">अथाद्रिशृङ्गमादाय हनुमान्मारुतात्मजः।</div><div class="line" style="position:relative;">अभिद्रुत्याददे प्राणान्वज्रवेगस्य रक्षसः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11571"><div class="verse-block verse-justify"><div class="line">नीलश्च महता ग्राव्णा दूपणावरजं हरिः।</div><div class="line" style="position:relative;">प्रमाथिनमभिद्रुत्य प्रममाथ महाबलः॥२९॥</div></div></div>
