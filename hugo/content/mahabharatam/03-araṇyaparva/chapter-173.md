<div class="verse-block-wrapper" id="03-araṇyaparva-b7079"><div class="verse-block verse-justify"><div class="line">अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7080"><div class="verse-block verse-justify"><div class="line">ततोऽश्मवर्षं सुमहत्प्रादुरासीत्समन्ततः।</div><div class="line" style="position:relative;">नगमात्रैः शिलाखण्डैस्तन्मां दृढमपीडयत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7081"><div class="verse-block verse-justify"><div class="line">तदहं वज्रसंकाशैः शरैरिनद्रास्त्रचोदितैः।</div><div class="line" style="position:relative;">अचूर्णयं वेगवद्भिः शतधैकैकमाहवे॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7082"><div class="verse-block verse-justify"><div class="line">चूर्ण्यमानेऽश्मवर्षे तु यावकः समजायत।</div><div class="line" style="position:relative;">तत्राश्मचूर्णा न्यपतन्पावकप्रकरा इव॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7083"><div class="verse-block verse-justify"><div class="line">ततोऽश्मवर्षे विहते जलवर्षं महत्तरम्।</div><div class="line" style="position:relative;">धाराभिरक्षमात्राभिः प्रादुरासीनममान्तिके॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7084"><div class="verse-block verse-justify"><div class="line">नभस प्रच्युता धारास्तिग्मवीर्याः सहस्रशः।</div><div class="line" style="position:relative;">आवृण्वन्सर्वतो व्योम दिशश्चोपदिशस्तथा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7085"><div class="verse-block verse-justify"><div class="line">धाराणां च निपातेन वायोर्विष्फूर्जितेन च।</div><div class="line" style="position:relative;">गर्जितेन च मेघानां न प्राज्ञायत किंचन॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7086"><div class="verse-block verse-justify"><div class="line">धारा दिवि च संबद्धा वसुधायां च सर्वशः।</div><div class="line" style="position:relative;">व्यामोहयन्त मां तत्रनिपतन्त्योऽनिशं भुवि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7087"><div class="verse-block verse-justify"><div class="line">तत्रोपदिष्टमिन्द्रेण दिव्यमस्त्रं विशोषणम्।</div><div class="line" style="position:relative;">दीप्तं प्राहिणवं घोरमशुप्यत्तेन तज्जलम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7088"><div class="verse-block verse-justify"><div class="line">हतेऽश्मवर्षे च मया जलवर्षे च शोषिते।</div><div class="line" style="position:relative;">मुमुचुर्दानवा मायामग्निं वायुं च भारत॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7089"><div class="verse-block verse-justify"><div class="line">ततोऽहमग्निं व्यधमं सलिलास्त्रेण सर्वशः।</div><div class="line" style="position:relative;">शैलेन च महास्त्रेण वायोरवेगमधारयम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7090"><div class="verse-block verse-justify"><div class="line">तस्यां प्रतिहतायां ते दानवा युद्धदुर्मदाः।</div><div class="line" style="position:relative;">प्राकुर्वन्विविधां मायां यौगपद्येन भारत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7091"><div class="verse-block verse-justify"><div class="line">ततो वर्षं प्रादुरभूत्सुमहद्रोमहर्षणम्।</div><div class="line" style="position:relative;">अस्त्राणां घोररूपाणामग्नेर्वायोस्तथाऽश्मनाम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7092"><div class="verse-block verse-justify"><div class="line">सा तु मायामयी वृष्टिः पीडयामास मां युधि।</div><div class="line" style="position:relative;">अथ घोरं तमस्तीव्रं प्रादुरासीत्समन्ततः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7093"><div class="verse-block verse-justify"><div class="line">तमसा संवृतेलोके घोरेण परुषेण च।</div><div class="line" style="position:relative;">हरयो विमुखाश्चासन्प्रास्खलच्चापि मातलिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7094"><div class="verse-block verse-justify"><div class="line">हस्ताद्धिरण्मयश्चास्य प्रतोदः प्रापतद्भुवि।</div><div class="line" style="position:relative;">असकृच्चाह मां भीतः किं करिष्याव इत्यपि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7095"><div class="verse-block verse-justify"><div class="line">मां च भीराविशत्तीव्रा तस्मिन्विगतचेतसि।</div><div class="line" style="position:relative;">स च मां विगतज्ञानः संत्रस्तमिदमब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7096"><div class="verse-block verse-justify"><div class="line">सुराणामसुराणां च संग्रामः सुमहानभूत्।</div><div class="line" style="position:relative;">अमृतार्थं पुरा पार्त स च दृष्टो मयाऽनघ॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7097"><div class="verse-block verse-justify"><div class="line">शम्बरस्य वधे घोरः संग्रामः सुमहानभूत्।</div><div class="line" style="position:relative;">सारथ्यं देवराजस्य तत्रापि कृतवानहम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7098"><div class="verse-block verse-justify"><div class="line">तथैव वृत्रस्य वधे संगृहीता हया मया।</div><div class="line" style="position:relative;">वैरोचनेर्मया युद्धं दृष्टं चापि सुदारुणम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7099"><div class="verse-block verse-justify"><div class="line">एते मया महाघोरः संग्रामाः पर्युपासिताः।</div><div class="line" style="position:relative;">न चापि विगतज्ञानो भूतपूर्वोस्मि पाण्डव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7100"><div class="verse-block verse-justify"><div class="line">पितामहेन संहारः प्रजानां विहितो ध्रुवम्।</div><div class="line" style="position:relative;">न हि युद्धमिदं युक्तमन्यत्र जगतः क्षयात्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7101"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा संस्तभ्यात्मानमात्मना।</div><div class="line" style="position:relative;">मोहयिष्यन्दानवानामहं मायाबलं महत्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7102"><div class="verse-block verse-justify"><div class="line">अब्रुवं मातलिं भीतं पश्य मे बुजयोर्बलम्।</div><div class="line" style="position:relative;">अस्त्राणां च प्रभावं वै धनुषो गाण्डिवस्य च॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7103"><div class="verse-block verse-justify"><div class="line">अद्यास्त्रमाययैतैषां मायामेतां सुदारुणाम्।</div><div class="line" style="position:relative;">विनिहन्मि तमश्चोग्रं मा भैः सूत स्तिरो भव॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7104"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वाहमसृजमस्त्रमायां नराधिप।</div><div class="line" style="position:relative;">मोहनीं सर्वभूतानां हिताय त्रिदिवौकसाम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7105"><div class="verse-block verse-justify"><div class="line">पीड्यमानासु मायासु तासु तास्वसुरोत्तमाः।</div><div class="line" style="position:relative;">पुनर्बहुविधा मायाः प्राकुर्वन्नमितौजसः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7106"><div class="verse-block verse-justify"><div class="line">पुनः प्रकाशमभवत्तमसा ग्रस्यते पुनः।</div><div class="line" style="position:relative;">भवत्यदर्शनो लोकः पुनरप्सु निमज्जति॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7107"><div class="verse-block verse-justify"><div class="line">सुसंगृहीतैर्हरिभिः प्रकाशे सति मातलिः।</div><div class="line" style="position:relative;">व्यचरत्स्यन्दनाग्र्येण संग्रामे रोमहर्षणे॥२९॥</div></div></div>
