<div class="verse-block-wrapper" id="03-araṇyaparva-b4177"><div class="verse-block verse-justify"><div class="line">एवं संभाषमाणे तु धौम्ये कौरवनन्दन।</div><div class="line" style="position:relative;">लोमशः स महातेजा ऋषिस्तत्राजगाम ह॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4178"><div class="verse-block verse-justify"><div class="line">तं पाण्डवाग्रजो राजा सगणो ब्राह्मणाश्च ते।</div><div class="line" style="position:relative;">उपातिष्ठन्महाभागं दिवि शक्रमिवामराः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4179"><div class="verse-block verse-justify"><div class="line">समभ्यर्च्य यथान्यायं धर्मपुत्रो युधिष्ठिरः।</div><div class="line" style="position:relative;">पप्रच्छागमने हेतुमटने च प्रयोजनम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4180"><div class="verse-block verse-justify"><div class="line">स पृष्टः पाण्डुपुत्रेण प्रीयमाणो महामनाः।</div><div class="line" style="position:relative;">उवाच श्लक्ष्णया वाचा हर्षयन्निव पाण्डवान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4181"><div class="verse-block verse-justify"><div class="line">संचरन्नस्मि कौन्तेय सर्वांल्लोकान्यदृच्छया।</div><div class="line" style="position:relative;">गतः शक्रस्य भवनं तत्रापश्यं सुरेश्वरम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4182"><div class="verse-block verse-justify"><div class="line">तव च भ्रातरं वीरमपश्यं सव्यसाचिनम्।</div><div class="line" style="position:relative;">शक्रस्यार्धासनगतं तत्र मे विस्मयो महान्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4183"><div class="verse-block verse-justify"><div class="line">आसीत्पुरुषशार्दूल दृष्ट्वा पार्थं तथागतम्।</div><div class="line" style="position:relative;">आह मां तत्र देवेशो गच्छ पाण्डुसुतान्प्रति॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4184"><div class="verse-block verse-justify"><div class="line">सोऽहमभ्यागतः क्षिप्रं दिदृक्षस्त्वां सहानुजम्।</div><div class="line" style="position:relative;">वचनात्पुरुहूतस्य पार्थस् च महात्मनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4185"><div class="verse-block verse-justify"><div class="line">आख्यास्ये ते प्रियं तात महत्पाण्डवनन्दन।</div><div class="line" style="position:relative;">भ्रातृभिः सहितो राजन्कृष्णया चैव तच्छृणु॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4186"><div class="verse-block verse-justify"><div class="line">यत्त्वयोक्तो महाबाहुरस्त्रार्थं भरतर्षभ।</div><div class="line" style="position:relative;">तदस्त्रमाप्तं पार्थेन रुद्रादप्रतिमं विभो॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4187"><div class="verse-block verse-justify"><div class="line">यत्तद्ब्रह्मशिरो नाम तपसा रुद्रमागमत्।</div><div class="line" style="position:relative;">अमृतादुत्थितं रौद्रं तल्लब्धं सव्यसाचिना॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4188"><div class="verse-block verse-justify"><div class="line">तत्समन्त्रं ससंहारं सप्रायश्चित्तमङ्गलम्।</div><div class="line" style="position:relative;">वज्रमस्त्राणि चान्यानि दण्डादीनि युधिष्ठिर॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4189"><div class="verse-block verse-justify"><div class="line">यमात्कुबेराद्वरुणादिन्द्राच्च कुरुनन्दन।</div><div class="line" style="position:relative;">अस्त्राण्यधीतवान्पार्थो दिव्यान्यमितविक्रमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4190"><div class="verse-block verse-justify"><div class="line">विश्वावसोस्तु तनयाद्गीतं नृत्यं च साम च।</div><div class="line" style="position:relative;">वादित्रं च यथान्यायं प्रत्यविन्दद्यथाविधि॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4191"><div class="verse-block verse-justify"><div class="line">एवं कृतास्त्रः कौन्तेयो गान्धर्वं वेदमात्मवान्।</div><div class="line" style="position:relative;">सुखं वसति बीभत्सुरनुजस्यानुजस्तव॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4192"><div class="verse-block verse-justify"><div class="line">यदर्थं मां सुरश्रेष्ठ इदं वचनमब्रवीत्।</div><div class="line" style="position:relative;">तच्च ते कथयिष्यामि युधिष्ठिर निबोध मे॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4193"><div class="verse-block verse-justify"><div class="line">भवान्मनुष्यलोकेऽपि गमिष्यति न संशयः।</div><div class="line" style="position:relative;">ब्रूयाद्युधिष्ठिरं तत्रवचनान्मे द्विजोत्तम॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4194"><div class="verse-block verse-justify"><div class="line">आगमिष्यति ते भ्राता कृतास्त्रः क्षिप्रमर्जुनः।</div><div class="line" style="position:relative;">सुरकार्यं महत्कृत्वा यदशक्यं दिवौकसैः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4195"><div class="verse-block verse-justify"><div class="line">तपसाऽपि त्वमात्मानं योजय भ्रातृभिः सह।</div><div class="line" style="position:relative;">तपसो हि परं नास्ति तपसा विन्दते महत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4196"><div class="verse-block verse-justify"><div class="line">अहं च कर्णं जानामि यथावद्भरतर्षभ।</div><div class="line" style="position:relative;">सत्यसन्धं महोत्साहं महावीर्यं महाबलम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4197"><div class="verse-block verse-justify"><div class="line">महाहवेष्वप्रतिमं महायुद्धविशारदम्।</div><div class="line" style="position:relative;">महाधनुर्धरं वीरं महास्त्रं वरवर्णिनम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4198"><div class="verse-block verse-justify"><div class="line">महेश्वरसुतप्रख्यमादित्यतनयं प्रभुम्।</div><div class="line" style="position:relative;">तथाऽर्जुनमतिस्कन्धं सहजोल्वणपौरुणम्॥२३॥</div></div></div>
