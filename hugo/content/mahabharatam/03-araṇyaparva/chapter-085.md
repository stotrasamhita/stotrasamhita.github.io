<div class="verse-block-wrapper" id="03-araṇyaparva-b4062"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4063"><div class="verse-block verse-justify"><div class="line">तान्सर्वानुत्सुकान्दृष्ट्वा पाण्डवान्दीनचेतसः।</div><div class="line" style="position:relative;">अश्वासयंस्तथा धौम्यो बृहस्पतिसमोऽब्रवीत्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4064"><div class="verse-block verse-justify"><div class="line">ब्राह्मणानुमतान्पुण्यानाश्रमान्भरतर्षभ।</div><div class="line" style="position:relative;">दिशस्तीर्थानि शैलांश्च शृणु मे वदतोऽनघ॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4065"><div class="verse-block verse-justify"><div class="line">याञ्श्रुत्वा गदतो राजन्विशोको भवितासि ह।</div><div class="line" style="position:relative;">द्रौपद्या चानया सार्धं भ्रातृभिश्च नरेश्वर॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4066"><div class="verse-block verse-justify"><div class="line">श्रवणाच्चैव तेषां त्वं पुण्यमाप्स्यसि पाण्डव।</div><div class="line" style="position:relative;">गत्वा शतगुणं चैव तेभ्य एव नरोत्तम॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4067"><div class="verse-block verse-justify"><div class="line">शृणु पूर्वां दिशं राजन्देवर्षिगणसेविताम्।</div><div class="line" style="position:relative;">रम्यां ते कथयिष्यामि युधिष्ठिर यथास्मृति॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4068"><div class="verse-block verse-justify"><div class="line">तस्यां रदेवर्षिजुष्टायां नैमिषं नाम भारत।</div><div class="line" style="position:relative;">यत्रतीर्थानि देवानां पुण्यानि च पृथक् पृथक्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4069"><div class="verse-block verse-justify"><div class="line">यत्र सा गोमती पुण्या रम्या देवर्षिसेविता।</div><div class="line" style="position:relative;">यज्ञभूमिश्च देवानां शामित्रं च विवस्वतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4070"><div class="verse-block verse-justify"><div class="line">तस्यां गिरिवरः पुण्यो गयो राजर्षिसत्कृतः।</div><div class="line" style="position:relative;">शिवं ब्रह्मसरो यत्रसेवितं त्रिदशर्षिभिः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4071"><div class="verse-block verse-justify"><div class="line">यदर्थे पुरुषव्याघ्र कीर्तयन्ति पुरातनाः।</div><div class="line" style="position:relative;">एष्टव्या बहवः पुत्रा यद्येकोपि गयां व्रजेत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4072"><div class="verse-block verse-justify"><div class="line">गौरीं वा वरयेत्कन्यां नीलं वा वृषमुत्सृजेत्।</div><div class="line" style="position:relative;">उत्तारयति संतत्या दश पूर्वान्दशावरान्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4073"><div class="verse-block verse-justify"><div class="line">महानदी च तत्रैव तथा गयशिरो नृप।</div><div class="line" style="position:relative;">यत्रासौ कीर्त्यते विप्रैरक्षय्यकरणो वटः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4074"><div class="verse-block verse-justify"><div class="line">यत्रदत्तं पितृभ्योऽन्नमक्षय्यं भवति प्रभो।</div><div class="line" style="position:relative;">सा च पुण्यजला तत्र फल्गुनामा महानदी॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4075"><div class="verse-block verse-justify"><div class="line">बहुमूलफला चापि कौशिकी भरतर्षभ।</div><div class="line" style="position:relative;">विश्वामित्रोऽध्यगाद्यत्र ब्राह्मणत्वं तपोधनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4076"><div class="verse-block verse-justify"><div class="line">गङ्गा यत्रनदी पुण्या यस्यास्तीरे भगीरथः।</div><div class="line" style="position:relative;">अयजत्तत्रबहुभिः क्रतुभिर्भूरिदक्षिणैः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4077"><div class="verse-block verse-justify"><div class="line">पाञ्चालेषु च कौरव्य कथयन्त्युत्पलावतम्।</div><div class="line" style="position:relative;">विश्वामित्रोऽयजद्यत्र शक्रेण सह कौशिकः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4078"><div class="verse-block verse-justify"><div class="line">यत्रानुवंशं भगवाञ्जामदग्न्यस्तथा जगौ।</div><div class="line" style="position:relative;">विश्वामित्रस्य तां दृष्ट्वा चिभूतिमतिमानुषीम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4079"><div class="verse-block verse-justify"><div class="line">कान्यकुब्जेऽपिबत्सोममिन्द्रेण सह कौशिकः।</div><div class="line" style="position:relative;">ततः क्षत्रादपाक्रामद्ब्राह्मणोस्मीति चाब्रवीत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4080"><div class="verse-block verse-justify"><div class="line">पवित्रमृषिभिर्जुष्टं पुण्यं पावनमुत्तमम्।</div><div class="line" style="position:relative;">गङ्गायमुनयोर्वीर संगमं लोकविश्रुतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4081"><div class="verse-block verse-justify"><div class="line">यत्रायजत भूतात्मा पूर्वमेव पितामहः।</div><div class="line" style="position:relative;">प्रयागमिति विख्यातं तस्माद्भरतसत्तम॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4082"><div class="verse-block verse-justify"><div class="line">अगस्त्यस्य तु राजेन्द्र तत्राश्रमवरो नृप।</div><div class="line" style="position:relative;">तत्तथा तापसारण्यं तापसैरुपशोभितम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4083"><div class="verse-block verse-justify"><div class="line">हिरण्यबिन्दुः कथितो गिरौ कालञ्जरे महान्।</div><div class="line" style="position:relative;">आगस्त्यपर्वतोरभ्यः पुण्यो गिरिवः शिवः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4084"><div class="verse-block verse-justify"><div class="line">महेन्द्रो नाम कौरव्य भार्गवस्य महात्मनः।</div><div class="line" style="position:relative;">अयजत्तत्रकौन्तेय पूर्वमेव पितामहः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4085"><div class="verse-block verse-justify"><div class="line">यत्रभागीरथी पुण्यां सरस्यासीद्युधिष्ठिर।</div><div class="line">यत्र सा ब्रह्मशालेति पुण्याख्याता विशांपते।</div><div class="line" style="position:relative;">धूतपाप्मभिराकीर्णा पुण्यं तस्याश्च दर्शनम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4086"><div class="verse-block verse-justify"><div class="line">पवित्रो मङ्गलीयश्च ख्यातो लोके सनातनः।</div><div class="line" style="position:relative;">केदारश्च मतङ्गस्य महानाश्रम उत्तमः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4087"><div class="verse-block verse-justify"><div class="line">कुण्डोदः पर्वतो रम्यो बहुमूलफलोदकः।</div><div class="line" style="position:relative;">नैषधस्तृषितो यत्र जलं शर्म च लब्धवान्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4088"><div class="verse-block verse-justify"><div class="line">यत्र देववनं पुण्यं तापसैरुपशोभितम्।</div><div class="line" style="position:relative;">बाहुदा च नदी यत्रनन्दा च गिरिमूर्धनि॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4089"><div class="verse-block verse-justify"><div class="line">तीर्थानि सरितः शैलाः पुण्यान्यायतनानि च।</div><div class="line" style="position:relative;">प्राच्यां दिशि महाराज कीर्तितानि मया तव॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4090"><div class="verse-block verse-justify"><div class="line">तिसृष्वन्यासु पुण्यानि दिक्षु तीर्थानि मे शृणु।</div><div class="line" style="position:relative;">सरितः पर्वतांश्चैव पुण्यान्यायतनानि च॥२९॥</div></div></div></div>
