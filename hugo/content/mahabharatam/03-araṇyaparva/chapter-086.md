<div class="verse-block-wrapper" id="03-araṇyaparva-b4094"><div class="verse-block verse-justify"><div class="line">यस्यामाख्यायते पुण्या दिशि गोदावरी नदी।</div><div class="line" style="position:relative;">बह्वारामा बहुजला तापसाचरिता शिवा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4095"><div class="verse-block verse-justify"><div class="line">वेणा भीमरथी चैव नद्यौ पापभयापहे।</div><div class="line" style="position:relative;">मृगद्विजसमाकीर्णे तापसालयभूषिते॥४॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b4097"><div class="verse-block verse-justify"><div class="line">अपि चात्र महायोगी मार्कण्डेयो महायशाः।</div><div class="line" style="position:relative;">अनुवंश्यां जगौ गाथां नृगस्य धरणीपतेः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4098"><div class="verse-block verse-justify"><div class="line">नृगस् यजमानस्य प्रत्यक्षमिति नः श्रुतम्।</div><div class="line" style="position:relative;">`मरुतः परिवेष्टारः सदस्याश्च दिवौकसः'॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4099"><div class="verse-block verse-justify"><div class="line">पयोष्ण्यां यजमानस्य वाराहे तीर्थ उत्तमे।</div><div class="line">उद्धृतं भूतलस्थं वावायुना समुदीरितम्।</div><div class="line" style="position:relative;">पयोष्ण्या हरते तोयं पापमामरणान्तिकम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4100"><div class="verse-block verse-justify"><div class="line">स्वर्गादुत्तुङ्गममलं विषाणं यत्र शूलिनः।</div><div class="line" style="position:relative;">स्वमात्मविहितं दृष्ट्वा मर्त्यः शिवपुरं व्रजेत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4101"><div class="verse-block verse-justify"><div class="line">एकतः सरितः सर्वा गङ्गाद्याः सलिलोच्चयाः।</div><div class="line" style="position:relative;">पयोष्णी चैकतः पुण्या तीर्थेभ्यो हि मता मम॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4102"><div class="verse-block verse-justify"><div class="line">माठरस्य वनं पुण्यं बहुमूलफलं शिवम्।</div><div class="line" style="position:relative;">यूपश्च भरतश्रेष्ठ वरुणस्रोतसे गिरौ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4103"><div class="verse-block verse-justify"><div class="line">प्रवेण्युत्तरपार्स्वे तु पुण्ये कण्वाश्रमे तथा।</div><div class="line" style="position:relative;">तापसानामरण्यानि कीर्तितानि यथाश्रुति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4104"><div class="verse-block verse-justify"><div class="line">वेदी शूर्पारके तात जमदग्नेर्महात्मनः।</div><div class="line" style="position:relative;">रम्या पाषाणतीर्था च पुरश्चन्द्रा च भारत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4105"><div class="verse-block verse-justify"><div class="line">अशोकतीर्थं तत्रैव कौन्ते य बहुलाश्रमम्।</div><div class="line" style="position:relative;">अगस्त्यतीर्थं पाण्ड्येषु वारुणं च युधिष्ठिर॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4106"><div class="verse-block verse-justify"><div class="line">कुमार्यः कथिताः पुण्याः पाण्ड्येष्वेव नरर्षभ।</div><div class="line" style="position:relative;">ताम्रपर्णी तु कौन्तेय कीर्तयिष्यामि तां शृणु॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4107"><div class="verse-block verse-justify"><div class="line">यत्र देवैस्तपस्तप्तं महदिच्छद्भिराश्रमे।</div><div class="line" style="position:relative;">गोकर्णमिति विख्यातं त्रिषु लोकेषु भारत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4108"><div class="verse-block verse-justify"><div class="line">शीततोयो बहुजलः पुण्यस्तात शिवश्च सः।</div><div class="line" style="position:relative;">ह्रदः परमदुष्प्रापो मानुषैरकृतात्मभिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4109"><div class="verse-block verse-justify"><div class="line">तत्र वृक्षतृणाद्यैश्च संपन्नः फलमूलवान्।</div><div class="line" style="position:relative;">आश्रमोऽगस्त्यशिष्यस्य पुण्यो देवसहे गिरौ॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4110"><div class="verse-block verse-justify"><div class="line">वैडूर्यपर्वतस्तत्र श्रीमान्मणिमयः शिवः।</div><div class="line" style="position:relative;">अगस्त्यस्याश्रमश्चैव बहुमूलफलोदकः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4111"><div class="verse-block verse-justify"><div class="line">सुराष्ट्रेष्वपि वक्ष्यामि पुण्यान्यायतनानि च।</div><div class="line" style="position:relative;">आश्रमान्सरितश्चैव सरांसि च नराधिप॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4112"><div class="verse-block verse-justify"><div class="line">चमसोद्भेदनं विप्रास्तत्रापि कथयन्त्युत।</div><div class="line" style="position:relative;">प्रभासं चोदधौ तीर्थं त्रिषु लोकेषु विश्रुतम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4113"><div class="verse-block verse-justify"><div class="line">तत्र पिण्डारकं नाम तापसाचरितं शिवम्।</div><div class="line" style="position:relative;">उज्जयन्तश्च शिखरी क्षिप्रं सिद्धिकरो महान्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4114"><div class="verse-block verse-justify"><div class="line">तत्र देवर्षिवीरेण नारदेनानुकीर्तितः।</div><div class="line" style="position:relative;">पुराणः श्रूयते श्लोकस्तं निबोध युधिष्ठिर॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4115"><div class="verse-block verse-justify"><div class="line">पुण्ये गिरौ सुराष्ट्रेषु मृगपक्षिनिषेविते।</div><div class="line" style="position:relative;">उज्जयन्ते स्म तप्ताङ्गो नाकपृष्ठे महीयते॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b4117"><div class="verse-block verse-justify"><div class="line">पुण्या द्वारवती तत्रयत्रासौ मधुसूदनः।</div><div class="line" style="position:relative;">साक्षाद्देवः पुराणोसौ स हि धर्मः सनातनः॥२६॥</div></div></div>
