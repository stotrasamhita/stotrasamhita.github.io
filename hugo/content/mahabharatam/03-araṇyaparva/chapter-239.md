<div class="verse-block-wrapper" id="03-araṇyaparva-b9780"><div class="verse-block verse-justify"><div class="line">ब्रवीषि यदिदं कर्ण सर्वं मनसि मे स्थितम्।</div><div class="line" style="position:relative;">न त्वभ्यनुज्ञां लप्स्यामि गमने यत्रपाण्डवाः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9781"><div class="verse-block verse-justify"><div class="line">परिदेवति तान्वीरान्धृतराष्ट्रो महीपतिः।</div><div class="line" style="position:relative;">मन्यतेऽभ्यधिकांश्चापि तपोयोगेन पाण्डवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9782"><div class="verse-block verse-justify"><div class="line">अथवाऽप्यनुबुध्येत नृपोऽस्माकं चिकीर्षितम्।</div><div class="line" style="position:relative;">एतामप्यायतिं रक्षन्नाभ्यनुज्ञातुमर्हति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9783"><div class="verse-block verse-justify"><div class="line">न हि द्वैतवने किंचिद्विद्यतेऽन्यत्प्रयोजनम्।</div><div class="line" style="position:relative;">उन्माथनमृते तेषां वनस्थानामपि द्विषाम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9784"><div class="verse-block verse-justify"><div class="line">जानासिहि यथा क्षत्ता द्यूतकाल उपस्थिते।</div><div class="line" style="position:relative;">अब्रवीद्यच्च मां त्वां च सौबलं वचनं तदा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9785"><div class="verse-block verse-justify"><div class="line">तानिसर्वाणि वाक्यानि यच्चान्यत्परिदेवितम्।</div><div class="line" style="position:relative;">विचिन्त्य निश्चय गच्छे गमनायेतराय वा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9786"><div class="verse-block verse-justify"><div class="line">ममापि हिमहान्हर्षो यदहं भीमफल्गुनौ।</div><div class="line" style="position:relative;">क्लिष्टावरण्ये पश्येयं कृष्णया सहिताविति॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9787"><div class="verse-block verse-justify"><div class="line">न तथा ह्याप्नुयां प्रीतिमवाप्य वसुधामिमाम्।</div><div class="line" style="position:relative;">दृष्ट्वा यथा पाण्डुसुतान्वल्कलाजिनवाससः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9788"><div class="verse-block verse-justify"><div class="line">किंनु स्यादधिकं तस्माद्यदहं द्रुपदात्मजाम्।</div><div class="line" style="position:relative;">द्रौपदीं कर्ण पश्येयं काषायवसनां वने॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9789"><div class="verse-block verse-justify"><div class="line">यदि मां दर्मराजश्च भीमसेनश्च पाण्डवः।</div><div class="line" style="position:relative;">युक्तं रपरमया लक्ष्म्या पश्येतां जीवितं भवेत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9790"><div class="verse-block verse-justify"><div class="line">उपायं न तु पश्यामि येन गच्छेम तद्वनम्।</div><div class="line" style="position:relative;">यथाचाभ्यनुजानीयाद्गच्छन्तं मां महीपतिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9791"><div class="verse-block verse-justify"><div class="line">स सौबलेन सहितस्तथा दुःशासनेन च।</div><div class="line" style="position:relative;">उपायं पश्य रनिपुणं येन गच्छेम तद्वनम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9792"><div class="verse-block verse-justify"><div class="line">अहमप्यनुगच्छामि गमनायेतराय च।</div><div class="line" style="position:relative;">कल्यमेव गमिष्यामि समीपं पार्थिवस्य ह॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9793"><div class="verse-block verse-justify"><div class="line">मयि तत्रोपविष्टे तु भीष्मे च कुरुसत्तमे।</div><div class="line" style="position:relative;">उपायो यो भवेद्दृष्टस्तं ब्रूयाः सहसौबलः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9794"><div class="verse-block verse-justify"><div class="line">ततो भीष्मस्य राज्ञश्च निशम्य रगमनं प्रति।</div><div class="line" style="position:relative;">व्यवसायं करिष्येऽहमनुनीय पितामहम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9795"><div class="verse-block verse-justify"><div class="line">तथेत्युक्त्वा तु ते सर्वेग्मुरावसथान्प्रति।</div><div class="line" style="position:relative;">व्युषितायां रजन्यां तु कर्णो राजानमभ्ययात्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9796"><div class="verse-block verse-justify"><div class="line">ततो दुर्योधनं कर्णः प्रहसन्निदमब्रवीत्।</div><div class="line" style="position:relative;">उपायः परिदृष्टोऽयं तं निबोधजनेश्वर॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9797"><div class="verse-block verse-justify"><div class="line">घोषा द्वैतवने सर्वेत्वत्प्रतीक्षा नराधिप।</div><div class="line" style="position:relative;">घोषयात्रापदेशेन गमिष्यामो न संशयः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9798"><div class="verse-block verse-justify"><div class="line">उचितं हि सदा गन्तुं घोषयात्रां विशांपते।</div><div class="line" style="position:relative;">एवं चेत्त्वां पिता राजन्समनुज्ञातुमर्हति॥२१॥</div></div></div>
