<div class="verse-block-wrapper" id="03-araṇyaparva-b10077"><div class="verse-block verse-justify"><div class="line">दुर्योधन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10078"><div class="verse-block verse-justify"><div class="line">अजानतस्ते राधेय नाभ्यसूयाम्यहं वचः।</div><div class="line" style="position:relative;">जानासि त्वं जिताञ्शत्रून्गन्धर्वां स्तेजसा मया॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10079"><div class="verse-block verse-justify"><div class="line">आयोधितास्तु गन्धर्वाः सुचिरं सोदरैर्मम।</div><div class="line" style="position:relative;">मया सह महाबाहो कृतश्चोभयतः क्षयः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10080"><div class="verse-block verse-justify"><div class="line">मायाधिकास्त्वयुध्यन्त यदा शूरा वियद्गताः।</div><div class="line" style="position:relative;">तदा नो न समं युद्धमभवत्खेचरैः सह॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10081"><div class="verse-block verse-justify"><div class="line">पराजयं च प्राप्ताः स्मो रणे बन्धनमेव च।</div><div class="line">सभृत्यामात्यपुत्राश्च सदारबलवाहनाः।</div><div class="line" style="position:relative;">उच्चैराकाशमार्गेण ह्रियमाणाः सुदुःखिताः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10082"><div class="verse-block verse-justify"><div class="line">अथ नः सैनिकाः केचिदमात्याश्च महारथाः।</div><div class="line" style="position:relative;">उपगम्याब्रुवन्दीनाः पाण्डवाञ्शरणप्रदान्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10083"><div class="verse-block verse-justify"><div class="line">एष दुर्योधनो राजा धार्तराष्ट्रः सहानुजः।</div><div class="line" style="position:relative;">सामात्यदारो ह्रियते गन्धर्वैर्दिवमाश्रितैः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10084"><div class="verse-block verse-justify"><div class="line">तं मोक्षयत भद्रं वः सहदारं नराधिपम्।</div><div class="line">परामर्शो मा भविष्यत्कुरुदारेषु सर्वशः।</div><div class="line" style="position:relative;">`इत्यब्रुवन्रणआन्मुक्ता धर्मराजमुपागताः'॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10085"><div class="verse-block verse-justify"><div class="line">एवमुक्ते तु धर्मात्मा ज्येष्ठः पाण्डुसुतस्तदा।</div><div class="line" style="position:relative;">प्रसाद्य सोदरान्सर्वानाज्ञापयत मोक्षणे॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10086"><div class="verse-block verse-justify"><div class="line">अथागम्य तमुद्देशं पाण्डवाः पुरुषर्षभाः।</div><div class="line" style="position:relative;">सान्त्वपूर्वमयाचन्त शक्ताः सन्तो महारथाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10087"><div class="verse-block verse-justify"><div class="line">यदा चास्मान्न मुमुचुर्गन्धर्वाः सान्त्विता अपि।</div><div class="line" style="position:relative;">`आकाशचारिणो वीरा नदन्तो जलदा इव'॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10088"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनश्च भीमश्च यमजौ च बलोत्कटौ।</div><div class="line" style="position:relative;">मुमुचुः शरवर्षाणि गन्धर्वान्प्रत्यनेकशः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10089"><div class="verse-block verse-justify"><div class="line">अथ सर्वे रणं मुक्त्वा प्रयाताः खेचरा दिवम्।</div><div class="line" style="position:relative;">अस्मानेवाभिकर्षन्तो दीनान्मुदितमानसाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10090"><div class="verse-block verse-justify"><div class="line">ततः समन्तात्पश्यामः शरजालेन वेष्टितम्।</div><div class="line" style="position:relative;">अमानुषाणि चास्त्राणि प्रयुञ्जानं घनंजयम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10091"><div class="verse-block verse-justify"><div class="line">समावृता दिशो दृष्ट्वा पाण्डवेन शितैः शरैः।</div><div class="line" style="position:relative;">धनंजयसखाऽऽत्मानं दर्शयामास वै तदा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10092"><div class="verse-block verse-justify"><div class="line">चित्रसेनः पाण्डवेन समाश्लिष्य परस्परम्।</div><div class="line" style="position:relative;">कुशलं परिपप्रच्छ तैः पृष्टश्चाप्यनामयम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10093"><div class="verse-block verse-justify"><div class="line">ते समेत्य तथाऽन्योन्यं सन्नाहान्विप्रमुच्य च।</div><div class="line" style="position:relative;">एकीभूतास्ततो वीरा गन्धर्वाः सह पाण्डवैः॥१७॥</div></div></div>
