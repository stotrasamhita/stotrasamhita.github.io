<div class="verse-block-wrapper" id="03-araṇyaparva-b9475"><div class="verse-block verse-justify"><div class="line">मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9476"><div class="verse-block verse-justify"><div class="line">यदा स्कन्देन मातृणामेवमेतत्प्रियं कृतम्।</div><div class="line" style="position:relative;">अथनैमब्रवीत्स्वाहा मम पुत्रस्त्वमौरसः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9477"><div class="verse-block verse-justify"><div class="line">इच्छाम्यहं त्वया दत्तां प्रीतिं परमदुर्लभाम्।</div><div class="line">तामब्रवीत्ततः स्कन्दः प्रीतिमिच्छसि कीदृशीम् ॥स्वाहोवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9478"><div class="verse-block verse-justify"><div class="line">दक्षस्याहं प्रिया कन्या स्वाहा नाम महाभुज।</div><div class="line" style="position:relative;">बाल्यात्प्रभृति नित्यं च जातकामा हुताशने॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9479"><div class="verse-block verse-justify"><div class="line">न स मां कामिनीं पुत्र सम्यग्जनाति पावकः।</div><div class="line">इच्छामि शाश्वतं वासं वस्तु पुत्र सहाग्निना ॥स्कन्द उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9480"><div class="verse-block verse-justify"><div class="line">हव्यं कव्यंच यत्किंचिद्द्विजा मन्त्रसुसंस्तुतम्।</div><div class="line" style="position:relative;">होष्यन्त्यग्नौ सदा देवि स्वाहेत्युक्त्वासमुद्धृतम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9481"><div class="verse-block verse-justify"><div class="line">अद्यप्रभृति दास्यन्ति सुवृत्ताः सत्पथे स्थिताः।</div><div class="line">एवमग्निसत््वया सार्धं सदा वत्स्यति शोभने ॥मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9482"><div class="verse-block verse-justify"><div class="line">एवमुक्ता ततः स्वाहा तुष्टास्कन्देन पूजिता।</div><div class="line" style="position:relative;">पावकेन समायुक्ता भर्त्रा स्कन्दमपूजयत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9483"><div class="verse-block verse-justify"><div class="line">ततो ब्रह्मा महासेनं प्रजापतिरथाब्रवीत्।</div><div class="line" style="position:relative;">अभिगच्छ महादेवं पितरं त्रिपुरार्दनम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9484"><div class="verse-block verse-justify"><div class="line">रुद्रेणाग्निं समाविश्य स्वाहामाविश्य चोमया।</div><div class="line" style="position:relative;">हितर्थं सर्वलोकानां जातस्त्वमपराजितः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9485"><div class="verse-block verse-justify"><div class="line">उमायोन्यां च रुद्रेण शुक्रं सिक्तं महात्मना।</div><div class="line" style="position:relative;">अस्मिन्गिरौ निपतितं मिञ्जिकमिञ्जिकं यतः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9486"><div class="verse-block verse-justify"><div class="line">`मिथुनं वै महाभाग तत्र तद्रुद्रसंभवम्'।</div><div class="line" style="position:relative;">संभूतं लोहितोदे तु शुक्रशेषमवापतत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9487"><div class="verse-block verse-justify"><div class="line">सूर्यरश्मिषु चाप्यनयदन्यच्चैवापतद्भुवि।</div><div class="line" style="position:relative;">आसक्तमन्यद्वृक्षेषु तदेवं प्चधाऽपतत्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9488"><div class="verse-block verse-justify"><div class="line">तत्र ते विविधाकारा गणा ज्ञेया मनीषिभिः।</div><div class="line" style="position:relative;">तव पारिषदा घोरा य एते पिशिताशिनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9489"><div class="verse-block verse-justify"><div class="line">एवमस्त्विति चाप्युक्त्वा महासेनो महेश्वरम्।</div><div class="line">अपूजयदमेयात्मा पितरं पितृवत्सलः ॥मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9490"><div class="verse-block verse-justify"><div class="line">अर्कपुष्पैस्तु ते पञ् गणाः पूज्या धनार्थिभिः।</div><div class="line" style="position:relative;">व्यादिप्रशमनार्थं चतेषां पूजां समाचरेत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9491"><div class="verse-block verse-justify"><div class="line">मिञ्जिकामिञ्जिकं चैव मिथुनं रुद्रसंभवम्।</div><div class="line" style="position:relative;">नमस्कार्यं सदैवेह बालानां हितमिच्छता॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9492"><div class="verse-block verse-justify"><div class="line">स्त्रियो मानुषमांसादा वृक्षका नाम नामतः।</div><div class="line" style="position:relative;">वृक्षेषु जातास्ता देव्यो नमस्कार्याः प्रजार्थिभिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9493"><div class="verse-block verse-justify"><div class="line">एषामेव पिशाचानामसङ्ख्येयगणाः स्मृताः।</div><div class="line" style="position:relative;">घण्टायाः सपताकायाः शृणु मे संभवं नृप॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9494"><div class="verse-block verse-justify"><div class="line">ऐरावतस्य घण्टे द्वे वैजयन्त्याविति श्रुते।</div><div class="line" style="position:relative;">गुहस्य ते स्वयं दत्ते क्रमेणानाय्य धीमता॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9495"><div class="verse-block verse-justify"><div class="line">एका तत्रविशाखस्य घण्टा स्कन्दस्य चापरा।</div><div class="line" style="position:relative;">पताका कार्तिकेयस् विशाखस्य च लोहिता॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9496"><div class="verse-block verse-justify"><div class="line">यानि क्रीडनकान्यस्य देवैर्दत्तानि वै तदा।</div><div class="line" style="position:relative;">तैरेव रमते देवो महासेनो महाबलः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9497"><div class="verse-block verse-justify"><div class="line">स संवृतः पिशाचानां गणैर्देवगणैस्तथा।</div><div class="line" style="position:relative;">शुशुभे काञ्चने शैले दीप्यमानः श्रिया वृत्तः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9498"><div class="verse-block verse-justify"><div class="line">तेन वीरेण शुशुभे स शैलः शुभकाननः।</div><div class="line" style="position:relative;">आदित्येनेवांशुमता मन्दरश्चारुकन्दरः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9499"><div class="verse-block verse-justify"><div class="line">संतानकवनैः फुल्लैः करवीरवनैरपि।</div><div class="line" style="position:relative;">पारिजातवनैश्चैव जपाशोकवनैस्तथा॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b9501"><div class="verse-block verse-justify"><div class="line">तत्रदेवगणाः सर्वेसर्वे देवर्षयस्तथा।</div><div class="line" style="position:relative;">मेघतूर्यरवाश्चैव क्षुब्धोदधिसमस्वनाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9502"><div class="verse-block verse-justify"><div class="line">तत्रदेवाश्च गन्धर्वा नृत्यन्तेऽप्सरसस्तथा।</div><div class="line" style="position:relative;">हृष्टानां तत्रभूतानां श्रूयते निनदो महान्॥२८॥</div></div></div>
