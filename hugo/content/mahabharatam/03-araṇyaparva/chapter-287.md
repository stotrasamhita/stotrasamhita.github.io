<div class="verse-block-wrapper" id="03-araṇyaparva-b11513"><div class="verse-block verse-justify"><div class="line">ततः प्रहस्तः सहसा समभ्येत्य विभीषणम्।</div><div class="line" style="position:relative;">गदया ताडयामास विनद्य रणकर्कशम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11514"><div class="verse-block verse-justify"><div class="line">स तयाऽभिहतो धीमान्गदया भीमवेगया।</div><div class="line" style="position:relative;">नाकम्पत महाबाहुर्हिमवानिव सुस्थिरः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11515"><div class="verse-block verse-justify"><div class="line">ततः प्रगृह्यविपुलां शतघण्टां विभीषणः।</div><div class="line" style="position:relative;">अनुमन्त्र्य महाशक्तिं चिक्षेपास् शिरः प्रति॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11516"><div class="verse-block verse-justify"><div class="line">पतन्त्या स तया वेगाद्राक्षसोऽशनिवेगया।</div><div class="line" style="position:relative;">हृतोत्तामङ्गो ददृशे वातरुग्ण इव द्रुमः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11517"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा निहतं सङ्ख्ये प्रहस्तं क्षणदाचरम्।</div><div class="line" style="position:relative;">अभिदुद्राव धूम्राक्षो वेगेन महता कपीन्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11518"><div class="verse-block verse-justify"><div class="line">तस् मेघोपमं सैन्यमापतद्भीमदर्शनम्।</div><div class="line" style="position:relative;">दृष्ट्वैव सहसा दीर्णा रणे वानरपुङ्गवाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11519"><div class="verse-block verse-justify"><div class="line">ततस्तान्सहसा दीर्णान्दृष्ट्वा वानरपुङ्गवान्।</div><div class="line" style="position:relative;">निर्ययौ कपिशार्दूलो हनूमान्मारुतात्मजः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11520"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वाऽवस्थितं सङ्ख्ये हरयः पवनात्मजम्।</div><div class="line" style="position:relative;">महत्या त्वरया राजत्संन्यवर्तन्त सर्वशः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11521"><div class="verse-block verse-justify"><div class="line">ततः शब्दो महानासीत्तुमुलो रोमहर्षणः।</div><div class="line" style="position:relative;">रामरावणसैन्यानामन्योन्यमभिधावताम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11522"><div class="verse-block verse-justify"><div class="line">तस्मिन्प्रवृत्ते संग्रामे घोरे रुधिरकर्दमे।</div><div class="line" style="position:relative;">क्षूम्राक्षः कपिसैन्यं तद्द्रावयामास पत्रिभिः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11523"><div class="verse-block verse-justify"><div class="line">तं स रक्षोमहामात्रमापतन्तं सपत्नजित्।</div><div class="line" style="position:relative;">प्रतिजग्राह हनुमांस्तरसा पवनात्मजः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11524"><div class="verse-block verse-justify"><div class="line">तयोर्युद्धमभूदधोरं हरिराक्षसवीरयोः।</div><div class="line" style="position:relative;">जीगीषतोर्युधाऽन्योन्यमिन्द्रप्रह्लादयोरिवं॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11525"><div class="verse-block verse-justify"><div class="line">कगदाभिः परिघैश्चैव राक्षसो जघ्निवान्कपिम्।</div><div class="line" style="position:relative;">कपिश्च जघ्निवान्रः सस्कन्धविटपैर्द्रुमैः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11526"><div class="verse-block verse-justify"><div class="line">ततस्तमतिकोपेन साश्वं सरथसारथिम्।</div><div class="line" style="position:relative;">धूम्राक्षमवधीत्क्रुद्धो हनूमान्मारुतात्मजः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11527"><div class="verse-block verse-justify"><div class="line">ततस्तं निहतं दृष्ट्वा धूम्राक्षं राक्षसोत्तमम्।</div><div class="line" style="position:relative;">हरयो जातविश्रम्भा जघ्नुरन्ये च सैनिकान्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11528"><div class="verse-block verse-justify"><div class="line">ते वध्यमाना हरिभिर्बलिभिर्जितकाशिभिः।</div><div class="line" style="position:relative;">राक्षसा भग्नसंकल्पा लङ्कामभ्यपतन्भयात्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11529"><div class="verse-block verse-justify"><div class="line">तेऽभिपत्य पुरं भग्ना हतशेषा निशाचराः।</div><div class="line" style="position:relative;">सर्वं राज्ञे यथावृत्तं रावणाय न्यवेदयन्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11530"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु रावणस्तेभ्यः प्रहस्तं निहतं युधि।</div><div class="line" style="position:relative;">धूम्राक्षं च महेष्वासं ससैन्यं सहराक्षसैः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11531"><div class="verse-block verse-justify"><div class="line">सुदीर्घमिव निःश्वस्य समुत्पत्य वरासनात्।</div><div class="line" style="position:relative;">उवाच कुम्भकर्णस्य कर्मकालोऽयमागतः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11532"><div class="verse-block verse-justify"><div class="line">इत्येवमुक्त्वा विविधैर्वादित्रैः सुमहास्वनैः।</div><div class="line" style="position:relative;">शयानमतिनिद्रालुं कुम्भकर्णमबोधयत्॥२१॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b11534"><div class="verse-block verse-justify"><div class="line">धन्योसि यस्य ते निद्रा कुम्भकर्णेयमीदृशी।</div><div class="line" style="position:relative;">य इदं दारुणं कालं न जानीषे महाभयम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11535"><div class="verse-block verse-justify"><div class="line">एष तीर्त्वाऽर्णवं रामः सेतुना हरिभिः सह।</div><div class="line" style="position:relative;">अवमत्येह नः सर्वान्करोति कदनं महत्॥२४॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b11537"><div class="verse-block verse-justify"><div class="line">तेन चैव प्रहस्तादिर्महान्नः स्वजनो हतः।</div><div class="line" style="position:relative;">तस्य नान्यो निहन्ताऽस्ति त्वामृतेशत्रुकर्शन॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11538"><div class="verse-block verse-justify"><div class="line">सदंशितोऽभिनिर्याहि त्वमद्य बलिनांवर।</div><div class="line" style="position:relative;">रामादीन्समरे सर्वाञ्जहि शत्रूनरिंदम॥२७॥</div></div></div>
