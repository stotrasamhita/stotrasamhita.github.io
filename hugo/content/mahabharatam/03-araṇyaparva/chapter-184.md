<div class="verse-block-wrapper" id="03-araṇyaparva-b7517"><div class="verse-block verse-justify"><div class="line">छादयन्तो महाघोषाः स्वं दिशश्च बलाहकाः।</div><div class="line" style="position:relative;">प्रववर्षुर्दिवारात्रमसिताः सततं तदा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7518"><div class="verse-block verse-justify"><div class="line">तपात्ययनिकेताश्च शतशोऽथ सहस्रशः।</div><div class="line" style="position:relative;">अपेतार्कप्रभाजालाः सविद्युन्मण्डप्रभाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7519"><div class="verse-block verse-justify"><div class="line">विरूढशष्पा धरणी मत्तदंशसरीसृपा।</div><div class="line" style="position:relative;">बभूव पयसा सिक्ता शान्तधूमरजोगणा॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7520"><div class="verse-block verse-justify"><div class="line">न स्म प्रज्ञायते किंचिदम्भसा समवस्तृते।</div><div class="line" style="position:relative;">समं वा विषमं वाऽपि नद्यो वा स्तावराणि वा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7521"><div class="verse-block verse-justify"><div class="line">क्षुब्धतोया महावेगाः श्वसमाना इवाशुगाः।</div><div class="line" style="position:relative;">सिन्धवः शोभयांचक्रुः काननानि तपात्यये॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7522"><div class="verse-block verse-justify"><div class="line">नदतां काननान्तेषु श्रूयन्ते विविधाः स्वनाः।</div><div class="line" style="position:relative;">वृष्टिभिस्ताड्यमानानां वराहमृगपक्षिणाम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7523"><div class="verse-block verse-justify"><div class="line">स्तोककाः शिखिनश्चैव पुंस्कोकिलगणैः सह।</div><div class="line" style="position:relative;">मत्ताः परिपतन्ति स्म दर्दुरश्चैव दर्पिताः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7524"><div class="verse-block verse-justify"><div class="line">तदा बहुविधाकारा प्रावृण्मेघानुनादिता।</div><div class="line" style="position:relative;">अभ्यतीता शिवा तेषां चरतां मरुधन्वसु॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7525"><div class="verse-block verse-justify"><div class="line">क्रौञ्चहंससमाकीर्णा शरत्प्रमुदीताऽभवत्।</div><div class="line" style="position:relative;">रूढकक्षवनप्रस्था प्रसन्नजलनिम्नगा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7526"><div class="verse-block verse-justify"><div class="line">विमलाकाशनक्षत्रा शरत्तेषां शिवाऽभषत्।</div><div class="line" style="position:relative;">मृगद्विजसमाकीर्णा पाण्डवानां महात्मनाम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7527"><div class="verse-block verse-justify"><div class="line">दृश्यन्ते शान्तरजसः क्षपाजलदशीतलाः।</div><div class="line" style="position:relative;">ग्रहनक्षत्रसङ्घैश्च सोमेन च विराजिताः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7528"><div class="verse-block verse-justify"><div class="line">कुमुदैः पुण्डरीकैश्च शीतवारिधराः शिवाः।</div><div class="line" style="position:relative;">नदीः पुष्करिणीश्चैव ददृशुः समलंकृताः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7529"><div class="verse-block verse-justify"><div class="line">ते वै मुमुदिरे वीराः प्रसन्नसलिलां शिवाम्।</div><div class="line" style="position:relative;">पश्यन्तो दृढधन्वानः परिपूर्णां सरस्वतीम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7530"><div class="verse-block verse-justify"><div class="line">तेषां पुण्यतमा रात्रिः पर्वसन्धौ स्म शारदी।</div><div class="line" style="position:relative;">तत्रैव वसतामासीत्कार्तिकी जनमेजय॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7531"><div class="verse-block verse-justify"><div class="line">पुण्यकृद्भिर्महासत्वैस्तापसैः सह पाण्डवाः।</div><div class="line" style="position:relative;">तत्सर्वं भरतश्रेष्ठ समूहुर्योगमुत्तमम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7532"><div class="verse-block verse-justify"><div class="line">तमिस्राभ्युदये तस्मिन्धौम्येन सह पाण्डवाः।</div><div class="line" style="position:relative;">सूतैः पौरोगवैः साकं काम्यकं प्रययुर्वनम्॥१८॥</div></div></div></div>
