<div class="verse-block-wrapper" id="03-araṇyaparva-b11642"><div class="verse-block verse-justify"><div class="line">संवृतोराक्षसैर्घेरैर्विविधायुधपाणिभिः।</div><div class="line" style="position:relative;">अभिदुद्राव रामं स पोथयन्हरियूथपान्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11643"><div class="verse-block verse-justify"><div class="line">तमाद्रवन्तं संक्रुद्ध मैन्दनीलनलाङ्गदाः।</div><div class="line" style="position:relative;">हनुमाञ्जाम्बवांश्चैव ससैन्याः पर्यवारयन्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11644"><div class="verse-block verse-justify"><div class="line">ते दशग्रीवसैन्यं तदृक्षवानरपुङ्गवाः।</div><div class="line" style="position:relative;">द्रुमैर्विध्वंसयांचक्रुर्दशग्रीवस्य पश्यतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11645"><div class="verse-block verse-justify"><div class="line">ततः स्वसैन्यमालोक्य वध्यमानमरातिभिः।</div><div class="line" style="position:relative;">मायावी चासृजन्मायां रावणो राक्षसाधिपः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11646"><div class="verse-block verse-justify"><div class="line">तस्य देहविनिष्क्रान्ताः शतशोऽथ सहस्रशः।</div><div class="line" style="position:relative;">राक्षसाः प्रत्यदृश्यन्त शरशक्त्यृष्टिपाणयः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11647"><div class="verse-block verse-justify"><div class="line">तान्रामो जघ्निवान्सर्वान्दिव्येनास्त्रेण राक्षसान्।</div><div class="line" style="position:relative;">अथ भूयोपि मायां स व्यदधाद्राक्षसाधिपः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11648"><div class="verse-block verse-justify"><div class="line">कृत्वा रामस् रूपाणि लक्ष्मणस्य च भारत।</div><div class="line" style="position:relative;">अभिदुद्राव रामं च लक्ष्मणं च दशाननः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11649"><div class="verse-block verse-justify"><div class="line">ततस्ते राममर्च्छन्तो लक्ष्मणं च क्षपाचराः।</div><div class="line" style="position:relative;">अभिपेतुस्तदा रामं प्रगृहीतशरासनाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11650"><div class="verse-block verse-justify"><div class="line">तां दृष्ट्वाराक्षसेन्द्रस् मायामिक्ष्वाकुनन्दनः।</div><div class="line" style="position:relative;">उवाच रामः सौमित्रिमसंभ्रान्तो बृहद्वचः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11651"><div class="verse-block verse-justify"><div class="line">जहीमान्राक्षसान्पापानात्मनः प्रतिरूपकान्।</div><div class="line" style="position:relative;">इत्युक्त्वाऽभ्यहनद्रामो लक्ष्मणश्चात्मरूपकान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11652"><div class="verse-block verse-justify"><div class="line">ततो हर्यश्वयुक्तेन रथेनादित्यवर्चसा।</div><div class="line">उपतस्थे रणे रामं मातलिः शक्रसारथिः ॥मातलिरुवाच।</div><div class="line" style="position:relative;">॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11653"><div class="verse-block verse-justify"><div class="line">अयं हर्यश्वयुग्जैत्रो मघोनः स्यन्दनोत्तमः।</div><div class="line" style="position:relative;">`त्वदर्थमिह संप्राप्तः संदेशाद्वै शतक्रतोः'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11654"><div class="verse-block verse-justify"><div class="line">अनेन शक्रः काकुत्स्थ समरे दैत्यदानवान्।</div><div class="line" style="position:relative;">शतशः पुरुषव्याघ्र रथोदारेण जघ्निवान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11655"><div class="verse-block verse-justify"><div class="line">तदनन नरव्याघ्र मया यत्तेन संयुगे।</div><div class="line" style="position:relative;">स्यन्दनेन जहिक्षिप्रं रावणं मा चिरं कृथाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11656"><div class="verse-block verse-justify"><div class="line">इत्युक्तो राघवस्तथ्यं वचोऽशङ्कत मातलेः।</div><div class="line" style="position:relative;">मायैषाराक्षसस्येति तमुवाच विबीषणः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11657"><div class="verse-block verse-justify"><div class="line">नेयं माया नरव्याघ्ररावणस् दुरात्मनः।</div><div class="line" style="position:relative;">तदातिष्ठ रथंशीघ्रमिमसैन्द्रं महाद्युते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11658"><div class="verse-block verse-justify"><div class="line">ततः प्रहृष्टः काकुत्स्थस्तथेत्युक्त्वा विभीषणम्।</div><div class="line" style="position:relative;">रथेनाभिपपाताथ दशग्रीवं रुषाऽन्वितः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11659"><div class="verse-block verse-justify"><div class="line">हाहाकुतानि भूतानि रावणे समभिद्रुते।</div><div class="line" style="position:relative;">सिंहनादाः सपटहादिति दिव्यास्तथाऽनदन्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11660"><div class="verse-block verse-justify"><div class="line">[दशकन्धरराजसून्वोस्तथा युद्धमभून्महत्।</div><div class="line" style="position:relative;">अलब्धोपममन्यत्रतयोरेव तथाऽभवत् ॥]॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11661"><div class="verse-block verse-justify"><div class="line">सरामाय महाघोरं विससर्ज निशाचरः।</div><div class="line" style="position:relative;">शूलमिन्द्राशनिप्रख्यं ब्रह्मदण्डभिवोद्यतम्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b11668"><div class="verse-block verse-justify"><div class="line">अल्पावशेषमायुश्च ततोऽमन्यन्त रक्षसः।</div><div class="line" style="position:relative;">ब्रह्मास्त्रोदीरणाच्छत्रोर्देवदानवकिंनराः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11669"><div class="verse-block verse-justify"><div class="line">ततः ससर्ज तं रामः शरमप्रतिमौजसम्।</div><div class="line" style="position:relative;">रावणान्तकरं घोरं ब्रह्मदण्डमिवोद्यतम्॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11670"><div class="verse-block verse-justify"><div class="line">मुक्तमात्रेण रामेण दूराकृष्टेन भारत।</div><div class="line">स तेन राक्षसश्रेष्ठः सरथः साश्वसारथिः।</div><div class="line" style="position:relative;">प्रजज्वाल महाज्वालेनाग्निनाभिपरिप्लुतः॥३१॥</div></div></div>
