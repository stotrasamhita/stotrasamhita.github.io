<div class="verse-block-wrapper" id="03-araṇyaparva-b1843"><div class="verse-block verse-justify"><div class="line">गतेषु लोकपालेषु पार्थः शत्रुनिबर्हणः।</div><div class="line" style="position:relative;">चिन्तयामास राजेन्द्र देवराजरथागमम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1844"><div class="verse-block verse-justify"><div class="line">तस्य चिन्तयमानस्य गुडाकेशस्य धीमतः।</div><div class="line" style="position:relative;">रथो मातलिसंयुक्त आजगाम महाप्रभः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1845"><div class="verse-block verse-justify"><div class="line">नभो वितिमिरं कुर्वञ्जलदान्पाटयन्निव।</div><div class="line">दिशः संपूरयन्नादैर्महामेघरवोपमैः।</div><div class="line" style="position:relative;">॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1846"><div class="verse-block verse-justify"><div class="line">असयः शक्तयो भीमा गदाश्चोग्रप्रदर्शनाः।</div><div class="line" style="position:relative;">दिव्यप्रभावाः प्रासाश्च विद्युतश्च महाप्रभाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1847"><div class="verse-block verse-justify"><div class="line">तथैवाशनयश्चैव चक्रयुक्तास्तुलागुडाः।</div><div class="line" style="position:relative;">वायुस्फोटाः सनिर्घाताः शङ्खमेघस्वनास्तथा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1848"><div class="verse-block verse-justify"><div class="line">तत्रनागा महाकाया ज्वलितास्याः सुदारुणाः।</div><div class="line" style="position:relative;">सिताभ्रकूटप्रतिमाः संहताश्च यथोपलाः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1849"><div class="verse-block verse-justify"><div class="line">दशवाजिसहस्राणि हरीणां वातरंहसाम्।</div><div class="line" style="position:relative;">वहन्ति यं नेत्रमुषं दिव्यं मायामयं रथम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1850"><div class="verse-block verse-justify"><div class="line">तत्रापश्यन्महानीलं वैजयन्तं महाप्रभम्।</div><div class="line" style="position:relative;">ध्वजमिन्दीवरश्यामं वंशं कनकभूषणम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1851"><div class="verse-block verse-justify"><div class="line">तस्मिन्रथे स्थितं सूतं तप्तहेमविभूषितम्।</div><div class="line" style="position:relative;">दृष्ट्वा पार्थो महाबाहुर्देवराजमतर्कयत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1852"><div class="verse-block verse-justify"><div class="line">तथा तर्कयतस्तस्य फल्गुनस्याथ मातलिः।</div><div class="line" style="position:relative;">सन्नतः प्रस्थितो भूत्वा वाक्यमर्जुनमब्रवीत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1853"><div class="verse-block verse-justify"><div class="line">भोभो शक्रात्मज श्रीमञ्शक्रस्त्वां द्रष्टुमिच्छति।</div><div class="line" style="position:relative;">आरोहतु भवाञ्शीघ्रं रथमिन्द्रस्य संभतम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1854"><div class="verse-block verse-justify"><div class="line">आह माममरश्रेष्ठः पिता तव शतक्रतुः।</div><div class="line" style="position:relative;">कुन्तीसुतमिह प्राप्तं पश्यन्तु त्रिदशालयाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1855"><div class="verse-block verse-justify"><div class="line">एष शक्रः परिवृतो देवैर्ऋषिगणैस्तथा।</div><div class="line" style="position:relative;">गन्धर्वैरप्सरोभिश्च त्वां दिदृक्षुः प्रतीक्षते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1856"><div class="verse-block verse-justify"><div class="line">अस्माल्लोकाद्देवलोकं पाकशासनशासनात्।</div><div class="line">मारोह त्वं मया सार्धं लब्धास्त्रः पुनरेष्यसि ॥अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1857"><div class="verse-block verse-justify"><div class="line">मातले गच्छ शीघ्रं त्वमारोहस्व रथोत्तमम्।</div><div class="line" style="position:relative;">राजसूयाश्वमेधानां शतैरपि सुदुर्लभम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1858"><div class="verse-block verse-justify"><div class="line">पार्थिवैः सुमहाभागैर्यज्वभिर्भूरिदक्षिणैः।</div><div class="line" style="position:relative;">दैवतैर्वा दुरारोहं दानवैर्वा रथोत्तमम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1859"><div class="verse-block verse-justify"><div class="line">नातप्ततपसा शक्य एष दिव्यो महारथः।</div><div class="line" style="position:relative;">द्रष्टुं वाऽप्यथवा स्प्रष्टुमारोढुं कुत एव च॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1860"><div class="verse-block verse-justify"><div class="line">त्वयि मतिष्ठिते साधो रथस्थे स्थिरवाजिनि।</div><div class="line">पश्चादहमथारोक्ष्ये मुकृती सत्पथं यथा ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1861"><div class="verse-block verse-justify"><div class="line">तस्यतद्वचनं श्रुत्वा मातलिः शक्रसारथिः।</div><div class="line" style="position:relative;">आरुरोह रथं शीघ्रं हयाञ्जग्राह रश्मिभिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1862"><div class="verse-block verse-justify"><div class="line">ततोऽर्जुनो हृष्टमना गङ्गायामाप्लुतः शुचिः।</div><div class="line" style="position:relative;">जजाप जप्यं कौन्तेयो विधिवत्कुरुनन्दनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1863"><div class="verse-block verse-justify"><div class="line">ततः पितृन्यथान्यायं तर्पयित्वा यथाविधि।</div><div class="line" style="position:relative;">मन्दरं शैलराजं तमाप्रष्टुमुपचक्रमे॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1864"><div class="verse-block verse-justify"><div class="line">साधूनां पुण्यशीलानां मुनीनां पुण्यकर्मणाम्।</div><div class="line" style="position:relative;">त्वं सदा संश्रयः शैल स्वर्गमार्गाभिकाङ्क्षिणाम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1865"><div class="verse-block verse-justify"><div class="line">त्वत्प्रसादात्सदा शैल ब्राह्मणाः क्षत्रिया विशः।</div><div class="line" style="position:relative;">स्वर्गं प्राप्ताश्चरन्ति स्म देवैः सह गतव्यथाः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1866"><div class="verse-block verse-justify"><div class="line">अद्रिराज महाशैल मुनिसंश्रय तीर्थवन्।</div><div class="line" style="position:relative;">गच्छाम्यामन्त्रयित्वा त्वां सुखमस्म्युषितस्त्वयि॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1867"><div class="verse-block verse-justify"><div class="line">तव सानूनि कुञ्जाश्च नद्यः प्रस्रवणानि च।</div><div class="line" style="position:relative;">तीर्थानि च सुपुण्यानि मया दृष्टान्यनेकशः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1868"><div class="verse-block verse-justify"><div class="line">सुसुगन्धाश्च वैर्योघास्त्वच्छरीरविनिःसृताः।</div><div class="line" style="position:relative;">अमृतास्वादसृशाः पीताः प्रस्रवणोदकाः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1869"><div class="verse-block verse-justify"><div class="line">शिशुर्यथा पितुश्चाङ्के सुखं शेते तटे तथा।</div><div class="line" style="position:relative;">मया तवाङ्के ललितं शैलराज महाप्रभो॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1870"><div class="verse-block verse-justify"><div class="line">अप्सरोगणसंकीर्णए ब्रह्मघोषानुनादिते।</div><div class="line" style="position:relative;">सुखमस्म्युषितः शैल तव सानुषु नित्यदा॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1871"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वार्जुनः शैलमामन्त्र्य परवीरहा।</div><div class="line" style="position:relative;">आरुरोह रथं दिव्यं द्योतयन्निव भास्करः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1872"><div class="verse-block verse-justify"><div class="line">स तेनादित्यरूपेण दिव्येनाद्भुतकर्मणा।</div><div class="line" style="position:relative;">ऊर्ध्वमाचक्रमे धीमान्प्रहृष्टः कुरुनन्दनः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1873"><div class="verse-block verse-justify"><div class="line">सोऽदर्शनपथं यातो मर्त्यानां धर्मचारिणाम्।</div><div class="line" style="position:relative;">ददर्शाद्भुतरूपाणि विमानानि सहस्रशः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1874"><div class="verse-block verse-justify"><div class="line">न तत्र सूर्यः सोमो वा द्योतते न च पावकः।</div><div class="line" style="position:relative;">स्वयैव प्रभया तत्र द्योतन्ते पुण्यलब्धया॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1875"><div class="verse-block verse-justify"><div class="line">तारारूपाणि यानीह दृश्यन्ते द्युतिमन्ति वै।</div><div class="line" style="position:relative;">आकाशे विप्रकृष्टत्वात्तनूनि सुमहान्त्यपि॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1876"><div class="verse-block verse-justify"><div class="line">तानि तत्र प्रभास्वन्ति रूपवन्ति च पाण्डवः।</div><div class="line" style="position:relative;">ददर्श स्वेषु धिष्ण्येषु दीप्तिमन्ति स्वयाऽर्चिषा॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1877"><div class="verse-block verse-justify"><div class="line">तत्र राजर्षयः सिद्धा वीराश्च निहता युधि।</div><div class="line" style="position:relative;">तपसा च जितं स्वर्गं संपेतुः शतसङ्घशः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1878"><div class="verse-block verse-justify"><div class="line">गन्धर्वाणां सहस्राणि मूर्यज्वलिततेजसाम्।</div><div class="line" style="position:relative;">गुह्यकारनामृषीणां च तथैवाप्सरसां गणान्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1879"><div class="verse-block verse-justify"><div class="line">लोकानात्मप्रभान्पश्यन्फल्गुनो विस्मयान्वितः।</div><div class="line" style="position:relative;">पप्रच्छ मातलिं प्रीत्या स चाप्येनमुवाच ह॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1880"><div class="verse-block verse-justify"><div class="line">एते सुकृतिनः पार्थ स्वेषु धिष्ण्येष्ववस्थिताः।</div><div class="line" style="position:relative;">तान्दृष्टवानसि विभो तारारूपाणि भूतले॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1881"><div class="verse-block verse-justify"><div class="line">ततोऽपश्यत्स्थितं द्वारि मत्तं विजयिनं गजम्।</div><div class="line" style="position:relative;">ऐरावतं चतुर्दन्तं कैलासमिव शृङ्गिणम्॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1882"><div class="verse-block verse-justify"><div class="line">स सिद्धमार्गमाक्रम्य कुरुपाण्डवसत्तमः।</div><div class="line" style="position:relative;">व्यरोचत यथापूर्वं मान्धाता पार्थिवोत्तमः॥४१॥</div></div></div>
