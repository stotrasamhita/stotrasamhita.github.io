<div class="verse-block-wrapper" id="03-araṇyaparva-b9699"><div class="verse-block verse-justify"><div class="line">मार्कण्डेयादिभिर्विप्रैः पाण्डवैश्च महात्मभिः।</div><div class="line" style="position:relative;">कथाभिरनुकूलाभिः सह स्तित्वा जनार्दनः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9700"><div class="verse-block verse-justify"><div class="line">ततस्तैः संविदं कृत्वा यथावन्मधुसूदनः।</div><div class="line" style="position:relative;">आरुरुक्षू रथं सत्यामाह्वयामास भारत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9701"><div class="verse-block verse-justify"><div class="line">सत्यभामा ततस्तत्र स्वजित्वा द्रुपदात्मजाम्।</div><div class="line" style="position:relative;">उवाच वचनं हृद्यं यथाभावं समाहितम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9702"><div class="verse-block verse-justify"><div class="line">कृष्णे माभूत्तवोत्कण्ठा मा व्यथा मा प्रजागरः।</div><div class="line" style="position:relative;">भर्तृभिर्देवसंकाशैर्जितां प्राप्स्यसि मेदिनीम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9703"><div class="verse-block verse-justify"><div class="line">न ह्येवं शीलसंपन्ना नैवं पूजितलक्षणाः।</div><div class="line" style="position:relative;">प्राप्नुवन्ति चिरं क्लेशं यथा त्वमसितेक्षणे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9704"><div class="verse-block verse-justify"><div class="line">अवश्यं च त्वया भूमिरियं निहतकण्टका।</div><div class="line" style="position:relative;">भर्तृभिः सहभोक्तव्या निर्द्वन्द्वेति श्रुतं मया॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9705"><div class="verse-block verse-justify"><div class="line">धार्तराष्ट्रवधं कृत्वावैराणि प्तियात्य च।</div><div class="line" style="position:relative;">युधिष्ठिरस्थां पृथिवीं द्रष्टासि द्रुपदात्मजे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9706"><div class="verse-block verse-justify"><div class="line">यास्ताः प्रव्राजपानां त्वां प्राहसन्दर्पमोहिताः।</div><div class="line" style="position:relative;">ताः क्षिप्रं हतसंकल्पा द्रक्ष्यसि त्वं कुरुस्त्रियः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9707"><div class="verse-block verse-justify"><div class="line">तव दुःखोपपन्नाया यैराचरितमप्रियम्।</div><div class="line" style="position:relative;">विद्धि संप्रस्थितान्सर्वांस्तान्कृष्णे यमसादनम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9708"><div class="verse-block verse-justify"><div class="line">पुत्रस्ते प्रतिविन्ध्यश्च सुतसोमस्तथाविधः।</div><div class="line" style="position:relative;">श्रुतकर्माऽर्जुनिश्चैव शतानीकश्च नाकुलिः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9709"><div class="verse-block verse-justify"><div class="line">सहदेवाच्च यो जातः श्रुतसेनस्तवात्मजः।</div><div class="line" style="position:relative;">सर्वेकुशलिनो वीराः कृतास्त्राश्च सुतास्तव॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9710"><div class="verse-block verse-justify"><div class="line">अभिमन्युरिव प्रीता द्वारवत्यां रता भृशम्।</div><div class="line" style="position:relative;">त्वमिवैषां सुभद्रा च प्रीत्या सर्वात्मना स्थिता॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9711"><div class="verse-block verse-justify"><div class="line">प्रीयते तव निर्द्वन्द्वा तेभ्यश्च विगतज्वरा।</div><div class="line" style="position:relative;">दुःखिता तेन दुःखेन सुखेन सुखिता तथा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9712"><div class="verse-block verse-justify"><div class="line">भजेत्सर्वात्मना चैव प्रद्युम्नजननी तथा।</div><div class="line" style="position:relative;">भानुप्रभृतिभिश्चैनान्विशिनष्टि च केशवः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9713"><div class="verse-block verse-justify"><div class="line">भोजनाच्छादने चैषां नित्यं मे श्वशुरः स्थितः।</div><div class="line" style="position:relative;">रामप्रभृतयः सर्वे भजन्त्यन्धकवृष्णयः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9714"><div class="verse-block verse-justify"><div class="line">तुल्यो हिप्रणयस्तेषां प्रद्युम्नस्य च भामिनि।</div><div class="line" style="position:relative;">एवमादि प्रियं सत्यंहृद्यमुक्त्वा मनोनुगम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9715"><div class="verse-block verse-justify"><div class="line">गमनाय मनश्चक्रेवासुदेवरथं प्रति।</div><div class="line" style="position:relative;">तां कृष्णां कृष्णमहिषी चकाराभिप्रदक्षिणम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9716"><div class="verse-block verse-justify"><div class="line">आरुरोह रथं शौरेः सत्यभामाऽथ भामिनी।</div><div class="line">स्मयित्वातु यदुश्रेष्ठो द्रौपदीं परिसान्त्व्य च।</div><div class="line" style="position:relative;">उपावर्त्य ततः शीघ्रैर्हयैः प्रायात्परंतपः॥१९॥</div></div></div></div>
