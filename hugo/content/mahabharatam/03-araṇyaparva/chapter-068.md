<div class="verse-block-wrapper" id="03-araṇyaparva-b2951"><div class="verse-block verse-justify"><div class="line">अथ दीर्घस्य कालस्य पर्णादो नाम वै द्विजः।</div><div class="line" style="position:relative;">प्रत्येत्य नगरं भैमीमिदं वचनमब्रवीत्॥२॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b2953"><div class="verse-block verse-justify"><div class="line">श्रावितश्च मया वाक्यं त्वदीयं स महाजने।</div><div class="line" style="position:relative;">ऋतुपर्णो महाभागो यथोक्तं वरवर्णिनि॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2954"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा नाब्रवीत्किंचिदृतुपर्णो नराधिपः।</div><div class="line" style="position:relative;">न च पारिषदः कश्चिद्भाष्यमाणो मयाऽसकृत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2955"><div class="verse-block verse-justify"><div class="line">अनुज्ञातं तु मां राज्ञा विजने कश्चिदब्रवीत्।</div><div class="line" style="position:relative;">ऋतुपर्णस्य पुरुषो बाहुको नाम नामतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2956"><div class="verse-block verse-justify"><div class="line">सूतस्तस्य नरेन्द्रस्य विरूपो ह्रस्वबाहुकः।</div><div class="line" style="position:relative;">शीघ्रयानेषु कुशलो मृष्टकर्ता च भोजने॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2957"><div class="verse-block verse-justify"><div class="line">स विनिःश्वस्य बंहुशो रुदित्वा च षुनःपुनः।</div><div class="line" style="position:relative;">कुशलं चैव मां पृष्ट्वा पश्चादिदमभाषत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2958"><div class="verse-block verse-justify"><div class="line">वैषम्यमपि संप्राप्ता गोपायन्ति कुलस्त्रियः।</div><div class="line" style="position:relative;">आत्मानमात्मना सत्यो जितः स्वर्गो न संशयः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2959"><div class="verse-block verse-justify"><div class="line">रहिता भर्तृभिश्चैव न कुप्यन्ति कदाचन।</div><div class="line" style="position:relative;">3-68-9bप्राणांश्चारित्रकवचान्धारयन्ति वरस्त्रियः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2960"><div class="verse-block verse-justify"><div class="line">विषमस्थेन मूढेन परिभ्रष्टसुखेन च।</div><div class="line" style="position:relative;">यत्सा तेन परित्यक्ता तत्र न क्रोद्धुमर्हति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2961"><div class="verse-block verse-justify"><div class="line">प्राणयात्रां परिप्रेप्सोः शकुनैर्हृतवाससः।</div><div class="line" style="position:relative;">आधिभिर्दह्यमानस्य श्यामा न क्रोद्धुमर्हति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2962"><div class="verse-block verse-justify"><div class="line">सत्कृताऽसत्कृता वाऽपि पतिं दृष्ट्वा तथागतम्।</div><div class="line" style="position:relative;">भ्रष्टराज्यं श्रित्या हीनं श्यामा न क्रोद्धुमर्हति॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2963"><div class="verse-block verse-justify"><div class="line">तस्य तद्वचनं श्रुत्वा त्वरितोऽहमिहागतः।</div><div class="line" style="position:relative;">श्रुत्वा प्रमाणं भवती राज्ञश्चैव निवेदयः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2964"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वाऽश्रुपूर्णाक्षी पर्णादस्य विशांपते।</div><div class="line" style="position:relative;">दमयन्ती रहोऽभ्येत्य मातरं प्रत्यभापत॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2965"><div class="verse-block verse-justify"><div class="line">अयमर्थौ न संवेद्यो भीमे मातः कदाचन।</div><div class="line" style="position:relative;">त्वत्सन्निधौ नियोक्ष्येऽहंसुदेवं द्विजसत्तमम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2966"><div class="verse-block verse-justify"><div class="line">यथा न नृपतिर्भीमः प्रतिपद्येत मे मतम्।</div><div class="line" style="position:relative;">यथा त्वया प्रकर्तव्यं मम चेत्प्रियमिच्छसि॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2967"><div class="verse-block verse-justify"><div class="line">यथा चाहं समानीता सुदेवेनाशु बान्धवान्।</div><div class="line" style="position:relative;">तेनैव मङ्गलेनाद्य सुदेवो यातु मा चिरम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2968"><div class="verse-block verse-justify"><div class="line">समानेतुं नलं मातरयोध्यां नगरीमितः।</div><div class="line" style="position:relative;">`ऋतुपर्णस्य नगरे निवसन्तमरिन्दमम्'॥१९॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b2971"><div class="verse-block verse-justify"><div class="line">त्वया हि मे बहुकृतंयदन्यो न करिष्यति।</div><div class="line" style="position:relative;">यद्भर्त्राऽहं समेष्यामि शीघ्रमेव द्विजोत्तम॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2972"><div class="verse-block verse-justify"><div class="line">स एवमुक्तोऽथाश्वास्य आशीर्वादैः सुमङ्गलैः।</div><div class="line" style="position:relative;">गृहानुपययौ चापि कृतार्थः सुमहामनाः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2973"><div class="verse-block verse-justify"><div class="line">ततः सुदेवमानाय्य दमयन्ती युधिष्ठिर।</div><div class="line" style="position:relative;">अब्रवीत्सन्निदौ मातुर्दुःखशोकसमन्विता॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2974"><div class="verse-block verse-justify"><div class="line">गत्वा सुदेव नगरीमयोध्यावासिनं नृपम्।</div><div class="line" style="position:relative;">ऋतुपर्णं वचो ब्रूहि पतिमन्यं चिकीर्षती॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2975"><div class="verse-block verse-justify"><div class="line">आस्थास्यति पुनर्भैमी दमयन्ती स्वयंवरम्।</div><div class="line" style="position:relative;">तत्र गच्छन्ति राजानो राजपुत्राश्च सर्वशः॥२६॥</div></div></div>
