<div class="verse-block-wrapper" id="03-araṇyaparva-b8830"><div class="verse-block verse-justify"><div class="line">मार्कण्येय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8831"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु विप्रेण धर्मव्याधो युधिष्ठिर।</div><div class="line">प्रत्युवाच यथा विप्रं तच्छृणुष्व नराधिप ॥व्याध उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8832"><div class="verse-block verse-justify"><div class="line">विज्ञानार्थं मनुष्याणां मनः पूर्वं प्रवर्तते।</div><div class="line" style="position:relative;">तत्प्राप्य कामं भजतेक्रोधं च द्विजसत्तम॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8833"><div class="verse-block verse-justify"><div class="line">ततस्तदर्थं यतते कर्म चारभते महत्।</div><div class="line" style="position:relative;">इष्टानां रूपगन्धानामभ्यासं च निषेवते॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8834"><div class="verse-block verse-justify"><div class="line">ततो रागः प्रभवति द्वेषश्च तदनन्तरम्।</div><div class="line" style="position:relative;">ततो लोभः प्रभवति मोहश्च तदनन्तरम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8835"><div class="verse-block verse-justify"><div class="line">तस् लोभाभिभूतस्य रागद्वेषहतस्य च।</div><div class="line" style="position:relative;">न धर्मे जायते बुद्धिर्व्याजाद्धर्मं करोति च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8836"><div class="verse-block verse-justify"><div class="line">व्याजेन चरते धर्ममर्थं व्याजेन रोचते।</div><div class="line" style="position:relative;">व्याजेन सिध्यमानेषु धनेषु द्विजसत्तम॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8837"><div class="verse-block verse-justify"><div class="line">तत्रैव रमते बुद्धिस्ततः पापं चिकीर्षति।</div><div class="line" style="position:relative;">सुहृद्भिर्वार्यमाणश्च पण्डितैश्च द्विजोत्तम॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8838"><div class="verse-block verse-justify"><div class="line">उत्तरं श्रुतिसंबद्धं ब्रवीत्यश्रुतियोजितम्।</div><div class="line" style="position:relative;">अधर्मस्त्रिविधस्तस्य वर्तते रागदोषजः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8839"><div class="verse-block verse-justify"><div class="line">पापं चिन्तयते चैव ब्रवीति च करोति च।</div><div class="line" style="position:relative;">तस्याधर्मप्रवृत्तस्य गुणा नश्यन्ति साधवः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8840"><div class="verse-block verse-justify"><div class="line">एकशीलाश्च मित्रत्वं भजन्ते पापकर्मिणः।</div><div class="line" style="position:relative;">स तेन दुःखमाप्नोति परत्र च विपद्यते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8841"><div class="verse-block verse-justify"><div class="line">पापात्मा भवति ह्येवं धर्मलाभं तु मे शृणु।</div><div class="line" style="position:relative;">यस्त्वेतान्प्रज्ञाया दोषान्पूर्वमेवानुपश्यति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8842"><div class="verse-block verse-justify"><div class="line">कुशलः सुखदुःखेषु सांधूंश्चाप्युपसेवते।</div><div class="line">तस्य साधुसमारम्भाद्बुद्ध्रिधर्मेषु राजते ॥ब्राह्मण उवाच।</div><div class="line" style="position:relative;">॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8843"><div class="verse-block verse-justify"><div class="line">ब्रवीषि सूनृतंधर्मं यस्य वक्ता न विद्यते।</div><div class="line">दिव्यप्रभावः सुमहानृषिरेव मतोसि मे ॥व्याध उवाच।</div><div class="line" style="position:relative;">॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8844"><div class="verse-block verse-justify"><div class="line">ब्राह्मणा वै महाभागाः पितरोऽग्रभुजः सदा।</div><div class="line" style="position:relative;">तेषां सर्वात्मना कार्यं प्रियं लोके मनीषिणा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8845"><div class="verse-block verse-justify"><div class="line">यत्तेषां च प्रियं तत्ते वक्ष्यामि द्विजसत्तम।</div><div class="line" style="position:relative;">नमस्कृत्वा ब्राह्मणेभ्यो ब्राह्मीं विद्यां निबोध मे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8846"><div class="verse-block verse-justify"><div class="line">इदं विश्वं जगत्सर्वमजगच्चापि सर्वशः।</div><div class="line" style="position:relative;">महाभूतात्मकं ब्रह्मन्नातः परतरं भवेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8847"><div class="verse-block verse-justify"><div class="line">महाभूतानि खं वायुरग्निरापस्तथा च भूः।</div><div class="line" style="position:relative;">शब्दः स्पर्शश्च रूपं च रसो गन्धश् तद्गुणाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8848"><div class="verse-block verse-justify"><div class="line">तेषामपि गुणाः सर्वे गुणवृत्तिः परस्परम्।</div><div class="line" style="position:relative;">पूर्वपूर्वगुणाः सर्वे क्रमशो गुणिषु त्रिषु॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8849"><div class="verse-block verse-justify"><div class="line">षष्ठी तु चेतना नाम मन इत्यभिधीयते।</div><div class="line" style="position:relative;">सप्तमी तु भवेद्बुद्धिरहंकारस्ततः परम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8850"><div class="verse-block verse-justify"><div class="line">इन्द्रियाणि च पञ्चात्मा रजः सत्वं तमस्तथा।</div><div class="line" style="position:relative;">इत्येष सप्तदशको राशिरव्यक्तसंज्ञकः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b8851"><div class="verse-block verse-justify"><div class="line">सर्वैरिहेन्द्रियार्थैस्तु व्यक्ताव्यक्तैः सुसंवृतैः।</div><div class="line">चतुर्विंसक इत्येष व्यकत्वाव्यक्तमयो गुणः।</div><div class="line" style="position:relative;">एतत्ते सर्वमाख्यातं किं भूयः श्रोतुमिच्छसि॥२२॥</div></div></div></div>
