<div class="verse-block-wrapper" id="03-araṇyaparva-b10462"><div class="verse-block verse-justify"><div class="line">प्रत्यक्षधर्मा भगवान्यस् तुष्टो हि कर्मभिः।</div><div class="line">सफलं तस्य जन्माहं मन्ये सद्धर्मचारिणः ॥व्यास उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10463"><div class="verse-block verse-justify"><div class="line">शिलोञ्छवृत्तिर्धर्मात्मा मुद्गलः संयतेन्द्रियः।</div><div class="line" style="position:relative;">आसीद्राजन्कुरुक्षेत्रे सत्यवागनसूयकः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10464"><div class="verse-block verse-justify"><div class="line">अतिथिव्रती क्रियावांश्च कापोतीं वृत्तिमास्थितः।</div><div class="line" style="position:relative;">सत्रमिष्टीकृतंनाम समुपास्ते महातपाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10465"><div class="verse-block verse-justify"><div class="line">सपुत्रदारो हि मुनिः पक्षाहारो बभूव ह।</div><div class="line" style="position:relative;">कपोतवृत्त्या पक्षेण व्रीहिद्रोयणमुपार्जयत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10466"><div class="verse-block verse-justify"><div class="line">दर्शं च पौर्णमासं च कुर्वन्विगतमन्सरः।</div><div class="line" style="position:relative;">देवतातिथिशेषेण कुरुते देहयापनम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10467"><div class="verse-block verse-justify"><div class="line">तत्रेन्द्रः सहितो देवैः साक्षात्रिभुवनेश्वरः।</div><div class="line" style="position:relative;">प्रत्यृह्णान्महाराज भागं पर्वणिपर्वणि॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10468"><div class="verse-block verse-justify"><div class="line">स पर्वकाल्यं कृत्वा तु मुनिवृत्त्या समन्वितः।</div><div class="line" style="position:relative;">अतिथिभ्यो ददावन्नं प्रहृष्टेनान्तरात्मना॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10469"><div class="verse-block verse-justify"><div class="line">व्रीहिद्रोणस्य तत्प्रीत्या ददतोऽन्नं महात्मनः।</div><div class="line" style="position:relative;">ऋषेर्मात्सर्यहीनस् वर्धत्यतिथिदर्शनात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10470"><div class="verse-block verse-justify"><div class="line">तच्छतान्यपि भुञ्जन्ति ब्राह्मणानां मनीषिणाम्।</div><div class="line" style="position:relative;">मुनेस्त्यागविशुद्ध्या तु तदन्नं वृद्धिमृच्छति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10471"><div class="verse-block verse-justify"><div class="line">तं तु रशुश्राव धर्मिष्ठं मुद्गलं संशितव्रतम्।</div><div class="line" style="position:relative;">दुर्वासा नृप दिग्वासास्तमथाभ्याजगाम ह॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10472"><div class="verse-block verse-justify"><div class="line">विभ्रच्चानियतं वेषमुन्मत्त इव पाण्डव।</div><div class="line" style="position:relative;">विकचः परुषा वाचो व्याहरन्विविधा मुनिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10473"><div class="verse-block verse-justify"><div class="line">अभिगम्याथ तं विप्रमुवाच मुनिसत्तमः।</div><div class="line" style="position:relative;">अन्नार्थिनमनुप्राप्तं विद्धि मां मुनिसत्तम॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10474"><div class="verse-block verse-justify"><div class="line">स्वागतं तेऽस्त्विति मुनिं मुद्गलः प्रत्यभाषत।</div><div class="line" style="position:relative;">पाद्यमाचमनीयं च प्रतिवेद्यान्नमुत्तमम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10475"><div class="verse-block verse-justify"><div class="line">प्रादात्स तपसोपात्तं क्षुधितायातिथिप्रियः।</div><div class="line" style="position:relative;">उन्मत्ताय परां श्रद्धामास्थाय स धृतव्रतः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10476"><div class="verse-block verse-justify"><div class="line">ततस्तदन्नं रसवत्स एव क्षुधयाऽन्वितः।</div><div class="line" style="position:relative;">बुभुजे कृत्स्नमुन्मत्तः प्रादात्तस्मै च मुद्गलः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10477"><div class="verse-block verse-justify"><div class="line">भुक्त्वा चान्नं ततः सर्वमुच्छिष्टेनात्मनस्ततः।</div><div class="line" style="position:relative;">अथाङ्गं लिलिपेऽन्नेन यथागतमगाच्च सः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10478"><div class="verse-block verse-justify"><div class="line">`तेनैवात्मानमालिप्य हसन्गायन्प्रधावति।</div><div class="line" style="position:relative;">नृत्यते धावते चैव बुद्ध्या तत्क्रोशते तथा'॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10479"><div class="verse-block verse-justify"><div class="line">एवं द्वितीये संप्राप्ते पर्वकाले मनीषिणः।</div><div class="line" style="position:relative;">आगम्य बुभुजे सर्वमन्नपुञ्छोपजीविनः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10480"><div class="verse-block verse-justify"><div class="line">निराहारस्तु स मुनिरुञ्छमार्जये पुनः।</div><div class="line" style="position:relative;">न चैनं विक्रियां नेतुमशकन्मुद्गलं क्षुधा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10481"><div class="verse-block verse-justify"><div class="line">न क्रोधो न च मात्सर्यं नावमानो न संभ्रमः।</div><div class="line" style="position:relative;">सपुत्रदारमुञ्छन्तमाविवेश द्विजोत्तमम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10482"><div class="verse-block verse-justify"><div class="line">तथा तमुञ्छधर्माणं दुर्वासा मुनिसत्तमम्।</div><div class="line" style="position:relative;">उपतस्थे यथाकालं षट्कृत्वः कृतनिश्चयः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10483"><div class="verse-block verse-justify"><div class="line">न चास् मनसः कश्चिद्विकारो दृश्यते मुनेः।</div><div class="line" style="position:relative;">शुद्धसत्वस्य शुद्धं स ददृशे निर्मलं मनः॥२४॥</div></div></div>
