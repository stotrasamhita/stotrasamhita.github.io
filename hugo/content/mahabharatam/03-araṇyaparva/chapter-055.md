<div class="verse-block-wrapper" id="03-araṇyaparva-b2409"><div class="verse-block verse-justify"><div class="line">बृहदश्व उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2410"><div class="verse-block verse-justify"><div class="line">वृते तु नैषधे भैम्या लोकपाला महौजसः।</div><div class="line" style="position:relative;">यान्तो ददृशुरायान्तं द्वापरं कलिना सह॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2411"><div class="verse-block verse-justify"><div class="line">अथाब्रवीत्कलिं शक्रः संप्रेक्ष्य बलवृत्रहा।</div><div class="line" style="position:relative;">द्वापरेण सहायेन कले ब्रूहि क्व यास्यसि॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2412"><div class="verse-block verse-justify"><div class="line">ततोऽब्रवीत्कलिः शक्रं दमयन्त्याः स्वयंवरम्।</div><div class="line" style="position:relative;">गत्वा हि वरयिष्ये तां मनो हि मम तां गतम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2413"><div class="verse-block verse-justify"><div class="line">तमब्रवीत्प्रहस्येन्द्रो निर्वृत्तः स स्वयंवरः।</div><div class="line" style="position:relative;">वृतस्तया नलो राजा पतिरस्मत्समीपतः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2414"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु शक्रेण कलिः कोपसमन्वितः।</div><div class="line" style="position:relative;">देवानामन्त्र्य तान्सर्वानुवाचेदं वचस्तदा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2415"><div class="verse-block verse-justify"><div class="line">देवानां मानुषं मध्ये यत्सा पतिमविन्दत।</div><div class="line" style="position:relative;">ननु तस्या भवेन्न्याय्यं विपुलं दण्डधारणम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2416"><div class="verse-block verse-justify"><div class="line">एवमुक्ते तु कलिना प्रत्यूचुस्ते दिवौकसः।</div><div class="line" style="position:relative;">अस्माभिः सभनुज्ञाते दमयन्त्या नलो वृतः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2417"><div class="verse-block verse-justify"><div class="line">का हि सर्वगुणोपेतं नाश्रयेत नलं नृपम्।</div><div class="line" style="position:relative;">यो वेद धर्मानखिलान्यथावच्चरितव्रतः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2418"><div class="verse-block verse-justify"><div class="line">योऽधीते चतुरो वेदान्सर्वानाख्यानपञ्चमान्।</div><div class="line" style="position:relative;">अहिंसानिरतो यश्च सत्यवादी दृढव्रतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2419"><div class="verse-block verse-justify"><div class="line">यस्मिन्दाक्ष्यं धृतिर्ज्ञानं तपः शौचं दमः शमः।</div><div class="line" style="position:relative;">ध्रुवाणि पुरुषव्याघ्रे लोकपालसमे नृपे॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2420"><div class="verse-block verse-justify"><div class="line">एवंरूपं नलं यो वै कामयेच्छपितुं कले।</div><div class="line" style="position:relative;">आत्मानं स शपेन्मूढो हन्यादात्मानमात्मना॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2421"><div class="verse-block verse-justify"><div class="line">एवंगुणं नलं यो वै कामयेच्छपितुं कले।</div><div class="line" style="position:relative;">कृच्छ्रे स नरके मञ्जेदगाधे विपुले ह्रदे॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2422"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा कलिं देवा द्वापरं च दिवं ययुः।</div><div class="line" style="position:relative;">ततो गतेषु देवेषु कलिर्द्वापरमब्रवीत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2423"><div class="verse-block verse-justify"><div class="line">संयन्तुं नोत्सहे कोपं नले वत्स्यामि द्वापर।</div><div class="line" style="position:relative;">भ्रंशयिष्यामि तं राज्यान्न भैम्या सह रंस्यते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2424"><div class="verse-block verse-justify"><div class="line">त्वमप्यक्षान्समाविश्य साहाय्यं कर्तुमर्हसि।</div><div class="line" style="position:relative;">`मम प्रियकृते ह्यस्मन्कृतवांश्च भविष्यसि'॥१६॥</div></div></div></div>
