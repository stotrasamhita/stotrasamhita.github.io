<div class="verse-block-wrapper" id="03-araṇyaparva-b1240"><div class="verse-block verse-justify"><div class="line">अतिवादान्मदाच्चैव मा धर्ममभिशङ्कथाः।</div><div class="line" style="position:relative;">धर्मातिशङ्की पुरुषस्तिर्यग्गतिपरायणः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1241"><div class="verse-block verse-justify"><div class="line">धर्मो यस्यातिशङ्क्यः स्यादार्षं वा दुर्बलात्मनः।</div><div class="line" style="position:relative;">वेदाच्छूद्र इवापेयात्स लोकादजरामरात्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1242"><div class="verse-block verse-justify"><div class="line">वेदाध्यायी धर्मपरः कुले जातो मनस्विनि।</div><div class="line" style="position:relative;">स्थविरेषु स योक्तव्यो राजभिर्धर्मचारिभिः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1243"><div class="verse-block verse-justify"><div class="line">पापीयान्स हि शूद्रेभ्यस्तस्करेभ्यो विशिष्यते।</div><div class="line" style="position:relative;">शास्त्रातिगो मन्दबुद्धिर्यो धर्ममतिशङ्कते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1244"><div class="verse-block verse-justify"><div class="line">प्रत्यक्षं हि त्वया दृष्ट ऋषिर्गच्छन्महातपाः।</div><div class="line" style="position:relative;">मार्कण्डेयोऽप्रमेयात्मा धर्मेण चिरजीविता॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1245"><div class="verse-block verse-justify"><div class="line">व्यासो वसिष्ठो मैत्रेयो नारदो लोमशः शुकः।</div><div class="line" style="position:relative;">अन्ये च ऋषयः सर्वे धर्मेणैव सुचेतसः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1246"><div class="verse-block verse-justify"><div class="line">प्रत्यक्षं पश्यसि ह्येतान्दिव्ययोगसमन्वितान्।</div><div class="line" style="position:relative;">शापानुग्रहणे शक्तान्देवेभ्योऽपि गरीयसः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1247"><div class="verse-block verse-justify"><div class="line">एते हि धर्ममेवादौ वर्णयन्ति सदाऽनधे।</div><div class="line" style="position:relative;">कर्तव्यममरप्रख्याः प्रत्यक्षागमबुद्धयः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1248"><div class="verse-block verse-justify"><div class="line">अतो नार्हसि कल्याणि धातारं धर्ममेव च।</div><div class="line" style="position:relative;">रजोमूढेन मनसा क्षेप्तुं शङ्कितुमेव च॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1249"><div class="verse-block verse-justify"><div class="line">उन्मत्तान्मन्यते बालः सर्वानागतनिश्चयान्।</div><div class="line" style="position:relative;">धर्मातिशङ्की नान्यस्मिन्प्रमाणमधिगच्छति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1250"><div class="verse-block verse-justify"><div class="line">इन्द्रियप्रीतिसंबद्धं यदिदं लोकसाक्षिकम्।</div><div class="line" style="position:relative;">एतावन्मन्यते बालो मोहमन्यत्र गच्छति॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1251"><div class="verse-block verse-justify"><div class="line">प्रायश्चित्तं न तस्यास्ति यो धर्ममतिशङ्कते।</div><div class="line" style="position:relative;">ध्यायन्स कृपणः पापो न लोकान्प्रतिपद्यते॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1252"><div class="verse-block verse-justify"><div class="line">प्रमाणाद्धि निवृत्तो हि वेदशास्त्रार्थनिन्दकः।</div><div class="line" style="position:relative;">कामलोभानुगो मूढो नरकं प्रतिपद्यते॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1253"><div class="verse-block verse-justify"><div class="line">आर्षं प्रमाणमुत्क्राम्य धर्मं न प्रतिपालयन्।</div><div class="line" style="position:relative;">सर्वशास्त्रातिगो मूढः शं जन्मसु न विन्दति॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1254"><div class="verse-block verse-justify"><div class="line">यस्तु नित्यं कृतमतिर्धर्ममेवाभिपद्यते।</div><div class="line" style="position:relative;">अशङ्कमानः कल्याणि सोऽमुत्रानन्त्यमश्नुते॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b1257"><div class="verse-block verse-justify"><div class="line">धर्म एव प्लवो नान्यः स्वर्गं द्रौपदि गच्छताम्।</div><div class="line" style="position:relative;">सैवः नौः सागरस्येव वणिजः पारमिच्छतः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1258"><div class="verse-block verse-justify"><div class="line">अफलो यदि धर्मः स्याच्चरितो धर्मचारिभिः।</div><div class="line" style="position:relative;">अप्रतिष्ठे तमस्येतज्जगन्मज्जेदनिन्दिते॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b1260"><div class="verse-block verse-justify"><div class="line">तपश्च ब्रह्मचर्यं च यज्ञः स्वाध्याय एव च।</div><div class="line" style="position:relative;">दानमार्जवमेतानि यदि स्युरफलानि वै॥२८॥</div></div></div>
