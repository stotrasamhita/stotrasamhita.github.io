<div class="verse-block-wrapper" id="03-araṇyaparva-b0700"><div class="verse-block verse-justify"><div class="line">समे निविष्टां सा सेना प्रभूतसलिलाशये।</div><div class="line" style="position:relative;">चतुरङ्गबलोपेता साल्वराजाभिपालिता॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0701"><div class="verse-block verse-justify"><div class="line">वर्जयित्वा श्मशानानि देवतायतनानि च।</div><div class="line" style="position:relative;">वल्मीकांश्चैव चैत्यांश्च संनिविष्टमभूद्बलम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0702"><div class="verse-block verse-justify"><div class="line">अनीकानां विभागेन पन्थानः सुकृतास्तथा।</div><div class="line" style="position:relative;">प्रथमा नवमाश्चैव साल्वस्य शिबिरे नृप॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0703"><div class="verse-block verse-justify"><div class="line">सर्वायुधसमोपेतं सर्वशस्त्रविशारदम्।</div><div class="line" style="position:relative;">रथनागाश्वकलिलं पदातिजनसंकुलम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0704"><div class="verse-block verse-justify"><div class="line">तुष्टपुष्टबलोपेतं वीरलक्षणलक्षितम्।</div><div class="line" style="position:relative;">विचित्रध्वजसन्नाहं विचित्ररथकार्मुकम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0705"><div class="verse-block verse-justify"><div class="line">संनिवेश्य च कौरव्य द्वारकायां नरर्षभ।</div><div class="line" style="position:relative;">अभिसारयामास तदा वेगैन पतगेन्द्रवत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0706"><div class="verse-block verse-justify"><div class="line">तदापतन्तं संदृश्य बलं साल्वपतेस्तदा।</div><div class="line" style="position:relative;">निर्याय योधयामासुः कुमारा वृष्णिनन्दनाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0707"><div class="verse-block verse-justify"><div class="line">असहन्तोऽभियानं तत्साल्वराजस्य कौरव।</div><div class="line" style="position:relative;">चारुदेष्णश्च साम्बश्च प्रद्युम्नश्च रमहारथः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0708"><div class="verse-block verse-justify"><div class="line">ते रथैर्दंशिताः सर्वेविचित्राभरणध्वजाः।</div><div class="line" style="position:relative;">संसक्ताः साल्वराजस्य बहुभिर्योधपुङ्गवैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0709"><div class="verse-block verse-justify"><div class="line">गृहीत्वा कार्मुकं साम्बः साल्वस्य सचिवं रणे।</div><div class="line" style="position:relative;">यीधयामास संहृष्टः क्षेमधूर्तिं चमूपतिम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0710"><div class="verse-block verse-justify"><div class="line">तस्य बाणमयं वर्षं जाम्बवत्याः सुतो महत्।</div><div class="line" style="position:relative;">मुमोच भरतश्रेष्ठ यथावर्षं सहस्रदृक्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0711"><div class="verse-block verse-justify"><div class="line">तद्बाणवर्षं तुमुलं विषेहे स चमूपतिः।</div><div class="line" style="position:relative;">क्षेमधूर्तिर्महाराज हिमवानिव निश्चलः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0712"><div class="verse-block verse-justify"><div class="line">ततः साम्बाय राजेन्द्र क्षेमधूर्तिरपि स्वयम्।</div><div class="line" style="position:relative;">मुमोच मायाविहितं शरजालं महत्तरम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0713"><div class="verse-block verse-justify"><div class="line">ततो मायामयं जालं माययैव विदार्यसः।</div><div class="line" style="position:relative;">साम्बः शरसहस्रेण रथमस्याभ्यवर्षत॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0714"><div class="verse-block verse-justify"><div class="line">ततः स विद्धः साम्बेन क्षेमधूर्तिश्चमूपतिः।</div><div class="line" style="position:relative;">अपायाज्जवनैरश्वैः साम्बबाणप्रपीडितः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0715"><div class="verse-block verse-justify"><div class="line">तस्मिन्विप्रद्रुते शरे साल्वस्याथ चमूपतौ।</div><div class="line" style="position:relative;">वेगवान्नाम दैतेयः सुतं मेऽभ्यद्रवद्बली॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0716"><div class="verse-block verse-justify"><div class="line">अभिपन्नस्तु राजेन्द्र साम्बो वृष्णिकुलोद्वहः।</div><div class="line" style="position:relative;">वेगं वेगवतो राजंस्तस्थौ वीरो विधारयन्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0717"><div class="verse-block verse-justify"><div class="line">स वेगवति कौन्तेय साम्बो वेगवतीं गदाम्।</div><div class="line" style="position:relative;">चिक्षेप तरसा वीरो व्याविद्ध्यन्सत्यविक्रमः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0718"><div class="verse-block verse-justify"><div class="line">तया त्वभिहतो राजन्वेगवान्न्यपतद्भुवि।</div><div class="line" style="position:relative;">वातरुग्णं इव क्षुण्णो जीर्णमूलोवनस्पतिः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0719"><div class="verse-block verse-justify"><div class="line">तस्मिन्निपतिते वीरे गदारुग्णे महासुरे।</div><div class="line" style="position:relative;">प्रविश्य महतीं सेनां योधयामास मे सुतः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b0720"><div class="verse-block verse-justify"><div class="line">चारुदेष्णेन संसक्तो विविन्ध्यो नाम दानवः।</div><div class="line" style="position:relative;">महारथः समाज्ञातो महाराज महाधनुः॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b0722"><div class="verse-block verse-justify"><div class="line">अन्योन्यस्याभिसंक्रुद्धावन्योन्यं जघ्नतुः शरैः।</div><div class="line" style="position:relative;">विनदन्तौ महाराज सिंहाविव महाबलौ॥२५॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b0724"><div class="verse-block verse-justify"><div class="line">स विविन्ध्याय सक्रोधः समाहूय महारथः।</div><div class="line" style="position:relative;">चिक्षेप मे सुतो राजन्स गतासुरथापतत्॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b0727"><div class="verse-block verse-justify"><div class="line">ततो निर्याय कौरव्य अवस्थाप्य च तद्बलम्।</div><div class="line" style="position:relative;">आनर्तानां महाराज प्रद्युम्नो वाक्यमब्रवीत्॥३०॥</div></div></div>
