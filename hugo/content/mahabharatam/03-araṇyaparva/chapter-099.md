<div class="verse-block-wrapper" id="03-araṇyaparva-b4483"><div class="verse-block verse-justify"><div class="line">युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4484"><div class="verse-block verse-justify"><div class="line">भूय एवाहमिच्छामि महर्षेस्तस्य धीमतः।</div><div class="line">कर्मणां विस्तरं श्रोतुमगस्त्यस्य द्विजात्तम ॥लोमश उवाच।</div><div class="line" style="position:relative;">॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4485"><div class="verse-block verse-justify"><div class="line">शृणु राजन्कथां दिव्यामद्भुतामतिमानुषीम्।</div><div class="line" style="position:relative;">अगस्त्यस्य महाराज प्रभावममितौजसः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4486"><div class="verse-block verse-justify"><div class="line">आसन्कृतयुगे घोरा दानवा युद्धदुर्मदाः।</div><div class="line" style="position:relative;">कालकेया इतिख्याता गणाः परमदारुणाः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4487"><div class="verse-block verse-justify"><div class="line">ते तु वृत्रं समाश्रित्य नानाप्रहरणोद्यताः।</div><div class="line" style="position:relative;">समन्तात्पर्यधावन्त महेन्द्रप्रमुखान्सुरान्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4488"><div class="verse-block verse-justify"><div class="line">ततो वृत्रवधे यत्नमकुर्वंस्त्रिदशाः पुरा।</div><div class="line" style="position:relative;">पुरंदरं पुरस्कृत्य ब्रह्माणमुपतस्थिरे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4489"><div class="verse-block verse-justify"><div class="line">कृताञ्जलींस्तु तान्सर्वान्परमेष्ठीत्युवाच ह।</div><div class="line" style="position:relative;">विदितं मे सुराः सर्वं यद्वः कार्यं चिकीर्षितम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4490"><div class="verse-block verse-justify"><div class="line">तमुपायं प्रवक्ष्यामि यथा वृत्रं वधिष्यथ।</div><div class="line" style="position:relative;">दधीच इतिविख्यातो महानृषिरुदारधीः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4491"><div class="verse-block verse-justify"><div class="line">तं गत्वा सहिताः सर्वेवरं वै संप्रयाचत।</div><div class="line" style="position:relative;">स वो दास्यति धर्मात्मा सुप्रीतेनान्तरात्मना॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4492"><div class="verse-block verse-justify"><div class="line">स वाच्यः सहितैः सर्वैर्भवद्भिर्जयकाङ्क्षिभिः।</div><div class="line" style="position:relative;">स्वान्यस्थीनि प्रयच्छेति त्रैलोक्यस् हिताय वै॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4493"><div class="verse-block verse-justify"><div class="line">स शरीरं समुत्सृज्य स्वान्यस्थीनि प्रदास्यति।</div><div class="line" style="position:relative;">तस्यास्थिभिर्महाघोरं वज्रं संस्क्रियतां दृढम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4494"><div class="verse-block verse-justify"><div class="line">महच्छत्रुहणं घोरं ष़डश्चं भीमनिःस्वनम्।</div><div class="line" style="position:relative;">तेन वज्रेण वै वृत्रं वधिष्यति शतक्रतुः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4495"><div class="verse-block verse-justify"><div class="line">एतद्वः सर्वमाख्यातं तस्माच्छीघ्रं विधीयताम्।</div><div class="line" style="position:relative;">एवमुक्तास्ततो देवा अनुज्ञाप्य पितामहम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4496"><div class="verse-block verse-justify"><div class="line">नारायणं पुरस्कृत्य दधीचस्याश्रमं ययुः।</div><div class="line" style="position:relative;">सरस्वत्याः परे पारे नानाद्रुमलतावृतम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4497"><div class="verse-block verse-justify"><div class="line">षट्पदोद्गीतनिनदैर्विघुष्टं सामगैरिव।</div><div class="line" style="position:relative;">पुंस्कोकिलरवोन्मिश्रं जीवंजीवकनादितम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4498"><div class="verse-block verse-justify"><div class="line">महिषैश्च वराहैश्च सृमरैश्चमरैरपि।</div><div class="line" style="position:relative;">तत्र तत्रानुचरितं शार्दूलभयवर्जितैः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4499"><div class="verse-block verse-justify"><div class="line">करेणुभिर्वारणैश्च प्रभिन्नकरटामुखैः।</div><div class="line" style="position:relative;">सरोवगाढैः क्रीडद्भिः समन्तादनुनादितम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4500"><div class="verse-block verse-justify"><div class="line">सिंहव्याघ्रैर्महानादान्नदद्भिरनुनादितम्।</div><div class="line" style="position:relative;">अपरैश्चापि संलीनैर्गुहाकन्दरशायिभिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4501"><div class="verse-block verse-justify"><div class="line">तेषु तेष्ववकाशेषु शोभितं सुमनोरमम्।</div><div class="line" style="position:relative;">त्रिविष्टपसमप्रख्यं दधीचाश्रममागमन्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4502"><div class="verse-block verse-justify"><div class="line">तत्रापश्यन्दधीचं ते दिवाकरसमद्युतिम्।</div><div class="line" style="position:relative;">जाज्वल्यमानं वपुषा यथा साक्षात्पितामहम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4503"><div class="verse-block verse-justify"><div class="line">तस्य पादौ सुरा राजन्नभिवाद्य प्रणम्य च।</div><div class="line" style="position:relative;">अयाचन्त वरं सर्वे यथोक्तं परमेष्ठिना॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4504"><div class="verse-block verse-justify"><div class="line">ततो दधीचः परमप्रतीतःसुरोत्तमांस्तानिदमभ्युवाच।</div><div class="line" style="position:relative;">करोमि यद्वो हितमद्य देवाःस्वं चापि देहंस्वयमुत्सृजामि॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4505"><div class="verse-block verse-justify"><div class="line">स एवमुक्त्वा द्विपदांवरिष्ठःप्राणान्वशी स्वान्सहसोत्ससर्ज।</div><div class="line" style="position:relative;">ततः सुरास्ते जगृहुः परासो-रस्थीनि तस्याथ यथोपदेशम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4506"><div class="verse-block verse-justify"><div class="line">प्रहृष्टरूपाश्च जयाय देवा-स्त्वष्टारमागम् तमर्थमूचुः।</div><div class="line" style="position:relative;">त्वष्टा तु तेषां वचनं निशम्यप्रहृष्टरूपः प्रयतः प्रयत्नात्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4507"><div class="verse-block verse-justify"><div class="line">चकार वज्रं भृशमुग्ररूपंकृत्वा च शक्रं स उवाच हृष्टः।</div><div class="line" style="position:relative;">अनेन वज्रप्रवरेण देवभस्मीकुरुष्वाद्य सुरारिमुग्रम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4508"><div class="verse-block verse-justify"><div class="line">ततो हतारिः सगणः सुखं वैप्रशाधिकृत्स्नं त्रिदिवं दिविष्ठः।</div><div class="line" style="position:relative;">त्वष्ट्रा तथोक्तस्तु पुरंदरस्त-द्वज्रं प्रहृष्टः प्रयतो ह्यगृह्नात्॥२६॥</div></div></div></div>
