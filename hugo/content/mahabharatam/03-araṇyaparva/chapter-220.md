<div class="verse-block-wrapper" id="03-araṇyaparva-b9047"><div class="verse-block verse-justify"><div class="line">अग्निर्यदा चैक एव बहुत्वं चास्य कर्मसु।</div><div class="line" style="position:relative;">दृश्यते भगवन्सर्वमेतदिच्छामि वेदितुम्॥४॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b9049"><div class="verse-block verse-justify"><div class="line">एतदिच्छाम्यहं त्वत्तः श्रोतुं भार्गवसत्तम।</div><div class="line">कौतूहलसमाविष्टो याथातथ्यं महामुने ॥मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9050"><div class="verse-block verse-justify"><div class="line">अत्राप्युदाहरन्तीममितिहासं पुरातनम्।</div><div class="line" style="position:relative;">यथा क्रुद्धो हुतवहस्तपस्तप्तुं वनं गतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9051"><div class="verse-block verse-justify"><div class="line">यथा च भगवानग्निः स्वयमेवाङ्गेराऽभवत्।</div><div class="line" style="position:relative;">संतापयंश्च प्रभया नाशयंस्तिमिरावलिम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9052"><div class="verse-block verse-justify"><div class="line">पुराङ्गिरा महाबाहो चचार तप उत्तमम्।</div><div class="line">आश्रमस्थो महाभागो हव्यवाहं विशेषयन्।</div><div class="line" style="position:relative;">यथाग्निर्भूत्वा तु तदा जगत्सर्वं व्यकाशयत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9053"><div class="verse-block verse-justify"><div class="line">तपश्चरंस्तु हुतभुक्संतप्तस्तस्य तेजसा।</div><div class="line" style="position:relative;">भृशं ग्लानश्चतेजस्वी न च किंचित्प्रजज्ञिवान्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9054"><div class="verse-block verse-justify"><div class="line">अथ संचिन्तयामास भगवान्हव्यवाहनः।</div><div class="line" style="position:relative;">अन्योऽग्निरिव लोकानां ब्रह्मणा संप्रकल्पितः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9055"><div class="verse-block verse-justify"><div class="line">अग्नित्वं विप्रनष्टं हि तप्यमानस् मे तपः।</div><div class="line" style="position:relative;">कथमग्निः पुनरहं भवेयमिति चिन्त्य सः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9056"><div class="verse-block verse-justify"><div class="line">अपश्यदग्निवल्लोकांस्तापयन्तं महामुनिम्।</div><div class="line" style="position:relative;">सोपासर्पच्छनैर्भीतस्तमुवाच तदाङ्गिराः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9057"><div class="verse-block verse-justify"><div class="line">शीघ्रमेव भवस्वाग्निस्त्वं पुनर्लोकभावनः।</div><div class="line" style="position:relative;">विज्ञातश्चासि लोकेषु त्रिषु संस्थानचारिषु॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9058"><div class="verse-block verse-justify"><div class="line">त्वमग्ने प्रथमः सृष्टो ब्र्हमणा तिमिरापर्हः।</div><div class="line">स्वस्थानं प्रतिपद्यस्व शीघ्रमेव तमो नुद ॥अग्निरुवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9059"><div class="verse-block verse-justify"><div class="line">नष्टकीर्तिरहं लोके भवाञ्जातो हुताशनः।</div><div class="line" style="position:relative;">भवन्तमेव ज्ञास्यन्ति पावकं न तु मां जनाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9060"><div class="verse-block verse-justify"><div class="line">निक्षिपम्यहमग्नित्वं त्वमग्निः प्रथमो भव।</div><div class="line">भविष्यामि द्वितीयोऽहं प्राजापत्यक एव च ॥अङ्गिरा उवाच।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b9061"><div class="verse-block verse-justify"><div class="line">कुरु पुण्यं प्रजास्वर्ग्यं भवाग्निस्तिमिरापहः।</div><div class="line">मां च देव कुरुष्वाग्ने प्रथमं पुत्रमञ्जसा ॥मार्कण्डेय उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
