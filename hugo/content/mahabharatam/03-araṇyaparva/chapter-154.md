<div class="verse-block-wrapper" id="03-araṇyaparva-b6261"><div class="verse-block verse-justify"><div class="line">स गत्वानलिनीं रम्यां राक्षसैरभिरक्षिताम्।</div><div class="line" style="position:relative;">कैलासशिरे रम्ये ददर्श शुभकानने॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6262"><div class="verse-block verse-justify"><div class="line">कुबेरभुवनाभ्याशे जातां पर्वतनिर्झरैः।</div><div class="line" style="position:relative;">सुरम्यां विपुलच्छायां नानाद्रुमलताकुलाम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6263"><div class="verse-block verse-justify"><div class="line">हरिताम्बुजसंछन्नां दिव्यां कनकपुष्कराम्।</div><div class="line" style="position:relative;">नानापक्षिजनाकीर्णां सूपतीर्थामकर्दमाम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6264"><div class="verse-block verse-justify"><div class="line">अतीव रम्यां सुजलां जातां पर्वतसानुषु।</div><div class="line" style="position:relative;">विचित्रभूतां लोकस्य शुभामद्भुतदर्शनाम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6265"><div class="verse-block verse-justify"><div class="line">तत्रामृतरसं शीतं लघु कुन्तीसुतः शुभम्।</div><div class="line" style="position:relative;">ददर्शविमलं तोयं पिबंश्च बहु पाण्डवः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6266"><div class="verse-block verse-justify"><div class="line">तां तु कपुष्करिणीं रम्यां दिव्यसौगन्धिकावृताम्।</div><div class="line" style="position:relative;">जातरूपमयैः पद्मैश्छन्नां परमगन्धिभिः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6267"><div class="verse-block verse-justify"><div class="line">वैडूर्यवरनालैश्च बहुचित्रैर्मनोरमैः।</div><div class="line" style="position:relative;">हंसकारण्डवोद्धूतैः सृजद्भिरमलं रजः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6268"><div class="verse-block verse-justify"><div class="line">आक्रीडं राजराजस् कुबेरस्य महात्मनः।</div><div class="line" style="position:relative;">गन्धर्वैरप्सरोभिश्च देवैश्च परमार्चिताम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6269"><div class="verse-block verse-justify"><div class="line">सेवितामृषिभिर्दिव्यैर्यक्षैः किंपुरुषैस्तथा।</div><div class="line" style="position:relative;">राक्षसैः किंनरैश्चापि गुप्तां वैश्रवणेन च॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6270"><div class="verse-block verse-justify"><div class="line">तां च दृष्ट्वैव कौन्तेयो भीमसेनो महाहलः।</div><div class="line" style="position:relative;">बभूव परमप्रीतो दिव्यंप्रेक्ष्य सरो महत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6271"><div class="verse-block verse-justify"><div class="line">तच्च क्रोवशा नाम राक्षसा राजशासनात्।</div><div class="line" style="position:relative;">रक्षन्ति शतसाहस्राश्चित्रायुधपरिच्छदाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6272"><div class="verse-block verse-justify"><div class="line">ते तु दृष्ट्वैव कौन्तेयमजिनैः परिवारितम्।</div><div class="line" style="position:relative;">रुक्माङ्गदधरं वीरं भीमं भीमपराक्रमम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6273"><div class="verse-block verse-justify"><div class="line">सायुधं बद्धनिस्त्रिंशमशङ्कितमरिंदमम्।</div><div class="line" style="position:relative;">पुष्करप्सुमुपायान्तमन्योन्यमभिचुक्रुशुः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6274"><div class="verse-block verse-justify"><div class="line">अयं पुरुषशार्दूलः सायुधोऽजिनसंवृतः।</div><div class="line" style="position:relative;">यच्चिकीर्षुरिह प्राप्तस्तत्संप्रष्टुमिहार्हथ॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6275"><div class="verse-block verse-justify"><div class="line">ततः सर्वे महाबाहुं समासाद्यवृकोदरम्।</div><div class="line" style="position:relative;">तेजोयुक्तमपृच्छन्त कस्त्वमाख्यातुमर्हसि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b6276"><div class="verse-block verse-justify"><div class="line">मुनिवेषधरश्चैव सायुधश्चैव लक्ष्यसे।</div><div class="line" style="position:relative;">यदर्थमाभिसंप्राप्तस्तदाचक्ष्यमहामते॥१७॥</div></div></div></div>
