<div class="verse-block-wrapper" id="03-araṇyaparva-b4705"><div class="verse-block verse-justify"><div class="line">ददाह सुमहातेजा मन्दबुद्धीन्स सागरान्।</div><div class="line" style="position:relative;">षष्टिं रतानि सहस्राणि युगपन्मुनिसत्तमः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4706"><div class="verse-block verse-justify"><div class="line">तान्दृष्ट्वा भस्मसाद्भूतान्नारदः सुमहातपाः।</div><div class="line" style="position:relative;">सगरान्तिकमागत्य तच्च तस्मै न्यवेदयत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4707"><div class="verse-block verse-justify"><div class="line">स तच्छ्रुत्वा वचो घोरं राजा मुनिमुखोद्गतम्।</div><div class="line" style="position:relative;">महूर्तं विमना भूत्वा स्थाणोर्वाक्यमचिन्तयत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4708"><div class="verse-block verse-justify"><div class="line">`स पुत्रनिधनोत्थेन दुःखेन समभिप्लुतः।</div><div class="line" style="position:relative;">आत्मानमात्मनाऽऽश्वास्य हयमेवान्वचिन्तयत्'॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4709"><div class="verse-block verse-justify"><div class="line">अशुमन्तं समाहूय असमञ्जसुतं तदा।</div><div class="line" style="position:relative;">पौत्रं भरतशार्दूल इदं वचनमब्रवीत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4710"><div class="verse-block verse-justify"><div class="line">षष्टिस्तात सहस्राणि पुत्राणाममितौजसाम्।</div><div class="line" style="position:relative;">कापिलं तेज आसाद्य मत्कृतेनिधनं गताः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4711"><div class="verse-block verse-justify"><div class="line">तव चापि पिता तात परित्यक्तो मयाऽनघ।</div><div class="line">धर्मं संरक्षमाणेन पौराणां हितमिच्छता ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4712"><div class="verse-block verse-justify"><div class="line">किमर्थं राजशार्दूलः सगरः पुत्रमात्मजम्।</div><div class="line">त्यक्तवान्दुस्त्यजं वीरं तन्मे ब्रूहि तपोधन ॥लोमश उवाच।</div><div class="line" style="position:relative;">॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4713"><div class="verse-block verse-justify"><div class="line">असमञ्ज इति ख्यातः सगरस्य सुतो ह्यभूत्।</div><div class="line" style="position:relative;">यं शैव्या जनयामास पौराणां स हि दारकान्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4714"><div class="verse-block verse-justify"><div class="line">`क्रीडतः सहसाऽऽसाद्य तत्रतत्र महीपते'।</div><div class="line" style="position:relative;">चूडासु क्रोशतो गृह्यनद्यां चिक्षेप दुर्बलान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4715"><div class="verse-block verse-justify"><div class="line">ततः पौराः समाजग्मुर्भयशोकपरिप्लुताः।</div><div class="line" style="position:relative;">सगरं चाभ्यभाषन्त सर्वे प्राञ्जलयः स्थिताः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4716"><div class="verse-block verse-justify"><div class="line">त्वं नस्त्राता महाराज परचक्रादिभिर्भयात्।</div><div class="line" style="position:relative;">असमञ्जभयाद्धोरात्ततो नस्त्रातुनर्हसि॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4717"><div class="verse-block verse-justify"><div class="line">पौराणां वचनं श्रुत्वा घोरं नृपतिसत्तमः।</div><div class="line" style="position:relative;">मुहूर्तं विमना भूत्वासचिवानिदमब्रवीत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4718"><div class="verse-block verse-justify"><div class="line">असमञ्ज पुरादद्य सुतो मे विप्रवास्यताम्।</div><div class="line" style="position:relative;">यदि वो मत्प्रियं कार्यमेतच्छीध्रं विधीयताम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4719"><div class="verse-block verse-justify"><div class="line">एवमुक्ता नरेन्द्रेण सचिवास्ते नराधिप।</div><div class="line" style="position:relative;">यथोक्तं त्वरिताश्चक्रुर्यथाऽऽज्ञापितवान्नृपः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4720"><div class="verse-block verse-justify"><div class="line">एतत्ते सर्वमाख्यातं यथा पुत्रो महात्मना।</div><div class="line" style="position:relative;">पौराणां हितकामेन सगरेण विवासितः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4721"><div class="verse-block verse-justify"><div class="line">अंशुमांस्तु महेष्वासो यदुक्तः सगरेण हि।</div><div class="line">तत्ते सर्वंप्रवक्ष्यामि कीर्त्यमानं निबोध मे ॥सगर उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4722"><div class="verse-block verse-justify"><div class="line">पितुश् तेऽहंत्यागेन पुत्राणां निधनेन च।</div><div class="line" style="position:relative;">अलाभेन तथाऽश्वस् परितप्यामि पुत्रक॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4723"><div class="verse-block verse-justify"><div class="line">तस्माद्दुःखाभिसंतप्तं यज्ञविघ्नाच्च मोहितम्।</div><div class="line">हयस्यानयनात्पौत्र नरकान्मां समुद्धर ॥लोमश उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4724"><div class="verse-block verse-justify"><div class="line">अंशुमानेवमुक्तस्तु सगरण महात्मना।</div><div class="line" style="position:relative;">जगाम दुःखात्तं देशं यत्रवै दारिता मही॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4725"><div class="verse-block verse-justify"><div class="line">स तु तेनैव मार्गेण समुद्रं प्रविवेश ह।</div><div class="line" style="position:relative;">अपश्यच्च महात्मानं कपिलं तुरगं च तम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4726"><div class="verse-block verse-justify"><div class="line">स दृष्ट्वा तेजसो राशिं पुराणमृपिसत्तमम्।</div><div class="line" style="position:relative;">प्रणम्य शिरसा भूमौ कार्यमस्मै न्यवेदयत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4727"><div class="verse-block verse-justify"><div class="line">ततः प्रीतो महाराज कपिलोंऽशुमतोऽभवत्।</div><div class="line" style="position:relative;">उवाच चैनं धर्मात्मा वरदोस्मीति भारत॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4728"><div class="verse-block verse-justify"><div class="line">स वव्रे तुरगं तत्र प्रथमं यज्ञकारणात्।</div><div class="line" style="position:relative;">द्वितीयमुदकं वव्रे पितॄणां पावनेच्छया॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4729"><div class="verse-block verse-justify"><div class="line">तमुवाच महातेजाः कपिलो मुनिपुङ्गवः।</div><div class="line" style="position:relative;">ददानि तव भद्रं ते यद्यत्प्रार्थयसेऽनघ॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4730"><div class="verse-block verse-justify"><div class="line">त्वयि क्षमा च धर्मश्च सत्यं चापि प्रतिष्ठितम्।</div><div class="line" style="position:relative;">त्वया कृतार्थः सगरः पुत्रवांश्च त्वया पिता॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4731"><div class="verse-block verse-justify"><div class="line">तव चैव प्रभावेन स्वर्गं यास्यन्ति सागराः।</div><div class="line" style="position:relative;">`शलभत्वं गता ये ते मम क्रोधहुताशने'॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4732"><div class="verse-block verse-justify"><div class="line">पौत्रश्च ते त्रिपथगां त्रिदिवादानयिष्यति।</div><div class="line" style="position:relative;">पावनार्थं सागराणां तोषयित्वा महेश्वरम्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4733"><div class="verse-block verse-justify"><div class="line">हयं नयस्व भद्रं ते याज्ञिं नरपुङ्गव।</div><div class="line" style="position:relative;">यज्ञः समाप्यतां तात सगरस्य महात्मनः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4734"><div class="verse-block verse-justify"><div class="line">अंशुमानेवमुक्तस्तु कपिलेन महात्मना।</div><div class="line" style="position:relative;">आजगाम हयं गृह्य यज्ञवाटं महात्मनः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4735"><div class="verse-block verse-justify"><div class="line">सोभिवाद्य ततः पादौ सगरस्य महात्मनः।</div><div class="line" style="position:relative;">मूर्ध्नि तनाप्युपाघ्रातस्तस्मै सर्वंनय्वेदयत्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4736"><div class="verse-block verse-justify"><div class="line">यथा दृष्टं श्रुतं चापि सागराणां क्षयं तथा।</div><div class="line" style="position:relative;">तं चास्मै हयमाचष्ट यज्ञवाटमुपागतम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4737"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा सगरो राजा पुत्रजं दुःखमत्यजत्।</div><div class="line" style="position:relative;">अंशुमन्तं च संपूज्यसमापयत तं क्रतुम्॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4738"><div class="verse-block verse-justify"><div class="line">समाप्तयज्ञः सगरो देवैः सर्वैः सभाजितः।</div><div class="line" style="position:relative;">पुत्रत्वे कल्पयामास समुद्रं वरुणालयम्॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4739"><div class="verse-block verse-justify"><div class="line">प्रशास्य सुचिरं कालं राज्यं राजीवलोचनः।</div><div class="line" style="position:relative;">पौत्रे भारं समवेश्य जगाम त्रिदिवं तदा॥३९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4740"><div class="verse-block verse-justify"><div class="line">अंशुमानपि धर्मात्मा महीं सागरमेखलाम्।</div><div class="line" style="position:relative;">प्रशशास महाराज यथैवास् पितामहः॥४०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4741"><div class="verse-block verse-justify"><div class="line">तस्य पुत्रः समभवद्दिलीपो नाम धर्मवित्।</div><div class="line" style="position:relative;">तस्मिन्राज्यं समाधाय अंशुमानपि संस्थितः॥४१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4742"><div class="verse-block verse-justify"><div class="line">दिलीपस्तु ततः श्रुत्वा पितॄणां निधनं महत्।</div><div class="line" style="position:relative;">पर्यतप्यत दुःखेन तेषां गतिमचिन्तयत्॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4743"><div class="verse-block verse-justify"><div class="line">गङ्गावतरणए यत्नं सुमहच्चाकरोन्नृपः।</div><div class="line" style="position:relative;">न चावतारयामास चेष्टमानो यथाबलम्॥४३॥</div></div></div>
