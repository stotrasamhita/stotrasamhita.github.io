<div class="verse-block-wrapper" id="03-araṇyaparva-b2297"><div class="verse-block verse-justify"><div class="line">तेभ्यः प्रतिज्ञाय नलः करिष्य इति भारत।</div><div class="line" style="position:relative;">अथैतान्परिपप्रच्छ कृताञ्जलिरुपस्थितः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2298"><div class="verse-block verse-justify"><div class="line">के वै भवन्तः कश्चासौ यस्याहं दूत ईप्सितः।</div><div class="line" style="position:relative;">किंच तत्रमया कार्यं कथयध्वं यथातथम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2299"><div class="verse-block verse-justify"><div class="line">एवमुक्ते नैषधेन मघवानभ्यभाषत।</div><div class="line" style="position:relative;">अमरान्वै निबोधास्मान्दमयन्त्यर्थमागतान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2300"><div class="verse-block verse-justify"><div class="line">अहमिन्द्रोऽयमग्निश्च तथैवायमपांपतिः।</div><div class="line" style="position:relative;">शरीरान्तकरो नॄणां यमोऽयमपि पार्थिव॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2301"><div class="verse-block verse-justify"><div class="line">त्वं वै समागतानस्मान्दमयन्त्यै निवेदय।</div><div class="line" style="position:relative;">लोकपाला महेन्द्राद्याः समायान्ति दिदृक्षवः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2302"><div class="verse-block verse-justify"><div class="line">प्राप्तुमिच्छन्ति देवास्त्वां शक्रोऽग्निर्वरुणो यमः।</div><div class="line" style="position:relative;">तेषामन्यतमं देवं पतित्वे वरयस्व ह॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2303"><div class="verse-block verse-justify"><div class="line">एवमुक्तः स शक्रेण नलः प्राञ्जलिरब्रवीत्।</div><div class="line" style="position:relative;">एकार्थसमवेतं मां न प्रेषयितुमर्हथ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2304"><div class="verse-block verse-justify"><div class="line">कथ हि जातसंकल्पः स्त्रियमुत्सृजते पुमान्।</div><div class="line" style="position:relative;">परार्थमीदृशं वक्तं तद्वै पश्यामरेश्वर॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2305"><div class="verse-block verse-justify"><div class="line">`एवमुक्तो नैषधेन मघवान्पुनरब्रवीत्।</div><div class="line">'करिष्य इतिसंश्रुत्य पूर्वमस्मासु नैषध।</div><div class="line" style="position:relative;">न करिष्यसि कस्मात्त्वं व्रज नैषध माचिरम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2306"><div class="verse-block verse-justify"><div class="line">`स वै त्वमागतानस्मान्दमयन्त्यै निवेदय।</div><div class="line">श्रेयसा योक्ष्यसे हि त्वं कुर्वन्नमरशासनम् ॥'बृहदश्व उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2307"><div class="verse-block verse-justify"><div class="line">एवमुक्तः स देवैस्तैर्नैषधः पुनरब्रवीत्।</div><div class="line" style="position:relative;">सुरक्षितानि वेश्मानि प्रवेष्टुं कथमुत्सहे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2308"><div class="verse-block verse-justify"><div class="line">प्रवेक्ष्यसीति तं शक्रः पुनरेवाभ्यभाषत।</div><div class="line" style="position:relative;">जगाम स तथेत्युक्त्वा दमयन्त्या निवेशनम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2309"><div class="verse-block verse-justify"><div class="line">ददर्श तत्र वैदर्भीं सखीगणसमावृताम्।</div><div class="line" style="position:relative;">देदीप्यमानां वपुषा श्रिया च वरवर्णिनीम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2310"><div class="verse-block verse-justify"><div class="line">अतीव सुकुमाराङ्गीं तनुमध्यां सुलोचनाम्।</div><div class="line" style="position:relative;">आक्षिपन्तीमिव च तां शशिनं स्वेन तेजसा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2311"><div class="verse-block verse-justify"><div class="line">तस्य दृष्ट्वैव ववृधे कामस्तां चारुहासिनीम्।</div><div class="line" style="position:relative;">सत्यं चिकीर्षमाणस्तुधारयामास हृच्छयम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2312"><div class="verse-block verse-justify"><div class="line">ततस्ता नैषधं दृष्ट्वा संभ्रान्ताः परमाङ्गनाः।</div><div class="line" style="position:relative;">आसनेभ्यः समुत्पेतुस्तेजसा तस्य धर्षिताः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2313"><div class="verse-block verse-justify"><div class="line">प्रशशंसुश्च मुप्रीता नलं ता विस्मयान्विताः।</div><div class="line" style="position:relative;">न चैनमभ्यभाषन्त मनोभिस्त्वभ्यपूजयन्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2314"><div class="verse-block verse-justify"><div class="line">अहो रूपमहो कान्तिरहो धैर्यं महात्मनः।</div><div class="line" style="position:relative;">कोऽयं देवोऽथवा यक्षो गन्धर्वो वा भविष्यति॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2315"><div class="verse-block verse-justify"><div class="line">न तास्तं शक्नुवन्ति स्म व्याहर्तुमपि किंचन।</div><div class="line" style="position:relative;">तेजसा धर्षितास्तस्य लज्जावत्यो वराङ्गनाः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2316"><div class="verse-block verse-justify"><div class="line">अथैनं स्मयमानेव स्मितपूर्वाभिभाषिणी।</div><div class="line" style="position:relative;">दमयन्ती नलं वीरमभ्यभाषत विस्मिता॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2317"><div class="verse-block verse-justify"><div class="line">कस्त्वं सर्वानवद्याङ्ग मम हृच्छयवर्धन।</div><div class="line" style="position:relative;">प्राप्तोस्यमरवद्वीर ज्ञातुमिच्छामि तेऽनघ॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2318"><div class="verse-block verse-justify"><div class="line">कथमागमनं चेह कथं चासि न लक्षितः।</div><div class="line" style="position:relative;">सुरक्षितं हि मे वेश्म राजा चैवोग्रशासनः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2319"><div class="verse-block verse-justify"><div class="line">एवमुक्तस्तु वैदर्भ्या नलस्तां प्रत्युवाच ह।</div><div class="line" style="position:relative;">नलं मां विद्धि कल्याणि देवदूतमिहागतम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2320"><div class="verse-block verse-justify"><div class="line">देवास्त्वां प्राप्तुमिच्छन्ति शक्रोऽग्निर्वरुणो यमः।</div><div class="line" style="position:relative;">तेषामन्यतमं देवं पतिं वरय शोभने॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2321"><div class="verse-block verse-justify"><div class="line">तेषामेव प्रभावेण प्रविष्टोऽहमलक्षितः।</div><div class="line" style="position:relative;">प्रविशन्तं न मां कश्चिदपश्यन्नाप्यवारयत्॥२६॥</div></div></div>
