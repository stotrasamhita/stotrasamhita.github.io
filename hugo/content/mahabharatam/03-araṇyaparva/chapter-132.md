<div class="verse-block-wrapper" id="03-araṇyaparva-b5410"><div class="verse-block verse-justify"><div class="line">एवमाशीः प्रयुक्ता हि दक्षेण यजता पुरा।</div><div class="line" style="position:relative;">इह ये वै मरिष्यन्ति ते वै स्वर्गजितो नराः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5411"><div class="verse-block verse-justify"><div class="line">एषा सरस्वती रम्या दिव्या चौघवती नदी।</div><div class="line" style="position:relative;">एतद्विनशनं नाम सरस्वत्या विशांपते॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5412"><div class="verse-block verse-justify"><div class="line">द्वारं निषादराष्ट्रस्य येषां दोषात्सरस्वती।</div><div class="line" style="position:relative;">प्रविष्टा पृथिवीं वीर मा निषादा हि मां विदुः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5413"><div class="verse-block verse-justify"><div class="line">एष वै चमसोद्भेदो यत्र दृश्या सरस्वती।</div><div class="line" style="position:relative;">यत्रैनामभ्यवर्तन्त दिव्याः पुण्याः समुद्रगाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5414"><div class="verse-block verse-justify"><div class="line">एतत्सिन्धोर्महत्तीर्थं यत्रागस्त्यमरिंदम।</div><div class="line" style="position:relative;">लोपामुद्रा समागम्य भर्तारमवृणीत वै॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5415"><div class="verse-block verse-justify"><div class="line">एतत्प्रकाशते तीर्थं प्रभासं भास्करद्युते।</div><div class="line" style="position:relative;">इन्द्रस्य दयितं पुण्यं पवित्रं पापनाशनम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5416"><div class="verse-block verse-justify"><div class="line">एतद्विष्णुपदं नाम दृश्यते तीर्थमुत्तमम्।</div><div class="line" style="position:relative;">एषां रम्या विपाशा च नदी परमपावनी॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5417"><div class="verse-block verse-justify"><div class="line">अत्र वै पुत्रशोकेन वसिष्ठो भगवानृषिः।</div><div class="line" style="position:relative;">बद्ध्वाऽत्मानं निपतितो विपाशः पुनरुत्थितः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5418"><div class="verse-block verse-justify"><div class="line">काश्मीरमण्डलं चैतत्सर्वपुण्यमरिंदम।</div><div class="line" style="position:relative;">महर्षिभिश्चाध्युषितं पश्येदं भ्रातृभिः सह॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5419"><div class="verse-block verse-justify"><div class="line">यत्रौत्तराणां सर्वेषामृषीणां नाहुषस्य च।</div><div class="line" style="position:relative;">अग्नेश्चैवात्र संवादः काश्यपश्य च भारत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5420"><div class="verse-block verse-justify"><div class="line">एतद्द्वारं महाराज मानसस्य प्रकाशते।</div><div class="line" style="position:relative;">वर्षमस्य गिरेर्मध्ये रामेण श्रीमता कृतम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5421"><div class="verse-block verse-justify"><div class="line">एष वातिकषण्डो वै प्रख्यातः सत्यविक्रमः।</div><div class="line" style="position:relative;">नात्यवर्तत यद्द्वारं विदेहादुत्तरं च यः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5422"><div class="verse-block verse-justify"><div class="line">इदमाश्चर्यमपरं देशेऽस्मिन्पुरुषर्षभ।</div><div class="line">क्षीणे युगे तु कौन्तेय शर्वस्य सह पार्षदैः।</div><div class="line" style="position:relative;">सहोमया च भवति दर्शनं कामरूपिणः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5423"><div class="verse-block verse-justify"><div class="line">अस्मिन्सरसि सत्रैर्वै चैत्रे मासि पिनाकिनम्।</div><div class="line" style="position:relative;">यजन्ते याजकाः सम्यक् परिवारं शुभार्थिनः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5424"><div class="verse-block verse-justify"><div class="line">अत्रोपस्पृश्य सरसि श्रद्दधानो जितेन्द्रियः।</div><div class="line" style="position:relative;">क्षीणपापः शुभाँल्लोकान्प्राप्नुते नात्र संशयः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5425"><div class="verse-block verse-justify"><div class="line">एष उज्जानको नाम पावकिर्यत्र शान्तवान्।</div><div class="line" style="position:relative;">अरुन्धतीसहायश्च वसिष्ठो भगवानृषिः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5426"><div class="verse-block verse-justify"><div class="line">ह्रदश्च कुशवानेष यत्र पद्मं कुशेशयम्।</div><div class="line" style="position:relative;">आश्रमश्चैव रुक्मिण्या यत्राशाम्यदकोपना॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5427"><div class="verse-block verse-justify"><div class="line">समाधीनां समासस्तु पाण्डवेय श्रुतस्त्वया।</div><div class="line" style="position:relative;">तं द्रक्ष्यसि महाराज भृगुतुन्दं महागिरिम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5428"><div class="verse-block verse-justify"><div class="line">वितस्तां पश्य राजेन्द्र सर्वपापप्रमोचनीम्।</div><div class="line" style="position:relative;">महर्षिभिश्चाध्युषितां शीततोयां सुनिर्मलाम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5429"><div class="verse-block verse-justify"><div class="line">जलां चोपजलां चैव यमुनामभितो नदीम्।</div><div class="line" style="position:relative;">उशीनरो वै यत्रेष्ट्वा वासवादत्यरिच्यत॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b5430"><div class="verse-block verse-justify"><div class="line">तां देवसमितिं तस्य वासवश्च विशांपते।</div><div class="line" style="position:relative;">अभ्यागच्छन्नृपवरं ज्ञातुमग्निश्च भारत॥२३॥</div></div></div>
