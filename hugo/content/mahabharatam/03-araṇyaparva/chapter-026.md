<div class="verse-block-wrapper" id="03-araṇyaparva-b1023"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1024"><div class="verse-block verse-justify"><div class="line">वसत्सु वै द्वैतवने पाण्डवेषु महात्मसु।</div><div class="line" style="position:relative;">अनुकीर्णं महारण्यं ब्राह्मणैः समपद्यत॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1025"><div class="verse-block verse-justify"><div class="line">ईर्यमाणेन सततं ब्रह्मघोषेण सर्वशः।</div><div class="line" style="position:relative;">ब्रह्मलोकसमं पुण्यमासीद्द्वैतवनं सरः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1026"><div class="verse-block verse-justify"><div class="line">यजुषामृचां साम्नां च गद्यानां चैव सर्वशः।</div><div class="line" style="position:relative;">आसीदुच्चार्यमाणानां निःस्वनो हृदयंगमः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1027"><div class="verse-block verse-justify"><div class="line">ज्याघोषश्चैव पार्थानां ब्रह्मघोषश्च धीमताम्।</div><div class="line" style="position:relative;">संसृष्टं ब्रह्मणा क्षत्रं भूय एव व्यरोचत॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1028"><div class="verse-block verse-justify"><div class="line">अथाब्रवीद्बको दाल्भ्यो धर्मराजं युधिष्ठिरम्।</div><div class="line" style="position:relative;">सन्ध्यां कौन्तेयमासीनमृषिभिः परिवारितम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1029"><div class="verse-block verse-justify"><div class="line">पश्य द्वैतवने पार्थ ब्राह्मणानां तपस्विनाम्।</div><div class="line" style="position:relative;">होमवेलां कुरुश्रेष्ठ संप्रज्वलितपावकाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1030"><div class="verse-block verse-justify"><div class="line">चरन्ति धर्मं पुण्येऽस्मिंस्त्वया गुप्ता धृतव्रताः।</div><div class="line" style="position:relative;">भृगवोऽङ्गिरसश्चैव वासिष्ठाः काश्यपैः सह॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1031"><div class="verse-block verse-justify"><div class="line">आगस्त्याश्च महाभागा आत्रेयाश्चोत्तमव्रताः।</div><div class="line" style="position:relative;">सर्वस्य जगतः श्रेष्ठा ब्राह्मणाः संगतास्त्वया॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1032"><div class="verse-block verse-justify"><div class="line">इदं तु वचनं पार्थ शृण्वेकाग्रमना मम।</div><div class="line" style="position:relative;">भ्रातृभिः सह कौन्तेय यत्त्वां वक्ष्यामि कौरव॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1033"><div class="verse-block verse-justify"><div class="line">ब्राह्म क्षत्रेण संसृष्टं क्षत्रं च ब्रह्मणा सह।</div><div class="line" style="position:relative;">उदीर्णे दहतः शत्रून्वनानीवाग्निमारुतौ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1034"><div class="verse-block verse-justify"><div class="line">नाब्राह्मणस्तात चिरं बुभूषे-दिच्छन्नमं लोकममुं च जेतुम्।</div><div class="line" style="position:relative;">विनीतधर्मार्थमपेतमोहंलब्ध्वा द्विजं हन्ति नृपः सपत्नान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1035"><div class="verse-block verse-justify"><div class="line">चरन्नैश्रेयसं धर्मं प्रजापालनकारितम्।</div><div class="line" style="position:relative;">नाध्यगच्छद्बलिर्लोके तीर्थमन्यत्र वै द्विजात्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1036"><div class="verse-block verse-justify"><div class="line">अनूनमासीदसुरस्य कामै-वैरोचनेः श्रीरपि चाक्षयाऽऽसीत्।</div><div class="line" style="position:relative;">लब्ध्वा महीं ब्राह्मणसंप्रयोगा-त्तेष्वाचरन्दुष्टमयथो व्यनश्यत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1037"><div class="verse-block verse-justify"><div class="line">नाब्राह्मणं भूमिरियं सभूति-र्वर्णं द्वितीयं भजते चिराय।</div><div class="line" style="position:relative;">समुद्रनेमिर्नमते तु तस्मैयं ब्राह्मणः शास्ति नयैर्विनीतम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1038"><div class="verse-block verse-justify"><div class="line">कुञ्चरस्येव संग्रामे परिगृह्याङ्कुशग्रहम्।</div><div class="line" style="position:relative;">ब्राह्माणैर्विप्रहीनस्य क्षत्रस्य क्षीयते बलम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1039"><div class="verse-block verse-justify"><div class="line">ब्राह्मण्यनुपमा दृष्टिः क्षात्रमप्रतिमं बलम्।</div><div class="line" style="position:relative;">तौ यदा चरतः सार्धं तदा लोकः प्रसीदती॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1040"><div class="verse-block verse-justify"><div class="line">यथा हि सुमहानग्निः कक्षं दहति सानिलः।</div><div class="line" style="position:relative;">तथा दहति राजन्यो ब्राह्मणेन समं रिपुम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1041"><div class="verse-block verse-justify"><div class="line">ब्राह्मणेष्वेव मेधावी बुद्धिपर्येषणं चरेत्।</div><div class="line" style="position:relative;">अलब्धस्य च लाभाय लब्धस्य परिवृद्धये॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1042"><div class="verse-block verse-justify"><div class="line">अलब्धलाभाय च लब्धवृद्धये।</div><div class="line">यथार्हतीर्थप्रतिपादनाय।</div><div class="line" style="position:relative;">यशस्विनं वेदविदं विपश्चितंबहुश्रुं ब्राह्मणमेव वासयेत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1043"><div class="verse-block verse-justify"><div class="line">ब्राह्मणेषूत्तमा भक्तिस्तव नित्यं युधिष्ठिर।</div><div class="line">तेन ते सर्वलोकेषु दीप्यते प्रथितं यशः ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1044"><div class="verse-block verse-justify"><div class="line">ततस्ते ब्राह्मणाः सर्वे बकं दाल्भ्यमंपूजयन्।</div><div class="line" style="position:relative;">युधिष्ठिरं स्तूयमाने भूयः सुमनसोऽभवन्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1045"><div class="verse-block verse-justify"><div class="line">द्वैपायनो नारदश्च जामदग्न्यः पृथुश्रवाः।</div><div class="line" style="position:relative;">इन्द्रद्युम्नो भालुकिश्च कृतचेताः सहस्रपात्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1046"><div class="verse-block verse-justify"><div class="line">कर्णश्रवाश्च मुञ्जश्च लवणाश्वश्च काश्यपः।</div><div class="line" style="position:relative;">हारीतः स्थूणकर्णश्च अग्निवेश्योऽथ शौनकः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b1047"><div class="verse-block verse-justify"><div class="line">कृतवाक्च सुवाश्चैव बृहदश्चो विभावसुः।</div><div class="line" style="position:relative;">ऊर्ध्वरेता वृषामित्रः सुहोत्रो होत्रवाहनः॥२५॥</div></div></div>
