<div class="verse-block-wrapper" id="03-araṇyaparva-b4039"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4040"><div class="verse-block verse-justify"><div class="line">भ्रातॄणां मतमाज्ञाय नारदस्य च धीमतः।</div><div class="line" style="position:relative;">पितामहसमं धौम्यं प्राह राजा युधिष्ठिरः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4041"><div class="verse-block verse-justify"><div class="line">मया स पुरुषव्याघ्रो जिष्णुः सत्यपराक्रमः।</div><div class="line" style="position:relative;">अस्त्रहेतोर्महाबाहुरमितात्मा विवासितः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4042"><div class="verse-block verse-justify"><div class="line">स हि वीरोऽनुरक्तश् समर्थश्च तपोधनः।</div><div class="line" style="position:relative;">कृती च भृशमप्यस्त्रे वासुदेव इव प्रभुः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4043"><div class="verse-block verse-justify"><div class="line">अहं ह्येतावुभौ ब्रह्मन्कृष्णावरिविघातिनौ।</div><div class="line" style="position:relative;">अभिजानामि विक्रान्तौ तथा व्यासः परन्तपौ॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4044"><div class="verse-block verse-justify"><div class="line">त्रियुगौ पुण्डरीकाक्षौ वासुदेवधनंजयौ।</div><div class="line" style="position:relative;">नारदोऽपितथा वेद योष्यशंसत्सदा मम॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4045"><div class="verse-block verse-justify"><div class="line">तथाऽहमपि जानामि नरनारायणावृषी।</div><div class="line" style="position:relative;">शक्तोऽयमित्यतो मत्वा मया स प्रेषितोऽर्जुनः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4046"><div class="verse-block verse-justify"><div class="line">इन्द्रादनवरः शक्रं सुरसूनुः सुराधिपम्।</div><div class="line" style="position:relative;">द्रष्टुमस्त्राणि चादातुमिन्द्रादिति विवासितः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4047"><div class="verse-block verse-justify"><div class="line">भीष्मद्रोणावतिरथौ कृपो द्रौणिश्च दुर्जयः।</div><div class="line" style="position:relative;">धृतराष्ट्रस्य पुत्रेण सुधृताः सुमहाबलाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4048"><div class="verse-block verse-justify"><div class="line">सर्वे वेदविदः शूराः सर्वेऽस्त्रकुशलास्तथा।</div><div class="line">`सर्वे महारथा मुख्याः सर्वे जितपरिश्रमाः'।</div><div class="line" style="position:relative;">योद्धुकामाश्च पार्थेन सततं ये महाबलाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4049"><div class="verse-block verse-justify"><div class="line">स च दिव्यास्त्रवित्कर्णः सूतपुत्रो महारथः।</div><div class="line">योऽस्त्रवेगानिलबलः शरार्चिस्तलनिःस्वनः।</div><div class="line" style="position:relative;">रजोधूमोऽस्त्रसंपातो धार्तराष्ट्रानिलोद्धतः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4050"><div class="verse-block verse-justify"><div class="line">निसृष्ट इव कालेन युगान्तज्वलनो यथा।</div><div class="line" style="position:relative;">मम सैन्यमयं कक्षं प्रधक्ष्यति न संशयः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4051"><div class="verse-block verse-justify"><div class="line">तं स कुष्णानिलोद्धूतो दिव्यास्त्रज्वलनो महान्।</div><div class="line" style="position:relative;">श्वेतवाजिबलाकाभृद्गण्डीवेन्द्रायुधोल्बणः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4052"><div class="verse-block verse-justify"><div class="line">संरब्धः शरधाराभिः सुधीप्तं कर्णपावकम्।</div><div class="line" style="position:relative;">उदीर्णोऽर्जुनमेघोऽयं शमयिष्यति संयुगे॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4053"><div class="verse-block verse-justify"><div class="line">स साक्षादेव सर्वाणि शक्रात्परपुरंजयः।</div><div class="line" style="position:relative;">दिव्यान्यस्त्राणि वीभत्सुर्यथावत्प्रतिपत्स्यते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4054"><div class="verse-block verse-justify"><div class="line">अलं स तेषां सर्वेषामिति मे धीयते मतिः।</div><div class="line" style="position:relative;">नास्ति त्वतिक्रिया तस् रणेऽरीणां प्रतिक्रिया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4055"><div class="verse-block verse-justify"><div class="line">ते वयं पाण्डवं सर्वे गृहीतास्त्रमरिंदमम्।</div><div class="line" style="position:relative;">द्रष्टारो न हि बीभत्सुर्भारमुद्यम्य सीदति॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4056"><div class="verse-block verse-justify"><div class="line">वयं तु तमृते वीरं वनेऽस्मिन्द्विपदांवर।</div><div class="line" style="position:relative;">अवधानं न गच्छामः काम्यके सह कृष्णया॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4057"><div class="verse-block verse-justify"><div class="line">भवानन्यद्वनं साधु बह्वन्नं फलवच्छुचि।</div><div class="line" style="position:relative;">आख्यातु रमणीयं च सेवितं पुण्यक्रमभिः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4058"><div class="verse-block verse-justify"><div class="line">यत्रकंचिद्वयं कालं वसन्तः सत्यविक्रमम्।</div><div class="line" style="position:relative;">प्रतीक्षामोऽर्जुनं वरं वृष्टिकामा इवाम्बुदम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4059"><div class="verse-block verse-justify"><div class="line">विविधानाश्रमान्कांश्चिद्द्विजातिभ्यः प्रतिश्रुतान्।</div><div class="line" style="position:relative;">सरांसि सरितश्चैव रमणीयांश्च पर्वतान्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4060"><div class="verse-block verse-justify"><div class="line">आचक्ष्व न हि मे ब्रह्मन्रोचते तमृतेऽर्जुनम्।</div><div class="line" style="position:relative;">वनेऽस्मिन्काम्यके वासो गच्छामोऽन्यां दिशंप्रति॥२२॥</div></div></div></div>
