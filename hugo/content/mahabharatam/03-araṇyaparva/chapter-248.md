<div class="verse-block-wrapper" id="03-araṇyaparva-b10063"><div class="verse-block verse-justify"><div class="line">कत्थनस्यावलिप्तस्य गर्वितस् च नित्यशः।</div><div class="line" style="position:relative;">सदा च पौरुषौदार्यैः पाण्डवानवमन्यतः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10064"><div class="verse-block verse-justify"><div class="line">दुर्योधनस् पापस्य नित्याहंकारवादिनः।</div><div class="line" style="position:relative;">प्रवेशो हास्तिनपुरे दुष्करः प्रतिभाति मे॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10065"><div class="verse-block verse-justify"><div class="line">तस्य लज्जानवितस्यैव शोकव्याकुलचेतसः।</div><div class="line">प्रवेशं विस्तरेण रत्वं वैशंपायन कीर्तय ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10066"><div class="verse-block verse-justify"><div class="line">धर्मराजनिसृष्स्तु धार्तराष्ट्रः सुयोधनः।</div><div class="line" style="position:relative;">लज्जयाऽधोमुखः सीदन्नुपासर्पत्सुदुःखितः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10067"><div class="verse-block verse-justify"><div class="line">स्वपुरं प्रययौ राजा चतुरङ्गबलानुगः।</div><div class="line" style="position:relative;">शोकोदहतया बुद्ध्या चिन्तयानः पराभवम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10068"><div class="verse-block verse-justify"><div class="line">विमुच्य पथि यानानि देशे सुयवसोदके।</div><div class="line">सन्निविष्टः शुभे रम्ये भूमिभागे यथेप्सितम्।</div><div class="line" style="position:relative;">हस्त्यश्वरथपादातं यथास्थानं न्यवेशयत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10069"><div class="verse-block verse-justify"><div class="line">अथोपविष्टं राजानं पर्यङ्के ज्वलनप्रभे।</div><div class="line">उपप्लुतं यथा सोमं राहुणा रात्रिसंक्षये।</div><div class="line" style="position:relative;">उपागम्याब्रवीत्कर्णो दुर्योधनमिदं तदा॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10070"><div class="verse-block verse-justify"><div class="line">दिष्ट्या जीवसि गान्धारे दिष्ठ्या नः संगमः पुनः।</div><div class="line" style="position:relative;">दिष्ट्या त्वया जिताश्चैव गन्धर्वाः कामरूपिणः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10071"><div class="verse-block verse-justify"><div class="line">दिष्ट्या समग्रान्पश्यामि भ्रातॄस्ते कुरुनन्दन।</div><div class="line" style="position:relative;">विजिगीषून्रणे युक्तान्निर्जितारीन्महारथान्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10072"><div class="verse-block verse-justify"><div class="line">अहं त्वभिद्रुतः सर्वैर्गन्धर्वैः पश्यतस्व।</div><div class="line">नाशक्नुवं स्थापयितुं दीर्यमाणां च वाहिनीम्।</div><div class="line" style="position:relative;">शरक्षताङ्गश्च भृशं व्यपयातोऽभिपीडितः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10073"><div class="verse-block verse-justify"><div class="line">इदं त्वत्यद्भुतं मन्ये यद्युष्मानिह भारत।</div><div class="line">अरिष्टानक्षसांश्चापि सदारबलवाहनान्।</div><div class="line" style="position:relative;">विमुक्तान्संप्रपश्यामि युद्धात्तस्मादमानुषात्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b10074"><div class="verse-block verse-justify"><div class="line">नैतस्य कर्ता लोकेऽस्मिन्पुमान्विद्यति भारत।</div><div class="line" style="position:relative;">यत्कृतं ते महाराज सह भ्रातृभिराहवे॥१४॥</div></div></div>
