<div class="verse-block-wrapper" id="03-araṇyaparva-b2229"><div class="verse-block verse-justify"><div class="line">आसीद्राजा नलो नाम वीरसेनसुतो बली।</div><div class="line" style="position:relative;">उपपन्नो गुणैरिष्टै रूपवानश्वकोविदः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2230"><div class="verse-block verse-justify"><div class="line">`यज्वा दानपतिर्दक्षः सदा शीलपुरस्कृतः'।</div><div class="line">अतिष्ठन्मनुजेन्द्राणां मूर्ध्नि देवपतिर्यथा।</div><div class="line" style="position:relative;">उपर्युपरि सर्वेषामादित्य इव तेजसा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2231"><div class="verse-block verse-justify"><div class="line">ब्रह्मण्यो वेदविच्छ्ररो निषधेषु महीपतिः।</div><div class="line" style="position:relative;">अक्षप्रियः सत्यवादी महानक्षौहिणीपतिः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2232"><div class="verse-block verse-justify"><div class="line">ईप्सितो वरनारीणामुदारः संयतेन्द्रियः।</div><div class="line" style="position:relative;">रक्षिताधन्विनां श्रेष्ठः साक्षादिव मनुः स्वयम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2233"><div class="verse-block verse-justify"><div class="line">तथैवासीद्विदर्भेषु भीमो भीमपराक्रमः।</div><div class="line" style="position:relative;">शूरः सर्वगुणैर्युक्तः प्रजाकामः स चाप्रजाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2234"><div class="verse-block verse-justify"><div class="line">स प्रजार्थे परं यत्नमकरोत्सुसमाहितः।</div><div class="line" style="position:relative;">तमभ्यगच्छद्ब्रह्मर्षिर्दमनो नाम भारत॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2235"><div class="verse-block verse-justify"><div class="line">तं स भीमः प्रजाकामस्तोषयामास धर्मवित्।</div><div class="line" style="position:relative;">महिष्या सह राजेनद््र सत्कारेण सुवर्चसम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2236"><div class="verse-block verse-justify"><div class="line">तस्मै प्रसन्नो दमनः सभार्याय वरं ददौ।</div><div class="line" style="position:relative;">कन्यारत्नं कुमारांश्च त्रीनुदारान्महायशाः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2237"><div class="verse-block verse-justify"><div class="line">दमयन्तीं दमं दान्तं दमनं च सुवर्चसम्।</div><div class="line" style="position:relative;">उपपन्नान्गुणैः सर्वैर्भीमान्भीमपराक्रमान्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2238"><div class="verse-block verse-justify"><div class="line">दमयन्ती तु रूपेण तेजसा वपुषा श्रिया।</div><div class="line" style="position:relative;">सौभाग्येन च लोकेषु यशः प्राप सुमध्यमा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2239"><div class="verse-block verse-justify"><div class="line">अथ तां वसि प्राप्ते दासीनां समलंकृतम्।</div><div class="line" style="position:relative;">शतं सखीनां च तदा पर्युपास्ते शचीमिव॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2240"><div class="verse-block verse-justify"><div class="line">तत्र स्म राजते भैमी सर्वाभरणभूषिता।</div><div class="line" style="position:relative;">सखीमध्येऽनवद्याङ्गी विद्युत्सौदामनी यथा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2241"><div class="verse-block verse-justify"><div class="line">अतीव रूपसंपन्ना श्रीरिवायतलोचना।</div><div class="line" style="position:relative;">न देवेषु न यक्षेषु तादृग्रूपवती क्वचित्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2242"><div class="verse-block verse-justify"><div class="line">मानुषेष्वपि चान्येषु दृष्टपूर्वाऽथवा श्रुता।</div><div class="line" style="position:relative;">चित्तप्रमाथिनी बाला देवानामपि सुन्दरी॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2243"><div class="verse-block verse-justify"><div class="line">नलश्च नरशार्दूलो रूपेणाप्रतिमो भुवि।</div><div class="line" style="position:relative;">कंदर्प इव रूपेण मूर्तिमानभवत्स्वयम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2244"><div class="verse-block verse-justify"><div class="line">तस्याः समीपे तु नलं प्रशशंसुः कुतूहलात्।</div><div class="line" style="position:relative;">नैषधस्य समीपे तु दमयन्तीं पुनः पुनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2245"><div class="verse-block verse-justify"><div class="line">तयोरदृष्टः कामोऽभूच्छृण्वतोः सततं गुणान्।</div><div class="line" style="position:relative;">अन्योन्यं प्रतिकौन्तेय स व्वर्धत हृच्छयः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2246"><div class="verse-block verse-justify"><div class="line">अशक्नवन्नलः कामं तदा धारयितुं हृदा।</div><div class="line" style="position:relative;">अन्तःपुरसमीपस्थे वन आस्ते रहोगतः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2247"><div class="verse-block verse-justify"><div class="line">स ददर्शै ततो हंसाञ्जातरूपपरिच्दान्।</div><div class="line" style="position:relative;">वने विचरतां तेषामेकं जग्राह पक्षिणम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2248"><div class="verse-block verse-justify"><div class="line">ततो।</div><div class="line">डन्तरिक्षगो वाचं व्याजहार नलं तदा।</div><div class="line" style="position:relative;">हन्तव्योस्मि न तेराजन्करिष्यामि तवप्रियम्॥२१॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b2256"><div class="verse-block verse-justify"><div class="line">दमयन्ति नलो नाम निषधेषु महीपतिः।</div><div class="line" style="position:relative;">अश्विनोः सदृशो रूपे न समास्तस्य मानुषाः॥२९॥</div></div></div>
