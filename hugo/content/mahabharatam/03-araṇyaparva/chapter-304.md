<div class="verse-block-wrapper" id="03-araṇyaparva-b12172"><div class="verse-block verse-justify"><div class="line">जनमेजय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12173"><div class="verse-block verse-justify"><div class="line">किं तद्गुह्यं न चाख्यातं कर्णायेहोष्णरश्मिना।</div><div class="line" style="position:relative;">कीदृशेकुण्डले ते च कवचं रचैव कीदृशम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12174"><div class="verse-block verse-justify"><div class="line">कुतश्च कवचं तस्यं कुण्डलेचैव सत्तम।</div><div class="line">एतदिच्छाम्यहं श्रोतुं तन्मे ब्रूहि तपोधन ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12175"><div class="verse-block verse-justify"><div class="line">अहं राजन्ब्रवीम्येतत्तस्य गुह्यं विभावसोः।</div><div class="line" style="position:relative;">यादृशे कुण्डले ते च कवचं वैव यादृशम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12176"><div class="verse-block verse-justify"><div class="line">कुन्तिभोजं पुरा राजन्ब्राह्मणः पर्युपस्थितः।</div><div class="line" style="position:relative;">तिग्मतेजा महान्प्रांशुः श्मश्रुदण्डजटाधरः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12177"><div class="verse-block verse-justify"><div class="line">दर्सनीयोऽनवद्याङ्गस्तेजसा प्रज्वलन्निव।</div><div class="line" style="position:relative;">मधुपिङ्गो मधुरवाक्तपःस्वाध्यायभूषणः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12178"><div class="verse-block verse-justify"><div class="line">स राजानं कुन्तिभोजमब्रवीत्सुमहातपाः।</div><div class="line" style="position:relative;">भिक्षामिच्छामि वै भोक्तुं तव गेहे विमत्सर॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12179"><div class="verse-block verse-justify"><div class="line">न मे व्यलीकं कर्तव्यं त्वया वा तव चानुगैः।</div><div class="line" style="position:relative;">एवं वत्स्यामि ते गेहे यदि ते रोचतेऽनघ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12180"><div class="verse-block verse-justify"><div class="line">यथाकामं च गच्छेयमागच्छेयं तथैव च।</div><div class="line" style="position:relative;">शय्यासने च मे राजन्नापराध्येत कश्चन॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12181"><div class="verse-block verse-justify"><div class="line">तमब्रवीत्कुन्तिभोजः प्रीतियुक्तमिदं वचः।</div><div class="line" style="position:relative;">एवमस्तु परंचेति पुनश्चैवमथाब्रवीत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12182"><div class="verse-block verse-justify"><div class="line">मम कन्या महाप्राज्ञ पृथा नाम यशस्विनी।</div><div class="line" style="position:relative;">शीलवृत्तान्विता साध्वी नियताऽनवमानिनी॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12183"><div class="verse-block verse-justify"><div class="line">उपस्थास्यति सा त्वां वै पूजयाऽनवमत्य च।</div><div class="line" style="position:relative;">तस्याश्च शीलवृत्तेन तुष्टिं समुपयास्यसि॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12184"><div class="verse-block verse-justify"><div class="line">एवमुक्त्वा तु तं विप्रमभिपूज्य यथाविधि।</div><div class="line" style="position:relative;">उवाच कन्यामभ्येत्य पृथां पृथुललोचनाम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12185"><div class="verse-block verse-justify"><div class="line">अयं वत्से महाभागो ब्राह्मणो वस्तुमिच्छति।</div><div class="line" style="position:relative;">मम गेहे मया चास्य तथेत्येवं प्रतिश्रुतम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12186"><div class="verse-block verse-justify"><div class="line">त्वयि वत्से परायत्तं ब्राह्मणस्याभिराधनम्।</div><div class="line" style="position:relative;">तन्मे वाक्यममिथ्या त्वं कर्तुमर्हसि कर्हिचित्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12187"><div class="verse-block verse-justify"><div class="line">अयं तपस्वी भगवान्स्वाध्यायनियतो द्विजः।</div><div class="line" style="position:relative;">यद्यद्ब्रूयान्महातेजास्तत्तद्देयमात्सरात्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12188"><div class="verse-block verse-justify"><div class="line">ब्राह्मणा हि परं तेजो ब्राह्मणा हि परं तपः।</div><div class="line" style="position:relative;">ब्राह्मणानां नमस्कारैः सूर्यो दिवि विराजते॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12189"><div class="verse-block verse-justify"><div class="line">अमानयन्हि दाण्डक्यो वातापिश्च महासुरः।</div><div class="line">निहतो ब्रह्मदण्डेन तालजङ्घस्तथैव च।</div><div class="line" style="position:relative;">`वैन्ध्याद्रिश्च समुद्रश्च नद्दुषश्च विहिंसितः'॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12190"><div class="verse-block verse-justify"><div class="line">सोयं वत्से महाभार आहितस्त्वयि सांप्रतम्।</div><div class="line" style="position:relative;">त्वं सदा नियता कुर्या ब्राह्मणस्याभिराधनम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12191"><div class="verse-block verse-justify"><div class="line">जानामि प्रणिधानं ते बाल्यात्प्रभृति नन्दिनि।</div><div class="line" style="position:relative;">ब्राह्मणेष्विह सर्वेषु गुरुबन्धुषु चैव ह॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12192"><div class="verse-block verse-justify"><div class="line">तथा प्रेष्येषु सर्वेषु मित्रसंबन्धिमातृषु।</div><div class="line" style="position:relative;">मयि चैव यथावत्त्वंसर्वमावृत्य वर्तसे॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12193"><div class="verse-block verse-justify"><div class="line">न ह्यतुष्टो जनोऽस्तीह पुरे चान्तःपुरे च ते।</div><div class="line" style="position:relative;">सम्यग्वृत्त्याऽनवद्याङ्गि तव भृत्यजनेष्वपि॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12194"><div class="verse-block verse-justify"><div class="line">संदेष्टव्यां तु मन्ये त्वां द्विजातिं कोपनं प्रति।</div><div class="line" style="position:relative;">पृथे बालेति कृत्वा वै सुता चासिममेति च॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12195"><div class="verse-block verse-justify"><div class="line">वृष्णीनां च कुले जाता शूरस् दयिता सुता।</div><div class="line" style="position:relative;">दत्ता प्रीतिमता मह्यं पित्रा बाला पुरास्वयम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12196"><div class="verse-block verse-justify"><div class="line">वसुदेवस् भगिनी सुतानां प्रवरा मम।</div><div class="line" style="position:relative;">अग्र्यमग्रे प्रतिज्ञाय तेनासि दुहिता मम॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12197"><div class="verse-block verse-justify"><div class="line">तादृशे हि कुले जाता कुले मम विवर्धिता।</div><div class="line" style="position:relative;">सुखात्सुखमनुप्राप्ता ह्रदाद्ध्रदमिवापगा॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12198"><div class="verse-block verse-justify"><div class="line">दौष्कुलेया विशेषेण कथंचित्प्रग्रहं गताः।</div><div class="line" style="position:relative;">बालभावाद्विकुर्वन्ति प्रायशः प्रमदाः शुभे॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12199"><div class="verse-block verse-justify"><div class="line">पृथे राजकुले जन्म रूपं चापि तवाद्भुतम्।</div><div class="line" style="position:relative;">तेन तेनासि संपन्ना समुपेता च भामिनि॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12200"><div class="verse-block verse-justify"><div class="line">सा त्वं दर्पं परित्यज्य दम्भं मानं च भामिनि।</div><div class="line" style="position:relative;">आराध्यवरदं विप्रं श्रेयसा योक्ष्यसे पृथे॥२९॥</div></div></div>
