<div class="verse-block-wrapper" id="03-araṇyaparva-b12341"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12342"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नेव काले तु धृतराष्ट्रस्य वै सखा।</div><div class="line" style="position:relative;">सूतोऽधिरथ इत्येव सदारो जाह्नवीं ययौ॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12343"><div class="verse-block verse-justify"><div class="line">तस्य भार्याऽभवद्राजन्रूपेणासदृशी भुवि।</div><div class="line">राधा नाम महाभागा न सा पुत्रमविन्दत।</div><div class="line" style="position:relative;">अपत्यार्थे परं यत्नमकरोच्च विशेषतः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12344"><div class="verse-block verse-justify"><div class="line">सा ददर्शाथ मञ्जूषामुह्यमानां यदृच्छया।</div><div class="line" style="position:relative;">दत्तरक्षाप्रतिसरामन्वालम्भनशोभनाम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12345"><div class="verse-block verse-justify"><div class="line">ऊर्मीतरङ्गैर्जाह्नव्याः समानीतामुपह्वरम्।</div><div class="line" style="position:relative;">`विवर्तमानां बहुशः पुनःपुनरितस्ततः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12346"><div class="verse-block verse-justify"><div class="line">ततः सा वायुना राजन्स्रोतसा च बलीयसा।</div><div class="line" style="position:relative;">उपानीतो यतः सूतः सभार्यो जलमाश्रितः'॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12347"><div class="verse-block verse-justify"><div class="line">सा तु कौतूहलात्प्राप्तां ग्राहयामास भामिनी।</div><div class="line" style="position:relative;">ततो निवेदयामास सूतस्याधिरथस्य वै॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12348"><div class="verse-block verse-justify"><div class="line">स तामुद्धृत्य मञ्जूषामुत्सार्य जलमन्तिकात्।</div><div class="line" style="position:relative;">यन्त्रैरुद्घाटयामास सोऽपश्यत्तत्रबालकम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12349"><div class="verse-block verse-justify"><div class="line">मृष्टकुण्डलयुक्तेन वदनेन विराजितम्।</div><div class="line" style="position:relative;">`परिम्लानमुखं बालं रुदन्तं क्षुधितं भृशम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12350"><div class="verse-block verse-justify"><div class="line">स तु तं परया लक्ष्म्या दृष्ट्वा युक्तं वरात्मजम्'।</div><div class="line">स सूतो भार्यया सार्धं विस्मयोत्फुल्ललोचनः।</div><div class="line" style="position:relative;">अङ्कमारोप्य तं बालं भार्यां वचनमब्रवीत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12351"><div class="verse-block verse-justify"><div class="line">इदमत्यद्भुतं भीरु यतो जातोस्मि भामिनि।</div><div class="line" style="position:relative;">दृष्टवान्देवगर्भोऽयं मन्येऽस्माकमुपागतः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12352"><div class="verse-block verse-justify"><div class="line">अनपत्यस्य पुत्रोऽयं देवैर्दत्तो ध्रुवं मम।</div><div class="line" style="position:relative;">इत्युक्त्वा तं ददौ पुत्रं राधायै स महीपते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12353"><div class="verse-block verse-justify"><div class="line">रप्रतिजग्राह तं राधा विधिवद्दिव्यरूपिणम्।</div><div class="line" style="position:relative;">पुत्रं कमलगर्भाभं देवगर्भं श्रिया वृतम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12354"><div class="verse-block verse-justify"><div class="line">`स्तन्यं समस्रवच्चास्य दैवादित्थ निश्चयः'।</div><div class="line">पुपोष चैनं विधिवद्ववृधे स च वीर्यवान्।</div><div class="line" style="position:relative;">ततः प्रभृति चाप्यन्ये प्राभवन्नौरसाः सुताः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12355"><div class="verse-block verse-justify"><div class="line">`नामकर्म च चक्रुस्ते कुण्डले तस् दृश्यते।</div><div class="line" style="position:relative;">कर्ण इत्येव तं बालं दृष्ट्वा कर्णं सकुण्डलम्'॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12356"><div class="verse-block verse-justify"><div class="line">वसुवर्मधरं दृष्ट्वा तं बालं हेमकुण्डलम्।</div><div class="line" style="position:relative;">नामास्य वसुषेणेति ततश्चक्रुर्द्विजातयः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12357"><div class="verse-block verse-justify"><div class="line">एवं स सूतपुत्रत्वं जगामामितविक्रमः।</div><div class="line" style="position:relative;">वसुषेण इतिख्यातो वृष इत्येव च प्रभुः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12358"><div class="verse-block verse-justify"><div class="line">सूतस्य ववृधेऽङ्गेषु ज्येष्ठः पुत्रः स वीर्यवान्।</div><div class="line" style="position:relative;">चारेण विदितश्चासीत्पृथया दिव्यवर्मभृत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12359"><div class="verse-block verse-justify"><div class="line">सूतस्त्वधिरथः पुत्रं विवृद्धं समयेन तम्।</div><div class="line" style="position:relative;">दृष्ट्वा प्रस्थापयामास पुरं वारणसाह्वयम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12360"><div class="verse-block verse-justify"><div class="line">तत्रोपसदनं चक्रे द्रोणस्येष्वस्त्रकर्मणि।</div><div class="line" style="position:relative;">सख्यं दुर्योधनेनैवमगमत्स च वीर्यवान्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12361"><div class="verse-block verse-justify"><div class="line">द्रोणात्कृपाच्च रामाच्च सोऽस्त्रग्रामं चतुर्विधम्।</div><div class="line" style="position:relative;">लब्ध्वा लोकेऽभवत्ख्यातः परमेष्वासतां गतः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12362"><div class="verse-block verse-justify"><div class="line">संधाय धार्तराष्ट्रेण पार्थानां विप्रिये रतः।</div><div class="line" style="position:relative;">योद्धुमाशंसते नित्यं फल्गुनेन महात्मना॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12363"><div class="verse-block verse-justify"><div class="line">सदा हि तस् स्पर्धाऽऽसीदर्जुनन विशांपते।</div><div class="line" style="position:relative;">अर्जुनस्य च कर्णेन यतो द्वन्द्वं बभूव ह॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12364"><div class="verse-block verse-justify"><div class="line">एतद्गुह्यं महाराज सूर्यस्यासीन्न संशयः।</div><div class="line" style="position:relative;">यः सूर्यसंभवः कर्णः कुर्यात्प्रतिकुले रतः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12365"><div class="verse-block verse-justify"><div class="line">तं तु कुण्डलिनं दृष्ट्वा वर्मणा च समन्वितम्।</div><div class="line" style="position:relative;">अवध्यं समरे मत्वा पर्यतप्यद्युधिष्ठिरः॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12366"><div class="verse-block verse-justify"><div class="line">यदा च कर्णो राजेन्द्र भानुमन्तं दिवाकरम्।</div><div class="line" style="position:relative;">स्तौति मध्यंदिने प्राप्ते प्राञ्जलिः सलिलोत्थितः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b12367"><div class="verse-block verse-justify"><div class="line">तत्रैनमुपतिष्ठन्ति ब्राह्मणा धनहेतुना।</div><div class="line" style="position:relative;">नादेयं तस्य तत्काले किंचिदस्ति द्विजातिषु॥२७॥</div></div></div>
