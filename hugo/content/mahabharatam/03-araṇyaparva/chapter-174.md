<div class="verse-block-wrapper" id="03-araṇyaparva-b7111"><div class="verse-block verse-justify"><div class="line">अर्जुन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7112"><div class="verse-block verse-justify"><div class="line">अदृश्यमानास्ते दैत्या योधयन्ति स्म मायया।</div><div class="line" style="position:relative;">अदृश्यनास्त्रवीर्येण तानप्यहमयोधयम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7113"><div class="verse-block verse-justify"><div class="line">गाण्डीवमुक्ता विशिखाः सम्यगस्त्रप्रचोदिताः।</div><div class="line" style="position:relative;">अच्छिन्दन्नुत्तमाङ्गानि यत्रयत्र स्म तेऽभवन्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7114"><div class="verse-block verse-justify"><div class="line">ततो निवातकवचा वध्यमाना मया युधि।</div><div class="line" style="position:relative;">संहृत्य रमायां सहसा प्राविशन्पुरमात्मनः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7115"><div class="verse-block verse-justify"><div class="line">व्यपयातेषु दैत्येषु प्रादुर्भूते च दर्शने।</div><div class="line" style="position:relative;">अपश्यं दानवांस्तत्र हताञ्शतसहस्रशः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7116"><div class="verse-block verse-justify"><div class="line">विनिष्पिष्टानि तत्रैषां शस्त्राण्याभरणानि च।</div><div class="line" style="position:relative;">शतशः स्म प्रदृश्यन्ते गात्राणि कवचानि च॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7117"><div class="verse-block verse-justify"><div class="line">हयानां नान्तरं ह्यासीत्पदाद्विचलितुं पदम्।</div><div class="line" style="position:relative;">उत्पत्य सहसा तस्थुरन्तरिक्षगमास्ततः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7118"><div class="verse-block verse-justify"><div class="line">ततो निवातकवचा व्योम संछाद्य केवलम्।</div><div class="line" style="position:relative;">अदृश्या ह्यभ्यवर्तन्त विसृजन्तः शिलोच्चयान्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7119"><div class="verse-block verse-justify"><div class="line">अन्तर्भूमिगताश्चान्ये हयानां चरणानथ।</div><div class="line" style="position:relative;">व्यगृह्णन्दानवा घोरा रथचक्रे च भारत॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7120"><div class="verse-block verse-justify"><div class="line">विनिगृह्य हयांश्चान्ये रथं च मम युध्यतः।</div><div class="line" style="position:relative;">सर्वतो मामविध्यन्त सरथं धरणीधरैः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7121"><div class="verse-block verse-justify"><div class="line">पर्वतैरुपचीयद्भिः पतद्भिश्च तथाऽपरैः।</div><div class="line" style="position:relative;">स देशो यत्रवर्तामि गुहेव समपद्यत॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7122"><div class="verse-block verse-justify"><div class="line">पर्वतैश्चाद्यमानोऽहं निगृहीतैश्च वाजिभिः।</div><div class="line" style="position:relative;">अगच्छं परमामार्तिं मातलिस्तदलक्षयत्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7123"><div class="verse-block verse-justify"><div class="line">लक्षयित्वा च मां भीतमिदं वचनमब्रवीत्।</div><div class="line" style="position:relative;">आर्जुनार्जुन मा भैस्त्वं वज्रमस्त्रमुदीरय॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7124"><div class="verse-block verse-justify"><div class="line">ततोऽहं तस्य तद्वाक्यं श्रुत्वा वज्रमुदीरयम्।</div><div class="line" style="position:relative;">देवराजस् दयितं वज्रमस्त्रं नराधिप॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7125"><div class="verse-block verse-justify"><div class="line">अचलं स्थानमासाद्य गाण्डीवमनुमन्त्र्य च।</div><div class="line" style="position:relative;">अमुञ्चं वज्रसंस्पर्शानायतान्निशिताञ्सरान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7126"><div class="verse-block verse-justify"><div class="line">ततो मायाश् ताः सर्वा निवातकवचांश्च तान्।</div><div class="line" style="position:relative;">ते वज्रचोदिता बाणा वज्रभूताः रसमाविशन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7127"><div class="verse-block verse-justify"><div class="line">ते वज्रवेगविहता दानवाः पर्वतोपमाः।</div><div class="line" style="position:relative;">इतरेतरमाश्लिष्य न्यपतन्पृथिवीतले॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7128"><div class="verse-block verse-justify"><div class="line">अन्तर्भूमौ च येऽगृह्णन्दानवा रथवाजिनः।</div><div class="line" style="position:relative;">अनुप्रविश्य तान्वाणाः प्राहिण्वन्यमसादनम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7129"><div class="verse-block verse-justify"><div class="line">हतैर्निवातकवचैर्निरस्तैः पर्वतोपमैः।</div><div class="line" style="position:relative;">समाच्छाद्यत देशः स विकीर्णैरिव पर्वतैः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7130"><div class="verse-block verse-justify"><div class="line">न हयानां क्षतिः काचिन्न रथस् न मातलेः।</div><div class="line" style="position:relative;">मम चादृश्यत तदा तदद्भुतमिवाभवत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7131"><div class="verse-block verse-justify"><div class="line">ततो मां प्रहसन्राजन्मातलिः प्रत्यभाषत।</div><div class="line" style="position:relative;">नैतदर्जुन देवेषु त्वयि वीर्यं यदीक्ष्यते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7132"><div class="verse-block verse-justify"><div class="line">हतेष्वसुरसङ्घेषु दारास्तेषां तु सर्वशः।</div><div class="line" style="position:relative;">प्राक्रोशन्नगरे तस्मिन्यथा शरदि लक्ष्मणाः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7133"><div class="verse-block verse-justify"><div class="line">ततो मातलिना सार्धमहं तत्पुरमभ्ययाम्।</div><div class="line" style="position:relative;">त्रासयन्रथघोषेण निवातकवचस्त्रियः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7134"><div class="verse-block verse-justify"><div class="line">तान्दृष्ट्वा दशसाहस्रान्मयूरसदृशान्हयान्।</div><div class="line" style="position:relative;">रथं च रविसंकाशं प्राद्रवन्गणशः स्त्रियः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7135"><div class="verse-block verse-justify"><div class="line">ताभिराभरणैः शब्दस्त्रासिताभिः समीरितः।</div><div class="line" style="position:relative;">शिलानामिव शैलेषु पतन्तीनामभूत्तदा॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7136"><div class="verse-block verse-justify"><div class="line">वित्रस्ता दैत्यनार्यस्ताः स्वानि वेश्मान्यथाविशन्।</div><div class="line" style="position:relative;">बहुरत्नविचित्राणि शातकुम्भमयानि च॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7137"><div class="verse-block verse-justify"><div class="line">तदद्भुताकारमहं दृष्ट्वा नगरमुत्तमम्।</div><div class="line" style="position:relative;">विशिष्टं देवनगरादपृच्छं मातलिं ततः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7138"><div class="verse-block verse-justify"><div class="line">इदमेवंविधं कस्माद्देवता न विशन्त्युत।</div><div class="line">पुरंदरपुराद्धीदं विशिष्टमिति लक्षये ॥मातलिरुवाच।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7139"><div class="verse-block verse-justify"><div class="line">आसीदिदं पुरा पार्थ देवराजस्य नः पुरम्।</div><div class="line" style="position:relative;">ततो निवातकवचैरितः प्रच्याविताः सुराः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7140"><div class="verse-block verse-justify"><div class="line">तपस्तप्त्वा महत्तीव्रं प्रसाद्य च पितामहम्।</div><div class="line" style="position:relative;">इदं वृतं निवासाय देवेभ्यश्चाभयं युधि॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7141"><div class="verse-block verse-justify"><div class="line">ततः शक्रेण भगवान्स्वयंभूरभिचोदितः।</div><div class="line" style="position:relative;">विधत्तां भगवानत्रेत्यात्मनो हितकाम्यया॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7142"><div class="verse-block verse-justify"><div class="line">तत उक्तो भगवता दिष्टमत्रेति भारत।</div><div class="line" style="position:relative;">भवितान्तस्त्वमप्येषां देहेनान्येन वृत्रहन्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7143"><div class="verse-block verse-justify"><div class="line">तत एषां वधार्थाय शक्रोऽस्त्राणि ददौ तव।</div><div class="line" style="position:relative;">न हि शक्याः सुरैर्हन्तुं य एते निहतास्त्वया॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b7144"><div class="verse-block verse-justify"><div class="line">कालस्य परिणामेन ततस्त्वमिह भारत।</div><div class="line" style="position:relative;">एषामन्तकरः प्राप्तस्तत्त्वया च कृतं तथा॥३४॥</div></div></div>
