<div class="verse-block-wrapper" id="03-araṇyaparva-b3374"><div class="verse-block verse-justify"><div class="line">वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3375"><div class="verse-block verse-justify"><div class="line">`धनञ्जयोत्सुकास्ते तु वने तस्मिन्महारथाः।</div><div class="line" style="position:relative;">न्यवसन्त महाभागा द्रौपद्या सह कृष्णया'॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3376"><div class="verse-block verse-justify"><div class="line">धनं जयोत्सुकानां तु भ्रातॄणां कृष्णया सह।</div><div class="line" style="position:relative;">श्रुत्वा वाक्यानि विमना धर्मराजोप्यजायत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3377"><div class="verse-block verse-justify"><div class="line">अथापश्यन्महात्मानं देवर्षिं तत्र नारदम्।</div><div class="line" style="position:relative;">दीप्यमानं श्रिया ब्राह्म्या दीप्ताग्निसमतेजसम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3378"><div class="verse-block verse-justify"><div class="line">तमागतमभिप्रेक्ष्य भ्रातृभिः सह धर्मराट्।</div><div class="line" style="position:relative;">प्रत्युत्थाय यथान्यायं पूजां चक्रे महात्मने॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3379"><div class="verse-block verse-justify"><div class="line">स तैः परिवृतः श्रीमान्भ्रातृभिः कुरुसत्तमः।</div><div class="line" style="position:relative;">विबभावतिदीप्तौजा देवैरिव शतक्रतुः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3380"><div class="verse-block verse-justify"><div class="line">यथा च वेदान्सावित्री याज्ञसेनी तथा पतीन्।</div><div class="line" style="position:relative;">न जहौ धर्मतः पार्थान्मेरुमर्कप्रभा यथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3381"><div class="verse-block verse-justify"><div class="line">`अर्ध्यं पाद्यमथानीय त्वभ्यवायदच्युतः।</div><div class="line" style="position:relative;">नारदस्तु महातेजाः स्वस्त्यस्त्वित्यभ्यभाषत॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3382"><div class="verse-block verse-justify"><div class="line">ततो युधिष्ठिरो राजा दृष्ट्वा देवर्षिसत्तमम्।</div><div class="line" style="position:relative;">यथार्हं पूजयामास विधिवत्कुरुनन्दनः'॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3383"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्य च तां पूजां नारदो भगवानृषिः।</div><div class="line" style="position:relative;">आश्वासयद्धर्मसुतं युक्तरूपमिवानघ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3384"><div class="verse-block verse-justify"><div class="line">उवाच च महात्मानं धर्मराजं युधिष्ठिरम्।</div><div class="line" style="position:relative;">ब्रूहि धर्मभृतां श्रेष्ठ केनार्थः किं ददानि ते॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3385"><div class="verse-block verse-justify"><div class="line">अथ धर्मसुतो राजा प्रणम्य भ्रातृभिः सह।</div><div class="line" style="position:relative;">उवाच प्राञ्जलिर्भूत्वा नारदं देवसंमितम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3386"><div class="verse-block verse-justify"><div class="line">न्वयि तुष्टे महाभाग सर्वलोकाभिपूजिते।</div><div class="line" style="position:relative;">कृतमित्येव मन्येऽहं प्रसादात्तव सुव्रत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3387"><div class="verse-block verse-justify"><div class="line">यदि त्वहमनुग्राह्यो भ्रातृभिः सहितोऽनघ।</div><div class="line" style="position:relative;">संदेहं मे सुनिश्रेष्ठ तत्वतश्छेत्तुमर्हसि॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3388"><div class="verse-block verse-justify"><div class="line">प्रदक्षिणां यः कुरुते पृथिवीं तीर्थतत्परः।</div><div class="line">किं फलं तस्य कार्त्स्न्येन तद्भवान्वक्तुमर्हति ॥नारद उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3389"><div class="verse-block verse-justify"><div class="line">शृणु राजन्नवहितो यथा भीष्मेण धीमता।</div><div class="line" style="position:relative;">पुलस्त्यस्य सकाशाद्वै सर्वमेतदुपश्रुतम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3390"><div class="verse-block verse-justify"><div class="line">पुरा भागीरथीतीरे भीष्मो धर्मभृतांवरः।</div><div class="line" style="position:relative;">पित्र्यं व्रतं समास्थाय न्यवसन्मुनिभिः सह॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3391"><div class="verse-block verse-justify"><div class="line">शुभे देशे तथा राजन्पुण्ये देवर्षिसेविते।</div><div class="line" style="position:relative;">गङ्गाद्वारे महाभाग देवगन्धर्वसेविते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3392"><div class="verse-block verse-justify"><div class="line">स पितॄंस्तर्पयामास देवांश्च परमद्युतिः।</div><div class="line" style="position:relative;">ऋषींश्च तर्पयामास विधिदृष्टेन कर्मणा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3393"><div class="verse-block verse-justify"><div class="line">कस्य चित्त्वथ कालस्य जपन्नेव महायशाः।</div><div class="line" style="position:relative;">ददर्शाद्भुतसंकाशं पुलस्त्यमृषिसत्तमम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3394"><div class="verse-block verse-justify"><div class="line">स तं दृष्ट्वोग्रतपसं दीप्यमानमिव श्रिया।</div><div class="line" style="position:relative;">प्रहर्षमतुलं लेभे विस्मयं परमं ययौ॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3395"><div class="verse-block verse-justify"><div class="line">उपस्थितं महाभागं पूजयामास भारत।</div><div class="line" style="position:relative;">भीष्मो धर्मभृतां श्रेष्ठो विधिदृष्टेन कर्मणा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3396"><div class="verse-block verse-justify"><div class="line">शिरसा चाघमादाय शुचिः प्रयतमानसः।</div><div class="line" style="position:relative;">नाम संकीर्तयामास तस्मिन्ब्रह्मर्षिसत्तमे॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b3397"><div class="verse-block verse-justify"><div class="line">भीष्मोऽहमस्मि भद्रं ते दासोऽस्मि तव सुव्रत।</div><div class="line" style="position:relative;">तव संदर्शनादेव मुक्तोऽहं सर्वकिल्वपैः॥२४॥</div></div></div>
