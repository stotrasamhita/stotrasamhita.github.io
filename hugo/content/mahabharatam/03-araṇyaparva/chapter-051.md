<div class="verse-block-wrapper" id="03-araṇyaparva-b2265"><div class="verse-block verse-justify"><div class="line">ततश्चिन्तापरा दीना विवर्णवदना कृशा।</div><div class="line" style="position:relative;">बभूवदमयन्ती तु निःश्वासपरमा तदा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2266"><div class="verse-block verse-justify"><div class="line">ऊर्ध्वदृष्टिर्ध्यानपरा बभूवोन्मत्तदर्शना।</div><div class="line" style="position:relative;">पाण्डुवर्णा क्षणेनाथ हृच्छयाविष्टचेतना॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2267"><div class="verse-block verse-justify"><div class="line">न शय्यासनभोगेषु रतिं विन्दति कर्हिचित्।</div><div class="line" style="position:relative;">न नक्तं न दिवा शेते हाहेति रुदती मुहुः॥५॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b2269"><div class="verse-block verse-justify"><div class="line">ततो विदर्भपतये दमयन्त्याः सखीजनः।</div><div class="line" style="position:relative;">न्यवेदयत्तामस्वस्थां दमयन्तीं नरेश्वरः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2270"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा नृपतिर्भीमो दमयन्तीसखीगणात्।</div><div class="line" style="position:relative;">किमर्थं दुहिता मेऽद्यनातिस्वस्थेव लक्ष्यते॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2271"><div class="verse-block verse-justify"><div class="line">स समीक्ष्य महीपालः स्वां सुतां प्राप्तयौवनाम्।</div><div class="line" style="position:relative;">अपश्यदात्मना कार्यं दमयन्त्याः स्वयंवरम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2272"><div class="verse-block verse-justify"><div class="line">स सन्निपातयामास महीपालान्विशांपतिः।</div><div class="line" style="position:relative;">एषोऽनुभूयतां चीराः स्वयंवर इति प्रभो॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2273"><div class="verse-block verse-justify"><div class="line">श्रुत्वा तु पार्थिवाः सर्वे दमयन्त्याः स्वयंवरम्।</div><div class="line" style="position:relative;">अभिजग्मुस्ततो वीरा राजानो भीमशासनात्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2274"><div class="verse-block verse-justify"><div class="line">हस्त्यश्वरथघोषेण नादयन्तो वसुंधराम्।</div><div class="line" style="position:relative;">विचित्रमाल्याभरणैर्बलैर्दृश्यैः स्वलंकृतैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2275"><div class="verse-block verse-justify"><div class="line">तेषां भीमो महाबाहुः पार्थिवानां महात्मनाम्।</div><div class="line" style="position:relative;">यथार्हमकरोत्पूजां तेऽवसंस्तत्र पूजिताः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2276"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नेव काले तु सुराणामृषिसत्तमौ।</div><div class="line" style="position:relative;">अटमानौ महात्मानाविन्द्रलोकमितो गतौ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2277"><div class="verse-block verse-justify"><div class="line">नारदः पर्वतश्चैव महाप्राज्ञौ महाव्रतौ।</div><div class="line" style="position:relative;">देवराजस् भवनं विविशाते सुपूजितौ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2278"><div class="verse-block verse-justify"><div class="line">तावर्चयित्वा मघवा ततः कुशलमव्ययम्।</div><div class="line">पप्रच्छानामयं चापि तयोः सर्वगतं विभुः ॥नारद उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2279"><div class="verse-block verse-justify"><div class="line">आवयोः कशलं देव सर्वत्रगतमीश्वर।</div><div class="line">लोके च मघवन्कृत्स्ने नृपाः कुशलिनो विभो ॥बृहदश्व उवाच।</div><div class="line" style="position:relative;">॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2280"><div class="verse-block verse-justify"><div class="line">नारदस्य वचः श्रुत्वा पप्रच्छ बलवृत्रहा।</div><div class="line" style="position:relative;">धर्मज्ञाः पृथिवीपालास्त्यक्तजीवितयोधिनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2281"><div class="verse-block verse-justify"><div class="line">शस्त्रेण निधनं काले ये गच्छन्त्यपराड्युखाः।</div><div class="line" style="position:relative;">अयं लोकोऽक्षयस्तेषां यथैव मम कामधुक्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b2282"><div class="verse-block verse-justify"><div class="line">क्वनु ते क्षत्रियाः शूरा नहि पश्यामि तानहम्।</div><div class="line" style="position:relative;">आगच्छतो महीपालान्दयितामनिथीन्मम॥१९॥</div></div></div>
