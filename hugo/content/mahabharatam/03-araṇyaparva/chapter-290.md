<div class="verse-block-wrapper" id="03-araṇyaparva-b11606"><div class="verse-block verse-justify"><div class="line">तौ वीरौ शरजालेन बद्धाविन्द्रजिता रणे।</div><div class="line" style="position:relative;">रेजतुः पुरुषव्याघ्रौ शकुन्ताविव पञ्जरे॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11607"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा निपतितौ भूमौ सर्वाङ्गेषु शराचितौ।</div><div class="line" style="position:relative;">सुग्रीवः कपिभिः सार्धं परिवार्योपतस्तिवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11608"><div class="verse-block verse-justify"><div class="line">सुषेणमैन्दद्विविदैः कुमुदेनाङ्गदेन च।</div><div class="line" style="position:relative;">हनुमननीलतारैश्च नलेन च कपीश्वरः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11609"><div class="verse-block verse-justify"><div class="line">ततस्तं देशमागम्य कृतकर्मा विभीषणः।</div><div class="line" style="position:relative;">बोधयामास तौ वीरौ प्रज्ञास्त्रेण प्रमोहितौ॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11610"><div class="verse-block verse-justify"><div class="line">विशल्यौ चापि सुग्रीवः क्षणेनैतौ चकार ह।</div><div class="line" style="position:relative;">विशल्यया महौषध्या दिव्यमन्त्रप्रयुक्तया॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11611"><div class="verse-block verse-justify"><div class="line">तौ लब्धसंज्ञौ नृवरौ विशल्यावुदतिष्ठताम्।</div><div class="line" style="position:relative;">उभौ गतक्लमौ चास्तां णेनैतौ महारथौ॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11612"><div class="verse-block verse-justify"><div class="line">ततो विभीषणः पार्थ राममिक्ष्वाकुनन्दनम्।</div><div class="line" style="position:relative;">उवाच विज्वरं दृष्ट्वा कृताञ्जलिरिदं वचः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11613"><div class="verse-block verse-justify"><div class="line">अयमम्भो गृहीत्वातु राजराजस् शासनात्।</div><div class="line" style="position:relative;">गुह्कोऽभ्यागतः श्लेतात्त्वत्सकाशमरिंदम॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11614"><div class="verse-block verse-justify"><div class="line">इदमम्भः कुबेरस्ते महाराज प्रयच्छति।</div><div class="line" style="position:relative;">अन्तर्हितानां भूतानां दर्शनार्थं परंतप॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11615"><div class="verse-block verse-justify"><div class="line">अनेन मृष्टनयनो भूतान्यन्तर्हितान्युत।</div><div class="line" style="position:relative;">भवान्द्रक्ष्यति यस्मै च भवानेतत्प्रदास्यति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11616"><div class="verse-block verse-justify"><div class="line">तथेति रामस्तद्वारि प्रतिगृह्याभिसंस्कृतम्।</div><div class="line" style="position:relative;">चकार नेत्रयोः शौचं लक्ष्मणश्च महामनाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11617"><div class="verse-block verse-justify"><div class="line">सुग्रीवजाम्बवन्तौ चहनुमानङ्गदस्तथा।</div><div class="line" style="position:relative;">मैन्दद्विविदनीलाश्च प्रायः प्लवगसत्तमाः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11618"><div class="verse-block verse-justify"><div class="line">तथासमभवच्चापि यदुवाच विभीषणः।</div><div class="line" style="position:relative;">क्षणेनातीन्द्रियाण्येषां चक्षुंष्यासन्युधिष्ठिर॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11619"><div class="verse-block verse-justify"><div class="line">इन्द्रजित्कृतकर्मा तु पित्रे कर्म तदाऽऽत्मनः।</div><div class="line" style="position:relative;">निवेद्य पुनरागच्छत्त्वरयाऽऽजिशिरःप्रति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11620"><div class="verse-block verse-justify"><div class="line">तमागतं तु संक्रुद्धं पुनरेव युयुत्सया।</div><div class="line" style="position:relative;">अभिदुद्राव सौमित्रिर्विभीषणमते स्थितः॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="03-araṇyaparva-b11622"><div class="verse-block verse-justify"><div class="line">तयोः समभवद्युद्धं तदाऽन्योन्यं जीगीषतोः।</div><div class="line" style="position:relative;">अतीव चित्रमाश्चर्यं शक्रप्रह्लादयोरिव॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b11623"><div class="verse-block verse-justify"><div class="line">अविध्यदिन्द्रजित्तीक्ष्णैः सौमित्रिं मर्मभेदिभिः।</div><div class="line" style="position:relative;">सौमित्रिश्चानलस्पर्शैरविध्यद्रावणिं शरैः॥२०॥</div></div></div>
