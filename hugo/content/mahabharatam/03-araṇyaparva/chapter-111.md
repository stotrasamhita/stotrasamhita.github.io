<div class="verse-block-wrapper" id="03-araṇyaparva-b4780"><div class="verse-block verse-justify"><div class="line">लोमश उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4781"><div class="verse-block verse-justify"><div class="line">एषा देवनदी पुण्या कौशिकी भरतर्षभ।</div><div class="line" style="position:relative;">विश्वामित्राश्रमो रम्य एष चात्र प्रकाशते॥२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4782"><div class="verse-block verse-justify"><div class="line">आश्रमश्चैव पुण्याख्यः काश्यपस्य महात्मनः।</div><div class="line" style="position:relative;">ऋश्यशृङ्गः सुतो यस् तपस्वी संयतेन्द्रियः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4783"><div class="verse-block verse-justify"><div class="line">तपसो यः प्रभावेन वर्षयामास वासवम्।</div><div class="line" style="position:relative;">अनावृष्ठ्यां भयाद्यस्य ववर्ष बलवृत्रहा॥४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4784"><div class="verse-block verse-justify"><div class="line">मृग्यां जातः स तेजस्वी काश्यपस्य सुतः प्रभुः।</div><div class="line" style="position:relative;">विषये लोमपादस्य यश्चकाराद्भुतं महत्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4785"><div class="verse-block verse-justify"><div class="line">निर्वर्तितेषु सस्येषु यस्मै शान्तां ददौ नृपः।</div><div class="line">लोमपादो दुहितरं सावित्रीं सविता यथा ॥युधिष्ठिर उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4786"><div class="verse-block verse-justify"><div class="line">ऋश्यशृङ्गः कथं मृग्यामुत्पन्नः काश्यपात्मजः।</div><div class="line" style="position:relative;">विरुद्धयोनिसंसर्गः कथं च तपसा युतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4787"><div class="verse-block verse-justify"><div class="line">किमर्थं च भयाच्छक्रस्तस्य बालस्य धीमतः।</div><div class="line" style="position:relative;">अनावृष्ट्यां प्रवृत्तायां ववर्ष बलवृत्रहा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4788"><div class="verse-block verse-justify"><div class="line">कथंरूपा च सा शान्ता राजपुत्री पतिव्रता।</div><div class="line" style="position:relative;">लोभयामास या चेतो मृगभूस्य तस्य वै॥९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4789"><div class="verse-block verse-justify"><div class="line">लोमपादश्च राजर्षिर्यदाऽश्रूयत धार्मिकः।</div><div class="line" style="position:relative;">कथं वै विषये तस्य नावर्षत्पाकशासनः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4790"><div class="verse-block verse-justify"><div class="line">एतन्मे भगवन्सर्वं विस्तरेण यथातथम्।</div><div class="line">वक्तुमर्हसि शुश्रूषोर्ऋश्यशृङ्गस् चेष्टितम् ॥लोमश उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4791"><div class="verse-block verse-justify"><div class="line">विभाण्डकस्य ब्रह्मर्षेस्तपसा भावितात्मनः।</div><div class="line" style="position:relative;">अमोघवीर्यस्य सतः प्रजापतिसमद्युतेः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4792"><div class="verse-block verse-justify"><div class="line">शृणु पुत्रो यथा जात ऋश्यशृङ्गः प्रतापवान्।</div><div class="line" style="position:relative;">महाह्रदे महातेजा बालः स्थविरसंभतः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4793"><div class="verse-block verse-justify"><div class="line">महाह्रदं समासाद्य काश्यपस्तपसि स्थितः।</div><div class="line" style="position:relative;">दीर्घकालं रिश्रान्त ऋषिर्देवर्षिसंमितः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4794"><div class="verse-block verse-justify"><div class="line">तस्य रेतः प्रचस्कन्द दृष्ट्वाऽप्सरसमुर्वशीम्।</div><div class="line">अप्सूपस्पृशतो राजन्मृगी तच्चापिबत्तदा।</div><div class="line" style="position:relative;">सह तोयेन तृषिता गर्भिणी चाभवत्ततः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4795"><div class="verse-block verse-justify"><div class="line">सा पुरोक्ता भगवता ब्रह्मणा लोककर्तृणा।</div><div class="line" style="position:relative;">देवकन्या मृगी भूत्वा मुनिं सूय विमोक्ष्यसे॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4796"><div class="verse-block verse-justify"><div class="line">अमोघत्वाद्विधेश्चैव भावित्वाद्दैवनिर्मितात्।</div><div class="line">तस्यां मृग्यां समभवत्तस्व पुत्रो महानृषिः।</div><div class="line" style="position:relative;">ऋश्यशृङ्गस्तपोनित्यो वन एवाभ्यवर्धत॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4797"><div class="verse-block verse-justify"><div class="line">तस्य शृङ्गं शिरसि वै राजन्नासीन्महात्मनः।</div><div class="line" style="position:relative;">तेनर्श्यशृङ्ग इत्येवं तदा स प्रथितोऽभवत्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4798"><div class="verse-block verse-justify"><div class="line">न तेन दृष्टपूर्वोऽन्यः पितुरन्यत्रमानुषः।</div><div class="line" style="position:relative;">तस्मात्तस्य मनो नित्यं ब्रह्मचर्येऽभवन्नृप॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4799"><div class="verse-block verse-justify"><div class="line">एतस्मिन्नेव काले तु सखा दशरशस्य वै।</div><div class="line" style="position:relative;">कलोमपाद इतिख्यातो ह्यङ्गानामीश्वरोऽभवत्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4800"><div class="verse-block verse-justify"><div class="line">तेन कामः कृतो मिथ्या ब्राह्मणेभ्य इति श्रुतिः।</div><div class="line">`दैवोपहतसत्त्वेन धर्मज्ञेनापि भारत'।</div><div class="line" style="position:relative;">स ब्राह्मणैः परित्यक्तस्तदा भरतसत्तम॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4801"><div class="verse-block verse-justify"><div class="line">पुरोहितापचाराच्च तस्य राज्ञो यदच्छया।</div><div class="line" style="position:relative;">न ववर्ष सहस्राक्षस्ततोऽपीड्यन्त वै प्रजाः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4802"><div class="verse-block verse-justify"><div class="line">स ब्राह्मणान्पर्यपृच्छत्तपोयुक्तान्मनीषिणः।</div><div class="line" style="position:relative;">प्रवर्षणे सुरेन्द्रस्य समर्थान्पृथिवीपते॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4803"><div class="verse-block verse-justify"><div class="line">कथं प्रवर्षेत्पर्जन्य उपायः परिमृश्यताम्।</div><div class="line" style="position:relative;">तमूचुश्चोदितास्ते तु स्वमतानि मनीषिणः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4804"><div class="verse-block verse-justify"><div class="line">तत्र त्वेको रमुनिवरस्तं राजानमुवाच ह।</div><div class="line" style="position:relative;">कुपितास्तव राजेन्द्र ब्राह्मणा निष्कृतिं चर॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4805"><div class="verse-block verse-justify"><div class="line">ऋश्यशृङ्गं मुनिसुतमानयस्व च पार्थिव।</div><div class="line" style="position:relative;">ऐणेयमनभिज्ञं च नारीणामार्जवे रतम्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4806"><div class="verse-block verse-justify"><div class="line">स चेदवतरेद्राजन्विषयं ते महातपाः।</div><div class="line" style="position:relative;">सद्यः प्रवर्षेत्पर्जन्य इति मे नास्ति संशयः॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4807"><div class="verse-block verse-justify"><div class="line">एतच्छ्रुत्वा वचो राजन्कृत्वा निष्कृतिमात्मनः।</div><div class="line" style="position:relative;">स गत्वा पुनरागच्छत्प्रसन्नेषु द्विजातिषु॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4808"><div class="verse-block verse-justify"><div class="line">राजानमागतं ज्ञात्वा प्रतिसंजगृहुः प्रजाः।</div><div class="line" style="position:relative;">`स च ता प्रतिजग्राह पितेव हितकृत्सदा'॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4809"><div class="verse-block verse-justify"><div class="line">ततोऽङ्गपतिराहूय सचिवान्मन्त्रकोविदान्।</div><div class="line" style="position:relative;">ऋश्यशृङ्गागमे यत्नमकरोन्मन्त्रनिश्चये॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4810"><div class="verse-block verse-justify"><div class="line">सोऽध्यगच्छदुपायं तु तैरमात्यैः सहाच्युतः।</div><div class="line" style="position:relative;">शास्त्रज्ञैरलमर्थज्ञौर्नीत्यां च परिनिष्ठितैः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4811"><div class="verse-block verse-justify"><div class="line">ततश्चानाययामास वारमुख्या महीपतिः।</div><div class="line" style="position:relative;">वेश्याः सर्वत्रनिष्णातास्ता उवाच स पार्थिवः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4812"><div class="verse-block verse-justify"><div class="line">ऋश्यशृङ्गमृषेः पुत्रमानयध्वमुपायतः।</div><div class="line" style="position:relative;">लोभयित्वाऽभिविश्चास्य विषयं मम शोभनाः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4813"><div class="verse-block verse-justify"><div class="line">ता राजभयभीताश्च शापभीताश्च योषितः।</div><div class="line" style="position:relative;">अशक्यमूचुस्तत्कार्यं विषण्णा गतचेतसः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4814"><div class="verse-block verse-justify"><div class="line">तत्र त्वेका जरद्योषा राजानमिदमब्रवीत्।</div><div class="line" style="position:relative;">प्रयतिष्ये महाराज तमानेतुं तपोधनम्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4815"><div class="verse-block verse-justify"><div class="line">अभिप्रेतांस्तु मे कामांस्त्वमनुज्ञातुमर्हसि।</div><div class="line" style="position:relative;">ततः शक्ष्याम्यानयितुमृश्यशृङ्गमृषेः सुतम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4816"><div class="verse-block verse-justify"><div class="line">तस्याः सर्वमभिप्रेतमन्वजानात्स पार्थिवः।</div><div class="line" style="position:relative;">धनं च प्रददौ भूरि रत्नानि विविधानि च॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="03-araṇyaparva-b4817"><div class="verse-block verse-justify"><div class="line">ततो रूपेण सपन्ना वयसा च महीपते।</div><div class="line" style="position:relative;">स्त्रिय आदाय काश्चित्सा जगाम वनमञ्जसा॥३८॥</div></div></div></div>
