<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0766"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्र महाबाहो कच्चित्ते वर्तते तपः।</div><div class="line" style="position:relative;">कच्चिन्मनस्ते प्रीणाति वनवासे नराधिप॥२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0767"><div class="verse-block verse-justify"><div class="line">कच्चिद्धृदि न ते शोको राजन्पुत्रविनाशजः।</div><div class="line" style="position:relative;">कच्चिज्ज्ञानानि सर्वाणि सुप्रसन्नानि तेऽनघ॥३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0768"><div class="verse-block verse-justify"><div class="line">कच्चिद्बुद्धिं दृढां कृत्वा चरस्यारण्यकं विधिम्।</div><div class="line" style="position:relative;">कच्चिद्वधूश्च गान्धारी न शोकेनाभिभूयते॥४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0769"><div class="verse-block verse-justify"><div class="line">महाप्रज्ञा बुद्धिमती देवी धर्मार्थदर्शिनी।</div><div class="line" style="position:relative;">आगमापायतत्त्वज्ञा कच्चिदेषा न शोचति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0770"><div class="verse-block verse-justify"><div class="line">कच्चित्कुन्ती च राजंस्त्वां शुश्रूषत्यनहङ्कृता।</div><div class="line" style="position:relative;">या परित्यज्य स्वं पुत्रं गुरुशुश्रूषणे रता॥६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0771"><div class="verse-block verse-justify"><div class="line">कच्चिद्धर्मसुतो राजा त्वया प्रत्यभिनन्दितः।</div><div class="line" style="position:relative;">भीमार्जुनयमाश्चैव कच्चिदेतेऽपि सान्तिताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0772"><div class="verse-block verse-justify"><div class="line">कच्चिन्नन्दसि दृष्ट्रैतान्कच्चिते निर्मलं मनः।</div><div class="line" style="position:relative;">कच्चिच्च शुद्धभावोसि जातज्ञानो नराधिप॥८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0773"><div class="verse-block verse-justify"><div class="line">एतद्धि त्रितयं श्रेष्ठं सर्वभूतेषु भारत।</div><div class="line" style="position:relative;">निर्वैरता महाराज सत्यमक्रोध एव च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0774"><div class="verse-block verse-justify"><div class="line">कच्चित्ते न च मोहोस्ति वनवासेन भारत।</div><div class="line" style="position:relative;">स्वदत्ते वन्यमन्नं वा उपवासोपि वा भवेत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0775"><div class="verse-block verse-justify"><div class="line">विदितं चापि राजेन्द्र विदुरस्य महात्मनः।</div><div class="line" style="position:relative;">गमनं विधिनाऽनेन धर्मस्य सुमहात्मनः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0776"><div class="verse-block verse-justify"><div class="line">माण्डव्यशापाद्धि स वै धर्मो विदुरतां गतः।</div><div class="line" style="position:relative;">महाबुद्धिर्महायोगी महात्मा सुमहामनाः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0777"><div class="verse-block verse-justify"><div class="line">बृहस्पतिर्वा देवेषु शुक्रो वाऽप्यसुरेषु च।</div><div class="line" style="position:relative;">न तथा बुद्धिसम्पन्नो यथा स पुरुषर्षभः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0778"><div class="verse-block verse-justify"><div class="line">तपोबलव्ययं कृत्वा सुचिरात्सम्भृतं तदा।</div><div class="line" style="position:relative;">माण्डव्येनर्षिणा धर्मो ह्यभिभूतः सनातनः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0779"><div class="verse-block verse-justify"><div class="line">नियोगाद्ब्रह्मणः पूर्वं मया स्वेन बलेन च।</div><div class="line" style="position:relative;">वैचित्रवीर्यके क्षेत्रे जातः स सुमहामतिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0780"><div class="verse-block verse-justify"><div class="line">भ्राता तव महाराज देवदेवः सनातनः।</div><div class="line" style="position:relative;">धारणान्मनसा ध्यानाद्य धर्म कवयो विदुः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0781"><div class="verse-block verse-justify"><div class="line">सत्येन संवर्धयति यो दमेन शमेन च।</div><div class="line" style="position:relative;">अहिंसया च दानेन तपसा च सनातनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0782"><div class="verse-block verse-justify"><div class="line">येन योगबलाज्जातः कुरुराजो युधिष्ठिरः।</div><div class="line" style="position:relative;">धर्म इत्येष नृपते प्राज्ञेनामितबुद्धिना॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0783"><div class="verse-block verse-justify"><div class="line">यथा वह्निर्यथा वायुर्यथाऽऽपः पृथिवी यथा।</div><div class="line" style="position:relative;">यथाऽऽकाशं तथा धर्म इह चामुत्र च स्थितः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0784"><div class="verse-block verse-justify"><div class="line">सर्वगश्चैव राजेन्द्र सर्वं व्याप्य चराचरम्।</div><div class="line" style="position:relative;">दृश्यते देवदेवैः स सिद्धैर्निर्मुक्तकल्मषैः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0785"><div class="verse-block verse-justify"><div class="line">यो हि धर्मः स विदुरो विदुरो यः स पाण्डवः।</div><div class="line" style="position:relative;">स एष राजन्दृश्यस्ते माण्डवः प्रेष्यवत्स्थितः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0786"><div class="verse-block verse-justify"><div class="line">प्रविष्टः स महात्मानं भ्राता ते बुद्धिसम्मतः।</div><div class="line" style="position:relative;">दृष्ट्वा महात्मा कौन्तेयं महायोगबलान्वितः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0787"><div class="verse-block verse-justify"><div class="line">त्वां चापि श्रेयसा योक्ष्ये नचिराद्भरतर्षभ।</div><div class="line" style="position:relative;">संशयच्छेदनार्थाय प्राप्तं मां विद्धि पुत्रक॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0788"><div class="verse-block verse-justify"><div class="line">न कृतं यैः पुरा कैश्चित्कर्म लोके महर्षिभिः।</div><div class="line" style="position:relative;">आश्चर्यभूतं तपसः फलं तद्दर्शयामि वः॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0789"><div class="verse-block verse-justify"><div class="line">किमिच्छसि महीपाल मत्तः प्राप्तुमभीप्सितम्।</div><div class="line" style="position:relative;">द्रष्ट्रुं स्प्रष्टुमथ श्रोतुं तत्कर्तास्मि तवानघ॥२५॥</div></div></div></div>
