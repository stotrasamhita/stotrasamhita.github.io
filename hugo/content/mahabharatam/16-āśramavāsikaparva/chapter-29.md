<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0738"><div class="verse-block verse-justify"><div class="line">ततस्तु राजन्नेतेषामाश्रमे पुण्यकर्मणाम्।</div><div class="line" style="position:relative;">शिवा नक्षत्रसम्पन्ना सा व्यतीयाय शर्वरी॥१॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0739"><div class="verse-block verse-justify"><div class="line">ततस्तत्र कथाश्चासंस्तेषां धर्मार्थलक्षणाः।</div><div class="line" style="position:relative;">विचित्रपदसञ्चारा नानाश्रुतिभिरन्विताः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0740"><div class="verse-block verse-justify"><div class="line">पाण्डवास्त्वभितो मातुर्धरण्यां सुषुपुस्तदा।</div><div class="line" style="position:relative;">उत्सृज्य तु महार्हाणि शयनानि नराधिप॥३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0741"><div class="verse-block verse-justify"><div class="line">यदाहारोऽभकवद्राजा धृतराष्ट्रो महामनाः।</div><div class="line" style="position:relative;">तदाहारा नृवींरास्ते न्यवसंस्तां निशां तदा॥४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0742"><div class="verse-block verse-justify"><div class="line">व्यतीतायां तु शर्वर्यां कृतपौर्वाह्णिकक्रियः।</div><div class="line" style="position:relative;">भ्रातृभिः सहितो राजा ददर्शाश्रममण्डलम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0743"><div class="verse-block verse-justify"><div class="line">सान्तः पुरपरीवारः सभृत्यः सपुरोहितः।</div><div class="line" style="position:relative;">यथासुखं यथोद्देशं धृतराष्ट्राभ्यनुज्ञया॥६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0744"><div class="verse-block verse-justify"><div class="line">ददर्श तत्र वेदीश्च सम्प्रज्वलितपावकाः।</div><div class="line" style="position:relative;">कृताभिषेकैर्मुनिभिर्हुताग्निभिरुपस्थिताः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0745"><div class="verse-block verse-justify"><div class="line">वानेयपुष्पनिकरैराज्यधूमोद्गमैरपि।</div><div class="line" style="position:relative;">ब्राह्मेण वपुषा युक्ता युक्ता मुनिगणस्य ताः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0746"><div class="verse-block verse-justify"><div class="line">मृगयूथैरनुद्विग्नैस्तत्रतत्र समाश्रितैः।</div><div class="line" style="position:relative;">अशङ्कितैः पक्षिगणैः प्रगीतैरिव च प्रभो॥९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0747"><div class="verse-block verse-justify"><div class="line">केकाभिर्नीलकण्ठानां दात्यूहानां च कूजितैः।</div><div class="line" style="position:relative;">कोकिलानां कुहुरवैः सुखैः श्रुतिमनोहरैः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0748"><div class="verse-block verse-justify"><div class="line">प्राधीतद्विजघोषैश्च क्वचित्क्वचिदलङ्कृतम्।</div><div class="line" style="position:relative;">फलमूलसमाहारैर्महद्भिश्चोपशोभितम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0749"><div class="verse-block verse-justify"><div class="line">ततः स राजा प्रददौ तापसार्थमुपाहृतान्।</div><div class="line" style="position:relative;">कलशान्काञ्चनान्राजंस्तथैवौदुम्बरानपि॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0750"><div class="verse-block verse-justify"><div class="line">अजिनानि प्रवेणीश्च स्रुक्स्रुवं च महीपतिः।</div><div class="line" style="position:relative;">कमण्डलूंश्चि स्थालीश्च पिठराणि च भारत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0751"><div class="verse-block verse-justify"><div class="line">भाजनानि च लौहानि पात्रीश्च विविधा नृप।</div><div class="line" style="position:relative;">यद्यदिच्छति यावच्च यदन्यदपि काङ्क्षितम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0752"><div class="verse-block verse-justify"><div class="line">एवं स राजा धर्मात्मा परीत्याश्रममण्डलम्।</div><div class="line" style="position:relative;">वसु विश्राण्य तत्सर्वं पुनरायान्महीपतिः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0753"><div class="verse-block verse-justify"><div class="line">कृताह्निकं च राजानं धृतराष्ट्रं महीपतिम्।</div><div class="line" style="position:relative;">ददर्शासीनमव्यग्रं गान्धारीसहितं तदा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0754"><div class="verse-block verse-justify"><div class="line">मातरं चाविदूरस्थां शिष्यवत्प्रणतां स्थिताम्।</div><div class="line" style="position:relative;">कुन्तीं ददर्श धर्मात्मा शिष्टाचारसमन्विताम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0755"><div class="verse-block verse-justify"><div class="line">स तमभ्यर्च्य राजानं नाम संश्राव्य चात्मनः।</div><div class="line" style="position:relative;">निषीदेत्यभ्यनुज्ञातो वृस्यामुपविवेश ह॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0756"><div class="verse-block verse-justify"><div class="line">भीमसेनादयश्चैव पाण्डव भरतर्षभ।</div><div class="line" style="position:relative;">अभिवाद्योपसङ्गृह्य निषेदुः पार्थिवाज्ञया॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0757"><div class="verse-block verse-justify"><div class="line">स तैः परिवृतो राजा शुशुभेऽतीव कौरवः।</div><div class="line" style="position:relative;">बिभ्रद्ब्राह्मीं श्रियं दीप्तां देवैरिव बृहस्पतिः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0758"><div class="verse-block verse-justify"><div class="line">तथा तेषूपविष्टेषु समाजग्मुर्महर्षयः।</div><div class="line" style="position:relative;">शतयूपप्रभृतयः कुरुक्षेत्रनिवासिनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0759"><div class="verse-block verse-justify"><div class="line">व्यासश्च भगवान्विप्रो देवर्षिगणसेवितः।</div><div class="line" style="position:relative;">वृतः शिष्यैर्महातेजा दर्शयामास पार्थिवम्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0760"><div class="verse-block verse-justify"><div class="line">ततः स राजा कौरव्यः कुन्तीपुत्रश्च धर्मराट्।</div><div class="line" style="position:relative;">भीमसेनादयश्चैव प्रत्युत्थायाभ्यवादयन्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0761"><div class="verse-block verse-justify"><div class="line">समागतस्ततो व्यासः शतयूपादिभिर्वृतः।</div><div class="line" style="position:relative;">धृतराष्ट्रं महीपालमास्यतामित्यभाषत॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0762"><div class="verse-block verse-justify"><div class="line">वरं तु विष्टरं कौश्यं कृष्णाजिनकुशोत्तरम्।</div><div class="line" style="position:relative;">प्रतिपेदे तदा व्यासस्तदर्थमुपकल्पितम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0763"><div class="verse-block verse-justify"><div class="line">ते च सर्वे द्विजश्रेष्ठा विष्टरेषु समन्ततः।</div><div class="line" style="position:relative;">द्वैपायनाभ्यनुज्ञाता निषेदुर्विपुलौजसः॥२६॥</div></div></div></div>
