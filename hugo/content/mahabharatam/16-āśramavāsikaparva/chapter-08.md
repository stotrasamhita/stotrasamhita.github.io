<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0245"><div class="verse-block verse-justify"><div class="line">पर्युपासनकाले तु विपरीतं विधीयते।</div><div class="line" style="position:relative;">आमर्दकाले राजेन्द्र व्यवसायस्ततोऽपरः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0246"><div class="verse-block verse-justify"><div class="line">व्यसनं भेदनं चैव शत्रूणां कारयेत्ततः।</div><div class="line" style="position:relative;">कर्षणं भीषणं चैव युद्धे चैव बहुक्षयम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0247"><div class="verse-block verse-justify"><div class="line">प्रयास्यमानो नृपतिस्त्रिविधं परिचिन्तयेत्।</div><div class="line" style="position:relative;">आत्मनश्चैव शत्रोश्च शक्तिं शास्त्रविशारदः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0248"><div class="verse-block verse-justify"><div class="line">उत्साहप्रभुशक्तिभ्यां मन्त्रशक्त्या च भारत।</div><div class="line" style="position:relative;">उपपन्नो नृपो यायाद्विपरीतं च वर्जयेत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0249"><div class="verse-block verse-justify"><div class="line">आददीत बलं राजा मौलं मित्रबलं तथा।</div><div class="line" style="position:relative;">अटवीबलं भृतं चैव तता श्रेणीबलं प्रभो॥७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0250"><div class="verse-block verse-justify"><div class="line">`मित्रामित्रबलं राजन्न्यायाद्वृद्ध्युदये धृतः।</div><div class="line">'तत्र मित्रबलं राजन्मौलं चैव विशिष्यते।</div><div class="line" style="position:relative;">श्रेणीबलं भृतं चैव तुल्ये एवेति मे मतिः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0251"><div class="verse-block verse-justify"><div class="line">तथाऽऽचारबलं चैव परस्परसमं नृप।</div><div class="line" style="position:relative;">विज्ञेयं बलकालेषु राज्ञा काल उपस्थिते॥९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0252"><div class="verse-block verse-justify"><div class="line">आपदश्चापि बोद्धव्या बहुरूपा नराधिप।</div><div class="line" style="position:relative;">भवन्ति राज्ञा कौरव्य यास्ताः पृथगतः शृणु॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0253"><div class="verse-block verse-justify"><div class="line">विकल्पा बहुधा राजन्नापदां पाण्डुनन्दन।</div><div class="line" style="position:relative;">सामादिभिरुपन्यस्य गमयेत्तान्नृपः सदा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0254"><div class="verse-block verse-justify"><div class="line">यात्रां गच्छेद्बलैर्युक्तो राजा षड्भिः परंतप।</div><div class="line" style="position:relative;">युक्तश्च देशकालाभ्यां बलैरात्मगुणैस्तथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0255"><div class="verse-block verse-justify"><div class="line">हृष्टपुष्टबलो गच्छेद्राजा वृद्ध्युदये रतः।</div><div class="line" style="position:relative;">अकृशश्चाप्यथो यायादनृतावपि पाण्डव॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0256"><div class="verse-block verse-justify"><div class="line">तूणाश्मानं वाजिरथप्रवाहांध्वजद्रुमैः संवृतकूलरोधसम्।</div><div class="line" style="position:relative;">पदातिनागैर्बहुकर्दमां नदींसपत्नानाशे नृपतिः प्रयोजयेत्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0257"><div class="verse-block verse-justify"><div class="line">अथोपपत्त्या शकटं पद्मवज्रं च भारत।</div><div class="line" style="position:relative;">उशना वेद यच्छास्त्रं तत्रैतद्विहितं विभो॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0258"><div class="verse-block verse-justify"><div class="line">चारयित्वा परबलं कृत्वा स्वबलदर्शनम्।</div><div class="line" style="position:relative;">स्वभूमौ योजयेद्युद्धं परभूमौ तथैव च॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0259"><div class="verse-block verse-justify"><div class="line">बलं प्रसादयेद्राजा निक्षिपेद्बलनो नरान्।</div><div class="line" style="position:relative;">ज्ञात्वा स्वविषयं तत्र सामादिभिरुपक्रमेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0260"><div class="verse-block verse-justify"><div class="line">सर्वथैव महाराज शरीरं धारयेदिह।</div><div class="line" style="position:relative;">प्रेत्य चेह च कर्तव्यमात्मनिःश्रेयसं परम्॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0262"><div class="verse-block verse-justify"><div class="line">एवं त्वया कुरुश्रेष्ठ वर्तितव्यं प्रजासु वै।</div><div class="line" style="position:relative;">उभयोर्लोकयोस्तात प्राप्तये नित्यमेव हि॥२०॥</div></div></div>
