<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0639"><div class="verse-block verse-justify"><div class="line">आज्ञापयामास ततः सेनां भरतसत्तमः।</div><div class="line" style="position:relative;">अर्जुनप्रमुखैर्गुप्तां लोकपालोपमैर्नरैः॥१॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0640"><div class="verse-block verse-justify"><div class="line">योगोयोग इति प्रीत्या ततः शब्दो महानभूत्।</div><div class="line" style="position:relative;">क्रोशतां सादिनां तत्र युज्यतां युज्यतामिति॥२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0641"><div class="verse-block verse-justify"><div class="line">केचिद्यानैर्नरा जग्मुः केचिदश्वैर्महाजवैः।</div><div class="line" style="position:relative;">काञ्चनैश्च रथैः केचिज्ज्वलितज्वलनोपमैः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0642"><div class="verse-block verse-justify"><div class="line">गजेन्द्रैश्च तथैवान्ये केचिदुष्ट्रैर्नराधिप।</div><div class="line" style="position:relative;">पदातयस्तथैवान्ये नखरप्रासयोधिनः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0643"><div class="verse-block verse-justify"><div class="line">पौरजानपदाश्चैव यानैर्बहुविधैस्तथा।</div><div class="line" style="position:relative;">अन्वयुः कुरुराजानं धृतराष्ट्रं दिदक्षवः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0644"><div class="verse-block verse-justify"><div class="line">स चापि राजवचनादाचार्यो गौतमः कृपः।</div><div class="line" style="position:relative;">सेनामादाय सेनानीः प्रययावाश्रमं प्रति॥६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0645"><div class="verse-block verse-justify"><div class="line">ततो द्विजैः परिवृतः कुरुराजो युधिष्ठिरः।</div><div class="line" style="position:relative;">संस्तूयमानो बहुभिः सूतमागधबन्दिभिः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0646"><div class="verse-block verse-justify"><div class="line">पाण्डुरेणातपत्रेण ध्रियमाणेन मूर्धनि।</div><div class="line" style="position:relative;">रतानीकेन महता निर्जगाम कुरूद्वहः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0647"><div class="verse-block verse-justify"><div class="line">गजैश्चाचलसंकाशैर्भीमकर्मा वृकोदरः।</div><div class="line" style="position:relative;">सज्जयन्त्रायुधोपेतैः प्रययौ पवनात्मजः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0648"><div class="verse-block verse-justify"><div class="line">माद्रीपुत्रावपि तथा हयारोहौ सुसंवृतौ।</div><div class="line" style="position:relative;">जग्मतुः शीघ्रगमनौ सन्नद्धकवचध्वजौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0649"><div class="verse-block verse-justify"><div class="line">अर्जुनश्च महातेजा रथेनादित्यवर्चसा।</div><div class="line" style="position:relative;">वशी श्वेतैर्हयैर्युक्तैर्दिव्येनान्वगमन्नृपम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0650"><div class="verse-block verse-justify"><div class="line">द्रौपदीप्रमुखाश्चापि स्त्रीसङ्घाः शिबिकागताः।</div><div class="line" style="position:relative;">स्त्र्यध्यक्षगुप्ताः प्रययुर्विसृजन्तोऽमितं वसु॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0651"><div class="verse-block verse-justify"><div class="line">समृद्धरथहस्त्यश्वं वेणुवीणानुनादितम्।</div><div class="line" style="position:relative;">शुशुभे पाण्डवं सैन्यं तत्तदा भरतर्षभ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0652"><div class="verse-block verse-justify"><div class="line">नदीतीरेषु रम्येषु सरःसु च विशाम्पते।</div><div class="line" style="position:relative;">वासान्कृत्वा क्रमेणाथ जग्मुस्ते कुरुपुङ्गवाः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0653"><div class="verse-block verse-justify"><div class="line">युयुत्सुश्च महातेजा धौम्यश्चैवि पुरोहितः।</div><div class="line" style="position:relative;">युधिष्ठिरस्य वचनात्पुरगुप्तिं प्रचक्रतुः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0654"><div class="verse-block verse-justify"><div class="line">ततो युधिष्ठिरो राजा कुरुक्षेत्रमवातरत्।</div><div class="line" style="position:relative;">क्रमोणोत्तीर्य यमुनां नदीं परमपाविनीम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0655"><div class="verse-block verse-justify"><div class="line">स ददर्शाश्रमं दूराद्राजर्षेस्तस्य धीमतः।</div><div class="line" style="position:relative;">शतयूपस्य कौरव्य धृतराष्ट्रस्य चैव ह॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="16-āśramavāsikaparva-b0656"><div class="verse-block verse-justify"><div class="line">ततः प्रमुदितः सर्वो जनस्तद्वनमञ्जसा।</div><div class="line" style="position:relative;">विवेश सुमहानादैरापूर्य भरतर्षभ॥१८॥</div></div></div></div>
