<div class="verse-block-wrapper" id="15-mausalaparva-b0193"><div class="verse-block verse-justify"><div class="line">राजा च भीमसेनश्च सहदेवश्च पाण्डवः।</div><div class="line" style="position:relative;">नकुलो याज्ञसेनी च षडेकमनसो वयम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="15-mausalaparva-b0196"><div class="verse-block verse-justify"><div class="line">इत्युक्त्वा दारुकमिदं वाक्यमाह धनंजयः।</div><div class="line" style="position:relative;">अमात्यान्वृष्णिवीराणां द्रष्टुमिच्छामि माचिरम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0197"><div class="verse-block verse-justify"><div class="line">इत्येवमुक्त्वा वचनं सुधर्मां यादवीं सभाम्।</div><div class="line" style="position:relative;">प्रविवेशार्जुनः शूरः शोचमानो महाभुजः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0198"><div class="verse-block verse-justify"><div class="line">तमासनगतं तत्र सर्वाः प्रकृतयस्तथा।</div><div class="line" style="position:relative;">ब्राह्मणा नागरास्तत्र परिवार्योपतस्थिरे॥८॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0199"><div class="verse-block verse-justify"><div class="line">तान्दीनमनसः सर्वान्विमूढान्गतचेतसः।</div><div class="line" style="position:relative;">उवाचेदं वचः काले पार्थो दीनतरस्तथा॥९॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0200"><div class="verse-block verse-justify"><div class="line">शक्रप्रस्थमहं नेष्ये पृष्णन्धकजनं स्वयम्।</div><div class="line" style="position:relative;">इदं तु नगरं सर्वं समुद्रः प्लावयिष्यति॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0201"><div class="verse-block verse-justify"><div class="line">सज्जीकुरुत यानानि रत्नानि विविधानि च।</div><div class="line" style="position:relative;">वज्रोऽयं भवतां राजा शक्रप्रस्थे भविष्यति॥११॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0202"><div class="verse-block verse-justify"><div class="line">सप्तमे दिवसे चैव रवौ विमल उद्गते।</div><div class="line" style="position:relative;">बहिर्वत्स्यामहे सर्वे सज्जीभावत माचिरम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0203"><div class="verse-block verse-justify"><div class="line">इत्युक्तास्तेन ते सर्वे पार्थेनाक्लिष्टकर्मणा।</div><div class="line" style="position:relative;">सज्जमाशु ततश्चक्रुः स्वसिद्ध्यर्थं समुत्सुकाः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0204"><div class="verse-block verse-justify"><div class="line">तां रात्रिमवसत्पार्थः केशवस्य निवेशने।</div><div class="line" style="position:relative;">महता शोकमोहेन सहसाऽभिपरिप्लुतः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0205"><div class="verse-block verse-justify"><div class="line">श्वोभूतेऽथ ततः शौरिर्वसुदेवः प्रतापवान्।</div><div class="line" style="position:relative;">युक्त्वाऽऽत्मानं महातेजा जगाम गतिमुत्तमां॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0206"><div class="verse-block verse-justify"><div class="line">ततः शब्दो महानासीद्वसुदेवनिवेशने।</div><div class="line" style="position:relative;">दारुणः क्रोशतीनां च रुदतीनां च योषिताम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0207"><div class="verse-block verse-justify"><div class="line">प्रकीर्णमूर्धजाः सर्वा विमुक्ताभरणस्रजः।</div><div class="line">उरांसि पाणिभिर्घ्नन्त्यो व्यलपन्तकरुणं स्त्रियः।</div><div class="line" style="position:relative;">॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0208"><div class="verse-block verse-justify"><div class="line">तं देवकी च भद्रा च रोहिणी मदिरा तथा।</div><div class="line" style="position:relative;">अन्वारोढुं व्यवसिता भर्तारं योषितां वराः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0209"><div class="verse-block verse-justify"><div class="line">ततः शौरिं नृयुक्तेन बहुमूल्येन भारत।</div><div class="line" style="position:relative;">यानेन महता पार्थो बहिर्निष्क्रामयत्तदा॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0210"><div class="verse-block verse-justify"><div class="line">तमन्वयुस्तत्रतत्रं दुःखसोकसमन्विताः।</div><div class="line" style="position:relative;">द्वारकावासिनः सर्वे पौरजानपदा हिताः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0211"><div class="verse-block verse-justify"><div class="line">तस्याश्वमेधिकं छत्रं दीप्यमानाश्च पावकाः।</div><div class="line" style="position:relative;">पुरस्तात्तस्य यानस्य याजकाश्च ततो ययुः॥२१॥</div></div></div>
<p class="prose">अनुजग्मुश्च तं वीरं देव्यस्ता वै स्वलङ्कृताः ॥स्त्रीसहस्रैः परिवृता वधूभिश्च सहस्रशः</p>
<div class="verse-block-wrapper" id="15-mausalaparva-b0213"><div class="verse-block verse-justify"><div class="line">यस्तु देशः प्रियस्तस्य जीवतोऽभून्महात्मनः।</div><div class="line" style="position:relative;">तत्रैनमुपसंकल्प्य वितृमेधं प्रचक्रिरे॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0214"><div class="verse-block verse-justify"><div class="line">तं चिताग्निगतं वीरं शूरपुत्रं वराङ्गनाः।</div><div class="line" style="position:relative;">ततोन्वारुरुहुः पत्न्यश्चतस्रः पतिलोकगाः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0215"><div class="verse-block verse-justify"><div class="line">तं वै चतसृभिः स्त्रीभिरन्वितं पाण्डुनन्दनः।</div><div class="line" style="position:relative;">अदाहयच्चन्दनैश्च गन्धैरुच्चावचैरपि॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0216"><div class="verse-block verse-justify"><div class="line">ततः प्रादुरभूच्छब्दः समिद्धस्य विभावसोः।</div><div class="line" style="position:relative;">सामगानां च निर्घोषो नराणां रुदतामपि॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0217"><div class="verse-block verse-justify"><div class="line">ततो वज्रप्रधानास्ते वृष्ण्यन्धककुमारकाः।</div><div class="line">सर्वे चैवोदकं चक्रुः स्त्रियश्चैव महात्मनः।</div><div class="line" style="position:relative;">॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0218"><div class="verse-block verse-justify"><div class="line">अलुप्तधर्मस्तं धर्मं कारयित्वा स फल्गुनः।</div><div class="line" style="position:relative;">जगाम वृष्णयो यत्र विनष्टा भरतर्षभ॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0219"><div class="verse-block verse-justify"><div class="line">स तान्दृष्ट्वा निपतितान्कदने भृशदुःखितः।</div><div class="line" style="position:relative;">बभूवातीव कौरव्यः प्राप्तकालं चकार ह॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0220"><div class="verse-block verse-justify"><div class="line">यथाप्रधानतश्चैव चक्रे सर्वास्तथा क्रियाः।</div><div class="line" style="position:relative;">ये हता ब्रह्मशापेन मुसलैरेरकोद्भवैः॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0221"><div class="verse-block verse-justify"><div class="line">`ततो भगवतो देहं दृष्ट्वा शिष्यः प्रलप्य च।</div><div class="line" style="position:relative;">स्मृत्वा तद्वचनं सर्वं मोहशोकोपबृंहितः ॥'॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0222"><div class="verse-block verse-justify"><div class="line">ततः शरीरे रामस्य वासुदेवस्य चोभयोः।</div><div class="line" style="position:relative;">अन्वीक्ष्य दाहयामास पुरुषैराप्तकारिभिः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0223"><div class="verse-block verse-justify"><div class="line">स तेषां विधिवत्कृत्वा प्रेतकार्याणि पाण्डवः।</div><div class="line" style="position:relative;">सप्तमे दिवसे प्रायाद्रथमारुह्य सत्वरः॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="15-mausalaparva-b0225"><div class="verse-block verse-justify"><div class="line">अनुजग्मुर्महात्मानं पाण्डुपुत्रं धनंजयम्।</div><div class="line" style="position:relative;">भृत्याश्चान्धकवृष्णीनां सादिनो रथिनश्च ये॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0226"><div class="verse-block verse-justify"><div class="line">वीरहीना वृद्धबालाः पौरजानपदास्तथा।</div><div class="line" style="position:relative;">ययुस्ते परिवार्याथ कलत्रं पार्थशासनात्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0227"><div class="verse-block verse-justify"><div class="line">कुञ्जरैश्च गजारोहा ययुः शैलनिर्भस्तथा।</div><div class="line" style="position:relative;">सपादरक्षैः संयुक्ताः सोत्तरायुधिका ययुः॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0228"><div class="verse-block verse-justify"><div class="line">पुत्राश्चान्धकवृष्णीनां सर्वे पार्थमनुव्रताः।</div><div class="line" style="position:relative;">ब्राह्मणाः क्षत्रिया वैश्याः शूद्राश्चैव महाधनाः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0229"><div class="verse-block verse-justify"><div class="line">दश षट् च सहस्राणि वासुदेवावरोधनम्।</div><div class="line" style="position:relative;">पुरस्कृत्य ययुर्वज्रं पौत्रं कृष्णस्य धीमतः॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0230"><div class="verse-block verse-justify"><div class="line">बहूनि च सहस्राणि प्रयुतान्यर्बुदानि च।</div><div class="line" style="position:relative;">भोजवृष्ण्यन्धकस्त्रीणां हतनाथानि निर्ययुः॥३९॥</div></div></div>

<div class="verse-block-wrapper" id="15-mausalaparva-b0235"><div class="verse-block verse-justify"><div class="line">काननेषु च रम्येषु पर्वतेषु नदीषु च।</div><div class="line" style="position:relative;">निवसन्नानयामास वृष्णिदारान्धनंजयः॥४४॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0236"><div class="verse-block verse-justify"><div class="line">स पञ्चनदमासाद्य धीमानतिसमृद्धिमत्।</div><div class="line" style="position:relative;">देशे गोपशुधान्याढ्ये निवासमकरोत्प्रभुः॥४५॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0237"><div class="verse-block verse-justify"><div class="line">ततो लोभः समभवद्दस्यूनां निहतेश्वराः।</div><div class="line" style="position:relative;">दृष्ट्वा स्त्रियो नीयमानाः पार्थेनैकेन भारत॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0238"><div class="verse-block verse-justify"><div class="line">ततस्ते पापकर्माणो लोभोपहतचेतसः।</div><div class="line" style="position:relative;">आभीरा मन्त्रयामासुः समेत्याशुभदर्शनाः॥४७॥</div></div></div>
