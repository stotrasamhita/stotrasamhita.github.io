<div class="verse-block-wrapper" id="15-mausalaparva-b0069"><div class="verse-block verse-justify"><div class="line">ततो जिगमिषन्तस्ते वृष्ण्यन्धकमहारथाः।</div><div class="line" style="position:relative;">सान्तःपुरास्तदा तीर्थयात्रामैच्छन्नरर्षभाः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0070"><div class="verse-block verse-justify"><div class="line">ततो मनोहरा हृष्टाःक पेयं मांसं मधूनि च।</div><div class="line" style="position:relative;">बहु नानाविधं चक्रुस्त्वरिताः कालचोदिताः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0071"><div class="verse-block verse-justify"><div class="line">ततः सीधुषु सक्ताश्च निर्ययुर्नगराद्बहिः।</div><div class="line" style="position:relative;">यानैरश्वैर्गजैश्चैव श्रीमन्तस्तिग्मतेजसः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0072"><div class="verse-block verse-justify"><div class="line">ततः प्रभासे न्यवसन्यथोद्दिष्टं यथागृहम्।</div><div class="line" style="position:relative;">प्रभूतभक्ष्यपेयास्ते सदारा यादवास्तदा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0073"><div class="verse-block verse-justify"><div class="line">निविष्टांस्तान्निशाम्याथ समुद्रान्ते स योगवित्।</div><div class="line" style="position:relative;">जगामामन्त्र्य तान्वीरानुद्धवोऽर्थविशारदः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0074"><div class="verse-block verse-justify"><div class="line">तं प्रस्थितं महात्मानमभिवाद्य कृताञ्जलिम्।</div><div class="line" style="position:relative;">जानन्विनाशं वृष्णीनां नैच्छद्वारयितुं हरिः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0075"><div class="verse-block verse-justify"><div class="line">ततः कालपरीतास्ते वृष्ण्यन्धकमहारथाः।</div><div class="line" style="position:relative;">अपश्यन्नुद्धवं यान्तं तेजसाऽऽवृत्य रोदसी॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0076"><div class="verse-block verse-justify"><div class="line">ब्राह्मणार्थेषु यत्सिद्धमन्नं तेषां महात्मनाम्।</div><div class="line" style="position:relative;">तद्वानरेभ्यः प्रददुः सुरागन्धसमन्वितम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0077"><div class="verse-block verse-justify"><div class="line">ततस्तूर्यशताकीर्णं नटनर्तकसंकुलम्।</div><div class="line" style="position:relative;">अवर्तत महापानं प्रभासे तिग्मतेजसाम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0078"><div class="verse-block verse-justify"><div class="line">कृष्णस्य सन्निधौ रामः सहितः कृतवर्मणा।</div><div class="line" style="position:relative;">अपिबद्युयुधानश्च गदो बभ्रुस्तथैव च॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0079"><div class="verse-block verse-justify"><div class="line">ततः परिषदो मध्ये युयुधानो मदोत्कटः।</div><div class="line" style="position:relative;">अब्रवीत्कृतवर्माणमवहास्यावमत्य च॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="15-mausalaparva-b0081"><div class="verse-block verse-justify"><div class="line">इत्युक्ते युयुधानेन पूजयामास तद्वचः।</div><div class="line" style="position:relative;">प्रद्युम्नो रथिनांश्रेष्ठो हार्दिक्यमवमत्य व॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0082"><div class="verse-block verse-justify"><div class="line">ततः परमसंक्रुद्धः कृतवर्मा तमब्रवीत्।</div><div class="line" style="position:relative;">निर्दिशन्निव सावज्ञं तदा सव्येन पाणिना॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="15-mausalaparva-b0083"><div class="verse-block verse-justify"><div class="line">भूरिश्रवाश्छिन्नबाहुर्युर्द्धे प्रायगतस्त्वया।</div><div class="line" style="position:relative;">वधेन सुनृशंसेन कथं वीरेण पातितः॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="15-mausalaparva-b0086"><div class="verse-block verse-justify"><div class="line">तच्छ्रुत्वा केशवस्याङ्कमगमद्रुदती तदा।</div><div class="line" style="position:relative;">सत्यभामा प्रकुपिता कोपयन्ती जनार्दनम्॥२५॥</div></div></div>
