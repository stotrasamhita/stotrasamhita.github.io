<div class="verse-block-wrapper" id="05-udyogaparva-b2637"><div class="verse-block verse-justify"><div class="line">सञ्जय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2638"><div class="verse-block verse-justify"><div class="line">अर्जुनो वासुदेवश्च धन्विनौ परमार्थितौ।</div><div class="line" style="position:relative;">कामादन्यत्र संभूतौ सर्वभावाय संमितौ॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2639"><div class="verse-block verse-justify"><div class="line">व्यामान्तरं समास्थय यथामुक्तं मनस्विनः।</div><div class="line" style="position:relative;">चक्रं तद्वासुदेवस्य मायया वर्तते विभो॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2640"><div class="verse-block verse-justify"><div class="line">सापह्नवं कौरवेषु पाण्डवानां सुसंमतम्।</div><div class="line" style="position:relative;">सारासारबलं ज्ञातुं तेजःपुञ्जावभासितम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2641"><div class="verse-block verse-justify"><div class="line">नरकं शम्बरं चैव कंसं चैद्यं च माधवः।</div><div class="line" style="position:relative;">जितवान्घोरसङ्काशान्क्रीडन्निव महाबलः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2642"><div class="verse-block verse-justify"><div class="line">पृथिवीं चान्तरिक्षं च द्यां चैव पुरुषोत्तमः।</div><div class="line" style="position:relative;">मनसैव विशिष्टात्मा नयत्यात्मवशं वशी॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2643"><div class="verse-block verse-justify"><div class="line">भूयो भूयो हि यद्राजन्पृच्छसे पाण्डवान्प्रति।</div><div class="line" style="position:relative;">सारासारबलं ज्ञातुं तत्समासेन मे शृणु॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2644"><div class="verse-block verse-justify"><div class="line">एकतो वा जगत्कृत्स्नमेकतो वा जनार्दनः।</div><div class="line" style="position:relative;">सारतो जगतः कृत्स्नादतिरिक्तो जनार्दनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2645"><div class="verse-block verse-justify"><div class="line">भस्म कुर्याञ्जगदिदं मनसैव जनार्दनः।</div><div class="line" style="position:relative;">न तु कृत्स्नं जगच्छक्तं किञ्चित्कर्तुं जनार्दने॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2646"><div class="verse-block verse-justify"><div class="line">यतः सत्यं यतो धर्मो यतो ह्रीरार्जवं यतः।</div><div class="line" style="position:relative;">ततो भवति गोविन्दो यतः कृष्णस्ततो जयः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2647"><div class="verse-block verse-justify"><div class="line">पृथिवीं चान्तरिक्षं च दिवं च पुरुषोत्तमः।</div><div class="line" style="position:relative;">विचेष्टयति भूतात्मा क्रीडन्निव जनार्दनः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2648"><div class="verse-block verse-justify"><div class="line">स कृत्वा पाण्डवान्सत्रं लोकं संमोहयन्निव।</div><div class="line" style="position:relative;">अधर्मनिरतान्मूढान्दग्धुमिच्छति ते सुतान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2649"><div class="verse-block verse-justify"><div class="line">कालचक्रं जगच्चक्रं युगचक्रं च केशवः।</div><div class="line" style="position:relative;">आत्मयोगेन भगवान्परिवर्तयतेऽनिशम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2650"><div class="verse-block verse-justify"><div class="line">कालस्य च हि मृत्योश्च जङ्गमस्थावरस्य च।</div><div class="line" style="position:relative;">ईष्टे हि भगवानेकः सत्यमेतद्ब्रवीमि ते॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2651"><div class="verse-block verse-justify"><div class="line">ईशन्नपि महायोगी सर्वस्य जगतो हरिः।</div><div class="line" style="position:relative;">कर्माण्यारभते कर्तुं कीनाश इव दुर्बलः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2652"><div class="verse-block verse-justify"><div class="line">तेन वञ्चयते लोकान्मायायोगेन केशवः।</div><div class="line" style="position:relative;">ये तमेव प्रपद्यन्ते न ते मुह्यन्ति मानवाः॥१६॥</div></div></div></div>
