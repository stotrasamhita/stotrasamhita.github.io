<div class="verse-block-wrapper" id="05-udyogaparva-b6606"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b6608"><div class="verse-block verse-justify"><div class="line">स्नातापवृत्तैस्तुरगैर्लब्धतोयैरविह्वलैः।</div><div class="line" style="position:relative;">प्रभाते चोदिते सूर्ये ततो युद्धमवर्तत॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6609"><div class="verse-block verse-justify"><div class="line">दृष्ट्वा भां तूर्णमायान्तं दंशितं स्यन्दने स्थितम्।</div><div class="line" style="position:relative;">अकरोद्रथमत्यर्थं रामः सज्जं प्रतापवान्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6610"><div class="verse-block verse-justify"><div class="line">ततोऽहं राममायान्तं दृष्ट्वा समरकाङ्क्षिणम्।</div><div class="line" style="position:relative;">धनुःश्रेष्ठं समित्सृज्य सहसावतरं रथात्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6611"><div class="verse-block verse-justify"><div class="line">अभिवाद्य तथैवाहं रथमारुह्य भारत।</div><div class="line" style="position:relative;">युयुत्सुर्जामदग्न्यस्य प्रमुखे वीतभीः स्थितः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6612"><div class="verse-block verse-justify"><div class="line">ततोऽहं शरवर्षेण महता समवाकिरम्।</div><div class="line" style="position:relative;">स च मां शरवर्षेण वर्षन्तं समवाकिरम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6613"><div class="verse-block verse-justify"><div class="line">संक्रुद्धो जामदग्न्यस्तु पुनरेव सुतेजितान्।</div><div class="line" style="position:relative;">संप्रैषीन्मे शरान्घोरान्दीप्तास्यानुरगानिव॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6614"><div class="verse-block verse-justify"><div class="line">ततोऽहं निशितैर्भल्लैः शतशोऽथ सहस्रशः।</div><div class="line" style="position:relative;">अच्छिदं सहसा राजन्नन्तरिक्षे पुनः पुनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6615"><div class="verse-block verse-justify"><div class="line">ततस्त्वस्त्राणि दिव्यानि जामदग्न्यः प्रतापवान्।</div><div class="line">मयि प्रयोजयामास तान्यहं प्रत्यषेधयम्।</div><div class="line" style="position:relative;">॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6616"><div class="verse-block verse-justify"><div class="line">अस्त्रैरेव महाबाहो चिकीर्षन्नधिकां क्रियाम्।</div><div class="line" style="position:relative;">ततो विदि महान्नादः प्रादुरासीत्समन्ततः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6617"><div class="verse-block verse-justify"><div class="line">ततोऽहमस्त्रं वायव्यं जामदग्न्ये प्रयुक्तवान्।</div><div class="line" style="position:relative;">प्रत्याजघ्ने च तद्रामो गुह्यकास्त्रेण भारत॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6618"><div class="verse-block verse-justify"><div class="line">ततोऽहमस्त्रमाग्नेयमनुमन्त्र्य प्रयुक्तवान्।</div><div class="line" style="position:relative;">वारुणेनैव तद्रामो वारयामास मे विभुः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6619"><div class="verse-block verse-justify"><div class="line">एवमस्त्राणि दिव्यानि रामस्याहमवारयम्।</div><div class="line" style="position:relative;">रामश्च मम तेजस्वी दिव्यास्त्रविदरिन्दमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6620"><div class="verse-block verse-justify"><div class="line">ततो मां सव्यतो राजन्रामः कुर्वन्द्विजोत्तमः।</div><div class="line" style="position:relative;">उरस्यविध्यत्संक्रुद्धो जामदग्न्यः प्रतापवान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6621"><div class="verse-block verse-justify"><div class="line">ततोऽहं भारतश्रेष्ठ संन्यषीदं रथोत्तमे।</div><div class="line" style="position:relative;">ततो मां कश्मलाविष्टं सूतस्तूर्णमुदावहत्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6622"><div class="verse-block verse-justify"><div class="line">ग्लायन्तं भरश्रेष्ठ रामबाणप्रपीडितम्।</div><div class="line" style="position:relative;">ततो मामपयातं वै भृशं विद्धमचेतसम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6623"><div class="verse-block verse-justify"><div class="line">रामस्यानुचरा हृष्टाः सर्वे दृष्ट्वा विचुक्रुशुः।</div><div class="line" style="position:relative;">अकृतव्रणप्रभृतयः काशिकन्या च भारत॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6624"><div class="verse-block verse-justify"><div class="line">ततस्तु लब्धसंज्ञोऽहं ज्ञात्वा सूतमथाब्रुवम्।</div><div class="line" style="position:relative;">याहि सूत यतो रामः सज्जोऽहं गतवेदनः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6625"><div class="verse-block verse-justify"><div class="line">ततो मामवहत्सूतो हयैः परमशोभितैः।</div><div class="line" style="position:relative;">नृत्यद्भिरिव कौरव्य मारुतप्रतिमैर्गतौ॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6626"><div class="verse-block verse-justify"><div class="line">ततोऽहं राममासाद्य बाणवर्षैश्च कौरव।</div><div class="line" style="position:relative;">अवाकिरं सुसंरब्धः संरब्धं च जिगीषया॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6627"><div class="verse-block verse-justify"><div class="line">तानापतत एवासौ रामो बाणानजिह्मगान्।</div><div class="line" style="position:relative;">बाणैरेवाच्छिनत्तूर्णमेकैकं त्रिभिराहवे॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6628"><div class="verse-block verse-justify"><div class="line">ततस्ते सूदिताः सर्वे मम बाणाः सुसंशिताः।</div><div class="line" style="position:relative;">रामबाणैर्द्विधा च्छिन्नाः शतशोऽथ सहस्रशः॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6629"><div class="verse-block verse-justify"><div class="line">ततः पुनः शरं दीप्तं सुप्रभं कालसंमितम्।</div><div class="line" style="position:relative;">असृजं जामदग्न्याय रामायाहं जिघांसया॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6630"><div class="verse-block verse-justify"><div class="line">तेन त्वभिहतो गाढं बाणवेगवशं गतः।</div><div class="line" style="position:relative;">मुमोह समरे रामो भूमौ च निपपात ह॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6631"><div class="verse-block verse-justify"><div class="line">ततो हाहाकृतं कसर्वं रामे भूतलमाश्रिते।</div><div class="line" style="position:relative;">जगद्भारत संविग्नं यथार्कपतने भवेत्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6632"><div class="verse-block verse-justify"><div class="line">तत एनं समुद्विग्नाः सर्व एवाभिदुद्रुवुः।</div><div class="line" style="position:relative;">तपोधनास्ते सहसा काश्या च कुरुनन्दन॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6633"><div class="verse-block verse-justify"><div class="line">तत एनं परिष्वज्य शनैराश्वासयंस्तदा।</div><div class="line" style="position:relative;">पाणिभिर्जलशीतैश्च जयाशीर्भिश्च कौरव॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6634"><div class="verse-block verse-justify"><div class="line">ततः स विह्वलं वाक्यं राम उत्थाय चाब्रवीत्।</div><div class="line" style="position:relative;">तिष्ठ भीष्म हतोऽसीति बाणं संधाय कार्मुके॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6635"><div class="verse-block verse-justify"><div class="line">स मुक्तो न्यपतत्तूर्णं सव्ये पार्श्वे महाहवे।</div><div class="line" style="position:relative;">येनाहं भृशमुद्विग्रो व्याघूर्णित इव द्रुमः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6636"><div class="verse-block verse-justify"><div class="line">हत्वा हयांस्ततो रामः शीघ्रास्त्रेण महाहवे।</div><div class="line" style="position:relative;">अवाकिरन्मां विस्रब्धो बाणैस्तैर्लोमवाहिभिः॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6637"><div class="verse-block verse-justify"><div class="line">ततोऽहमपि शीघ्रास्त्रं समरप्रतिवारणम्।</div><div class="line" style="position:relative;">अवासृजं महाबाहो तेन्तराधिष्ठिताः शराः॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6638"><div class="verse-block verse-justify"><div class="line">रामस्य मम चैवाशु व्योमावृत्य समन्ततः।</div><div class="line" style="position:relative;">न स्म सूर्यः प्रतपति शरजालसमावृतः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6639"><div class="verse-block verse-justify"><div class="line">मातरिश्वा ततस्तस्मिन्मेघरुद्ध इवाभवत्।</div><div class="line" style="position:relative;">ततो वायोः प्रकम्पाच्च सूर्यस्य च गभस्तिभिः॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6640"><div class="verse-block verse-justify"><div class="line">अभिघातप्रभावाच्च पावकः समजायत।</div><div class="line" style="position:relative;">ते शराः स्वसमुत्थेन प्रदीप्ताश्चित्रभानुना॥३५॥</div></div></div>
