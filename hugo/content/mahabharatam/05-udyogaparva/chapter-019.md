<div class="verse-block-wrapper" id="05-udyogaparva-b0609"><div class="verse-block verse-justify"><div class="line">युयुधानस्ततो वीरः सात्वतानां महारथः।</div><div class="line" style="position:relative;">महता चतुरङ्गेण बलेनागाद्युधिष्ठिरम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0610"><div class="verse-block verse-justify"><div class="line">तस्य योधा महावीर्या नानादेशसमागताः।</div><div class="line" style="position:relative;">नानाप्रहरणा वीराः शोभयांचक्रिरे बलम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0611"><div class="verse-block verse-justify"><div class="line">परश्वथैर्भिण्डिपालैः शूलतोमरमुद्गरैः।</div><div class="line" style="position:relative;">परिघैर्यष्टिभिः पाशैः करवालैश्च निर्मलैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0612"><div class="verse-block verse-justify"><div class="line">खङ्गकार्मुकनिर्यूहैः शरैश्च विविधैरपि।</div><div class="line" style="position:relative;">तैलधौतैः प्रकाशद्भिस्तदशोभत वै बलम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0613"><div class="verse-block verse-justify"><div class="line">तस्य मेघप्रकाशस्य सौवर्णौः शोभितस्य च।</div><div class="line" style="position:relative;">बभूव रूपं सैन्यस्य मेघस्येव सविद्युतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0614"><div class="verse-block verse-justify"><div class="line">अक्षौहिणी तु सा सेना तदा यौधिष्ठिरं बलम्।</div><div class="line" style="position:relative;">प्रविश्यान्तर्दधे राजन्सागरं कुनदी यथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0615"><div class="verse-block verse-justify"><div class="line">तथैवाक्षौहिणीं गृह्य चेदीनामृषभो बली।</div><div class="line" style="position:relative;">धृष्टकेतुरुपागच्चत्पाण्डवानमितौजसः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0616"><div class="verse-block verse-justify"><div class="line">मागधश्च जयत्सेनो जारासन्धिर्महाबलः।</div><div class="line" style="position:relative;">अक्षौहिण्यैव सैन्यस्य धर्मराजमुपागमत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0617"><div class="verse-block verse-justify"><div class="line">तथैव पाण्ड्यो राजेन्द्र सागरानूपवासिभिः।</div><div class="line" style="position:relative;">वृतो बहुविधैर्योधैर्युधिष्ठिरमुपागमत्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0618"><div class="verse-block verse-justify"><div class="line">तस्य सैन्यमतीवासीत्तस्मिन्बलसमागमे।</div><div class="line" style="position:relative;">प्रेक्षणीयतरं राजन्सुवेषं बलवत्तदा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0619"><div class="verse-block verse-justify"><div class="line">` केकयाश्च नरव्याघ्राः सोदराः पञ्च पार्थिवाः।</div><div class="line" style="position:relative;">संहर्षयन्तः कौन्तेयोनक्षौहिण्या समागताः ॥'॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0620"><div class="verse-block verse-justify"><div class="line">द्रुपदस्याप्यभूत्सेना नानादेशसमागतैः।</div><div class="line" style="position:relative;">शोभिता पुरुषैः शूरैः पुत्रैश्चास्य महारथैः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0621"><div class="verse-block verse-justify"><div class="line">तथैव राजा मत्स्यानां विराटो वाहिनीपतिः।</div><div class="line" style="position:relative;">पार्वतीयैर्महीपालैः सहितः पाण्डवानयात्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0622"><div class="verse-block verse-justify"><div class="line">इतश्चेतश्च पाण्डूनां समाजग्मुर्महात्मनाम्।</div><div class="line" style="position:relative;">अक्षौहिण्यस्तु सप्तैव विविधध्वजसङ्कुलाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0623"><div class="verse-block verse-justify"><div class="line">युयुत्समानाः कुरुभिः पाण्डवान्समहर्षयन्।</div><div class="line" style="position:relative;">तथैव धार्तराष्ट्रस्य हर्षं समभिर्धयन्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0624"><div class="verse-block verse-justify"><div class="line">भगदत्तो महीपालः सेनामक्षौहिणीं ददौ।</div><div class="line" style="position:relative;">तस्य चीनैः किरातैश्च काञ्चनैरिव संवृतम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0625"><div class="verse-block verse-justify"><div class="line">बभौ बलमनाधृष्यं कर्णिकारवनं यथा।</div><div class="line" style="position:relative;">तथा भूरिश्रवाः शूरः शल्यश्च कुरुनन्दन॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0626"><div class="verse-block verse-justify"><div class="line">दुर्योधनमुपायातावक्षौहिण्या पृथक्पृथक्।</div><div class="line" style="position:relative;">कृतवर्मा च हार्दिक्यो भोजान्धकुकुरैः सह॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0627"><div class="verse-block verse-justify"><div class="line">अक्षौहिण्यैव सेनाया दुर्योधनमुपागमत्।</div><div class="line" style="position:relative;">तस्य तैः पुरुषव्याघ्रैर्वनमालाधरैर्बलम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0628"><div class="verse-block verse-justify"><div class="line">अशोभत यथा मत्तैर्वनं प्रक्रीडितैर्गजैः।</div><div class="line" style="position:relative;">जयद्रथमुखाश्चान्ये सिन्धुसौवीरवासिनः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0629"><div class="verse-block verse-justify"><div class="line">आजग्मुः पृथिवीपालाः कम्पयन्त इवाचलान्।</div><div class="line" style="position:relative;">तेषामक्षौहिणी सेना बहुला विबभौ तदा॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0630"><div class="verse-block verse-justify"><div class="line">विधूयमानो वातेन बहुरूप इवाम्बुदः।</div><div class="line" style="position:relative;">सुदक्षिणश्च काम्भोजो यवनैश्च शकैस्तथा॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0631"><div class="verse-block verse-justify"><div class="line">उपाजगाम कौरव्यमक्षौहिण्या विशांपते।</div><div class="line" style="position:relative;">तस्य सेनासमावायः शलभानामिवाबभौ॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0632"><div class="verse-block verse-justify"><div class="line">स च संप्राप्य कौरव्यं तत्रैवान्तर्दधे तदा।</div><div class="line" style="position:relative;">तथा महिष्मतीवासी नीलो नीलायुधैः सह॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0633"><div class="verse-block verse-justify"><div class="line">महीपालो महावीर्यैर्दक्षिणापथवासिभिः।</div><div class="line" style="position:relative;">आवन्त्यौ च महीपालौ महाबलसुसंवृतौ॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0634"><div class="verse-block verse-justify"><div class="line">पृथगक्षौहिणीभ्यां तावभियातौ सुयोधनम्।</div><div class="line" style="position:relative;">ततस्ततस्तु सर्वेषां भूमिपानां महात्मनाम्॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0635"><div class="verse-block verse-justify"><div class="line">तिस्रोऽन्याः समवर्तन्त वाहिन्यो भरतर्षभ।</div><div class="line" style="position:relative;">एवमेकादशावृत्ताः सेना दुर्योधनस्य ताः॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0636"><div class="verse-block verse-justify"><div class="line">युयुत्समानाः कौन्तेयान्नानाध्वजसमाकुलाः।</div><div class="line" style="position:relative;">न हास्तिनपुरे राजन्नवकाशोऽभवत्तदा॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0637"><div class="verse-block verse-justify"><div class="line">राज्ञां स्वबलमुख्यानां प्राधान्येनापि भारत।</div><div class="line" style="position:relative;">ततः प़ञ्चनदं चैव कृत्स्नं च कुरुजाङ्गलम्॥३०॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b0639"><div class="verse-block verse-justify"><div class="line">वारणं वाटधानं च यामुनश्चैव पर्वतः।</div><div class="line" style="position:relative;">एष देशः सुविस्तीर्णः प्रभूतधनधान्यवान्॥३२॥</div></div></div>
