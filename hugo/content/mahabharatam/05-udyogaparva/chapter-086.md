<div class="verse-block-wrapper" id="05-udyogaparva-b3203"><div class="verse-block verse-justify"><div class="line">धृतराष्ट्र उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3204"><div class="verse-block verse-justify"><div class="line">उपप्लाव्यादिह क्षत्तरुपायातो जनार्दनः।</div><div class="line" style="position:relative;">वृकस्थले निवसति स च प्रातरिहैष्यति॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3205"><div class="verse-block verse-justify"><div class="line">आहुकानामधिपतिः पुरोगः सर्वसात्वताम्।</div><div class="line" style="position:relative;">महामना महावीर्यो महासत्वो जनार्दनः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3206"><div class="verse-block verse-justify"><div class="line">स्फीतस्य वृष्णिराष्ट्रस्य भर्ता गोप्ता च माधवः।</div><div class="line" style="position:relative;">त्रयाणामपि लोकानां भगवान्प्रपितामहः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3207"><div class="verse-block verse-justify"><div class="line">वृष्ण्यन्धकाः सुमनसो यस्य प्रज्ञामुपासते।</div><div class="line" style="position:relative;">आदित्या वसवो रुद्रा यथा बुद्धिं बृहस्पतेः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3208"><div class="verse-block verse-justify"><div class="line">तस्मै पूजां प्रयोक्ष्यामि दाशार्हाय महात्मने।</div><div class="line" style="position:relative;">प्रत्यक्षं तव धर्मज्ञ तां मे कथयतः श्रुणु॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3209"><div class="verse-block verse-justify"><div class="line">एकवर्णैः सुक्लृप्ताङ्गैर्बाह्लिजातैर्हयोत्तमैः।</div><div class="line" style="position:relative;">चतुर्यक्तान्रथांस्तस्मै रौक्मान्दास्यामि षोडश॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3210"><div class="verse-block verse-justify"><div class="line">नित्यप्रभिन्नान्मातङ्गानीषादन्तान्प्रहारिणः।</div><div class="line" style="position:relative;">अष्टानुचरमेकैकमष्टौ दास्यामि कौरव॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3211"><div class="verse-block verse-justify"><div class="line">दासीनामप्रजातानां शुभानां रुक्मवर्चसाम्।</div><div class="line" style="position:relative;">शतमस्मै प्रदास्यामि दासानामपि तावताम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3212"><div class="verse-block verse-justify"><div class="line">आविकं च सुखस्पर्शं पार्वतीयैरुपाहृतम्।</div><div class="line" style="position:relative;">तदप्यस्मै प्रदास्यामि सहस्त्राणि दाशाष्ट च॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3213"><div class="verse-block verse-justify"><div class="line">अजिनानां सहस्राणि चीनदेशोद्भवानि च।</div><div class="line" style="position:relative;">तान्यप्यस्मै प्रदास्यामि यावदर्हति केशवः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3214"><div class="verse-block verse-justify"><div class="line">दिवा रात्रौ च भात्येष सुतेजा विमलो मणिः।</div><div class="line" style="position:relative;">तमप्यस्मै प्रदास्यामि तमर्हति हि केशवः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3215"><div class="verse-block verse-justify"><div class="line">एकेनाभिपतत्यह्ना योजनानि चतुर्दश।</div><div class="line" style="position:relative;">यानमश्वतरीयुक्तं दास्ये तस्मै तदप्यहम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3216"><div class="verse-block verse-justify"><div class="line">यावन्ति वाहनान्यस्य यावन्तः पुरुषाश्च ते।</div><div class="line" style="position:relative;">ततोष्टगुणमप्यस्मै भोज्यं दास्याम्यहं सदा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3217"><div class="verse-block verse-justify"><div class="line">मम पुत्राश्च पौत्रश्च सर्वे दुर्योधनादृते।</div><div class="line" style="position:relative;">प्रत्युद्यास्यन्ति दाशार्हं रथैर्मृष्टैः स्वलङ्कृताः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3218"><div class="verse-block verse-justify"><div class="line">स्वलङ्कृताश्च कल्याण्यः पादैरेव सहस्रशः।</div><div class="line" style="position:relative;">वारमुख्या महाभागं प्रत्युद्यास्यन्ति केशवम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3219"><div class="verse-block verse-justify"><div class="line">नगरादपि याः काश्चिद्गमिष्यन्ति जनार्दनम्।</div><div class="line" style="position:relative;">द्रष्टुं कन्याश्च कल्याण्यस्ताश्च यास्यन्त्यनावृताः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3220"><div class="verse-block verse-justify"><div class="line">सस्त्रीपुरुषबालं च नगरं मधुसूदनम्।</div><div class="line" style="position:relative;">उदीक्षतां महात्मानं भानुमन्तमिव प्रजाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3221"><div class="verse-block verse-justify"><div class="line">महाध्वजपताकाश्च क्रियन्तां सर्वतो दिशः।</div><div class="line" style="position:relative;">जलावसिक्तो विरजाः पन्थास्तस्येति चान्वशात्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3222"><div class="verse-block verse-justify"><div class="line">दुःशासनस्य च गृहं दुर्योधनगृहाद्वरम्।</div><div class="line" style="position:relative;">तदद्य क्रियतां क्षिप्रं सुसंमृष्टमलङ्कृतम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3223"><div class="verse-block verse-justify"><div class="line">एतद्धि रुचिराकारैः प्रासादैरुपशोभितम्।</div><div class="line" style="position:relative;">शिवं च रमणीयं च सर्वर्तु सुमहाधनम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3224"><div class="verse-block verse-justify"><div class="line">सर्वमस्मिन्गृहे रत्नं मम दुर्योधनस्य च।</div><div class="line" style="position:relative;">यद्यदर्हति वार्ष्णेयस्तत्तद्देयमसंशयम्॥२२॥</div></div></div></div>
