<div class="verse-block-wrapper" id="05-udyogaparva-b6072"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6073"><div class="verse-block verse-justify"><div class="line">सुदक्षिणस्तु काम्भोजो रथ एकगुणो मतः।</div><div class="line" style="position:relative;">तवार्थसिद्धिमाकाङ्क्षन्योत्स्यते समरे परैः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6074"><div class="verse-block verse-justify"><div class="line">एतस्य रथसिंहस्य तवार्थे राजसत्तम।</div><div class="line" style="position:relative;">पराक्रमं यथेन्द्रस्य द्रक्ष्यन्ति कुरवो युधि॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6075"><div class="verse-block verse-justify"><div class="line">एतस्य रथवंशे हि तिग्मवेगप्रहारिणः।</div><div class="line" style="position:relative;">काम्भोजानां महाराज शलभानामिवायतिः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6076"><div class="verse-block verse-justify"><div class="line">नीलो माहिष्यतीवासी नीलवर्मा रथस्तव।</div><div class="line" style="position:relative;">रथवंशेन कदनं शत्रूणां वै करिष्यति॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6077"><div class="verse-block verse-justify"><div class="line">कृतवैरः पुरा चैव सहदेवेन मारिष।</div><div class="line" style="position:relative;">योत्स्यते सततं राजंस्तवार्थे कुरुनन्दन॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6078"><div class="verse-block verse-justify"><div class="line">विन्दानुविन्दावावन्त्यौ संमतौ रथसत्तमौ।</div><div class="line" style="position:relative;">कृतिनौ समरे तात दृढवीर्यपराक्रमौ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6079"><div class="verse-block verse-justify"><div class="line">एतौ तौ पुरुषव्याघ्रौ रिपुसैन्यं प्रधक्ष्यतः।</div><div class="line" style="position:relative;">गदाप्रासासिनाराचैस्तोमरैश्च करच्यतैः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6080"><div class="verse-block verse-justify"><div class="line">युद्धाभिकामौ समरे क्रीडन्ताविव यूथपौ।</div><div class="line" style="position:relative;">यूथमध्ये महाराज विचरन्तौ कृतान्तवत्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6081"><div class="verse-block verse-justify"><div class="line">त्रिगर्ता भ्रातरः पञ्च रथोदारा मता मम।</div><div class="line" style="position:relative;">कृतवैराश्च पार्थैस्ते विराटनगरे तदा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6082"><div class="verse-block verse-justify"><div class="line">मकरा इव राजेन्द्र समुद्धततरङ्गिणीम्।</div><div class="line" style="position:relative;">गङ्गां विक्षोभयिष्यन्ति पार्थानां युधि वाहिनी॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6083"><div class="verse-block verse-justify"><div class="line">ते रथाः पञ्च राजेन्द्र येषां सत्यरथो मुखम्।</div><div class="line" style="position:relative;">एते योत्स्यन्ति सङ्ग्रामे संस्मरन्तः पुराकृतमक्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6084"><div class="verse-block verse-justify"><div class="line">व्यलीकं पाण्डवेयेन भीमसेनानुजेन ह।</div><div class="line" style="position:relative;">दिशो विजयता राजञ्श्वेतवाहेन भारत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6085"><div class="verse-block verse-justify"><div class="line">ते हनिष्यन्ति पार्थानां तानासाद्य महारथान्।</div><div class="line" style="position:relative;">वरान्वरान्महेष्वासान्क्षत्रियाणां धुरंधरान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6086"><div class="verse-block verse-justify"><div class="line">लक्ष्णणस्तव पुत्रश्च तथा दुःशासनस्य च।</div><div class="line" style="position:relative;">उभौ तौ पुरुषव्याघ्रौ संग्रामेष्वपलायिनौ॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6087"><div class="verse-block verse-justify"><div class="line">तरुणौ सुकुमारौ च राजपुत्रौ रतस्विनौ।</div><div class="line" style="position:relative;">युद्धानां च विशेषज्ञौ प्रणेतारौ च सर्वशः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6088"><div class="verse-block verse-justify"><div class="line">रथौ तौ कुरुशार्दूल मतौ मे रथसत्तमौ।</div><div class="line" style="position:relative;">क्षत्रधर्मरतौ वीरौ महत्कर्म करिष्यतः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6089"><div class="verse-block verse-justify"><div class="line">दण्डधारो महाराज रथ एको नरर्षभ।</div><div class="line" style="position:relative;">योत्स्यते तव सङ्ग्रामे स्वेन सैन्येन पालितः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6090"><div class="verse-block verse-justify"><div class="line">बृहद्बलस्तथा राजा कौसल्यो रथसत्तमः।</div><div class="line" style="position:relative;">रथो मम मतस्तात महावेगपराक्रमः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6091"><div class="verse-block verse-justify"><div class="line">एष योत्स्यति सङ््ग्रामे स्वान्बन्धून्संप्रहर्षयन्।</div><div class="line" style="position:relative;">उग्रायुधो महेष्वासो धार्तराष्ट्रहिते रतः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6092"><div class="verse-block verse-justify"><div class="line">कृपः शारद्वतो रजन्रथयूथपयूथपः।</div><div class="line" style="position:relative;">प्रियान्प्राणान्परित्यज्य प्रधक्ष्यति रिपूंस्तव॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6093"><div class="verse-block verse-justify"><div class="line">गौतमस्य महर्षेर्य आचार्यस्य शरद्वतः।</div><div class="line" style="position:relative;">कार्तिकेय इवाजयः शरस्तम्बात्सुतोऽभवत्॥२२॥</div></div></div>
