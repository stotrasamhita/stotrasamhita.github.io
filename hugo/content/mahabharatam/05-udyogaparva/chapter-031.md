<div class="verse-block-wrapper" id="05-udyogaparva-b0985"><div class="verse-block verse-justify"><div class="line">बलं जिज्ञासमानस्य आचक्षीथा यथातथम्।</div><div class="line" style="position:relative;">अथ मन्त्र मन्त्रयित्वा याथातथ्येन हृष्टवत्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0986"><div class="verse-block verse-justify"><div class="line">गावल्गणे कुरून्गत्वा धृतराष्ट्रं महाबलम्।</div><div class="line" style="position:relative;">अभिवाद्योपसंगृह्य ततः पृच्छेरनामयम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0987"><div class="verse-block verse-justify"><div class="line">ब्रूयाश्चैनं त्वमासीनं कुरुभिः परिवारितम्।</div><div class="line" style="position:relative;">तवैव राजन्वीर्येण सुखं जीवन्ति पाण्डवाः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0988"><div class="verse-block verse-justify"><div class="line">तव प्रसादाद्बालास्ते प्राप्ता राज्यमरिन्दम।</div><div class="line" style="position:relative;">राज्ये तान्स्थापयित्वाऽग्रे नोपेक्षस्व विनश्यतः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0989"><div class="verse-block verse-justify"><div class="line">सर्वमप्येतदेकस्य नालं सञ्जय कस्यचित्।</div><div class="line" style="position:relative;">तात संहत्य जीवामो द्विषतां मा वशं गमः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0990"><div class="verse-block verse-justify"><div class="line">तथा भीष्मं शान्तनवं भारतानां पितामहम्।</div><div class="line" style="position:relative;">शिरसाऽभिवदेथास्त्वं मम नाम प्रकीर्तयन्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0991"><div class="verse-block verse-justify"><div class="line">अभिवाद्य च वक्तव्यस्ततोऽस्माकं पितामहः।</div><div class="line" style="position:relative;">भवता शन्तनोर्वंशो निमग्नः पुनरुद्धृतः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0992"><div class="verse-block verse-justify"><div class="line">स त्वं कुरु तथा तात स्वमतेन पितामह।</div><div class="line" style="position:relative;">यथा जीवन्ति ते पौत्राः प्रीतिमन्तः परस्परम्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0993"><div class="verse-block verse-justify"><div class="line">तथैव विदुरं ब्रूयाः कुरूणां मन्त्रधारिणम्।</div><div class="line" style="position:relative;">अयुद्धं सौम्य भाषस्व हितकामो युधिष्ठिरे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0994"><div class="verse-block verse-justify"><div class="line">अथ दुर्योधनं ब्रूया राजपुत्रममर्षणम्।</div><div class="line" style="position:relative;">मध्ये कुरूणामासीनमनुनीय पुनः पुनः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0995"><div class="verse-block verse-justify"><div class="line">अपापां यदुपैक्षस्त्वं कृष्णामेतां सभागताम्।</div><div class="line" style="position:relative;">तद्दुःखमतितिक्षाम मा वधीष्म कुरूनिति॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0996"><div class="verse-block verse-justify"><div class="line">एवं पूर्वापरान्क्लेशानतितिक्षन्त पाण्डवाः।</div><div class="line" style="position:relative;">बलीयांसोऽपि सन्तो यत्तत्सर्वं कुरवो विदुः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0997"><div class="verse-block verse-justify"><div class="line">यन्नः प्रव्राजयेः सौम्य अजिनैः प्रतिवासितान्।</div><div class="line" style="position:relative;">तद्दुःखमतितिक्षाम मा वधीषय कुरूनिति॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0998"><div class="verse-block verse-justify"><div class="line">यत्कुन्तीं समतिक्रम्य कृष्णां केशेष्वधर्षयत्।</div><div class="line" style="position:relative;">दुःशासनस्तेऽनुमते तच्चास्माभिरुपेक्षितम्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b0999"><div class="verse-block verse-justify"><div class="line">अथोचितं स्वकं भागं लभेमहि परन्तप।</div><div class="line" style="position:relative;">निवर्तय परद्रव्याद्बुद्धिं गृद्धां नरर्षभ॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b1000"><div class="verse-block verse-justify"><div class="line">शान्तिरेवं भवेद्राजन्प्रीतिश्चैव परस्परम्।</div><div class="line" style="position:relative;">राज्यैकदेशमपि नः प्रयच्छ शममिच्छताम्॥१९॥</div></div></div>
