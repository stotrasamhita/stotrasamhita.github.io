<div class="verse-block-wrapper" id="05-udyogaparva-b5673"><div class="verse-block verse-justify"><div class="line">यः किंपुरुषसिंहस्य गन्धमादनवासिनः।</div><div class="line" style="position:relative;">कृत्स्नं शिष्यो धनुर्वेदं चतुष्पादमवाप्तवान्॥४॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b5675"><div class="verse-block verse-justify"><div class="line">त्रीण्येवैतानि दिव्यानि धनूंषि दिविचारिणाम्।</div><div class="line">वारुणं गाण्डिवं तत्र माहेन्द्रं विजयं धनुः।</div><div class="line" style="position:relative;">शार्ङ्ग तु वैष्णवं प्राहुर्दिव्यं तेजोमयं धनुः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5676"><div class="verse-block verse-justify"><div class="line">धारयामास तत्कृष्णः परसेनाभयावहम्।</div><div class="line" style="position:relative;">गाण्डीवं पावकाल्लेभे खाण्डवे पाकशासनिः॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5677"><div class="verse-block verse-justify"><div class="line">द्रुमाद्रुक्मी महातेजा विजयं प्रत्यपद्यत।</div><div class="line" style="position:relative;">संछिद्य मौरवान्पाशान्निहत्य मुरमोजसा॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5678"><div class="verse-block verse-justify"><div class="line">निर्जित्य नरकं भौममाहृत्य मणिकुण्डले।</div><div class="line" style="position:relative;">षोडश स्त्रीसहस्राणि रत्नानि विविधानि च॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5679"><div class="verse-block verse-justify"><div class="line">प्रतिपेदे हृषीकेशः शार्ङ्गं च धनुरुत्तमम्।</div><div class="line" style="position:relative;">रुक्मी तु विजयं लब्ध्वा धनुर्मेघनिभस्वनम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5680"><div class="verse-block verse-justify"><div class="line">विभीषयन्निव जगत्पाण्डवानभ्यवर्तत।</div><div class="line" style="position:relative;">नामृष्यत पुरा योऽसौ स्वबाहुलगर्वितः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5681"><div class="verse-block verse-justify"><div class="line">रुक्मिण्या हरणं वीरो वासुदेवेन धीमता।</div><div class="line" style="position:relative;">कृत्वा प्रतिज्ञां नाहत्वा निवर्तिष्ये जनार्दनम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5682"><div class="verse-block verse-justify"><div class="line">ततोऽन्वधावद्वार्ष्णेयं सर्वशस्त्रभृतां वरः।</div><div class="line" style="position:relative;">सेनया चतुरङ्गिण्या महत्या दूरपातया॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5683"><div class="verse-block verse-justify"><div class="line">विचित्रायुधवर्मिण्या गङ्ग्येव प्रवृद्धया।</div><div class="line" style="position:relative;">स समासाद्य वार्ष्णेयं योगानामीश्वरं प्रभुम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5684"><div class="verse-block verse-justify"><div class="line">व्यंसितो व्रीडितो राजन्नाजगाम स कुण्डिनम्।</div><div class="line" style="position:relative;">यत्रैव कृष्णेन रणे निर्जितः परवीरहा॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5685"><div class="verse-block verse-justify"><div class="line">तत्र भोजकटं नाम कृतं नगरमुत्तमम्।</div><div class="line" style="position:relative;">सैन्येन महता तेन प्रभूतगजवाजिना॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b5689"><div class="verse-block verse-justify"><div class="line">युधिष्ठिरस्तु तं राजा प्रत्युद्गम्याभ्यपूजयत्।</div><div class="line" style="position:relative;">स पूजितः पाण्डुपुत्रैर्यथान्यायं सुसंस्तुतः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5690"><div class="verse-block verse-justify"><div class="line">प्रतिगृह्य तु तान्सर्वान्विश्रान्तः सहसैनिकः।</div><div class="line" style="position:relative;">उवाच मध्ये वीराणां कुन्तीपुत्रं धनञ्जयम्॥२१॥</div></div></div>
