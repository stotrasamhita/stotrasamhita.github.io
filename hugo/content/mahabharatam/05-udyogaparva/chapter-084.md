<div class="verse-block-wrapper" id="05-udyogaparva-b3154"><div class="verse-block verse-justify"><div class="line">जनमेजय उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3156"><div class="verse-block verse-justify"><div class="line">तस्य प्रयाणे यान्यसन्निमित्तानि महात्मनः।</div><div class="line" style="position:relative;">तानि मे श्रृणु सर्वाणि दैवान्यौत्पातिकानि च॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3157"><div class="verse-block verse-justify"><div class="line">अनभ्रेऽशनिनिर्घोषः सविद्युत्समजायत।</div><div class="line" style="position:relative;">अन्वगेव च पर्जन्यः प्रावर्षद्विघने भृशम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3158"><div class="verse-block verse-justify"><div class="line">प्रत्यगूहुर्महानद्यः प्राङ्मुखाः सिन्धुसप्तमाः।</div><div class="line" style="position:relative;">विपरिता दिशः सर्वा न प्राज्ञायत किंचन॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3159"><div class="verse-block verse-justify"><div class="line">प्राज्वलन्नग्नयो राजन्पृथिवी समकम्पत।</div><div class="line" style="position:relative;">उदपानाश्च कुम्भाश्च प्रासिञ्चञ्शतशो जलम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3160"><div class="verse-block verse-justify"><div class="line">तमःसंवृतमप्यासीत्सर्वं जगदिदं तथा।</div><div class="line" style="position:relative;">न दिशो नादिशो राजन्प्रज्ञायन्तेस्म रेणुना॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3161"><div class="verse-block verse-justify"><div class="line">प्रादुरासीन्महाञ्छब्दः खेशरीरमदृश्यत।</div><div class="line" style="position:relative;">सर्वेषु राजन्देशेषु तदद्भुतमिवाभवत्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3162"><div class="verse-block verse-justify"><div class="line">प्रामथ्नाद्धास्तिनपुरं वातो दक्षिणपश्चिमः।</div><div class="line" style="position:relative;">आरुजन्गणशो वृक्षान्परुषोऽशनिनिःस्वनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3163"><div class="verse-block verse-justify"><div class="line">यत्रयत्र च वार्ष्णेयो वर्तते पथि भारत।</div><div class="line" style="position:relative;">तत्रतत्र सुखो वायुः सर्वं चासीत्प्रदक्षिणम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3164"><div class="verse-block verse-justify"><div class="line">ववर्ष पुष्पवर्षं च कमलानि च भूरिशः।</div><div class="line" style="position:relative;">समश्च पन्था निर्दुःखो व्यपेतकुशकण्टकः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3165"><div class="verse-block verse-justify"><div class="line">संस्तुतो ब्राह्मणैर्गीर्भिस्तत्रतत्र सहस्रशः।</div><div class="line" style="position:relative;">अर्च्यते मधुपर्कैश्च वसुभिश्च वसुप्रदः॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3166"><div class="verse-block verse-justify"><div class="line">तं किरन्ति महात्मानं वन्यैः पुष्पैः सुगन्धिभिः।</div><div class="line" style="position:relative;">स्त्रियः पथि समागम्य सर्वभूतहिते रतम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3167"><div class="verse-block verse-justify"><div class="line">स शालिभवनं रम्यं सर्वसस्यसमाचितम्।</div><div class="line" style="position:relative;">सुख परमधर्मिष्ठमभ्यगाद्भरतर्षभ॥१४॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3169"><div class="verse-block verse-justify"><div class="line">नित्यं हृष्टाः सुमनसो भारतैरभिरक्षिताः।</div><div class="line" style="position:relative;">नोद्विग्नाः परचक्राणां व्यसनानामकोविदाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3170"><div class="verse-block verse-justify"><div class="line">उपप्लाव्यादथागम्य जनाः पुरनिवासिनः।</div><div class="line" style="position:relative;">यथ्यतिष्ठन्त सहिता विष्वक्सेनदिदृक्षया॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3171"><div class="verse-block verse-justify"><div class="line">ते तु सर्वे समायान्तमग्निमिद्धमिव प्रभुम्।</div><div class="line" style="position:relative;">अर्चयामासुरर्चार्हं देशातिथिमुपस्थितम्॥१८॥</div></div></div>
