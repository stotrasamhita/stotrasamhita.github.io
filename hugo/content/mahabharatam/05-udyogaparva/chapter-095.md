<div class="verse-block-wrapper" id="05-udyogaparva-b3590"><div class="verse-block verse-justify"><div class="line">अशिष्टा गतमर्यादा लोभेन हृतचेतसः।</div><div class="line" style="position:relative;">स्वेषु बन्धुषु मुख्येषु तद्वेत्थ पुरुषर्षभ॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3591"><div class="verse-block verse-justify"><div class="line">सेयमापन्महाघोरा कुरुष्वेव समुत्थिता।</div><div class="line" style="position:relative;">उपेक्ष्यमाणा कौरव्य पृथिवीं घातयिष्यति॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3592"><div class="verse-block verse-justify"><div class="line">शक्या चेयं शमयितुं त्वं चेदिच्छसि भारत।</div><div class="line" style="position:relative;">न दुष्करो ह्यत्र शमो मतो मे भरतर्षभ॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3593"><div class="verse-block verse-justify"><div class="line">त्वय्यधीनः शमो राजन्मयि चैव विशांपते।</div><div class="line" style="position:relative;">पुत्रान्स्थापय कौरव्य स्थापयिष्याम्यहं परान्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3594"><div class="verse-block verse-justify"><div class="line">आज्ञा तव हि राजेन्द्र कार्या पुत्रैः सहान्वयैः।</div><div class="line" style="position:relative;">हितं बलवदप्येषां तिष्ठतां तव शासने॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3595"><div class="verse-block verse-justify"><div class="line">तव चैव हितं राजन्पाण्डवानामथो हितम्।</div><div class="line" style="position:relative;">शमे प्रयतमानस्य तव शासनकाङ्क्षिणः॥१६॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3598"><div class="verse-block verse-justify"><div class="line">न हि त्वां पाण्डवैर्जेतुं रक्ष्यमाणं महात्मभिः।</div><div class="line" style="position:relative;">इन्द्रोपि देवैः सहितः प्रसहेत कुतो नृपाः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3599"><div class="verse-block verse-justify"><div class="line">यत्र भीष्मश्च द्रोणश्च कृपः कर्णो विविंशतिः।</div><div class="line" style="position:relative;">अश्वत्थामा विकर्णश्च सोमदत्तोऽथ बाह्लिकः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3600"><div class="verse-block verse-justify"><div class="line">सैन्धवश्च कलिङ्गश्च काम्भोजश्च सुदक्षिणः।</div><div class="line" style="position:relative;">युधिष्ठिरो भीमसेनः सव्यसाची यमौ तथा॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3601"><div class="verse-block verse-justify"><div class="line">सात्यकिश्च महातेजा युयुत्सुश्च महारथः।</div><div class="line" style="position:relative;">को नु तान्विपरीतात्मा युद्ध्येत भरतर्षभ॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3602"><div class="verse-block verse-justify"><div class="line">लोकस्येश्वरतां भूयः शत्रुभिश्चाप्यधृष्यताम्।</div><div class="line" style="position:relative;">प्राप्स्यसि त्वममित्रघ्न सहितः कुरुपाण्डवैः॥२३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3604"><div class="verse-block verse-justify"><div class="line">स त्वं पुत्रैश्च पौत्रैश्च पितृभिर्भ्रातृभिस्तथा।</div><div class="line" style="position:relative;">सुहृद्भिः सर्वतो गुप्तः सुखं शक्ष्यसि जीवितुं॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3605"><div class="verse-block verse-justify"><div class="line">एतानेव पुरोधाय यत्कृत्य च यथा पुरा।</div><div class="line" style="position:relative;">अखिलां भोक्ष्यसे सर्वां पृथिवीं पृथिवीपते॥२६॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3607"><div class="verse-block verse-justify"><div class="line">तैरेवोपार्जितां भूमिं भोक्ष्यसे च परन्तप।</div><div class="line" style="position:relative;">यदि संपत्स्यसे पुत्रैः सहामात्यैर्नराधिप॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3608"><div class="verse-block verse-justify"><div class="line">संयुगे वै महाराज दृश्यते सुमहान्क्षयः।</div><div class="line" style="position:relative;">क्षये चोभयतो राजन्कं धर्ममनुपश्यसि॥२९॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3610"><div class="verse-block verse-justify"><div class="line">शूराश्च हि कृतास्त्राश्च सर्वे युद्धाभिकाङ्क्षिणः।</div><div class="line" style="position:relative;">पाण्डवास्तावकाश्चैव तान्रक्ष महतो भयात्॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3611"><div class="verse-block verse-justify"><div class="line">न पश्येम कुरून्सर्वान्पाण्डवांश्चैव संयुगे।</div><div class="line" style="position:relative;">क्षीणानुभयतः शूरान्राथिनो रथिभिर्हतान्॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3612"><div class="verse-block verse-justify"><div class="line">समवेताः पृथिव्यां हि राजानो राजसत्तम।</div><div class="line" style="position:relative;">अमर्षवशमापन्ना नाशयेयुरिमाः प्रजाः॥३३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3616"><div class="verse-block verse-justify"><div class="line">सुवाससः स्रग्विणश्च सत्कृता भरतर्षभ।</div><div class="line" style="position:relative;">अमर्षं च निराकृत्य वैराणि च परन्तप॥३६॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3620"><div class="verse-block verse-justify"><div class="line">आहुस्त्वां पाण्डवा राजन्नभिवाद्य प्रसाद्य च।</div><div class="line" style="position:relative;">भवतः शासनाद्दुःखसमुभूतं सहानुगैः॥४०॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3622"><div class="verse-block verse-justify"><div class="line">स्थाता नः समये तस्मिन्पितेति कृतनिश्चयाः।</div><div class="line" style="position:relative;">नाहास्म समयं तात तच्च नो ब्राह्मणा विदुः॥४२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3623"><div class="verse-block verse-justify"><div class="line">तस्मिन्नः समये तिष्ठ स्थितानां भरतर्षभ।</div><div class="line">नित्यं संक्लेशिता राजन्स्वराज्यांशं लभेमहि।</div><div class="line" style="position:relative;">॥४३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3624"><div class="verse-block verse-justify"><div class="line">त्वं धर्ममर्थं संजानन्सम्यङ्वस्त्रातुमर्हसि ॥गुरुत्वं भवति प्रेक्ष्य बहून्क्लेशांस्तितिक्ष्महे।</div><div class="line" style="position:relative;">॥४४॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3626"><div class="verse-block verse-justify"><div class="line">वर्तामहे त्वयि च तां त्वं च वर्तस्व नस्तथा ॥पित्रा स्थापयितव्या हि वयमुत्पथमास्थिताः।</div><div class="line" style="position:relative;">॥४६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3627"><div class="verse-block verse-justify"><div class="line">संस्थापय पथिप्वस्मांस्तिष्ठ धर्मे सुवर्त्मनि ॥आहुश्चेमां परिषदं पुत्रास्ते भरतर्षभ।</div><div class="line" style="position:relative;">॥४७॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3631"><div class="verse-block verse-justify"><div class="line">ये धर्ममनुपश्यन्तस्तूष्णीं ध्यायन्त आसते।</div><div class="line" style="position:relative;">ते सत्यमाहुर्धर्म्यं च न्याय्यं च भरतर्षभ॥५०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3632"><div class="verse-block verse-justify"><div class="line">शक्यं किमन्यद्वक्तुं ते दानादन्यञ्जनेश्वर।</div><div class="line" style="position:relative;">ब्रुवन्तु ते महीपालाः सभायां ये समासते॥५१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3633"><div class="verse-block verse-justify"><div class="line">धर्मार्थौ संप्रधार्यैव यदि सत्यं ब्रवीम्यहम्।</div><div class="line" style="position:relative;">प्रमुञ्चोमान्मृत्युपाशात्क्षत्रियान्पुरुषर्षभ॥५२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3634"><div class="verse-block verse-justify"><div class="line">प्रशाम्य भरतश्रेष्ठ मा मन्युवशमन्वगाः।</div><div class="line">पित्र्यं तेभ्यः प्रदायांशं पाण्डवेभ्यो यथोचितम्।</div><div class="line" style="position:relative;">ततः सपुत्रः सिद्धार्थो भुङ्क्ष भोगान्परन्तप॥५३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3635"><div class="verse-block verse-justify"><div class="line">अजातशत्रुं जानीषे स्थितं धर्मे सतां सदा।</div><div class="line" style="position:relative;">सपुत्रे त्वयि वृत्तिं च वर्तते यां नराधिप॥५४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3636"><div class="verse-block verse-justify"><div class="line">दाहितश्च निरस्तश्च त्वामेवोपाश्रितः पुनः।</div><div class="line" style="position:relative;">इन्द्रप्रस्थं त्वयैवासौ सपुत्रेण विवासितः॥५५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3637"><div class="verse-block verse-justify"><div class="line">स तत्र विवसन्सर्वान्वशमानीय पार्थिवान्।</div><div class="line" style="position:relative;">त्वन्मुखानकरोद्राजन्न च त्वामत्यवर्तत॥५६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3638"><div class="verse-block verse-justify"><div class="line">तस्यैवं वर्तमानस्य सौबलेन जिहीर्षता।</div><div class="line" style="position:relative;">राष्ट्राणि धनधान्यं च प्रयुक्तः परमोपधिः॥५७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3639"><div class="verse-block verse-justify"><div class="line">स तामवस्थां संप्राप्य कृष्णां प्रेक्ष्य सभां गताम्।</div><div class="line" style="position:relative;">क्षत्रधर्मादमेयात्मा नाकम्पत युधिष्ठिरः॥५८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3640"><div class="verse-block verse-justify"><div class="line">अहं तु तव तेषां च श्रेय इच्छामि भारत।</div><div class="line" style="position:relative;">धर्मादर्थात्सुखाच्चैव मा राजन्नीनशः प्रजाः॥५९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3641"><div class="verse-block verse-justify"><div class="line">अनर्थमर्थं मन्वानोऽप्यर्थं चानर्थमात्मनः।</div><div class="line" style="position:relative;">लोभेऽतिप्रसृतान्पुत्रान्निगृह्णीष्व विशांपते॥६०॥</div></div></div>
