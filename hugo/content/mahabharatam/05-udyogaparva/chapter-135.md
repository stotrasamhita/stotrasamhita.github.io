<div class="verse-block-wrapper" id="05-udyogaparva-b4874"><div class="verse-block verse-justify"><div class="line">अहो क्षत्रसमाचारो यत्र मामितरं यथा।</div><div class="line" style="position:relative;">नियोजयसि युद्धाय परमातेव मां तथा॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4875"><div class="verse-block verse-justify"><div class="line">ईदृशं वचनं ब्रूयाद्भवती पुत्रमेकजम्।</div><div class="line" style="position:relative;">किं नु ते मामपश्यन्त्याः पृथिव्या अपि सर्वया॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4876"><div class="verse-block verse-justify"><div class="line">किमाभरणकृत्येन किं भोगैर्जीवितेन वा।</div><div class="line">मयि वा सङ्गरहते प्रियपुत्रे विशेषतः ॥मातोवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4877"><div class="verse-block verse-justify"><div class="line">सर्वावस्था हि विदुषां तात धर्मार्थकारणात्।</div><div class="line" style="position:relative;">तावेवाभिसमीक्ष्याहं सञ्जय त्वामचूजुदम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4878"><div class="verse-block verse-justify"><div class="line">स समीक्ष्य क्रमोपेतो मुख्यः कालोऽयमागतः।</div><div class="line" style="position:relative;">अस्मिंश्चेदागते काले कार्यं न प्रतिपद्यसे॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4879"><div class="verse-block verse-justify"><div class="line">असंभावितरूपस्त्वमानृशंस्यं करिष्यसि।</div><div class="line" style="position:relative;">तं त्वामयशसा स्पृष्टं न ब्रूयां यदि सञ्जय॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4880"><div class="verse-block verse-justify"><div class="line">खरीवात्सल्यमाहुस्तन्निःसामर्थ्यमहेतुकम्।</div><div class="line" style="position:relative;">सद्भिर्विगर्हितं मार्गं त्यज मूर्खनिषेवितम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4881"><div class="verse-block verse-justify"><div class="line">अविद्या वै महत्यस्ति यामिमां संश्रिताः प्रजाः।</div><div class="line" style="position:relative;">तव स्याद्यदि सद्वृत्तं तेन मे त्वं प्रियो भवेः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4882"><div class="verse-block verse-justify"><div class="line">धर्मार्थगुणयुक्तेन नेतरेण कथञ्चन।</div><div class="line" style="position:relative;">दैवमानुषयुक्तेन सद्भिराचरितेन च॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4883"><div class="verse-block verse-justify"><div class="line">यो ह्येवमविनीतेन रमते पुत्रनप्तृणा।</div><div class="line" style="position:relative;">अनुत्थानवता चापि दुर्विनीतेन दुर्धिया॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4884"><div class="verse-block verse-justify"><div class="line">रमते यस्तु पुत्रेण मोघं तस्य प्रजाफलम्।</div><div class="line" style="position:relative;">अकुर्वन्तो हि कर्माणि कुर्वन्तो निन्दितानि च॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4885"><div class="verse-block verse-justify"><div class="line">सुखं नैवेह नामुत्र लभन्ते पुरुषाधमाः।</div><div class="line">युद्धाय क्षत्रियः सृष्टः सञ्जयेह जयाय च।</div><div class="line" style="position:relative;">`क्रूराय कर्मणे नित्यं प्रजानां परिपालने ॥'॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4886"><div class="verse-block verse-justify"><div class="line">जयन्वा वध्यमानो वा प्राप्नोतीन्द्रसलोकताम्।</div><div class="line">न शक्रभवने पुण्ये दिवि तद्विद्यते सुखम्।</div><div class="line" style="position:relative;">यदमित्रान्वशे कृत्वा क्षत्रियः सुखमेधते॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4887"><div class="verse-block verse-justify"><div class="line">मन्युना दह्यमानेह पुरुषेण मनस्विना।</div><div class="line" style="position:relative;">निकृतेनेह बहुशः शत्रून्प्रति जिगीषया॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4888"><div class="verse-block verse-justify"><div class="line">आत्मानं वा परित्यज्य शत्रुं वा विनिपात्य च।</div><div class="line">प्राप्यते नेह शान्तिर्हि नित्यमेव तु सञ्जय।</div><div class="line" style="position:relative;">अतोन्येन प्रकारेण शान्तिरस्य कुतो भवेत्॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4889"><div class="verse-block verse-justify"><div class="line">इह प्राज्ञो हि पुरुषः स्वल्पमप्रियमिच्छति।</div><div class="line" style="position:relative;">यस्य स्वल्पं प्रियं लोके ध्रुवं तस्याल्पमप्रियम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4890"><div class="verse-block verse-justify"><div class="line">प्रियाभावाच्च पुरुषो नैव प्राप्नोति शोभनम्।</div><div class="line">ध्रुवं चाभावमभ्येति गत्वा गङ्गेव सागरम् ॥पुत्र उवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4891"><div class="verse-block verse-justify"><div class="line">नेयं मतिस्त्वया वाच्या मातः पुत्रे विशेषतः।</div><div class="line">कारुण्यमेव पश्य त्वं भूत्वेह जडमूकवत् ॥मातोवाच।</div><div class="line" style="position:relative;">॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4892"><div class="verse-block verse-justify"><div class="line">अतो मे भूयसी नन्दिर्यदेवमनुपश्यसि।</div><div class="line" style="position:relative;">चोद्यं मां चोदयस्येतद्भृशं वै चोदयामि ते॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4893"><div class="verse-block verse-justify"><div class="line">अथ त्वां पूजयिष्यामि हत्वा वै सर्वसैन्धवान्।</div><div class="line">अहं पश्यामि विजयं कृच्छ्रभाविनमेव ते ॥पुत्र उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4894"><div class="verse-block verse-justify"><div class="line">अकोशस्यासहायस्य कुतः सिद्धिर्जयो मम।</div><div class="line" style="position:relative;">इत्यवस्थां विदित्वैतामात्मनात्मनि दारुणाम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4895"><div class="verse-block verse-justify"><div class="line">राज्याद्भावो निवृत्तो मे त्रिदिवादिव दृष्कृतः।</div><div class="line" style="position:relative;">ईदृशं भवती कंचिदुपायमनुपश्यति॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4896"><div class="verse-block verse-justify"><div class="line">तन्मे परिणतप्रज्ञे सम्यक्प्रब्रूहि पृच्छते।</div><div class="line">करिष्यामि हि तत्सर्वं यथावदनुशासनम् ॥मातोवाच।</div><div class="line" style="position:relative;">॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4897"><div class="verse-block verse-justify"><div class="line">पुत्र नात्मावमन्तव्यः पूर्वाभिरसमृद्धिभिः।</div><div class="line">अभूत्वा हि भवन्त्यर्था भूत्वा नश्यन्ति चापरे।</div><div class="line" style="position:relative;">अमर्षेणैव चाप्यर्था नारब्धव्याः सुबालिशैः॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4898"><div class="verse-block verse-justify"><div class="line">सर्वेषां कर्मणां तात फले नित्यमनित्यता।</div><div class="line" style="position:relative;">अनित्यमिति जानन्तो न भवन्ति भवन्ति च॥२७॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b4900"><div class="verse-block verse-justify"><div class="line">अथ द्वैगुण्यमीहायां फलं भवति वा नवा।</div><div class="line" style="position:relative;">यस्य प्रागेव विदिता सर्वार्थानामनित्यता॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4901"><div class="verse-block verse-justify"><div class="line">नुदेद्वृद्धिसमृद्धी स प्रतिकूले नृपात्मज।</div><div class="line" style="position:relative;">उत्थातव्यं जागृतव्यं योक्तव्यं भूतिकर्मसु॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4902"><div class="verse-block verse-justify"><div class="line">भविष्यतीत्येव मनः कृत्वा सततमव्यथैः।</div><div class="line" style="position:relative;">मङ्गलानि पुरस्कृत्य ब्राह्मणांश्चेश्वरैः सह॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4903"><div class="verse-block verse-justify"><div class="line">प्राज्ञस्य नृपतेराशु वृद्धिर्भवति पुत्रक।</div><div class="line" style="position:relative;">अभिवर्तति लक्ष्मीस्तं प्राचीमिव दिवाकरः॥३२॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b4906"><div class="verse-block verse-justify"><div class="line">स्पर्धिनश्चैव ये केचित्तान्युक्त उपधारय।</div><div class="line">एतेन त्वं प्रकारेण महतो भेत्स्यसे गणान्।</div><div class="line" style="position:relative;">महावेग इवोद्धूतो मातरिश्वा बलाहकान्॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b4907"><div class="verse-block verse-justify"><div class="line">तेषामग्रप्रदायी स्याः कल्पोत्थायी प्रियंवदः।</div><div class="line" style="position:relative;">ते त्वां प्रियं करिष्यन्ति पुरोधास्यन्ति च ध्रुवम्॥३६॥</div></div></div>
