<div class="verse-block-wrapper" id="05-udyogaparva-b2579"><div class="verse-block verse-justify"><div class="line">शकुनीनामिहार्थाय पाशं भूमावयोजयत्।</div><div class="line" style="position:relative;">कश्चिच्छाकुनिकस्तात पूर्वेषामिति शुश्रुम्॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2580"><div class="verse-block verse-justify"><div class="line">तस्मिंस्तौ शकुनौ बद्धौ युगपत्सहचारिणौ।</div><div class="line" style="position:relative;">तावुपादाय तं पाशं जग्मतुः खचरावुभौ॥३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b2582"><div class="verse-block verse-justify"><div class="line">तथा तमनुधावन्तमृगयुं शकुनार्थिनम्।</div><div class="line" style="position:relative;">आश्रमस्थो मुनिः कश्चिद्ददर्शाथ कृताह्निकः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2583"><div class="verse-block verse-justify"><div class="line">तावन्तरिक्षगौ शीघ्रमनुयान्तं महीचरम्।</div><div class="line" style="position:relative;">श्लोकेनानेन कौरव्य पप्च्छ स मुनिस्तदा॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2584"><div class="verse-block verse-justify"><div class="line">विचित्रमिदमाश्चर्यं मृगहन्प्रतिभाति मे।</div><div class="line">प्लवमानौ हि खचरौ पदातिरनुधावसि ॥शाकुनिक उवाच।</div><div class="line" style="position:relative;">॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2585"><div class="verse-block verse-justify"><div class="line">पाशमेकमुभावेतौ सहितौ हरतो मम।</div><div class="line">यत्र वै विवदिष्येते तत्र मे वशमेष्यतः ॥विदुर उवाच।</div><div class="line" style="position:relative;">॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2586"><div class="verse-block verse-justify"><div class="line">तौ विवादमनुप्राप्तौ शकुनौ मृत्युसन्धितौ।</div><div class="line" style="position:relative;">विगृह्य च स्रुदुर्बुद्धी पृथिव्यां संनिपेततुः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2587"><div class="verse-block verse-justify"><div class="line">तौ युध्यमानौ संरब्धौ मृत्युपाशवशानुगौ।</div><div class="line" style="position:relative;">उपसृत्य प्रमत्तौ तौ जग्राह मृगहा तदा॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2588"><div class="verse-block verse-justify"><div class="line">एवं ये ज्ञातयोऽर्थेषु मिथो गच्छन्ति विग्रहम्।</div><div class="line" style="position:relative;">ते मृत्युवशमायान्ति शकुनाविव विग्रहात्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2589"><div class="verse-block verse-justify"><div class="line">संभोजनं संकथनं संप्रश्नोऽथ समागमः।</div><div class="line" style="position:relative;">एतानि ज्ञातिकार्याणि न विरोधः कदाचन॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2590"><div class="verse-block verse-justify"><div class="line">ये स्म काले सुमनसः सर्वे वृद्धानुपासते।</div><div class="line" style="position:relative;">सिंहगुप्तमिवारण्यमप्रधृष्या भवन्ति ते॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2591"><div class="verse-block verse-justify"><div class="line">येऽर्थं सन्ततमासाद्य दीना इव समासते।</div><div class="line" style="position:relative;">श्रियं ते संप्रयच्छन्ति द्विषद्भ्यो भरतर्षभ॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2592"><div class="verse-block verse-justify"><div class="line">धूमायन्ते व्यपेतानि ज्वलन्ति सहितानि च।</div><div class="line" style="position:relative;">धृतराष्ट्रोल्मुकानीव ज्ञातयो भरतर्षभ॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2593"><div class="verse-block verse-justify"><div class="line">इदमन्यत्प्रवक्ष्यामि यथा दृष्टं गिरौ मया।</div><div class="line" style="position:relative;">श्रुत्वा तदपि कौरव्य यथा श्रेयस्तथा कुरु॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2594"><div class="verse-block verse-justify"><div class="line">वयं किरातैः सहिता गच्छामो गिरिमुत्तरम्।</div><div class="line" style="position:relative;">ब्राह्मणैर्देवकल्पैश्च विद्याजम्भकवार्तिकैः॥१७॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b2598"><div class="verse-block verse-justify"><div class="line">अचक्षुर्लभते चक्षुर्वृद्धो भवति वै युवा।</div><div class="line" style="position:relative;">इति ते कथयन्तिस्म ब्राह्मणा जम्भसाधकाः॥२१॥</div></div></div>
