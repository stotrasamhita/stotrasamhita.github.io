<div class="verse-block-wrapper" id="05-udyogaparva-b6426"><div class="verse-block verse-justify"><div class="line">दुःख द्वयोरिदं भद्रे कतरस्य चिकीर्षसि।</div><div class="line" style="position:relative;">प्रतिकर्तव्यमबले तत्त्वं वत्से वदस्व मे॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6427"><div class="verse-block verse-justify"><div class="line">यदि सौभपतिर्भद्रे नियोक्तव्यो मतस्तव।</div><div class="line" style="position:relative;">नियोक्ष्यति महात्मा स रामस्त्वद्धितकाम्यया॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6428"><div class="verse-block verse-justify"><div class="line">अथापगेयं भीष्मं त्वं रामेणेच्छसि धीमता।</div><div class="line" style="position:relative;">रणे विनिर्जितं द्रष्टुं कुर्यात्तदपि भार्गवः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6429"><div class="verse-block verse-justify"><div class="line">सृञ्जयस्य वचः श्रुत्वा तव चैव शुचिस्मिते।</div><div class="line">यदत्र ते भृशं कार्यं तदद्यैव विचिन्त्यताम् ॥अम्बोवाच।</div><div class="line" style="position:relative;">॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6430"><div class="verse-block verse-justify"><div class="line">अपनीतास्मि भीष्मेण भगवन्नविजानता।</div><div class="line" style="position:relative;">नाभिजानाति मे भीष्मो ब्रह्मन्साल्वगतं मनः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6431"><div class="verse-block verse-justify"><div class="line">एतद्विचार्य मनसा भवानेतद्विनिश्चयम्।</div><div class="line" style="position:relative;">विचिनोतु यथान्यायं विधानं क्रियतां तथा॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6432"><div class="verse-block verse-justify"><div class="line">भीष्मे वा कुरुशार्दूले शाल्वराजेऽथवा पुनः।</div><div class="line" style="position:relative;">उभयोरेव वा ब्रह्मन्युक्तं यत्तत्समाचर॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6433"><div class="verse-block verse-justify"><div class="line">निवेदितं मया ह्येतद्दुःखमूलं यथातथम्।</div><div class="line">विधानं तत्र भगवन्कर्तुमर्हसि युक्तितः ॥अकृतव्रण उवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6434"><div class="verse-block verse-justify"><div class="line">उपपदमिदं भद्रे यदेवं वरवर्णिनि।</div><div class="line" style="position:relative;">धर्मं प्रतिवचो ब्रूयाः श्रृणु चदं वचो मम॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6435"><div class="verse-block verse-justify"><div class="line">यदि त्वामापगेयो वै न नयेद्गजसाह्वयम्।</div><div class="line" style="position:relative;">साल्वस्त्वा शिरसा भीरु गृह्णीयाद्रामचोदितः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6436"><div class="verse-block verse-justify"><div class="line">तेन त्वं निर्जिता भद्रे यस्मान्नीतासि भामिनि।</div><div class="line" style="position:relative;">संशयः साल्वराजस्य तेन त्वयि सुमध्यमे॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6437"><div class="verse-block verse-justify"><div class="line">भीष्मः पुरुषमानी च जितकाशी तथैव च।</div><div class="line">तस्मात्प्रतिक्रिया युक्ता भीष्मे कारयितुं तव ॥अम्बोवाच।</div><div class="line" style="position:relative;">॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6438"><div class="verse-block verse-justify"><div class="line">ममाप्येष सदा ब्रह्मन्हृदि कामोऽभिवर्तते।</div><div class="line" style="position:relative;">घातयेयं यदि रणे भीष्ममित्येव नित्यदा॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6439"><div class="verse-block verse-justify"><div class="line">भीष्मं वा साल्वराजं वा यं वा दोषेण गच्छसि।</div><div class="line">प्रशाधि तं महाबाहोयत्कृतेऽहं सुदुःखिता ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6440"><div class="verse-block verse-justify"><div class="line">एवं कथयतामेव तेषां स दिवसो गतः।</div><div class="line" style="position:relative;">रात्रिश्च भरतश्रेष्ठ सुखशीतोष्णमारुता॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6441"><div class="verse-block verse-justify"><div class="line">ततो रामः प्रादुरासीत्प्रज्वलन्निव तेजसा।</div><div class="line" style="position:relative;">शिष्यैः परिवृतो राजञ्जटाचीरधरो मुनिः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6442"><div class="verse-block verse-justify"><div class="line">धनुष्पाणिरदीनात्मा खङ्गं बिभ्रत्परश्वधी।</div><div class="line" style="position:relative;">विरजा राजशार्दूल सृञ्जयं सोऽभ्ययान्नृपम्॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6443"><div class="verse-block verse-justify"><div class="line">ततस्तं तापसा दृष्ट्वा स च राजा महातपाः।</div><div class="line" style="position:relative;">तस्थुः प्राञ्जलयो राजन्सा च कन्या तपस्विनी॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6444"><div class="verse-block verse-justify"><div class="line">पूजयामासुरव्यग्रा मधुपर्केण भार्गवम्।</div><div class="line" style="position:relative;">अर्चितश्च यथान्यायं निषसाद सहैव तैः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6445"><div class="verse-block verse-justify"><div class="line">ततः पूर्वव्यतीतानि कथयन्तौ स्म तावुभौ।</div><div class="line" style="position:relative;">आसातां जामदग्न्यश्च सृञ्जयश्चैव भारत॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6446"><div class="verse-block verse-justify"><div class="line">तथा कथान्ते राजर्षिर्भृगुश्रेष्ठं महाबलम्।</div><div class="line" style="position:relative;">उवाच मधुरं काले रामं वचनमर्थवत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6447"><div class="verse-block verse-justify"><div class="line">रामेयं मम दौहित्री काशिराजसुता प्रभो।</div><div class="line" style="position:relative;">अस्याः श्रृणु यथातत्त्वं कार्यं कार्यविशारद॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6448"><div class="verse-block verse-justify"><div class="line">परमं कथ्यतां चेति तां रामः प्रत्यभाषत।</div><div class="line" style="position:relative;">ततः साभ्यवदद्रामं ज्वलन्तमिव पावकम्॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6449"><div class="verse-block verse-justify"><div class="line">ततोऽभिवाद्य चरणौ रामस्य शिरसा शुभौ।</div><div class="line" style="position:relative;">स्पृष्ट्वा पद्मदलाभाभ्यां पाणिभ्यामग्रतः स्थिता॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6450"><div class="verse-block verse-justify"><div class="line">रुरोद सा शोकवती बाष्पव्याकुललोचना।</div><div class="line">प्रपेदे शरणं चैव शरण्यं भृगुनन्दनम् ॥राम उवाच।</div><div class="line" style="position:relative;">॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6451"><div class="verse-block verse-justify"><div class="line">यथा त्वं सृंजयस्यास्य तथा मे त्वं नृपात्मजे।</div><div class="line">ब्रूहि यत्ते मनोदुःखं करिष्ये वचनं तव ॥अम्बोवाच।</div><div class="line" style="position:relative;">॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6452"><div class="verse-block verse-justify"><div class="line">भगवञ्शरणं त्वाद्य प्रपन्नाऽस्मि महाव्रतम्।</div><div class="line">शोकपङ्कार्णवान्मग्नं घोरादुद्धर मां विभो ॥भीष्म उवाच।</div><div class="line" style="position:relative;">॥२८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6453"><div class="verse-block verse-justify"><div class="line">तस्याश्च दृष्ट्वा रूपं च वपुश्चाभिनवं पुनः।</div><div class="line" style="position:relative;">सौकुमार्यं परं चैव रामश्चिन्तापरोऽभवत्॥२९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6454"><div class="verse-block verse-justify"><div class="line">किमियं वक्ष्यतीत्येवं विममर्श भृगूद्वहः।</div><div class="line" style="position:relative;">इति दध्यौ चिरं रामः कृपयाभिपरिप्लुतः॥३०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6455"><div class="verse-block verse-justify"><div class="line">कथ्यतामिति सा भूयो रामेणोक्ता शुचिस्मिता।</div><div class="line" style="position:relative;">सर्वमेव यथातत्त्वं कथयामास भार्गवे॥३१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6456"><div class="verse-block verse-justify"><div class="line">तच्छुत्वा जामदग्र्यस्तु राजपुत्र्या वचस्तदा।</div><div class="line">उवाच तां वरारोहां निश्चित्यार्थविनिश्चयम् ॥राम उवाच।</div><div class="line" style="position:relative;">॥३२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6457"><div class="verse-block verse-justify"><div class="line">प्रेषयिष्यामि भीष्माय कुरुश्रेष्ठाय भामिनि।</div><div class="line" style="position:relative;">करिष्यति वचो मह्यं श्रुत्वा च स नराधिपः॥३३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6458"><div class="verse-block verse-justify"><div class="line">न चेत्करिष्यति वचो मयोक्तं जाह्नवीसुतः।</div><div class="line" style="position:relative;">धक्ष्याम्यहं रणे भद्रे सामात्यं शस्त्रतेजसा॥३४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6459"><div class="verse-block verse-justify"><div class="line">अथवा ते मतिस्तत्र राजपुत्रि न वर्तते।</div><div class="line">यावत्साल्वपतिं वीरं योजयाम्यत्र कर्मणि ॥अम्बोवाच।</div><div class="line" style="position:relative;">॥३५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6460"><div class="verse-block verse-justify"><div class="line">विसर्जिताऽहं भीष्मेण श्रुत्वैव भृगुनन्दन।</div><div class="line" style="position:relative;">साल्वराजगतं भावं मम पूर्वं मनीषितम्॥३६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6461"><div class="verse-block verse-justify"><div class="line">सौभराजमुपेत्याहमवोचं दुर्वचं वचः।</div><div class="line" style="position:relative;">न च मां प्रत्यगृह्णात्स चारित्र्यपरिशङ्कितः॥३७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6462"><div class="verse-block verse-justify"><div class="line">एतत्सर्वं विनिश्चित्य स्वबुद्ध्या भृगुनन्दन।</div><div class="line" style="position:relative;">यदत्रौपयिकं कार्यं तच्चिन्तयितुमर्हसि॥३८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6463"><div class="verse-block verse-justify"><div class="line">मम तु व्यसनस्यास्य भीष्मो मूलं महाव्रतः।</div><div class="line" style="position:relative;">येनाहं वशमानीता समत्क्षिप्य बलात्तदा॥३९॥</div></div></div>
