<div class="verse-block-wrapper" id="05-udyogaparva-b5635"><div class="verse-block verse-justify"><div class="line">बृहस्पतिसं बुद्ध्या क्षमया पृथिवीसमम्।</div><div class="line" style="position:relative;">समुद्रमिव गाम्भीर्ये हिमवन्तमिव स्थिरम्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b5637"><div class="verse-block verse-justify"><div class="line">रणयज्ञे प्रवितते सुभीमे लोमहर्षणे।</div><div class="line" style="position:relative;">दीक्षितं चिररात्राय श्रुत्वा तत्र युधिष्ठिरः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5638"><div class="verse-block verse-justify"><div class="line">किमब्रवीन्महाबाहुः सर्वशस्त्रभृतां वरः।</div><div class="line">भीमसेनार्जुनौ वापि कृष्णो वा प्रत्यभाषत ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5639"><div class="verse-block verse-justify"><div class="line">आपद्धर्मार्थकुशलो महाबुद्धिर्युधिष्ठिरः।</div><div class="line" style="position:relative;">सर्वान्भ्रातॄन्समानीय वासुदेवं च शाश्वतम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5640"><div class="verse-block verse-justify"><div class="line">उवाच वदतां श्रेष्ठः सान्त्वपूर्वमिदं वचः।</div><div class="line" style="position:relative;">पर्याक्रामत सैन्यानि यत्तास्तिष्ठत दंशिताः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5641"><div class="verse-block verse-justify"><div class="line">पितामहेन वो युद्धं पूर्वमेव भविष्यति।</div><div class="line">तस्मात्सप्तसु सेनासु प्रणेतॄन्मम पश्यत ॥कृष्ण उवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5642"><div class="verse-block verse-justify"><div class="line">यथार्हति भवान्वक्तुमस्मिन्काले ह्युपस्थिते।</div><div class="line" style="position:relative;">तथेदमर्थवद्वाक्यमुक्तं ते भरतर्षभ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5643"><div class="verse-block verse-justify"><div class="line">रोचते मे महाबाहो क्रियतां यदनन्तरम्।</div><div class="line">नायकास्तव सेनायां क्रियन्तामिह सप्त वै ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5644"><div class="verse-block verse-justify"><div class="line">ततो द्रुपदमानाय्य विराटं शिनिपुङ्गवम्।</div><div class="line">धृष्टद्युम्नं च पाञ्चाल्यं धृष्टकेतुं च पार्थिव।</div><div class="line" style="position:relative;">शिखण्डिनं च पाञ्चाल्यं सहदेवं च मागधम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5645"><div class="verse-block verse-justify"><div class="line">एतान्सप्त महाभागान्वीरान्युद्धाभिकाङ्क्षिणः।</div><div class="line" style="position:relative;">सेनाप्रणेतॄन्विधिवदभ्यषिञ्चद्युधिष्ठिरः॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5646"><div class="verse-block verse-justify"><div class="line">सर्वसेनापतिं चात्र धृष्टद्युम्नं चकार ह।</div><div class="line" style="position:relative;">द्रोणान्तहेतोरुत्पन्नो य इद्धाञ्जातवेदसः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5647"><div class="verse-block verse-justify"><div class="line">सर्वेषामेव तेषां तु समस्तानां महात्मनाम्।</div><div class="line" style="position:relative;">सेनापतिपतिं चक्रे गडाकेशं धनञ्जयम्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5648"><div class="verse-block verse-justify"><div class="line">अर्जुनस्यापि नेता च संयन्ता चैव वाजिनाम्।</div><div class="line" style="position:relative;">सङ्कर्णणानुजः श्रीमान्महाबुद्धिर्जनार्दनः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5649"><div class="verse-block verse-justify"><div class="line">तद्दृष्ट्वोपस्थितं युद्धं समासन्नं महात्ययम्।</div><div class="line" style="position:relative;">प्राविशद्भवनं राज्ञः पाण्डवानां हलायुधः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5650"><div class="verse-block verse-justify"><div class="line">सहाक्रूरप्रभृतिभिर्ददसाम्बोद्धवादिभिः।</div><div class="line" style="position:relative;">रौक्मिणेयाहुकसुतैश्चारुदेष्णपुरोगमैः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5651"><div class="verse-block verse-justify"><div class="line">वृष्णिमुख्यैरधिगतैर्व्याघ्रैरिव बलोत्कटैः।</div><div class="line" style="position:relative;">अभिगुप्तो महाबाहुर्मरुद्भिरिव वासवः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5652"><div class="verse-block verse-justify"><div class="line">नीलकौशेयवसनः कैलासशिखरोपमः।</div><div class="line" style="position:relative;">सिंहखेलगतिः श्रीमान्मदरक्तान्तलोचनः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5653"><div class="verse-block verse-justify"><div class="line">तं दृष्ट्वा धर्मराजश्च केशवश्च महाद्युतिः।</div><div class="line" style="position:relative;">उदतिष्ठत्ततः पार्थो भीमकर्मा वृकोदरः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b5654"><div class="verse-block verse-justify"><div class="line">गाण्डीवधन्वा ये चान्ये राजानस्तत्र केचन।</div><div class="line" style="position:relative;">पूजयाञ्चक्रिरे ते वै समायान्तं हलायुधम्॥२२॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b5663"><div class="verse-block verse-justify"><div class="line">तच्च मे नाकरोद्वाक्यं त्वदर्थे मधुसूदनः।</div><div class="line" style="position:relative;">निर्विष्टः सर्वभावेन धनञ्जयमवेक्ष्य ह॥३१॥</div></div></div>
