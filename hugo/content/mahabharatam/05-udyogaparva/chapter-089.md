<div class="verse-block-wrapper" id="05-udyogaparva-b3271"><div class="verse-block verse-justify"><div class="line">प्रातरुत्थाय कृष्णस्तु कृतवान्सर्वमाह्निकम्।</div><div class="line" style="position:relative;">ब्राह्मणैरभ्यनुज्ञातः प्रययौ नगरं प्रति॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3272"><div class="verse-block verse-justify"><div class="line">तं प्रयान्तं महाबाहुमनुज्ञाप्य महाबलम्।</div><div class="line" style="position:relative;">पर्यवर्तन्त ते सर्वे वृकस्थलनिवासिनः॥३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3274"><div class="verse-block verse-justify"><div class="line">धार्तराष्ट्रास्तमायान्तं प्रत्युञ्जग्मुः स्वलङ्कृताः।</div><div class="line" style="position:relative;">दुर्योधनादृते सर्वे भीष्मद्रोणकृपादयः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3275"><div class="verse-block verse-justify"><div class="line">पौराश्च बहुला राजन्हृषीकेशं दिदृक्षवः।</div><div class="line" style="position:relative;">यानैर्बहुविधैरन्ये पद्भिरेव तथाऽपरे॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3276"><div class="verse-block verse-justify"><div class="line">स वै पथि समागम्य भीष्मेणाक्लिष्टकर्मणा।</div><div class="line" style="position:relative;">द्रोणेन धार्तराष्ट्रैश्च तैर्वृतो नगरं ययौ॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3277"><div class="verse-block verse-justify"><div class="line">कृष्णसंमाननार्थं च नगरं समलङ्कृतम्।</div><div class="line" style="position:relative;">बभूव राजमार्गश्च बहुरत्नसमाचितः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3278"><div class="verse-block verse-justify"><div class="line">न च कश्चिद्गृहे राजंस्तदाऽऽसीद्भरतर्षभ।</div><div class="line" style="position:relative;">न स्त्री न वृद्धो न शिशुर्वासुदेवदिदृक्षया॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3279"><div class="verse-block verse-justify"><div class="line">राजमार्गे नरास्तस्मिन्संस्तुवन्त्यवनिं गताः।</div><div class="line" style="position:relative;">तस्मिन्काले महाराज हृषीकेशप्रवेशने॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3280"><div class="verse-block verse-justify"><div class="line">आवृतानि वरस्त्रीभिर्गृहाणि सुमहान्त्यपि।</div><div class="line" style="position:relative;">प्रचलन्तीव भारेण दृश्यन्तेस्म महीतले॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3281"><div class="verse-block verse-justify"><div class="line">तथा च गतिमन्तस्ते वासुदेवस्य वाजिनः।</div><div class="line" style="position:relative;">प्रनष्टगतयोऽभूवन्राजमार्गे नरैर्वृते॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3282"><div class="verse-block verse-justify"><div class="line">स गृहं धृतराष्ट्रस्य प्राविशच्छत्रुकर्शनः।</div><div class="line" style="position:relative;">पाण्डुरैः पुण्डरीकाक्षः प्रासादैरुपशोभितम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3283"><div class="verse-block verse-justify"><div class="line">तिस्रः कक्ष्या व्यतिक्रम्य केशवो राजवेश्मनः।</div><div class="line" style="position:relative;">वैचित्रवीर्यं राजानमभ्यगच्छदरिन्दमः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3284"><div class="verse-block verse-justify"><div class="line">अभ्यागच्छति दाशार्हे प्रज्ञाचक्षुर्नराधिपः।</div><div class="line" style="position:relative;">तहैव द्रोणभीष्माभ्यामुदतिष्ठन्महायशाः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3285"><div class="verse-block verse-justify"><div class="line">कृपश्च सोमदत्तश्च महाराजश्च बाह्लिकः।</div><div class="line" style="position:relative;">आसनेभ्योऽचलन्सर्वे पूजयन्तो जनार्दनम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3286"><div class="verse-block verse-justify"><div class="line">ततो राजानमासाद्य धृतराष्ट्रं यशस्विनम्।</div><div class="line" style="position:relative;">सभीष्मं पूजयामास वार्ष्णेयो वाग्भिरञ्जसा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3287"><div class="verse-block verse-justify"><div class="line">तेषु धर्मानुपूर्वी तां प्रयुज्य मधुसूदनः।</div><div class="line" style="position:relative;">यथावयः समीयाय राजभिः सह माधवः॥१८॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b3290"><div class="verse-block verse-justify"><div class="line">अथ गां मधुपर्कं चाप्युदकं च जनार्दने।</div><div class="line" style="position:relative;">उपजह्नुर्यथान्यायं धृतराष्ट्रपुरोहिताः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3291"><div class="verse-block verse-justify"><div class="line">कृतातिथ्यस्तु गोविन्दः सर्वान्परिहसन्कुरून्।</div><div class="line" style="position:relative;">आस्ते सांबन्धिकं कुर्वन्कुरुभिः परिवारितः॥२२॥</div></div></div>
