<div class="verse-block-wrapper" id="05-udyogaparva-b2850"><div class="verse-block verse-justify"><div class="line">भीम उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2851"><div class="verse-block verse-justify"><div class="line">यथायथैव शान्तिः स्यात्कुरूणां मधुसूदन।</div><div class="line" style="position:relative;">तथातथैव भाषेथा मा स्म युद्धेन भीषयेः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2852"><div class="verse-block verse-justify"><div class="line">अमर्षी जातसंरम्भः श्रेयोद्वेषी महामनाः।</div><div class="line" style="position:relative;">नोग्रं दुर्योधनो वाच्यः साम्नैवेनं समाचरेः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2853"><div class="verse-block verse-justify"><div class="line">प्रकृत्या पापसत्वश्च तुल्यचेतास्तु दस्युभिः।</div><div class="line" style="position:relative;">ऐश्वर्यमदमत्तश्च कृतवैरश्च पाण्डवैः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2854"><div class="verse-block verse-justify"><div class="line">अदीर्घदर्शी निष्ठूरी क्षेप्ता क्रूरपराक्रमः।</div><div class="line" style="position:relative;">दीर्घमन्युरनेयश्च पापात्मा निकृतिप्रियः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2855"><div class="verse-block verse-justify"><div class="line">म्रियेतापि न भज्येत नैव जह्यात्स्वकं मतम्।</div><div class="line" style="position:relative;">तादृशेन शमः कृष्ण मन्ये परमदुष्करः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2856"><div class="verse-block verse-justify"><div class="line">सुहृदामप्यवाचीनस्त्यक्तधर्मा प्रियानृतः।</div><div class="line" style="position:relative;">प्रतिहन्त्ये सुहृदां वाचश्चैव मनांसि च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2857"><div class="verse-block verse-justify"><div class="line">स मन्युवशमापन्नः स्वभावं दुष्टमास्थितः।</div><div class="line" style="position:relative;">स्वभावात्पापमभ्येति तृणैश्छन्न इवोरगः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2858"><div class="verse-block verse-justify"><div class="line">दुर्योधनो हि यत्सेनः सर्वथा विदितस्तव।</div><div class="line" style="position:relative;">यच्छीलो यत्स्वभावश्च यद्बलो यत्पराक्रमः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2859"><div class="verse-block verse-justify"><div class="line">पुरा प्रसन्नाः कुरवः सहपुत्रास्तथा वयम्।</div><div class="line" style="position:relative;">इन्द्रज्येष्ठा इवाभूम मोदमानाः सबान्धवाः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2860"><div class="verse-block verse-justify"><div class="line">दुर्योधनस्य क्रोधेन भरता मधुसूदन।</div><div class="line" style="position:relative;">धक्ष्यन्ते शिशिरापाये वनानीव हुताशनैः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2861"><div class="verse-block verse-justify"><div class="line">अष्टादशेमे राजानः प्रख्यात मधुसूदन।</div><div class="line" style="position:relative;">ये समुच्चिच्छिदुर्ज्ञातीन्सुहृदश्च सबान्धवान्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2862"><div class="verse-block verse-justify"><div class="line">असुराणां समृद्धानां ज्वलतामिव तेजसा।</div><div class="line" style="position:relative;">पर्यायकाले धर्मस्य प्राप्ते कलिरजायत॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2863"><div class="verse-block verse-justify"><div class="line">हैहयानामुदावर्तो नीपानां जनमेजयः।</div><div class="line" style="position:relative;">बहुलस्तालजङ्घानां कृमीणामुद्धतो वसुः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2864"><div class="verse-block verse-justify"><div class="line">अजबिन्दुः सुवीराणां सुराष्ट्राणां रुषर्द्धिकः।</div><div class="line" style="position:relative;">अर्कजश्च बलीहानां चीनानां धौतमूलकः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2865"><div class="verse-block verse-justify"><div class="line">हयग्रीवो विदेहानां वरयुश्च महौजसाम्।</div><div class="line" style="position:relative;">बाहुः सुन्दरवंशानां दीप्ताक्षाणां पुरूरवाः॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2866"><div class="verse-block verse-justify"><div class="line">सहजश्चेदिमत्स्यानां प्रवीराणां वृषध्वजः।</div><div class="line" style="position:relative;">धारणश्चन्द्रवत्सानां मुकुटानां विगाहनः॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2867"><div class="verse-block verse-justify"><div class="line">शमश्च नन्दिवेगानामित्येते कुलपांसनाः।</div><div class="line" style="position:relative;">युगान्ते कृष्ण संभूताः कुलेषु पुरुषाधमाः॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2868"><div class="verse-block verse-justify"><div class="line">अप्ययं नः कुरूणां स्याद्युगान्ते कालसंभृतः।</div><div class="line" style="position:relative;">दुर्योधनः कुलाङ्गारो जघन्यः पापपूरुषः॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2869"><div class="verse-block verse-justify"><div class="line">तस्मान्मृदु शनैर्ब्रूया धर्मार्थसहितं हितम्।</div><div class="line" style="position:relative;">कामानुबद्धं बहुलं नोग्रमुग्रपराक्रमम्॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2870"><div class="verse-block verse-justify"><div class="line">अपि दुर्योधनं कृष्ण सर्वे वयमधश्चराः।</div><div class="line" style="position:relative;">नीचैर्भूत्वाऽनुयास्यामो मा स्म नो भरता नशन्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2871"><div class="verse-block verse-justify"><div class="line">अप्युदासीनवृत्तिः स्याद्यथा नः कुरुभिः सह।</div><div class="line" style="position:relative;">वासुदेव तथा कार्यं न कुरूननयः स्पृशेत्॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2872"><div class="verse-block verse-justify"><div class="line">वाच्यः पितामहो वृद्धो ये च कृष्ण समासदः।</div><div class="line" style="position:relative;">भ्रातॄणामस्तु सौभ्रात्रं धार्तराष्ट्रः प्रशाम्यताम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b2873"><div class="verse-block verse-justify"><div class="line">अहमेतद्ब्रवीम्येवं राजा चैव प्रशंसति।</div><div class="line" style="position:relative;">अर्जुनो नैव युद्धार्थी भूयसी हि दयाऽर्जुने॥२४॥</div></div></div></div>
