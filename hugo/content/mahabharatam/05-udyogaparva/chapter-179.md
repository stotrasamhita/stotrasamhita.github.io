<div class="verse-block-wrapper" id="05-udyogaparva-b6565"><div class="verse-block verse-justify"><div class="line">भीष्म उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6566"><div class="verse-block verse-justify"><div class="line">तमहं स्मयन्निव रणे प्रत्यभाषं व्यवस्थितम्।</div><div class="line" style="position:relative;">भूमिष्ठं नोत्सहे योद्धुं भवन्तं रथमास्थितः॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6567"><div class="verse-block verse-justify"><div class="line">आरोह स्यन्दनं वीर कवचं च महाभुज।</div><div class="line" style="position:relative;">बधान समरे राम यदि योद्धुं मयेच्छसि॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6568"><div class="verse-block verse-justify"><div class="line">ततो मामब्रवीद्रामः स्मयमानो रणाजिरे।</div><div class="line" style="position:relative;">रथो मे मेदिनी भीष्म वाहा वेदाः सदश्ववत्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6569"><div class="verse-block verse-justify"><div class="line">सूतश्च मातरिश्वा वै कवचं वेदमातरः।</div><div class="line" style="position:relative;">सुसंवीतो रणे ताभिर्योत्स्येऽहं कुरुनन्दन॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6570"><div class="verse-block verse-justify"><div class="line">एवं ब्रुवाणो गान्धारे रामो मां सत्यविक्रमः।</div><div class="line" style="position:relative;">शरव्रातेन महता सर्वतः प्रत्यवारयत्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6571"><div class="verse-block verse-justify"><div class="line">ततोऽपश्यं जामदग्न्यं रथमध्ये व्यवस्थितम्।</div><div class="line" style="position:relative;">सर्वायुधवरे श्रीमत्यद्भुतोपमदर्शने॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6572"><div class="verse-block verse-justify"><div class="line">मनसा विहिते पुण्ये विस्तीर्णे नगरोपभे।</div><div class="line" style="position:relative;">दिव्याश्वयुजि सन्नद्वे काञ्चनेन विभूषिते॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6573"><div class="verse-block verse-justify"><div class="line">कवचेन महाबाहो सोमार्ककृतलक्ष्मणा।</div><div class="line" style="position:relative;">धनुर्धरो बद्धतूणो बद्धगोधाङ्गुलित्रवान्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6574"><div class="verse-block verse-justify"><div class="line">सारथ्यं कृतवांस्तत्र युयुत्सोरकुतव्रणः।</div><div class="line" style="position:relative;">सखा वेदविदत्यन्तं दयितो भार्गवस्य ह॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6575"><div class="verse-block verse-justify"><div class="line">आह्वयानः स मां युद्धे मनो हर्षयतीव मे।</div><div class="line" style="position:relative;">पुनः पुनरभिक्रोशन्नभियाहीति भार्गवः॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6576"><div class="verse-block verse-justify"><div class="line">तमादित्यमिवोद्यन्तमनाधृष्यं महाबलम्।</div><div class="line" style="position:relative;">क्षत्रियान्तकरं राममेकमेकः समासदूम्॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6577"><div class="verse-block verse-justify"><div class="line">ततोऽहं बाणपातेषु त्रिषु वाहान्निगृह्य वै।</div><div class="line" style="position:relative;">अवतीर्य धनुर्न्यस्य पदातिर्ऋषिसत्तमम्॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6578"><div class="verse-block verse-justify"><div class="line">अभ्यागच्छं तदा राममर्चिष्यन्द्विजसत्तमम्।</div><div class="line" style="position:relative;">अभिवाद्य चैनं विधिवदब्रुवं वाक्यमुत्तमम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6579"><div class="verse-block verse-justify"><div class="line">योत्स्ये त्वया रणे राम सदृशेनाधिकेन वा।</div><div class="line">गुरुणा धर्मशीलेन जयमाशास्व मे विभो ॥राम उवाच।</div><div class="line" style="position:relative;">॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6580"><div class="verse-block verse-justify"><div class="line">एवमेतत्कुरुश्रेष्ठ कर्तव्यं भूतिमिच्छता।</div><div class="line" style="position:relative;">धर्मो ह्येष महाबाहो विशिष्टैः सह युध्यताम्॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6581"><div class="verse-block verse-justify"><div class="line">शपेयं त्वां नचेदेवमागच्छेथा विशांपते।</div><div class="line" style="position:relative;">युध्यस्व त्वं रणे यत्तो धैर्यमालम्ब्य कौरव॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6582"><div class="verse-block verse-justify"><div class="line">न तु ते जयमाशासे त्वां विजेतुमहं स्थितः।</div><div class="line" style="position:relative;">गच्छ युध्यस्व धर्मेण प्रीतोऽस्मि चरितेन ते॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6583"><div class="verse-block verse-justify"><div class="line">ततोऽहं तं नमस्कृत्य रथमारुह्य सत्वरः।</div><div class="line" style="position:relative;">प्राध्मापयं रणे शङ्खं पुनर्हेमपरिष्कृतम्॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6584"><div class="verse-block verse-justify"><div class="line">ततो युद्धं समभवन्मम तस्य च भारत।</div><div class="line" style="position:relative;">दिवसान्सुबहून्राजन्परस्परजिगीषया॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6585"><div class="verse-block verse-justify"><div class="line">स मे तस्मिन्रणे पूर्वं प्राहरत्कङ्खपत्रिभिः।</div><div class="line" style="position:relative;">वष्ट्या शतैश्च नवभिः शराणां नतषर्वणाम्॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6586"><div class="verse-block verse-justify"><div class="line">चत्वारस्तेन मे वाहाः सूतश्चैव विशांपते।</div><div class="line" style="position:relative;">प्रतिरुद्धास्तथैवाहं समरे दंशितः स्थितः॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6587"><div class="verse-block verse-justify"><div class="line">नमस्कृत्य च देवेभ्यो ब्राह्मणेभ्यो विशेषतः।</div><div class="line" style="position:relative;">तमहं स्मयन्निव रणे प्रत्यभाषं व्यवस्थितम्॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6588"><div class="verse-block verse-justify"><div class="line">आचार्यता मानिता मे निर्मर्यादे ह्यपि त्वयि।</div><div class="line" style="position:relative;">भूयश्च शृणु मे ब्रह्मन्संपदं धर्मसंग्रहे॥२४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6589"><div class="verse-block verse-justify"><div class="line">ये ते वेदाः शरीरस्था .... यच्च ते महत्।</div><div class="line" style="position:relative;">तपश्च ते महत्तप्तं न तेभ्यः प्रहराम्यहम्॥२५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6590"><div class="verse-block verse-justify"><div class="line">प्रहरे क्षत्रधर्मस्य यं त्वं राम समाश्रितः।</div><div class="line" style="position:relative;">ब्राह्मणः क्षत्रियत्वं हि याति शस्त्रसमुद्यमात्॥२६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6591"><div class="verse-block verse-justify"><div class="line">पश्य मे धनुषो वीर्यं पश्य बाह्वोर्बलं मम।</div><div class="line" style="position:relative;">एष ते कार्मुक वीर छिनद्मि निशितेषुणा॥२७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b6592"><div class="verse-block verse-justify"><div class="line">तस्याहं निशितं भल्लं चिक्षेप भरतर्षभ।</div><div class="line" style="position:relative;">तेनास्य धनुषः कोटिं छित्वा भूमावपातयम्॥२८॥</div></div></div>
