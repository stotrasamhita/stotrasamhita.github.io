<div class="verse-block-wrapper" id="05-udyogaparva-b3007"><div class="verse-block verse-justify"><div class="line">सहदेव उवाच।</div><div class="line" style="position:relative;">॥१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3008"><div class="verse-block verse-justify"><div class="line">यदेतत्कथितं राज्ञा धर्म एष सनातनः।</div><div class="line" style="position:relative;">यथा च युद्धमेव स्यात्तथा कार्यमरिन्दम॥२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3009"><div class="verse-block verse-justify"><div class="line">यदि प्रसममिच्छेयुः कुरवः पाण्डवैः सह।</div><div class="line" style="position:relative;">तथापि युद्धं दाशार्ह योजयेथाः सहैव तैः॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3010"><div class="verse-block verse-justify"><div class="line">कथं नु दृष्ट्वा पाञ्चालीं तथा कृष्ण सभागताम्।</div><div class="line" style="position:relative;">अवधेन प्रशाम्येत मम मन्युः सुयोधने॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3011"><div class="verse-block verse-justify"><div class="line">यदि भीमार्जुनौ कृष्ण धर्मराजश्च धार्मिकः।</div><div class="line" style="position:relative;">धर्ममुत्सृज्य तेनाहं योद्धुमिच्छामि संयुगे॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3012"><div class="verse-block verse-justify"><div class="line">ब्रूहि मद्वचनं कृष्ण सुयोधनमपण्डितम्।</div><div class="line">कृच्छ्रे वने वा वस्तव्यं पुरे वा नागसाह्वये ॥सात्यकिरुवाच।</div><div class="line" style="position:relative;">॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3013"><div class="verse-block verse-justify"><div class="line">सत्यमाह महाबाहो सहदेवो महामतिः।</div><div class="line" style="position:relative;">दुर्योधनवधे शान्तिस्तस्य कोपस्य मे भवेत्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3014"><div class="verse-block verse-justify"><div class="line">न जानासि यथा दृष्ट्वा चीराजिनधरान्वने।</div><div class="line" style="position:relative;">तवापि मन्युरुद्धूतो दुःखितान्प्रेक्ष्य पाण्डवान्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3015"><div class="verse-block verse-justify"><div class="line">तस्मान्माद्रीसुतः शूरो यदाह रणकर्कशः।</div><div class="line">वचनं सर्वयोधानां तन्मतं पुरुषोत्तम ॥वैशंपायन उवाच।</div><div class="line" style="position:relative;">॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3016"><div class="verse-block verse-justify"><div class="line">एवं वदि वाक्यं तु युयुधाने महामतौ।</div><div class="line" style="position:relative;">सुभीमः सिंहनादोऽभूद्योधानां तत्र सर्वशः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3017"><div class="verse-block verse-justify"><div class="line">सर्वे हि सर्वशो वीरास्तद्वचः प्रत्यपूजयन्।</div><div class="line" style="position:relative;">साधुसाध्विति शैनेयं हर्षयन्तो युयुत्सवः॥११॥</div></div></div></div>
