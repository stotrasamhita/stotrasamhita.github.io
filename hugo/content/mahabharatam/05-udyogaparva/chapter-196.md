<div class="verse-block-wrapper" id="05-udyogaparva-b7093"><div class="verse-block verse-justify"><div class="line">चेदिकाशिकरूशानां नेतारं दृढविक्रमम्।</div><div class="line" style="position:relative;">सेनापतिममित्रघ्नं धृष्टकेतुमथादिशत्॥३॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b7095"><div class="verse-block verse-justify"><div class="line">ते शूराश्चित्रवर्माणस्तप्तकुण्डलधारिणः।</div><div class="line" style="position:relative;">आज्यावसिक्ता ज्वलिता धिष्ण्येष्विव हुताशनाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7096"><div class="verse-block verse-justify"><div class="line">अशोभन्त महेष्वासा ग्रहाः प्रज्वलिता इव।</div><div class="line" style="position:relative;">अथ सैन्यं यथायोगं पूजयित्वा नरर्षभः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7097"><div class="verse-block verse-justify"><div class="line">दिदेश तान्यनीकानि प्रयाणाय महीपतिः।</div><div class="line" style="position:relative;">तेषां युधिष्ठिरो राजा ससैन्यानां महात्मनाम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7098"><div class="verse-block verse-justify"><div class="line">व्यादिदेश सबाह्यानां भक्ष्यभोज्यमनुत्तमम्।</div><div class="line" style="position:relative;">स गजाश्वमनुष्याणां ये च शिल्पोपजीविनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7099"><div class="verse-block verse-justify"><div class="line">अभिमन्युं बृहन्तं च द्रौपदेयांश्च सर्वशः।</div><div class="line" style="position:relative;">धृष्टद्युम्नमुखानेतान्प्राहिणोत्पाण्डुनन्दनः॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7100"><div class="verse-block verse-justify"><div class="line">भीमं च युयुधानं च पाण्डवं च धनंजयम्।</div><div class="line" style="position:relative;">द्वितीयं प्रेषयामास बलस्कन्धं युधिष्टिरः॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7101"><div class="verse-block verse-justify"><div class="line">भाण्डं समारोपयतां चरतां संप्रधावताम्।</div><div class="line" style="position:relative;">हृष्टानां तत्र योधानां शब्दो दिवमिवास्पशत्॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7102"><div class="verse-block verse-justify"><div class="line">स्वयमेव ततः पश्चाद्विराटद्रपदान्वितः।</div><div class="line" style="position:relative;">अथापरैर्महीपालैः सह प्रायान्महीपतिः॥१२॥</div></div></div>

<div class="verse-block-wrapper" id="05-udyogaparva-b7104"><div class="verse-block verse-justify"><div class="line">ततः पुनरनीकानि न्ययोजयत बुद्धिमान्।</div><div class="line" style="position:relative;">मोहयन्धृतराष्ट्रस्य पुत्राणां बुद्धिनिश्चयम्॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7105"><div class="verse-block verse-justify"><div class="line">द्रौपदेयान्महेष्वासानभिमन्युं च पाण्डवः।</div><div class="line" style="position:relative;">नकुलं सहदेवं च सर्वांश्चैव प्रभद्रकान्॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b7106"><div class="verse-block verse-justify"><div class="line">दश चाश्वसहस्राणि द्विसहस्त्राणि दन्तिनाम्।</div><div class="line" style="position:relative;">अयुतं च पदातीनां रथाः पञ्चशतं तथा॥१६॥</div></div></div>
