<div class="verse-block-wrapper" id="05-udyogaparva-b3829"><div class="verse-block verse-justify"><div class="line">एष शेषः स्थितो नागो येनेयं धार्यते सदा।</div><div class="line" style="position:relative;">तपसा लोकमुख्येन प्रभावसहिता मही॥३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3830"><div class="verse-block verse-justify"><div class="line">श्वेताचलनिभाकारो दिव्याभरणभूषितः।</div><div class="line" style="position:relative;">सहस्रं धारयन्मूर्ध्नां ज्वालाजिह्वो महाबलः॥४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3831"><div class="verse-block verse-justify"><div class="line">इह नानाविधाकारा नानाविधविभूषणाः।</div><div class="line" style="position:relative;">सुरसायाः सुता नागा निवसन्ति गतव्यथाः॥५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3832"><div class="verse-block verse-justify"><div class="line">मणिस्वस्तिकचक्राङ्काः कमण्डलुकलक्षणाः।</div><div class="line" style="position:relative;">सहस्रसंख्या बलिन सर्वे रौद्राः स्वभावतः॥६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3833"><div class="verse-block verse-justify"><div class="line">सहस्रशिरसः केचित्केचित्पञ्चशताननाः।</div><div class="line" style="position:relative;">शतशीर्षास्तथा केचित्केचित्रिशिरसोऽपि च॥७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3834"><div class="verse-block verse-justify"><div class="line">द्विपञ्चशिरसः केचित्केचित्सप्तसुखास्तथा।</div><div class="line" style="position:relative;">महाभोगा महाकायाः पर्वताभोगभोगिनः॥८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3835"><div class="verse-block verse-justify"><div class="line">बहूनीह सहस्राणि प्रयुतान्यर्बुदानि च।</div><div class="line" style="position:relative;">नागानामेकवंशानां यथाश्रेष्ठं तु मे श्रुणु॥९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3836"><div class="verse-block verse-justify"><div class="line">वासुकिस्तक्षकश्चैव कर्कोटकधनञ्जयौ।</div><div class="line" style="position:relative;">कालीयो नहुषश्चैव कम्बलाश्वतरावुभौ॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3837"><div class="verse-block verse-justify"><div class="line">बाह्यकुण्डो मणिर्नागस्तथैवापूरणः खगः।</div><div class="line" style="position:relative;">वामनश्चैलपत्रश्च कुकुरः कुकुणस्तथा॥११॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3838"><div class="verse-block verse-justify"><div class="line">आर्यको नन्दकश्चैव तथा कलशपोतकौ।</div><div class="line" style="position:relative;">कैलासकः पिञ्जरको नागश्चैरावतस्तथा॥१२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3839"><div class="verse-block verse-justify"><div class="line">सुमनोमुखो दधिमुखः शङ्खो नन्दोपनन्दकौ।</div><div class="line" style="position:relative;">आप्तः कोटरकश्चैव शिखी निष्ठूरिकस्तथा॥१३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3840"><div class="verse-block verse-justify"><div class="line">तित्तिरिर्हस्तिभद्रश्च कुमुदो माल्यपिण्डकः।</div><div class="line" style="position:relative;">द्वौ पद्मौ पुण्डरीकश्च पुष्पो मुद्गरपर्णकः॥१४॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3841"><div class="verse-block verse-justify"><div class="line">करवीरः पीठरकः संवृत्तो वृत्त एव च।</div><div class="line" style="position:relative;">पिण्डारो बिल्वपत्रश्च मूषिकादः शिरीषकः॥१५॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3842"><div class="verse-block verse-justify"><div class="line">दिलीपः शङ्खशीर्षश्च ज्योतिष्कोऽथापराजितः।</div><div class="line" style="position:relative;">कौरव्यो धृतराष्ट्रश्च कुहुरः कृशकस्तथा॥१६॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3843"><div class="verse-block verse-justify"><div class="line">विरजा धारणश्चैव सुबाहुर्मुखरो जयः।</div><div class="line" style="position:relative;">बधिरान्धौ विशुण्डिश्च विरसः सुरसस्तथा॥१७॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3844"><div class="verse-block verse-justify"><div class="line">एते चान्ये च बहवः कश्यपस्यात्मजाः स्मृताः।</div><div class="line">मातले पश्य यद्यत्र कश्चित्ते रोचते वरः ॥कण्व उवाच।</div><div class="line" style="position:relative;">॥१८॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3845"><div class="verse-block verse-justify"><div class="line">मातलिस्त्वेकमव्यग्रः सततं संनिरीक्ष्य वै।</div><div class="line">पप्रच्छ नारदं तत्र प्रीतिमानिव चाभवत् ॥मातलिरुवाच।</div><div class="line" style="position:relative;">॥१९॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3846"><div class="verse-block verse-justify"><div class="line">स्थितो य एष पुरतः कौरव्यस्यार्यकस्य तु।</div><div class="line" style="position:relative;">द्युतिमान्दर्शनीयश्च कस्यैष कुलनन्दनः॥२०॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3847"><div class="verse-block verse-justify"><div class="line">कः पिता जननी चास्य कतमस्यैष भोगिनः।</div><div class="line" style="position:relative;">वंशस्य कस्यैष महान्केतुभूत इव स्थितः॥२१॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3848"><div class="verse-block verse-justify"><div class="line">प्रणिधानेन धैर्येण रूपेण वयसा च मे।</div><div class="line">मनः प्रविष्टो देवर्षे गुणकेश्याः पतिर्वरः ॥कण्व उवाच।</div><div class="line" style="position:relative;">॥२२॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3849"><div class="verse-block verse-justify"><div class="line">मातलिं प्रीतमनसं दृष्ट्वा सुमुखदर्शनात्।</div><div class="line">निवेदयामास तदा माहात्म्यं जन्म कर्म च ॥नारद उवाच।</div><div class="line" style="position:relative;">॥२३॥</div></div></div>
<div class="verse-block-wrapper" id="05-udyogaparva-b3850"><div class="verse-block verse-justify"><div class="line">ऐरावतकुले जातः सुमुखो नाम नागराट्।</div><div class="line" style="position:relative;">आर्यकस्य मतः पौत्रो दौहित्रो वामनस्य च॥२४॥</div></div></div>
