<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0001"><div class="verse-block"><div class="line">जगदवन विधौ त्वं जागरूका भवानि</div><div class="line pada-even">तव तु जननि निद्रामात्मवत्कल्पयित्वा।</div><div class="line">प्रतिदिवसमहं त्वां बोधयामि प्रभाते</div><div class="line pada-even" style="position:relative;">त्वयि कृतमपराधं सर्वमेतं क्षमस्व॥</div></div></div>

<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0006"><div class="verse-block"><div class="line">शुभं दिशतु नो देवी कामाक्षी सर्वमङ्गला</div><div class="line pada-even">शुभं दिशतु नो देवी कामकोटी-मठेशः।</div><div class="line">शुभं दिशतु तच्छिष्य-सद्गुरुर्नो जयेन्द्रो</div><div class="line pada-even" style="position:relative;">सर्वं मङ्गलमेवास्तु मङ्गलानि भवन्तु नः॥</div></div></div>
<h3 class="subheading">सुप्रभातम्</h3>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0008"><div class="verse-block"><div class="line">कामाक्षि देव्यम्ब तवाऽऽर्द्रदृष्ट्या</div><div class="line pada-even">मूकः स्वयं मूककविर्यथाऽसीत्।</div><div class="line">तथा कुरु त्वं परमेशजाये</div><div class="line pada-even" style="position:relative;">त्वत्पादमूले प्रणतं दयार्द्रे॥१॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0009"><div class="verse-block verse-justify"><div class="line">उत्तिष्ठोत्तिष्ठ वरदे उत्तिष्ठ जगदीश्वरि।</div><div class="line" style="position:relative;">उत्तिष्ठ जगदाधारे त्रैलोक्यं मङ्गलं कुरु॥२॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0010"><div class="verse-block"><div class="line">शृणोषि कच्चिद् ध्वनिरुत्थितोऽयं</div><div class="line pada-even">मृदङ्गभेरीपटहानकानाम्।</div><div class="line">वेदध्वनिं शिक्षितभूसुराणां</div><div class="line pada-even" style="position:relative;">शृणोषि भद्रे कुरु सुप्रभातम्॥३॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0011"><div class="verse-block"><div class="line">शृणोषि भद्रे ननु शङ्खघोषं</div><div class="line pada-even">वैतालिकानां मधुरं च गानम्।</div><div class="line">शृणोषि मातः पिककुक्कुटानां</div><div class="line pada-even" style="position:relative;">ध्वनिं प्रभाते कुरु सुप्रभातम्॥४॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0012"><div class="verse-block"><div class="line">मातर्निरीक्ष्य वदनं भगवान् शशाङ्को</div><div class="line pada-even">लज्जान्वितः स्वयमहो निलयं प्रविष्टः।</div><div class="line">द्रष्टुं त्वदीय वदनं भगवान् दिनेशो</div><div class="line pada-even" style="position:relative;">ह्यायाति देवि सदनं कुरु सुप्रभातम्॥५॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0013"><div class="verse-block"><div class="line">पश्याम्ब केचिद् धृतपूर्णकुम्भाः</div><div class="line pada-even">केचिद् दयार्द्रे धृतपुष्पमालाः।</div><div class="line">काचित् शुभाङ्ग्यो ननुवाद्यहस्ताः</div><div class="line pada-even" style="position:relative;">तिष्ठन्ति तेषां कुरु सुप्रभातम्॥६॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0014"><div class="verse-block"><div class="line">भेरीमृदङ्गपणवानकवाद्यहस्ताः</div><div class="line pada-even">स्तोतुं महेशदयिते स्तुतिपाठकास्त्वाम्।</div><div class="line">तिष्ठन्ति देवि समयं तव काङ्क्षमाणाः</div><div class="line pada-even" style="position:relative;">ह्युत्तिष्ठ दिव्यशयनात् कुरु सुप्रभातम्॥७॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0015"><div class="verse-block"><div class="line">मातर्निरीक्ष्य वदनं भगवान् त्वदीयं</div><div class="line pada-even">नैवोत्थितः शशिधिया शयितस्तवाङ्के।</div><div class="line">सम्बोधयाऽऽशु गिरिजे विमलं प्रभातं</div><div class="line pada-even" style="position:relative;">जातं महेशदयिते कुरु सुप्रभातम्॥८॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0016"><div class="verse-block"><div class="line">अन्तश्चरन्त्यास्तव भूषणानां</div><div class="line pada-even">झल्झल्ध्वनिं नूपुरकङ्कणानाम्।</div><div class="line">श्रुत्वा प्रभाते तव दर्शनार्थी</div><div class="line pada-even" style="position:relative;">द्वारि स्थितोऽहं कुरु सुप्रभातम्॥९॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0017"><div class="verse-block"><div class="line">वाणी पुस्तकमम्बिके गिरिसुते पद्मानि पद्मासना</div><div class="line pada-even">रम्भा त्वम्बरडम्बरं गिरिसुता गङ्गा च गङ्गाजलम्।</div><div class="line">काली तालयुगं मृदङ्गयुगलं बृन्दा च नन्दा तथा</div><div class="line pada-even" style="position:relative;">नीला निर्मलदर्पण-धृतवती तासां प्रभातं शुभम्॥१०॥</div></div></div>
<div class="verse-block-wrapper" id="kamakshi-suprabhatam-b0018"><div class="verse-block"><div class="line">उत्थाय देवि शयनाद्भगवान् पुरारिः</div><div class="line pada-even">स्नातुं प्रयाति गिरिजे सुरलोकनद्याम्।</div><div class="line">नैको हि गन्तुमनघे रमते दयार्द्रे</div><div class="line pada-even" style="position:relative;">ह्युत्तिष्ठ देवि शयनात् कुरु सुप्रभातम्॥११॥</div></div></div>
